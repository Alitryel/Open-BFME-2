from search import *
import itertools,re

LOOPS=[
 'for(int i=0;i<8;++i){if(game->getConstSlot(i)&&game->getConstSlot(i)->isAI()){found=true;break;}}',
 'int i=0;do{if(game->getConstSlot(i)&&game->getConstSlot(i)->isAI()){found=true;break;}}while(++i<8);',
 'for(int i=0;i<8;++i){if(game->getConstSlot(i)&&game->getConstSlot(i)->isAI())return true;}',
 'int i=0;do{if(game->getConstSlot(i)&&game->getConstSlot(i)->isAI())return true;}while(++i<8);',
]

def source(flags):
 params,order,graph,output,scoped=flags
 retloop=bool(graph&2)
 loop=LOOPS[graph&3]
 if graph&4:
  if retloop:
   code='if(!enable)return false; if(!game->getConstSlot(game->v13())->m_hasMap)return false; if(!observer)return true; if(!game->v12())return false; '+loop+'return false;'
  else:
   code='bool found=false; if(!enable)return found; if(!game->getConstSlot(game->v13())->m_hasMap)return found; if(!observer)return true; if(!game->v12())return found; '+loop+'return found;'
 else:
  if retloop:
   code='if(enable&&game->getConstSlot(game->v13())->m_hasMap){if(!observer)return true;else if(game->v12()){'+loop+'}} return false;'
  else:
   code='bool found=false; if(enable&&game->getConstSlot(game->v13())->m_hasMap){if(!observer)found=true;else if(game->v12()){'+loop+'}} return found;'
 if scoped: code='{'+code+'}'
 declarations=[('GameInfo *const &'if params&1 else'GameInfo *')+'game',('const bool &'if params&2 else'bool')+' enable',('const bool &'if params&4 else'bool')+' observer']
 args=['game','enable','observer']
 if order:declarations.reverse();args.reverse()
 if output:
  code=re.sub(r'return ([^;]+);',r'{result=\1;return;}',code)
  declarations.append('bool &result');args.append('found')
  call='bool found; calcPreview('+','.join(args)+');'
  ret='void'
 else:
  call='bool found=calcPreview('+','.join(args)+');'
  ret='bool'
 definition='static __forceinline '+ret+' calcPreview('+','.join(declarations)+'){'+code+'}\n'
 begin=BODY.index('\n\t\tbool found = false;')
 end=BODY.index('\n\t\t((AptMapPreview *)&m_60)')
 body=BODY[:begin]+'\n\t\t'+call+BODY[end:]
 return HEAD+definition+MARK+body

if __name__=='__main__':
 run([('foundhelper-%03d'%n,source(f))for n,f in enumerate(itertools.product(range(8),range(2),range(8),range(2),range(2)))], 'found_helper')
