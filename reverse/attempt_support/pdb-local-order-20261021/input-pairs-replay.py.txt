from pathlib import Path

anchor='int _RTC_GetSrcLine(unsigned long address,char *source,int sourcelen,int *pline,char **moduleName) {'
pre,body=base.split(anchor)
variants=[]
# Full-body adapters change only the internal parameter order, retaining the canonical wrapper ABI.
args=[('unsigned long','address'),('char *','source'),('int','sourcelen'),('int *','pline'),('char **','moduleName')]
for k,order in enumerate([(4,0,1,2,3),(4,3,2,1,0),(3,4,0,1,2)]):
 sig=','.join(args[i][0]+' '+args[i][1] for i in order);call=','.join(args[i][1] for i in order)

# Both fields are genuinely consumed; all constructors/copies are visible, with no dummy storage or operations.
for rev,copy in itertools.product([False,True],[False,True]):
 fields='char **moduleName;unsigned long address;' if rev else 'unsigned long address;char **moduleName;'
 init='moduleName(n),address(a)' if rev else 'address(a),moduleName(n)'
 cp='InputPair(const InputPair& o):moduleName(o.moduleName),address(o.address){}' if copy else ''
 helper='struct InputPair{'+fields+'InputPair(unsigned long a,char **n):'+init+'{}'+cp+'};\n'

 changed=changed.replace(' char *errname=(char *)_alloca(1024);',' char *errname=(char *)_alloca(1024);\n InputPair inputs(address,moduleName);')
 variants.append((f'pair{int(rev)}{int(copy)}',pre+helper+anchor+changed))
