def text_for(shape,profile,opt,cpu,gl=False):
 text=re.sub(r'// cl:.*',f'// cl: {opt} /G{cpu} '+('/GL ' if gl else '')+'/arch:SSE /DNDEBUG /MD /EHsc',shapes[shape])
 return ('// compiler: '+profile+'\n' if profile!='rtm' else '')+text

def native():
 jobs=[]
 for shape,profile,opt,cpu in itertools.product(shapes,['rtm',*profiles],['/O1','/O2','/O1 /Og-'],[5,6,7]):
  jobs.append((f'n{len(jobs):03}',text_for(shape,profile,opt,cpu),dict(shape=shape,profile=profile,opt=opt,cpu=cpu)))
 return front.execute('native',jobs)

def intrinsics():
 # Recognition probes only. An unresolved intrinsic is a setup result, never a candidate.
 jobs=[]
 for profile,hint in itertools.product(['rtm',*profiles],['__nop','_nop']):
  text=f'// cl: /O1 /G6 /MD\nextern "C" void {hint}(void);\n#pragma intrinsic({hint})\nextern "C" void recognitionProbe(){{'+(hint+'();')*16+'}\n'
  if profile!='rtm':text='// compiler: '+profile+'\n'+text
  jobs.append((f'i{len(jobs):03}',text,dict(profile=profile,hint=hint,symbol='_recognitionProbe')))
 return front.execute('intrinsics',jobs)

def link_one(job):
 n,shape,profile,opt,cpu=job;name=f'l{n:03}';src=OUT/(name+'.cpp');obj=src.with_suffix('.obj')
 src.write_text(text_for(shape,profile,opt,cpu,True))
 cmd,env=build.compiler_command(src,obj);cmd.insert(-1,'-Fd'+str(obj.with_suffix('.pdb').relative_to(ROOT)))
 cp=subprocess.run(cmd,cwd=ROOT,env=env,capture_output=True,text=True)
 row=dict(name=name,shape=shape,profile=profile,opt=opt,cpu=cpu,compile_command=cmd,compile_returncode=cp.returncode,compile_log=cp.stdout+cp.stderr)
 if cp.returncode:return row
 rel=str(src.relative_to(ROOT).with_suffix(''));sdk=build.vc71_root()/'Vc7/PlatformSDK/Lib'
 cmd=['wine',str(ROOT/'build/trust237-round5'/('ltcg-'+profile)/'link.exe'),'/nologo','/DLL','/NOENTRY','/INCREMENTAL:NO','/LTCG','/NODEFAULTLIB','/EXPORT:'+NAME,'/OUT:'+rel+'.dll','/MAP:'+rel+'.map',str(obj.relative_to(ROOT)),'msvcrt.lib']
 cmd += [str((sdk/x).relative_to(ROOT)) for x in ['Kernel32.Lib','Shell32.Lib','WinTrust.Lib']]
 lp=subprocess.run(cmd,cwd=ROOT,env=env,capture_output=True,text=True)
 row.update(link_command=cmd,link_returncode=lp.returncode,link_log=lp.stdout+lp.stderr)
 if lp.returncode:return row
 pe=pefile.PE(str(src.with_suffix('.dll')));base=pe.OPTIONAL_HEADER.ImageBase
 start=next(e.address for e in pe.DIRECTORY_ENTRY_EXPORT.symbols if e.name.decode()==NAME)
 addresses=[int(m.group(1),16)-base for line in src.with_suffix('.map').read_text().splitlines() if (m:=re.match(r'\s+0001:[0-9a-fA-F]+\s+\S+\s+([0-9a-fA-F]{8})\s',line))]
 end=min(a for a in addresses if a>start)
 raw=pe.get_data(start,end-start).rstrip(b'\xcc')
 md=capstone.Cs(capstone.CS_ARCH_X86,capstone.CS_MODE_32);md.detail=True
 ins=list(md.disasm(raw,base+start));assert sum(i.size for i in ins)==len(raw) and ins[-1].mnemonic=='ret'
 imports={i.address:i.name.decode() for dll in pe.DIRECTORY_ENTRY_IMPORT for i in dll.imports if i.name}
 targetpe=pefile.PE(str(ROOT/'baselines/bfme2/workshop-vanilla-1.06/files/game.dat'))
 targetimports={i.name.decode():i.address for dll in targetpe.DIRECTORY_ENTRY_IMPORT for i in dll.imports if i.name}
 targets={'memset':0x6291ae,'WinVerifyTrust':0x655652}
 norm=bytearray(raw);calls=[]
 for i in ins:
  if i.mnemonic!='call':continue
  off=i.address-base-start
  if i.operands[0].type==capstone.x86.X86_OP_MEM:
   api=imports[i.operands[0].mem.disp];norm[off+2:off+6]=struct.pack('<I',targetimports[api])
  else:
   thunk=pe.get_data(i.operands[0].imm-base,6);assert thunk[:2]==b'\xff\x25'
   api=imports[struct.unpack('<I',thunk[2:])[0]]
   norm[off+1:off+5]=struct.pack('<i',targets[api]-RVA-off-5)
  calls.append((off,api))
 diffs=[i for i,(a,b) in enumerate(itertools.zip_longest(norm,TARGET)) if a!=b]
 row.update(size=len(raw),raw=raw.hex(),normalized=norm.hex(),normalized_sha256=hashlib.sha256(norm).hexdigest(),diff=len(diffs),prefix=next((i for i,(a,b) in enumerate(zip(norm,TARGET)) if a!=b),min(len(norm),len(TARGET))),nops=[i.address-base-start for i in ins if i.mnemonic=='nop'],nop16=b'\x90'*16 in raw,calls=calls)
 return row

def ltcg():
 jobs=[(n,*values) for n,values in enumerate(itertools.product(shapes,profiles,['/O1','/O2','/O1 /Og-'],[5,6,7]))]
 with concurrent.futures.ThreadPoolExecutor(max_workers=8) as pool:rows=list(pool.map(link_one,jobs))
 (OUT/'ltcg.json').write_text(json.dumps(dict(tool_inputs=inputs,results=rows),indent=2)+'\n')
 good=[r for r in rows if r.get('link_returncode')==0]
 print('ltcg',len(rows),'good',len(good),'distinct',len({r['normalized_sha256'] for r in good}),'NOP16',sum(r['nop16'] for r in good),'with_nops',sum(bool(r['nops']) for r in good),flush=True)
 for r in sorted({r['normalized_sha256']:r for r in good}.values(),key=lambda r:(r['diff'],-r['prefix']))[:5]:
  print({k:r[k] for k in ['name','shape','profile','opt','cpu','size','diff','prefix','nops']},flush=True)
 for r in rows:
  if r.get('link_returncode')!=0:print('ERROR',r,flush=True)
 return rows
