from pathlib import Path
import json
from wdbg import Dbg
W=Path(__file__).resolve().parent;d=Dbg(W/'base.cpp',W/'steps.obj',W/'steps.log');active=False;records=[]
try:
 d.prompt();d.prompt('set $CanDeferOnBPByAddr = 1');d.prompt('break *0x107821a7')
 for _ in range(150):
  out=d.prompt('cont',60)
  if 'terminated' in out:break
  e=d.reg('eip')
  if e==0x107821a7:
   f=d.words(d.reg('esp')+4,1)[0];fn=d.string(d.words(d.words(f,1)[0]+4,1)[0])
   if '?bfmeTwoCGD@' not in fn:continue
   active=True;d.prompt('break *0x10775acc');continue
  if not active:continue
  operand=d.words(d.reg('esp')+4,1)[0];w=d.words(operand,8)
  if d.words(w[6],8)[7]!=370:continue
  print('vptr entry',hex(e),flush=True)
  for i in range(600):
   regs={n:d.reg(n) for n in ['eip','eax','ecx','edx','ebx','esi','edi','ebp','esp']}
   rec=dict(regs=regs,insn=d.prompt('x/i $eip'),stack=d.words(regs['esp'],12))
   records.append(rec)
   if regs['eip']==0x10775a87:break
   d.prompt('nexti',60)
  break
finally:
 (W/'steps.json').write_text(json.dumps(records,indent=1));print('steps',len(records),flush=True)
 if d.p.poll() is None:
  try:d.prompt('quit')
  except Exception:d.p.kill()
 d.log.close()
