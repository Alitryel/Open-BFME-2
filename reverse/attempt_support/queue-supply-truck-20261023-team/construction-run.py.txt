from pathlib import Path
import sys,json
ROOT=Path('build/queueSupplyTruck-research')
OUT=ROOT/'construction'
sys.path.insert(0,str((ROOT/'experiments').resolve()))
from trial import run

def block_end(s,a):
    depth=1;i=s.index('{',a)+1
    while depth:
        if s[i]=='{':depth+=1
        if s[i]=='}':depth-=1
        i+=1
    return i

def transform(s,kind,force):
    fn=s.index('void AIPlayer::queueSupplyTruck( void )')
    a=s.index('\t\t\tconst ThingTemplate *tTemplate =',fn)
    loop=s.index('while (tTemplate)',a)
    b=block_end(s,loop)
    if kind=='scan':
        body=s[a:b].replace('break;','return;')
        decl='void buildSupplyTruck(BuildListInfo *info)'
        repl='\t\t\tbuildSupplyTruck(info);'
    else:
        start=s.index('if (factory) {',a)
        a=s.index('{',start)+1
        b=block_end(s,start)-1
        body=s[a:b].replace('break;','')
        decl='void enqueueSupplyTruck(BuildListInfo *info, const ThingTemplate *tTemplate, Object *factory)'
        repl='\n\t\t\t\t\t\tenqueueSupplyTruck(info,tTemplate,factory);\n\t\t\t\t\t\tbreak;\n\t\t\t\t\t'
    s=s[:a]+repl+s[b:]
    s=s.replace('\tTeamInQueue *m_teamBuildQueue;', '\t'+decl+';\n\tTeamInQueue *m_teamBuildQueue;',1)
    pos=s.index('void AIPlayer::queueSupplyTruck( void )')
    qualifier='__forceinline' if force else 'inline'
    impl=qualifier+' '+decl.replace('buildSupplyTruck','AIPlayer::buildSupplyTruck').replace('enqueueSupplyTruck','AIPlayer::enqueueSupplyTruck')+'\n{\n'+body+'\n}\n'
    return s[:pos]+impl+s[pos:]

results=[]
for context,path in [('base',ROOT/'extraction/base_control.cpp'),('first',ROOT/'extraction/base_first_ret_force.cpp')]:
    s=path.read_text()
    for kind in ['scan','enqueue']:
        for force in [False,True]:
            name=context+'_'+kind+('_force' if force else '')
            p=OUT/(name+'.cpp');p.write_text(transform(s,kind,force))
            r=run(p);results.append(r);print(json.dumps(r),flush=True)
(OUT/'results.json').write_text(json.dumps(results,indent=2))
