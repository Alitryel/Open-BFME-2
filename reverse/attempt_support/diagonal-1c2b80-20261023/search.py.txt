from driver import *

def gen():
 yield 'baseline',BASE
 # Widen real offset calculations and explicitly narrow to the 32-bit index.
 # All nine taps remain consumed; no extra load, test, barrier or restriction.
 for group,kind,mask in itertools.product(range(3),range(2),range(1,512)):
  b=BASE
  cast=['__int64','unsigned __int64'][kind]
  taps=['',' - pitch - 1',' + pitch - 1',' - pitch + 1',' + pitch + 1',' - 2 * pitch - 2',' + 2 * pitch + 2',' + 2 * pitch - 2',' - 2 * pitch + 2']
  for j in ([0,2] if group==0 else ([1,3] if group==1 else range(4))):
   for k,tail in enumerate(taps):
    if mask&(1<<k):
     old='src[(col+%d)%s]'%(j,tail)
     new='src[(int)((%s)(col+%d)%s)]'%(cast,j,tail)
     assert old in b,old
     b=b.replace(old,new)
  yield 'wide-%d-%d-%03d'%(group,kind,mask),b
if __name__=='__main__':run(list(gen()),'wide')
