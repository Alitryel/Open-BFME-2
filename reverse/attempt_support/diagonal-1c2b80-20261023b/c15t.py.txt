import sys
# c15 count at T = first kind-3 global refill after the first kind-15 refill
def c15t(tag):
    ids=[int(x) for x in open('tr/%s.cnt.ids'%tag).read().split()]
    gs=[int(x) for x in open('tr/%s.cnt.g'%tag).read().split()]
    n15=0; seen15=False
    for i in range(len(ids)-1):
        x=ids[i]
        if x>=0: n15+=1
        if gs[i+1]!=gs[i]:
            if x>=0: seen15=True
            elif seen15: return n15
    return -1
if __name__=='__main__':
    for t in sys.argv[1:]: print(t,c15t(t))
