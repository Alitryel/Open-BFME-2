from pathlib import Path
import subprocess,sys,re,json
base=Path('reverse/attempts/0x002ce30a.cpp').read_text()
out=Path('build/resume-eh-local')
sym='?privateFireWeapon@Weapon@@QAE_NPBVObject@@PBUCoord3D@@0H1_N2IPAH@Z'
def typed(s,fire=True,special=True,ptr=False):
    arg='const WeaponBonus*' if ptr else 'const WeaponBonus&'
    use='&bonus' if ptr else 'bonus'
    if fire:
        s=s.replace('const Coord3D*,int,bool,Weapon*,int);','const Coord3D*,'+arg+',bool,Weapon*,int);')
        s=s.replace(',&targetPos,(int)&bonus,',',&targetPos,'+use+',').replace(',victimPos,(int)&bonus,',',victimPos,'+use+',')
    if special:
        s=s.replace('void rva002CAA9D(int,const void*,const void*,int);','void rva002CAA9D(int,const void*,const void*,'+arg+');')
        s=s.replace('sourceObj,victimObj,(int)&bonus);','sourceObj,victimObj,'+use+');')
    return s
variants={'base':base,'fire-ref':typed(base,True,False),'both-ref':typed(base),'both-ptr':typed(base,ptr=True)}
results=[]
for name,s in variants.items():
    p=out/('turn51-weapon-'+name+'.cpp');p.write_text(s)
    log=out/('turn51-weapon-'+name+'.txt')
    proc=subprocess.run([sys.executable,'tools/align_diff.py',str(p),sym,'0x002ce30a','955'],stdout=subprocess.PIPE,stderr=subprocess.STDOUT,text=True)
    log.write_text(proc.stdout)
    lines=[x for x in proc.stdout.splitlines() if any(y in x.lower() for y in ('retail:', 'compiled:', 'structural','memory','register','size delta','error','bytes'))]
    print(name,proc.returncode,'\n'.join(lines[:18]),flush=True)
    results.append({'variant':name,'exit':proc.returncode,'summary':lines})
(out/'turn51-weapon-types-results.json').write_text(json.dumps(results,indent=2))
