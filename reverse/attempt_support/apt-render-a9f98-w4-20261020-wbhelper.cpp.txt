// and a TU whose render auto-inlines it emits that orphan copy without any caller.
static void __fastcall aptMatrixMul(V2 &out, const V2 &in, const AptMatrix &m)
{

	out.Y = in.X * m.b + in.Y * m.d + m.ty;
}
