import trials,itertools,concurrent.futures,json
specs=[]
decl='template <> _String_base<wchar_t, allocator<wchar_t> >::~_String_base();'
proxy=trials.proxy.replace('EXCEPTION','').replace('sizeof(T)==0','_DiagLeak').replace('namespace _STL {','namespace _STL { enum { _DiagLeak = 0 };',1)
for attribute,depth,position,lifetime in itertools.product(['','inline','__forceinline'],[0,1,255],['before','after'],[0,1]):
 label=f'base-a{attribute or "plain"}-d{depth}-{position}-l{lifetime}'
 definition=f'\nnamespace _STL {{\n#pragma inline_depth({depth})\ntemplate <> {attribute} _String_base<wchar_t, allocator<wchar_t> >::~_String_base() {{ _M_deallocate_block(); }}\n#pragma inline_depth(255)\n}}\n'
 src=trials.bank
 if lifetime:src=src.replace('#define private public',proxy+'\n#define private public',1)
 if position=='before':src=src.replace(decl,'}\n'+definition+'\nnamespace _STL {')
 else:src+=definition
 specs.append((label,src))
for attribute,depth,position,visible in itertools.product(['','inline','__forceinline'],[0,1,255],['before','after'],[0,1]):
 label=f'whole-a{attribute or "plain"}-d{depth}-{position}-v{visible}'
 definition=f'\nnamespace _STL {{\n#pragma inline_depth({depth})\ntemplate <> {attribute} basic_string<wchar_t>::~basic_string() {{ _Destroy(this->_M_start, this->_M_finish+1); }}\n#pragma inline_depth(255)\n}}\n'
 src=trials.bank
 if visible:src=src.replace(decl,'')
 if position=='before':src=src.replace('typedef ostreambuf_iterator<wchar_t', '}\n'+definition+'\nnamespace _STL {\ntypedef ostreambuf_iterator<wchar_t',1)
 else:
  src=src.replace('typedef ostreambuf_iterator<wchar_t', 'template <> basic_string<wchar_t>::~basic_string();\ntypedef ostreambuf_iterator<wchar_t',1)
  src+=definition
 specs.append((label,src))
with concurrent.futures.ThreadPoolExecutor(max_workers=8) as p:r=list(p.map(trials.run,specs))
(trials.scratch/'definitions.json').write_text(json.dumps(r,indent=2)+'\n')
