import sys,itertools,json,time,re
from pathlib import Path
from concurrent.futures import ThreadPoolExecutor
sys.path.insert(0,'build/fxinit-seat22');import search as h
h.HERE=Path('build/fxinit-split22').resolve()
NAMES=['Get_Tex_Offset','Get_Location_Offset','Get_Normal_Offset','Get_Diffuse_Offset','Get_Basis_Offset','Get_Basis_Second_Offset','Get_Basis_End','Get_Extension_Data_Offset','Get_FVF_Size']
def variants():
 for mask,form,fields in itertools.product(range(512),range(2),range(2)):
  src=h.BASE;a=src.index('struct BfmeFVFDescriptor {');b=src.index('extern void BFME_DX8_Thread_Lock',a);decl=src[a:b]
  if fields:
   # Preserve fvf/flags/count/format and constructor's ABI; vary only offsets and stride.
   decl=re.sub(r'^ unsigned (stride|locationOffset|normalOffset|blendOffset|textureOffsets|diffuseOffset|specularOffset|basisOffset|basisSecondOffset|basisEnd|extensionDataOffset)\b',r' int \1',decl,flags=re.M)
  for k,name in enumerate(NAMES):
   if mask>>k&1:
    ret='int ' if form==0 else ('const int &'if fields else'const unsigned &')
    decl=decl.replace('unsigned '+name+'(',ret+name+'(')
  src=src[:a]+decl+src[b:]
  yield f'desctypes_{mask:03}_{form}{fields}',src
if __name__=='__main__':
 t=time.time()
 with ThreadPoolExecutor(max_workers=8)as pool:r=list(pool.map(h.run_one,variants()))
 r.sort(key=lambda x:(x.get('nd',99999),x.get('cost',99999)))
 (h.HERE/'desctypes-results.json').write_text(json.dumps(r,indent=2))
 print(json.dumps({'count':len(r),'valid':sum('nd'in x and not x.get('unresolved')for x in r),'seconds':time.time()-t,'best':r[:8]},indent=2))
