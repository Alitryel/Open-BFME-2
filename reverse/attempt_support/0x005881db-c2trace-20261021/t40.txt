import sys,os
sys.path.insert(0,'/work/Open-BFME-2/build/c2tr')
from wdbg import Dbg
# t40.py SRC OBJ : trace 40941 for record named WN (env), extra bps XB=hex,hex printing regs
src,obj=sys.argv[1:3]; WN=os.environ.get('WN','node'); XB=[int(x,16) for x in os.environ.get('XB','').split(',') if x]
d=Dbg(src,obj,obj+'.log')
d.prompt();d.prompt('set $CanDeferOnBPByAddr = 1')
B=0x10700000
bps=[0x83acd,0x82978,0x412dd,0x412e2,0xa90e6]+XB; added=set()
for b in bps:
    o=d.prompt('break *0x%x'%(B+b))
    if 'Breakpoint' in o and 'Unable' not in o: added.add(b)
names={}
def nm(sym):
    if sym in names: return names[sym]
    sw=d.words(sym,1); pw=d.words(sw[0],2) if sw and sw[0]>0x100000 else []
    n=d.string(pw[1]).strip() if len(pw)>1 and pw[1]>0x100000 else '?'
    n=n.strip('"').lstrip('_'); names[sym]=n; return n
on=False;k=0;tr=False;n40=0
R=['eax','ecx','edx','ebx','esi','edi','ebp','esp']
while True:
    out=d.prompt('cont')
    if 'terminated' in out: break
    eip=d.reg('eip')-B
    for b in bps:
        if b not in added and b!=eip: d.prompt('break *0x%x'%(B+b)); added.add(b)
    added.add(eip)
    if eip==0x83acd:
        ebp=d.reg('ebp'); f=d.words(ebp,1)[0]; curfn=d.string(d.words(f+4,1)[0]); on='updateMeleeAttack' in curfn and d.reg('ebx')==0; k=0; continue
    if not on: continue
    if eip==0x82978: k+=1; continue
    if eip==0xa90e6:
        esi=d.reg('esi'); w=d.words(esi,2)
        if nm(w[0])==WN and os.environ.get('CLR'):
            d.prompt('set *(int*)0x%x = %d'%(esi+4,w[1]&~0x400)); print('R%d clr flag4 %x'%(k,w[1]),flush=True)
        continue
    if eip==0x412dd:
        esi=d.reg('esi'); w=d.words(esi,18)
        tr = nm(w[0])==WN
        if tr:
            n40+=1; t=w[15]; tw=d.words(t,16)
            print('R%d 40941 %s:%x w15=%x tuple=%s'%(k,nm(w[0]),w[7],t,' '.join('%x'%x for x in tw)),flush=True)
            op=tw[10]; i=0
            while op and i<8:
                ow=d.words(op,10); s=ow[7]; sw=d.words(s,8) if s>0x100000 else []
                print('   op %x: %s | sym %s'%(op,' '.join('%x'%x for x in ow),' '.join('%x'%x for x in sw)),flush=True)
                op=ow[0]; i+=1
        continue
    if eip==0x412e2:
        if tr: print('   ret %x'%d.reg('eax'),flush=True)
        tr=False; continue
    if tr:
        print('   bp %x %s'%(eip,' '.join('%s=%x'%(r,d.reg(r)) for r in R)),flush=True)
print('done',flush=True)
