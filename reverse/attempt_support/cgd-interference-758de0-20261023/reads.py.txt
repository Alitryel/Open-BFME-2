from pathlib import Path
import itertools,json
W=Path(__file__).resolve().parent
base=(W/'bank.cpp').read_text()
helper='''
__forceinline ContactData *readData(ContactData *const volatile &p) { return p; }
__forceinline ContactSource *readSource(ContactSource *const volatile &p) { return p; }
__forceinline void *bridgeReadSource(ContactData *s,const ContactVector &p,const ContactVector &n) { return readSource(s->source)->prepare(p,n); }
'''
head,body=base.split('void BfmeThingCGD::bfmeTwoCGD() {')
head+=helper+'\nvoid BfmeThingCGD::bfmeTwoCGD() {'
old1='node->state=node->first->source->test(node->second->source->prepare(node->point,node->normal))?1:2;'
old2='node->first->source->respond(bridge(node->second,node->point,node->normal));'
old3='node->second->source->respond(altPrepare(node->first->source,node->point,ContactVector(node->normal).negate()));'
def src(side,mode):
 p='node->'+side
 return p+'->source' if mode==0 else 'readData('+p+')->source' if mode==1 else 'readSource('+p+'->source)'
out=W/'reads';out.mkdir(exist_ok=True);meta={}
for control in range(2):
 for modes in itertools.product(range(3),repeat=6):
  n=len(meta);s=head+body
  s=s.replace(old1,'node->state='+src('first',modes[0])+'->test('+src('second',modes[1])+'->prepare(node->point,node->normal))?1:2;')
  inn='bridge('+('readData(node->second)' if modes[3]==1 else 'node->second')+',node->point,node->normal)' if modes[3]!=2 else 'bridgeReadSource(node->second,node->point,node->normal)'
  s=s.replace(old2,src('first',modes[2])+'->respond('+inn+');')
  inn='altPrepare('+src('first',modes[5])+',node->point,ContactVector(node->normal).negate())' if control==0 else src('first',modes[5])+'->prepare(node->point,ContactVector(node->normal).negate())'
  s=s.replace(old3,src('second',modes[4])+'->respond('+inn+');')
  name='v%04d'%n;(out/(name+'.cpp')).write_text(s);meta[name]=dict(control=control,modes=modes)
(W/'reads-meta.json').write_text(json.dumps(meta,indent=1));print('generated',len(meta))
