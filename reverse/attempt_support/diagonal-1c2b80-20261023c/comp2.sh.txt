#!/bin/sh
# comp2.sh src tag: count, then inject (1279-first15-B) kind-15 at AT 1 (or kind-3 chunks when negative), print state + nds
cd /work/Open-BFME-2/build/db
src=$1; t=$2
python3 count15.py "$src" "$t" >/dev/null 2>&1
[ -s tr/$t.cnt.ids ] || python3 count15.py "$src" "$t" >/dev/null 2>&1
set -- $(python3 st.py "$t" 2>/dev/null | tail -1 | python3 -c "import ast,sys;d=ast.literal_eval(sys.stdin.read());print(d['first15'],d['A'],d['B'],d['tail'])")
F=$1; A=$2; B=$3; tl=$4
nd=$(python3 dd.py "$src" | awk '{print $5}')
N=$((1279-F-B)); plan=""
if [ $N -gt 0 ]; then plan="15:1:$N"; elif [ $N -lt 0 ]; then M=$(( (31-N)/32 )); N2=$((N+32*M)); plan="3:1:$((32*M))"; [ $N2 -gt 0 ] && plan="$plan,15:1:$N2"; fi
nc=-
if [ -n "$plan" ]; then
  k=k_$t
  timeout 900 python3 inject3.py "$src" "$k" "$plan" > tr/$k.out 2>&1
  nc=$(python3 scoreobj.py tr/$k.obj 2>&1 | awk '{print $NF}')
  rm -f tr/$k.obj tr/$k.log tr/$k.out
fi
echo "$t f15=$F A=$A B=$B tail=$tl nd=$nd plan=$plan comp=$nc"
