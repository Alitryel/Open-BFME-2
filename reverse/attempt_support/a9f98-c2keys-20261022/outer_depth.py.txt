from search import *
BASE=(HERE/'owned-abi.cpp').read_text();HEAD,BODY=BASE.split('void AptShapeContainer::render()',1)
CALLS=list(re.finditer(r'(aptMatrixMul(?:UVFirst|NamedY|NamedProducts|Indexed|LineSecond|SolidThird)?)\((v->(?:pos|uv)|a|b), ([^,]+), ([^;]+)\);',BODY))
def make(depths):
    head=HEAD;body=BODY
    for n,site in reversed(list(enumerate([1,5]))):
        c=CALLS[site];old=c[1];new='aptOuterDepth'+str(site)
        helper=re.search(r'static __forceinline void '+old+r'\([^}]+\}',HEAD)[0].replace(old,new)
        for ai,axis in enumerate('XY'):
            dep=depths[n*2+ai]
            if dep:helper=re.sub(r'(out\.'+axis+r' = )([^;]+);',lambda m:m[1]+'('*dep+m[2]+')'*dep+';',helper)
        head+='\n'+helper+'\n';body=body[:c.start()]+c[0].replace(old,new)+body[c.end():]
    return head+'void AptShapeContainer::render()'+body
if __name__=='__main__':
    batch([('outerDepth_'+''.join(map(str,d)),make(d)) for d in itertools.product(range(6),repeat=4)])
