from pathlib import Path
import sys,itertools
ROOT=Path.cwd();OUT=ROOT/'build/trust237-round13'
sys.path.insert(0,str(ROOT/'build/trust237-round9'));import frontends as f
f.OUT=OUT
# New family: cleanup itself is duplicated across status-dependent edges.
# Earlier equal-result checks placed one cleanup before the status branch.
# Every arm calls LocalFree exactly once and returns one for every u32.
conditions=['u==0u','u==0x800b0100u','(long)u<0','u<=1u']
forms=[
 'if(P){C;return R;}C;return R;',
 'if(P){C;r=R;}else{C;r=R;}return r;',
 'if(P)goto left;goto right;left:C;goto done;right:C;goto done;done:return R;',
 'switch(u&3u){case 0:C;return R;case 1:C;return R;default:C;return R;}',
]
returns=['1','r+1','(int)((u&1u)<=1u)','(int)((u|1u)!=0u)']
jobs=[]
for cond,form,ret,helper,cpu,opt in itertools.product(conditions,range(4),returns,[False,True],[6,7],['O1','O2']):
    # In the assignment form r cannot occur on both sides of its own update;
    # use an independent, constant-zero return seed for that shape.
    expr=ret.replace('r+1','seed+1') if form==1 else ret
    cleanup='releaseArgs(argv)' if helper else 'LocalFree(argv)'
    tail=forms[form].replace('P',cond).replace('C',cleanup).replace('R',expr)
    src=f.base.replace(f.marker,'long status=WinVerifyTrust(0,&action,&data);unsigned long u=(unsigned long)status;int r=0;int seed=0;'+tail)
    if helper:
        src=src.replace('int Rva00237D3DCheck() {','__forceinline void releaseArgs(unsigned short **p) throw(){LocalFree(p);}\nint Rva00237D3DCheck() {')
    src=src.replace('/O1 /G7',f'/{opt} /G{cpu}')
    jobs.append((f'f{len(jobs):04}',src,dict(condition=cond,form=form,result=ret,helper=helper,cpu=cpu,opt=opt)))
if __name__=='__main__':f.execute('funnels',jobs)
