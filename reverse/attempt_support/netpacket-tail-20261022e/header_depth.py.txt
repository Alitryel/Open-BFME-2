import itertools
from search import *
from header_weight import source
items=[]
for mode,levels in itertools.product([1,2],itertools.product(range(3),repeat=6)):
 if max(levels)<2 or min(levels)==2:continue
 mask=sum((1<<i) for i,x in enumerate(levels) if x)
 items.append((f'hdepth_{mode}_'+''.join(map(str,levels)),source(mode,0,mask,levels)))
for mode,pos,other in itertools.product([1,2],range(6),itertools.product(range(2),repeat=5)):
 levels=list(other);levels.insert(pos,3);levels=tuple(levels)
 mask=sum((1<<i) for i,x in enumerate(levels) if x)
 items.append((f'hdepth_{mode}_'+''.join(map(str,levels)),source(mode,0,mask,levels)))
batch(items,'header-depth')
