import itertools
from pathlib import Path
b = Path('sw2.cpp').read_text(encoding='latin-1')
NAME = "Path *Pathfinder::internal_findHierarchicalPath("
A = ("     HierarchyPoint delta(way->location);\n     delta.x-=obj->position[0];delta.y-=obj->position[1];delta.z-=obj->position[2];\n"
     "     float dist=delta.x*delta.x+delta.y*delta.y;\n")
assert b.count(A) == 1
CT = {'n': None, 'y': "y=c.y;x=c.x;z=c.z;", 'z': "z=c.z;y=c.y;x=c.x;"}
SUBS = {'xyz': 'xyz', 'yxz': 'yxz', 'xy': 'xy', 'yx': 'yx', 'zxy': 'zxy'}
for ct, so, do, pp in itertools.product('nyz', SUBS, (0,1), (0,1)):
    u = b
    typ = 'HierarchyPoint'
    if CT[ct]:
        u = u.replace(NAME, f"struct HPD : public Coord3D {{ HPD(const Coord3D &c) {{ {CT[ct]} }} HPD &operator=(const Coord3D &c) {{ *(Coord3D *)this=c;return *this; }} }};\n" + NAME, 1)
        typ = 'HPD'
    P = "pos->" if pp else "obj->position"
    idx = {'x': 0, 'y': 1, 'z': 2}
    sub = ''.join(f"delta.{c}-={P}{c};" if pp else f"delta.{c}-=obj->position[{idx[c]}];" for c in so)
    dist = "delta.x*delta.x+delta.y*delta.y" if do == 0 else "delta.y*delta.y+delta.x*delta.x"
    r = (f"     const Coord3D *pos=(const Coord3D *)obj->position;\n" if pp else "") + f"     {typ} delta(way->location);\n     {sub}\n     float dist={dist};\n"
    u = u.replace(A, r)
    Path(f'v25/{ct}{so}{do}{pp}.cpp').write_text(u, encoding='latin-1', newline='')
