import sys,itertools,json,time
from pathlib import Path
from concurrent.futures import ThreadPoolExecutor,as_completed
sys.path.insert(0,'build/fxinit-seat22');import search as h
h.HERE=Path('build/fxinit-edges22').resolve()
PASS='*(unsigned *)(vertices+desc.Get_Basis_End())=pass;'
ONE='*(int *)(vertices+desc.Get_Basis_End())=remap[links[0][i]];'
def variants():
 for order,polarity,guard,loc,mask in itertools.product(itertools.permutations(range(4)),range(4),range(3),range(2),[2,8]):
  labels=['CopyBoth','StorePosition','CopyNormal','StoreNormal']
  p='positions[1]';n='normals[1]'
  if polarity&1:pt=f'if(!{p}) goto CopyNormal;goto StorePosition;'
  else:pt=f'if({p}) goto StorePosition;goto CopyNormal;'
  if polarity&2:nt=f'if(!{n}) goto FinishSkin;goto StoreNormal;'
  else:nt=f'if({n}) goto StoreNormal;goto FinishSkin;'
  bodies=[pt,h.P[len('if(positions[1]) '):]+'goto CopyNormal;',nt,h.N[len('if(normals[1]) '):]+'goto FinishSkin;']
  pn=''
  for k,j in enumerate(order):
   body=bodies[j];nextname=labels[order[k+1]]if k<3 else'FinishSkin'
   if body.endswith('goto '+nextname+';'):body=body[:-len('goto '+nextname+';')]
   pn+=labels[j]+':\n'+body+'\n'
  cond=[f'!{p}&&!{n}',f'!((unsigned){p}|(unsigned){n})',f'{p}=={n}&&!{p}'][guard]
  entry='if('+cond+') goto CopyBoth;goto FinishSkin;'
  arm='if(m_Passes>1) {'+PASS+'goto FinishSkin;}\nelse if(m_BonesPerVertex>0 && links[0]) {\nif(m_BonesPerVertex==1) {'+ONE+'goto FinishSkin;}\nelse if(m_BonesPerVertex==2) {\n'+h.R+'\n'+h.W+'\ngoto CopyBoth;\n'+(pn if not loc else'')+'}\nelse {'+(entry if mask&8 else'goto FinishSkin;')+'}\n}\nelse {'+(entry if mask&2 else'goto FinishSkin;')+'}\n'
  # Blocks emitted inside the two-bone scope must not fall out through a following else; final FinishSkin is still after the complete if.
  if loc:arm+=pn
  arm+='FinishSkin:;\n'
  yield 'copyblocks_'+''.join(map(str,order))+f'_{polarity}{guard}{loc}{mask}',h.BASE[:h.START]+arm+h.BASE[h.END:]
if __name__=='__main__':
 t=time.time();items=list(variants());r=[]
 with ThreadPoolExecutor(max_workers=8)as pool:
  for f in as_completed([pool.submit(h.run_one,x)for x in items]):
   r.append(f.result())
   if len(r)%512==0:print('tested',len(r),'best',min((x for x in r if 'nd'in x),key=lambda x:(x['nd'],x['cost'])),flush=True)
 r.sort(key=lambda x:(x.get('nd',99999),x.get('cost',99999)))
 (h.HERE/'copyblocks-results.json').write_text(json.dumps(r,indent=2))
 print(json.dumps({'count':len(r),'valid':sum('nd'in x and not x.get('unresolved')for x in r),'seconds':time.time()-t,'best':r[:4]},indent=2))
