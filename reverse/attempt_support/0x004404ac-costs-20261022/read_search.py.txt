from search import *
import re

def source(n):
    body=BODY
    j=0
    def flag(m):
        nonlocal j
        i=j;j+=1
        return '((const volatile bool &)enable)' if n&(1<<i) else 'enable'
    body=re.sub(r'\benable\b',flag,body)
    j=0
    def observer(m):
        nonlocal j
        i=j;j+=1
        # The declaration is not a read.
        if i==0:return 'observer'
        return '((const volatile bool &)observer)' if n&(1<<(5+i)) else 'observer'
    body=re.sub(r'\bobserver\b',observer,body)
    return HEAD+MARK+body

if __name__=='__main__':run([('read-%04d'%n,source(n)) for n in range(4096)],'reads')
