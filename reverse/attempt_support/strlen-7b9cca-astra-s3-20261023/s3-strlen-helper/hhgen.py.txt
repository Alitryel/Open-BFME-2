#!/usr/bin/env python3
# astra-s3 helper seat: isolated strlen variant runner (scratch TU per variant).
# Usage: hhgen.py <phase>
#   phase 1 -> hh_corpus  -> hh.jsonl   (core corpus)
#   phase 2 -> hh2_corpus -> hh2.jsonl  (round-2 targeted corpus)
#   phase 3 -> flag/profile sweep of the best shapes from hh.jsonl+hh2.jsonl -> hh3.jsonl
import sys, json, subprocess, concurrent.futures, itertools, time
from pathlib import Path

ROOT = Path("/home/claude/decomp/s3")
sys.path.insert(0, str(ROOT / "tools"))
sys.path.insert(0, str(Path(__file__).resolve().parent))
import build  # noqa: E402

OUT = ROOT / "build" / "s3-strlen-helper"
TARGET = build.read_target_bytes(0x7B9CCA, 24)

def run_one(job):
    idx, tag, flags, profile, ext, body = job
    src = OUT / ("hh_%04d.%s" % (idx, ext))
    obj = src.with_suffix(".obj")
    if obj.exists():
        obj.unlink()
    head = ("// compiler: %s\n" % profile if profile else "") + "// cl: %s\n" % flags
    src.write_text(head + body + "\n")
    rec = dict(index=idx, tag=tag, flags=flags, profile=profile, ext=ext, src=str(src.name))
    try:
        command, env = build.compiler_command(src, obj)
        p = subprocess.run(command, cwd=ROOT, env=env, capture_output=True, text=True, timeout=120)
        rec["returncode"] = p.returncode
        rec["diagnostic"] = (p.stdout + p.stderr).strip()[:600]
        if p.returncode == 0:
            symbols = build.defined_code_symbols(obj)
            matches = [s for s in symbols if "rva007B9CCAStrLen" in s]
            if not matches:
                rec["error"] = "symbol not defined"
            else:
                buf, relocs = build.read_object_symbol_bytes(obj, matches[0], code_only=True)
                rec["size"] = len(buf)
                rec["hex"] = buf.hex()
                rec["relocs"] = relocs
                rec["exact"] = buf == TARGET and not relocs
                rec["diff"] = sum(a != b for a, b in itertools.zip_longest(buf, TARGET))
    except Exception as exc:
        rec["error"] = "%s: %s" % (type(exc).__name__, exc)
    return rec

def main():
    phase = sys.argv[1] if len(sys.argv) > 1 else "1"
    if phase == "1":
        import hh_corpus as corpus
        logname = "hh.jsonl"
        jobs = [(i, t, f, p, e, b) for i, (t, f, p, e, b) in enumerate(corpus.VARIANTS)]
    elif phase == "2":
        import hh2_corpus as corpus
        logname = "hh2.jsonl"
        jobs = [(500 + i, t, f, p, e, b) for i, (t, f, p, e, b) in enumerate(corpus.VARIANTS)]
    elif phase == "3":
        import hh3_corpus as corpus
        logname = "hh3.jsonl"
        jobs = [(800 + i, t, f, p, e, b) for i, (t, f, p, e, b) in enumerate(corpus.VARIANTS)]
    elif phase == "4":
        import hh4_corpus as corpus
        logname = "hh4core.jsonl"
        jobs = [(1200 + i, t, f, p, e, b) for i, (t, f, p, e, b) in enumerate(corpus.VARIANTS)]
    elif phase == "5":
        import hh5_corpus as corpus
        logname = "hh5core.jsonl"
        jobs = [(1500 + i, t, f, p, e, b) for i, (t, f, p, e, b) in enumerate(corpus.VARIANTS)]
    elif phase == "6":
        import hh6_corpus as corpus
        logname = "hh6core.jsonl"
        jobs = [(1800 + i, t, f, p, e, b) for i, (t, f, p, e, b) in enumerate(corpus.VARIANTS)]
    else:
        # phase 5: sweep flag/profile variants over the best distinct shapes so far
        import hh_corpus, hh2_corpus, hh3_corpus, hh4_corpus, hh5_corpus, hh6_corpus
        bodies = {}
        for v in (hh_corpus.VARIANTS + hh2_corpus.VARIANTS + hh3_corpus.VARIANTS
                  + hh4_corpus.VARIANTS + hh5_corpus.VARIANTS + hh6_corpus.VARIANTS):
            bodies.setdefault(v[0], (v[3], v[4]))
        shapes = {}
        for name in ("hh.jsonl", "hh2.jsonl", "hh3.jsonl", "hh4core.jsonl", "hh5core.jsonl", "hh6core.jsonl"):
            try:
                for line in open(OUT / name):
                    r = json.loads(line)
                    if r.get("hex") is not None and r.get("returncode") == 0:
                        shapes.setdefault((r["diff"], r["size"], r["hex"]), []).append(r)
            except FileNotFoundError:
                pass
        picked = []
        for key in sorted(shapes):
            if len(picked) >= 5:
                break
            for r in shapes[key]:
                if r["tag"] in bodies:
                    picked.append((r["tag"],) + bodies[r["tag"]])
                    break
        jobs = []
        idx = 0
        for tag, ext, body in picked:
            for flags in ["/O1 /G6 /MD", "/O1 /G5 /MD", "/O2 /G6 /MD"]:
                jobs.append((950 + idx, "%s|F%s" % (tag, flags.replace(" ", "")), flags, "", ext, body))
                idx += 1
            for prof in ["vc71-sp1", "vc71-ddk-4035"]:
                jobs.append((950 + idx, "%s|P%s" % (tag, prof), "/O1 /G6 /MD", prof, ext, body))
                idx += 1
        logname = "hh5.jsonl"
        print("phase5 jobs:", len(jobs))
    build.ensure_reference_current()
    start = time.time()
    with (OUT / logname).open("w") as fh, concurrent.futures.ThreadPoolExecutor(max_workers=2) as pool:
        for i, rec in enumerate(pool.map(run_one, jobs)):
            fh.write(json.dumps(rec) + "\n")
            fh.flush()
            if rec.get("exact") or rec.get("diff", 99) <= 12 or i % 25 == 0:
                print(json.dumps({k: rec.get(k) for k in ("tag", "size", "diff", "exact", "hex", "returncode", "error")}), flush=True)
    print("DONE", len(jobs), "seconds", time.time() - start, flush=True)

if __name__ == "__main__":
    main()
