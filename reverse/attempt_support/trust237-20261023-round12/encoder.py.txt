from pathlib import Path
import hashlib,itertools,json,struct,sys
sys.path.insert(0,'build/trust237-round4/venv/lib/python3.12/site-packages')
import pefile,unicorn
from unicorn import Uc,UC_ARCH_X86,UC_MODE_32
from unicorn.x86_const import UC_X86_REG_EAX,UC_X86_REG_EIP,UC_X86_REG_ESP,UC_X86_REG_EBX,UC_X86_REG_EBP,UC_X86_REG_ESI,UC_X86_REG_EDI,UC_X86_REG_EDX
ROOT=Path.cwd();OUT=ROOT/'build/trust237-round12'
path=ROOT/'reference/open-bfme-1/inputs/toolchains/vs2003/Program Files/Microsoft Visual Studio .NET 2003/Vc7/bin/c2.dll'
data=path.read_bytes();sha=hashlib.sha256(data).hexdigest()
assert sha=='bcd28f39b1798b07989c30a1df471cf87fc20e3802d130901a08f83800e6b81e'
pe=pefile.PE(data=data);base=pe.OPTIONAL_HEADER.ImageBase;image=pe.get_memory_mapped_image()
uc=Uc(UC_ARCH_X86,UC_MODE_32);uc.mem_map(base,(len(image)+4095)&~4095);uc.mem_write(base,image)
OUTBUF=0x2000000;NODE=0x2001000;SETTINGS=0x2002000;STACK=0x3000000;ESP=STACK+0x8000;STOP=0x4000000
uc.mem_map(OUTBUF,0x3000);uc.mem_map(STACK,0x10000);uc.mem_map(STOP,0x1000)
uc.mem_write(0x10883370,struct.pack('<I',SETTINGS))
table=[]
for n in range(1,16):
    address=struct.unpack('<I',pe.get_data(0x10870c5c-base+4*n,4))[0]
    table.append(pe.get_data(address-base,n))
def model(n):return table[14]*(n//15)+(table[n%15-1] if n%15 else b'')
sections=[]
for va in [0x10870c5c,0x10870d1c]:
    section=pe.get_section_by_rva(va-base)
    assert not section.Characteristics&0x80000000
    sections.append(dict(address=hex(va),section=section.Name.rstrip(b'\0').decode(),characteristics=hex(section.Characteristics),writable=False))
# Opcode A9 and node type 0D are observed in round7's positive trace.
# Byte9=1 is constructor state; |2 is the identified alignment-node flag.
# The observed word at +A is 6000; the encoder masks it with 0FFF.
# That masked word, or stored byte +16 when zero, chooses the padding size.
# Argument 3 is context, not size: the entry overwrites its stack home.
rows=[]
for context,flags,word,stored in itertools.product([0,1,16],[1,3],[high|n for high in [0,0x6000] for n in range(65)],[1,7,16]):
    node=bytearray(0x40);struct.pack_into('<I',node,4,0xa9)
    node[8]=0xd;node[9]=flags;struct.pack_into('<H',node,0xa,word);node[0x16]=stored
    uc.mem_write(NODE,bytes(node));uc.mem_write(OUTBUF,b'\xcc'*0x400)
    uc.mem_write(ESP,struct.pack('<5I',STOP,NODE,OUTBUF,context,0))
    uc.reg_write(UC_X86_REG_ESP,ESP);uc.reg_write(UC_X86_REG_EDX,0)
    saved=[(UC_X86_REG_EBX,0x11111111),(UC_X86_REG_ESI,0x22222222),(UC_X86_REG_EDI,0x33333333),(UC_X86_REG_EBP,0x44444444)]
    for reg,value in saved:uc.reg_write(reg,value)
    uc.emu_start(0x107b2250,STOP,count=4096)
    assert uc.reg_read(UC_X86_REG_EIP)==STOP
    size=(word&0xfff) or stored;raw=bytes(uc.mem_read(OUTBUF,size))
    assert uc.reg_read(UC_X86_REG_EAX)==size,(context,flags,word,stored,size,hex(uc.reg_read(UC_X86_REG_EAX)),bytes(uc.mem_read(OUTBUF,32)).hex())
    assert uc.reg_read(UC_X86_REG_ESP)==ESP+20
    assert all(uc.reg_read(reg)==value for reg,value in saved)
    assert bytes(uc.mem_read(OUTBUF+size,1))==b'\xcc'
    assert raw==model(size),(context,flags,word,stored,raw.hex())
    rows.append(dict(context_argument=context,flags=flags,attribute_word=hex(word),stored=stored,size=size,raw=raw.hex(),nop16=raw==b'\x90'*16))
assert len(rows)==2340 and not any(r['nop16'] for r in rows)
result=dict(compiler_sha256=sha,entry='0x107b2250',stop=hex(STOP),unicorn=unicorn.__version__,executions=len(rows),context_arguments=[0,1,16],flags=[1,3],attribute_lengths=[0,64],attribute_high_bits=['0x0','0x6000'],stored_lengths=[1,7,16],all_model_abi_checks_passed=True,nop16_count=0,npad16=model(16).hex(),read_only_tables=sections,rows_sha256=hashlib.sha256(json.dumps(rows,sort_keys=True).encode()).hexdigest())
(OUT/'encoder.json').write_text(json.dumps(result,indent=2)+'\n')
print(result)
