from pathlib import Path
import sys,itertools,concurrent.futures,json
sys.path.insert(0,str(Path(__file__).parent));import trials
trials.scratch=Path(__file__).resolve().parent
base=(trials.scratch/'bank.cpp').read_text();trials.symbol=base.splitlines()[0][3:]
header=Path('vendor/stlport/stl/_string.h').read_text()
old='~basic_string() { _Destroy(this->_M_start, this->_M_finish + 1); }'
assert header.count(old)==1
ctor='template <> _String_base<wchar_t,allocator<wchar_t> >::_String_base(const allocator<wchar_t>&);'
dtor='template <> _String_base<wchar_t, allocator<wchar_t> >::~_String_base();'
proxy='''
#define _STLP_alloc_proxy _DiagVendorProxy
#include <stl/_alloc.h>
#undef _STLP_alloc_proxy
namespace _STL {
enum { _DiagLeak = 0 };
template<class Value,class T,class Alloc>
class _STLP_alloc_proxy:public Alloc {
public:
 Value _M_data;
 _STLP_alloc_proxy(const Alloc&a,Value p):Alloc(a),_M_data(p) {}
 ~_STLP_alloc_proxy() EXCEPTION { if (_DiagLeak) free(_M_data); }
};
}
'''
specs=[]
for implicit,p,visible,opaque in itertools.product([0,1],[0,1,2],[0,1],[0,1]):
 label=f'implicit-i{implicit}-p{p}-v{visible}-o{opaque}'
 s=base
 if implicit:
  d=trials.scratch/label/'stl';d.mkdir(parents=True,exist_ok=True)
  (d/'_string.h').write_text(header.replace(old,'// POD character destruction is implicit in this diagnostic.'))
  s=s.replace('// cl:','// cl: /I'+str(d.parent.relative_to(Path.cwd())),1)
 if p:s=s.replace('#define private public',proxy.replace('EXCEPTION','throw()' if p==2 else '')+'\n#define private public',1)
 if visible:s=s.replace(dtor,'')
 if not opaque:s=s.replace(ctor,'')
 specs.append((label,s))
with concurrent.futures.ThreadPoolExecutor(max_workers=8) as pool:r=list(pool.map(trials.run,specs))
(trials.scratch/'implicit_proxy.json').write_text(json.dumps(r,indent=2))
