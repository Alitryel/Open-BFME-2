from itertools import product
from pathlib import Path
W=Path(__file__).resolve().parent
bank=(W/'bank.cpp').read_text();control=(W/'uniform-control.cpp').read_text()
D=W/'contracts';D.mkdir(exist_ok=True)
cv=['',' const',' volatile',' const volatile']
attrs=['','__declspec(nothrow) ','__declspec(noalias) ','__declspec(restrict) ']
for i,(p,t,r,a,b,c) in enumerate(product(range(4),range(4),range(4),range(4),range(2),range(2))):
 s=control if c else bank
 if b:
  s=s.replace('bridge(node->second,node->point,node->normal)','node->second->source->prepare(node->point,node->normal)')
 s=s.replace('virtual void *prepare(const ContactVector &point, const ContactVector &normal)=0;',attrs[a]+'virtual void *prepare(const ContactVector &point, const ContactVector &normal)'+cv[p]+'=0;')
 s=s.replace('virtual bool test(void *info)=0;','virtual bool test(void *info)'+cv[t]+'=0;')
 s=s.replace('virtual void respond(void *info)=0;','virtual void respond(void *info)'+cv[r]+'=0;')
 (D/('v%04d.cpp'%i)).write_text(s)
print('generated',i+1)
