cb=open('build/q8/cb.cpp').read()
H='void AIPlayer::queueSupplyTruck( void )'
cb=cb.replace(H,'extern int g_x1; extern int g_x2;\n'+H,1)
lines=cb.split('\n')
start=[i for i,l in enumerate(lines) if l.startswith(H)][0]
n=0
for i in range(start+2,len(lines)-2):
    l=lines[i].rstrip()
    if not l.strip(): continue
    if l.strip().startswith('//') or l.strip().startswith('/*'): continue
    if not (l.endswith(';') or l.endswith('{') or l.endswith('}')): continue
    if l.strip().startswith('for') or l.strip().startswith('while') or l.strip().startswith('if') and not l.endswith('{'): continue
    if l.strip()=='} else {': continue
    L=lines[:i+1]+['if (g_x1) g_x2=1;']+lines[i+1:]
    open('build/q9/y/p%04d.cpp'%(i+1),'w').write('\n'.join(L))
    n+=1
print(n)
