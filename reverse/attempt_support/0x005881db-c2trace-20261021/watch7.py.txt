import sys
sys.path.insert(0,'/work2/Open-BFME-2/build/c2tr')
from wdbg import Dbg
# watch7.py SRC OBJ ADDR FROMROUND [IDX]: watch dword ADDR between 82978 entries FROMROUND..FROMROUND+1 (updateMeleeAttack cls0)
src,obj=sys.argv[1:3]; A=int(sys.argv[3],16); R=int(sys.argv[4])
d=Dbg(src,obj,obj+'.log')
d.prompt();d.prompt('set $CanDeferOnBPByAddr = 1')
bps=[0x83acd,0x82978]; added=set()
for b in bps:
    o=d.prompt('break *0x%x'%(0x10700000+b))
    if 'Breakpoint' in o and 'Unable' not in o: added.add(b)
on=False;k=0;w=False
while True:
    out=d.prompt('cont')
    if 'terminated' in out: break
    eip=d.reg('eip')-0x10700000
    if w and ('Hardware watchpoint' in out or 'atchpoint' in out) and eip not in (0x83acd,0x82978):
        v=d.words(A,1)[0]; esp=d.reg('esp'); st=d.words(esp,96)
        print('W eip=%x val=%x idx=%x regs eax=%x ecx=%x edx=%x ebx=%x esi=%x edi=%x stk=%s'%(eip,v,d.words(A-0x24+0x1c,1)[0],d.reg('eax'),d.reg('ecx'),d.reg('edx'),d.reg('ebx'),d.reg('esi'),d.reg('edi'),' '.join('%x@%d'%(x-0x10700000,i) for i,x in enumerate(st) if 0x10701000<=x<0x10870000)),flush=True)
        continue
    for b in bps:
        if b not in added and b!=eip: d.prompt('break *0x%x'%(0x10700000+b)); added.add(b)
    added.add(eip)
    if eip==0x83acd:
        ebp=d.reg('ebp'); f=d.words(ebp,1)[0]; curfn=d.string(d.words(f+4,1)[0]); on='updateMeleeAttack' in curfn and d.reg('ebx')==0; k=0; continue
    if not on: continue
    if eip==0x82978:
        k+=1
        if k==R:
            print(d.prompt('watch *(int*)0x%x'%A)[-200:],flush=True); w=True
        elif k>R: break
d.p.kill()
