from pathlib import Path
import json,csv
out=Path(__file__).resolve().parent
rows=[]
for label in ('baseline','gh','optional'):
 p=out/label/'regions.json'
 if not p.exists():continue
 regions=json.loads(p.read_text())
 for r in regions:
  members={k:[i for i,b in enumerate(r['blocks']) if r['index'] in b['sets'][k]] for k in ('0x4c','0x50')}
  refs=[(i,t['opcode'],t['origin_refs']) for i,b in enumerate(r['blocks']) for t in b['tuples'] if t['origin_refs']]
  row=dict(control=label,round=r['round'],role=r['role'],index=r['index'],priority=r['priority'],cost=r['cost'],references=r['refs'],span_blocks=len(r['blocks']),membership_4c=len(members['0x4c']),membership_50=len(members['0x50']))
  rows.append(row)
  if r['role']=='BASE_VTABLE':print(row,'members',members,'refs',refs)
 with (out/label/'membership.json').open('w') as f:json.dump([{'round':r['round'],'role':r['role'],'index':r['index'],'origin':r['origin'],'block_addresses':[b['address'] for b in r['blocks']],'membership':{k:[i for i,b in enumerate(r['blocks']) if r['index'] in b['sets'][k]] for k in ('0x4c','0x50')}} for r in regions],f,indent=2)
with (out/'allocation.csv').open('w') as f:
 w=csv.DictWriter(f,fieldnames=list(rows[0]));w.writeheader();w.writerows(rows)
