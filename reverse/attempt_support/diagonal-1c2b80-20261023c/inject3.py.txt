import sys,json
sys.path.insert(0,'/work/Open-BFME-2/build/c2tr')
from wdbg import Dbg
# inject3.py src tag "K:AT:N,..."  K=15 or 3: consume N ids of kind K after its AT-th arena allocation
src=sys.argv[1]; tag=sys.argv[2]
plan={15:[],3:[]}
for x in sys.argv[3].split(','):
    if not x: continue
    k,a,n=map(int,x.split(':')); plan[k].append((a,n))
for k in plan: plan[k].sort()
BP={15:0x1076045d,3:0x10760853}; AR={15:0x108a79b0,3:0x108a79ac}
W='/work/Open-BFME-2/build/db/tr/'
d=Dbg(src,W+tag+'.obj',W+tag+'.log')
active=False; cnt={15:0,3:0}; pend={15:0,3:0}; log=[]
try:
    d.prompt();d.prompt('set $CanDeferOnBPByAddr = 1');d.prompt('break *0x10760539')
    while True:
        o=d.prompt('cont',300)
        if 'terminated' in o or not o.strip(): break
        if '0x10760539' in o:
            if active: continue
            w=d.words(d.reg('eax'),1);s=d.words(w[0],2);fn=d.string(s[1])
            if 'DiagonalBlur' in fn:
                active=True
                for k in (15,3):
                    if plan[k]: d.prompt('break *0x%x'%BP[k])
            continue
        if not active: continue
        for k in (15,3):
            if ('0x%x'%BP[k]) in o:
                cnt[k]+=1
                while plan[k] and plan[k][0][0]==cnt[k]: pend[k]+=plan[k].pop(0)[1]
                if pend[k]:
                    d.prompt('finish',60)
                    ar=d.words(AR[k],1)[0]; cur,end=d.words(ar+4,2)
                    room=(end-cur)//0x60; t=min(room,pend[k])
                    d.prompt('set *(int*)0x%x = 0x%x'%(ar+4,cur+t*0x60)); pend[k]-=t; log.append((k,cnt[k],t))
                break
        if not any(plan.values()) and not any(pend.values()):
            d.prompt('delete'); active=False; break_out=True
            # let it run to completion
            while True:
                o=d.prompt('cont',300)
                if 'terminated' in o or not o.strip(): break
            break
finally:
    print('log',log,flush=True)
    if d.p.poll() is None:
        try: d.prompt('quit')
        except Exception: d.p.kill()
