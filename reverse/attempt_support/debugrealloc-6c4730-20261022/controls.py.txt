"""Replay from the repository root; writes only private build/ objects.

The historical bank is intentional: it isolates the footer change. Production
also replaces its obsolete Rva00033150 alias with the recovered allocator owner.
"""
from pathlib import Path
import hashlib
import json
import subprocess
import sys

ROOT = Path.cwd()
sys.path.insert(0, str(ROOT / "tools"))
from variant_search import Scorer
from locate import object_functions

NAME = "?rva006C4730@GeneralAllocatorDebug@@QAEPAXPAXIH@Z"
REV = "1bf83f28eb5e87b1e456076677ae5d0836ce6134"
bank = subprocess.run(
    ["git", "show", REV + ":reverse/attempts/0x006c4730.cpp"],
    cwd=ROOT, capture_output=True, check=True,
).stdout.decode("utf-8")
old = """\t\t\tchar *end = !(h & 2) ? (char *)p + (h & 0x7FFFFFF8) + 4 : (char *)p + (h & 0x7FFFFFF8);
\t\t\t*(unsigned short *)(end - 10) = 0;"""
assert bank.count(old) == 1
size = "unsigned n=!(h&2)?(h&0x7ffffff8)+4:(h&0x7ffffff8);"
reuse = "if(!(h&2)){h&=0x7ffffff8;h+=4;}else h&=0x7ffffff8;"
pointer = "unsigned short*t;if(!(h&2)){unsigned n=(h&0x7ffffff8)+4;t=(unsigned short*)((char*)p+n-10);}else{unsigned n=h&0x7ffffff8;t=(unsigned short*)((char*)p+n-10);}*t=0;"
forms = {
    "bank": old,
    "separate-size-word": size + "*(unsigned short*)((char*)p+n-10)=0;",
    "per-arm-named-pointer": pointer,
    "separate-size-intrinsic": size + "char*t=(char*)p+n-10;memset(t,0,2);",
    "reuse-header-word": reuse + "char*t=(char*)p+h-10;*(unsigned short*)t=0;",
    "reuse-header-intrinsic-WINNER": reuse + "char*t=(char*)p+h-10;memset(t,0,2);",
}
out = ROOT / "build/debugrealloc_controls_replay"
scorer = Scorer(NAME, 0x6C4730, 798, out)
results = []
for label, form in forms.items():
    result = scorer(bank.replace(old, form))
    assert result["fitness"] >= 0, (label, result)
    obj = out / ("c%d.obj" % scorer.n)
    raw = next(body.rstrip(b"\xcc") for name, body, relocs in object_functions(obj) if name == NAME)
    exact = result["exact"] and len(raw) == 798
    row = dict(control=label, extent=len(raw), fitness=result["fitness"],
               exact=exact, raw_sha256=hashlib.sha256(raw).hexdigest())
    results.append(row)
    print(json.dumps(row), flush=True)
    if label.endswith("WINNER"):
        assert exact, "Winning source no longer reproduces; compare compiler and inputs"
(out / "results.json").write_text(json.dumps(results, indent=2) + "\n")
