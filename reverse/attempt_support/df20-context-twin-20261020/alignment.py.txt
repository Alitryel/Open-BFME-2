import trials,itertools,concurrent.futures,json
src=trials.bank.replace('__declspec(align(8)) money_base::pattern','money_base::pattern')
targets={'none':None,'currency':'string_type __curr_sym','sign':'string_type __sign','digits':'string_type __new_digits','locale':'locale __loc','pattern':'money_base::pattern __format','width':'size_t __width','value_length':'size_t __value_length','length':'size_t __length','fill':'size_t __fill_amt','negative':'bool __is_negative','policy':'_MoneySignPolicy __sign_policy','plus':'char_type __plus'}
specs=[]
for label,target in targets.items():
 for intbool,unscoped in itertools.product([0,1],repeat=2):
  s=src
  if target:s=s.replace(target,'__declspec(align(8)) '+target)
  if intbool:s=s.replace('bool __is_negative','int __is_negative')
  if unscoped:
   start=s.index('  {\n    for (int __i = 0; __i < 4; ++__i)')
   end=s.index('\n\n',start)
   s=s[:start]+s[start:end].replace('  {\n','  int __j;\n',1).replace('for (int __i = 0; __i < 4; ++__i)','for (__j = 0; __j < 4; ++__j)').replace('__format.field[__i]','__format.field[__j]').rsplit('  }',1)[0]+s[end:]
  specs.append((f'align-{label}-b{intbool}-u{unscoped}',s))
with concurrent.futures.ThreadPoolExecutor(max_workers=8) as p:r=list(p.map(trials.run,specs))
(trials.scratch/'alignment.json').write_text(json.dumps(r,indent=2)+'\n')
