"""Build trial copies of the home unit with the bank's edits plus a candidate body."""
import os
import re
import subprocess
import sys
from pathlib import Path

ROOT = Path(__file__).resolve().parents[2]
HOME = ROOT / 'Code/GameEngine/Source/GameLogic/Object/Contain/HordeContainIface11CSlots.cpp'
OUT = ROOT / 'build/w2-46d2c3/trials'
SYM = '?rva0046D2C3@HordeContain@@UAEPAVObject@@H@Z'


def rep(s, old, new, count=1):
    assert s.count(old) == count, (old, s.count(old))
    return s.replace(old, new)


def absolutize(text):
    home_dir = HOME.parent

    def fix(m):
        rel = m.group(1)
        p = (home_dir / rel).resolve()
        if p.exists():
            return '#include "%s"' % p.as_posix()
        return m.group(0)
    return re.sub(r'#include "([^"]+)"', fix, text)


def home_text(edits=None):
    raw = HOME.read_bytes().decode('utf-8')
    h = raw.replace('\r\n', '\n')
    h = rep(h, 'virtual Object *rva0046D27A() = 0; virtual void gap69() = 0;',
            'virtual Object *rva0046D27A() = 0; virtual Object *rva0046D2C3(int kind) = 0;')
    h = rep(h, '\tvirtual Object *rva0046D27A();\n\tvirtual void rva0046D7AF(int a1);',
            '\tvirtual Object *rva0046D27A();\n\tvirtual Object *rva0046D2C3(int kind);\n\tvirtual void rva0046D7AF(int a1);')
    h = rep(h, 'class ThingTemplate\n{\npublic:\n',
            'class ThingTemplate\n{\npublic:\n\tbool isKindOf(KindOfType t) const;\n')
    # KindOfType must precede ThingTemplate.
    h = rep(h, 'enum KindOfType { KINDOF_INVALID = -1 };\n', '')
    h = rep(h, 'class ThingTemplate\n{\npublic:\n', 'enum KindOfType { KINDOF_INVALID = -1 };\nclass ThingTemplate\n{\npublic:\n')
    h = rep(h, '\tconst ThingTemplate *getTemplate() const { return m_template; }\n',
            '\tconst ThingTemplate *getTemplate() const { return m_template; }\n\tbool isTemplateKindOf(KindOfType kind) const { return getTemplate()->isKindOf(kind); }\n')
    if edits:
        for old, new in edits:
            h = rep(h, old, new)
    return h

def write_trial(name, body, edits=None, absolute=True):
    OUT.mkdir(parents=True, exist_ok=True)
    text = home_text(edits) + body
    if absolute:
        text = absolutize(text)
    p = OUT / (name + '.cpp')
    p.write_text(text, encoding='utf-8', newline='\n')
    obj = ROOT / 'build/match' / (p.relative_to(ROOT).as_posix().replace('/', '_')[:-4] + '.obj')
    return p


def measure(name, body, edits=None):
    p = write_trial(name, body, edits)
    rel = p.relative_to(ROOT).as_posix()
    sys.path.insert(0, str(ROOT / 'tools'))
    import build
    obj = build.obj_path(p)
    for ext in ('.obj', '.deps.json'):
        q = Path(str(obj)[:-4] + ext)
        if q.exists():
            q.unlink()
    env = dict(os.environ, BFME_OBJSTORE='off', BFME_HOST_TAG='fleet')
    proc = subprocess.run([sys.executable, 'tools/align_diff.py', rel, SYM, '0x0046D2C3', '175'],
                          cwd=ROOT, capture_output=True, text=True, env=env)
    out = proc.stdout + proc.stderr
    (OUT / (name + '.align.txt')).write_text(out, encoding='utf-8')
    return out

# ===== run.py =====
"""Measure many candidate bodies in parallel and summarize the register webs."""
import re
import sys
from concurrent.futures import ThreadPoolExecutor
from pathlib import Path

sys.path.insert(0, str(Path(__file__).resolve().parent))
import prep


def summarize(name, out):
    m = re.search(r'ours (\d+) B', out)
    size = m.group(1) if m else '?'
    m = re.search(r'structural regions: (\d+)', out)
    regions = m.group(1) if m else '?'
    mem = re.search(r'(\d+)\s+memory', out)
    reg = re.search(r'(\d+)\s+register', out)
    exact = 'EXACT' if 'structural regions: 0' in out and mem and int(mem.group(1)) <= 2 and reg and int(reg.group(1)) == 0 else ''
    err = ''
    if 'error' in out.lower() and size == '?':
        err = [l for l in out.splitlines() if 'error' in l.lower()][:2]
    return '%-28s size=%s regions=%s mem=%s reg=%s %s %s' % (
        name, size, regions, mem.group(1) if mem else '?', reg.group(1) if reg else '?', exact, err)


def run(bodies, edits_for=None, workers=6):
    def go(k):
        edits = edits_for(k) if edits_for else None
        out = prep.measure(k, bodies[k], edits)
        return summarize(k, out)
    with ThreadPoolExecutor(workers) as ex:
        for line in ex.map(go, list(bodies)):
            print(line, flush=True)


def webs(name):
    """Registers of this/end/it/k/obj read off our object's disassembly."""
    import capstone
    sys.path.insert(0, str(prep.ROOT / 'tools'))
    import build
    from locate import object_functions
    obj = build.obj_path(prep.OUT / (name + '.cpp'))
    fn = {n: s for n, s, r in object_functions(obj)}.get(prep.SYM)
    if fn is None:
        return 'no-symbol'
    md = capstone.Cs(capstone.CS_ARCH_X86, capstone.CS_MODE_32)
    ins = [(i.mnemonic, i.op_str) for i in md.disasm(fn.rstrip(b'\xcc'), 0)]
    res = {}
    for i, (m, o) in enumerate(ins):
        if 'this' not in res and m == 'mov' and o.endswith(', ecx'):
            res['this'] = o.split(',')[0]
        if 'end' not in res and m == 'mov' and o.endswith('dword ptr [eax]') and 'call' in [x[0] for x in ins[:i]]:
            res['end'] = o.split(',')[0]
            nm, no = ins[i + 1]
            res['it'] = no.split(',')[0]
        if 'k' not in res and m == 'mov' and '+ 0x54]' in o:
            nm, no = ins[i + 1]
            res['k'] = no.split(',')[0]
        if 'obj' not in res and m == 'mov' and o.endswith(', eax') and i > 0 and ins[i - 1][0] == 'call' and 'k' in res:
            res['obj'] = o.split(',')[0]
    return ' '.join('%s=%s' % (k, res.get(k, '?')) for k in ('this', 'end', 'it', 'k', 'obj'))


def run2(bodies, edits_for=None, workers=6):
    def go(k):
        edits = edits_for(k) if edits_for else None
        out = prep.measure(k, bodies[k], edits)
        return summarize(k, out) + '  ' + webs(k)
    with ThreadPoolExecutor(workers) as ex:
        for line in ex.map(go, list(bodies)):
            print(line, flush=True)
