from pathlib import Path
import itertools

root = Path(__file__).parent
base = Path('reverse/attempts/0x0027893e.cpp').read_text()
out = root / 'guard_inline'
out.mkdir(parents=True, exist_ok=True)
tail_at = base.index('\t\tif (!(m_status & mask80) && !obj->isEffectivelyDead()) {')
prefix = base[:tail_at]
for member, guard, eventarg, objectref, force, separate, tail in itertools.product(range(2),range(4),range(2),range(2),range(2),range(2),range(2)):
    # A complete real status transition is now the inlining unit. No dummy
    # uses, extra observable reads or operations are added.
    qualifier = '__forceinline' if force else 'inline'
    op = 'Object &object' if objectref else 'Object *object'
    oe = '&object' if objectref else 'object'
    eventparams = 'const int &eventSource' if eventarg else 'const ThingTemplate *eventTemplate, bool revealed'
    event = 'eventSource' if eventarg else '(revealed ? eventTemplate->m_revealedEvent : eventTemplate->m_shroudedEvent)'
    status = 'm_status' if member else 'status'
    params = op+', '+eventparams+', int mask80'
    if not member: params='unsigned int &status, '+params
    if tail: params += ', int &notifyEvent'
    rettype = 'bool' if tail else 'void'
    success = 'notifyEvent = '+event+'; return true;' if tail else ('int event = '+event+'; if (event != -1) { Object *nearest = ((Drawable *)('+oe+'))->rva00272AD5(); if (nearest) TheEva->reportEvaEvent((EvaEventID)event, nearest->getPosition(), ('+oe+')->getPosition()); }')
    dead='('+oe+')->isEffectivelyDead()'
    write=status+' |= mask80; '+success
    fail='return false;' if tail else 'return;'
    body=[
        'if (!('+status+' & mask80) && !'+dead+') { '+write+' } '+fail,
        'if ('+status+' & mask80) { '+fail+' } if ('+dead+') { '+fail+' } '+write,
        'if (!('+status+' & mask80)) { if (!'+dead+') { '+write+' } } '+fail,
        'do { if ('+status+' & mask80) break; if ('+dead+') break; '+write+' } while (false); '+fail,
    ][guard]
    helpers=[]; decls=[]; calls=[]
    for arm in range(2):
        name='notifyStatusInView'+(str(arm) if separate else '')
        signature=qualifier+' '+rettype+' '+(('Drawable::' if member else '')+name)+'('+params+')'
        if arm==0 or separate:
            helpers.append(signature+' { '+body+' }')
            if member: decls.append(qualifier+' '+rettype+' '+name+'('+params+');')
        args=['*obj' if objectref else 'obj']
        args += ['tmpl->m_'+('shroudedEvent' if arm else 'revealedEvent')] if eventarg else ['tmpl','false' if arm else 'true']
        args += ['mask80']
        if not member: args.insert(0,'m_status')
        if tail: args += ['notifyEvent']
        calls.append(name+'('+', '.join(args)+')')
    source=prefix
    if tail:
        source += '\t\tif ('+calls[0]+') goto completeNotify;\n\t} else {\n\t\tif (((Rva00270260 *)this)->rva00270260()) return;\n\t\tint mask80=0x80;\n\t\tif ('+calls[1]+') goto completeNotify;\n\t}\n\treturn;\ncompleteNotify:\n'
        source += '\tif (notifyEvent != -1) { Object *nearest=((Drawable *)obj)->rva00272AD5(); if (nearest) TheEva->reportEvaEvent((EvaEventID)notifyEvent,nearest->getPosition(),obj->getPosition()); }\n}\n'
    else:
        source=source.replace('\tint notifyEvent;\n','')
        source += '\t\t'+calls[0]+';\n\t} else {\n\t\tif (((Rva00270260 *)this)->rva00270260()) return;\n\t\tint mask80=0x80;\n\t\t'+calls[1]+';\n\t}\n}\n'
    if member:
        source=source.replace('\tvoid onComingOutOfShroud();','\tvoid onComingOutOfShroud();\n\t'+'\n\t'.join(decls))
    at=source.index('void Drawable::onComingOutOfShroud()')
    source=source[:at]+'\n'.join(helpers)+'\n'+source[at:]
    name=f'm{member}g{guard}e{eventarg}o{objectref}f{force}s{separate}t{tail}.cpp'
    (out/name).write_text(source)
print(len(list(out.glob('*.cpp'))))
