from pathlib import Path
import sys,itertools
sys.path.insert(0,str(Path('build/trust237-round3').resolve()))
import search
mode=sys.argv[1]
forms=[
 'return P;',
 'int r=0; if(P) ++r; return r;',
 'int r=0; if(P) goto yes; return r; yes:return r+1;',
 'int r=0; do{if(!(P))break; ++r;}while(0); return r;',
 'bool r=P; return r;',
 'int r=0; for(unsigned i=(P);i;--i) ++r; return r;',
 'int r=0; switch((bool)(P)){case true:++r;break;case false:break;}return r;',
 'return (P)?1:0;',
]
if mode=='wide':
 preds=[
 'x<=0xffffffffui64',
 '(x|0x100000000ui64)!=0ui64',
 '(x+1ui64)>x',
 'x*x<=0xfffffffe00000001ui64',
 '(x^(x+1ui64))!=0ui64',
 'x%0x100000000ui64==x',
 'x/0x100000000ui64==0ui64',
 '((x<<32)>>32)==x',
 '(~x>>32)==0xffffffffui64',
 '((x<<32)|(x^0xffffffffui64))!=0ui64',
 'x*0x100000001ui64==((x<<32)+x)',
 'x+(0xffffffffui64-x)==0xffffffffui64',
 '(x*(x+1ui64)&1ui64)==0ui64',
 '(x|1ui64)%2ui64==1ui64',
 '(x&0xffffffff00000000ui64)==0ui64',
 '(unsigned __int64)(unsigned char)u<=255ui64',
 '(x^(x>>32))==x',
 '(x&(x-1ui64))<x || x==0ui64',
 '(x&(x+1ui64))<=x',
 '(x|(x+1ui64))>=x',
 'x==0ui64 || x/x==1ui64',
 '(x*(x+1ui64))%2ui64==0ui64',
 '((x>>31)+((x^0x80000000ui64)>>31))==1ui64',
 '(x-1ui64)<x || x==0ui64',
 ]
 choices=itertools.product(range(len(preds)),range(8),range(2),range(2),range(2),range(2))
 jobs=[]
 for pi,fi,timing,cpu,helper,scope in choices:
  pred=preds[pi];helpers=''
  if helper:
   helpers='__forceinline bool widePredicate(unsigned __int64 x,unsigned long u){return '+pred+';}\n'
   pred='widePredicate(x,u)'
  tail=forms[fi].replace('P','('+pred+')')
  if timing:tail='unsigned __int64 x=(unsigned __int64)u; LocalFree(argv); '+tail
  else:tail='LocalFree(argv); unsigned __int64 x=(unsigned __int64)u; '+tail
  if scope:tail='{'+tail+'}'
  s=search.base.replace(search.marker,'    long status=WinVerifyTrust(0,&action,&data); unsigned long u=(unsigned long)status; '+tail)
  s=s.replace('int Rva00237D3DCheck() {',helpers+'int Rva00237D3DCheck() {')
  if not cpu:s=s.replace('/G7','/G6')
  jobs.append((f'w{len(jobs):04}',s,dict(pred=pi,form=fi,timing=timing,cpu=cpu,helper=helper,scope=scope)))
elif mode=='float':
 preds=['x==x','!(x!=x)','x-x==0','x>=0 || x<0','x<=4294967296.0','x>=0 && x<=4294967296.0','x*0==0','x+1>x || x+1==x','x*x>=0','x/4294967296.0<=1','x+(-x)==0','x>=0']
 choices=itertools.product(range(len(preds)),range(4),range(3),range(2),range(2),range(3))
 jobs=[]
 for pi,fi,typ,timing,cpu,fp in choices:
  ty=['float','double','long double'][typ]
  tail=forms[fi].replace('P','('+preds[pi]+')')
  if timing:tail=ty+' x=('+ty+')u; LocalFree(argv); '+tail
  else:tail='LocalFree(argv); '+ty+' x=('+ty+')u; '+tail
  s=search.base.replace(search.marker,'    long status=WinVerifyTrust(0,&action,&data); unsigned long u=(unsigned long)status; '+tail)
  if not cpu:s=s.replace('/G7','/G6')
  s=s.replace('/arch:SSE',['/arch:SSE','/arch:SSE /Op','/arch:IA32'][fp])
  jobs.append((f'f{len(jobs):04}',s,dict(pred=pi,form=fi,type=typ,timing=timing,cpu=cpu,fp=fp)))
else:raise SystemExit(mode)
search.execute(mode,jobs)
