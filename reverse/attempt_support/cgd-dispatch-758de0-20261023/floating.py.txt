from itertools import product
from pathlib import Path
W=Path(__file__).resolve().parent
bank=(W/'bank.cpp').read_text();uniform=(W/'uniform-control.cpp').read_text()
D=W/'floating';D.mkdir(exist_ok=True)
for idx,(opts,copy,control) in enumerate(product(product(range(3),repeat=6),range(2),range(2))):
 s=uniform if control else bank
 defs='union FloatBits {const void *p;float f;};\n'
 body='FloatBits u;u.p=p;return u.f;' if not copy else 'float f; __builtin_dummy;return f;'
 if copy:
  defs+='extern "C" void *__cdecl memcpy(void *,const void *,unsigned);\n#pragma intrinsic(memcpy)\n'
  body='float f;memcpy(&f,&p,4);return f;'
 defs+='__forceinline float floatBits(const void *p) {'+body+'}\n'
 for fast in range(2):
  cc='__fastcall ' if fast else ''
  defs+='''struct FloatSource%d {
 virtual void slot0()=0;virtual void slot1()=0;virtual void slot2()=0;
 virtual void *%sprepare(float p,float n)=0;
 virtual void slot4()=0;virtual void slot5()=0;virtual void slot6()=0;virtual void slot7()=0;
 virtual bool %stest(float info)=0;
 virtual void %srespond(float info)=0;
};\n'''%(fast,cc,cc,cc)
 inners=['node->second->source->prepare(node->point,node->normal)', 'node->second->source->prepare(node->point,node->normal)' if control else 'bridge(node->second,node->point,node->normal)', 'node->first->source->prepare(node->point,ContactVector(node->normal).negate())' if control else 'altPrepare(node->first->source,node->point,ContactVector(node->normal).negate())']
 for j in range(3):
  inner='node->'+('first' if j==2 else 'second')+'->source'
  outer='node->'+('second' if j==2 else 'first')+'->source'
  meth='test' if j==0 else 'respond'
  old=outer+'->'+meth+'('+inners[j]+')'
  newinner=inners[j]
  if opts[2*j]:
   normal='ContactVector(node->normal).negate()' if j==2 else 'node->normal'
   newinner='((FloatSource%d *)%s)->prepare(floatBits(&node->point),floatBits(&(%s)))'%(opts[2*j]-1,inner,normal)
  if opts[2*j+1]:new='((FloatSource%d *)%s)->%s(floatBits(%s))'%(opts[2*j+1]-1,outer,meth,newinner)
  else:new=outer+'->'+meth+'('+newinner+')'
  assert old in s,(idx,j)
  s=s.replace(old,new,1)
 a=s.index('struct ContactData {');s=s[:a]+defs+s[a:]
 (D/('v%04d.cpp'%idx)).write_text(s)
print('generated',idx+1)
