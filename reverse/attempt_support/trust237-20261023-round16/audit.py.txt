target_path=ROOT/'baselines/bfme2/workshop-vanilla-1.06/files/game.dat'
b=target_path.read_bytes();p=pefile.PE(data=b);base=p.OPTIONAL_HEADER.ImageBase
assert hashlib.sha256(b).hexdigest()=='f008b587570bad693981dc7218588c81d192a1e064b0f7f861539c51156a7640'

resolved,unresolved=resolve_bytes(raw,rel,target,0x237d3d,226,{'_memset':[0x6291ae],'_WinVerifyTrust@12':[0x655652]})
assert not unresolved
diff=[i for i,(x,y) in enumerate(zip(resolved,target)) if x!=y]
assert diff==list(range(205,221))
imports={i.address:i.name.decode() for d in p.DIRECTORY_ENTRY_IMPORT for i in d.imports if i.name}

apis=[]
for off,typ,name in rel:
    if typ==6:
        va=struct.unpack_from('<I',target,off)[0];assert imports[va]==names[name]
        apis.append(dict(symbol=name,iat_va=hex(va),target_name=imports[va]))
    else:
        assert typ==20
        rva=0x237d3d+off+4+struct.unpack_from('<i',target,off)[0]
        assert rva=={'_memset':0x6291ae,'_WinVerifyTrust@12':0x655652}[name]
        thunk=p.get_data(rva,6);assert thunk[:2]==b'\xff\x25'
        va=struct.unpack_from('<I',thunk,2)[0]
        assert imports[va]=={'_memset':'memset','_WinVerifyTrust@12':'WinVerifyTrust'}[name]
        apis.append(dict(symbol=name,thunk_rva=hex(rva),target_name=imports[va]))
