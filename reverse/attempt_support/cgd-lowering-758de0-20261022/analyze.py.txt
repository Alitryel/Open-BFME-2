import json,difflib,collections
from pathlib import Path
import search
W=Path(__file__).resolve().parent
T=search.T
bounds=[(0,0x94),(0x94,0xaa),(0xaa,0xbe),(0xbe,0xe1),(0xe1,0x136),(0x136,0x140),(0x140,333)]
def score(data):
 A=search.insns(data);sm=difflib.SequenceMatcher(None,[(x[0],x[1]) for x in A],[(x[0],x[1]) for x in T],autojunk=False)
 amounts=[0]*len(bounds)
 for tag,a,b,c,d in sm.get_opcodes():
  if tag!='equal':continue
  for t in T[c:d]:
   for i,(lo,hi) in enumerate(bounds):
    if lo<=t[3]<hi:amounts[i]+=t[2]
 return amounts
bank=score((W/'visible/v0000.bin').read_bytes());print('bank',bank,sum(bank))
summary={'bank':bank,'bounds':bounds,'families':{}}
for fam in ['compound2','callconv','podparam','visible','exprlife','wrapalias','pairshape','bytereturn','refresult']:
 p=W/(fam+'-results.json');rs=json.loads(p.read_text());good=[r for r in rs if 'error' not in r];uniq={r['sha']:r for r in good}
 rows=[]
 for r in uniq.values():
  path=W/fam/(r['name']+'.bin')
  r['pair_regions']=score(path.read_bytes());r['pair_sum']=sum(r['pair_regions']);rows.append(r)
 best=max(rows,key=lambda r:r['pair_sum']);regionbest=[max(rows,key=lambda r:r['pair_regions'][i]) for i in range(len(bounds))]
 better=[]
 for i,r in enumerate(regionbest):
  if r['pair_regions'][i]>bank[i]:better.append({'region':bounds[i],'record':r})
 info={'total':len(rs),'compiled':len(good),'distinct':len(uniq),'best':best,'region_improvements':better,'sizes':dict(collections.Counter(r['size'] for r in good))}
 summary['families'][fam]=info
 print(fam,'total',len(rs),'compiled',len(good),'distinct',len(uniq),'best',best['name'],best['pair_sum'],best['pair_regions'],'improved',[(b['region'],b['record']['name'],b['record']['pair_regions']) for b in better])
(W/'trials-summary.json').write_text(json.dumps(summary,indent=2))
