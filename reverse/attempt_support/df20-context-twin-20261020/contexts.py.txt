import trials,itertools,concurrent.futures,json
from pathlib import Path
construct=Path('vendor/stlport/stl/_construct.h').read_text()
specs=[('baseline',trials.bank)]
start=trials.bank.index('  string_type __sign = __intl ?')
end=trials.bank.index('#pragma inline_depth(255)',start)
orig=trials.bank[start:end]
expr=orig.split(' = ',1)[1].strip().rstrip(';')
for destroy,visible,nothrow,direct,constant in itertools.product(['generic','scalar-loop','empty-throw'],[0,1],[0,1],[0,1],[0,1]):
 label=f'destroy-{destroy}-v{visible}-nt{nothrow}-d{direct}-c{constant}'
 src=trials.bank
 body={'generic':'__destroy(__first, __last, (wchar_t*)0);','scalar-loop':'for (; __first != __last; ++__first) __first->~wchar_t();','empty-throw':'if (sizeof(wchar_t)==0) free(__first);'}[destroy]
 # A typedef is needed to spell a scalar pseudo-destructor with this compiler.
 if destroy=='scalar-loop':body='typedef wchar_t W; for (; __first != __last; ++__first) __first->~W();'
 override=construct.replace('inline void _Destroy(wchar_t*, wchar_t*) {}','inline void _Destroy(wchar_t* __first, wchar_t* __last) { '+body+' }')
 d=trials.scratch/label/'stl';d.mkdir(parents=True,exist_ok=True)
 (d/'_construct.h').write_text(override)
 src=src.replace('// cl:','// cl: /I'+str(d.parent.relative_to(trials.root)),1)
 decl='template <> _String_base<wchar_t, allocator<wchar_t> >::~_String_base();'
 if visible: src=src.replace(decl,'')
 elif nothrow:src=src.replace(decl,decl[:-1]+' throw();')
 if direct:src=src.replace(orig,'  string_type __sign('+expr+');\n')
 if constant:
  src=src.replace('string_type __sign','const string_type __sign').replace('string_type __curr_sym','const string_type __curr_sym')
 specs.append((label,src))
with concurrent.futures.ThreadPoolExecutor(max_workers=8) as p:r=list(p.map(trials.run,specs))
(trials.scratch/'contexts.json').write_text(json.dumps(r,indent=2)+'\n')
