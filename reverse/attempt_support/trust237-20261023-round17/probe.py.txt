def path(u,p):return u._cstr(p).lower().rstrip('\\')
def file_attributes(u,a):
    global last_error
    p=path(u,a(0))
    if p in dirs:return 0x10
    if p in files:return 0x20
    last_error=2;return 0xffffffff
def create_directory(u,a):
    global last_error
    p=path(u,a(0))
    if p in dirs:last_error=183;return 0
    dirs.add(p);last_error=0;return 1
def remove_directory(u,a):
    global last_error
    p=path(u,a(0))
    if p not in dirs:last_error=3;return 0
    if any(x.startswith(p+'\\') for x in dirs|set(files)):last_error=145;return 0
    dirs.remove(p);return 1
def create_file(u,a):
    global last_error
    p=path(u,a(0));disp=a(4)
    assert not p.startswith('\\\\.'), 'device path needs actual driver: '+p
    if disp in (1,2,4,5):
        if disp==1 and p in files:last_error=80;return 0xffffffff
        if disp in (1,2,5) or p not in files:files[p]=bytearray()
    elif disp==3 and p not in files:last_error=2;return 0xffffffff
    else:assert disp==3, 'unknown creation disposition'
    h=0x73000000+4*len(handles);handles[h]=[p,0];last_error=0;return h
def read_file(u,a):
    assert not a(4), 'overlapped read'
    p,pos=handles[a(0)];data=files[p][pos:pos+a(2)]
    u.uc.mem_write(a(1),bytes(data));u._write32(a(3),len(data));handles[a(0)][1]+=len(data);return 1
def write_file(u,a):
    assert not a(4), 'overlapped write'
    p,pos=handles[a(0)];data=bytes(u.uc.mem_read(a(1),a(2)))
    if pos>len(files[p]):files[p].extend(b'\0'*(pos-len(files[p])))
    files[p][pos:pos+len(data)]=data;handles[a(0)][1]+=len(data);u._write32(a(3),len(data));return 1
def file_pointer(u,a):
    p,pos=handles[a(0)];n=a(1);n=n if n<0x80000000 else n-0x100000000
    high=u._read32(a(2)) if a(2) else 0
    assert high in (0,0xffffffff), '64-bit file pointer'
    pos=({0:0,1:pos,2:len(files[p])}[a(3)])+n

    handles[a(0)][1]=pos
    if a(2):u._write32(a(2),0)

def file_size(u,a):
    if a(1):u._write32(a(1),0)
    return len(files[handles[a(0)][0]])
def global_alloc(u,a):
    assert not a(0)&2, 'GMEM_MOVEABLE unsupported'

def module_filename(u,a):
    assert a(0) in (0,u.base), 'unknown module'
    path=b'C:\\BFME2\\game.dat'
    n=min(len(path),a(2)-1)
    u.uc.mem_write(a(1),path[:n]+b'\0')

def temp_path(u,a):

class Probe(Unpacker):
    def _api(self,uc,addr,size,data):
        dll,name=self.thunks[addr]
        esp=uc.reg_read(UC_X86_REG_ESP)
        argc=10 if name=='CreateProcessA' else APIS.get(name,(4,))[0]
        rec=dict(dll=dll,name=name,caller=hex(self._read32(esp)),args=[hex(self._read32(esp+4+4*i)) for i in range(argc)])
        if name=='CreateProcessA':
            rec['application']=self._cstr(self._read32(esp+4));rec['command_line']=self._cstr(self._read32(esp+8))
        if name in ('GetFileAttributesA','CreateDirectoryA','CreateFileA','LoadLibraryA','GetProcAddress'):
            ptr=self._read32(esp+4+(4 if name=='GetProcAddress' else 0))
            if ptr>0xffff:rec['path_or_name']=self._cstr(ptr)
        if len(calls)<4096: calls.append(rec)
        super()._api(uc,addr,size,data)
