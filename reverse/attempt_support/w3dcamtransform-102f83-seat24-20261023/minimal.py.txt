import search as h
import itertools

def minimal():
    for bits in itertools.product(range(2),repeat=3):
        s=h.BASE
        for i,b in enumerate(bits):
            if b:s=s.replace('Row[%d][1] = (float)( c*tmp1 + s*tmp2);'%i,'Row[%d][1] = (float)( c*tmp1 + (s*tmp2));'%i)
        yield 'minimal_'+''.join(map(str,bits)),s
h.minimal=minimal
h.run('minimal')
