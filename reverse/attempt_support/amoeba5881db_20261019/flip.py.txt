import sys,importlib.util
sys.path.insert(0,'D:/LAN/openbfme2/Open-BFME-2/build/nbr/am/b3')
import h
from pathlib import Path
from concurrent.futures import ThreadPoolExecutor
spec=importlib.util.spec_from_file_location('s',sys.argv[1]); m=importlib.util.module_from_spec(spec); spec.loader.exec_module(m)
def one(item):
    name,t=item
    f=Path(h.D+name+'.cpp'); f.write_text(t)
    obj=f.with_suffix('.obj')
    ok,tx,c=h.build.try_compile_source(f,obj)
    if not ok: return name+' ERR'
    data,rel=h.build.read_object_symbol_bytes(obj,h.SYM)
    ins=[(i.address,i.mnemonic,i.op_str) for i in h.md.disasm(bytes(data),0)]
    for k,(a,mn,o) in enumerate(ins):
        if a>0x40 and mn=='mov' and o.endswith('+ 8]') and o.startswith('edi'):
            return '%s %s %s size=%d'%(name,'FLIP' if 'eax' in o else 'same',o,len(data))
    return name+' ?'
with ThreadPoolExecutor(8) as ex:
    for r in ex.map(one,m.V.items()): print(r,flush=True)
