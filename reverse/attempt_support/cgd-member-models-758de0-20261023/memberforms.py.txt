from pathlib import Path
from itertools import product
W=Path(__file__).resolve().parent
bank=(W/'bank.cpp').read_text();uniform=(W/'uniform-control.cpp').read_text()
out=W/'memberforms';out.mkdir(exist_ok=True)
models=[]
# The raw member pointer is diagnostic: zero this/vbase adjustments retain
# the ordinary ECX receiver and stack arguments; inspect emitted calls too.
for idx,(model,init,order,cc,sites,control) in enumerate(product(range(3),range(4),range(2),range(2),range(8),range(2))):
 s=uniform if control else bank
 kind=['__multiple_inheritance','__virtual_inheritance',''][model]
 size=[8,12,16][model]
 defs=('#pragma pointers_to_members(full_generality, virtual_inheritance)\n' if model==2 else '')
 defs+='struct '+kind+' MultiSource;\n'
 defs+='typedef void *(MultiSource::*PrepareM)(const ContactVector &,const ContactVector &);\n'
 defs+='typedef bool (MultiSource::*TestM)(void *);\n'
 defs+='typedef void (MultiSource::*RespondM)(void *);\n'
 defs+='struct VBase { int dummy; };struct OtherBase { int value; }; struct MultiSource'+(' : virtual VBase' if model else ' : VBase,OtherBase')+' { void *p(const ContactVector &,const ContactVector &); bool t(void *);void r(void *); };\n'
 defs+='typedef char check_mfp[sizeof(PrepareM)=='+str(size)+'?1:-1];\n'
 defs+='template<class T> union PointerView { T method; struct { void *target; int delta['+str(size//4-1)+']; } raw; };\n'
 for name,ret,ty,slot,param,args in [('prep','void *','PrepareM',3,'const ContactVector &p,const ContactVector &n','p,n'),('check','bool','TestM',8,'void *info','info'),('response','void','RespondM',9,'void *info','info')]:
  src='((void **)*(void ***)s)['+str(slot)+']'
  zero=''.join('v.raw.delta['+str(i)+']=0;' for i in range(size//4-1))
  if init==0:body='PointerView<'+ty+'> v;'+(zero+'v.raw.target='+src+';' if order else 'v.raw.target='+src+';'+zero)
  elif init==1:body='PointerView<'+ty+'> v;v.method=0;v.raw.target='+src+';'
  elif init==2:body='PointerView<'+ty+'> v;v.method=&MultiSource::'+{'prep':'p','check':'t','response':'r'}[name]+';v.raw.target='+src+';'
  else:body='PointerView<'+ty+'> v;'+zero+'void *t='+src+';v.raw.target=t;'
  body+=('return ' if ret!='void' else '')+'(((MultiSource *)s)->*v.method)('+args+');'
  defs+='__forceinline '+ret+' '+('__fastcall ' if cc else '')+name+'(ContactSource *s,'+param+'){'+body+'}\n'
 h,t=s.split('void BfmeThingCGD::bfmeTwoCGD() {',1)
 if sites&1:
  t=t.replace('node->second->source->prepare(node->point,node->normal)','prep(node->second->source,node->point,node->normal)')
  t=t.replace('bridge(node->second,node->point,node->normal)','prep(node->second->source,node->point,node->normal)')
  t=t.replace('node->first->source->prepare(node->point,ContactVector(node->normal).negate())','prep(node->first->source,node->point,ContactVector(node->normal).negate())')
  t=t.replace('altPrepare(node->first->source,node->point,ContactVector(node->normal).negate())','prep(node->first->source,node->point,ContactVector(node->normal).negate())')
 if sites&2:t=t.replace('node->first->source->test(','check(node->first->source,')
 if sites&4:
  t=t.replace('node->first->source->respond(','response(node->first->source,').replace('node->second->source->respond(','response(node->second->source,')
 h=h.replace('struct ContactData {',defs+'struct ContactData {')
 (out/('v%04d.cpp'%idx)).write_text(h+'void BfmeThingCGD::bfmeTwoCGD() {'+t)
 models.append([model,init,order,cc,sites,control])
import json
(W/'memberforms-models.json').write_text(json.dumps(models))
print('generated',len(models))
