"""emo: instruction-stream diff of our /Od body of EmotionTrackerUpdate::update against WB's twin (va 0x1230dd0).
Call targets named on our side from relocations and mapped onto WB callee addresses through aligned equal blocks.
usage: odemo.py OBJ [--from 0] [--to 99999] [--name MANGLED --wb VA --wbsize N]"""
import sys, re, difflib, collections, argparse
from pathlib import Path
ROOT = Path(__file__).resolve().parents[2]
sys.path.insert(0, str(ROOT / 'build/agAG/t')); sys.path.insert(0, str(ROOT / 'build/agAH/t')); sys.path.insert(0, str(ROOT / 'tools'))
from odcmp import wb_body, md
from locate import object_functions
from odfront import short

ap = argparse.ArgumentParser(); ap.add_argument('obj')
ap.add_argument('--name', default='?update@EmotionTrackerUpdate@@UAE?AW4UpdateSleepTime@@XZ')
ap.add_argument('--wb', default='0x1230dd0'); ap.add_argument('--wbsize', type=int, default=4655)
ap.add_argument('--stats', action='store_true'); ap.add_argument('--loose', action='store_true')
a = ap.parse_args()


def ours(obj):
    for n, span, relocs in object_functions(Path(obj)):
        if n == a.name: return span.rstrip(b'\xcc'), relocs
    raise SystemExit('not found')


def norm(i, cname=None):
    mn, ops = i.mnemonic, i.op_str
    if mn == 'call':
        if cname: return 'call ' + cname
        return 'call ' + (ops if ops.startswith('dword') else '')
    if mn.startswith('j'): return mn
    ops = re.sub(r'ebp [-+] 0x[0-9a-f]+', 'S', ops)
    ops = re.sub(r'\b0x[0-9a-f]{4,}\b', 'A', ops)
    if a.loose:
        ops = re.sub(r'(e[abcds][xip]) [-+] 0x[0-9a-f]+\]', r'\1+D]', ops)
        ops = re.sub(r'\b(eax|ecx|edx|esi|edi|ebx)\b', 'R', ops)
        ops = re.sub(r'\b(al|cl|dl|bl)\b', 'R8', ops)
        ops = re.sub(r', 0x[0-9a-f]+$', ', I', ops)
    return mn + ' ' + ops


span, relocs = ours(a.obj)
rel = {off: sym for off, typ, sym in relocs}
A = list(md.disasm(span, 0))
wb, base = wb_body(int(a.wb, 16), a.wbsize)
B = list(md.disasm(wb, base))
def ocall(i): return short(rel.get(i.address + 1)) if i.mnemonic == 'call' else None
def wcall(i):
    try: return int(i.op_str, 16)
    except ValueError: return None
wname = {}
for rnd in range(3):
    na = [norm(i, ocall(i)) for i in A]; nb = [norm(i, wname.get(wcall(i)) or (('W%x' % wcall(i)) if wcall(i) else None)) for i in B]
    if rnd == 0:
        na = [norm(i) if i.mnemonic == 'call' else x for i, x in zip(A, na)]; nb = [norm(i) if i.mnemonic == 'call' else x for i, x in zip(B, nb)]
    sm = difflib.SequenceMatcher(a=na, b=nb, autojunk=False)
    ops = sm.get_opcodes()
    cmap = collections.defaultdict(collections.Counter)
    for tag, i1, i2, j1, j2 in ops:
        if tag != 'equal': continue
        for x, y in zip(range(i1, i2), range(j1, j2)):
            if A[x].mnemonic == 'call':
                cmap[wcall(B[y])][ocall(A[x])] += 1
    wname = {k: v.most_common(1)[0][0] for k, v in cmap.items() if k is not None}
eq = 0; tot = 0
for tag, i1, i2, j1, j2 in ops:
    if tag == 'equal':
        eq += i2 - i1; tot += i2 - i1
        continue
    tot += max(i2 - i1, j2 - j1)
    if a.stats: continue
    print('== %s ours[%x..%x] wb[%x..%x]' % (tag, A[i1].address if i1 < len(A) else 0, A[i2 - 1].address if i2 > i1 else 0, B[j1].address if j1 < len(B) else 0, B[j2 - 1].address if j2 > j1 else 0))
    for k in range(max(i2 - i1, j2 - j1)):
        x = A[i1 + k] if i1 + k < i2 else None; y = B[j1 + k] if j1 + k < j2 else None
        os_ = ('%05x %-6s %s%s' % (x.address, x.mnemonic, x.op_str, (' <' + str(ocall(x)) + '>') if x.mnemonic == 'call' else '')) if x else ''
        ws_ = ('%x %-6s %s%s' % (y.address, y.mnemonic, y.op_str, (' <' + str(wname.get(wcall(y))) + '>') if y.mnemonic == 'call' else '')) if y else ''
        print('   O %-60s | W %s' % (os_[:60], ws_[:80]))
print('ours %d wb %d equal %d of %d aligned rows' % (len(A), len(B), eq, tot))
