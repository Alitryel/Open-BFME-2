from pathlib import Path
import sys,json,itertools
sys.path.insert(0,'tools')
from variant_search import Scorer
from zh_sweep import carve
base=Path('build/pdb-context-input.cpp.txt').read_text()
start=base.index('int _RTC_GetSrcLine(')
head,body=base[:start],base[start:]
types={'address':'unsigned long address','moduleName':'char **moduleName',
       'allocation':'char *allocation','errorName':'char *&errorName',
       'line':'int *line','source':'char *source'}
actual={'address':'address','moduleName':'moduleName','allocation':'(char *)_alloca(1024)',
        'errorName':'errname','line':'pline','source':'source'}
s=Scorer('?_RTC_GetSrcLine@@YAHKPADHPAHPAPAD@Z',0x75b177,847,Path('build/pdb-allocation-order').resolve())
records=[]
for order in itertools.permutations(['address','moduleName','allocation']):
 for mode in ['before-assignment','after-assignment-control','both-sides']:
  args=list(order)+['errorName','line','source']
  helper='extern "C" void _ReadWriteBarrier(void);\n#pragma intrinsic(_ReadWriteBarrier)\n'
  helper+='static __forceinline unsigned long resetLookupOutputs('+','.join(types[x] for x in args)+') {\n'
  if mode!='after-assignment-control':helper+='_ReadWriteBarrier();\n'
  helper+='errorName=allocation;\n'
  if mode in ['after-assignment-control','both-sides']:helper+='_ReadWriteBarrier();\n'
  helper+='*line=0;source[0]=0;*moduleName=0;return address;\n}\n'
  changed=body.replace('char *errname=(char *)_alloca(1024);','char *errname;')
  changed=changed.replace('*pline=0;source[0]=0;*moduleName=0;','')
  changed=changed.replace('info=GetImageInfo(address);','info=GetImageInfo(resetLookupOutputs('+','.join(actual[x] for x in args)+'));')
  src=head+helper+changed
  r=s(src);obj=s.dir/('c%d.obj'%s.n)
  extent=next((len(b) for n,b,_ in carve(obj.read_bytes()) if n==s.symbol),None) if obj.exists() else None
  diffs=[i for i,(a,b) in enumerate(zip(r.get('compiled',b''),r.get('target',b''))) if a!=b]
  item=dict(order=list(order),mode=mode,n=s.n,extent=extent,differences=diffs,**{k:v for k,v in r.items() if k not in ('compiled','target')})
  records.append(item);print(item,flush=True)
  (s.dir/'results.json').write_text(json.dumps(records,indent=2)+'\n')
  if r['exact'] and extent==847:
   (s.dir/'exact.cpp').write_text(src);raise SystemExit(0)
