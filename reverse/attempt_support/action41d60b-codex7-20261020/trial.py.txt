import sys,subprocess,concurrent.futures,json,re,difflib,hashlib
from pathlib import Path
sys.path.insert(0,'tools')
import build,capstone
from locate import object_functions
ROOT=build.ROOT
D=ROOT/'build/action41d60b-codex8'
seed=(ROOT/'reverse/attempts/0x0041d60b.cpp').read_text()
NAME='?canDoSpecialPowerAtLocation@ActionManager@@QAE_NPBVObject@@PBUCoord3D@@W4CommandSourceType@@PBVSpecialPowerTemplate@@0I_N@Z'
retail=build.read_target_bytes(0x41d60b,3709)
md=capstone.Cs(capstone.CS_ARCH_X86,capstone.CS_MODE_32)
R=list(md.disasm(retail[:3521],0)); RI={i.address:n for n,i in enumerate(R)}
def score(obj):
 name,blob,rel=next(x for x in object_functions(obj) if x[0]==NAME)
 code=len(blob)-188
 C=list(md.disasm(blob[:code],0));CI={i.address:n for n,i in enumerate(C)}
 holes={j for off,typ,nm in rel for j in range(off,off+4)}
 seq=difflib.SequenceMatcher(a=[i.mnemonic for i in R],b=[i.mnemonic for i in C],autojunk=False)
 diff=[]; structural=0; stack=0
 for tag,a,b,c,d in seq.get_opcodes():
  if tag!='equal':structural+=max(b-a,d-c);continue
  for x,y in zip(R[a:b],C[c:d]):
   rb=bytearray(x.bytes);cb=bytearray(y.bytes)
   for j in range(y.size):
    if y.address+j in holes and j<len(rb): rb[j]=cb[j]=0
   if x.mnemonic.startswith('j') or x.mnemonic=='call':
    if x.op_str.startswith('0x') and y.op_str.startswith('0x'):
     if x.mnemonic=='call':continue
     if RI.get(int(x.op_str,16))==CI.get(int(y.op_str,16)):continue
   if rb!=cb:
    diff.append([hex(x.address),x.mnemonic,x.op_str,y.op_str]);stack+=('ebp' in x.op_str and 'ebp' in y.op_str)
 return dict(size=len(blob),code=code,frame=C[2].op_str,insns=len(C),structural=structural,nd=len(diff)+structural,stack=stack,diff=diff,sha256=hashlib.sha256(blob).hexdigest())
def run(item):
 label,src=item;p=D/(label+'.cpp');o=p.with_suffix('.obj');p.write_text(src)
 try:
  cmd,env=build.compiler_command(p,o)
  res=subprocess.run(cmd,cwd=ROOT,env=env,stdout=subprocess.PIPE,stderr=subprocess.STDOUT)
  p.with_suffix('.log').write_bytes(res.stdout)
  if res.returncode:return label,dict(error=res.stdout.decode(errors='replace')[-1200:])
  s=score(o);p.with_suffix('.json').write_text(json.dumps(s,indent=2));return label,{k:v for k,v in s.items() if k!='diff'}
 except BaseException as e:return label,dict(error=str(e))
def batch(items):
 with concurrent.futures.ThreadPoolExecutor(max_workers=8) as ex:
  out=list(ex.map(run,items))
 (D/('results-'+sys.argv[1]+'.json')).write_text(json.dumps(dict(out),indent=2))
 for l,s in out: print(l,json.dumps(s),flush=True)
if __name__=='__main__':
 if sys.argv[1]=='base':batch([('base',seed)])
