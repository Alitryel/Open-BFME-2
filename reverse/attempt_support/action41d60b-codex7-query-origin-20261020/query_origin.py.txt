import sys,itertools
sys.path.insert(0,'build/action41d60b-codex8'); import trial
sources=[trial.seed,(trial.D/'providers1_15.cpp').read_text()]
patterns=[
('geometry','BfmeWideResult','const Coord3D *loc,float radius','loc,radius','ThePartitionManager->iterateObjectsInRange(loc,radius+500.0f,1,&FILTER(7),0)'),
('placement','BfmeWideResult','const Coord3D *loc,float radius','loc,radius','ThePartitionManager->iterateObjectsInRange(loc,radius+500.0f,0,&FILTER(0x78),0)'),
('healing','BfmeWideResult','const Object *obj,const Coord3D *loc,int radius','obj,loc,radius','ThePartitionManager->iterateObjectsInRange(loc,(float)radius,0,Rva0026119DFilter().link(&FILTER(7))->link(&Rva00260EB1Filter(obj,4,false)),0)'),
('closest99','Object *','const Coord3D *loc','loc','ThePartitionManager->getClosestObject(loc,200.0f,0,&FILTER(0x3c))'),
('closest72','Object *','const Coord3D *loc,int radius','loc,radius','ThePartitionManager->getClosestObject(loc,(float)radius,0,&FILTER(0x3c))')]
items=[]
for base,s in enumerate(sources):
 for mask in range(64):
  t=s;helpers=[]
  for bit,(name,ret,args,use,expr) in enumerate(patterns):
   if mask>>bit&1:
    assert t.count(expr)==1,(name,t.count(expr))
    t=t.replace(expr,'query_'+name+'('+use+')')
    helpers.append('static __forceinline '+ret+' query_'+name+'('+args+') { return '+expr+'; }\n')
  if mask&32:
   expr='target->isAnyKindOf(*(const BitFlags<69> *)&Rva0006EE7A(0,2,7))'
   if base:
    old='Rva0006EE7A mask(0,2,7);\n    if(target->isAnyKindOf(*(const BitFlags<69> *)&mask))'
    assert old in t
    t=t.replace(old,'if(query_kind(target))')
   else:
    assert expr in t;t=t.replace(expr,'query_kind(target)')
   helpers.append('static __forceinline bool query_kind(const Object *target) { return '+expr+'; }\n')
  i=t.index('bool ActionManager::canDoSpecialPowerAtLocation(')
  t=t[:i]+''.join(helpers)+t[i:]
  items.append(('qorigin%d_%02d'%(base,mask),t))
trial.batch(items)
