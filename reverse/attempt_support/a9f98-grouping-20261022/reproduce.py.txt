#!/usr/bin/env python3
"""Isolated diagnostic, NOT a byte/identity/data gate or pin admission.
Run from the repository root: python3 reverse/attempt_support/a9f98-grouping-20261022/reproduce.py.txt
DIR32 slots are masked as in variant.py. The two end-list mappings are hypotheses
supported by WB/native callers, not newly admitted symbols.csv pins.
"""
import argparse, hashlib, json, subprocess, sys
from pathlib import Path
root = Path.cwd()
sys.path.insert(0, str(root / 'tools'))
import build, capstone
from variant import resolve_bytes
p = argparse.ArgumentParser()
p.add_argument('--source', default='reverse/attempts/0x000a9f98.cpp')
p.add_argument('--out', default='build/a9f98-grouping-recheck')
a = p.parse_args()
source = root / a.source
scratch = root / a.out
scratch.mkdir(parents=True, exist_ok=True)
trial = scratch / 'candidate.cpp'
obj = scratch / 'candidate.obj'
trial.write_bytes(source.read_bytes())
command, env = build.compiler_command(trial, obj)
r = subprocess.run(command, cwd=root, env=env, capture_output=True, text=True)
(scratch / 'compiler.log').write_text(r.stdout + r.stderr)
if r.returncode:
    raise SystemExit(r.returncode)
symbol = '?render@AptShapeContainer@@QAEXXZ'
rva, size = 0xA9F98, 1471
retail = build.read_target_bytes(rva, size)
body, rels = build.read_object_symbol_bytes(obj, symbol, code_only=True)
mapping = build.load_symbol_map()
mapping['?rva001100A6@BfmeHub982@@QAIXPAX@Z'] = [0x1100A6]
mapping['?rva001100C5@BfmeHub982@@QAIXPAX@Z'] = [0x1100C5]
resolved, unresolved = resolve_bytes(body, rels, retail, rva, min(size, len(body)), mapping)
def disasm(blob):
    c = capstone.Cs(capstone.CS_ARCH_X86, capstone.CS_MODE_32)
    return [(i.address, i.mnemonic, i.op_str) for i in c.disasm(blob, 0)]
target, ours = disasm(retail), disasm(resolved)
diffs = [(t, c) for t, c in zip(target, ours) if t != c]
receipt = dict(source_sha256=hashlib.sha256(source.read_bytes()).hexdigest(),
               raw_body_sha256=hashlib.sha256(body).hexdigest(),
               diagnostic_body_sha256=hashlib.sha256(resolved).hexdigest(),
               target_sha256=hashlib.sha256(retail).hexdigest(),
               size=len(body), instruction_count_delta=len(ours)-len(target),
               differing_instructions=len(diffs),
               differing_bytes=sum(x != y for x, y in zip(retail, resolved)),
               unresolved=unresolved,
               relocations_masked='DIR32; variant.resolve_bytes',
               diagnostic_end_list_mappings=['0x1100A6', '0x1100C5'])
(scratch / 'receipt.json').write_text(json.dumps(receipt, indent=2) + '\n')
(scratch / 'resolved.bin').write_bytes(resolved)
lines=[]
for t, c in diffs:
    lines.append(f'+{t[0]:04X} / RVA {rva+t[0]:08X}: retail {t[1]} {t[2]} | ours {c[1]} {c[2]}')
(scratch / 'residue.txt').write_text('\n'.join(lines) + '\n')
print(json.dumps(receipt, indent=2))
print('\n'.join(lines))
