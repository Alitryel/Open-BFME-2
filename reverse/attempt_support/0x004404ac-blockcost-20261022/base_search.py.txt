from search import *
import re

def source(n):
 kind=n%8;pos=(n//8)%8;use=n//64
 typ=['unsigned long','long','unsigned int','int','void *','char *','unsigned char *','const unsigned long &'][kind]
 marks=['\n\tGameInfo *game','\n\tif (!game)','\n\tif (slot == -1 || slot >= 8)',
 '\n\tbool observer','\n\tGameWindow *hero','\n\tGameWindow **team','\n\tif (slot == game->v13())','\n\t\tbool found']
 at=BODY.index(marks[pos]);body=BODY[:at];tail=BODY[at:]
 init='this'if kind==4 else '('+(typ.replace('const ','').replace(' &',''))+')this'
 body+='\n '+typ+' base = '+init+';'
 access='((AptMpGameSetup *)base)->'
 if use in [1,2]:tail=re.sub(r'(?<![.>\w])m_[A-Za-z0-9_]+\b',lambda m:access+m.group(0),tail)
 if use==2:tail=re.sub(r'\b(HandleHeroSelection|UpdateHeroDisplay)\(',lambda m:access+m.group(1)+'(',tail)
 if use==3:tail=tail.replace('((AptMapPreview *)&m_60)', '((AptMapPreview *)((char *)base+0x60))')
 if use==0:tail=tail.replace('((AptMapPreview *)&m_60)', '((AptMapPreview *)((unsigned long)base+0x60))')
 return HEAD+MARK+body+tail
if __name__=='__main__':run([('base-%03d'%n,source(n))for n in range(256)],'base')
