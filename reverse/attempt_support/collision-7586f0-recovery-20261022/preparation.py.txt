from search import *
import itertools
cut=BASE.index('void CollisionManagerImpl::rva007586F0()')
pre,body=BASE[:cut],BASE[cut:]
items=[]
for n,(indexed,which,place,form,zeros) in enumerate(itertools.product(range(2),range(1,4),range(5),range(4),range(4))):
 if place==0 and zeros==2:continue
 s=body
 if indexed:s=s.replace('SortList *B = A + 1;','SortList *B = &lists[src+1];')
 stmt=('A = &lists[src];' if which&1 else '')+(' B = &lists[src+1];' if which&2 else '')
 if form==1:stmt=stmt.replace('&lists[src+1]','A+1')
 if form==2:stmt=stmt.replace('&lists[src]','&lists[(src^2)^2]')
 if form==3:stmt=stmt.replace('&lists[src]','lists+src').replace('&lists[src+1]','lists+src+1')
 if place==0:s=s.replace('o->tail = e;', 'o->tail = e; '+stmt)
 if place==1:s=s.replace('o->count++;','o->count++; '+stmt)
 if place==2:s=s.replace('from = B;','from = B; '+stmt).replace('from = A;','from = A; '+stmt)
 if place==3:s=s.replace('out ^= 1;','out ^= 1; '+stmt)
 if place==4:
  # Input preparation belongs to every selection path and its continuation.
  s=s.replace('b--;','b--; '+stmt).replace('a--;','a--; '+stmt)
 if zeros==1:
  s=s.replace('lists[out + 1].count = 0;\n\t\t\tlists[out].count = 0;', 'if (A->count) { lists[out+1].count=0; lists[out].count=0; } else { lists[out+1].count=0; lists[out].count=0; }')
 if zeros==2:
  s=s.replace('o->tail = e;\n\t\t\t\t\t}', 'o->tail = e; o->count++;\n\t\t\t\t\t}')
  s=s.replace('\n\t\t\t\t\to->count++;','')
 if zeros==3:
  s=s.replace('out ^= 1;','if (A->count) out ^= 1; else out ^= 1;')
 items.append(('prep-%04d'%n,pre+s))
batch(items,'preparation')
