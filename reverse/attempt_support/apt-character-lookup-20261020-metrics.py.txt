from pathlib import Path
import sys,difflib,json
from collections import Counter
root=Path(__file__).resolve().parents[1];sys.path.insert(0,str(root/'tools'));import build,align_diff,capstone
name='?Rva006EFBC0ObjectMemberLookup@@YAPAVAptValue@@PAV1@PBVEAStringC@@@Z'
def measure(source):
    obj=build.obj_path(source);raw,rels=build.read_object_symbol_bytes(obj,name,code_only=True)
    offsets={o for o,k,n in rels if k==6}
    starts=[o for o in offsets if o>len(raw)//2 and all(o+4*n in offsets for n in range(4)) and o-4 not in offsets]
    assert starts,source
    first=min(starts)
    row={'name':name,'source':source.relative_to(root).as_posix(),'target_rva':'0x006EFBC0','target_size':str(max(5936,len(raw))),'status':'matched','notes':'','export_rva':''}
    patch=build.compile_function(row,build.load_symbol_map(),obj)
    md=capstone.Cs(capstone.CS_ARCH_X86,capstone.CS_MODE_32)
    a=align_diff.disasm(md,raw[:first]);b=align_diff.disasm(md,build.read_target_bytes(0x006EFBC0,5504))
    sm=difflib.SequenceMatcher(a=[x[1] for x in a],b=[x[1] for x in b],autojunk=False)
    kinds=Counter();regions=[]
    for tag,i1,i2,j1,j2 in sm.get_opcodes():
        if tag!='equal':regions.append({'tag':tag,'retail':b[j1][0] if j1<len(b) else None,'a':i2-i1,'b':j2-j1});continue
        for x,y in zip(a[i1:i2],b[j1:j2]):
            if x[2]!=y[2]:
                if any(x[0]<o+4 and o<x[0]+x[3] for o,k,n in rels):kinds['reloc']+=1
                else:kinds[align_diff.classify(x[2],y[2])]+=1
    result={'source':source.name,'whole_extent':len(raw),'code_and_alignment':first,'target_code_and_alignment':5504,'structural':len(regions),'kinds':dict(kinds),'regions':regions,'unresolved':patch['unresolved'],'tail':[str(x) for x in a[-9:]]}
    return result
if __name__=='__main__':
    out=[]
    for arg in sys.argv[1:]:
        p=(root/arg).resolve();r=measure(p);out.append(r);print(json.dumps(r),flush=True)
    (root/'build/apt-character-contracts-oct20/code-metrics.json').write_text(json.dumps(out,indent=2)+'\n')
