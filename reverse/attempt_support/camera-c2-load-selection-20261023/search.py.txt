import concurrent.futures as cf
import difflib,json,re,subprocess,sys,time,hashlib
from pathlib import Path
ROOT=Path(__file__).resolve().parents[2]
sys.path.insert(0,str(ROOT/'tools'))
import build,capstone
from variant import resolve_bytes
OUT=Path(__file__).resolve().parent
NAME='?adjustCameraMovement@PolygonTrigger@@QAE_NPBUCoord3D@@0PAU2@_N@Z'
RVA,SIZE=0x2e3aa8,723
BASE=(OUT/'base.cpp').read_text()
TARGET=build.read_target_bytes(RVA,SIZE)
SMAP=build.load_symbol_map()

def seq(raw,relocs=()):
 md=capstone.Cs(capstone.CS_ARCH_X86,capstone.CS_MODE_32);md.detail=True
 ins=list(md.disasm(raw,RVA));norm=[]
 for i in ins:
  op=i.op_str
  if i.group(capstone.CS_GRP_JUMP):op='BRANCH'
  elif i.group(capstone.CS_GRP_CALL) and i.operands[0].type==capstone.x86.X86_OP_IMM:op='CALL'
  elif any(i.address-RVA<=off<i.address-RVA+i.size for off,_,_ in relocs):
   op=re.sub(r'(?<![a-z0-9_])(?:0x[0-9a-f]+|\d+)(?![a-z0-9_])','ADDR',op)
  else:op=re.sub(r'0x[0-9a-f]{6,}','ADDR',op)
  norm.append((i.mnemonic,op))
 return ins,norm

TI,TN=seq(TARGET)
def metrics(obj):
 raw,rr=build.read_object_symbol_bytes(obj,NAME,code_only=True)
 ins,norm=seq(raw,rr);sm=difflib.SequenceMatcher(None,norm,TN,autojunk=False)
 nd=sum(max(b-a,d-c)for tag,a,b,c,d in sm.get_opcodes()if tag!='equal')
 fixed,unresolved=(resolve_bytes(raw,rr,TARGET,RVA,SIZE,SMAP)
                   if len(raw)==SIZE else (raw,[]))
 return dict(size=len(raw),nd=nd,fit=sm.ratio(),exact=len(raw)==SIZE and fixed==TARGET and not unresolved,
             unresolved=sorted(set(unresolved)),sha=hashlib.sha256(raw).hexdigest())

def run(item):
 label,text=item
 src,obj=OUT/(label+'.cpp'),OUT/(label+'.obj');src.write_text(text)
 command,env=build.compiler_command(src,obj)
 p=subprocess.run(command,cwd=ROOT,env=env,capture_output=True,text=True)
 if p.returncode:return dict(id=label,error=(p.stdout+p.stderr)[-600:])
 return dict(id=label,**metrics(obj))



import itertools
TAIL='remaining*=0.4f;along.normalize();along.scale(remaining);Coord3D next(candidate);next.add(&along);'
Q=[0,0xc0,0xc3,0xc7,0xe7,0xf7,0xfe,0xff]
ORDERS=[['px','cx','py','pz','sx','cy','sy','sz'],['px','cx','py','pz','cy','sx','sy','sz'],['px','py','pz','cx','cy','sx','sy','sz'],['cx','px','py','pz','sx','cy','sy','sz'],['px','cx','sx','py','cy','sy','pz','sz'],['px','cx','py','sx','cy','py2','pz','sy','sz']]
# Last duplicate py event removed: all families retain one multiplication per component.
ORDERS[-1]=['px','cx','py','sx','cy','pz','sy','sz']
def make(q,order,decl,ops=0):
 def read(e,b):return '*(const volatile float*)&'+e if q&(1<<b) else e
 rx,ry,rz=[read('remaining',b) for b in range(3)]
 ax,ay,az=[read('along.'+c,b+3) for b,c in enumerate('xyz')]
 cx,cy=[read('candidate.'+c,b+6) for b,c in enumerate('xy')]
 mul=lambda a,b,k:a+'*'+b if not ops&(1<<k) else b+'*'+a
 E={'px':'px='+mul(rx,ax,0)+';','py':'py='+mul(ry,ay,1)+';','pz':'pz='+mul(az,rz,2)+';',
 'cx':'cx='+cx+';','cy':'cy='+cy+';','sx':'next.x=cx+px;','sy':'next.y=cy+py;','sz':'next.z=candidate.z+pz;'}
 tail='remaining*=0.4f;along.normalize();Coord3D next;float '+','.join(decl)+';'+''.join(E[e] for e in order)
 return BASE.replace(TAIL,tail)

def family_qual():
 items=[('base-normal',BASE)]
 # Access qualification hypotheses now measured on the constructor-repaired bank.
 # Each volatile lvalue models one existing retail load only; no new memory access.
 for mask in range(512):
  t=BASE
  tail='remaining*=0.4f;along.normalize();'
  tail+='float sx='+ ('*(const volatile float*)&remaining' if mask&1 else 'remaining')+'*'+ ('*(const volatile float*)&along.x' if mask&8 else 'along.x')+';'
  tail+='float sy='+ ('*(const volatile float*)&remaining' if mask&2 else 'remaining')+'*'+ ('*(const volatile float*)&along.y' if mask&16 else 'along.y')+';'
  tail+='float sz='+ ('*(const volatile float*)&remaining' if mask&4 else 'remaining')+'*'+ ('*(const volatile float*)&along.z' if mask&32 else 'along.z')+';'
  tail+='Coord3D next;next.x='+ ('*(const volatile float*)&candidate.x' if mask&64 else 'candidate.x')+'+sx;next.y='+ ('*(const volatile float*)&candidate.y' if mask&128 else 'candidate.y')+'+sy;next.z='+ ('*(const volatile float*)&candidate.z' if mask&256 else 'candidate.z')+'+sz;'
  t=t.replace('remaining*=0.4f;along.normalize();along.scale(remaining);Coord3D next(candidate);next.add(&along);',tail)
  items.append((f'qual-{mask:03x}',t))
 rows=[];start=time.time()
 with cf.ThreadPoolExecutor(max_workers=8) as pool, (OUT/'qual.jsonl').open('w') as log:
  for r in pool.map(run,items):
   rows.append(r);log.write(json.dumps(r)+'\n');log.flush()
   if r.get('exact'):print('EXACT',r,flush=True)
 good=sorted((r for r in rows if 'error'not in r),key=lambda r:(r['nd'],abs(r['size']-SIZE)))
 print('count',len(rows),'errors',len(rows)-len(good),'distinct',len({r['sha'] for r in good}),'seconds',round(time.time()-start,1))
 for r in good[:12]:print(r)

def family_split():
 items=[]
 # Split the z read and multiply. Its read interval can determine allocation
 # while its use position can determine final scheduling independently.
 orders=[['px','cx','py','sx','cy','sy','sz'],['px','cx','py','cy','sx','sy','sz'],['px','py','cx','sx','cy','sy','sz']]
 for q in [0xc3,0xe3,0xf3,0xff]:
  for oi,order in enumerate(orders):
   for za in range(7):
    for zm in range(za+1,9):
     ev=order.copy();ev.insert(za,'az');ev.insert(zm,'mz')
     if ev.index('mz')>ev.index('sz'):continue
     for mode in range(16):
      r=lambda e,b:'*(const volatile float*)&'+e if q>>b&1 else e
      E={'px':'px='+r('remaining',0)+'*along.x;',
         'py':'py='+r('remaining',1)+'*'+r('along.y',4)+';',
         'az':'az='+r('along.z',5)+';',
         'mz':'pz=az*remaining;',
         'cx':'cx='+r('candidate.x',6)+';',
         'cy':'cy='+r('candidate.y',7)+';',
         'sx':'next.x=cx+px;','sy':'next.y=cy+py;','sz':'next.z=candidate.z+pz;'}
      if mode&1:E['az']='az=along.z;'
      if mode&2:E['mz']='pz=remaining*az;'
      z='remaining*=0.4f;along.normalize();Coord3D next;float px,py,pz,cx,cy,az;'
      if mode&4:z=z.replace('px,py,pz,cx,cy,az','az,pz,cx,cy,px,py')
      for i,e in enumerate(ev):
       z+=E[e]
       if mode&8 and e=='px':z+='_ReadWriteBarrier();'
      t=BASE.replace(TAIL,z)
      if mode&8:t=t.replace('#include <math.h>','#include <math.h>\nextern "C" void _ReadWriteBarrier();\n#pragma intrinsic(_ReadWriteBarrier)')
      items.append((f'split-q{q:02x}-o{oi}-a{za}-m{zm}-v{mode:x}',t))
 rows=[];start=time.time()
 with cf.ThreadPoolExecutor(max_workers=8) as pool,(OUT/'split.jsonl').open('w')as log:
