from pathlib import Path
from itertools import product
W=Path(__file__).resolve().parent
base=(W/'bank.cpp').read_text()
D=W/'vectors';D.mkdir(exist_ok=True)

def vector(name,k):
 if k==0:return 'typedef ContactVector '+name+';\n'
 stem='struct '+name
 lead=''
 if k==1:fields='float x,y,z;'
 elif k==2:
  lead='struct '+name+'Base { float x,y,z; };\n';stem+=' : '+name+'Base';fields=''
 elif k==3:fields='union { struct { float x,y,z; }; float values[3]; };'
 elif k==4:fields='union {float x;unsigned ix;}; union {float y;unsigned iy;}; union {float z;unsigned iz;};'
 elif k==5:
  lead='struct '+name+'Tag {};\n';stem+=' : '+name+'Tag';fields='float x,y,z;'
 elif k==6:
  lead='template<class R>struct RoleVector {float x,y,z;};\n' if name=='PVec' else ''
  lead+='struct '+name+'Role {};\n';stem+=' : RoleVector<'+name+'Role>';fields=''
 else:fields='float x,y,z;'
 body=fields+' '+name+'(const '+name+'&v){x=v.x;y=v.y;z=v.z;} '+name+'&negate(){x=-x;y=-y;z=-z;return *this;}'
 if k==7:body+=' ~'+name+'(){}'
 return lead+stem+' {'+body+'};\n'

for i,(pk,nk,abi,order,ret,bridge) in enumerate(product(range(8),range(8),range(2),range(2),range(2),range(2))):
 s=base
 if not bridge:
  s=s.replace('__forceinline void *__cdecl bridge(ContactData *s,const ContactVector &p,const ContactVector &n) { return s->source->prepare(p,n); }\n','')
  s=s.replace('bridge(node->second,node->point,node->normal)','node->second->source->prepare(node->point,node->normal)')
 if not abi:
  a=s.index('struct PArg {');z=s.index('struct ContactData {')
  s=s[:a]+s[z:]
  s=s.replace('altPrepare(node->first->source,node->point,ContactVector(node->normal).negate())','node->first->source->prepare(node->point,ContactVector(node->normal).negate())')
 p=vector('PVec',pk);n=vector('NVec',nk)
 # The template declaration is shared and must precede either specialization.
 if pk==6 or nk==6:
  p=p.replace('template<class R>struct RoleVector {float x,y,z;};\n','')
  n=n.replace('template<class R>struct RoleVector {float x,y,z;};\n','')
  extra='template<class R>struct RoleVector {float x,y,z;};\n'
 else:extra=''
 defs=extra+(n+p if order else p+n)
 s=s.replace('struct ContactSource {',defs+'struct ContactSource {')
 s=s.replace('const ContactVector &point, const ContactVector &normal','const PVec &point, const NVec &normal')
 s=s.replace('const ContactVector &point,const ContactVector &normal','const PVec &point,const NVec &normal')
 s=s.replace('const ContactVector &p,const ContactVector &n','const PVec &p,const NVec &n')
 s=s.replace('ContactVector point,normal;','PVec point; NVec normal;')
 s=s.replace('ContactVector(node->normal)','NVec(node->normal)')
 if abi:
  # Keep the existing diagnostic eight-byte pair ABI while splitting its payload types.
  s=s.replace('struct PArg { const ContactVector * p; PArg(const ContactVector &p):p(&p){} };','struct PArg { const PVec * p; PArg(const PVec &p):p(&p){} };\nstruct NArg { const NVec *p; NArg(const NVec &n):p(&n){} };')
  s=s.replace('PArg point,normal;', 'PArg point; NArg normal;')
 if ret:
  s=s.replace('struct ContactSource {','struct ContactInfo {};\nstruct ContactSource {')
  s=s.replace('void *prepare(', 'ContactInfo *prepare(').replace('void * __fastcall preparePair(', 'ContactInfo * __fastcall preparePair(')
  s=s.replace('void *info','ContactInfo *info').replace('inline void *altPrepare','inline ContactInfo *altPrepare').replace('__forceinline void *__cdecl bridge','__forceinline ContactInfo *__cdecl bridge')
 (D/('v%04d.cpp'%i)).write_text(s)
print('generated',i+1)
