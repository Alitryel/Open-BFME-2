from pathlib import Path
import json
import struct
import sys

ROOT = Path(__file__).resolve().parents[3]
import sys as _s; OUT = Path(__file__).resolve().parent / 'costs' / _s.argv[1]
OUT.mkdir(parents=True, exist_ok=True)
sys.path.insert(0, str(ROOT / 'build/army-causal-20261021'))
sys.path.insert(0, str(ROOT / 'build/camera-causal-20261021/decode'))
import trace_pair as T
import symbol_decode as D

SYM = '?rva00102F83@W3DCamTransform@@QAEXPAUCameraTransformInput@@@Z'
state = {'pass': 0, 'active': False, 'globs': {}, 'captured': False}

def signed(v): return struct.unpack('<i', struct.pack('<I', v))[0]

def enrich_operand(t, o):
    if o.get('kind') in (591, 592):
        symbol = D.decoded_symbol(t, o['p6'])
        o['decoded_symbol'] = symbol
        o['camera_component'] = D.camera_component(symbol)
    for key in ('left', 'right'):
        if key in o: enrich_operand(t, o[key])

def tuple_info(t, ptr):
    result = T.S.tuple_info(t, ptr)
    if result:
        for o in result['src'] + result['dst']: enrich_operand(t, o)
    return result

def costs(t):
    return {str(p): {'component': info['component'],
                    'cost': signed(t.word(p + 64)),
                    'refs': t.word(p + 36)}
            for p, info in state['globs'].items()}

def capture(t, info, ctx, record):
    phase = info['name']
    if phase == '@Spillcosts@12':
        state['pass'] += 1
        record['pass'] = state['pass']
        if state['captured']: return
        all_globs = D.globs(t)
        state['globs'] = {g['address']: {'component': D.camera_component(g['symbol'])[0], 'initial': g}
                          for g in all_globs if D.camera_component(g['symbol'])}
        state['active'] = bool(state['globs'])
        if not state['active']: return
        state['captured'] = True
        record['camera_globs'] = state['globs']
        record['costs_before'] = costs(t)
        def returned(t, ctx):
            record['costs_after'] = costs(t)
            state['active'] = False
        t.on_return(ctx, returned, pop_bytes=12)
        return
    if not state['active']:
        t.records.pop()
        return
    if phase.startswith('CostWrite'):
        if phase == 'CostWriteSource': pointer, new, insn = ctx.Esi, ctx.Ebx, ctx.Ebp
        elif phase == 'CostWriteDefinition': pointer, new, insn = ctx.Ebp, ctx.Ebx, ctx.Esi
        elif phase == 'CostWriteStore': pointer, new, insn = ctx.Esi, ctx.Ecx, t.word(ctx.Esp + 0x40)
        else: pointer, new, insn = ctx.Ebp, ctx.Edx, ctx.Esi
        if pointer not in state['globs']:
            t.records.pop()
            return
        old = signed(t.word(pointer + 64))
        record.update(glob=pointer, component=state['globs'][pointer]['component'],
                      old=old, new=signed(new), delta=signed(new)-old,
                      tuple=tuple_info(t,insn), pass_number=state['pass'])
        return
    record['pass'] = state['pass']
    record['costs_before'] = costs(t)
    if phase == '@CmdUpdateCost@12':
        record['tuple'] = tuple_info(t, t.word(ctx.Esp + 4))
        record['excluded_glob'] = ctx.Edx
    elif phase == '@CmdSubCost@8':
        record['tuple'] = tuple_info(t, ctx.Ecx)
        record['symbol'] = D.decoded_symbol(t, ctx.Eax)
    else:
        record['symbol'] = D.decoded_symbol(t, ctx.Ecx)
    record['ebp_available'] = t.word(t.base + 0x1a7b30)
    def returned(t, ctx):
        record['ret'] = signed(ctx.Eax)
        record['costs_after'] = costs(t)
    t.on_return(ctx, returned, pop_bytes=8 if phase == '@CmdUpdateCost@12' else 0)

cpp = ROOT / 'build/camera-causal-20261021/claude' / _s.argv[2]
command, env = T.TC.build.compiler_command(cpp, OUT / (_s.argv[1]+'.obj'))
T.TC.OUT = T.LAB
t = T.Tracer(_s.argv[1], command, SYM, on_hit=capture, env=env)
T.TC.OUT = OUT
t.maps['LoadCost8002f'] = 0x8002f
t.maps.update(CostWriteSource=0x82692, CostWriteDefinition=0x84c3e, CostWriteReload=0x8281c, CostWriteStore=0x8cb73)
t.names = ['@Spillcosts@12'] + T.COST_HOOKS + ['CostWriteSource', 'CostWriteDefinition', 'CostWriteReload', 'CostWriteStore']
t.run(timeout=180)
print('component globs:', len(state['globs']), flush=True)
