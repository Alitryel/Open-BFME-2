import sys, os, subprocess, json, hashlib, difflib
from pathlib import Path
sys.path.insert(0,str(Path('tools').resolve()))
import build, capstone
from locate import object_functions
BASE=Path(__file__).resolve().parent
NAME='?queueSupplyTruck@AIPlayer@@IAEXXZ'
MD=capstone.Cs(capstone.CS_ARCH_X86,capstone.CS_MODE_32)
TARGET=build.read_target_bytes(0x4f37b0,1537)
def insns(b): return list(MD.disasm(b,0))
def run(src):
 src=Path(src).resolve(); obj=src.with_suffix('.obj'); command,env=build.compiler_command(src,obj)
 env.update({'WINEDEBUG':'-all','DISPLAY':'','WAYLAND_DISPLAY':''})
 result=subprocess.run(command,cwd=build.ROOT,env=env,capture_output=True,text=True)
 src.with_suffix('.compile.log').write_text(result.stdout+result.stderr)
 if result.returncode: return {'name':src.stem,'error':(result.stdout+result.stderr)[-2500:]}
 body,relocs=next((b,r) for n,b,r in object_functions(obj) if n==NAME)
 body=body.rstrip(b'\xcc'); A=insns(body); B=insns(TARGET)
 src.with_suffix('.dis').write_text('\n'.join(f'{i.address:04x} {i.mnemonic} {i.op_str}' for i in A))
 structural=0; mem=0; reg=0
 for tag,i1,i2,j1,j2 in difflib.SequenceMatcher(a=[x.mnemonic for x in A],b=[x.mnemonic for x in B],autojunk=False).get_opcodes():
  if tag!='equal': structural+=1; continue
  for a,b in zip(A[i1:i2],B[j1:j2]):
   if a.op_str==b.op_str:continue
   if 'ptr [' in a.op_str and 'ptr [' in b.op_str:mem+=1
   elif a.op_str.startswith('0x') and b.op_str.startswith('0x'):continue
   else:
    import re
    if any(int(x,16)>=0x400000 for x in re.findall(r'0x[0-9a-f]+',b.op_str)): continue
    reg+=1
 # Build-list loop guard and stores of base destructor vtable are independent discriminators.
 idx=next((n for n,i in enumerate(A) if '0x278]' in i.op_str),None)
 head=' | '.join(i.mnemonic+' '+i.op_str for i in A[idx+1:idx+5]) if idx is not None else '?'
 vf=[(off,sym) for off,typ,sym in relocs if '??_7Rva000421C8' in sym]
 facts={'name':src.stem,'size':len(body),'regions':structural,'mem':mem,'reg':reg,'head':head,'vft_relocs':vf,'sha':hashlib.sha256(body).hexdigest()[:12]}
 src.with_suffix('.json').write_text(json.dumps(facts,indent=2)); return facts
if __name__=='__main__':
 for src in sys.argv[1:]: print(json.dumps(run(src)),flush=True)
