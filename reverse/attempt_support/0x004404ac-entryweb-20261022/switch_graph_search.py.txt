from search import *
import re

CONDS=['slot == -1 || slot >= 8','combo.m_window','observer','m_handicap[slot]','hero','m_rules.m_00 == 1','m_playerTemplate[slot]','*team','slot == game->v13()','enable && game->getConstSlot(game->v13())->m_hasMap']

def ws(text,i):
 while i<len(text) and text[i].isspace():i+=1
 return i

def matching(text,i):
 opening=text[i];closing={'(':')','{':'}','[':']'}[opening];level=1;j=i+1
 while level:
  level+=(text[j]==opening)-(text[j]==closing);j+=1
 return j

def statement(text,i):
 i=ws(text,i)
 if text[i]=='{':return matching(text,i)
 if text.startswith('if',i) and not(text[i+2].isalnum()or text[i+2]=='_'):
  p=ws(text,i+2);q=matching(text,p);end=statement(text,q);other=ws(text,end)
  if text.startswith('else',other):return statement(text,other+4)
  return end
 # All selected unbraced statements are calls, assignments or returns.
 paren=bracket=0;j=i
 while True:
  if text[j]=='(':paren+=1
  elif text[j]==')':paren-=1
  elif text[j]=='[':bracket+=1
  elif text[j]==']':bracket-=1
  elif text[j]==';'and paren==0 and bracket==0:return j+1
  j+=1

def transform(text,mask):
 # Work inside-out by first transforming the final matching header.
 matches=[]
 for m in re.finditer(r'\bif\s*\(',text):
  p=text.index('(',m.start());q=matching(text,p);cond=text[p+1:q-1].strip()
  if cond in CONDS and mask&(1<<CONDS.index(cond)):matches.append(m.start())
 if not matches:return text
 i=matches[-1];p=text.index('(',i);q=matching(text,p);cond=text[p+1:q-1]
 end=statement(text,q);then=text[q:end];other=ws(text,end);final=end;otherwise=''
 if text.startswith('else',other):
  final=statement(text,other+4);otherwise=text[other+4:final]
 repl='switch (!!('+cond+')) {case 1: {'+then+'} break; default: {'+otherwise+'} break;}'
 # Remove this header from the recursive set by spelling its condition once
 # only in the switch. The remaining headers keep their original conditions.
 return transform(text[:i]+repl+text[final:],mask)

def source(mask):return HEAD+MARK+transform(BODY,mask)
if __name__=='__main__':run([('switchgraph-%04d'%n,source(n))for n in range(1024)], 'switch_graph')
