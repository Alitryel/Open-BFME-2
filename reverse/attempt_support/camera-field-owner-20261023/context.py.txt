import itertools,sys,json,concurrent.futures as cf,time
from pathlib import Path
from check import run,OUT
ROOT=OUT.parents[1]
seeds={
 'clean':(OUT/'base.cpp').read_text(),
 'bank':(ROOT/'reverse/attempts/0x002e3aa8.cpp').read_text(),
 'near':(ROOT/'reverse/attempt_support/camera-c2-load-selection-20261023/no-barrier.cpp.txt').read_text()}
def variants():
 for tag,s in seeds.items():
  yield tag+'-base',s
  pred=s[s.index('bool PolygonTrigger::rva002E3A39('):s.index('inline __declspec(noinline) float Coord2D::length')]
  # Move the actual predicate past the parent. This changes emitted context
  # but preserves its visible definition for the compiler's deferred analysis.
  yield tag+'-pred-after',s.replace(pred,'')+'\n'+pred
  # Diagnostic controls force the real preceding predicate to be framed.
  yield tag+'-pred-frame',s.replace(pred,'#pragma optimize("y",off)\n'+pred+'#pragma optimize("y",on)\n')
  # The real immediate retail predecessor is the pair getter.
  decl='class Rva002E3A8DHolder {public:Rva002E3A8DPair*get(Rva002E3A8DPair*,int);};'
  fixed='class Rva002E3A8DHolder {public:char pad[8];Rva002E3A8DPair *array;__declspec(noinline) Rva002E3A8DPair*get(Rva002E3A8DPair*,int);};'
  getter='Rva002E3A8DPair *Rva002E3A8DHolder::get(Rva002E3A8DPair *out,int index){Rva002E3A8DPair *slot=&array[index];out->x=slot->x;out->yBits=slot->yBits;return out;}\n'
  assert decl in s
  z=s.replace(decl,fixed)
  marker='bool PolygonTrigger::adjustCameraMovement('
  for mode in ['before','after','before-frame']:
   g=getter
   if mode.endswith('frame'):g='#pragma optimize("y",off)\n'+getter+'#pragma optimize("y",on)\n'
   if mode=='after':out=z+'\n'+g
   else:out=z.replace(marker,g+marker)
   yield tag+'-get-'+mode,out
  # All four math bodies are genuine emitted providers; test their supported
  # definition order before or after the parent, without artificial functions.
  import re
  defs=re.findall(r'inline __declspec\(noinline\) (?:float|void) Coord[23]D::[^\n]+\n',s)
  assert len(defs)==4
  bare=s
  for d in defs:bare=bare.replace(d,'')
  for k,d in enumerate(defs):
   yield tag+f'-math{k}-before',bare.replace(marker,d+marker)+'\n'+''.join(x for x in defs if x!=d)
   yield tag+f'-math{k}-last',bare+'\n'+''.join(x for x in defs if x!=d)+d

if __name__=='__main__':
 start=time.time();items=list(variants());rows=[]
 with cf.ThreadPoolExecutor(max_workers=8)as p,(OUT/'context.jsonl').open('w')as log:
  for r in p.map(run,items):
   rows.append(r);log.write(json.dumps(r)+'\n');log.flush()
   if r.get('exact'):print('EXACT',r,flush=True)
 for r in rows:print(r)
 print('count',len(rows),'seconds',round(time.time()-start,1))
