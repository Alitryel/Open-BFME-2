from pathlib import Path
import sys,itertools,subprocess,json,hashlib
sys.path.insert(0,'tools')
import build
p=Path('build/trust237-round4').resolve();root=Path.cwd();rows=[]
parents=[Path('reverse/attempt_support/trust237-20261022-round3/pad0046.cpp.txt'),Path('reverse/attempts/0x00237d3d.cpp')]
for i,(parent,cpu,arch) in enumerate(itertools.product(parents,[3,4],['SSE','default'])):
 text=parent.read_text().replace('/G6',f'/G{cpu}').replace('/G7',f'/G{cpu}').replace('/arch:SSE',f'/arch:{arch}' if arch!='default' else '')
 src=p/f'old{i:02}.cpp';obj=src.with_suffix('.obj');src.write_text(text)
 cmd,env=build.compiler_command(src,obj);r=subprocess.run(cmd,env=env,cwd=root,capture_output=True,text=True)
 out=dict(parent=str(parent),cpu=cpu,arch=arch,returncode=r.returncode,output=r.stdout+r.stderr)
 if r.returncode==0:
  name=next(s for s in build.defined_code_symbols(obj) if s.startswith('?alignmentProbe@') or s.startswith('?Rva00237D3DCheck@'))
  raw,_=build.read_object_symbol_bytes(obj,name,code_only=True);out.update(size=len(raw),raw=raw.hex(),sha256=hashlib.sha256(raw).hexdigest())
 rows.append(out)
(p/'legacy_cpu.json').write_text(json.dumps(rows,indent=2))
for r in rows:print({k:v for k,v in r.items() if k!='raw'})
