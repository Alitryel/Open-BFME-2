from search import *
import itertools
s=(W/'base.cpp').read_text();vs=[]
for visible,spec,placement,outline,merge in itertools.product(range(3),range(2),range(2),range(2),range(2)):
 o=s
 q='const'+(' throw()' if spec else '')
 o=o.replace('virtual float ground(float,float,Coord3D*);','virtual float ground(float,float,Coord3D*)'+q+';')
 o=o.replace('virtual float layer(float,float,int,Coord3D*,bool);','virtual float layer(float,float,int,Coord3D*,bool)'+q+';')
 body=''
 attr='inline __declspec(noinline) ' if outline else 'inline '
 if visible in [0,2]:body+=attr+'float LocoTerrainView::ground(float,float,Coord3D*n)'+q+'{if(n){n->x=0.0f;n->y=0.0f;n->z=0.0f;}return 0;}\n'
 if visible in [1,2]:body+=attr+'float LocoTerrainView::layer(float,float,int,Coord3D*n,bool)'+q+'{if(n){n->x=0.0f;n->y=0.0f;n->z=0.0f;}return 0;}\n'
 if placement==0:o=o.replace('bool Locomotor::rva001E7ECA',body+'bool Locomotor::rva001E7ECA')
 else:o+='\n'+body
 if merge:
  start=o.index('class TerrainLogic\n');end=o.index('class LocoTerrainView:',start)
  o=o[:start]+o[end:]
  o=o.replace('class LocoTerrainView:','class TerrainLogic:')
  o=o.replace('class TerrainLogic:public LocoSlots<6>\n{\n public:', 'class TerrainLogic:public LocoSlots<6>\n{\n public:\n PathfindLayerEnum getHighestLayerForDestination(const Coord3D*,bool);')
  at=o.index('class BfmeFixedStorage0004543D')
  o=o[:at]+'extern TerrainLogic*TheTerrainLogic;\n'+o[at:]
  o=o.replace('LocoTerrainView','TerrainLogic')
 vs.append((f'virtualvis_{visible}{spec}{placement}{outline}{merge}',o))
run(vs,'virtualvis48')
