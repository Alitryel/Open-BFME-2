from pathlib import Path
import datetime,hashlib,json,os,subprocess,sys,time
ROOT=Path.cwd();OUT=ROOT/'build/trust237-round15'
label=sys.argv[1];assert label in ('old','new')
def utc():return datetime.datetime.now(datetime.timezone.utc).isoformat()
def git(*args):return subprocess.check_output(['git',*args],cwd=ROOT).decode().strip()
def production_fingerprint():
 raw=subprocess.check_output(['git','ls-tree','-r','HEAD','Code','tools','.githooks','reverse/functions.csv','reverse/symbols.csv','reverse/data.csv'],cwd=ROOT)
 return hashlib.sha256(raw).hexdigest()
receipt=dict(started_utc=utc(),bfme2_revision=git('rev-parse','HEAD'),
 donor=git('-C','reference/open-bfme-1','rev-parse','HEAD'),
 donor_status=git('-C','reference/open-bfme-1','status','--porcelain'),
 production_tree_sha256=production_fingerprint(),command=['./build.sh'],
 environment=dict(BUILD_POOL='8',PYTHONUNBUFFERED='1'))
assert receipt['bfme2_revision']=='0fa5a5b477dbcce1ec612e730ed954120ff59f11'
assert receipt['donor']=={'old':'f890413e75e671748215ba2ef3304aa7953993e9','new':'f36a4db066aa61edeee90085a31d628eb829001d'}[label]
assert receipt['donor_status']==''
log=OUT/('bfme1-'+label+'-gate.log');start=time.monotonic()
with log.open('wb') as output:
 cp=subprocess.run(['./build.sh'],cwd=ROOT,env={**os.environ,**receipt['environment']},stdout=output,stderr=subprocess.STDOUT)
receipt.update(finished_utc=utc(),elapsed_seconds=time.monotonic()-start,exit=cp.returncode,
 log_sha256=hashlib.sha256(log.read_bytes()).hexdigest(),
 production_tree_after_sha256=production_fingerprint())
(OUT/('bfme1-'+label+'-gate.json')).write_text(json.dumps(receipt,indent=2)+'\n')
print(json.dumps(receipt),flush=True)
sys.exit(cp.returncode)
