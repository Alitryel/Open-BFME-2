from pathlib import Path
import json,concurrent.futures
import search
W=Path(__file__).resolve().parent;out=W/'ablate';out.mkdir(exist_ok=True)
b=(W/'base.cpp').read_text();header,body=b.split('void BfmeThingCGD::bfmeTwoCGD() {',1)
# Deliberately non-equivalent probes isolate address sharing. Never bank these.
models={'base':body}
call='node->second->source->respond(altPrepare(node->first->source,node->point,ContactVector(node->normal).negate()));'
assert call in body
for n,expr in [('reverse-normal','node->normal'),('reverse-offset','*(const ContactVector *)((char *)node+24)'),('reverse-cursor','current->point'),('reverse-copy','ContactVector(node->point)')]:
 models[n]=body.replace(call,call.replace('node->point',expr))
models['no-reverse-call']=body.replace(call,'')
models['no-reverse-block']=body[:body.rfind('  if(node->first->source==0')]+'\n }\n}\n'
ps=[]
for n,t in models.items():
 p=out/(n+'.cpp');p.write_text(header+'void BfmeThingCGD::bfmeTwoCGD() {'+t);ps.append(p)
with concurrent.futures.ThreadPoolExecutor(8) as pool:rs=list(pool.map(search.run,ps))
(W/'ablate-results.json').write_text(json.dumps(rs,indent=2))
for r in rs:print(r)
