from pathlib import Path
import sys,subprocess,json,re,hashlib
sys.path.insert(0,'tools');import build
out=Path('build/resume-eh-local');out.mkdir(exist_ok=True)
s=Path('reverse/attempts/0x004b1633.cpp').read_text()
variants={'baseline':s,'color-signed':s.replace('inline unsigned long GameMakeColor','inline int GameMakeColor')}
colors=['GameMakeColor(0,0,255,255)','GameMakeColor(0,255,255,255)','GameMakeColor(255,0,0,255)']
for typ in ['unsigned long','const unsigned long','int','const int']:
 for scope in ['debug','entry']:
  v=s
  for i,c in enumerate(colors):v=v.replace(c,'debugColor'+str(i))
  decl='\n'.join(typ+' debugColor'+str(i)+'='+c+';' for i,c in enumerate(colors))+'\n'
  anchor='        Coord3D pos;' if scope=='debug' else '    EmotionUpdateData *data='
  variants['used-colors-'+typ.replace(' ','-')+'-'+scope]=v.replace(anchor,decl+anchor,1)
variants['named-max-float']=s.replace('float closestDist=3.402823466e+38f;','const float maxDistance=3.402823466e+38f;\n    float closestDist=maxDistance;')
variants['seconds-frame-unit']=s.replace('g_rva00DBA4E4*0x3c','g_rva00DBA4E4*(2*30)')
variants['actual-color-type-and-units']=variants['color-signed'].replace('g_rva00DBA4E4*0x3c','g_rva00DBA4E4*(2*30)')
expr=['shroudRange*shroudRange','data->range20*data->range20','data->range24*data->range24']
for mask in range(1,8):
 v=s
 for i,e in enumerate(expr):
  if mask&(1<<i):v=v.replace(e,'('+e+')')
 variants['fp-square-parens-'+str(mask)]=v
helper='inline float emotionSquare(float value) { return value*value; }\n'
for mask in [1,6,7]:
 v=s.replace('UpdateSleepTime EmotionTrackerUpdate::update()',helper+'UpdateSleepTime EmotionTrackerUpdate::update()')
 for i,e in enumerate(expr):
  if mask&(1<<i):v=v.replace(e,'emotionSquare('+e.split('*')[0]+')')
 variants['float-square-helper-'+str(mask)]=v
assert len(variants)==23
sym='?update@EmotionTrackerUpdate@@UAE?AW4UpdateSleepTime@@XZ';rva=0x4b1633;size=2200
retail=build.read_target_bytes(rva,size);results=[]
for name,src in variants.items():
 p=out/('turn55-emotion-'+name+'.cpp');p.write_bytes(src.encode())
 proc=subprocess.run([sys.executable,'tools/align_diff.py',str(p),sym,hex(rva),str(size)],stdout=subprocess.PIPE,stderr=subprocess.STDOUT,text=True)
 p.with_suffix('.txt').write_bytes(proc.stdout.encode())
 try:
  raw,relocs=build.read_object_symbol_bytes(build.obj_path(p),sym,code_only=True);mask=set()
  for off,kind,target in relocs:mask.update(range(off,off+4))
  diffs=[i for i,(a,b) in enumerate(zip(raw,retail)) if i not in mask and a!=b]
  row={'name':name,'extent':len(raw),'fixed_diff':len(diffs)+abs(len(raw)-size),'first':diffs[:8],'structural':re.search(r'== structural regions: (\d+)',proc.stdout)[1],'sha256':hashlib.sha256(raw).hexdigest()}
 except Exception as e:row={'name':name,'exit':proc.returncode,'error':str(e)}
 print(json.dumps(row),flush=True);results.append(row)
 (out/'turn55-emotion-results.json').write_bytes(json.dumps(results,indent=2).encode())
