from pathlib import Path
import sys,itertools
ROOT=Path.cwd();OUT=ROOT/'build/trust237-round13'
sys.path.insert(0,str(ROOT/'build/trust237-round9'));import frontends as f
f.OUT=OUT
# k0166 is 227 bytes: its 3-byte AND BL,imm8 is the one byte that
# prevents a 16-byte internal arithmetic region and aligned return suffix.
# Test whether preserving the mask's narrow type shares the already-loaded
# AL bound and selects a 2-byte register AND instead.
types=['unsigned char','unsigned short','unsigned int','unsigned long']
values=['2u','3u','0xfffffffeu','0xffffffffu']
forms=['lo&=bound;return lo<=bound;',
 'lo=(T)(lo&bound);return lo<=bound;',
 'return (T)(lo&bound)<=bound;',
 'return bound>=(T)(bound&lo);']
jobs=[]
for typ,val,form,mask,rettype,ref in itertools.product(types,values,forms,[1,127],['int','bool'],[False,True]):
    bound='const '+typ+' &bound=('+typ+')'+str(mask)+'u;' if ref else typ+' bound='+str(mask)+'u;'
    helper='__forceinline '+rettype+' rangePredicate(unsigned long raw){'+typ+' lo=('+typ+')raw;'+bound+form.replace('T',typ)+'}\n'
    tail='if(u=='+val+'){releaseArgs(argv);r=rangePredicate(u);}else{releaseArgs(argv);r=rangePredicate(u);}return r;'
    src=f.base.replace(f.marker,'long status=WinVerifyTrust(0,&action,&data);unsigned long u=(unsigned long)status;int r=0;'+tail)
    src=src.replace('int Rva00237D3DCheck() {',helper+'__forceinline void releaseArgs(unsigned short **p) throw(){LocalFree(p);}\nint Rva00237D3DCheck() {').replace('/O1 /G7','/O1 /G6')
    jobs.append((f't{len(jobs):04}',src,dict(type=typ,value=val,form=form,mask=mask,return_type=rettype,bound_reference=ref)))
if __name__=='__main__':f.execute('typed',jobs)
