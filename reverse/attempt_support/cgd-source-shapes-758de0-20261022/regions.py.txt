from pathlib import Path
import sys,json,difflib,hashlib
sys.path.insert(0,str(Path(__file__).resolve().parent));import search
p=search.DIR;T=search.insns(search.target);ts=[x[:2] for x in T]
regions=[(0,333),(0x8c,0xaa),(0xbe,0xe1),(0x136,0x140)]
best=[(0,None)]*len(regions);seen=set();rows=[]
for report in p.glob('*-results.json'):
 family=report.stem[:-8]
 if family in ['factorial','proxy','refgetters']:continue
 for r in json.loads(report.read_text()):
  if 'error' in r or r['sha'] in seen:continue
  path=p/family/(r['name']+'.bin')
  if not path.exists():continue
  seen.add(r['sha']);A=search.insns(path.read_bytes());ats=[x[:2] for x in A]
  sm=difflib.SequenceMatcher(None,ats,ts,autojunk=False);covered=set()
  for a,b,c in sm.get_matching_blocks():covered.update(range(b,b+c))
  counts=[sum(T[k][2] for k in covered if lo<=T[k][3]<hi) for lo,hi in regions]
  row=dict(path=str(path.relative_to(p)),sha=r['sha'],size=r['size'],relocs=r['relocs'],counts=counts,positional=r['positional']);rows.append(row)
  for i,c in enumerate(counts):
   if c>best[i][0]:best[i]=(c,row)
print('distinct',len(seen))
for i,v in enumerate(best):print(i,v)
print('baseline',[r for r in rows if r['sha']=='fb1009de36137ad2a9d51456fb0b4b6f75176beeca9c0e6292549ec39c845eb3'])
(p/'region-scores.json').write_text(json.dumps({'regions':regions,'best':best,'rows':rows},indent=1))
