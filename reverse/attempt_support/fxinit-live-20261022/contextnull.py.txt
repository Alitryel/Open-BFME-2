import sys,itertools,json,time
from pathlib import Path
from concurrent.futures import ThreadPoolExecutor,as_completed
sys.path.insert(0,'build/fxinit-seat22')
import search as h
h.HERE=Path('build/fxinit-live22').resolve()
Z=['0','m_BonesPerVertex-2','m_BonesPerVertex^2','2-m_BonesPerVertex','m_BonesPerVertex!=2','!links[0]','m_BonesPerVertex&~2','(unsigned)m_BonesPerVertex>>2']
def variants():
 for p,n,stage,comparison in itertools.product(range(len(Z)),range(len(Z)),range(4),range(3)):
  src=h.BASE;decls=[]
  for k,(which,index) in enumerate([('positions',p),('normals',n)]):
   z='(unsigned)('+Z[index]+')';ptr=which+'[1]'
   if stage:
    var='nullValue'+str(k);decls.append('unsigned '+var+'='+z+';');z=var
   e=['(unsigned)'+ptr+'!='+z,'(unsigned)'+ptr+'>'+z,'((unsigned)'+ptr+'^'+z+')!=0'][comparison]
   src=src.replace('if('+ptr+')','if('+e+')')
  ds=''.join(decls)
  if stage==1:src=src.replace(h.R,ds+h.R)
  if stage==2:src=src.replace(h.W,ds+h.W)
  if stage==3:src=src.replace('if((unsigned)positions[1]',ds+'if((unsigned)positions[1]',1) if comparison<2 else src.replace('if(((unsigned)positions[1]',ds+'if(((unsigned)positions[1]',1)
  yield f'contextnull_{p}{n}{stage}{comparison}',src
if __name__=='__main__':
 start=time.time()
 with ThreadPoolExecutor(max_workers=8) as pool:r=list(pool.map(h.run_one,variants()))
 r.sort(key=lambda x:(x.get('nd',99999),x.get('cost',99999)))
 (h.HERE/'contextnull-results.json').write_text(json.dumps(r,indent=2))
 print(json.dumps({'count':len(r),'valid':sum('nd'in x and not x.get('unresolved')for x in r),'seconds':time.time()-start,'best':r[:5]},indent=2))
