from headmerge import *
items=[]
for condition,before,inside,yescopy,nocopy,contract in itertools.product(range(6),range(2),range(2),range(2),range(2),range(2)):
 if not (before or inside or yescopy):continue
 if not (before or nocopy):continue
 objbase='(draw=p)' if inside else 'p'
 cond=[
  'p && (obj='+objbase+'->m_object)!=0',
  'p ? (obj='+objbase+'->m_object)!=0 : false',
  'p && (obj='+objbase+'->m_object)',
  '!(!p || !(obj='+objbase+'->m_object))',
  '(obj=p?'+objbase+'->m_object:0)!=0',
  '(obj=0, p && (obj='+objbase+'->m_object)!=0)',
 ][condition]
 form=('draw=p; 'if before else '')+'if('+cond+') {'+('draw=p; 'if yescopy else '')+body+'\n} else {'+('draw=p; 'if nocopy else '')+terrain+'}\n'
 src=BASE[:lo]+decl+form+BASE[hi:]
 if not contract:src=src.replace('template <> void StringBase<char>::releaseBuffer() throw();\n','')
 items.append(('cc%d%d%d%d%d%d'%(condition,before,inside,yescopy,nocopy,contract),src))
if __name__=='__main__':batch(items,'conditioncopy')
