from pathlib import Path
import sys,subprocess,json
root=Path(__file__).resolve().parents[1];sys.path.insert(0,str(root/'tools'))
out=root/'build/string29530-provider-oct20';out.mkdir(exist_ok=True)
base=(root/'reverse/attempts/0x00029530.cpp').read_text()
sym='??$Rva000179B0UninitializedCopy@PADPAD@_STL@@YAPADPAD00@Z'
native='''inline void *__copy_trivial(const void *__first,const void *__last,void *__result) {
 return (__last == __first) ? __result :
  ((char*)memmove(__result,__first,((const char*)__last-(const char*)__first))) +
  ((const char*)__last-(const char*)__first);
}'''
chain=base[:base.index('template <class _Size>')].replace('void *__copy_trivial(const void *__first, const void *__last, void *__result);',native)+'}\n'
actual='''// stlport
#include <memory>
namespace _STL {
template<class I,class O> inline O Rva000179B0UninitializedCopy(I a,I b,O c) {
 return uninitialized_copy(a,b,c);
}
template char *Rva000179B0UninitializedCopy<char*,char*>(char*,char*,char*);
}
'''
variants={}
for opt in ['/O1','/O2']:
    for kind,text in [('chain',chain),('headers',actual)]:
        if kind=='chain':
            text=text.replace('/Od /Ob1 /DNDEBUG /MD',opt+' /Ob2 /DNDEBUG /MD')
            text+='namespace _STL {template char *Rva000179B0UninitializedCopy<char*,char*>(char*,char*,char*);}\n'
        else:text='// cl: '+opt+' /Ob2 /EHsc /MD /D_STLP_USE_STATIC_LIB /D_STLP_USE_MALLOC\n'+text
        variants[kind+'-'+opt[1:]]=text
import build
from locate import object_functions
rows=[]
for key,s in variants.items():
    src=out/(key+'.cpp');src.write_text(s)
    p=subprocess.run([sys.executable,'tools/explain_mismatch.py',sym,'--rva','0x179B0','--size','41','--source',str(src)],cwd=root,capture_output=True,text=True,encoding='utf8',errors='replace',timeout=300)
    log=p.stdout+p.stderr;src.with_suffix('.mismatch.txt').write_text(log)
    obj=build.obj_path(src)
    fs=list(object_functions(obj)) if obj.exists() else []
    helpers=[(n,len(b)) for n,b,r in fs if n==sym]
    summary=[l for l in log.splitlines() if l.startswith(('classification:','first diff:','OK:','unresolved:'))]
    print(key,helpers,summary,log[-300:] if not fs else '',flush=True)
    rows.append(dict(key=key,helpers=helpers,summary=summary))
(out/'results.json').write_text(json.dumps(rows,indent=2))
