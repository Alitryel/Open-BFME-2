struct BlurCursor {unsigned char *source;unsigned char *dest;};
static __forceinline void blurImpl(BlurCursor cursor,int pitch)
{
unsigned char *source=cursor.source;
unsigned char *dest=cursor.dest;

 unsigned char *src = source;
 unsigned char *dst = dest;
 int row, col;
 for (row = 0; row < 8; row++)
 {
  for (col = 0; col < 8; col++)
  {
   dst[col] = (unsigned char)((16 + 8 * src[col]
   + 4 * (src[col - pitch - 1] + src[col + pitch - 1] + src[col - pitch + 1] + src[col + pitch + 1])
   + 2 * (src[col - 2 * pitch - 2] + src[col + 2 * pitch + 2] + src[col + 2 * pitch - 2] + src[col - 2 * pitch + 2])) >> 5);

  }
  dst += pitch;
  src = src + (pitch);

 }
 for (row = 0; row < 8; row++)
 {
  for (col = 0; col < 8; col++)
  {
   int value = (1 + 6 * src[col] - src[col - 1] - src[col - pitch] - src[col + pitch] - src[col + 1]) >> 1;
   if (value < 0)
   value = 0;
   else if (value > 255)
   value = 255;
   dst[col] = (unsigned char)value;

  }
  dst += pitch;
  src += pitch;

 }


}
extern "C" void DiagonalBlur(void *instance,unsigned char *source,unsigned char *dest,int pitch)
{
BlurCursor cursor={source,dest};blurImpl(cursor,pitch);
}
