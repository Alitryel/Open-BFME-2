import sys,os
sys.path.insert(0,'/work/Open-BFME-2/build/c2tr')
from wdbg import Dbg
# fp2.py SRC OBJ : PFL-style allowed-set forcing at round entry + trace a9634 calls for WN
src,obj=sys.argv[1:3]; WN=os.environ.get('WN','node').split(',')
PF=[(int(a),int(b,16),int(c,16)) for a,b,c in (x.split(':') for x in os.environ.get('PFL','').split(',') if x)]
d=Dbg(src,obj,obj+'.log')
d.prompt();d.prompt('set $CanDeferOnBPByAddr = 1')
B=0x10700000
bps=[0x83acd,0x82978,0xa9634,0xa8f54]; added=set()
for b in bps:
    o=d.prompt('break *0x%x'%(B+b))
    if 'Breakpoint' in o and 'Unable' not in o: added.add(b)
names={}
def nm(sym):
    if sym in names: return names[sym]
    sw=d.words(sym,1); pw=d.words(sw[0],2) if sw and sw[0]>0x100000 else []
    n=d.string(pw[1]).strip() if len(pw)>1 and pw[1]>0x100000 else '?'
    n=n.strip('"').lstrip('_'); names[sym]=n; return n
def rec_of(i):
    p=d.words(0x108807b8+(i&0x3ff)*4,1)[0]
    while p:
        w=d.words(p,13)
        if w[7]==i: return p,w
        p=w[12]
    return None,None
def members(hdr):
    out=[]; ch=d.words(hdr,1)[0] if hdr else 0
    while ch:
        ww=d.words(ch,3)
        if len(ww)<3: break
        base,nx,bits=ww; out+=[base+j for j in range(32) if bits>>j&1]; ch=nx
    return out
on=False;k=0
while True:
    out=d.prompt('cont')
    if 'terminated' in out: break
    eip=d.reg('eip')-B
    for b in bps:
        if b not in added and b!=eip: d.prompt('break *0x%x'%(B+b)); added.add(b)
    added.add(eip)
    if eip==0x83acd:
        ebp=d.reg('ebp'); f=d.words(ebp,1)[0]; curfn=d.string(d.words(f+4,1)[0]); on='updateMeleeAttack' in curfn and d.reg('ebx')==0; k=0; continue
    if not on: continue
    if eip==0x82978:
        k+=1
        p,w=rec_of(0xb)
        print('R%d entry node:b fl=%x w9=%x allowed=%s reg=%x'%(k,w[1],w[9],members(w[8]) if w[8] else None,d.words(w[4]+0x1c,1)[0] if w[4] else 0),flush=True)
        for r,i,v in PF:
            if r==100+k:
                p,w=rec_of(i); ch=d.words(w[8],1)[0]; d.prompt('set *(int*)0x%x = %d'%(ch+8,v)); print('  aset %x -> %x'%(i,v),flush=True)
        if k>14: break
        continue
    if eip==0xa9634:
        esp=d.reg('esp'); ra,a1,a2=d.words(esp,3); w=d.words(a1,10)
        if nm(w[0]) in WN:
            st=d.words(esp,40)
            print('  a9634 R%d %s:%x ra=%x ras=%s'%(k,nm(w[0]),w[7],ra-B,[hex(x-B) for x in st if B<x<B+0x200000][:8]),flush=True)
        continue
    if eip==0xa8f54:
        print('  a8f54 R%d'%k,flush=True); continue
print('done',flush=True)
