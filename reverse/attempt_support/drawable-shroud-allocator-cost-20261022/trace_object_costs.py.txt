import sys,json,re
from pathlib import Path
sys.path.insert(0,str(Path(__file__).resolve().parent))
from wdbg import Dbg
src,obj=sys.argv[1:3];d=Dbg(src,obj,obj+'.debug.log')
d.prompt();d.prompt('set $CanDeferOnBPByAddr = 1')
BASE=0x10700000;focus=False;roundno=0;armed=False;names={}
def name(sym):
 if sym not in names:
  sw=d.words(sym,2)
  if sw and sw[0]>0x100000:
   fw=d.words(sw[0],2)
   names[sym]=d.string(fw[1]) if len(fw)>1 and fw[1]>0x100000 else '?'
  else:names[sym]='anon:'+','.join(hex(w) for w in sw)
 return names[sym]
def members(h):
 p=d.words(h,1)[0] if h else 0;out=[];seen=set()
 while p and p not in seen:
  seen.add(p);a,n,b=d.words(p,3);out.extend(a+i for i in range(32) if b>>i&1);p=n
 return out
def info(p):
 w=d.words(p,18)
 return dict(name=name(w[0]),index=w[7],words=w,allowed=members(w[8]),register=d.words(w[4]+0x1c,1)[0] if w[4] else None)
d.prompt('break *0x%x'%(BASE+0x83acd))
while True:
 out=d.prompt('cont')
 if 'terminated' in out:break
 ip=d.reg('eip')-BASE
 if ip==0x83acd:
  f=d.words(d.reg('ebp'),1)[0];fn=d.string(d.words(f+4,1)[0]);focus='onComingOutOfShroud' in fn and d.reg('ebx')==0;roundno=0
  if focus:
   print(json.dumps({'kind':'enter','name':fn}),flush=True)
   if not armed:
    for off in [0x82978,0x82a4d,0x831f7,0xa8f54,0x82273,0x82692,0x84c3e,0x8281c,0x8cb73]:d.prompt('break *0x%x'%(BASE+off))
    armed=True
  continue
 if not focus:continue
 if ip==0x82978:roundno+=1
 elif ip==0x82a4d:
  print(json.dumps({'kind':'pop','round':roundno,'record':info(d.reg('eax'))}),flush=True)
 elif ip==0x831f7:
  st=d.words(d.reg('esp'),4)
  print(json.dumps({'kind':'colored','round':roundno,'record':info(st[2])}),flush=True)
 elif ip==0xa8f54:
  st=d.words(d.reg('esp'),4)
  print(json.dumps({'kind':'split','round':roundno,'members':members(st[2])}),flush=True)
 elif ip==0x82273:
  print(json.dumps({'kind':'done','round':roundno}),flush=True);focus=False

 elif ip in [0x82692,0x84c3e,0x8281c,0x8cb73]:
  if ip==0x82692:p=d.reg('esi');new=d.reg('ebx');t=d.reg('ebp')
  elif ip==0x84c3e:p=d.reg('ebp');new=d.reg('ebx');t=d.reg('esi')
  elif ip==0x8281c:p=d.reg('ebp');new=d.reg('edx');t=d.reg('esi')
  else:p=d.reg('esi');new=d.reg('ecx');t=d.words(d.reg('esp')+0x40,1)[0]
  record=info(p)
  if '_obj' in record['name']:
   print(json.dumps({'kind':'cost-write','phase':hex(ip),'round':roundno,'record':record,'new':new,'tuple':d.words(t,16)}),flush=True)
