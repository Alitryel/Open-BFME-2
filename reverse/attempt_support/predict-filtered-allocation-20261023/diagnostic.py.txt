"""Causal compiler diagnostics only. Never land these objects or use shared cache.

Private debugger-controlled objects test whether an initial register priority
can explain the remaining Y role and stack-home rotation. Ordinary, unmodified
compiler output remains the sole acceptance authority.
"""
from wdbg import *
import concurrent.futures as cf
from search import metrics,BASE,OUT

def run(item):
 symbol_id,value=item
 tag=f'diagnostic-s{symbol_id}-v{value}'
 src=OUT/(tag+'.cpp');obj=OUT/(tag+'.obj')
 src.write_text(BASE)
 d=Dbg(str(src),str(obj),str(OUT/(tag+'.log')))
 active=False;edits=[]
 try:
  d.prompt();d.prompt('set $CanDeferOnBPByAddr = 1');d.prompt('break *0x10760539')
  for i in range(500):
   o=d.prompt('cont',30)
   if 'terminated' in o or not o.strip():break
   if '0x10760539' in o:
    w=d.words(d.reg('eax'),1);sy=d.words(w[0],2);fn=d.string(sy[1])
    if active:break
    if '?Rva009B5530Prepare@@' in fn:
     active=True;d.prompt('break *0x10782b26')
    continue
   if active and '0x10782b26' in o:
    addr=d.reg('eax');w=d.words(addr,4);sw=d.words(w[0],8)
    site=d.words(d.reg('esp'),1)[0]
    if sw[7]==symbol_id and site==0x1078474f:
     d.prompt('set *(unsigned int *)0x%x = %d'%(addr+12,value))
     edits.append([sw[7],w[3],value])
 finally:
  if d.p.poll() is None:
   try:d.prompt('quit')
   except Exception:d.p.kill()
  d.log.close()
 if not obj.exists():return dict(id=tag,error='no private object',edits=edits)
 return dict(id=tag,edits=edits,**metrics(obj))

if __name__=='__main__':
 items=[(sym,v) for sym in [13,17] for v in [0,50,90,125,130,159,180,256,400]]
 with cf.ThreadPoolExecutor(max_workers=4) as pool:
  for result in pool.map(run,items):print(json.dumps(result),flush=True)
