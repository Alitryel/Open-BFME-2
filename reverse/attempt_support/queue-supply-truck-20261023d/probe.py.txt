# probe.py SRC.cpp -> writes SRC.asm (cl /FAs listing) using the ledger compiler command
import sys,subprocess,os
sys.path.insert(0,'/work2/Open-BFME-2/tools')
import build
from pathlib import Path
src=Path(sys.argv[1]).resolve(); out=src.with_suffix('.obj')
cmd,env=build.compiler_command(src,out)
asm=src.with_suffix('.asm')
cmd=cmd+['/FA','/Fa'+build.wine_path(asm)]
r=subprocess.run(cmd,cwd=build.ROOT,env=env,stdout=subprocess.PIPE,stderr=subprocess.STDOUT,text=True)
if r.returncode: print(r.stdout[-3000:]); sys.exit(1)
