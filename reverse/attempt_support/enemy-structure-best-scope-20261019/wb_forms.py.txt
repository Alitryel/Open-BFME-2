bank = Path('reverse/attempts/0x005735c6.cpp').read_text()
i = bank.index('void AITargetHeuristicEnemyStructure::findBestTarget(')

pre_bool = pre.replace('int isValidTarget(Thing*,void*);', 'bool isValidTarget(Thing*,void*);')
OLDOBJ = 'class Object{public:char pad[0x38];Coord3D pos;char pad44[0x74-0x44];ObjectID id;};'
NEWOBJ = 'class Object{public:const Coord3D *getPosition() const {return &pos;} ObjectID getID() const {return id;} char pad[0x38];Coord3D pos;char pad44[0x74-0x44];ObjectID id;};'
def body(innerform, keyt, dist, bestinit):
    K = keyt
    L = ['void AITargetHeuristicEnemyStructure::findBestTarget(AITarget *target,Player *attacker,Player *defender){',
         ' Record *a=(Record*)g_tacticalAIRegistry->rva002A8F24(attacker);',
         ' IdRange *units=(IdRange*)a->units->get();',
         ' Record *b=(Record*)g_tacticalAIRegistry->rva002A8F24(defender);',
         ' IdRange *structures=(IdRange*)b->structures->get();',
         ' _STL::map<%s,int> counts;' % K,
         ' for(IdRange::iterator it=units->begin();it!=units->end();++it){',
         '  Object *unit=TheGameLogic->findObjectByID(*it);',
         '  if(!unit) continue;']
    if bestinit == 'a':
        L += ['  %s best=(%s)0;' % (K, K), '  float bestDistance=-1.0f;']
    else:
        L += ['  float bestDistance=-1.0f;', '  %s best=(%s)0;' % (K, K)]
    L += ['  for(IdRange::iterator s=structures->begin();s!=structures->end();++s){',
          '   Object *obj=TheGameLogic->findObjectByID(*s);']
    if innerform == 'cont':
        L += ['   if(!obj || !isValidTarget((Thing*)obj,attacker)) continue;']
    elif innerform == 'cont2':
        L += ['   if(!obj) continue;', '   if(!isValidTarget((Thing*)obj,attacker)) continue;']
    else:
        L += ['   if(obj && isValidTarget((Thing*)obj,attacker)){']
    if dist == 'acc':
        L += ['    float dx=obj->getPosition()->x-unit->getPosition()->x;float dy=obj->getPosition()->y-unit->getPosition()->y;']
    else:
        L += ['    float dx=obj->pos.x-unit->pos.x;float dy=obj->pos.y-unit->pos.y;']
    L += ['    float distance=dx*dx+dy*dy;',
          '    if(bestDistance<0.0f || distance<bestDistance){bestDistance=distance;best=%s;}' % ('obj->getID()' if dist=='acc' else 'obj->id')]
    if innerform == 'if': L += ['   }']
    L += ['  }',
          '  if(best){_STL::map<%s,int>::iterator i=counts.find(best);if(i!=counts.end())++i->second;else counts[best]=1;}' % K,
          ' }',
          ' if(!counts.empty()){', '  %s bestId=(%s)0;' % (K, K), '  int maximum=0;',
          '  for(_STL::map<%s,int>::iterator i=counts.begin();i!=counts.end();++i){' % K,
          '   if(i->second>maximum){bestId=i->first;maximum=i->second;}', '  }',
          '  target->setTarget(TheGameLogic->findObjectByID((ObjectID)bestId),g_tacticalAIRegistry->fallbackWeight);', ' }', '}', '']
    return '\n'.join(L)
jobs = []
