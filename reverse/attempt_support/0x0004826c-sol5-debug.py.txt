from trials import *
import os,select,re

tag=sys.argv[1] if len(sys.argv)>1 else 'base'
c=COMMAND.copy(); c[0]='/usr/bin/winedbg'; c[-2]='-Fo'+(DIR/'debug.obj').relative_to(ROOT).as_posix(); c[-1]=(DIR/(tag+'.cpp')).relative_to(ROOT).as_posix()
p=subprocess.Popen(c,cwd=ROOT,env=ENV,stdin=subprocess.PIPE,stdout=subprocess.PIPE,stderr=subprocess.STDOUT)
log=open(DIR/(tag+'-pack.log'),'wb')
def prompt(command=None):
    if command is not None: p.stdin.write((command+'\n').encode()); p.stdin.flush()
    data=b''; deadline=time.time()+30
    while not data.endswith(b'Wine-dbg>'):
        if time.time()>deadline: raise TimeoutError(data[-200:])
        if not select.select([p.stdout],[],[],1)[0]: continue
        chunk=os.read(p.stdout.fileno(),16384)
        if not chunk: return data
        data+=chunk
    log.write(data); log.flush()
    return data.decode(errors='replace')
prompt(); prompt('set $CanDeferOnBPByAddr = 1'); prompt('break *0x10788642')
records=[]
for i in range(5000):
    out=prompt('cont')
    if 'has terminated' in out: break
    name=prompt('x/s *(unsigned int*)(*(unsigned int*)$edx + 4)')
    record=prompt('x/16x $edx')
    regs=prompt('info reg')
    rows=re.findall(r'0x[0-9a-f]+:\s+((?:[0-9a-f]{8}\s*){4})',record)
    vals=[int(v,16) for row in rows for v in row.split()]
    rec={'name':name.strip(),'data':vals,'regs':regs.strip()}
    if name.startswith('_draw\n') or any(name.startswith(v+'\n') for v in ['_fuel','_burnRate','_resistance']):
        rec['instruction']=prompt('x/20x $edi')
        rec['operand']=prompt('x/16x $ebp')
        rec['caller']=prompt('x/4x $esp')
    records.append(rec)
log.close()
p.stdin.write(b'quit\n'); p.stdin.flush()
try: p.wait(timeout=5)
except subprocess.TimeoutExpired: p.terminate()
(DIR/(tag+'-pack.json')).write_text(json.dumps(records,indent=2))
print('records',len(records))
for r in records:
    if any(s in r['name'] for s in ['draw','fuel','resistance','burnRate','name','shader','polyPerFrame','maxBurn','decay']): print(r['name'],r['data'])
