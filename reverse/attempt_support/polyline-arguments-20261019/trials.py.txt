# run.py
from pathlib import Path
import sys,json
sys.path.insert(0,'tools')
from variant_search import Scorer,diff_text
import build,capstone,reloc_ledger as rl
p=Path('build/lane-r-poly').resolve();s=(p/'banked.cpp').read_text()

(p/'corrected.cpp').write_text(fixed)
variants=[('bank',s),('corrected',fixed)]
score=Scorer('?distSqToSegment@@YAMPBUCoord2D@@00@Z',0x4eacf5,197,p/'objects')
cs=capstone.Cs(capstone.CS_ARCH_X86,capstone.CS_MODE_32)
(p/'retail.txt').write_text('\n'.join(f'{i.address:08X} {i.mnemonic} {i.op_str}' for i in cs.disasm(build.read_target_bytes(0x4eacf5,197+227),0x4eacf5)))
r=[]
for name,src in variants:
 v=score(src);a,b=v.pop('compiled',b''),v.pop('target',b'');v.update(name=name,length=len(a));print(v,flush=True)
 (p/(name+'.diff.txt')).write_text(diff_text(a,b,200));r.append(v)
(p/'results.json').write_text(json.dumps(r,indent=2))


# inline.py
from pathlib import Path
import sys,json
sys.path.insert(0,'tools')
from variant_search import Scorer,diff_text
import build,capstone,reloc_ledger as rl
p=Path('build/lane-r-poly').resolve();s=(p/'banked.cpp').read_text()
fixed=s.replace('const Coord2D *b, const Coord2D *a, const Coord2D *p','const Coord2D *p, const Coord2D *a, const Coord2D *b')
(p/'corrected.cpp').write_text(fixed)
variants=[]
start=fixed.index('\n{',fixed.index('static __declspec(noinline) float distSqToSegment'))
end=fixed.index('class Polyline',start)
for passing in ['const Coord2D &','Coord2D']:
 for ordering in [0,1]:
  helpers='__forceinline Coord2D sub2('+passing+' a,'+passing+' b){return Coord2D(a.x-b.x,a.y-b.y);}\n'
  helpers+='__forceinline float dot2('+passing+' a,'+passing+' b){return '+('a.x*b.x+a.y*b.y' if not ordering else 'a.y*b.y+a.x*b.x')+';}\n'
  helpers+='__forceinline float dist2('+passing+' a,'+passing+' b){Coord2D d=sub2(a,b);return dot2(d,d); }\n'
  helpers+='__forceinline Coord2D scaled2('+passing+' a,float t){return Coord2D(a.x*t,a.y*t);}\n'
  helpers+='__forceinline Coord2D add2('+passing+' a,'+passing+' b){return Coord2D(a.x+b.x,a.y+b.y);}\n'
  for tail in ['separate','shared']:
   body='{Coord2D v=sub2(*b,*a);Coord2D w=sub2(*p,*a);float c1=dot2(w,v);'
   if tail=='separate':body+='if(c1<=0.0f)return dist2(*a,*p);Coord2D e=sub2(*b,*a);float c2=dot2(e,e);if(c1>=c2)return dist2(*b,*p);Coord2D pb=add2(*a,scaled2(v,c1/c2));return dist2(pb,*p); }\n'
   else:body+='Coord2D q;if(c1<=0.0f)q=*a;else {Coord2D e=sub2(*b,*a);float c2=dot2(e,e);if(c1>=c2)q=*b;else {Coord2D pb=add2(*a,scaled2(v,c1/c2));return dist2(pb,*p);}}return dist2(q,*p); }\n'
   v=fixed[:start]+body+fixed[end:];v=v.replace('static __declspec(noinline) float distSqToSegment',helpers+'static __declspec(noinline) float distSqToSegment')
   variants.append(('coord_'+str(len(variants)),v))
for form in ['ref','value']:
 for order in [0,1]:
  helpers='__forceinline float square2('+('const float &' if form=='ref' else 'float')+' x){return x*x;}\n'
  v=fixed.replace('return x * x + y * y;','return '+('square2(x)+square2(y)' if not order else 'square2(y)+square2(x)')+';')
  v=v.replace('static __declspec(noinline) float distSqToSegment',helpers+'static __declspec(noinline) float distSqToSegment')
  variants.append(('square_'+str(len(variants)),v))
score=Scorer('?distSqToSegment@@YAMPBUCoord2D@@00@Z',0x4eacf5,197,p/'inline-objects')
cs=capstone.Cs(capstone.CS_ARCH_X86,capstone.CS_MODE_32)
(p/'retail.txt').write_text('\n'.join(f'{i.address:08X} {i.mnemonic} {i.op_str}' for i in cs.disasm(build.read_target_bytes(0x4eacf5,197+227),0x4eacf5)))
r=[]
for name,src in variants:
 v=score(src);a,b=v.pop('compiled',b''),v.pop('target',b'');v.update(name=name,length=len(a));print(v,flush=True)
 (p/(name+'.cpp')).write_text(src);(p/(name+'.diff.txt')).write_text(diff_text(a,b,200));r.append(v)
(p/'inline-results.json').write_text(json.dumps(r,indent=2))


# ctor.py

import sys,json

(p/'corrected.cpp').write_text(fixed)
variants=[]
tcase='float y = t * dy + a->y - p->y;\n\t\t\tfloat x = t * dx + a->x - p->x;\n\t\t\treturn x * x + y * y;'
endcase='float x = cx - p->x;\n\tfloat y = cy - p->y;\n\treturn x * x + y * y;'
for ctor in ['pair','dotargs','refargs']:
 for end in [False,True]:
  for reverse in [False,True]:
   v=fixed
   if ctor=='pair':
    expression='Coord2D q(t*dx+a->x-p->x,t*dy+a->y-p->y);return '+('q.y*q.y+q.x*q.x;' if reverse else 'q.x*q.x+q.y*q.y;')
    final='Coord2D q(cx-p->x,cy-p->y);return '+('q.y*q.y+q.x*q.x;' if reverse else 'q.x*q.x+q.y*q.y;')
   else:
    ty='float' if ctor=='dotargs' else 'const float &'
    helper='__forceinline float sumSquares('+ty+' x,'+ty+' y){return '+('y*y+x*x;' if reverse else 'x*x+y*y;')+'}\n'
    v=v.replace('static __declspec(noinline) float distSqToSegment',helper+'static __declspec(noinline) float distSqToSegment')
    expression='return sumSquares(t*dx+a->x-p->x,t*dy+a->y-p->y);'
    final='return sumSquares(cx-p->x,cy-p->y);'
   v=v.replace(tcase,expression)
   if end:v=v.replace(endcase,final)
   variants.append(('ctor_'+str(len(variants)),v))
score=Scorer('?distSqToSegment@@YAMPBUCoord2D@@00@Z',0x4eacf5,197,p/'ctor-objects')

r=[]
for name,src in variants:
 v=score(src);a,b=v.pop('compiled',b''),v.pop('target',b'');v.update(name=name,length=len(a));print(v,flush=True)
 (p/(name+'.cpp')).write_text(src);(p/(name+'.diff.txt')).write_text(diff_text(a,b,200));r.append(v)

(p/'corrected.cpp').write_text(fixed)
variants=[]
base=(p/'ctor_3.cpp').read_text()
for early in [False,True]:
 for dot in ['scalar','reverse','coord','coord_reverse']:
  for length in ['scalar','reverse','coord','coord_reverse']:
   v=base
   if early:
    v=v.replace('cx = a->x; cy = a->y;','Coord2D q(a->x-p->x,a->y-p->y);return q.y*q.y+q.x*q.x;').replace('cx = b->x; cy = b->y;','Coord2D q(b->x-p->x,b->y-p->y);return q.y*q.y+q.x*q.x;')

   elif dot.startswith('coord'):

    v=v.replace('apx * dx + apy * dy','ap.x*dx+ap.y*dy' if dot=='coord' else 'ap.y*dy+ap.x*dx')
   if length=='reverse':v=v.replace('ex * ex + ey * ey','ey*ey+ex*ex')
   elif length.startswith('coord'):
    v=v.replace('float ex = b->x - a->x;\n\t\tfloat ey = b->y - a->y;','Coord2D e(b->x-a->x,b->y-a->y);')
    v=v.replace('ex * ex + ey * ey','e.x*e.x+e.y*e.y' if length=='coord' else 'e.y*e.y+e.x*e.x')
   variants.append(('tails_'+str(len(variants)),v))

 v=score(src);a,b=v.pop('compiled',b''),v.pop('target',b'');v.update(name=name,length=len(a));print(v,flush=True)

# abi.py
from pathlib import Path
import sys,json
sys.path.insert(0,'tools')
from variant_search import Scorer,diff_text
import build,capstone,reloc_ledger as rl
p=Path('build/lane-r-poly').resolve();s=(p/'banked.cpp').read_text()
fixed=s.replace('const Coord2D *b, const Coord2D *a, const Coord2D *p','const Coord2D *p, const Coord2D *a, const Coord2D *b')
(p/'corrected.cpp').write_text(fixed)
variants=[]
base=(p/'tails_3.cpp').read_text()
import itertools,re
args={'p':'&p','a':'&prev','b':'&cur'}
for perm in itertools.permutations('pab'):
 v=base.replace('const Coord2D *p, const Coord2D *a, const Coord2D *b',', '.join('const Coord2D *'+x for x in perm)).replace('distSqToSegment(&p, &prev, &cur)','distSqToSegment('+','.join(args[x] for x in perm)+')')
 variants.append(('param_'+''.join(perm),v))
for flag in ['/O2','/Ob2','/Op','/G6','/Oy-']:
 variants.append(('flags_'+flag[1:],base.replace('/MD /DNDEBUG','/MD /DNDEBUG '+flag)))
v=base.replace('float cx, cy;','const Coord2D *closest;').replace('cx = a->x; cy = a->y;','closest=a;').replace('cx = b->x; cy = b->y;','closest=b;').replace('Coord2D q(cx-p->x,cy-p->y);','Coord2D q(closest->x-p->x,closest->y-p->y);')
variants.append(('pointer_tail',v))
v=base.replace('float cx, cy;','Coord2D closest;').replace('cx = a->x; cy = a->y;','closest=*a;').replace('cx = b->x; cy = b->y;','closest=*b;').replace('Coord2D q(cx-p->x,cy-p->y);','Coord2D q(closest.x-p->x,closest.y-p->y);')
variants.append(('object_tail',v))
for mask in range(1,8):
 v=base
 for n,letter in enumerate('pab'):
  if mask&(1<<n):
   v=re.sub(r'const Coord2D \*'+letter+r'\b','const Coord2D &'+letter,v).replace(letter+'->',letter+'.')
 v=v.replace('distSqToSegment(&p, &prev, &cur)','distSqToSegment('+','.join(args[c][1:] if mask&(1<<i) else args[c] for i,c in enumerate('pab'))+')')
 variants.append(('references_'+str(mask),v))
score=Scorer('?distSqToSegment@@YAMPBUCoord2D@@00@Z',0x4eacf5,197,p/'abi-objects')
cs=capstone.Cs(capstone.CS_ARCH_X86,capstone.CS_MODE_32)
(p/'retail.txt').write_text('\n'.join(f'{i.address:08X} {i.mnemonic} {i.op_str}' for i in cs.disasm(build.read_target_bytes(0x4eacf5,197+227),0x4eacf5)))
r=[]
for name,src in variants:
 import re
 if name.startswith('references_'):
  obj=(p/'abi-objects'/('ref_'+name+'.obj'));source=obj.with_suffix('.cpp');source.write_text(src)
  build.try_compile_source(source,obj)
  sections,symbols=rl.parse_coff(obj.read_bytes());actual=next(x['name'] for x in symbols.values() if x['name'].startswith('?distSqToSegment@@') and x['section']>0)
  score.symbol=actual
 else:score.symbol='?distSqToSegment@@YAMPBUCoord2D@@00@Z'
 v=score(src);a,b=v.pop('compiled',b''),v.pop('target',b'');v.update(name=name,length=len(a));print(v,flush=True)
 (p/(name+'.cpp')).write_text(src);(p/(name+'.diff.txt')).write_text(diff_text(a,b,200));r.append(v)
(p/'abi-results.json').write_text(json.dumps(r,indent=2))
