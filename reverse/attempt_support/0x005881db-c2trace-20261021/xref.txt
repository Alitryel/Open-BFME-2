import sys,capstone,struct,re
P='/work/Open-BFME-2/reference/open-bfme-1/inputs/toolchains/vs2003/Program Files/Microsoft Visual Studio .NET 2003/Vc7/bin/c2.dll'
data=open(P,'rb').read()
tg=[int(x,16) for x in sys.argv[1:]]
# scan for E8/E9/0F8x/EB/7x rel targets
for i in range(0x1000,min(len(data),0x200000)-6):
    b=data[i]
    if b in (0xe8,0xe9):
        t=i+5+struct.unpack('<i',data[i+1:i+5])[0]
        if t in tg: print('%x %s -> %x'%(i,'call' if b==0xe8 else 'jmp',t))
    elif b==0x0f and 0x80<=data[i+1]<=0x8f:
        t=i+6+struct.unpack('<i',data[i+2:i+6])[0]
        if t in tg: print('%x jcc -> %x'%(i,t))
    elif 0x70<=b<=0x7f or b==0xeb:
        t=i+2+struct.unpack('<b',data[i+1:i+2])[0]
        if t in tg: print('%x jshort? -> %x'%(i,t))
