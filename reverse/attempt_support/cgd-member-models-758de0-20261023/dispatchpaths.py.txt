from pathlib import Path
from itertools import product
import json
W=Path(__file__).resolve().parent
bank=(W/'bank.cpp').read_text();uniform=(W/'uniform-control.cpp').read_text()
D=W/'dispatchpaths';D.mkdir(exist_ok=True)
preds=['node->state&1','node->key0&1','index&1','node->first==node->second']
models=[]
for idx,(scope,pred,sites,inners,outers,bridge,control) in enumerate(product(range(2),range(4),range(8),range(2),range(2),range(2),range(2))):
 s=uniform if control else bank
 # The zero-offset base and overrides share exactly the same three slots.
 a=s.index('struct ContactSource {');b=s.index('};',a)+2
 block=s[a:b]
 base=block.replace('struct ContactSource {','struct BaseSource {')
 overrides='struct ContactSource : BaseSource { virtual void *prepare(const ContactVector &,const ContactVector &)=0;virtual bool test(void *)=0;virtual void respond(void *)=0;};'
 s=s[:a]+base+'\n'+overrides+s[b:]
 normal='node->normal';reverse='ContactVector(node->normal).negate()'
 first='bridge(node->second,node->point,node->normal)' if not control else 'node->second->source->prepare(node->point,node->normal)'
 last='node->first->source->prepare(node->point,'+reverse+')' if control else 'altPrepare(node->first->source,node->point,'+reverse+')'
 old=[
 'node->state=node->first->source->test(node->second->source->prepare(node->point,node->normal))?1:2;',
 'node->first->source->respond('+first+');',
 'node->second->source->respond('+last+');']
 helper=''
 for j in range(3):
  r='node->second->source' if j<2 else 'node->first->source'
  o='node->first->source' if j<2 else 'node->second->source'
  n=normal if j<2 else reverse
  def call(alt):
   rr='((BaseSource *)'+r+')' if alt and inners else r
   oo='((BaseSource *)'+o+')' if alt and outers else o
   ic=rr+'->prepare(node->point,'+n+')'
   if j==1 and bridge and not(alt and inners) and not control:ic='bridge(node->second,node->point,node->normal)'
   if j==2 and not control:ic=last # mixed reverse remains bank's diagnostic ABI.
   return oo+('->test(' if j==0 else '->respond(')+ic+')'
  c0=call(False);c1=call(True)
  if not sites>>j&1:new=('node->state='+c0+'?1:2;' if j==0 else c0+';')
  elif scope==0:
   stmt=lambda c:'node->state='+c+'?1:2;' if j==0 else c+';'
   new='if('+preds[pred]+') {'+stmt(c0)+'} else {'+stmt(c1)+'}'
  else:
   name='pathcall'+str(j)
   helper+='__forceinline '+('bool' if j==0 else 'void')+' '+name+'(ContactNode *node,unsigned tag){if(tag){'+('return ' if j==0 else '')+c0+';}else{'+('return ' if j==0 else '')+c1+';}}\n'
   new=('node->state=' if j==0 else '')+name+'(node,'+preds[pred]+')'+('?1:2;' if j==0 else ';')
  assert old[j] in s,(idx,j)
  if j==0:s=s.replace('if(node->state==0)\n   '+old[j],'if(node->state==0) { '+new+' }')
  else:s=s.replace(old[j],new)
 s=s.replace('class BfmeThingCGD {',helper+'class BfmeThingCGD {')
 (D/('v%04d.cpp'%idx)).write_text(s)
 models.append([scope,pred,sites,inners,outers,bridge,control])
(W/'dispatchpaths-models.json').write_text(json.dumps(models))
print('generated',len(models))
