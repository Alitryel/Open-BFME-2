from pathlib import Path
from itertools import product
W=Path(__file__).resolve().parent
base=(W/'base.cpp').read_text();out=W/'visible';out.mkdir(exist_ok=True)
for i,(p,t,r,ctx) in enumerate(product(range(4),range(4),range(4),range(4))):
 s=base
 protos=['virtual void *prepare(const ContactVector &point, const ContactVector &normal)=0;', 'virtual bool test(void *info)=0;', 'virtual void respond(void *info)=0;']
 names=['prepare','test','respond']
 heads=['void *prepare(const ContactVector &point, const ContactVector &normal)', 'bool test(void *info)','void respond(void *info)']
 bodies=['return (void *)&point;', 'return info!=0;', '(void)info;']
 for j,n in enumerate([p,t,r]):
  if n==0:continue
  if n==1:new='virtual '+heads[j]+';'
  elif n==2:new='virtual __declspec(noinline) '+heads[j]+' { '+bodies[j]+' }'
  else:new='virtual '+heads[j]+' { '+bodies[j]+' }'
  s=s.replace(protos[j],new)
 if ctx&2:s=s.replace('struct ContactSource {','struct ContactSource;\nstruct BaseSource { virtual void slot0()=0; virtual void slot1()=0; virtual void slot2()=0; };\nstruct ContactSource : BaseSource {').replace(' virtual void slot0()=0;\n virtual void slot1()=0;\n virtual void slot2()=0;\n','')
 if ctx&1:s=s.replace('struct ContactSource {','struct __declspec(novtable) ContactSource {').replace('struct ContactSource : BaseSource {','struct __declspec(novtable) ContactSource : BaseSource {')
 (out/('v%04d.cpp'%i)).write_text(s)
print(i+1)
