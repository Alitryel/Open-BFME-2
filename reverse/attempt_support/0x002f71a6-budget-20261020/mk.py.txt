# Rebuilds the register-budget probes for 0x002F71A6 from the bank.
# Usage: python mk.py <bank.cpp> <outdir>; writes dA, dK, dE, dL, dP, dQ (.cpp).
import sys
from pathlib import Path

bank = Path(sys.argv[1]).read_text(encoding='latin-1')
out = Path(sys.argv[2])
NAME = "Path *Pathfinder::internal_findHierarchicalPath("


def put(name, text):
    (out / (name + '.cpp')).write_text(text, encoding='latin-1', newline='')


def cut(text, start, stop):
    i = text.index(start)
    j = text.index(stop, i)
    return text[:i] + text[j:]


def body_only(text, old, new):
    i = text.index(NAME)
    return text[:i] + text[i:].replace(old, new)


# dA: the mineshaft cursor as a plain for-scoped Waypoint* (union keeps only the cell).
a = bank.replace(
    "for(candidate.waypoint=((HierarchyTerrain *)TheTerrainLogic)->getFirstWaypoint();"
    "candidate.waypoint;candidate.waypoint=candidate.waypoint->next) {",
    "for(Waypoint *way=((HierarchyTerrain *)TheTerrainLogic)->getFirstWaypoint();way;way=way->next) {")
i = a.index("for(Waypoint *way=")
j = a.index("if(m_srcMineshaft && m_destMineshaft && m_srcMineshaft!=m_destMineshaft &&")
a = a[:i] + a[i:j].replace("candidate.waypoint", "way") + a[j:]
put('dA', a)

# dK: no union at all; the closest cell is a plain function-scope pointer.
k = a.replace(" union CandidateStorage {Waypoint *waypoint;PathfindCell *cell;};CandidateStorage candidate;\n", "")
k = k.replace("candidate.cell=0;", "PathfindCell *closestCell=0;").replace("candidate.cell", "closestCell")
put('dK', k)

end = k.rindex("}")
# dE: body cut just before the main search loop (cursor enregistered).
e = k.index(" PathfindCell *closestCell=0;")
put('dE', k[:e] + " *adjustedTo=adjustTo;return 0;\n" + k[end:])

# dL: main loop reduced to its spine (no bridge, goal switch, portal or scan blocks).
tail = ("  if(reachedGoal){\n   if(parentCell!=goalCell)hierarchySetParent(goalCell,parentCell);\n"
        "   m_isTunneling=false;\n   Path *path=(Path *)MakeHierarchicalPathPassable(")
i = k.index("  if(!reachedGoal && m_zoneManager.rva00532069(")
j = k.index(tail, k.index("  if(m_zoneManager.rva005315C8("))
l = k[:i] + k[j:]
l = cut(l, "  Int k; unsigned short examinedZones[16];Int examinedCount=0;", " }\n if(closestOK && closestCell")
put('dL', l)

# dP: spine without barriers and without the final vector erase (one variable over budget).
p = body_only(l, "_ReadWriteBarrier();", "")
p = cut(p, " _STL::vector<Gen_p12pod> *hops", " m_isTunneling=false;rva002F40E7();goalCell->ReleaseInfo();return 0;\n}")
put('dP', p)

# dQ: dP without the unused extent aggregate (at budget: retail's register shape returns).
q = p.replace("  Disp32IntPair extent;((Rva005315B0IntPairField *)&m_zoneManager)->get(extent);\n", "")
assert q != p
put('dQ', q)
