import sys, subprocess, re, os
D = os.path.relpath(os.path.dirname(os.path.abspath(__file__))).replace(os.sep, '/') + '/'
# usage (from the repo root): python build/<dir>/shape.py <name>...  compiles build/<dir>/<name>.cpp, prints build-list loop head shape and vftable hoist
for n in sys.argv[1:]:
    out = subprocess.run([sys.executable, D + 'dumpobj.py', f'{D}{n}.cpp', 'queueSupplyTruck'],
                         capture_output=True, text=True)
    t = out.stdout
    open(f'{D}{n}.dis.txt', 'w').write(t + out.stderr)
    L = [l.strip() for l in t.splitlines()]
    if not L:
        print(n, 'FAILED', out.stderr[-800:]); continue
    size = L[0].split()[-1]
    head = ''; hi = 0
    for i, l in enumerate(L):
        if '0x278]' in l:
            head = ' / '.join(x.split(None, 1)[1] for x in L[i+1:i+4]); hi = i; break
    guarded = head.startswith('test')
    # destructor stores: three 'mov dword ptr [ebp - X], R' right after the search; find R for the 0x108 slot
    reg = '?'; src = ''
    pat = re.compile(r'\S+ mov dword ptr \[ebp - 0x\w+\], (e\w\w)$')
    for i in range(hi, len(L) - 4):
        ms = [pat.match(x) for x in L[i:i+5]]
        regs = [m.group(1) for m in ms if m]
        if len(regs) >= 3 and len(set(regs[:3])) == 1:
            reg = regs[0]
            for j in range(i - 1, 0, -1):
                if re.match(rf'\S+ (mov|xor|lea|pop) {reg}, ', L[j]):
                    src = f'{L[j]} ({i-j} before)'; break
            break
    print(f'{n}: {size}B guarded={guarded} vtreg={reg} <- {src} | {head[:90]}')
