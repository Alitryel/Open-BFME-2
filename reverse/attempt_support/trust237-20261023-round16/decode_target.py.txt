from pathlib import Path
import sys,hashlib,json,time
sys.path.insert(0,str(Path(__file__).parent))
import rtpatch
out=Path(__file__).parent
b=(out/'BFME2_1.06_English.exe').read_bytes()
assert hashlib.sha256(b).hexdigest()=='14d9f7159e7f2410a9b501d0f5a71fa8039922aea2e5d3b5071e0fa7bb054867'
r=rtpatch.Reader(b,0x1ec001);entry=rtpatch.read_entry(r,True)
assert r.p==0x1ec035,(hex(r.p),entry)
assert entry['size']==0xbbafa8 and entry['name']=='game.dat',entry
csize=int.from_bytes(b[0x1ebffd:0x1ec001],'little')
print('entry',entry,'compressed',csize,'stream',hex(r.p),flush=True)
t=time.monotonic();decoded=rtpatch.decompress(b,r.p,r.p+csize,entry['size'])
print('decoded',len(decoded),'seconds',time.monotonic()-t,flush=True)
checks=rtpatch.rolling_both(decoded);print('checksums',checks,flush=True)
result=dict(entry=entry,stream=r.p,csize=csize,decoded_size=len(decoded),checksums=checks,
 sha256=hashlib.sha256(decoded).hexdigest(),checks_pass=(len(decoded),*checks)==(entry['size'],entry['w1'],entry['w2']))
(out/'decoded-game.json').write_text(json.dumps(result,indent=2)+'\n')
assert result['checks_pass'],result
(out/'game-patch.dat').write_bytes(decoded)
print('verified',result['sha256'],flush=True)
