from pathlib import Path
import sys,json,itertools
ROOT=Path(__file__).resolve().parents[2]
sys.path.insert(0,str(ROOT/'tools'))
from variant_search import Scorer
out=Path(__file__).parent/'length-carriers'
s=Scorer('?distSqToSegment@@YAMPBVCoord2D@@00@Z',0x4eacf5,197,out)
rows=[]
for basenum,order,kind,formula in itertools.product((18,30),range(2),range(5),range(4)):
 text=(Path(__file__).parent/f'partial-coords/c{basenum}.cpp').read_text()
 if kind==0:
  decl='Coord2D delta;'
  x='delta.x';y='delta.y'
 elif kind==1:
  decl='WallBoundaryPoint delta;'
  x='delta.x';y='delta.y'
 elif kind==2:
  decl='float delta[2];'
  x='delta[0]';y='delta[1]'
 elif kind==3:
  decl='float lengthX,lengthY;'
  x='lengthX';y='lengthY'
 else:
  decl='Coord3D delta;'
  x='delta.x';y='delta.y'
 assigns=[x+' = lineEnd->x - lineStart->x;',y+' = lineEnd->y - lineStart->y;']
 if order:assigns.reverse()
 expression=(f'sqr({y}) + sqr({x})' if formula==0 else f'sqr({x}) + sqr({y})' if formula==1 else f'{y} * {y} + {x} * {x}' if formula==2 else f'({y} * {y}) + ({x} * {x})')
 repl=decl+'\n\t'+'\n\t'.join(assigns)+'\n\tfloat lineLenSqr = '+expression+';'
 assert 'float lineLenSqr = calcDistSquared(*lineEnd, *lineStart);' in text
 text=text.replace('float lineLenSqr = calcDistSquared(*lineEnd, *lineStart);',repl)
 r=s(text);got=r.get('compiled',b'');target=r.get('target',b'')
 row={'forms':[basenum,order,kind,formula],'file':f'c{s.n}.cpp','size':len(got),'diff':sum(a!=b for a,b in zip(got,target))+abs(len(got)-len(target)),'exact':r['exact'],'fitness':r['fitness'],'class':r['class']}
 rows.append(row)
 if s.n==1 or r['exact'] or row['diff']<12:print(json.dumps(row),flush=True)
 if r['exact']:(out/'win.cpp').write_bytes(text.encode());break
(out/'results.json').write_text(json.dumps(rows,indent=2))
print('completed',len(rows),'best',min(rows,key=lambda r:(not r['exact'],-r['fitness'])),flush=True)
