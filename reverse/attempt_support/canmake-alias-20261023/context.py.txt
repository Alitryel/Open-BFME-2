"""Frozen 218-byte diagnostic input, stored as the bank's tracked Git blob."""
from pathlib import Path
import re,subprocess
FROZEN_218='c6d8c81f8f95c7131c3d592d2d62dbdf44594825'
HOME=Path('Code/GameEngine/Source/GameLogic/SkirmishAI/AIWallBuilder').resolve()
def bank_text(helper_first=True):
 text=subprocess.check_output(['git','cat-file','blob',FROZEN_218],text=True)
 text=re.sub(r'#include "([^"]+)"',lambda m:'#include "'+(HOME/m[1]).resolve().as_posix()+'"',text)
 if helper_first:
  helper=next(line for line in text.splitlines()if line.startswith('struct MakeCodeLife '))
  text=helper+'\n'+text.replace(helper+'\n','',1)
 return text
