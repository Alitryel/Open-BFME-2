import sys,itertools,json,time
from pathlib import Path
from concurrent.futures import ThreadPoolExecutor,as_completed
sys.path.insert(0,'build/fxinit-seat22');import search as h
h.HERE=Path('build/fxinit-thread22').resolve();base=Path('build/fxinit-edges22/trials/skipjoin_020000.cpp').read_text()
old='if(!positions[1] && !normals[1]) goto CopyBoth;'
assert old in base
def variants():
 for shape,qual,disp,cmpform,extra in itertools.product(range(4),range(4),range(4),range(4),range(2)):
  p,n='positions[1]','normals[1]'
  # These are existing pointer reads, selectively qualified; no stores added.
  if qual&1:p='*(const Vector3 *volatile *)&positions[1]'
  if qual&2:n='*(const Vector3 *volatile *)&normals[1]'
  tests=[lambda a:'!('+a+')',lambda a:'(unsigned)('+a+')<1',lambda a:'(unsigned)('+a+')<=0',lambda a:'('+a+')==0']
  pp,nn=tests[cmpform](p),tests[cmpform](n)
  # Reverse antecedent order so the immediately preceding test names P,
  # matching the first optional-copy successor observed in C2's early IR.
  forms=[f'if({nn} && {pp}) goto CopyBoth;',f'if({nn}) {{if({pp}) goto CopyBoth;}}',f'if(!({nn})) goto FinishSkin; if({pp}) goto CopyBoth;',f'if({nn}) {{if(!({pp})) goto FinishSkin; goto CopyBoth;}}']
  s=base.replace(old,forms[shape])
  if disp&1:s=s.replace('m_BonesPerVertex>0 && links[0]','*(volatile int *)&m_BonesPerVertex>0 && links[0]')
  if disp&2:s=s.replace('m_BonesPerVertex>0 && links[0]','m_BonesPerVertex>0 && *(const unsigned short *volatile *)&links[0]')
  if extra:s=s.replace('CopyBoth:\n'+h.P,'CopyBoth:\nif(!normals[1]) { '+h.P+' } else { '+h.P+' }')
  yield f'revguard_{shape}{qual}{disp}{cmpform}{extra}',s
items=list(variants());r=[];t=time.time()
with ThreadPoolExecutor(max_workers=8)as pool:
 for f in as_completed([pool.submit(h.run_one,x)for x in items]):r.append(f.result())
r.sort(key=lambda x:(x.get('nd',99999),x.get('cost',99999)));(h.HERE/'revguard-results.json').write_text(json.dumps(r,indent=2))
print(json.dumps({'count':len(r),'resolved':sum('nd'in x and not x.get('unresolved')for x in r),'seconds':time.time()-t,'best':r[:5]},indent=2))
