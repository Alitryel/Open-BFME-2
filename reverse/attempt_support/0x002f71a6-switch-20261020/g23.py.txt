import itertools
from pathlib import Path
b = Path('sw2.cpp').read_text(encoding='latin-1')
NAME = "Path *Pathfinder::internal_findHierarchicalPath("
A1 = "     delta.x-=obj->position[0];delta.y-=obj->position[1];delta.z-=obj->position[2];\n     float dist=delta.x*delta.x+delta.y*delta.y;\n"
A2 = "     delta.x-=rawTo->x;delta.y-=rawTo->y;delta.z-=rawTo->z;\n     dist=delta.x*delta.x+delta.y*delta.y;\n"
for s in (A1, A2): assert b.count(s) == 1
HELP = {
 'f': "static inline void coordSub(Coord3D *c,const Coord3D *p){c->x-=p->x;c->y-=p->y;c->z-=p->z;}\nstatic inline float coordLenSq2D(const Coord3D *c){return c->x*c->x+c->y*c->y;}\n",
 'r': "static inline void coordSub(Coord3D &c,const Coord3D &p){c.x-=p.x;c.y-=p.y;c.z-=p.z;}\nstatic inline float coordLenSq2D(const Coord3D &c){return c.x*c.x+c.y*c.y;}\n",
}
def sub(k, d, p): return f"coordSub(&{d},{p});" if k == 'f' else f"coordSub({d},*{p});"
def ln(k, d): return f"coordLenSq2D(&{d})" if k == 'f' else f"coordLenSq2D({d})"
for k, s1, l1, s2, l2 in itertools.product('fr', (0,1), (0,1), (0,1), (0,1)):
    if not (s1 or l1 or s2 or l2): continue
    u = b.replace(NAME, HELP[k] + NAME, 1)
    p1 = "(const Coord3D *)obj->position"
    r1 = (f"     {sub(k,'delta',p1)}\n" if s1 else "     delta.x-=obj->position[0];delta.y-=obj->position[1];delta.z-=obj->position[2];\n")
    r1 += (f"     float dist={ln(k,'delta')};\n" if l1 else "     float dist=delta.x*delta.x+delta.y*delta.y;\n")
    r2 = (f"     {sub(k,'delta','rawTo')}\n" if s2 else "     delta.x-=rawTo->x;delta.y-=rawTo->y;delta.z-=rawTo->z;\n")
    r2 += (f"     dist={ln(k,'delta')};\n" if l2 else "     dist=delta.x*delta.x+delta.y*delta.y;\n")
    u = u.replace(A1, r1).replace(A2, r2)
    Path(f'v23/{k}{s1}{l1}{s2}{l2}.cpp').write_text(u, encoding='latin-1', newline='')
