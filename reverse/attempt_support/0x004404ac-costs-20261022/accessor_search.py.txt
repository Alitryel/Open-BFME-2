from search import *

def source(n):
    head,body=HEAD,BODY
    fields=[('m_game','Rva0043DA65 *',''), ('m_colorCombo','MpGameSetupComboRef &','[slot]'),
            ('m_handicap','GameWindow *','[slot]'), ('m_hero','GameWindow *','[slot]'),
            ('m_playerTemplate','GameWindow *','[slot]'),('m_team','GameWindow *&','[slot]')]
    methods=[]
    for i,(f,t,suffix) in enumerate(fields):
        k=(n//(3**i))%3
        if not k:continue
        arg='int slot' if suffix else ''
        call='slot' if suffix else ''
        getter='getField%d'%i
        if k==1:methods.append('%s %s(%s) { return %s%s; }'%(t,getter,arg,f,suffix))
        else:
            methods.append('%s %sInner(%s) { return %s%s; }'%(t,getter,arg,f,suffix))
            methods.append('%s %s(%s) { return %sInner(%s); }'%(t,getter,arg,getter,call))
        body=body.replace(f+suffix, getter+'('+call+')')
    head=head.replace('\nprivate:\n\tunsigned char m_pad000', '\n'+ '\n'.join(methods)+'\nprivate:\n\tunsigned char m_pad000')
    return head+MARK+body

if __name__=='__main__':run([('access-%04d'%n,source(n)) for n in range(729)],'accessor')
