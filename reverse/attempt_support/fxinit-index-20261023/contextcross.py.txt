def variants():
 # Four actual bodies, 4 placements each: opaque, before, after, or explicitly
 # noinline before. Includes ordinary external definitions absent in prior
 # all-inline contexts. Interactions are tested factorially.
 for choices in itertools.product(range(4),repeat=4):
  pre='';post=''
  for body,choice in zip([vc,vd,ib.ctors,ib.dtors],choices):
   if choice==1:pre+=body+'\n'
   elif choice==2:post+=body+'\n'
   elif choice==3:pre+='__declspec(noinline) '+body+'\n'
  s=BASE.replace('bool FXShaderGeometry::Init(',pre+'\nbool FXShaderGeometry::Init(')+'\n'+post
  yield 'context_'+''.join(map(str,choices)),s
if __name__=='__main__':
 items=list(variants());t=time.time()
 with ThreadPoolExecutor(max_workers=8)as pool:r=list(pool.map(h.run_one,items))
