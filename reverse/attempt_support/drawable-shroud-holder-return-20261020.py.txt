from pathlib import Path
import sys,subprocess,json
root=Path(__file__).resolve().parents[1];out=root/'build/shroud-stlport-bitset-oct20';base=(out/'checked-kind-budget100.cpp').read_text()
provider=(root/'Code/GameEngine/Source/GameLogic/Object/Update/Rva004A1828.cpp').read_text().replace('// cl: /DNDEBUG /MD','',1)
results=[]
for typ,mangled in [('void *','PAX'),('Rva004A1828Holder *','PAVRva004A1828Holder@@')]:
    for flags in ['Ob1','Ob2-noinline']:
        pcode=provider.replace('virtual int check();','virtual '+typ+' check();').replace('int Rva004A1828Get(',typ+' Rva004A1828Get(').replace('int result = entry->m_item.check();',typ+' result = entry->m_item.check();')
        source=base.replace('int Rva004A1828Get(Rva004A1828Owner *owner);',typ+' Rva004A1828Get(Rva004A1828Owner *owner);')
        if flags=='Ob1':source=source.replace('/Ob2','/Ob1',1)
        else:
            source=source.replace(typ+' Rva004A1828Get(', '__declspec(noinline) '+typ+' Rva004A1828Get(',1)
        key=('voidptr' if typ=='void *' else 'holderptr')+'-'+flags
        p=out/('return-'+key+'.cpp');p.write_bytes((source+'\n'+pcode).encode())
        name='?Rva004A1828Get@@YA'+mangled+'PAURva004A1828Owner@@@Z'
        for label,n,rva,size in [('parent','?onComingOutOfShroud@Drawable@@QAEXXZ','0x0027893E','747'),('provider',name,'0x004A1828','36')]:
            r=subprocess.run([sys.executable,'tools/explain_mismatch.py',n,'--rva',rva,'--size',size,'--source',p.relative_to(root).as_posix()],cwd=root,capture_output=True,text=True,encoding='utf8',errors='replace',timeout=240)
            log=r.stdout+r.stderr;p.with_suffix('.'+label+'.log').write_bytes(log.encode())
            d={'key':key,'label':label,'name':n,'exit':r.returncode,'exact':'OK: bytes match' in log,'summary':[x for x in log.splitlines() if x.startswith(('compiled symbol size:','classification:','unresolved:','first diff:'))]}
            if r.returncode:d['error']=log[-700:]
            results.append(d);(out/'holder-return-results.json').write_text(json.dumps(results,indent=2));print(json.dumps(d),flush=True)
