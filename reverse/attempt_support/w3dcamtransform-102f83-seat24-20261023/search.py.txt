import sys, json, subprocess, difflib, re, itertools, time
from pathlib import Path
from concurrent.futures import ThreadPoolExecutor, as_completed
ROOT=Path(subprocess.check_output(['git','rev-parse','--show-toplevel'],text=True).strip())
sys.path.insert(0,str(ROOT/'tools'))
import build, capstone, permute
import hashlib
INPUT=subprocess.check_output(['git','show','ce76006fa6:reverse/attempts/0x00102f83.cpp'])
assert hashlib.sha256(INPUT).hexdigest()=='f585abbaa3d1d5ef987cdfd827deebefca3c10b4a332114a9145f93f9f7ffb03'
BASE=INPUT.decode().replace('\r\n','\n')
HERE=Path(__file__).parent
NAME='?rva00102F83@W3DCamTransform@@QAEXPAUCameraTransformInput@@@Z'
RVA=0x102f83
SIZE=4299
TARGET=build.read_target_bytes(RVA,SIZE)
MAP=build.load_symbol_map()

def insns(blob,relocs=()):
    md=capstone.Cs(capstone.CS_ARCH_X86,capstone.CS_MODE_32)
    relocation={o for o,t,n in relocs}
    out=[]
    for x in md.disasm(blob,0):
        ops=x.op_str
        if x.mnemonic.startswith('j') or x.mnemonic=='call': ops='TARGET' if not '[' in ops else ops
        if any(x.address<=r<x.address+x.size for r in relocation):
            ops=re.sub(r'0x[0-9a-f]+|\b0\b','ADDR',ops)
        else:
            ops=re.sub(r'0x[0-9a-f]{6,}', 'ADDR',ops)
        ops=ops.replace('fs:[ADDR]','fs:[0]')
        out.append((x.address,x.mnemonic,ops,x.size))
    return out
TINS=insns(TARGET)

def run_one(item):
    label,src=item
    path=HERE/'trials'/f'{label}.cpp';path.parent.mkdir(exist_ok=True)
    path.write_text(src)
    obj=path.with_suffix('.obj')
    command,env=build.compiler_command(path,obj)
    p=subprocess.run(command,cwd=ROOT,env=env,capture_output=True,text=True)
    if p.returncode:
        path.with_suffix('.errors').write_text(p.stdout+p.stderr)
        return {'id':label,'error':'\n'.join(line for line in (p.stdout+p.stderr).splitlines() if 'error ' in line)[:1800]}
    raw,rels=build.read_object_symbol_bytes(obj,NAME,code_only=True)
    row={'name':NAME,'source':str(path.relative_to(ROOT)),'target_rva':hex(RVA),'target_size':str(SIZE),'notes':''}
    try:
        patch=build.compile_function(row,MAP,obj)
        full=patch['bytes']; unr=patch['unresolved']
        nd=sum(a!=b for a,b in zip(full,TARGET))+abs(len(full)-SIZE)
    except BaseException as e: nd=9999; unr=[str(e)]
    ours=insns(raw,rels)
    seq=difflib.SequenceMatcher(a=[x[1] for x in TINS],b=[x[1] for x in ours],autojunk=False)
    regions=0;struct=0;ops=0;delta=[]
    for tag,a,b,c,d in seq.get_opcodes():
        if tag!='equal':
            regions+=1;struct+=max(b-a,d-c)
            delta.append({'region':tag,'retail':[TINS[a:b]],'ours':[ours[c:d]]})
        else:
            for x,y in zip(TINS[a:b],ours[c:d]):
                if x[2]!=y[2]: ops+=1;delta.append({'retail':x,'ours':y})
    region_cost={};
    for h in delta:
        ri=h['retail'];oi=h['ours']
        if isinstance(ri[0],int): addr=ri[0];count=1
        else:
            rr=ri[0];oo=oi[0];addr=(rr or oo)[0][0];count=max(len(rr),len(oo))
        bucket=next((key for lo,hi,key in [(0,0x400,'entry'),(0x400,0x430,'pitch1'),(0x430,0x500,'angle1'),(0x500,0x570,'translate1'),(0x570,0x768,'live_matrix'),(0x768,0x800,'translate2'),(0x800,0x8b8,'pitch'),(0x8b8,0xbe0,'mid'),(0xbe0,0xc98,'shaker'),(0xc98,0xf20,'tail1'),(0xf20,0x1100,'final')] if lo<=addr<hi),'other')
        region_cost[bucket]=region_cost.get(bucket,0)+count
    result={'region_cost':region_cost,'id':label,'size':len(raw),'nd':nd,'regions':regions,'struct':struct,'ops':ops,'cost':struct+ops,'unresolved':unr,'fitness':permute.fitness(full,TARGET)}
    path.with_suffix('.json').write_text(json.dumps({'result':result,'diff':delta},indent=2))
    path.with_suffix('.dis').write_text('\n'.join(f'{a:04x} {m} {o}' for a,m,o,s in ours))
    return result

def parens():
    for bits in itertools.product(range(2),repeat=8):
        s=BASE
        a=s.index('    __forceinline void Rotate_X(float theta)');b=s.index('    __forceinline void Rotate_Y(float theta)',a)
        part=s[a:b]
        for i,term in enumerate(['c*tmp1','s*tmp2','-s*tmp1','c*tmp2']):
            if bits[i]:part=part.replace(term,'('+term+')')
        if bits[4]:part=part.replace('(float)(', '(float)((').replace(';\n',';\n')
        # Group each sum through an explicit outer FP paren, without changing arithmetic.
        if bits[4]:
            part=re.sub(r'(Row\[\d\]\[\d\] = \(float\)\(\(.*)\);',r'\1));',part)
        if bits[5]:part=part.replace('-s*tmp1','-(s*tmp1)')
        if bits[6]:part=part.replace('tmp1 = Row[0][1];','const float &sine=s;const float &cosine=c;\n\ttmp1 = Row[0][1];').replace('c*tmp','cosine*tmp').replace('s*tmp','sine*tmp')
        if bits[7]:part=part.replace('tmp1 = Row[0][1];','const Matrix3D &matrix=*this;\n\ttmp1 = Row[0][1];');part=re.sub(r'(tmp[12] = )Row',r'\1matrix.Row',part) if bits[7] else part
        yield 'parens_'+''.join(map(str,bits)),s[:a]+part+s[b:]

def run(mode):
    from concurrent.futures import ProcessPoolExecutor
    items=list(globals()[mode]());rs=[];t=time.time()
    with ProcessPoolExecutor(max_workers=8) as ex:
        for f in as_completed([ex.submit(run_one,x) for x in items]):
            rs.append(f.result())
            if len(rs)%32==0:print('tested',len(rs),'best',max(rs,key=lambda z:z.get('fitness',-1)),flush=True)
    rs.sort(key=lambda z:-z.get('fitness',-1))
    (HERE/(mode+'-results.json')).write_text(json.dumps(rs,indent=2))
    print(json.dumps({'count':len(rs),'seconds':time.time()-t,'best':rs[:5]},indent=2),flush=True)
if __name__=='__main__':run(sys.argv[1])
