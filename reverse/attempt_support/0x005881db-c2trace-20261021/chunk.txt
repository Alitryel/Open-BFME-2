import sys
sys.path.insert(0,'/work/Open-BFME-2/build/c2tr')
from wdbg import Dbg
# chunk.py SRC OBJ ROUND : print a8f54 set header + chunk addresses at round ROUND
src,obj,R=sys.argv[1],sys.argv[2],int(sys.argv[3])
d=Dbg(src,obj,obj+'.log')
d.prompt();d.prompt('set $CanDeferOnBPByAddr = 1')
B=0x10700000
bps=[0x83acd,0x82978,0xa8f54]; added=set()
for b in bps:
    o=d.prompt('break *0x%x'%(B+b))
    if 'Breakpoint' in o and 'Unable' not in o: added.add(b)
on=False;k=0
while True:
    out=d.prompt('cont')
    if 'terminated' in out: break
    eip=d.reg('eip')-B
    for b in bps:
        if b not in added and b!=eip: d.prompt('break *0x%x'%(B+b)); added.add(b)
    added.add(eip)
    if eip==0x83acd:
        ebp=d.reg('ebp'); f=d.words(ebp,1)[0]; curfn=d.string(d.words(f+4,1)[0]); on='updateMeleeAttack' in curfn and d.reg('ebx')==0; k=0; continue
    if not on: continue
    if eip==0x82978: k+=1; continue
    if eip==0xa8f54 and k==R:
        esp=d.reg('esp'); st=d.words(esp,4); hdr=st[2]
        ch=d.words(hdr,1)[0]; print('R',R,'ra=%x set=%x'%(st[0]-B,hdr),flush=True)
        while ch:
            base,nx,bits=d.words(ch,3); print(' chunk %x base %x bits %08x'%(ch,base,bits),flush=True); ch=nx
        break
d.p.kill()
