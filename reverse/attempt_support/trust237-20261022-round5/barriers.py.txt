import itertools
import search

kinds = [
    ('noop', '', '__noop(u);'),
    ('noop16', '', '__noop(u,u,u,u,u,u,u,u,u,u,u,u,u,u,u,u);'),
    ('assume', '', '__assume((u&1u)<=1u);'),
    ('read', 'extern "C" void _ReadBarrier();\n#pragma intrinsic(_ReadBarrier)\n', '_ReadBarrier();'),
    ('write', 'extern "C" void _WriteBarrier();\n#pragma intrinsic(_WriteBarrier)\n', '_WriteBarrier();'),
    ('readwrite', 'extern "C" void _ReadWriteBarrier();\n#pragma intrinsic(_ReadWriteBarrier)\n', '_ReadWriteBarrier();'),
    ('readwrite_noop', 'extern "C" void _ReadWriteBarrier();\n#pragma intrinsic(_ReadWriteBarrier)\n', '_ReadWriteBarrier();__noop(u);'),
    ('noop_assume', '', '__noop(u);__assume((u&1u)<=1u);'),
]
shapes = ['return 1;', 'int r=0;if((u&1u)<=1u)goto yes;return r;yes:return r+1;', 'return (unsigned __int64)(unsigned char)u<=255ui64;']
jobs = []
for (kind,decl,stmt),count,pos,shape,cpu in itertools.product(kinds,[1,16],range(3),range(3),[6,7]):
    trust='long status=WinVerifyTrust(0,&action,&data);unsigned long u=(unsigned long)status;'
    free='LocalFree(argv);'
    ret=shapes[shape]
    if pos==0:trust+=stmt*count
    elif pos==1:free+=stmt*count
    else:
        decl+='__forceinline void barrierHelper(unsigned long u){'+stmt*count+'}\n'
        free+='barrierHelper(u);'
    text=search.base.replace(search.marker,trust+free+ret)
    text=text.replace('int Rva00237D3DCheck() {',decl+'int Rva00237D3DCheck() {').replace('/G7',f'/G{cpu}')
    jobs.append((f'n{len(jobs):04}',text,dict(kind=kind,count=count,position=pos,shape=shape,cpu=cpu)))
search.execute('barriers',jobs)
