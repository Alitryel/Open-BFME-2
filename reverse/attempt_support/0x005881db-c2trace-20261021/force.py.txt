import sys,json
sys.path.insert(0,'/work2/Open-BFME-2/build/c2tr')
from wdbg import Dbg
# usage: force.py SRC OBJ target class round name idx newcost   (modifies cost at entry of 0x82978 round N)
src,obj,target=sys.argv[1:4]; cls=int(sys.argv[4]); rnd=int(sys.argv[5]); edits=sys.argv[6:]
d=Dbg(src,obj,obj+'.log')
d.prompt();d.prompt('set $CanDeferOnBPByAddr = 1')
bps=[0x83acd,0x82978]; added=set()
for b in bps:
    o=d.prompt('break *0x%x'%(0x10700000+b))
    if 'Breakpoint' in o and 'Unable' not in o: added.add(b)
curfn='';curcls=-1;k=0;done=False
while True:
    out=d.prompt('cont')
    if 'terminated' in out: break
    eip=d.reg('eip')-0x10700000
    for b in bps:
        if b not in added and b!=eip: d.prompt('break *0x%x'%(0x10700000+b)); added.add(b)
    added.add(eip)
    if done: continue
    if eip==0x83acd:
        ebp=d.reg('ebp'); f=d.words(ebp,1)[0]; curfn=d.string(d.words(f+4,1)[0]); curcls=d.reg('ebx'); k=0; continue
    if target not in curfn or curcls!=cls: continue
    k+=1
    if k!=rnd: continue
    buckets=d.words(0x108807b8,1024)
    for b in buckets:
        p=b
        while p:
            w=d.words(p,18)
            for e in edits:
                idx,field,val=e.split(':')
                if w[7]==int(idx,16):
                    a=p+4*int(field); d.prompt('set *(int*)0x%x = %d'%(a,int(val)))
                    print('set idx',idx,'w%s'%field,'from',hex(w[int(field)]),'to',val,'readback',hex(d.words(a,1)[0]),flush=True)
            p=w[12]
    done=True
    d.prompt('delete break 1'); d.prompt('delete break 2'); d.prompt('delete break 3')
print('finished',flush=True)
