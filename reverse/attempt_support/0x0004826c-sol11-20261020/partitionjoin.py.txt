from headmerge import *
# Arithmetic split OR became a full-width TEST late enough to preserve draw's
# third packing reference. Cross that late fold with redundant final guards;
# prior factjoin used only early pointer/ordering predicates.
masks=[0xffff,0xff,0x7fffffff,0x80000000,0x55555555,0x33333333,0x01010101,1]
items=[]
for f,mask,syntax,nul,fin,order,contract in itertools.product([8,9],range(8),range(3),range(2),range(3),range(2),range(2)):
 a='((unsigned)p & 0x%xu)'%masks[mask]; b='((unsigned)p & 0x%xu)'%(0xffffffff^masks[mask])
 cond=[a+' || '+b,'('+a+' != 0) || ('+b+' != 0)','!('+a+' == 0 && '+b+' == 0)'][syntax]
 form=forms[f].replace('obj=0','obj='+('(Object*)p' if nul else '0'))
 form=form.replace('p?p->m_object:0','p?p->m_object:'+('(Object*)p' if nul else '0'))
 form=form.replace('if(p)','if('+cond+')').replace('obj=p?','obj=('+cond+')?')
 if order:form=form.replace('draw=p; obj=p->m_object;','obj=p->m_object; draw=p;')
 form=form.replace('if(obj)','if('+['obj','p && obj','obj && p'][fin]+')')
 src=BASE[:lo]+decl+form+'\n'+BASE[hi:]
 if not contract:src=src.replace('template <> void StringBase<char>::releaseBuffer() throw();\n','')
 items.append(('pj%02d%d%d%d%d%d%d'%(f,mask,syntax,nul,fin,order,contract),src))
if __name__=='__main__':batch(items,'partitionjoin')
