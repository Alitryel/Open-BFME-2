from pathlib import Path
import runpy,itertools,concurrent.futures,json,re
OUT=Path('build/canmake-joins-20261023/scalar-receiver').resolve();OUT.mkdir(parents=True,exist_ok=True)
r=runpy.run_path('reverse/attempt_support/canmake-00596e15-20261022/replay.py.txt');r['run'].__globals__['OUT']=OUT
text=runpy.run_path('reverse/attempt_support/canmake-alias-20261023/context.py.txt')['bank_text']()
prefix,body=text.rsplit('int AIBuildableWallSegment::canMake(Player *player)',1);prefix=prefix[prefix.index('\n')+1:]
tests=[]
for bits,typ,ctor,dtor,fall in itertools.product(range(1,8),range(4),range(3),range(2),range(2)):
 ty=['int','unsigned int','AIBuildableWallSegment*','void*'][typ]
 helper='class AIBuildableWallSegment; struct MakeCodeLife { '+ty+' code; '
 if ctor==0:helper+='__forceinline MakeCodeLife(AIBuildableWallSegment*p):code(('+ty+')p){} '
 elif ctor==1:helper+='__forceinline MakeCodeLife(AIBuildableWallSegment*p):code(0){code=('+ty+')p;} '
 else:helper+='__forceinline MakeCodeLife():code(0){} '
 helper+='__forceinline ~MakeCodeLife(){'+('code=0;'if dtor else '')+'} };\n'
 b=body.replace('MakeCodeLife selected;','MakeCodeLife selected(this);'if ctor<2 else 'MakeCodeLife selected;selected.code=('+ty+')this;')
 for i,member in enumerate(['m_startNodeID','getWorldPosition()','m_totalCost']):
  if bits&(1<<i):b=re.sub(r'(?<![\w.>])'+re.escape(member),lambda m:'((AIBuildableWallSegment*)selected.code)->'+m[0],b)
 if fall:b=b.replace('return AIBuildableStructure::canMake(player);','return ((AIBuildableWallSegment*)selected.code)->AIBuildableStructure::canMake(player);')
 if typ>=2:b=re.sub(r'selected.code=([029]);',r'selected.code=('+ty+r')\1;',b)
 b=b.replace('return selected.code;','return (int)selected.code;')
 tests.append((f'scalar-{bits}{typ}{ctor}{dtor}{fall}',helper+prefix+'\nint AIBuildableWallSegment::canMake(Player *player)'+b))
import os
case=os.environ.get('CANMAKE_CASE')
if case:tests=[x for x in tests if x[0]==case];assert tests,case
results=[]
with concurrent.futures.ThreadPoolExecutor(max_workers=8)as pool:
 for i,d in enumerate(pool.map(r['run'],tests),1):
  results.append(d)
  if i%84==0 or d.get('exact_candidate'):print(json.dumps(dict(done=i,total=len(tests),best=sorted(results,key=lambda x:(x.get('nd',999),-x.get('score',0)))[:1])),flush=True)
(OUT/'results.json').write_text(json.dumps(results,indent=2)+'\n')
print(json.dumps(dict(trials=len(results),errors=sum('error'in x for x in results),unresolved=sum(bool(x.get('unresolved'))for x in results),exact=sum(x.get('exact_candidate',False)for x in results),bodies=len({x.get('masked_code_sha256')for x in results}))),flush=True)
