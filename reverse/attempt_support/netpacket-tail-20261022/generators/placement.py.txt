import sys,itertools,re
sys.path.insert(0,'build/netpacket-20261022');from search import *
# Exact non-debug allocation mechanism from the donor GameMemory.h: inline
# class-specific new plus magic-enum placement form, both forward to global new.
classes=['NetCommandList','NetCommandRef','NetAckStage1CommandMsg','NetAckStage2CommandMsg','NetAckBothCommandMsg'];items=[]
for modes in itertools.product(range(3),repeat=5):
 b=BANK
 for cl,mode in zip(classes,modes):
  if not mode:continue
  ops='\n inline void *operator new(unsigned int s) { return ::operator new(s); }\n inline void operator delete(void *p) { ::operator delete(p); }\n'
  if mode==2:
   ops+=' enum '+cl+'MagicEnum { '+cl+'_GLUE_NOT_IMPLEMENTED = 0 };\n inline void *operator new(unsigned int s, '+cl+'MagicEnum) { return ::operator new(s); }\n inline void operator delete(void *p, '+cl+'MagicEnum) { ::operator delete(p); }\n'
   b=b.replace('new '+cl+'(', 'new('+cl+'::'+cl+'_GLUE_NOT_IMPLEMENTED) '+cl+'(')
  b,n=re.subn(r'(class '+cl+r'\s*\{\s*public:)',lambda m:m[0]+ops,b);assert n==1
 items.append(('placement_'+''.join(map(str,modes)),b))
batch(items,'placement-new')
