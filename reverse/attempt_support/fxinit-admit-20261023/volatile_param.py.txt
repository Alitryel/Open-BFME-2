import sys,json,itertools,time
from pathlib import Path
from concurrent.futures import ThreadPoolExecutor,as_completed
sys.path.insert(0,'build/fxinit-seat22');import search as h
h.HERE=Path('build/fxinit-admit23').resolve()
def variants():
 for kind,scope,which,form in itertools.product(range(6),range(4),range(3),range(3)):
  if kind==0:typ='const Vector3 *volatile';expr='p';arg='{n}[1]'
  if kind==1:typ='const Vector3 *const volatile &';expr='p';arg='{n}[1]'
  if kind==2:typ='volatile unsigned';expr='(const Vector3 *)p';arg='(unsigned){n}[1]'
  if kind==3:typ='const volatile unsigned';expr='(const Vector3 *)p';arg='(unsigned){n}[1]'
  if kind==4:typ='const Vector3 *';expr='p';arg='{n}[1]'
  if kind==5:typ='const Vector3 *const &';expr='p';arg='{n}[1]'
  check=[expr,expr+'!=0','!('+expr+'==0)'][form]
  # A volatile by-value parameter has a real local home. No cast removes
  # volatility from its copy read, and the conditional copy stays in the arm.
  if scope==0:helper='static __forceinline bool StreamPresent('+typ+' p){return '+check+';}\n'
  elif scope==1:helper='static __forceinline void CopyPresent('+typ+' p,unsigned i,Vector3 *dest){if('+check+') *dest=('+expr+')[i];}\n'
  else:
   helper='struct StreamInput{'+typ+' p; __forceinline StreamInput('+typ+' q):p(q){} '
   if scope==2:helper+='__forceinline operator bool() const {return '+check+';}};\n'
   else:helper+='__forceinline void copy(unsigned i,Vector3 *dest) const{if('+check+') *dest=('+expr+')[i];}};\n'
  s=h.BASE
  for k,(n,offset,old)in enumerate([('positions','Location',h.P),('normals','Normal',h.N)]):
   if which!=2 and which!=k:continue
   a=arg.format(n=n);dest='(Vector3 *)(vertices+desc.Get_'+offset+'_Offset(1))'
   if scope==0:new=old.replace('if('+n+'[1])','if(StreamPresent('+a+'))')
   elif scope==1:new='CopyPresent('+a+',i,'+dest+');'
   elif scope==2:new=old.replace('if('+n+'[1])','if(StreamInput('+a+'))')
   else:new='StreamInput('+a+').copy(i,'+dest+');'
   s=s.replace(old,new)
  s=s.replace('bool FXShaderGeometry::Init(',helper+'bool FXShaderGeometry::Init(')
  yield f'volatileparam_{kind}{scope}{which}{form}',s
if __name__=='__main__':
 t=time.time();rs=[]
 with ThreadPoolExecutor(max_workers=8)as ex:
  for f in as_completed([ex.submit(h.run_one,x)for x in variants()]):rs.append(f.result())
 rs.sort(key=lambda z:(z.get('nd',99999),z.get('cost',99999)))
 (h.HERE/'volatileparam-results.json').write_text(json.dumps(rs,indent=2))
 print(json.dumps({'count':len(rs),'seconds':time.time()-t,'errors':sum('error'in z for z in rs),'best':rs[:6]},indent=2))
