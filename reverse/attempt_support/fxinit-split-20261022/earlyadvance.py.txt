import sys,itertools,json,time,re
from pathlib import Path
from concurrent.futures import ThreadPoolExecutor
sys.path.insert(0,'build/fxinit-seat22');import search as h
h.HERE=Path('build/fxinit-split22').resolve()
LOOP='   for(unsigned i=0;i<(unsigned)mesh->Get_Vertex_Count();++i) {'
END='    vertices+=desc.Get_FVF_Size();'
POINTS=['    *(Vector3 *)(vertices+desc.Get_Location_Offset(0))','    for(int t=0;t<textureCount;++t)', '    if(basis)', '    if(m_Passes>1)']
def variants():
 for point,mode,iform,vform,scope,sgn in itertools.product(range(4),range(1,4),range(3),range(3),range(2),range(2)):
  src=h.BASE;start=src.index(LOOP);end=src.index(END,start)+len(END);part=src[start:end];at=part.index(POINTS[point]);tail=part[at:];prefix=part[:at];decl=[];step=[]
  if mode&1:
   src=src.replace(LOOP,LOOP.replace(';++i)', ';)'));prefix=prefix.replace(LOOP,LOOP.replace(';++i)', ';)'))
   if iform==0:
    decl.append(('int 'if sgn else'unsigned ')+'previousIndex;');step.append('previousIndex=i++;');ix='previousIndex'
   elif iform==1:step.append('++i;');ix='(i-1)'
   else:decl.append(('int 'if sgn else'unsigned ')+'previousIndex;');step.append('previousIndex=i;i=previousIndex+1;');ix='previousIndex'
   tail=re.sub(r'\bi\b',ix,tail)
  if mode&2:
   tail=tail.replace(END,'')
   if vform==0:
    decl.append('char *previousVertex;');step.append('previousVertex=vertices;vertices+=desc.Get_FVF_Size();');vx='previousVertex'
   elif vform==1:step.append('vertices+=desc.Get_FVF_Size();');vx='(vertices-desc.Get_FVF_Size())'
   else:decl.append('unsigned previousVertex;');step.append('previousVertex=(unsigned)vertices;vertices=(char *)(previousVertex+desc.Get_FVF_Size());');vx='((char *)previousVertex)'
   tail=re.sub(r'\bvertices\b',vx,tail)
  ds=''.join(decl);st=''.join(step)
  if scope==0:prefix=prefix.replace('{','{'+ds,1)
  else:prefix=ds+prefix
  # Re-find after any header spelling change.
  start=src.index('   for(unsigned i=0;i<(unsigned)mesh->Get_Vertex_Count();');end=src.index(END,start)+len(END)
  src=src[:start]+prefix+st+tail+src[end:]
  yield f'earlyadvance_{point}{mode}{iform}{vform}{scope}{sgn}',src
if __name__=='__main__':
 t=time.time()
 with ThreadPoolExecutor(max_workers=8)as pool:r=list(pool.map(h.run_one,variants()))
 r.sort(key=lambda x:(x.get('nd',99999),x.get('cost',99999)))
 (h.HERE/'earlyadvance-results.json').write_text(json.dumps(r,indent=2))
 print(json.dumps({'count':len(r),'valid':sum('nd'in x and not x.get('unresolved')for x in r),'seconds':time.time()-t,'best':r[:8]},indent=2))
