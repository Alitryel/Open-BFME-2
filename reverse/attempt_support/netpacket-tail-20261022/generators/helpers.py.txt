import sys,itertools,re
sys.path.insert(0,'build/netpacket-20261022'); from search import *
SIGN='NetCommandList *NetPacket::getCommandList()\n'
H,B=BANK.split(SIGN,1)
ZPOS=B.index("} else if (m_packet[i] == 'Z') {")
ZEND=B.index("\t\t} else {\n\t\t\trva0058E57B",ZPOS)
starts=['\t\t\tlastCommand = new NetCommandRef(msg);','\t\t\tdelete lastCommand;', '\t\t\tNetCommandRef *ref = retval->addMessage(msg);','\t\t\tmsg->setTiming(timestamp, frame);']
# Inlined cleanup helpers preserve the same operations; different output forms
# test whether the index restoration becomes an allocator fixup at inline exit.
items=[]
for startidx,passref,retkind,includeidx,applyD in itertools.product(range(4),range(8),range(3),range(2),range(2)):
 start=B.rfind(starts[startidx],ZPOS,ZEND); tail=B[start:ZEND]
 # Variables passed by reference or copied from the caller determine lifetimes.
 params=[('NetCommandMsg *','msg'),('NetCommandRef *','lastCommand'),('NetCommandList *','retval'),('UnsignedInt','timestamp'),('UnsignedInt','frame'),('UnsignedByte','playerID'),('UnsignedByte','commandType'),('UnsignedInt','commandID'),('Bool','commandIDSet'),('UnsignedByte','relay'),('Int','offset'),('Int','i')]
 live=[(t,n) for t,n in params if re.search(r'\b'+n+r'\b',tail)]
 if not includeidx:tail=tail.replace('\t\t\ti = offset;\n',''); live=[(t,n) for t,n in live if n not in ('i','offset')]
 retname=['','lastCommand','i'][retkind]
 if retname=='i' and not includeidx:continue
 decl=[]
 for t,n in live:
  if n in ['commandID','commandIDSet','i','lastCommand']:ref=True
  else:ref=bool(passref&(1<<(['msg','retval','offset'].index(n)))) if n in ['msg','retval','offset'] else False
  if n==retname:ref=False
  decl.append(t+(' &' if ref else ' ')+n)
 rettype='void' if not retname else ('NetCommandRef *' if retname=='lastCommand' else 'Int')
 body=tail + ('\treturn '+retname+';\n' if retname else '')
 name=f'npFinish{startidx}_{passref}_{retkind}_{includeidx}_{applyD}'
 helper='static __forceinline '+rettype+' '+name+'('+', '.join(decl)+')\n{\n'+body+'}\n\n'
 call='\t\t\t'+(retname+' = ' if retname else '')+name+'('+', '.join(n for t,n in live)+');\n'
 if not includeidx:call+='\t\t\ti = offset;\n'
 nb=B[:start]+call+B[ZEND:]
 if applyD:
  if tail not in nb[:ZPOS]:continue
  nb=nb.replace(tail,call,1)
 items.append((name,H+helper+SIGN+nb))
batch(items,'helpers')
