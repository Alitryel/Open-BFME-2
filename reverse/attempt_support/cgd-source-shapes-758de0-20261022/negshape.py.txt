from pathlib import Path
import sys,itertools
sys.path.insert(0,str(Path(__file__).resolve().parent));import search
D=search.DIR/'negshape';D.mkdir(exist_ok=True);b=search.DIR.joinpath('base.cpp').read_text()
ctors=[
 'x=v.x; y=v.y; z=v.z;',
 'x=v.x*1.0f; y=v.y*1.0f; z=v.z*1.0f;',
 'x=-(-v.x); y=-(-v.y); z=-(-v.z);',
 'float a=v.x,b=v.y,c=v.z; x=a; y=b; z=c;',
 '*this=v;',
 'set(v.x,v.y,v.z);',
 'x=v.x/1.0f; y=v.y/1.0f; z=v.z/1.0f;',
 'x=1.0f*v.x; y=1.0f*v.y; z=1.0f*v.z;',
]
negexpr=['-V','V*-1.0f','-1.0f*V','V/-1.0f','-(V*1.0f)','(V*1.0f)*-1.0f','-(-(-V))','-(V/1.0f)']
for n,(ctor,nx,ny,nz,shape) in enumerate(itertools.product(range(8),range(8),range(8),range(8),range(2))):
 # Balanced fractional 2048 design: retain interactions without redundant all-order set.
 if ((ctor+nx+ny+nz)&1)!=shape:continue
 s=b.replace('x=v.x; y=v.y; z=v.z;',ctors[ctor])
 neg=' '.join(v+'='+negexpr[k].replace('V',v)+';' for v,k in zip('xyz',[nx,ny,nz]))
 if shape:neg=' '.join(neg.split()[::-1])
 s=s.replace('x=-x; y=-y; z=-z;',neg)
 s=s.replace(' ContactVector &negate()', ' void set(float a,float b,float c) { x=a; y=b; z=c; }\n ContactVector &negate()')
 (D/('n%05d.cpp'%n)).write_text(s)
