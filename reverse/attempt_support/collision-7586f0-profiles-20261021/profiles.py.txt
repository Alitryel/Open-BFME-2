from pathlib import Path
import sys,json
root=Path.cwd();sys.path.insert(0,str(root/'tools'))
from variant_search import Scorer,diff_text
from locate import object_functions
out=root/'build/collision-finish';src=(out/'banked.cpp').read_text()
sym='?rva007586F0@CollisionManagerImpl@@QAEXXZ'
variants=[]
for form in ['A + 1','&lists[src + 1]']:
 for profile in ['', 'vc71-sp1','vc71-ddk-4035']:
  variants.append(('pointer'+str(len(variants)), ('// compiler: '+profile+'\n' if profile else '')+src.replace('A + 1',form)))
for expr in ['&pairs[src >> 1][1]','pairs[src >> 1] + 1','&pairs[(unsigned)src / 2][1]']:
 variants.append(('pair'+str(len(variants)),src.replace('SortList lists[4];','union { SortList lists[4]; SortList pairs[2][2]; };').replace('A + 1',expr)))
# Independent layout expression: named halves, same measured 4x12 byte storage.
for expr in ['&pairs[src >> 1].second','&pairs[src / 2].second']:
 variants.append(('named'+str(len(variants)),src.replace('SortList lists[4];','struct Pair { SortList first, second; }; union { SortList lists[4]; Pair pairs[2]; };').replace('A + 1',expr)))
scorer=Scorer(sym,0x7586f0,460,out/'objects')
results=[]
for name,s in variants:
 r=scorer(s);a,b=r.pop('compiled',b''),r.pop('target',b'')
 extent=None
 obj=scorer.dir/('c'+str(scorer.n)+'.obj')
 if obj.exists():
  extent=next((len(v) for n,v,rel in object_functions(obj) if n==sym),None)
 r.update(name=name,extent=extent,different_bytes=sum(x!=y for x,y in zip(a,b))+abs(len(a)-len(b)))
 (out/(name+'.cpp')).write_text(s)
 if a:(out/(name+'.diff.txt')).write_text(diff_text(a,b,200))
 results.append(r);(out/'results.json').write_text(json.dumps(results,indent=2));print(json.dumps(r),flush=True)
