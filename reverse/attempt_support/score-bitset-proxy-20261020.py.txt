from pathlib import Path
import subprocess,sys,json
root=Path(__file__).resolve().parents[1]
d=root/'build/score-bitset-inlining-oct20'
s=(d/'budget100-repeat.cpp').read_text()
s=s.replace('return kinds.test(bit); }','return kinds.test(bit); }\n    __forceinline bool testProxy(int bit) { return kinds[bit]; }')
s=s.replace('const ScoreTemplateKindView *mine=','ScoreTemplateKindView *mine=').replace('const ScoreTemplateKindView *theirs=','ScoreTemplateKindView *theirs=').replace('(const ScoreTemplateKindView *)','(ScoreTemplateKindView *)')
results=[]
for mine,theirs in [('testProxy','testKind'),('testKind','testProxy'),('testProxy','testProxy')]:
    for order in ['normal','other-first']:
        text=s.replace('mine->testKind(90)','mine->'+mine+'(90)').replace('theirs->testKind(90)','theirs->'+theirs+'(90)')
        if order=='other-first':text=text.replace('if(mine->'+mine+'(90)!=theirs->'+theirs+'(90))','if(theirs->'+theirs+'(90)!=mine->'+mine+'(90))')
        key='proxy-'+mine+'-'+theirs+'-'+order;p=d/(key+'.cpp');p.write_bytes(text.encode())
        r=subprocess.run([sys.executable,'tools/explain_mismatch.py','??MS4SortElem8@@QBE_NABU0@@Z','--rva','0x005EC52E','--size','102','--source',p.relative_to(root).as_posix()],cwd=root,capture_output=True,text=True,encoding='utf8',errors='replace')
        log=r.stdout+r.stderr;p.with_suffix('.log').write_bytes(log.encode())
        results.append({'key':key,'exit':r.returncode,'exact':'OK: bytes match' in log,'summary':[x for x in log.splitlines() if x.startswith(('compiled symbol size:','classification:','unresolved:','first diff:'))]})
(d/'proxy-results.json').write_bytes((json.dumps(results,indent=2)+'\n').encode())
