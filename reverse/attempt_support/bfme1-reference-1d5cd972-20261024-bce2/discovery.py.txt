from pathlib import Path
import datetime, hashlib, json, os, subprocess, sys, time
ROOT=Path.cwd(); OUT=ROOT/'build/resume-eh-local/turn71-reference'
sys.path.insert(0,str(ROOT/'tools'))
def utc(): return datetime.datetime.now(datetime.timezone.utc).isoformat()
commands=[
 ('bfme1-scan',['py','-3','tools/bfme1_sweep.py','scan']),
 ('bfme1-ranked',['py','-3','tools/bfme1_sweep.py','ranked']),
 ('bfme1-near',['py','-3','tools/bfme1_sweep.py','near']),
 ('lift',['py','-3','reference/open-bfme-1/tools/lift_lane.py','--limit','1000']),
 ('zh-compile',['py','-3','tools/zh_sweep.py','compile','--force','--jobs','2']),
 ('zh-match',['py','-3','tools/zh_sweep.py','match']),
 ('zh-packets',['py','-3','-c',"import sys;from pathlib import Path;sys.path.insert(0,'tools');import zh_sweep;zh_sweep.PACKET_DIR=Path.cwd()/'build/resume-eh-local/turn71-reference/packets';sys.argv=['zh_sweep','packets'];zh_sweep.main()"]),
]
receipt=dict(started=utc(),donor=subprocess.check_output(['git','-C','reference/open-bfme-1','rev-parse','HEAD']).decode().strip(),steps=[])
mode=sys.argv[1] if len(sys.argv)>1 else 'all'
if mode=='zh-only': commands=[pair for pair in commands if pair[0]=='zh-compile']
elif mode=='search-only': commands=[pair for pair in commands if pair[0]!='zh-compile']
receipt_path=OUT/('discovery-'+mode+'.json')
for label,cmd in commands:
    step=dict(name=label,command=cmd,started=utc());receipt['steps'].append(step)
    receipt_path.write_text(json.dumps(receipt,indent=2)+'\n')
    log=OUT/(label+'.txt');start=time.monotonic()
    with log.open('wb') as f: cp=subprocess.run(cmd,stdout=f,stderr=subprocess.STDOUT,env={**os.environ,'PYTHONUNBUFFERED':'1'})
    step.update(exit=cp.returncode,elapsed_seconds=time.monotonic()-start,finished=utc(),log_sha256=hashlib.sha256(log.read_bytes()).hexdigest())
    receipt_path.write_text(json.dumps(receipt,indent=2)+'\n')
    if cp.returncode: sys.exit(cp.returncode)
print(json.dumps(receipt),flush=True)
