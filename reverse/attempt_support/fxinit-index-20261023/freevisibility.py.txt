import sys,json,itertools,time
from pathlib import Path
from concurrent.futures import ThreadPoolExecutor
sys.path.insert(0,'build/fxinit-seat22');import search as h
h.HERE=Path('build/fxinit-index22').resolve()
DECL='typedef void (__cdecl *GameFreeFunction)(void *,int);\nextern "C" GameFreeFunction __gameMemFreePtr;\n'
# Real rowed 17B wrapper, independently visible for both already-resolved
# C++ spellings the bank uses. No alias or new pin is added.
BODY='void __cdecl NAME(void *block) {__gameMemFreePtr(block,3);}\n'
def variants():
 for f,a,tree in itertools.product(range(6),range(6),range(3)):
  x=h.BASE;pre=DECL;post=''
  for name,choice in [('free',f),('Rva00030830GameFree',a)]:
   body=BODY.replace('NAME',name)
   if choice==1:pre+='__declspec(noinline) '+body
   elif choice==2:pre+='inline __declspec(noinline) '+body
   elif choice==3:post+='__declspec(noinline) '+body
   elif choice==4:post+=body
   elif choice==5:pre+='__forceinline '+body
  if tree:
   x=x.replace('struct _Rb_tree_header_proxy {_Rb_tree_node_base *_M_data;};','struct _Rb_tree_header_proxy {_Rb_tree_node_base *_M_data; ~_Rb_tree_header_proxy() {if(_M_data) ::free((void*)_M_data);} };')
   x=x.replace(' ~Rva0018C262();',' __declspec(noinline) ~Rva0018C262();\n void rva0018C316();')
   body='Rva0018C262::~Rva0018C262() {rva0018C316();}\n'
   if tree==1:pre+=body
   else:post+=body
  # Body definitions go after all type declarations and before Init; they are
  # not put before _STL declarations to avoid unrelated first-use effects.
  x=x.replace('bool FXShaderGeometry::Init(',pre+'bool FXShaderGeometry::Init(')+'\n'+post
  yield 'freevis_'+str(f)+str(a)+str(tree),x
if __name__=='__main__':
 t=time.time()
 with ThreadPoolExecutor(max_workers=8)as ex:r=list(ex.map(h.run_one,variants()))
 r.sort(key=lambda x:(x.get('cost',99999),x.get('nd',99999)))
 (h.HERE/'free-results.json').write_text(json.dumps(r,indent=2))
 print(json.dumps({'count':len(r),'seconds':time.time()-t,'errors':sum('error'in x for x in r),'best':r[:5]},indent=2))
