import sys,itertools,json,time
from pathlib import Path
from concurrent.futures import ThreadPoolExecutor,as_completed
sys.path.insert(0,'build/fxinit-seat22');import search as h
h.HERE=Path('build/fxinit-edges22').resolve()
LOOP='   for(unsigned i=0;i<(unsigned)mesh->Get_Vertex_Count();++i) {'
def variants():
 for anchor,scope,form,mask,pn in itertools.product(range(2),range(3),range(3),range(1,16),range(4)):
  src=h.BASE;origin=['tangents','binormals'][anchor]
  delta='(vectorCursor-(unsigned)'+origin+')'
  index=[delta+'/sizeof(Vector3)','('+delta+'>>2)/3','('+delta+'/3)>>2'][form]
  index='('+index+')';decl='unsigned vectorCursor=(unsigned)'+origin+(';'if scope==2 else'+i*sizeof(Vector3);')
  lo=src.index(LOOP);hi=src.index(' m_IB=new',lo);body=src[lo:hi]
  for k,stream in enumerate(['links[0]','links[1]','weights[0]','weights[1]']):
   if mask>>k&1:body=body.replace(stream+'[i]',stream+'['+index+']')
  for k,stream in enumerate(['positions[1]','normals[1]']):
   if pn>>k&1:body=body.replace(stream+'[i]','*(const Vector3 *)(vectorCursor+((unsigned)'+stream+'-(unsigned)'+origin+'))')
  if scope==0:body=body.replace(LOOP,LOOP+decl)
  if scope==1:body=body.replace('    if(m_Passes>1)',decl+'\n    if(m_Passes>1)')
  if scope==2:body=decl+'\n'+body.replace('    vertices+=desc.Get_FVF_Size();','    vertices+=desc.Get_FVF_Size(); vectorCursor+=sizeof(Vector3);')
  yield f'crossstride_{anchor}{scope}{form}{mask:02x}{pn}',src[:lo]+body+src[hi:]
if __name__=='__main__':
 t=time.time();items=list(variants());r=[]
 with ThreadPoolExecutor(max_workers=8)as pool:
  for f in as_completed([pool.submit(h.run_one,x)for x in items]):
   r.append(f.result())
   if len(r)%512==0:print('tested',len(r),'best',min((x for x in r if 'nd'in x),key=lambda x:(x['nd'],x['cost'])),flush=True)
 r.sort(key=lambda x:(x.get('nd',99999),x.get('cost',99999)))
 (h.HERE/'crossstride-results.json').write_text(json.dumps(r,indent=2))
 print(json.dumps({'count':len(r),'valid':sum('nd'in x and not x.get('unresolved')for x in r),'seconds':time.time()-t,'best':r[:4]},indent=2))
