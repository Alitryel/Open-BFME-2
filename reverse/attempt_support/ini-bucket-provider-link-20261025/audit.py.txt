"""Bounded, read-only before/after ownership overlay and identity audit."""
from pathlib import Path
import collections
import copy
import hashlib
import json
import shutil
import sys

sys.path.insert(0, str(Path.cwd() / "tools"))
import allowed_symbols as allowed
import build
import link_census
import link_check

out = Path("build/ini-bucket-link-20261025")
control = json.loads((out / "control.json").read_text(encoding="utf-8"))
sources = ["Code/GameEngine/Source/Common/INI/INI.cpp",
           "Code/GameEngine/Source/Common/System/INI_ctor.cpp",
           "Code/GameEngine/Source/Common/BfmeObject476Copy.cpp",
           "Code/GameEngine/Source/GameClient/MessageStream/SelectionIDMap.cpp",
           "Code/Libraries/Source/WWVegas/WWLib/stlport_lower_bound_uint.cpp"]
rows = link_census.ledger()
live = {p.relative_to(build.ROOT).as_posix() for p in link_census.compile_sources(rows)}
idx = link_check.load_index()
index_hash = hashlib.sha256(link_check.INDEX.read_bytes()).hexdigest()
retired = {s: b["object"] for s, b in idx["blockers"].items() if s not in live}
link_check.forget(idx, retired.values())
truth = link_census.RetailTruth(rows)
objects = [build.obj_path(Path.cwd() / s) for s in sources]
bank = out / "control"
bank.mkdir(exist_ok=True)
before_object = bank / objects[0].name
shutil.copyfile(out / "before-obj", before_object)
weights = link_check.source_bytes(set(sources))
results = {}
for state, obj in [("before", before_object), ("after", objects[0])]:
    view = copy.deepcopy(idx)
    link_check.refresh(view, [obj, *objects[1:]], truth)
    results[state] = {}
    for source, object_path in zip(sources, [obj, *objects[1:]]):
        result = link_check.check_object(object_path, view, truth, source)
        results[state][source] = {k: sorted(result[k]) for k in
                                 ("unresolved", "duplicates", "comdat", "selected", "addresses")}
    print(state, {k: len(v) for k, v in results[state][sources[0]].items() if v})
assert any(results["before"][sources[0]].values())
assert not any(results["after"][sources[0]].values()), results["after"][sources[0]]
for source in sources[1:]:
    for key in results["before"][source]:
        old = {tuple(x) if isinstance(x, (list, tuple)) else x
               for x in results["before"][source][key]}
        new = {tuple(x) if isinstance(x, (list, tuple)) else x
               for x in results["after"][source][key]}
        assert new <= old, (source, key, new - old)
assert hashlib.sha256(link_check.INDEX.read_bytes()).hexdigest() == index_hash
ident, scoped = allowed.load(sources)
found = allowed.scan(ident, scoped)
wrong = [r for r in found if r[5] in allowed.WRONG]
assert not [r for r in wrong if r[0]["source"] == sources[0]], wrong
helper = control["wrong_bindings"][0]["symbol"]
new = allowed.Obj(objects[0])
body, size, relocs = new.body(new.by_name[helper])
assert size == 59 and ident.certify(str(objects[0]), helper, 0x148DDF) == "retail"
ctor_call = [r for r in found if r[0]["source"] == sources[0]
             and int(r[0]["target_rva"], 16) == 0x2D281 and r[4] == 0x148DDF]
assert len(ctor_call) == 1 and ctor_call[0][5] == "fold", ctor_call
ledger_counts = collections.Counter(r["source"] for r in rows if r["source"] in sources)
report = {
    "input_revision": control["revision"], "index": idx["meta"], "index_sha256": index_hash,
    "scope": "Paired bounded ownership overlay: retired census objects removed in memory and five current/control objects refreshed; other providers retain published facts. Census and global current-ledger closure await measurement.",
    "sources": sources, "row_counts": dict(ledger_counts),
    "current_authored_nonpadding_bytes": weights, "retired_index_objects": retired,
    "before": results["before"], "after": results["after"],
    "identity_counts": dict(collections.Counter(r[5] for r in found)),
    "identity_wrong_findings": len(wrong), "helper_extent": size, "helper_verdict": "retail",
    "provider_identity_findings": [{"source": r[0]["source"], "row": r[0]["name"],
                                    "symbol": r[3], "target": hex(r[4]), "class": r[5],
                                    "detail": r[6]} for r in wrong],
    "helper_sha256": hashlib.sha256(body).hexdigest(),
    "constructor_call": {"offset": ctor_call[0][1], "class": ctor_call[0][5],
                         "target": "0x00148DDF", "symbol": helper},
    "objects": {s: {"source_sha256": hashlib.sha256(Path(s).read_bytes()).hexdigest(),
                    "object_sha256": hashlib.sha256(o.read_bytes()).hexdigest(),
                    "ordinary_receipt_current": build.compile_is_current(Path.cwd() / s, o)}
                for s, o in zip(sources, objects)},
}
assert all(d["ordinary_receipt_current"] for d in report["objects"].values())
(out / "paired-result.json").write_text(json.dumps(report, indent=2) + "\n", encoding="utf-8")
allowed.write_refs(found, out / "cohort-refs.tsv")
print("Verified rows:", len(scoped), "wrong bindings:", len(wrong))
print("INI bounded clean scope:", weights[sources[0]], "authored nonpadding bytes")
for source in sources:
    print(source, "before", {k: len(v) for k, v in results["before"][source].items() if v},
          "after", {k: len(v) for k, v in results["after"][source].items() if v})
