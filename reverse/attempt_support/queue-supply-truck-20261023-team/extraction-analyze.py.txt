# Function definitions only, read via AST by region-audit.py.

def coff(path):
 data=path.read_bytes(); syms=build.read_object_symbols(data);secs=[]
 for ix in range(build.u16(data,2)):
  at=20+ix*40; ptr=build.u32(data,at+20);size=build.u32(data,at+16);rp=build.u32(data,at+24);rc=build.u16(data,at+32)
  rel={build.u32(data,rp+j*10):syms[build.u32(data,rp+j*10+4)]['name'] for j in range(rc)}
  secs.append(dict(body=data[ptr:ptr+size] if ptr else b'',relocs=rel,name=data[at:at+8].rstrip(b'\0').decode()))
 return syms,secs

def eh(path):
 syms,secs=coff(path)
 hs=next(s for s in syms if s['name']=='__ehhandler$'+NAME);hsec=secs[hs['section']-1]
 fr=next(v for k,v in hsec['relocs'].items() if hs['value']<=k<hs['value']+5)
 fs=next(s for s in syms if s['name']==fr);fsec=secs[fs['section']-1];base=fs['value']
 magic,states=struct.unpack_from('<II',fsec['body'],base)
 umname=fsec['relocs'][base+8];um=next(s for s in syms if s['name']==umname);usec=secs[um['section']-1]
 actions=[]
 for n in range(states):
  at=um['value']+n*8;prev=struct.unpack_from('<i',usec['body'],at)[0];aname=usec['relocs'][at+4]
  act=next(s for s in syms if s['name']==aname);asec=secs[act['section']-1]
  ends=sorted({s['value'] for s in syms if s['section']==act['section'] and s['value']>act['value']})
  end=ends[0] if ends else len(asec['body'])
  listing=[]
  for i in insns(asec['body'][act['value']:end]):
   rel=[sym for off,sym in asec['relocs'].items() if act['value']+i.address<=off<act['value']+i.address+i.size]
   listing.append(i.mnemonic+' '+i.op_str+(' ['+','.join(rel)+']' if rel else ''))
  actions.append(dict(state=n,prev=prev,action=listing))
 return dict(magic=hex(magic),states=states,actions=actions)

def normalize(s):
 s=s.replace('?rva0026C3AC@AICommandInterface','?aiDock@AICommandInterface').replace('PAVRva002D06CA@@','PAVThingFactory@@')
 if '??queueSupplyTruck@AIPlayer' in s:
  s=re.sub(r'@\?[A-Z]+@(?=\?\?queueSupplyTruck)', '@?LOCAL@',s)
 return s

def inventory(body,relocs):
 out=[]
 for i in insns(body):
  if i.mnemonic!='call':continue
  sym=next((s for off,t,s in relocs if i.address<=off<i.address+i.size),None)
  if sym: kind='direct:'+normalize(sym)
  elif '[ebp' in i.op_str:kind='indirect:Team PMF'
  else:
   m=re.search(r'\+ (0x[0-9a-f]+)\]',i.op_str)
   kind='indirect:virtual '+(m.group(1) if m else '0')
  out.append(dict(offset=hex(i.address),instruction=i.mnemonic+' '+i.op_str,kind=kind))
 return out
