from pathlib import Path
import subprocess,sys,json
out=Path('build/resume-eh-local/turn64-camera-tail');out.mkdir(exist_ok=True)
base=Path('build/resume-eh-local/turn64-camera-planar/points.cpp').read_text()
view='''class CameraSpatial : public Coord3D {
public:
 CameraSpatial() {}
 CameraSpatial(float px,float py,float pz) {x=px;y=py;z=pz;}
 CameraSpatial(const Coord3D &p) {x=p.x;y=p.y;z=p.z;}
 CameraSpatial(const CameraSpatial &p) {x=p.x;y=p.y;z=p.z;}
 CameraSpatial &operator=(const Coord3D &p) {x=p.x;y=p.y;z=p.z;return *this;}
 CameraSpatial &operator=(const CameraSpatial &p) {x=p.x;y=p.y;z=p.z;return *this;}
 __forceinline CameraSpatial &Scale(float r) {x*=r;y*=r;z*=r;return *this;}
 __forceinline CameraSpatial &Add(const Coord3D &p) {x+=p.x;y+=p.y;z+=p.z;return *this;}
};
'''
variants={}
for scope in ['candidate','all']:
 s=base.replace('bool PolygonTrigger::adjustCameraMovement(',view+'bool PolygonTrigger::adjustCameraMovement(',1)
 for n in (['candidate'] if scope=='candidate' else ['candidate','along','cur','used','next']):s=s.replace('Coord3D '+n,'CameraSpatial '+n)
 variants['assignment_'+scope]=s
for chain in ['separate','combined','copy_scaled','scale_result']:
 s=base.replace('bool PolygonTrigger::adjustCameraMovement(',view+'bool PolygonTrigger::adjustCameraMovement(',1)
 for n in ['along','next']:s=s.replace('Coord3D '+n,'CameraSpatial '+n)
 old='along.scale(remaining);CameraSpatial next(candidate);next.add(&along);'
 assert old in s
 tail={
 'separate':'along.Scale(remaining);CameraSpatial next(candidate);next.Add(along);',
 'combined':'CameraSpatial next(candidate);next.Add(along.Scale(remaining));',
 'copy_scaled':'CameraSpatial scaled(along);scaled.Scale(remaining);CameraSpatial next(candidate);next.Add(scaled);',
 'scale_result':'CameraSpatial &scaled=along.Scale(remaining);CameraSpatial next(candidate);next.Add(scaled);'}[chain]
 variants[chain]=s.replace(old,tail)
sym='?adjustCameraMovement@PolygonTrigger@@QAE_NPBUCoord3D@@0PAU2@_N@Z';results=[]
for key,s in variants.items():
 p=out/(key+'.cpp');p.write_bytes(s.encode())
 r=subprocess.run([sys.executable,'tools/explain_mismatch.py',sym,'--rva','0x002E3AA8','--size','723','--source',p.as_posix()],stdout=subprocess.PIPE,stderr=subprocess.STDOUT,text=True)
 p.with_suffix('.txt').write_bytes(r.stdout.encode());d={'variant':key,'exit':r.returncode,'summary':[x for x in r.stdout.splitlines() if x.startswith(('compiled symbol size:','classification:','first diff:','unresolved:'))]}
 if r.returncode:d['tail']=r.stdout[-900:]
 results.append(d);print(d,flush=True)
 (out/'results.json').write_bytes((json.dumps(results,indent=2)+'\n').encode())
