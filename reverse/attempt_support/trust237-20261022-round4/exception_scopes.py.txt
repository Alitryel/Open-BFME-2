from pathlib import Path
import sys,itertools
sys.path.insert(0,str(Path('build/trust237-round4').resolve()))
import search
kinds=['seh_finally','seh_filter','cpp_rethrow','empty','empty_throw','empty_force','empty_force_throw','empty_noinline']
counts=[1,4,16]
shapes=['return 1;','int r=0;if((u&1u)<=1u)goto yes;return r;yes:return r+1;','return (unsigned __int64)(unsigned char)u<=255ui64;']
def wrap(code,kind,count):
 if kind=='seh_finally':
  for _ in range(count):code='__try{'+code+'}__finally{}'
 elif kind=='seh_filter':
  for _ in range(count):code='__try{'+code+'}__except(0){}'
 elif kind=='cpp_rethrow':
  for _ in range(count):code='try{'+code+'}catch(...){throw;}'
 else:
  for n in range(count):code='{Empty guard'+str(n)+';'+code+'}'
 return code
jobs=[]
for kind,count,pos,shape,cpu,eh in itertools.product(kinds,counts,range(4),range(3),[6,7],['EHsc','EHa']):
 trust='status=WinVerifyTrust(0,&action,&data);';free='LocalFree(argv);';tail=shapes[shape]
 if pos==0:trust=wrap(trust,kind,count)
 elif pos==1:free=wrap(free,kind,count)
 elif pos==2:tail=wrap(tail,kind,count)
 body='long status;'+trust+'unsigned long u=(unsigned long)status;'+free+tail
 if pos==3:body=wrap(body,kind,count)
 s=search.base.replace(search.marker,body)
 if kind.startswith('empty'):
  directive='__forceinline ' if 'force' in kind else ('__declspec(noinline) ' if 'noinline' in kind else '')
  exc=' throw()' if 'throw' in kind else ''
  guard='struct Empty{'+directive+'~Empty()'+exc+'{}};\n'
  s=s.replace('int Rva00237D3DCheck() {',guard+'int Rva00237D3DCheck() {')
 s=s.replace('/G7',f'/G{cpu}').replace('/EHsc',f'/{eh}')
 jobs.append((f'e{len(jobs):04}',s,dict(kind=kind,count=count,position=pos,shape=shape,cpu=cpu,eh=eh)))
search.execute('exception_scopes',jobs)
