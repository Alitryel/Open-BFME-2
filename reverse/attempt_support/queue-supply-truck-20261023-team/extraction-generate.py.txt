from pathlib import Path
import json
ROOT=Path('build/queueSupplyTruck-research')
OUT=ROOT/'extraction'
base=(ROOT/'experiments/canonical.cpp').read_text().replace('\t\t\t\t\t\t_ReadWriteBarrier();\n','')
gh=(ROOT/'cfg/clean_f_nonull_center_c03_exit_condition_goto.cpp').read_text()

def loop_at(s, start):
    a=s.index('PlayerTeamList::const_iterator it;',start)
    a=s.rfind('\n',0,a)+1
    b=s.index('for (it =',a)
    op=s.index('{',b); n=1; end=op+1
    while n:
        if s[end]=='{': n+=1
        elif s[end]=='}': n-=1
        end+=1
    return a,end

def transform(s, selected, ref=False, force=False):
    fun=s.index('void AIPlayer::queueSupplyTruck( void )')
    a,b=loop_at(s,fun)
    c,d=loop_at(s,b)
    e,f=loop_at(s,d)
    definitions=[]; declarations=[]; replacements=[]
    prefix='__forceinline' if force else 'inline'
    if 'first' in selected:
        body=s[a:b]
        if ref:
            decl='void countHarvesters(Int &totalHarvesters)'
            definition=f'{prefix} void AIPlayer::countHarvesters(Int &totalHarvesters)\n{{\n{body}\n}}\n'
            replacement='\tcountHarvesters(totalHarvesters);'
        else:
            decl='Int countHarvesters()'
            definition=f'{prefix} Int AIPlayer::countHarvesters()\n{{\n\tInt totalHarvesters = 0;\n{body}\n\treturn totalHarvesters;\n}}\n'
            # Remove the redundant caller initialization only for return-valued helper.
            st=s.index('\tInt totalHarvesters;',fun)
            replacements.append((st,a,'\tInt totalHarvesters = countHarvesters();\n'))
            replacement=''
        if ref: replacements.append((a,b,replacement))
        else: replacements.append((a,b,replacement))
        definitions.append(definition); declarations.append(decl)
    if 'recount' in selected:
        body=s[c:d]
        decl='Int recountHarvesters(Object *supplyCenter)'
        definition=f'{prefix} Int AIPlayer::recountHarvesters(Object *supplyCenter)\n{{\n\tInt curGatherers = 0;\n{body}\n\treturn curGatherers;\n}}\n'
        st=s.rfind('\t\t\t\tInt curGatherers = 0;',fun,c)
        assert st>fun
        replacements.append((st,d,'\t\t\t\tInt curGatherers = recountHarvesters(supplyCenter);'))
        definitions.append(definition); declarations.append(decl)
    if 'loose' in selected:
        body=s[e:f].replace('return;','return true;')
        decl='Bool reassignLooseHarvester(BuildListInfo *info)'
        definition=f'{prefix} Bool AIPlayer::reassignLooseHarvester(BuildListInfo *info)\n{{\n{body}\n\treturn false;\n}}\n'
        replacements.append((e,f,'\t\t\tif (reassignLooseHarvester(info)) return;'))
        definitions.append(definition); declarations.append(decl)
    for a,b,repl in sorted(replacements,reverse=True):s=s[:a]+repl+s[b:]
    s=s.replace('private:\n\tTeamInQueue *m_teamBuildQueue;', 'private:\n\t'+';\n\t'.join(declarations)+';\n\tTeamInQueue *m_teamBuildQueue;',1)
    pos=s.index('void AIPlayer::queueSupplyTruck( void )')
    return s[:pos]+'\n'.join(definitions)+'\n'+s[pos:]

variants=[('first_ret',('first',),False),('first_ref',('first',),True),('recount_ret',('recount',),False),('loose_bool',('loose',),False),('counts',('first','recount'),False),('all',('first','recount','loose'),False)]
manifest=[]
for context,s in [('base',base),('gh',gh)]:
    for name,selected,ref in variants:
        name=context+'_'+name
        p=OUT/(name+'.cpp')
        p.write_text(transform(s,selected,ref))
        manifest.append({'name':name,'context':context,'extractions':selected,'count_ref':ref,'force':False})
(OUT/'manifest.json').write_text(json.dumps(manifest,indent=2))
