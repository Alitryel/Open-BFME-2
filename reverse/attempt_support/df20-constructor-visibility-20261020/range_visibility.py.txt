from pathlib import Path
import sys,itertools,concurrent.futures,json,re
sys.path.insert(0,'build/df20-scalars-20261020');import trials
trials.scratch=Path.cwd()/'build/df20-scalars-20261020'
base=Path('build/df20-scalars-20261020/bank.cpp').read_text();trials.symbol=base.splitlines()[0][3:]
ctor='template <> template <> __forceinline basic_string<wchar_t>::basic_string(const wchar_t* first,const wchar_t* last,const allocator<wchar_t>& a) : _String_base<wchar_t,allocator<wchar_t> >(a) { _M_range_initialize(first,last,forward_iterator_tag()); }'
assert ctor in base
specs=[]
for extra,ctorform,base_decl,range_decl,tag in itertools.product([0,1],['primary','specialized'],[0,1],[0,1],['named','temporary']):
 if ctorform=='primary' and tag!='temporary':continue
 s=base
 if extra:s=s.replace('// cl:','// cl: /D_STLP_NEEDS_EXTRA_TEMPLATE_CONSTRUCTORS',1)

 if range_decl:decl+='template <> template <> void basic_string<wchar_t>::_M_range_initialize(const wchar_t*,const wchar_t*,const forward_iterator_tag&);\n'
 decl+='}\n'
 s=s.replace('#include <locale>',decl+'#include <locale>',1)
 if ctorform=='primary':s=s.replace(ctor,'')
 else:
  if tag=='named':s=s.replace(ctor,ctor.replace('{ _M_range_initialize(first,last,forward_iterator_tag()); }','{ forward_iterator_tag tag; _M_range_initialize(first,last,tag); }'))
 specs.append((f'range-x{extra}-{ctorform}-b{base_decl}-r{range_decl}-t{tag}',s))
with concurrent.futures.ThreadPoolExecutor(max_workers=8) as p:r=list(p.map(trials.run,specs))
