from pathlib import Path
import sys,json
sys.path.insert(0,'tools');import build,eh_verify
folder=Path('build/df20-scalars-20261020');name=(folder/'bank.cpp').read_text().splitlines()[0][3:]
game=eh_verify.Image();results=[]
for path in folder.glob('*.cpp'):
 try:
  obj=eh_verify.Obj(build.obj_path(path));sy=obj.by_name[name];sections=obj.eh_sections(sy['section'])
  places,conflicts=eh_verify.section_placements(game,obj,dict(sy,retail_rva=0xdf20),sections)
  out={'label':path.stem,'conflicts':len(conflicts),'sections':[]}
  for s in sections:
   at=places.get(s['index']);d=s['data'];mask=set()
   for o,i,t in s['relocs']:mask.update(range(o,o+4))
   diff=sum(i not in mask and b!=game.bytes[at+i] for i,b in enumerate(d)) if at is not None else None
   out['sections'].append({'name':s['name'],'size':len(d),'nonreloc_diff':diff})
  results.append(out)
 except Exception as e:results.append({'label':path.stem,'error':str(e)})
(folder/'eh-scan.json').write_text(json.dumps(results,indent=2))
ok=[x for x in results if 'sections' in x and not x['conflicts'] and sorted(s['size'] for s in x['sections'])==[212,337]]
ok.sort(key=lambda x:sum(s['nonreloc_diff'] or 0 for s in x['sections']))
print('objects',len(results),'same shape',len(ok))
for x in ok[:12]:print(x)
