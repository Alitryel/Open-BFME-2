"""Make the constructor's return value live through the quote call."""
from pathlib import Path
import concurrent.futures,itertools,json,re,runpy
OUT=Path('build/canmake-alias-20261023/lifetime-initial-values').resolve();OUT.mkdir(parents=True,exist_ok=True)
r=runpy.run_path('reverse/attempt_support/canmake-00596e15-20261022/replay.py.txt');r['run'].__globals__['OUT']=OUT
base=runpy.run_path('reverse/attempt_support/canmake-alias-20261023/context.py.txt')['bank_text']();prefix,body=base.rsplit('int AIBuildableWallSegment::canMake(Player *player)',1)
tests=[]
for init,where,typ,mask,get in itertools.product([0,2,9],range(3),range(4),range(4),range(4)):
 p=prefix;b=body
 type_=['int','unsigned int','unsigned short','signed char'][typ]
 p=p.replace('int code;',type_+' code;').replace('MakeCodeLife():code(0)','MakeCodeLife():code('+str(init)+')')
 if where:
  b=b.replace('MakeCodeLife selected;','')
  if where==1:b=b.replace('Rva0039205C result;','MakeCodeLife selected;\nRva0039205C result;')
  else:b=b.replace('if (failed)','MakeCodeLife selected;\nif (failed)')
 count=[0]
 def replace(m):
  val=int(m[1])
  if val!=init:return m[0]
  bit=count[0];count[0]+=1
  return ''if mask&(1<<bit)else m[0]
 b=re.sub(r'selected.code=(\d+);',replace,b)
 expr=['selected.code','selected.get()','(int)selected.code','(unsigned int)selected.code'][get]
 b=b.replace('return selected.code;','return '+expr+';')
 tests.append((f'initial-{init}{where}{typ}{mask}{get}',p+'\nint AIBuildableWallSegment::canMake(Player *player)'+b))
results=[]
with concurrent.futures.ThreadPoolExecutor(max_workers=8)as pool:
 for i,d in enumerate(pool.map(r['run'],tests),1):
  results.append(d)
  if i%192==0 or d.get('exact_candidate'):print(json.dumps(dict(done=i,total=len(tests),best=sorted(results,key=lambda x:(x.get('nd',999),-x.get('score',0)))[:2])),flush=True)
(OUT/'results.json').write_text(json.dumps(results,indent=2)+'\n')
print(json.dumps(dict(trials=len(results),errors=sum('error'in x for x in results),unresolved=sum(bool(x.get('unresolved'))for x in results),exact=sum(x.get('exact_candidate',False)for x in results),bodies=len({x.get('masked_code_sha256')for x in results}))),flush=True)
