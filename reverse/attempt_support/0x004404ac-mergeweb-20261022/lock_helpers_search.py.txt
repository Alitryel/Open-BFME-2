MERGED=(OUT/'merged-control.cpp').read_text();H,B=MERGED.split(MARK)

def source(f):
 kind,ret,ref,place,qual=f
 h=H;b=B
 typ=['bool','char','unsigned char','int'][ret]
 q=['inline','__forceinline'][qual]
 # Existing rule read, inlined at the same two sites: no extra call or read.
 if kind==0:
  decl=q+' '+typ+' lockedValue(){return m_rules.m_04==1;}'
  expr='lockedValue()'
 elif kind==1:
  decl=q+' '+typ+' lockedValue(int rule){return rule==1;}'
  expr='lockedValue(m_rules.m_04)'
 elif kind==2:
  decl='static '+q+' '+typ+' lockedValue('+('const int &'if ref else 'int ')+'rule){return rule==1;}'
  expr='lockedValue(m_rules.m_04)'
 elif kind==3:
  decl='static '+q+' '+typ+' lockedValue('+('const AptMpGameSetup &'if ref else 'const AptMpGameSetup *')+'owner){return '+('owner.'if ref else 'owner->')+'m_rules.m_04==1;}'
  expr='lockedValue('+('*this'if ref else'this')+')'
 elif kind==4:
  decl=q+' void lockedValue('+('int &'if ret==3 else typ+' &')+'value){value=m_rules.m_04==1;}'
  expr=None
 else:
  decl=q+' int ruleValue(int which){return ((const int *)&m_rules)[which];} '+q+' '+typ+' lockedValue(){return ruleValue(1)==1;}'
  expr='lockedValue()'
 if place:
  # Keep helper bodies after the target, with declarations in the owner.
  # For in-class definitions leave the exact language-level inline contract.
  pass
 h=h.replace('\tvoid rva004404AC(bool enable, int slot);',decl+'\n\tvoid rva004404AC(bool enable, int slot);')
 old='bool locked = m_rules.m_04 == 1;'
 if expr is None:b=b.replace(old,typ+' locked; lockedValue(locked);')
 else:b=b.replace(old,typ+' locked='+expr+';')
 if ref and kind in(0,1,5):b=b.replace(typ+' locked=', 'const '+typ+' &locked=')
 # Place changes only the two independent branch spellings, not helper order.
 if place:b=b.replace('!locked)', '(!locked))').replace('&& (!locked))','&& (!locked))')
 return h+MARK+b
