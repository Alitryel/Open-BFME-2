from pathlib import Path
import os
import subprocess
import sys

sys.path.insert(0, 'tools')
import build

p = Path('build/worker-73bfe0')
if Path('build/STOP_WORKER').exists():
    sys.exit('STOP_WORKER present')
with (p/'hypotheses.md').open('a') as f:
    f.write('\n\n## Native differential trace probe\n\nLink the already compiled bank and semantic candidate separately with one callback recorder. Execute the complete native 874-byte body in a local 32-bit process, patching only its two external calls to that recorder. Compare mutated coordinate arrays, char result, all scanline arguments, updater grid/mask contents and consistent updater identity across calls. Exercise duplicate vertices, horizontal edges, asymmetric coordinates, reverse winding, cyclic rotations and early callback failures. Prediction: the bank diverges and the corrected candidate agrees. Refutation: any corrected trace differs; no whole-program or general polygon correctness claim follows from passing these vectors.\n')
src = build.ROOT/p/'trace_harness_base.cpp'
obj = build.ROOT/p/'trace_harness_base.obj'
cmd, env = build.compiler_command(src, obj)
with (p/'trace-base-compile.log').open('w') as log:
    log.write(repr(cmd)+'\n'); log.flush()
    r = subprocess.run(cmd, env=env, stdout=log, stderr=subprocess.STDOUT)
if r.returncode:
    sys.exit(r.returncode)
for trial in sys.argv[1:] or ['before','traversal','hskip']:
    if Path('build/STOP_WORKER').exists():
        sys.exit('STOP_WORKER present')
    exe = p/f'trace-{trial}.exe'
    link = str(build.vc71_root()/'Vc7/bin/link.exe')
    cmd = [link, '-nologo', '-subsystem:console', f'-out:{exe}', str(obj),
           str(build.obj_path(build.ROOT/p/f'{trial}.cpp')), 'kernel32.lib']
    with (p/f'trace-{trial}-link.log').open('w') as log:
        log.write(repr(cmd)+'\n'); log.flush()
        r = subprocess.run(cmd, env=env, stdout=log, stderr=subprocess.STDOUT)
    print(trial, 'link exit', r.returncode)
    if r.returncode:
        continue
    with (p/f'trace-{trial}.log').open('w') as log:
        r = subprocess.run([str(exe)]+(['stop-after-three'] if trial=='before' else []), env=env, stdout=log, stderr=subprocess.STDOUT, timeout=30)
        log.write(f'process_exit={r.returncode}\n')
    print((p/f'trace-{trial}.log').read_text())
