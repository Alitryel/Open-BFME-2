from search import *

def address(n):
    bits=n&255;kind=(n>>8)&3
    if kind==0:decl='AptMapPreview *preview';value='(AptMapPreview *)&m_60';call='preview->rva0057C597(found);'
    elif kind==1:decl='AptMpGameSetup *owner';value='this';call='((AptMapPreview *)&owner->m_60)->rva0057C597(found);'
    elif kind==2:decl='char *preview';value='(char *)&m_60';call='((AptMapPreview *)preview)->rva0057C597(found);'
    else:decl='Rva0057E3DB *preview';value='&m_60';call='((AptMapPreview *)preview)->rva0057C597(found);'
    var=decl.rsplit(' ',1)[1].lstrip('*')
    prep=var+' = '+value+';'
    b=BODY[:BODY.index('\tif (slot == game->v13())')]
    b+='''
    if (slot == game->v13()) {
        %s = %s;
        bool found = false;
        if (enable && game->getConstSlot(game->v13())->m_hasMap) {
            @0
            if (!observer) { found = true; @1 }
            else if (game->v12()) {
                @2
                for (int i=0; i<8; ++i) {
                    @3
                    if (game->getConstSlot(i) && game->getConstSlot(i)->isAI()) {
                        found = true; @4 break;
                    }
                    @5
                }
                @6
            }
        } else { @7 }
        %s
    }
}
'''%(decl,value,call)
    for i in range(8):b=b.replace('@%d'%i,prep if bits>>i&1 else '')
    return HEAD+MARK+b

def duplicate(n):
    bits=n&15;mode=n>>4
    head,body=HEAD,BODY
    body=body[:body.index('\tif (slot == game->v13())')]
    call='((AptMapPreview *)&m_60)->rva0057C597(found);'
    leaves=[]
    for i in range(4):
        leaves.append((call+' goto PreviewDone;') if bits>>i&1 else 'goto PreviewCall;')
    start='bool found = false;'
    t,f='found = true;',''
    if mode==1:start='bool found;';f='found = false;'
    if mode==2:start='int found = 0;'
    if mode==3:start='bool found = false;';f='found = false;'
    body+='''
    if (slot == game->v13()) {
        %s
        if (enable && game->getConstSlot(game->v13())->m_hasMap) {
            if (!observer) { %s %s }
            else if (game->v12()) {
                %s
                for (int i=0; i<8; ++i) {
                    if (game->getConstSlot(i) && game->getConstSlot(i)->isAI()) {
                        found = true; break;
                    }
                }
                %s
            } else { %s %s }
        } else { %s %s }
        PreviewCall: %s
        PreviewDone: ;
    }
}
'''%(start,t,leaves[0],f,leaves[1],f,leaves[2],f,leaves[3],call)
    return head+MARK+body

if __name__=='__main__':
    run([('addr-%04d'%n,address(n)) for n in range(1024)] + [('dup-%02d'%n,duplicate(n)) for n in range(64)],'address')
