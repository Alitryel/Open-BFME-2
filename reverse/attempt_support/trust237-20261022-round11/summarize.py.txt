from pathlib import Path
import json, hashlib, sys, itertools
ROOT=Path.cwd(); OUT=ROOT/'build/trust237-round11'
sys.path.insert(0,str(ROOT/'tools'))
import build, capstone
from variant import resolve_bytes
NAME='?Rva00237D3DCheck@@YAHXZ'; RVA=0x237d3d; SIZE=226
target=build.read_target_bytes(RVA,SIZE)
rows=json.loads((OUT/'swaps.json').read_text())
good=[r for r in rows if 'error' not in r]
corrected=0
allmul_objects=0
for r in good:
    raw,rel=build.read_object_symbol_bytes(OUT/(r['name']+'.obj'),NAME,code_only=True)
    assert hashlib.sha256(raw).hexdigest()==r['sha256']
    assert rel==[tuple(x) for x in r['relocations']]
    allmul_objects+=any(name=='__allmul' for _,_,name in rel)
    md=capstone.Cs(capstone.CS_ARCH_X86,capstone.CS_MODE_32)
    assert [i.address for i in md.disasm(raw,0) if i.mnemonic=='nop']==r['nops']
    resolved,unresolved=resolve_bytes(raw,rel,target+b'\0'*max(0,len(raw)-SIZE),RVA,len(raw),{'_memset':[0x6291ae],'_WinVerifyTrust@12':[0x655652],'__allmul':[0x629580]})
    corrected+=bool(r['unresolved'])
    r.update(diff=sum(a!=b for a,b in itertools.zip_longest(resolved,target)),prefix=next((i for i,(a,b) in enumerate(zip(resolved,target)) if a!=b),min(len(raw),SIZE)),unresolved=unresolved)
assert len(good)==512 and not any(r['unresolved'] or r['nops'] or r['nop16'] for r in good)
(OUT/'swaps.json').write_text(json.dumps(rows,indent=2)+'\n')
controls=json.loads((OUT/'controls.json').read_text())
for c in controls:
    assert c['returncode']==0
    for name,r in c['functions'].items():
        raw,rel=build.read_object_symbol_bytes(OUT/f"control-g{c['cpu']}.obj",'_'+name,code_only=True)
        assert raw.hex()==r['raw'] and hashlib.sha256(raw).hexdigest()==r['sha256']
        assert rel==r['relocations']==[]
for name in controls[0]['functions']:
    assert len({c['functions'][name]['sha256'] for c in controls})==1
trace=json.loads((OUT/'swap-control.trace.json').read_text())
assert trace['ended'] and trace['object_exists'] and trace['records']==[]
assert hashlib.sha256((ROOT/trace['source']).read_bytes()).hexdigest()==trace['source_sha256']
for name,r in controls[0]['functions'].items():
    raw,rel=build.read_object_symbol_bytes(OUT/'swap-control.trace.obj','_'+name,code_only=True)
    assert raw.hex()==r['raw'] and rel==[]
raw,rel=build.read_object_symbol_bytes(ROOT/'build/scratch/0x00237d3d.obj',NAME,code_only=True)
assert len(raw)==219 and hashlib.sha256(raw).hexdigest()=='23875f743d6816184d590366a5617a42b8e18aedc479c4d75864e0f6d61de0c9'
emulation=json.loads((OUT/'swaps-emulation.json').read_text())
emulation.pop('rows');emulation['detailed_receipt_sha256']=hashlib.sha256((OUT/'swaps-emulation.json').read_bytes()).hexdigest()
unique={r['sha256']:r for r in good}
best=sorted(unique.values(),key=lambda r:(r['diff'],-r['prefix']))[:4]
c2=ROOT/'reference/open-bfme-1/inputs/toolchains/vs2003/Program Files/Microsoft Visual Studio .NET 2003/Vc7/bin/c2.dll'
result=dict(attempts=len(rows),successful=len(good),distinct_raw=len(unique),with_nop=0,with_nop16=0,unresolved_rel32=0,objects_using_allmul=allmul_objects,best_diagnostic_cases=[{k:r[k] for k in ['name','meta','size','prefix','diff','eq','sha256']} for r in best],bank=dict(size=len(raw),sha256=hashlib.sha256(raw).hexdigest(),score='69/86',prefix=205,differences=21),emulation=emulation,compiler_sha256=hashlib.sha256(c2.read_bytes()).hexdigest(),trace=trace,controls=dict(cpus=[c['cpu'] for c in controls],functions_identical_across_cpus=True,functions=controls[0]['functions']))
(OUT/'summary.json').write_text(json.dumps(result,indent=2)+'\n')
print('audited',len(rows),'compiles;',len(unique),'raw bodies; no NOP or bank improvement;',emulation['executions'],'executions passed;',allmul_objects,'objects use matched allmul; corrected',corrected,'existing receipts without recompiling')
