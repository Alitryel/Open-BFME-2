from wdbg import Dbg
from pathlib import Path
import sys,json,time
W=Path('build/locomotor-zero-cfg');label=sys.argv[1] if len(sys.argv)>1 else 'baseline'; force=int(sys.argv[2]) if len(sys.argv)>2 else -1
outname=label+('-force'+str(force) if force>=0 else '-rounds')
d=Dbg(str(W/(label+'.cpp')),str(W/(outname+'.obj')),str(W/(outname+'-debugger.log')))
fn='';cls=-1;snaps=[];start=time.time();armed=False;rnd=0
bps=[0x83acd,0x82978,0x82a4d,0x82273]
def zero(p):
 w=d.words(p,30)
 if len(w)<30:return
 sy=d.words(w[0],16)
 if len(sy)<11 or sy[:2]!=[0,0x100d]:return
 op=d.words(sy[10],8)
 if len(op)<7 or op[1]!=0x24b or op[4:7]!=[0,0,0]:return
 return dict(addr=p,w=w,sym=sy,op=op)
def snap(phase):
 rows=[]
 for head in d.words(0x108807b8,1024):
  p=head
  while p:
   z=zero(p);w=z['w'] if z else d.words(p,13)
   if z:rows.append(z)
   p=w[12]
 snaps.append(dict(phase=phase,rnd=rnd,rows=rows))
 print(phase,rnd,[(r['w'][7],r['w'][3],r['w'][9],r['w'][16]) for r in rows],flush=True)
try:
 d.prompt();d.prompt('set $CanDeferOnBPByAddr = 1');d.prompt('break *0x10783acd')
 for step in range(1000):
  out=d.prompt('cont')
  if 'terminated' in out:break
  if time.time()-start>100:raise TimeoutError('overall deadline')
  off=d.reg('eip')-0x10700000
  if not armed:
   armed=True
   for off2 in bps[1:]:d.prompt('break *0x%x'%(0x10700000+off2))
  if off==0x83acd:
   ebp=d.reg('ebp');p=d.words(ebp,1)[0];fn=d.string(d.words(p+4,1)[0]);cls=d.reg('ebx');rnd=0
  if 'rva001E7ECA' not in fn or cls!=0:continue
  if off==0x82978:rnd+=1;snap('round')
  if off==0x82a4d:
   p=d.reg('eax');z=zero(p)
   if z:
    w=z['w'];snaps.append(dict(phase='color',rnd=rnd,row=z))
    print('COLOR',rnd,'idx',w[7],'priority',w[3],'refs',w[9],'cost',w[16],flush=True)
    if force==w[7]:d.prompt('set *(int*)0x%x = 0'%(p+0x40));print('forced cost=0',flush=True)
  if off==0x82273:snap('finished')
 (W/(outname+'.json')).write_text(json.dumps(snaps,indent=2))
finally:
 if d.p.poll() is None:
  try:d.prompt('quit')
  except Exception:d.p.kill()
 d.log.close()
