from trials import *
# New source hypothesis: pure query/copy contracts can remove EH live sets
# without removing a call or changing its ABI. Each declaration is a probe,
# not a new claim about the original header spelling.
spells=[
 'BfmeFixedStorage128(const BfmeFixedStorage128 &that);',
 'bool _M_is_any() const;',
 'int rva00046827();',
 'void *rva0026F0F0(const void *mask);',
 'int get() const;',
 'const Coord3D *getPosition() const;',
 'int Get_Int(const char *name, int def);',
]
items=[]
for mask in range(256):
 s=BASE
 for k,decl in enumerate(spells):
  assert s.count(decl)==1,(k,s.count(decl))
  if mask>>k&1:s=s.replace(decl,decl[:-1]+' throw();')
 if not mask&128:s=s.replace('template <> void StringBase<char>::releaseBuffer() throw();\n','')
 items.append(('query%03d'%mask,s))
batch(items,'query-contract-interaction')
