import sys,itertools,json,time
from pathlib import Path
from concurrent.futures import ThreadPoolExecutor
sys.path.insert(0,'build/fxinit-seat22')
import search as h
h.HERE=Path('build/fxinit-live22').resolve()
LOOP='   for(unsigned i=0;i<(unsigned)mesh->Get_Vertex_Count();++i) {'
def variants():
 for flow,fieldref,arraymode,mapconst,descptr,order in itertools.product(range(2),range(2),range(3),range(2),range(2),range(6)):
  src=h.BASE
  body=h.ARM
  if flow:
   body='''if(m_Passes>1) {*(unsigned *)(vertices+desc.Get_Basis_End())=pass;return;}
if(m_BonesPerVertex<=0 || !links[0]) return;
if(m_BonesPerVertex==1) {*(int *)(vertices+desc.Get_Basis_End())=remap[links[0][i]];return;}
if(m_BonesPerVertex!=2) return;
'''+h.R+'\n'+h.W+'\n'+h.P+'\n'+h.N
  body=body.replace('m_Passes','passes').replace('m_BonesPerVertex','bones')
  q='const int &'if fieldref else'int '
  groups=[[(q+'passes','m_Passes'),(q+'bones','m_BonesPerVertex')],[(('const 'if mapconst else'')+'_STL::vector<short> &remap','remap'),(('const BfmeFVFDescriptor *desc'if descptr else'const BfmeFVFDescriptor &desc'),('&desc'if descptr else'desc'))],[]]
  for ty,n in [('unsigned short','links'),('unsigned short','weights'),('Vector3','positions'),('Vector3','normals')]:
   if arraymode==0:groups[2].append((f'const {ty} *const *{n}',n))
   if arraymode==1:groups[2].append((f'const {ty} *const (&{n})[2]',n))
   if arraymode==2:
    for k in range(2):groups[2].append((f'const {ty} *{n}{k}',f'{n}[{k}]'));body=body.replace(f'{n}[{k}]',f'{n}{k}')
  if descptr:body=body.replace('desc.','desc->')
  groups=[groups[k]for k in list(itertools.permutations(range(3)))[order]]
  params=[('char *vertices','vertices'),('unsigned pass','pass'),('unsigned i','i')]+sum(groups,[])
  definition='static __forceinline void SkinVertexDispatch('+','.join(p[0]for p in params)+') {\n'+body+'\n}\n'
  call='SkinVertexDispatch('+','.join(p[1]for p in params)+');\n'
  src=src[:h.START]+call+src[h.END:]
  src=src.replace('bool FXShaderGeometry::Init(',definition+'bool FXShaderGeometry::Init(')
  yield 'dispatchinline_'+''.join(map(str,(flow,fieldref,arraymode,mapconst,descptr,order))),src
if __name__=='__main__':
 start=time.time()
 with ThreadPoolExecutor(max_workers=8) as pool:r=list(pool.map(h.run_one,variants()))
 r.sort(key=lambda x:(x.get('nd',99999),x.get('cost',99999)))
 (h.HERE/'dispatchinline-results.json').write_text(json.dumps(r,indent=2))
 print(json.dumps({'count':len(r),'valid':sum('nd'in x and not x.get('unresolved')for x in r),'seconds':time.time()-start,'best':r[:5]},indent=2))
