from pathlib import Path
import sys,json
sys.path.insert(0,'tools')
from variant_search import Scorer
from zh_sweep import carve
base=Path('build/pdb-context-input.cpp.txt').read_text()
at=base.index('int _RTC_GetSrcLine(');head,body=base[:at],base[at:]
mod='static_cast<char **volatile &>(moduleName)'
addr='static_cast<volatile unsigned long &>(address)'
variants=[]
for label,code in [('self-module','moduleName='+mod+';'),
                   ('self-module-address','moduleName='+mod+';address='+addr+';'),
                   ('self-address-module-control','address='+addr+';moduleName='+mod+';')]:
 variants.append((label,head+body.replace('char *errname=(char *)_alloca(1024);','char *errname=(char *)_alloca(1024);'+code)))
for label,change in [('first-module-only',body.replace('*pline=0;source[0]=0;*moduleName=0;','*pline=0;source[0]=0;*'+mod+'=0;')),
                     ('first-address-only',body.replace('info=GetImageInfo(address);','info=GetImageInfo('+addr+');'))]:
 variants.append((label,head+change))
changed=body.replace('*pline=0;source[0]=0;*moduleName=0;','*pline=0;source[0]=0;*'+mod+'=0;').replace('info=GetImageInfo(address);','info=GetImageInfo('+addr+');')
variants.append(('first-both',head+changed))
s=Scorer('?_RTC_GetSrcLine@@YAHKPADHPAHPAPAD@Z',0x75b177,847,Path('build/pdb-direct-views').resolve())
records=[]
for label,src in variants:
 r=s(src);obj=s.dir/('c%d.obj'%s.n)
 extent=next((len(b) for n,b,_ in carve(obj.read_bytes()) if n==s.symbol),None) if obj.exists() else None
 diffs=[i for i,(a,b) in enumerate(zip(r.get('compiled',b''),r.get('target',b''))) if a!=b]
 item=dict(trial=label,n=s.n,extent=extent,differences=diffs,**{k:v for k,v in r.items() if k not in ('compiled','target')})
 records.append(item);(s.dir/'results.json').write_text(json.dumps(records,indent=2)+'\n')
 print({k:v for k,v in item.items() if k!='differences'},'diff_count',len(diffs),'first',diffs[:18],flush=True)
 if r['exact'] and extent==847:(s.dir/'exact.cpp').write_text(src);break
