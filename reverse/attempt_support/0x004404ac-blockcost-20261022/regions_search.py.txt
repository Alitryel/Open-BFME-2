from search import *
import re
# Extract five existing sequences. Change inline parameter webs, not behavior.
A=BODY.index('\n\tif (m_handicap[slot])')
B=BODY.index('\n\tGameWindow *hero')
C=BODY.index('\n\tif (m_playerTemplate[slot])')
D=BODY.index('\n\tGameWindow **team')
E=BODY.index('\n\tif (slot == game->v13())')
F=BODY.rfind('\n}')
segments=[BODY[A:B],BODY[B:C],BODY[C:D],BODY[D:E],BODY[E:F]]

def source(n):
 mask=n%32;kind=(n//32)%16;placement=n//512
 typI='const int &' if kind&1 else 'int'
 typB='const bool &' if kind&2 else 'bool'
 typG='GameInfo *const &' if kind&4 else 'GameInfo *'
 static=bool(kind&8)
 head=HEAD;body=BODY;decls=[];defs=[]
 for k,s in enumerate(segments):
  if not mask&(1<<k):continue
  params=typI+' slot, '+typB+' enable, '+typB+' observer'
  args='slot,enable,observer'
  if k==4:params+=', '+typG+'game';args+=',game'
  name='cost_region%d'%k
  if static:
   params='AptMpGameSetup *self, '+params;args='this,'+args
   s=re.sub(r'(?<![.>\w])m_[A-Za-z0-9_]+',lambda m:'self->'+m.group(0),s)
   s=re.sub(r'\b(HandleHeroSelection|UpdateHeroDisplay)\(',lambda m:'self->'+m.group(1)+'(',s)
  decls.append(('static 'if static else '')+'__forceinline void '+name+'('+params+');')
  defs.append('__forceinline void AptMpGameSetup::'+name+'('+params+') {'+s+'\n}')
  body=body.replace(segments[k],'\n\t'+name+'('+args+');')
 head=head.replace('\tvoid rva004404AC(bool enable, int slot);','\tvoid rva004404AC(bool enable, int slot);\n'+ '\n'.join(decls))
 if placement:return head+MARK+body+'\n'+'\n'.join(defs)+'\n'
 return head+'\n'+'\n'.join(defs)+'\n'+MARK+body
if __name__=='__main__':run([('reg-%04d'%n,source(n)) for n in range(1024)],'regions')
