{

 if((ah|bh)==0)return __emulu(al,bl);
unsigned cross=(unsigned)__emulu(ah,bl);cross+=(unsigned)__emulu(al,bh);U64 low=__emulu(al,bl);return low+((U64)cross<<32);
}
