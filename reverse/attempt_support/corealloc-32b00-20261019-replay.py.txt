from pathlib import Path
import sys
sys.path.insert(0, 'tools')
import build
root = Path.cwd()
out = root / 'build/replay-corealloc-32b00'
out.mkdir(parents=True, exist_ok=True)
bank = (root / 'reverse/attempts/0x00032b00.cpp').read_text(encoding='utf-8')
old = bank
for field in ['m_releaseDirect', 'm_unknownF', 'm_releaseAllowed']:
    old = old.replace('*(volatile unsigned char *)&core->' + field + ' =', 'core->' + field + ' =')
ref = bank.replace(
    'const unsigned int &chunkSize = (unsigned int)core - (unsigned int)chunk + coreSize;\n\t\tchunk[1] = chunkSize | chunkFlags;',
    'const unsigned int &chunkSize = (chunk[1] = (unsigned int)core - (unsigned int)chunk + coreSize);\n\t\tchunk[1] = chunkSize | chunkFlags;')
symbol = '?rva00032B00@GeneralAllocator@Allocator@EA@@QAEPAXI@Z'
symbols = build.load_symbol_map()
for label, source in [('old-prefix', old), ('ordered-prefix', bank), ('extra-store-diagnostic', ref)]:
    path = out / (label + '.cpp')
    path.write_text(source, encoding='utf-8', newline='\n')
    obj = path.with_suffix('.obj')
    ok, message, _ = build.try_compile_source(path, obj)
    if not ok:
        raise SystemExit(message)
    row = dict(name=symbol, target_rva='0x00032B00', target_size='343',
               source=path.relative_to(root).as_posix(), notes='')
    patch = build.compile_function(row, symbols, obj)
    raw, _ = build.read_object_symbol_bytes(obj, symbol, 343)
    print(label, 'size', len(raw), 'exact', len(raw) == 343 and patch['bytes'] == patch['target'],
          'unresolved', patch['unresolved'])
