# Every family starts from the three-row Rotate_X winner.
for i in range(3):
    h.BASE=h.BASE.replace('Row[%d][1] = (float)( c*tmp1 + s*tmp2);'%i,'Row[%d][1] = (float)( c*tmp1 + (s*tmp2));'%i)


# matrix
def matrix():
    for bits in itertools.product(range(2),repeat=10):
        s=h.BASE
        a=s.index('    __forceinline void mulVector3');b=s.index('    __forceinline void mulCameraAngleVector',a);c=s.index('    __forceinline void Rotate_X',b)
        p=s[a:b];q=s[b:c]
        for i,(r,k) in enumerate(itertools.product(range(3),range(2))):
            term='Row[%d][%d]*v.%s'%(r,k,'XYZ'[k])
            if bits[i]:p=p.replace(term,'('+term+')')
        for i in range(2):
            r=i+1
            if bits[6+i]:q=q.replace('(Row[%d][0]*v.X+Row[%d][1]*v.Y)'%(r,r),'Row[%d][0]*v.X+Row[%d][1]*v.Y'%(r,r))
        if bits[8]:p=re.sub(r'(float [xyz]=)([^;]+);',r'\1(\2);',p)
        if bits[9]:
            lines=q.splitlines();ix=next(i for i,l in enumerate(lines) if 'float x='in l);iy=next(i for i,l in enumerate(lines) if 'float y='in l);lines[ix],lines[iy]=lines[iy],lines[ix];q='\n'.join(lines)+'\n'
        yield 'matrix_'+''.join(map(str,bits)),s[:a]+p+q+s[c:]
h.matrix=matrix

# valuepaths
def valuepaths():
    ts=[
      'p.X+=v->pos.x;p.Y+=v->pos.y;return p;',
      'return Vector3(p.X+v->pos.x,p.Y+v->pos.y,p.Z);',
      'Vector3 result(p);result.X+=v->pos.x;result.Y+=v->pos.y;return result;',
      'float x=p.X+v->pos.x;float y=p.Y+v->pos.y;return Vector3(x,y,p.Z);',
      'p.X=(p.X+v->pos.x);p.Y=(p.Y+v->pos.y);return p;',
      'p.X=(p.X)+v->pos.x;p.Y=(p.Y)+v->pos.y;return p;',
      'p.X=p.X+(v->pos.x);p.Y=p.Y+(v->pos.y);return p;',
      'p.X=v->pos.x+p.X;p.Y=v->pos.y+p.Y;return p;',
    ]
    ps=[
      's.X=(s.X-t.X)/p+t.X;s.Y=(s.Y-t.Y)/p+t.Y;return s;',
      'return Vector3((s.X-t.X)/p+t.X,(s.Y-t.Y)/p+t.Y,s.Z);',
      'float x=(s.X-t.X)/p+t.X;float y=(s.Y-t.Y)/p+t.Y;return Vector3(x,y,s.Z);',
      's.X=((s.X-t.X)/p)+t.X;s.Y=((s.Y-t.Y)/p)+t.Y;return s;',
      's.X=((s.X-t.X)/p+t.X);s.Y=((s.Y-t.Y)/p+t.Y);return s;',
      's.X=t.X+((s.X-t.X)/p);s.Y=t.Y+((s.Y-t.Y)/p);return s;',
      'float inversePitch=1.0f/p;s.X=(s.X-t.X)*inversePitch+t.X;s.Y=(s.Y-t.Y)*inversePitch+t.Y;return s;',
      'Vector3 result(s);result.X=(s.X-t.X)/p+t.X;result.Y=(s.Y-t.Y)/p+t.Y;return result;',
    ]
    for t,p,ref,out in itertools.product(range(len(ts)),range(len(ps)),range(2),range(2)):
        s=h.BASE.replace(ts[0],ts[t]).replace(ps[0],ps[p])
        if ref:
            s=s.replace('translated(Vector3 p,','translated(const Vector3 &in,')
            s=s.replace('const CameraTransformInput *v) {','const CameraTransformInput *v) {Vector3 p(in);')
            s=s.replace('pitchCamera(Vector3 s,','pitchCamera(const Vector3 &in,').replace('const Vector3 &t,float p) {','const Vector3 &t,float p) {Vector3 s(in);')
        if out:
            # Actual Vector3 Set(const Vector3&) copy spelling for returned values.
            s=s.replace('\tvoid Set(Real x, Real y, Real z)','\t__forceinline void Set(const Vector3 &v) {X=v.X;Y=v.Y;Z=v.Z;}\n\tvoid Set(Real x, Real y, Real z)')
            for var in ['sourcePos','targetPos']:s=s.replace(var+'=translated('+var+',v);',var+'.Set(translated('+var+',v));')
            s=s.replace('sourcePos=pitchCamera(sourcePos,targetPos,v->fxPitch);','sourcePos.Set(pitchCamera(sourcePos,targetPos,v->fxPitch));')
        yield 'valuepaths_%d%d%d%d'%(t,p,ref,out),s
h.valuepaths=valuepaths

# ctor
def ctor():
    terms=['a.Z*s','a.Y*s','a.Z*s','a.X*s','a.Y*s','a.X*s']
    for bits in itertools.product(range(2),repeat=10):
        s=h.BASE
        a=s.index('    __forceinline Matrix3D(const Vector3 &a, float angle)');b=s.index('    __forceinline Matrix3D &operator=',a)
        part=s[a:b]
        # Six independently grouped off-diagonal sine products, in row order.
        lines=part.splitlines();j=0
        for i,line in enumerate(lines):
            if '(float)(' in line and '*s' in line:
                if bits[j]:lines[i]=line.replace(terms[j],'('+terms[j]+')')
                j+=1
        assert j==6
        part='\n'.join(lines)+'\n'
        if bits[6]:part=part.replace('cosf(angle);','(cosf(angle));')
        if bits[7]:part=part.replace('sinf(angle);','(sinf(angle));')
        if bits[8]:part=part.replace('        Row[0].Set(','        float oneMinusC=1.0f-c;\n        Row[0].Set(').replace('(1.0f - c)','oneMinusC')
        if bits[9]:
            x=part.index('        Row[0].Set(')
            if bits[8]:x=part.index('        float oneMinusC')
            part=part[:x]+'        Set(a,s,c);\n    }\n    __forceinline void Set(const Vector3 &a,float s,float c) {\n'+part[x:]
        yield 'ctor_'+''.join(map(str,bits)),s[:a]+part+s[b:]
h.ctor=ctor

# refs
def refs():
    exprs=['sourcePos.X','sourcePos.Y','targetPos.X','targetPos.Y']
    names=['sourceX','sourceY','targetX','targetY']
    for mask,scope,pitch in itertools.product(range(16),range(4),range(4)):
        s=h.BASE
        selected=[i for i in range(4) if mask>>i&1]
        decl=''.join('float &%s=%s;'%(names[i],exprs[i]) for i in selected)
        call='sourcePos=translated(sourcePos,v);targetPos=translated(targetPos,v);'
        body=''
        for i in range(4):
            if i%2==0 and not any(x in selected for x in [i,i+1]):
                v=exprs[i].split('.')[0];body+=v+'=translated('+v+',v);';continue
            if i%2 and not any(x in selected for x in [i-1,i]):continue
            body+=(names[i] if i in selected else exprs[i])+'+=v->pos.'+('x' if i%2==0 else 'y')+';'
        if scope==0:s=s.replace('Vector3 sourcePos,targetPos;','Vector3 sourcePos,targetPos;'+decl).replace(call,body)
        elif scope==1:s=s.replace('    // translate to current XY position','    '+decl+'\n    // translate to current XY position').replace(call,body)
        elif scope==2:s=s.replace(call,'{'+decl+body+'}')
        else:
            s=s.replace('static __forceinline Vector3 translated(Vector3 p,const CameraTransformInput *v) {p.X+=v->pos.x;p.Y+=v->pos.y;return p;}',
                        'static __forceinline Vector3 translated(Vector3 p,const CameraTransformInput *v) {float &x=p.X;float &y=p.Y;x+=v->pos.x;y+=v->pos.y;return p;}')
        if pitch==1:s=s.replace('s.X=(s.X-t.X)/p+t.X;s.Y=(s.Y-t.Y)/p+t.Y;return s;', 'float &x=s.X;float &y=s.Y;const float &tx=t.X;const float &ty=t.Y;x=(x-tx)/p+tx;y=(y-ty)/p+ty;return s;')
        if pitch==2:s=s.replace('sourcePos=pitchCamera(sourcePos,targetPos,v->fxPitch);','{float &sx=sourcePos.X;float &sy=sourcePos.Y;const float &tx=targetPos.X;const float &ty=targetPos.Y;sx=(sx-tx)/v->fxPitch+tx;sy=(sy-ty)/v->fxPitch+ty;}')
        if pitch==3:s=s.replace('sourcePos=pitchCamera(sourcePos,targetPos,v->fxPitch);','{const float &p=v->fxPitch;sourcePos=pitchCamera(sourcePos,targetPos,p);}')
        yield 'refs_%02d%d%d'%(mask,scope,pitch),s
h.refs=refs

# indexed
def indexed():
    for bits in itertools.product(range(2),repeat=9):
        s=h.BASE.replace('\tvoid Set(Real x, Real y, Real z)','\t__forceinline float &operator[](int i){return (&X)[i];}\n\t__forceinline const float &operator[](int i)const{return (&X)[i];}\n\tvoid Set(Real x, Real y, Real z)')
        for i,var in enumerate(['sourcePos','targetPos','sourcePos','targetPos']):
            token=var+'=translated('+var+',v);'
            # replace only this call, retaining the other initial/live calls.
            nth=i//2
            starts=[m.start() for m in re.finditer(re.escape(token),s)]
            # Earlier replacements remove the initial call of that variable.
            idx=0 if (nth==0 or bits[i-2]) else 1
            if bits[i]:
                a=starts[idx];s=s[:a]+var+'[0]+=v->pos.x;'+var+'[1]+=v->pos.y;'+s[a+len(token):]
        if bits[4]:s=s.replace('sourcePos=pitchCamera(sourcePos,targetPos,v->fxPitch);','sourcePos[0]=(sourcePos[0]-targetPos[0])/v->fxPitch+targetPos[0];sourcePos[1]=(sourcePos[1]-targetPos[1])/v->fxPitch+targetPos[1];')
        if bits[5]:
            a=s.index('    } else {\n        sourcePos.X=(sourcePos.X-targetPos.X)/v->fxPitch')
            b=s.index('    if(v->positionHold)',a)
            p=s[a:b]
            for v in ['sourcePos','targetPos']:
                for i,axis in enumerate('XY'):p=p.replace(v+'.'+axis,v+'['+str(i)+']')
            s=s[:a]+p+s[b:]
        if bits[6]:
            # Remaining real component operations in the main body only.
            a=s.index('void W3DCamTransform::rva00102F83')
            p=s[a:]
            for v in ['sourcePos','targetPos']:
                for i,axis in enumerate('XYZ'):p=p.replace(v+'.'+axis,v+'['+str(i)+']')
            s=s[:a]+p

            for v in ['s','t','p']:
                for i,axis in enumerate('XY'):p=p.replace(v+'.'+axis,v+'['+str(i)+']')
            s=s[:a]+p+s[b:]
        if bits[8]:
            a=s.index('    __forceinline void mulVector3');b=s.index('    __forceinline void mulCameraAngleVector',a)
            s=s[:a]+re.sub(r'(float [xyz]=)([^;]+);',r'\1(\2);',s[a:b])+s[b:]

# scalar_lifetime
def scalar_lifetime():
    for trans,pitch,scope,chain,group,matrix in itertools.product(range(5),range(4),range(2),range(2),range(2),range(2)):
        s=h.BASE;decl=[]
        names={1:['xValue','yValue','xValue','yValue'],2:['sxValue','syValue','txValue','tyValue'],3:['xyValue']*4,4:['localX','localY','localX','localY']}
        def store(var,axis,name,rhs):
            if group:rhs='('+rhs+')'
            return var+'.'+axis+'='+name+'='+rhs+';' if chain else name+'='+rhs+';'+var+'.'+axis+'='+name+';'
        if trans:
            n=names[trans]
            if trans!=4:decl+=list(dict.fromkeys(n))
            for i,var in enumerate(['sourcePos','targetPos']):
                p=''.join(store(var,axis,n[2*i+j],var+'.'+axis+'+v->pos.'+'xy'[j]) for j,axis in enumerate('XY'))
                if trans==4:p='{float localX,localY;'+p+'}'
                s=s.replace(var+'=translated('+var+',v);',p)
        if pitch:
            if pitch==1:pn=['pitchX','pitchY'];decl+=pn
            if pitch==2:
                pn=names[trans][:2] if trans in [1,2,3] else ['xValue','yValue'];decl+=pn
            if pitch==3:pn=['pitchX','pitchY']
            p=''.join(store('sourcePos',axis,pn[j],'(sourcePos.'+axis+'-targetPos.'+axis+')/v->fxPitch+targetPos.'+axis) for j,axis in enumerate('XY'))
            if pitch==3:p='{float pitchX,pitchY;'+p+'}'
            s=s.replace('sourcePos=pitchCamera(sourcePos,targetPos,v->fxPitch);',p)
        if decl:
            d='float '+','.join(dict.fromkeys(decl))+';'
            if scope:s=s.replace('    // translate to current XY position','    '+d+'\n    // translate to current XY position')
            else:s=s.replace('    Vector3 sourcePos,targetPos;','    Vector3 sourcePos,targetPos;\n    '+d)
        if matrix:
            a=s.index('    __forceinline void mulVector3');b=s.index('    __forceinline void mulCameraAngleVector',a)
            s=s[:a]+re.sub(r'(float [xyz]=)([^;]+);',r'\1(\2);',s[a:b])+s[b:]
        yield 'scalar_lifetime_'+''.join(map(str,[trans,pitch,scope,chain,group,matrix])),s
h.scalar_lifetime=scalar_lifetime

# trig
def trig():
    def exp(n,f):
        return [f+'f(angle)',f+'(angle)','(float)'+f+'((double)angle)','((float)'+f+'((double)angle))','(float)('+f+'((double)angle))','((float)('+f+'((double)angle)))','('+f+'f(angle))','(('+f+'f(angle)))'][n]
    for co,si,forward,matrix in itertools.product(range(8),range(8),range(2),range(2)):
        s=h.BASE
        a=s.index('    __forceinline Matrix3D(const Vector3');b=s.index('    __forceinline Matrix3D &operator=',a)
        p=s[a:b].replace('cosf(angle)',exp(co,'cos')).replace('sinf(angle)',exp(si,'sin'))
        if forward:
            p=p.replace('__forceinline Matrix3D(const Vector3 &a, float angle) {','__forceinline Matrix3D(const Vector3 &a, float angle) {Set(a,angle);}\n    __forceinline void Set(const Vector3 &a,float angle) {')
            p=p.replace('        Row[0].Set(','        Set(a,s,c);\n    }\n    __forceinline void Set(const Vector3 &a,float s,float c) {\n        Row[0].Set(',1)
        s=s[:a]+p+s[b:]
        if matrix:
            a=s.index('    __forceinline void mulVector3');b=s.index('    __forceinline void mulCameraAngleVector',a)
            s=s[:a]+re.sub(r'(float [xyz]=)([^;]+);',r'\1(\2);',s[a:b])+s[b:]
        yield 'trig_'+''.join(map(str,[co,si,forward,matrix])),s
