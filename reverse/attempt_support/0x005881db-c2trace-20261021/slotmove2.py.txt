import sys,os
sys.path.insert(0,'/work2/Open-BFME-2/build/c2tr')
from wdbg import Dbg
# usage: slotmove.py SRC OBJ "X>Y;..." [nocolor-var@round ...]
# moves list entry X to directly after Y in the frame-slot list (idx or name; name:idx)
src,obj,moves=sys.argv[1:4]; forces=sys.argv[4:]
d=Dbg(src,obj,obj+'.log')
d.prompt();d.prompt('set $CanDeferOnBPByAddr = 1')
bps=[0x83acd,0x82978,0x82273,0x886f6]; added=set()
for b in bps:
    o=d.prompt('break *0x%x'%(0x10700000+b))
    if 'Breakpoint' in o and 'Unable' not in o: added.add(b)
names={}
def nm(sym):
    if sym in names: return names[sym]
    sw=d.words(sym,1); pw=d.words(sw[0],2) if sw and sw[0]>0x100000 else []
    n=d.string(pw[1]).strip() if len(pw)>1 and pw[1]>0x100000 else '?'
    n=n.strip('"').lstrip('_')
    names[sym]=n; return n
fv={};dec={}
for f in forces:
    if f.startswith('dec='):
        v,c=f[4:].split(':'); dec[v]=int(c); continue
    v,r=f.split('@'); off,val=0x24,1
    if ':' in r: r,ov=r.split(':'); off,val=(int(x,0) for x in ov.split('='))
    fv.setdefault(int(r),[]).append((v,off,val))
curfn='';curcls=-1;k=0;phase=0;done=False;final=None
def key(p):
    w=d.words(p,8); return '%s:%x'%(nm(p),w[7])
while True:
    out=d.prompt('cont')
    if 'terminated' in out: break
    eip=d.reg('eip')-0x10700000
    for b in bps:
        if b not in added and b!=eip: d.prompt('break *0x%x'%(0x10700000+b)); added.add(b)
    added.add(eip)
    if eip==0x83acd:
        ebp=d.reg('ebp'); f=d.words(ebp,1)[0]; curfn=d.string(d.words(f+4,1)[0]); curcls=d.reg('ebx'); k=0; continue
    if 'updateMeleeAttack' not in curfn: continue
    if eip==0x82273 and curcls==0:
        phase=1
        if dec:
            d.prompt('break *0x%x'%(0x10700000+0x89a9e)); added.add(0x89a9e); bps.append(0x89a9e)
        continue
    if eip==0x89a9e and phase==1 and not done:
        s_=d.reg('eax'); n_=nm(s_)
        if os.environ.get('PW') and n_ in os.environ['PW'].split(','): print('W',n_,hex(s_),d.reg('ebx'),d.reg('edx'),flush=True)
        if dec.get(n_,0)>0:
            dec[n_]-=1; d.prompt('set $edx = $edx - 1'); print('dec',n_,flush=True)
        continue
    if eip==0x82978 and curcls==0 and phase==0:
        k+=1
        for v,off,val in fv.get(k,[]):
            for b in d.words(0x108807b8,1024):
                p=b
                while p:
                    w=d.words(p,18)
                    sw=d.words(w[0],1); pw=d.words(sw[0],2) if sw and sw[0]>0x100000 else []
                    n=d.string(pw[1]).strip().strip('"').lstrip('_') if len(pw)>1 and pw[1]>0x100000 else '?'
                    if (w[9]>=2 or off!=0x24) and n==v and (not os.environ.get('MINIDX') or w[7]>=int(os.environ['MINIDX'],16)) and (not os.environ.get('IDX') or w[7]==int(os.environ['IDX'],16)):
                        d.prompt('set *(int*)0x%x = %d'%(p+off,val)); print('round',k,'set',v,hex(w[7]),hex(off),val,flush=True)
                    p=w[12]
        continue
    if eip==0x886f6 and phase==1 and not done:
        done=True
        head=d.reg('ebx')
        lst=[];p=head
        while p:
            lst.append(p); p=d.words(p+0x30,1)[0]
        keys=[key(p) for p in lst]
        print('BEFORE',' '.join(keys),flush=True)
        import os
        if os.environ.get('DUMPSYM'):
            for p_,k_ in zip(lst,keys):
                w=d.words(p_,24); print('SYM %-18s %s'%(k_,' '.join('%x'%v for v in w)),flush=True)
        def find(s):
            for i,kk in enumerate(keys):
                if kk==s or kk.split(':')[0]==s: return i
            raise SystemExit('missing '+s)
        for mv in moves.split(';'):
            if not mv: continue
            x,y=mv.split('>')
            i=find(x); e=lst.pop(i); ke=keys.pop(i)
            j=find(y); lst.insert(j+1,e); keys.insert(j+1,ke)
        print('AFTER ',' '.join(keys),flush=True)
        for i,p in enumerate(lst):
            nx=lst[i+1] if i+1<len(lst) else 0
            pv=lst[i-1] if i>0 else 0
            d.prompt('set *(int*)0x%x = 0x%x'%(p+0x30,nx)); d.prompt('set *(int*)0x%x = 0x%x'%(p+0x34,pv))
        if lst[0]!=head:
            d.prompt('set $ebx = 0x%x'%lst[0])
            hp=d.words(0x108a78a4,1)[0]
            if hp==head: d.prompt('set *(int*)0x108a78a4 = 0x%x'%lst[0])
        d.prompt('break *0x%x'%(0x10700000+0xb2b47)); added.add(0xb2b47); bps.append(0xb2b47); final=lst
        continue
    if eip==0xb2b47 and done and final:
        out=[]
        for p in final:
            w=d.words(p,16); out.append('%s=%x'%(key(p),w[9]&0xffff))
        print('OFF',' '.join(out),flush=True); final=None
print('done',flush=True)
