from pathlib import Path
import sys,subprocess,json,re
sys.path.insert(0,'tools');import build
out=Path('build/resume-eh-local');base=Path('reverse/attempts/0x002e3aa8.cpp').read_text().replace('// cl: /O1','// cl: /O1 /G7',1)
sym='?adjustCameraMovement@PolygonTrigger@@QAE_NPBUCoord3D@@0PAU2@_N@Z';rva=0x2e3aa8;size=723
dot='along.x*delta.x+along.y*delta.y'
variants={'baseline':base,
 'dot-products':base.replace(dot,'(along.x*delta.x)+(along.y*delta.y)'),
 'dot-memory-leaves':base.replace(dot,'along.x*(delta.x)+along.y*(delta.y)'),
 'dot-live-leaves':base.replace(dot,'(along.x)*delta.x+(along.y)*delta.y'),
 'dot-result':base.replace(dot,'(('+dot+'))')}
for kind,body in [('plain','x+=p->x;y+=p->y;z+=p->z;'),('receiver','x=(x)+p->x;y=(y)+p->y;z=(z)+p->z;'),('argument','x=x+(p->x);y=y+(p->y);z=z+(p->z);'),('result','x=(x+p->x);y=(y+p->y);z=(z+p->z);')]:
 decl='struct CameraNext54 : Coord3D { CameraNext54(const Coord3D &v):Coord3D(v) {} __forceinline void add(const Coord3D *p){'+body+'} };\n'
 s=base.replace('bool PolygonTrigger::adjustCameraMovement',decl+'bool PolygonTrigger::adjustCameraMovement',1).replace('Coord3D next(candidate);','CameraNext54 next(candidate);')
 variants['tail-'+kind]=s
variants['cast-products']=base.replace(dot,'((float)(along.x*delta.x))+((float)(along.y*delta.y))')
variants['cast-leaves']=base.replace(dot,'along.x*((float)(delta.x))+along.y*((float)(delta.y))')
variants['cast-remaining']=base.replace('float remaining=deltaLen-usedLen;','float remaining=((float)(deltaLen-usedLen));').replace('remaining*=0.4f;','remaining=((float)(remaining*0.4f));')
ops=base.replace('along.sub(&cur);','along-=cur;').replace('along.scale(-1.0f);','along*=-1.0f;').replace('used.sub(start);','used-=*start;').replace('along.scale(remaining);','along*=remaining;').replace('next.add(&along);','next+=along;')
variants['operators']=ops
variants['operators-cast-products']=ops.replace(dot,'((float)(along.x*delta.x))+((float)(along.y*delta.y))')
if len(sys.argv)>1:variants={k:v for k,v in variants.items() if sys.argv[1] in k}
retail=build.read_target_bytes(rva,size);results=[]
for name,src in variants.items():
 p=out/('turn54-camera-g7-'+name+'.cpp');p.write_bytes(src.encode())
 proc=subprocess.run([sys.executable,'tools/align_diff.py',str(p),sym,hex(rva),str(size)],stdout=subprocess.PIPE,stderr=subprocess.STDOUT,text=True)
 p.with_suffix('.txt').write_bytes(proc.stdout.encode())
 try:
  raw,relocs=build.read_object_symbol_bytes(build.obj_path(p),sym,code_only=True);mask=set()
  for off,kind,target in relocs:mask.update(range(off,off+4))
  diffs=[i for i,(a,b) in enumerate(zip(raw,retail)) if i not in mask and a!=b]
  row={'name':name,'extent':len(raw),'fixed_diff':len(diffs)+abs(len(raw)-size),'first':diffs[:8],'structural':re.search(r'== structural regions: (\d+)',proc.stdout)[1]}
 except Exception as e:row={'name':name,'exit':proc.returncode,'error':str(e)}
 print(json.dumps(row),flush=True);results.append(row)
 (out/('turn54-camera-g7-results-'+(sys.argv[1] if len(sys.argv)>1 else 'all')+'.json')).write_bytes(json.dumps(results,indent=2).encode())
