 '?TheGameLogic@@3PAVGameLogic@@A':0xdfe78c,

def run(item):
    key,source=item
    path=OUT/(key+'.cpp');obj=path.with_suffix('.obj')
    path.write_text(source)
    command,env=build.compiler_command(path,obj)
    if any(f.lower() in ('-zi','/zi') for f in command):
        command.insert(-1,'-Fd'+path.with_suffix('.pdb').relative_to(ROOT).as_posix())
    p=subprocess.run(command,cwd=ROOT,env=env,capture_output=True,text=True)
    if p.returncode:
        (OUT/(key+'.err')).write_text(p.stdout+p.stderr)
        return dict(key=key,error=(p.stdout+p.stderr)[-350:])
    raw,relocs=build.read_object_symbol_bytes(obj,SYMBOL,code_only=True)
    raw=raw.rstrip(b'\xcc')
    patched=bytearray(raw)
    unresolved=[]
    for off,typ,name in relocs:
        if off+4>len(raw):continue
        if typ==20:
            candidates=MAP.get(name)
            if candidates:
                patched[off:off+4]=struct.pack('<i',candidates[0]-(0x596e15+off+4))
            else:unresolved.append(name)
        elif typ==6:
            if name in DIR_ADDR:patched[off:off+4]=struct.pack('<I',DIR_ADDR[name])
            else:unresolved.append(name)

    md=capstone.Cs(capstone.CS_ARCH_X86,capstone.CS_MODE_32)
    a=[(i.mnemonic,i.op_str) for i in md.disasm(patched,0)]
    b=[(i.mnemonic,i.op_str) for i in md.disasm(TARGET,0)]
    # Relocations and branch distances are excluded from the instruction score.
    def norm(x):
        m,op=x
        if m=='call' and not 'ptr' in op:op='CALL'
        if m.startswith('j'):op='BRANCH'
        op=re.sub(r'0x[0-9a-f]{6,}','ADDR',op)
        return m,op
    na=list(map(norm,a));nb=list(map(norm,b))
    nd=sum(max(i2-i1,j2-j1) for tag,i1,i2,j1,j2 in difflib.SequenceMatcher(None,na,nb,autojunk=False).get_opcodes() if tag!='equal')
    ratio=lambda x,y:difflib.SequenceMatcher(None,x,y,autojunk=False).ratio()
    score=0.5*ratio([x[0] for x in a],[x[0] for x in b])+0.3*ratio(a,b)+0.2*ratio(patched,TARGET)
    canonical=bytearray(raw)
    for off,_,_ in relocs:
        if off+4<=len(canonical):canonical[off:off+4]=b'\0'*4
    d=dict(key=key,size=len(raw),nd=nd,score=round(score,6),
           exact_candidate=bytes(patched)==TARGET and not unresolved,
           masked_code_sha256=hashlib.sha256(canonical).hexdigest(),unresolved=unresolved)

    (OUT/(key+'.asm')).write_text('\n'.join(f'{i.address:03x} {i.mnemonic} {i.op_str}' for i in md.disasm(patched,0)))
    return d
def batch(items,label):
    items=list(items)
    with concurrent.futures.ThreadPoolExecutor(max_workers=8) as pool:
        results=list(pool.map(run,items))
    (OUT/(label+'.json')).write_text(json.dumps(results,indent=2))
    for d in sorted(results,key=lambda d:(d.get('nd',999),-d.get('score',0)))[:16]:print(json.dumps(d),flush=True)
    print('tested',len(results),'compile errors',sum('error' in d for d in results),flush=True)
    for d in results:
        if d.get('exact_candidate'):print('CANDIDATE; normal full gates still required',d['key'],flush=True)
