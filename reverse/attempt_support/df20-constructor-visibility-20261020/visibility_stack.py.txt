from pathlib import Path
import sys,itertools,concurrent.futures,json
sys.path.insert(0,'build/df20-scalars-20261020');import trials
trials.scratch=Path.cwd()/'build/df20-scalars-20261020'
base=(trials.scratch/'range-x0-specialized-b1-r0-ttemporary.cpp').read_text();trials.symbol=base.splitlines()[0][3:]
wide='template <> _String_base<wchar_t,allocator<wchar_t> >::_String_base(const allocator<wchar_t>&);'
copy='template <> basic_string<wchar_t>::basic_string(const basic_string<wchar_t>&);'
specs=[]
for loc,nt,narrow,copyvis,tag in itertools.product(['plain','const','ref'],[0,1],[0,1],[0,1],[0,1]):
 s=base
 if loc=='const':s=s.replace('  locale __loc =','  const locale __loc =')
 if loc=='ref':s=s.replace('  locale __loc =','  const locale& __loc =')
 if nt:s=s.replace(wide,wide[:-1]+' throw();')
 if narrow:s=s.replace(wide,wide+'\ntemplate <> _String_base<char,allocator<char> >::_String_base(const allocator<char>&);')
 if copyvis:s=s.replace(copy,'')
 if tag:
  # Expose the source tag itself rather than an empty derived tag, preserving
  # __copy's selected random-access overload and runtime semantics.
  s=s.replace('struct _MoneySignPolicy : random_access_iterator_tag {};','typedef random_access_iterator_tag _MoneySignPolicy;')
 specs.append((f'visstack-l{loc}-n{nt}-c{narrow}-v{copyvis}-t{tag}',s))
with concurrent.futures.ThreadPoolExecutor(max_workers=8) as p:r=list(p.map(trials.run,specs))
(trials.scratch/'visibility_stack.json').write_text(json.dumps(r,indent=2))
