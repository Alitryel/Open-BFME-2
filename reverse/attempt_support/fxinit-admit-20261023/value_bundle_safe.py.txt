# Corrected hoist subset: avoid forming out-of-arm pointers or subtracting nulls.
import sys,json,time
from pathlib import Path
from concurrent.futures import ThreadPoolExecutor,as_completed
sys.path.insert(0,'build/fxinit-admit23');import value_bundle as v
h=v.h
def variants():
 for label,source in v.variants():
  keys=label.split('_')[1];shape,order,ctor,byref,local,placement,sgn=map(int,keys)
  if not placement or (ctor and not local):continue
  if shape==0:
   # Retain the original current-vertex view only in its valid arm;
   # the hoisted view contains base pointers and consumes i at the call.
   source=source.replace('links[0]+i','links[0]').replace('links[1]+i','links[1]')
   source=source.replace('v.map[*v.a]+(v.map[*v.b]<<8)','v.map[v.a[i]]+(v.map[v.b[i]]<<8)')
   source=source.replace('v){return','v,unsigned i){return').replace('PackInputValue(packInput);','PackInputValue(packInput,i);')
  else:source=source.replace('links[1]-links[0]','(links[1] ? links[1]-links[0] : 0)')
  yield label.replace('valuebundle_','valuebundlesafe_'),source
if __name__=='__main__':
 t=time.time();rs=[]
 with ThreadPoolExecutor(max_workers=8)as ex:
  for f in as_completed([ex.submit(h.run_one,x)for x in variants()]):rs.append(f.result())
 rs.sort(key=lambda z:(z.get('nd',99999),z.get('cost',99999)))
 (h.HERE/'valuebundlesafe-results.json').write_text(json.dumps(rs,indent=2))
 print(json.dumps({'count':len(rs),'seconds':time.time()-t,'errors':sum('error'in z for z in rs),'best':rs[:6]},indent=2))
