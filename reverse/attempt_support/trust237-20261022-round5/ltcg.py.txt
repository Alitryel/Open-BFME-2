from pathlib import Path
import concurrent.futures, hashlib, itertools, json, re, shutil, struct, subprocess, sys
sys.path.insert(0, 'tools')
import build, compiler_profiles, capstone, pefile
import search

ROOT=Path.cwd();OUT=search.OUT
sdk=build.vc71_root()/'Vc7/PlatformSDK/Lib'
profiles={'vc71-sp1':compiler_profiles.SP1_FILES,'vc71-ddk-4035':compiler_profiles.DDK4035_FILES}
tool_inputs={}
for profile,files in profiles.items():
    lane=OUT/('ltcg-'+profile);lane.mkdir(exist_ok=True)
    for name,expected in files.items():
        source=ROOT/'vendor'/profile/'bin'/name
        assert hashlib.sha256(source.read_bytes()).hexdigest()==expected
        shutil.copy2(source,lane/name)
    shutil.copy2(build.vc71_root()/'Vc7/bin/link.exe',lane/'link.exe')
    tool_inputs[profile]={name:hashlib.sha256((lane/name).read_bytes()).hexdigest() for name in list(files)+['link.exe']}

targetpe=pefile.PE('baselines/bfme2/workshop-vanilla-1.06/files/game.dat')
targetimports={i.name.decode():i.address for dll in targetpe.DIRECTORY_ENTRY_IMPORT for i in dll.imports if i.name}
targets={'memset':0x6291ae,'WinVerifyTrust':0x655652}

def run(job):
    n,profile,opt,cpu,shape=job
    name=f'g{n:03}';src=OUT/(name+'.cpp');obj=src.with_suffix('.obj')
    body=search.base if shape==0 else (OUT/'bank.cpp').read_text()
    if shape==2:
        body=body.replace('int r=0; if(((u&1u)<=1u)) goto yes; return r; yes:return r+1;', 'return (unsigned __int64)(unsigned char)u<=255ui64;')
    assert ('unsigned __int64' in body)==(shape==2)
    body=re.sub(r'// cl:.*',f'// cl: {opt} /G{cpu} /GL /arch:SSE /DNDEBUG /MD /EHsc',body)
    src.write_text(f'// compiler: {profile}\n'+body)
    cmd,env=build.compiler_command(src,obj)
    cp=subprocess.run(cmd,cwd=ROOT,env=env,capture_output=True,text=True)
    result=dict(name=name,profile=profile,opt=opt,cpu=cpu,shape=shape,compile_returncode=cp.returncode,compile_log=cp.stdout+cp.stderr)
    if cp.returncode:return result
    rel=str(src.relative_to(ROOT).with_suffix(''))
    cmd=['wine',str(OUT/('ltcg-'+profile)/'link.exe'),'/nologo','/DLL','/NOENTRY','/INCREMENTAL:NO','/LTCG','/NODEFAULTLIB','/EXPORT:'+search.NAME,'/OUT:'+rel+'.dll','/MAP:'+rel+'.map',str(obj.relative_to(ROOT)),'msvcrt.lib']
    cmd += [str((sdk/x).relative_to(ROOT)) for x in ['Kernel32.Lib','Shell32.Lib','WinTrust.Lib']]
    lp=subprocess.run(cmd,cwd=ROOT,env=env,capture_output=True,text=True)
    result.update(link_returncode=lp.returncode,link_log=lp.stdout+lp.stderr)
    if lp.returncode:return result
    pe=pefile.PE(str(src.with_suffix('.dll')));base=pe.OPTIONAL_HEADER.ImageBase
    start=next(e.address for e in pe.DIRECTORY_ENTRY_EXPORT.symbols if e.name.decode()==search.NAME)
    lines=src.with_suffix('.map').read_text().splitlines()
    addresses=[int(m.group(1),16)-base for l in lines if (m:=re.match(r'\s+0001:[0-9a-fA-F]+\s+\S+\s+([0-9a-fA-F]{8})\s',l))]
    end=min(a for a in addresses if a>start)
    raw=pe.get_data(start,end-start).rstrip(b'\xcc')
    md=capstone.Cs(capstone.CS_ARCH_X86,capstone.CS_MODE_32);md.detail=True
    instructions=list(md.disasm(raw,base+start))
    assert instructions[-1].mnemonic=='ret', [(i.mnemonic,i.op_str) for i in instructions[-5:]]
    imports={i.address:i.name.decode() for dll in pe.DIRECTORY_ENTRY_IMPORT for i in dll.imports if i.name}
    norm=bytearray(raw);unresolved=[]
    for ins in instructions:
        if ins.mnemonic!='call':continue
        offset=ins.address-base-start
        if ins.operands[0].type==capstone.x86.X86_OP_MEM:
            addr=ins.operands[0].mem.disp;imp=imports.get(addr)
            if imp in targetimports:norm[offset+2:offset+6]=struct.pack('<I',targetimports[imp])
            else:unresolved.append(hex(ins.address))
        else:
            thunk=pe.get_data(ins.operands[0].imm-base,6)
            imp=imports.get(struct.unpack('<I',thunk[2:])[0]) if thunk[:2]==b'\xff\x25' else None
            if imp in targets:norm[offset+1:offset+5]=struct.pack('<i',targets[imp]-search.RVA-offset-5)
            else:unresolved.append(hex(ins.address))
    result.update(size=len(raw),raw=raw.hex(),normalized_sha256=hashlib.sha256(norm).hexdigest(),diff=sum(a!=b for a,b in itertools.zip_longest(norm,search.TARGET)),prefix=next((i for i,(a,b) in enumerate(zip(norm,search.TARGET)) if a!=b),min(len(norm),search.SIZE)),nops=sum(i.mnemonic=='nop' for i in instructions),nop16=b'\x90'*16 in raw,unresolved=unresolved)
    return result

jobs=[(n,*values) for n,values in enumerate(itertools.product(profiles,['/O1','/O2','/O1 /Og-'],[5,6,7],range(3)))]
with concurrent.futures.ThreadPoolExecutor(max_workers=8) as pool:results=list(pool.map(run,jobs))
(OUT/'ltcg.json').write_text(json.dumps(dict(tool_inputs=tool_inputs,results=results),indent=2))
ok=[r for r in results if r.get('link_returncode')==0]
print('LTCG TOTAL',len(results),'GOOD',len(ok),'UNIQUE',len({r['normalized_sha256'] for r in ok}),'NOP16',sum(r['nop16'] for r in ok),'SOME_NOP',sum(r['nops']>0 for r in ok))
for r in sorted(ok,key=lambda r:(r['diff'],-r['prefix']))[:5]:print({k:v for k,v in r.items() if k not in ['raw','compile_log','link_log']})
for r in [r for r in results if r.get('link_returncode')!=0][:3]:print(r)
