"""Diagnostic inverse search over forced Spillcosts glob costs (never landable).
Each evaluation compiles the source under the tracer, overwrites the cost of chosen
globs (by index in symbol_decode.globs order) at the first SSE Spillcosts return,
and scores the emitted body against retail (changed-instruction count).
usage: invsearch.py label src.cpp [rounds]"""
from pathlib import Path
import json, re, struct, sys, difflib
H = Path(__file__).resolve().parent
ROOT = H.parents[2]
sys.path.insert(0, str(ROOT / 'build/army-causal-20261021'))
sys.path.insert(0, str(ROOT / 'build/camera-causal-20261021/decode'))
sys.path.insert(0, str(ROOT / 'tools'))
import trace_pair as T
import symbol_decode as D
import build, capstone, pefile

SYM = '?rva00102F83@W3DCamTransform@@QAEXPAUCameraTransformInput@@@Z'
label, rel = sys.argv[1], sys.argv[2]
rounds = int(sys.argv[3]) if len(sys.argv) > 3 and sys.argv[3].isdigit() else 1
OUT = H / 'inv' / label
OUT.mkdir(parents=True, exist_ok=True)
pe = pefile.PE(str(ROOT / 'baselines/bfme2/workshop-vanilla-1.06/files/game.dat'))
md = capstone.Cs(capstone.CS_ARCH_X86, capstone.CS_MODE_32)


def norm(i):
    s = i.op_str
    if i.mnemonic.startswith('j') or i.mnemonic == 'call':
        s = re.sub(r'^0x[0-9a-f]+$', 'target', s)
    return i.mnemonic + ' ' + re.sub(r'\[(?:0x[0-9a-f]+|0)\]', '[global]', s)


RETAIL = [norm(i) for i in md.disasm(pe.get_data(0x102f83, 4299), 0)]


def changed(raw):
    ours = [norm(i) for i in md.disasm(raw, 0)]
    m = difflib.SequenceMatcher(a=RETAIL, b=ours, autojunk=False)
    return sum(max(b - a, d - c) for t, a, b, c, d in m.get_opcodes() if t != 'equal')


def signed(v): return struct.unpack('<i', struct.pack('<I', v))[0]


def evaluate(force, tag='eval'):
    state = {'done': False, 'globs': None}

    def capture(t, info, ctx, record):
        if info['name'] != '@Spillcosts@12' or state['done']:
            t.records.pop(); return
        globs = D.globs(t)
        if not any(D.camera_component(g['symbol']) for g in globs):
            t.records.pop(); return
        state['done'] = True

        def returned(t, ctx):
            info = []
            for n, g in enumerate(globs):
                comp = D.camera_component(g['symbol'])
                sym = g['symbol']
                info.append({'n': n, 'name': comp[0] if comp else (sym.get('name') if isinstance(sym, dict) else None) or f"#{sym.get('id') if isinstance(sym, dict) else '?'}",
                             'refs': t.word(g['address'] + 36), 'cost': signed(t.word(g['address'] + 64))})
                if n in force:
                    t.write(g['address'] + 64, struct.pack('<i', force[n]))
            state['globs'] = info
        t.on_return(ctx, returned, pop_bytes=12)

    obj = OUT / f'{tag}.obj'
    cmd, env = T.TC.build.compiler_command(H / rel, obj)
    T.TC.OUT = T.LAB
    t = T.Tracer(tag, cmd, SYM, on_hit=capture, env=env)
    T.TC.OUT = OUT
    t.names = ['@Spillcosts@12']
    t.run(timeout=240)
    raw, _ = build.read_object_symbol_bytes(obj, SYM, code_only=True)
    return changed(raw), len(raw), state['globs']


log = open(OUT / 'search.log', 'a')
force = {}
seed = OUT / 'seed.json'
if seed.exists():
    force = {int(k): v for k, v in json.loads(seed.read_text()).items()}
best, size, globs = evaluate(force, 'base')
if '--xy' in sys.argv and not force:
    for g in globs:
        if g['name'] in ('sourcePos.X', 'sourcePos.Y', 'targetPos.X', 'targetPos.Y') and g['refs'] >= 10:
            force[g['n']] = -7 if g['name'].startswith('source') else -6
    seed.write_text(json.dumps(force))
    best, size, globs = evaluate(force, 'base')
print('base', best, size, flush=True); log.write(f'base {best} {size} {force}\n'); log.flush()
for rnd in range(rounds):
    improved = False
    order = sorted(globs, key=lambda g: -g['refs'])
    for g in order:
        n = g['n']
        cur = force.get(n, g['cost'])
        for trial in ((-20,) if cur > 0 else (60,)):
            f = dict(force); f[n] = trial
            score, sz, _ = evaluate(f)
            msg = f"r{rnd} glob {n} {g['name']} refs {g['refs']} cost {g['cost']} -> {trial}: {score} ({sz}B)"
            log.write(msg + '\n'); log.flush()
            if score < best:
                best, force, improved = score, f, True
                print('ACCEPT', msg, flush=True)
                (OUT / 'best.json').write_text(json.dumps({'changed': best, 'force': force}, indent=1))
    if not improved:
        break
print('final', best, force, flush=True)
