from pathlib import Path
import collections,difflib,hashlib,itertools,json,sys
sys.path.insert(0,'tools');import build,capstone
from variant import resolve_bytes
OUT=Path('build/trust237-round12');NAME='?Rva00237D3DCheck@@YAHXZ'
target=build.read_target_bytes(0x237d3d,226)
decoder=capstone.Cs(capstone.CS_ARCH_X86,capstone.CS_MODE_32)
target_mn=[i.mnemonic for i in decoder.disasm(target,0)]
rows=json.loads((OUT/'carriers.json').read_text())
assert len(rows)==1152 and all('error' not in r for r in rows)
unresolved=collections.Counter();unit_groups={};helper_raw={};suffix=[]
symbols={'_memset','__imp__GetCommandLineW@0','__imp__CommandLineToArgvW@8','_WinVerifyTrust@12','__imp__LocalFree@4'}
for r in rows:
    obj=OUT/(r['name']+'.obj')
    raw,rel=build.read_object_symbol_bytes(obj,NAME,code_only=True)
    assert raw.hex()==r['raw'] and hashlib.sha256(raw).hexdigest()==r['sha256']
    assert rel==[tuple(t) for t in r['relocations']]
    res,unres=resolve_bytes(raw,rel,target+b'\0'*max(0,len(raw)-226),0x237d3d,len(raw),{'_memset':[0x6291ae],'_WinVerifyTrust@12':[0x655652]})
    assert unres==r['unresolved']
    assert sum(a!=b for a,b in itertools.zip_longest(res,target))==r['diff']
    assert next((n for n,(a,b) in enumerate(zip(res,target)) if a!=b),min(len(raw),226))==r['prefix']
    instructions=list(decoder.disasm(raw,0));mn=[i.mnemonic for i in instructions]
    eq=sum(c-b for a,b,c,d,e in difflib.SequenceMatcher(a=mn,b=target_mn,autojunk=False).get_opcodes() if a=='equal')
    assert eq==r['eq'] and len(raw)==r['size']
    assert not r['nop16'] and not r['nops'] and all(i.mnemonic!='nop' for i in instructions)
    assert b'\x90'*16 not in raw
    if raw.endswith(bytes.fromhex('405e5bc9c3')):suffix.append(r['name'])
    for entry in unres:unresolved[json.dumps(entry,sort_keys=True)]+=1
    bodies={};todo=[NAME]
    while todo:
        name=todo.pop()
        if name in bodies:continue
        code,rels=build.read_object_symbol_bytes(obj,name,code_only=True)
        bodies[name]=dict(raw=code.hex(),relocations=rels)
        todo += [s for _,_,s in rels if s not in symbols and s not in bodies]
        if name!=NAME:
            helper_raw[hashlib.sha256(code).hexdigest()]=code.hex()
            assert b'\x90'*16 not in code and all(i.mnemonic!='nop' for i in decoder.disasm(code,0))
    key=hashlib.sha256(json.dumps(bodies,sort_keys=True).encode()).hexdigest()
    unit_groups.setdefault(key,[]).append(r['name'])
emulation=json.loads((OUT/'carriers-emulation.json').read_text())
assert emulation['object_count']==1153 and emulation['all_passed']
emulated={r['group']:r for r in emulation['rows']}
for key,names in unit_groups.items():
    assert key in emulated and set(names)<=set(emulated[key]['sources'])
assert len(unit_groups)==emulation['distinct_unit_code_and_relocations']==36
raw,rel=build.read_object_symbol_bytes(Path('build/scratch/0x00237d3d.obj'),NAME,code_only=True)
bank_sha=hashlib.sha256(raw).hexdigest()
assert bank_sha=='23875f743d6816184d590366a5617a42b8e18aedc479c4d75864e0f6d61de0c9'
compact=lambda r:{k:r[k] for k in ['name','meta','size','diff','prefix','eq','sha256','unresolved']}
unique={r['sha256']:r for r in rows}
best=[compact(r) for r in sorted(unique.values(),key=lambda r:(r['diff'],-r['prefix']))[:4]]
result=dict(inputs=dict(bfme2='4bae5b8bf6',bfme1='42bba2ddbd5cb895d99293583891d3b7315fa49e'),
    compiles=len(rows),successful=len(rows),distinct_main_raw=len(unique),
    sequential_object_hash_relocation_score_instruction_audit=True,
    fully_resolved_main_rel32=sum(not r['unresolved'] for r in rows),unresolved_main_diagnostics=dict(unresolved),
    distinct_same_unit_helpers=len(helper_raw),helper_raw=helper_raw,nop_body_count=0,nop16_body_count=0,
    retail_increment_restore_suffix_count=len(suffix),best=best,
    bank=dict(size=len(raw),score='69/86=0.802326',masked_prefix=205,positional_differences=21,sha256=bank_sha),
    emulation={k:v for k,v in emulation.items() if k!='rows'},
    emulation_receipt_sha256=hashlib.sha256((OUT/'carriers-emulation.json').read_bytes()).hexdigest(),
    full_encoder=json.loads((OUT/'encoder.json').read_text()))
assert result['retail_increment_restore_suffix_count']==0
(OUT/'summary.json').write_text(json.dumps(result,indent=2)+'\n')
print({k:v for k,v in result.items() if k not in ['best','helper_raw','emulation','full_encoder']})
