# Native 107626A8/10762764 admits equality edge facts only for register/memory
# versus a literal operand. Delay computed zero until after this admission.
pre='''extern "C" unsigned long __cdecl _byteswap_ulong(unsigned long);
extern "C" unsigned int __cdecl _rotl(unsigned int,int);
extern "C" unsigned int __cdecl _rotr(unsigned int,int);
#pragma intrinsic(_byteswap_ulong,_rotl,_rotr)
'''
u='((unsigned)p)'
exprs=[
'0u', '_rotl(0u,(int)v)', '_rotr(0u,(int)v)',
'_byteswap_ulong(v)^_byteswap_ulong(v)',
'_rotl(v,(int)v)^_rotl(v,(int)v)',
'_rotr(v,(int)v)-_rotr(v,(int)v)',
'(_rotl(v,13)^_rotl(v,13))',
'(v/((v|1u)))-(v/((v|1u)))',
'0u/(v|1u)', '0u%(v|1u)',
'((v&0xffffu)-(unsigned short)v)',
'((v>>16)-(unsigned short)(v>>16))',
'((v<<16)^(v&0xffffu)*65536u)',
'((v|1u)&(v^0xffffffffu)&0xfffffffeu)',
'((v&1u)*(1u-(v&1u)))',
'((v&255u)/256u)',
'((v>>31)>>1)',
'_rotl(v,0)^v', '_rotr(v,32)^v',
'(_byteswap_ulong(_byteswap_ulong(v))-v)',
'(_rotr(_rotl(v,13),13)-v)',
'((v+1u)-(1u+v))',
'((v*3u)-(v+v+v))',
'((v/3u)*3u+(v%3u)-v)',
'((v/7u)*7u+(v%7u)-v)',
'((v/10u)*10u+(v%10u)-v)',
'((v*5u)-(v+(v<<2)))',
'((v|0x55555555u)&(v^0xffffffffu)&0xaaaaaaaau)',
'((v&0x55555555u)&(v^0xffffffffu))',
'((v^0xffffffffu)+v+1u)',
'((v<1u)?v:0u)',
'((v>0u)?0u:v)',
]
def generate():
 for ei,e in enumerate(exprs):
  for helper,local,rev,order,contract,nul,rel in itertools.product(range(2),repeat=7):
   expression='zeroWord((unsigned)p)' if helper else e.replace('v',u)
   lead=decl
   if local:lead+='unsigned zero='+expression+';';expression='zero'
   if ei in (3,4,17,18) and not helper and not local:expression='('+expression+')'
   lhs,rhs=('(unsigned)p',expression) if not rev else (expression,'(unsigned)p')
   op=('>' if not rev else '<') if rel else '!='
   test=lhs+op+rhs
   yes='draw=p; obj=p->m_object;' if not order else 'obj=p->m_object; draw=p;'
   no='draw=p; obj='+('(Object*)p' if nul else '0')+';'
   src=BASE[:lo]+lead+'if('+test+') {'+yes+'} else {'+no+'} if(obj) {'+body+'\n} else {'+terrain+'}\n'+BASE[hi:]
   if not contract:src=src.replace('template <> void StringBase<char>::releaseBuffer() throw();\n','')
   src=src.replace('#define W(s)',pre+'#define W(s)')
   if helper:src=src.replace('void W3DDisplay::gatherDebugStats(void)','static __forceinline unsigned zeroWord(unsigned v) {return '+e+';}\nvoid W3DDisplay::gatherDebugStats(void)')
   yield ('zc%02d%d%d%d%d%d%d%d'%(ei,helper,local,rev,order,contract,nul,rel),src)
if __name__=='__main__':batch(list(generate()),'zerocompare')
