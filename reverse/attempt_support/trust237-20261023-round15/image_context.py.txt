from pathlib import Path
import hashlib,json,re,sys
ROOT=Path.cwd();OUT=ROOT/'build/trust237-round15'
sys.path.insert(0,str(ROOT/'tools'))
import capstone,pefile
md=capstone.Cs(capstone.CS_ARCH_X86,capstone.CS_MODE_32)
result=[]
for name in ['game.dat','worldbuilder.exe']:
 path=ROOT/'baselines/bfme2/workshop-vanilla-1.06/files'/name
 raw=path.read_bytes();pe=pefile.PE(data=raw)
 sections=[]
 for section in pe.sections:
  if not section.Characteristics&0x20000000:continue
  data=section.get_data();runs=list(re.finditer(b'\x90{2,}',data))
  long=[]
  for run in runs:
   if len(run[0])<16:continue
   off=run.start();end=run.end();rva=section.VirtualAddress+off
   after=data[end:end+24]
   long.append(dict(rva=hex(rva),length=len(run[0]),before=data[max(0,off-24):off].hex(),after=after.hex(),decoded_after=[dict(rva=hex(i.address),text=i.mnemonic+' '+i.op_str) for i in md.disasm(after,section.VirtualAddress+end)]))
  sections.append(dict(name=section.Name.rstrip(b'\0').decode(),rva=hex(section.VirtualAddress),raw_size=len(data),runs_ge2=len(runs),maximum_run=max([len(x[0]) for x in runs],default=0),long_runs=long))
 result.append(dict(file=name,sha256=hashlib.sha256(raw).hexdigest(),sections=sections))
(OUT/'image-context.json').write_text(json.dumps(result,indent=2)+'\n')
for item in result:
 for sec in item['sections']:
  print(item['file'],sec['name'],'runs>=2',sec['runs_ge2'],'max',sec['maximum_run'],'long',[(r['rva'],r['length']) for r in sec['long_runs']],flush=True)
