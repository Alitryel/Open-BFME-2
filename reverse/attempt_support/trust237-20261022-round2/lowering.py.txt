import sys,itertools
from pathlib import Path
sys.path.insert(0,str(Path('build/trust237-round2').resolve()))
import search
mode=sys.argv[1] if len(sys.argv)>1 else 'bitfield'
jobs=[]
if mode=='bitfield':
    # All expressions produce the value 1 on the target's 32-bit unsigned ABI.
    reps=[
      ('struct Narrow { unsigned int one:1; };','Narrow n; n.one=u|1u;','n.one'),
      ('struct Narrow { unsigned char one:1; };','Narrow n; n.one=u|1u;','n.one'),
      ('struct Narrow { unsigned short one:1; };','Narrow n; n.one=u|1u;','n.one'),
      ('struct Narrow { bool one:1; };','Narrow n; n.one=(bool)(u|1u);','n.one'),
      ('struct Narrow { int one:1; };','Narrow n; n.one=u|1u;','-n.one'),
      ('struct Narrow { unsigned int zero:1,one:1; };','Narrow n; n.one=(u<<1u)|1u;','n.one'),
      ('struct Narrow { unsigned int one:1; };','Narrow n={u|1u};','n.one'),
      ('struct Narrow { unsigned int one:1; Narrow(unsigned int x):one(x) {} };','Narrow n(u|1u);','n.one'),
    ]
    forms=[
      'return V;', 'int r=0; r+=V; return r;',
      'int r=0; if(V) ++r; return r;',
      'int r=0; while(V) {++r;break;} return r;',
      'int r=0; do {if(!V) break; ++r;} while(0); return r;',
      'int r=0; switch(V){case 0:break;case 1:++r;break;} return r;',
      'bool v=V; return v;',
      'int r=V; return r?1:0;',
    ]
    for ri,fi,timing,scope in itertools.product(range(len(reps)),range(len(forms)),range(2),range(2)):
        decl,init,value=reps[ri];tail=forms[fi].replace('V','('+value+')')
        region=(init+' LocalFree(argv); '+tail) if timing else ('LocalFree(argv); '+init+' '+tail)
        if scope:region='{ '+region+' }'
        body='    long status=WinVerifyTrust(0,&action,&data); unsigned long u=(unsigned long)status; '+region
        text=search.base.replace(search.marker,body).replace('int Rva00237D3DCheck() {',decl+'\nint Rva00237D3DCheck() {')
        jobs.append((f'n{len(jobs):04}',text,dict(rep=ri,form=fi,timing=timing,scope=scope)))
elif mode=='assume':
    for pi,form,timing in itertools.product([0,1,5,6,8,9,10,11,12,13,16,17,18,22,23],range(5),range(2)):
        pred=search.PRED[pi]
        if form==0:tail='__assume('+pred+'); return '+pred+';'
        elif form==1:tail='if(!('+pred+')) __assume(0); return '+pred+';'
        elif form==2:tail='bool p='+pred+'; __assume(p); int r=0; if(p) ++r; return r;'
        elif form==3:tail='__assume('+pred+'); int r=0; while('+pred+') {++r;break;} return r;'
        else:tail='__assume('+pred+'); return 1;'
        if timing:tail='__assume('+pred+'); LocalFree(argv); '+tail
        else:tail='LocalFree(argv); '+tail
        text=search.base.replace(search.marker,'    long status=WinVerifyTrust(0,&action,&data); unsigned long u=(unsigned long)status; '+tail)
        jobs.append((f'a{len(jobs):04}',text,dict(pred=pi,form=form,timing=timing)))
else:raise SystemExit(mode)
search.execute(mode,jobs)
