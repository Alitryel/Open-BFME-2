"""WB-evidenced sleep getter and shared weapon-action inlining probe."""
from pathlib import Path
from itertools import product
import re
import json
import sys

ROOT = Path(__file__).resolve().parent.parent
sys.path.insert(0, str(ROOT / 'tools'))
import permute

OUT = ROOT / 'build/dual-weapon-probe'
OUT.mkdir(exist_ok=True)
HOME = ROOT / 'Code/GameEngine/Source/GameLogic/Object/Behavior/DualWeaponBehaviorCtor.cpp'
BASE = (ROOT / 'reverse/attempts/0x0045af14.cpp').read_text()
# Scratch files do not inherit the production TU's region-voted flags.
# These are exactly the flags compiler_command() selects for the home unit.
BASE = BASE.replace('// cl: /GX /MD /DNDEBUG', '// cl: /O1 /arch:SSE /G7 /GX /MD /DNDEBUG')
BASE = re.sub(r'#include "([.][.]/[^\"]+)"',
              lambda m: '#include "' + (HOME.parent / m[1]).resolve().as_posix() + '"', BASE)
SYMBOL = '?update@DualWeaponBehavior@@UAE?AW4UpdateSleepTime@@XZ'
MARK = 'UpdateSleepTime DualWeaponBehavior::update()'
SET = 'obj->setWeaponSetFlag(WEAPONSET_CLOSE);\n\t\t\tobj->setCloseCondition();'
CHECK = '''if (!obj->getWeaponSetFlags()->test(7))
\t\t{
\t\t\tobj->setWeaponSetFlag(WEAPONSET_CLOSE);
\t\t\tobj->setCloseCondition();
\t\t}'''


def source_for(sleep, action, force):
    source = BASE
    definitions = ''
    tag = '__forceinline' if force else 'inline'
    if sleep != 'none':
        if sleep == 'free':
            definitions += f'static {tag} UpdateSleepTime normalSleep() {{ return (UpdateSleepTime)BfmeLogicFramesPerSecond; }}\n'
        else:
            suffix = ' const' if sleep == 'const' else ''
            source = source.replace('virtual UpdateSleepTime update();\nprivate:',
                                    'virtual UpdateSleepTime update();\n' +
                                    f'    UpdateSleepTime normalSleep(){suffix};\nprivate:')
            definitions += f'{tag} UpdateSleepTime DualWeaponBehavior::normalSleep(){suffix} {{ return (UpdateSleepTime)BfmeLogicFramesPerSecond; }}\n'
        source = source.replace('return (UpdateSleepTime)BfmeLogicFramesPerSecond;', 'return normalSleep();')
    if action != 'none':
        member = action.endswith('Member')
        ensure = action.startswith('ensure')
        body = 'setWeaponSetFlag(WEAPONSET_CLOSE); setCloseCondition();'
        if ensure:
            body = 'if (!getWeaponSetFlags()->test(7)) { ' + body + ' }'
        if member:
            definitions += f'static {tag} void weaponClose(Object *obj) {{ obj->weaponClose(); }}\n'
            source = source.replace(' public:\n bool isKindOf',
                                    ' public:\n ' + tag + ' void weaponClose() { ' + body + ' }\n bool isKindOf')
        else:
            for name in ('setWeaponSetFlag', 'setCloseCondition', 'getWeaponSetFlags'):
                body = body.replace(name + '(', 'obj->' + name + '(')
            definitions += f'static {tag} void weaponClose(Object *obj) {{ {body} }}\n'
        if ensure:
            assert source.count(CHECK) == 3
            source = source.replace(CHECK, 'weaponClose(obj);')
        else:
            assert source.count(SET) == 4
            source = source.replace(SET, 'weaponClose(obj);')
    return source.replace(MARK, definitions + '\n' + MARK)


def main():
    scorer = permute.Scorer('0x0045AF14', SYMBOL, 527, OUT)
    records = []
    best = None
    for sleep, action, force in product(('none', 'free', 'member', 'const'),
                                        ('none', 'setFree', 'setMember', 'ensureFree', 'ensureMember'),
                                        (False, True)):
        source = source_for(sleep, action, force)
        fit, exact = scorer.score(source)
        r = dict(sleep=sleep, action=action, force=force, fitness=fit, exact=exact)
        if fit >= 0 and scorer.last:
            a,b = scorer.last
            x,y = permute._instructions(a),permute._instructions(b)
            nd = sum(i != j for i,j in zip(x,y)) + abs(len(x)-len(y))
            r.update(size=len(a), differences=nd, unresolved=scorer.unresolved)
            if best is None or fit > best:
                best = fit
                (OUT / 'best.cpp').write_text(source)
                (OUT / 'best.bin').write_bytes(a)
                print(json.dumps(r), flush=True)
            if exact:
                (OUT / 'win.cpp').write_text(source)
                records.append(r)
                break
        else:
            r['failed'] = True
        records.append(r)
    (OUT / 'results.json').write_text(json.dumps(records,indent=2))
    print('finished',len(records),'trials',flush=True)


if __name__ == '__main__':
    main()
