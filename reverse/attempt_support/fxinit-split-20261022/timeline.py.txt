import sys,json,re
from pathlib import Path
sys.path.insert(0,'build/netpacket-20261022e')
from wdbg import Dbg
src,obj=sys.argv[1:3];d=Dbg(src,obj,obj+'.debug.log');d.prompt();d.prompt('set $CanDeferOnBPByAddr = 1');d.prompt('break *0x10783aac')
on=False;bps=[];events=[];round=0

def members(h):
 p=d.words(h,1)[0] if h else 0;v=[];seen=set()
 while p and p not in seen:
  seen.add(p);a,n,b=d.words(p,3);v.extend(a+i for i in range(32) if b>>i&1);p=n
 return v

def rec(idx):
 p=d.words(0x108807b8+4*(idx&1023),1)[0]
 while p:
  w=d.words(p,18)
  if w[7]==idx:return {'ptr':p,'w':w}
  p=w[12]
 return {'idx':idx}

def record(p):return {'ptr':p,'w':d.words(p,18)}
while True:
 out=d.prompt('cont')
 if 'terminated'in out:break
 ip=d.reg('eip')-0x10700000
 if ip==0x83aac:
  f=d.words(d.reg('ebp'),1)[0];fn=d.string(d.words(f+4,1)[0]);yes='?Init@FXShaderGeometry' in fn and d.reg('ebx')==0
  if yes and not on:
   on=True
   for v in [0x82a4d,0x831f7,0xa8f54,0x8ab54,0x8ab66,0x8ab85,0x8292a,0x82978]:
    s=d.prompt('break *0x%x'%(v+0x10700000));n=int(re.search(r'Breakpoint (\d+)',s)[1]);bps.append(n)
    if v==0x8292a:d.prompt('condition %d (*(int *)($eax+0x44)==647)'%n)
  elif not yes and on:
   for n in bps:d.prompt('disable %d'%n)
   on=False
  continue
 if not on:continue
 e={'ip':hex(ip),'round':round}
 if ip==0x82978:round+=1;continue
 if ip==0x82a4d:e['kind']='pop';e['rec']=record(d.reg('eax'))
 elif ip==0x831f7:
  st=d.words(d.reg('esp'),4);e['kind']='colored';e['rec']=record(st[2]);e['neighbors']=members(st[3])
 elif ip==0xa8f54:
  e['kind']='split';e['members']=members(d.words(d.reg('esp')+8,1)[0])
 elif ip in (0x8ab54,0x8ab66,0x8ab85):
  e['kind']='rebuild';e['regs']={r:d.reg(r) for r in ['eax','ebx','ecx','edx','esi','edi','esp','ebp']};e['stack']=d.words(e['regs']['esp'],16)
  e['set']=members(d.words(0x10882158+4*e['regs']['ebx'],1)[0]);e['records']=[rec(i) for i in e['set']]
 else:
  sp=d.reg('esp');e['kind']='cost';e['rec']=record(d.reg('eax'));e['factor']=d.reg('edx');e['newcost']=d.reg('ebx');e['block']=d.words(sp+0x20,1)[0]
  e['lists']={}
  for off in [0x10,0x1c]:
   p=d.words(sp+off,1)[0];ls=[];seen=set()
   while p and p not in seen:
    seen.add(p);r=record(p);ls.append(r);p=r['w'][5]
   e['lists'][str(off)]=ls
 events.append(e);Path(obj+'.timeline.json').write_text(json.dumps(events))
 if e['kind']=='rebuild':print(round,e['ip'],'active',len(e['set']),flush=True)
 elif 'rec'in e:
  w=e['rec']['w']
  if w[17] in [644,645,647] or w[7] in [51,109,129,1]:print(round,e['kind'],w[7],w[17],w[3],e.get('factor',''),flush=True)
print('DONE',len(events),flush=True)
