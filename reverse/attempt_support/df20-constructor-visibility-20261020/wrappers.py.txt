from pathlib import Path
import sys,itertools,concurrent.futures,json
sys.path.insert(0,'build/df20-scalars-20261020');import trials
trials.scratch=Path.cwd()/'build/df20-scalars-20261020'
base=Path('build/df20-scalars-20261020/bank.cpp').read_text();trials.symbol=base.splitlines()[0][3:]
specs=[]
for mask,depth,opaque,desdepth in itertools.product(range(1,8),[1,2,3,255],[0,1],[0,255]):
 s=base
 if not opaque:s=s.replace('template <> _String_base<wchar_t, allocator<wchar_t> >::~_String_base();','')
 wrapper='''
#pragma inline_depth(0)
struct _MoneyString : basic_string<wchar_t> {
 typedef basic_string<wchar_t> Base;
 __forceinline _MoneyString(const Base& s) : Base(s) {}
#pragma inline_depth(255)
 __forceinline _MoneyString(const wchar_t* f,const wchar_t* l) : Base(f,l) {}
#pragma inline_depth(DEPTH)
 __forceinline ~_MoneyString() {}
};
#pragma inline_depth(255)
'''.replace('DEPTH',str(desdepth))
 s=s.replace('struct _MoneySignPolicy :',wrapper+'struct _MoneySignPolicy :',1)
 for bit,name in enumerate(['curr_sym','sign','new_digits']):
  if mask&(1<<bit):s=s.replace('string_type __'+name,'_MoneyString __'+name)
 # The pragma in force at the parent's closing brace controls nonzero depth.
 pos=s.rfind('\n}\n}')
 s=s[:pos]+f'\n#pragma inline_depth({depth})'+s[pos:]
 specs.append((f'wrap-m{mask}-d{depth}-o{opaque}-e{desdepth}',s))
with concurrent.futures.ThreadPoolExecutor(max_workers=8) as p:r=list(p.map(trials.run,specs))
(trials.scratch/'wrappers.json').write_text(json.dumps(r,indent=2))
