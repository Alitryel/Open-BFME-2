from headmerge import *

# Unlike nulljoin's obj=0 replacement, this explicitly replaces the ternary
# false value too. On the null edge p is an honest null Object pointer.
nulls=['0','(Object*)p','reinterpret_cast<Object*>(p)',
       '(Object*)((unsigned)p)','(Object*)((unsigned)p & 0xffffffffu)']
tests=['p','p!=0','(unsigned)p>0','(unsigned)p>=1']
items=[]
for f,n,c,order,contract in itertools.product([8,9,10,12],range(len(nulls)),range(len(tests)),range(2),range(2)):
 form=forms[f].replace('obj=0','obj='+nulls[n]).replace('p?p->m_object:0','p?p->m_object:'+nulls[n])
 form=form.replace('if(p)','if('+tests[c]+')').replace('obj=p?','obj=('+tests[c]+')?')
 if order:form=form.replace('draw=p; obj=p->m_object;','obj=p->m_object; draw=p;')
 src=BASE[:lo]+decl+form+'\n'+BASE[hi:]
 if not contract:src=src.replace('template <> void StringBase<char>::releaseBuffer() throw();\n','')
 items.append(('carry%02d%d%d%d%d'%(f,n,c,order,contract),src))
if __name__=='__main__':batch(items,'nullcarry')
