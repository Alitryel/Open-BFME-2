import sys,json,itertools,time
from pathlib import Path
from concurrent.futures import ThreadPoolExecutor,as_completed
sys.path.insert(0,'build/fxinit-seat22')
import search as h
h.HERE=Path('build/fxinit-cost22').resolve()
BASE=h.BASE

def variants(mode):
 if mode in ['relativecancel','relativecanceltypes']:
  for anchor,arith,ptype,stage,ref,order,mexpr in itertools.product(range(2),range(8),range(3),range(2),range(2),range(2),range(2)):
   if mode=='relativecanceltypes' and (ptype!=2 or ref!=1):continue
   src=BASE;k=anchor;j=1-anchor
   a=f'links[{k}][i]';b=f'links[{j}][i]';decl=''
   if stage:
    decl=f'int primary={a}; int secondary={b};';a='primary';b='secondary'
   a='((int)'+a+')';b='((int)'+b+')'
   base='remap.begin()' if not mexpr else 'remap._M_start'
   ptypes=['const short *','const char *','unsigned']
   if ptype==0:init=f'{base}+{a}';unit=1;access='*pivot'
   if ptype==1:init=f'(const char *){base}+{a}*2';unit=2;access='*(const short *)pivot'
   if ptype==2:init=f'(unsigned){base}+{a}*2';unit=2;access='*(const short *)pivot'
   decl+=ptypes[ptype]+(' const &' if ref else '')+' pivot='+init+';'
   aa=f'({a}*{unit})';bb=f'({b}*{unit})'
   other=[f'pivot+({bb}-{aa})',f'(pivot+{bb})-{aa}',f'(pivot-{aa})+{bb}',f'({bb}-{aa})+pivot',f'pivot+(({b}-{a})*{unit})',f'({b}-{a})*{unit}+pivot',f'pivot-({aa}-{bb})',f'pivot-(({a}-{b})*{unit})'][arith]
   other='*(const short *)('+other+')'
   vals=[access,other] if not anchor else [other,access]
   expr=vals[0]+'+('+vals[1]+'<<8)' if not order else '('+vals[1]+'<<8)+'+vals[0]
   src=src.replace(h.R,decl+'*(int *)(vertices+desc.Get_Basis_End())='+expr+';')
   yield f'{mode}_{anchor}{arith}{ptype}{stage}{ref}{order}{mexpr}',src
 if mode=='relativebase':
  for anchor,ty,ref,helper,order,form,copyguard,staging in itertools.product(range(2),range(2),range(2),range(2),range(2),range(2),range(2),range(2)):
   src=BASE
   # Both indices are valid in the same remap allocation. Their signed
   # difference fits int (each input is uint16), including negative offsets.
   k=anchor;j=1-anchor
   typ=['int','unsigned'][ty]
   a=f'links[{k}][i]';b=f'links[{j}][i]'
   decl=''
   if staging:
    decl+=typ+f' primary={a}; '+typ+f' secondary={b};'
    a='primary';b='secondary'
   if ref:
    decl+='const short &pivot=remap['+a+'];'
    aa='pivot';bb='(&pivot)['+f'(int)({b})-(int)({a})'+']'
   else:
    decl+='const short *pivot=&remap['+a+'];'
    aa='*pivot';bb='pivot['+f'(int)({b})-(int)({a})'+']'
   if form:
    bb=bb.replace('(&pivot)[','*(&pivot+(').replace('pivot[','*(pivot+(')[:-1]+'))'
   vals=[aa,bb] if anchor==0 else [bb,aa]
   expr=vals[0]+'+('+vals[1]+'<<8)'
   if order:expr='('+vals[1]+'<<8)+'+vals[0]
   if helper:
    hdecl='static __forceinline int FxRelativeBone(short *map,unsigned a,unsigned b) {'
    hd='const short *pivot=map+a;return '+('*pivot+(pivot[(int)b-(int)a]<<8)' if not anchor else 'pivot[(int)b-(int)a]+(*pivot<<8)')+';}\n'
    src=src.replace('bool FXShaderGeometry::Init(',hdecl+hd+'bool FXShaderGeometry::Init(')
    expr=f'FxRelativeBone(remap.begin(),links[{k}][i],links[{j}][i])';decl=''
   replacement=decl+'*(int *)(vertices+desc.Get_Basis_End())='+expr+';'
   src=src.replace(h.R,replacement)
   if copyguard:src=src.replace('if(positions[1])','if(*(const Vector3 *volatile *)&positions[1])')
   yield f'relativebase_{anchor}{ty}{ref}{helper}{order}{form}{copyguard}{staging}',src
 if mode=='mapbase':
  for kind,decl,where,one,order,readptr in itertools.product(range(4),range(4),range(4),range(2),range(2),range(2)):
   src=BASE
   expr=['*(short *volatile *)&remap._M_start','(short *)*(volatile unsigned *)&remap._M_start','*(short **)&remap._M_start','remap.begin()'][kind]
   typ=['short *','short *const &','short *volatile ','const short *'][decl]
   declaration=typ+'mapBase='+expr+';\n'
   lo=src.index('  for(unsigned pass=0;');hi=src.index(' m_IB=new',lo)
   core=src[lo:hi]
   remapexpr='mapBase' if not readptr else '*(short *const *)&mapBase'
   newR=h.R.replace('remap[','('+remapexpr+')[')
   core=core.replace(h.R,newR)
   if one and where!=3:core=core.replace('remap[links[0][i]];', '('+remapexpr+')[links[0][i]];')
   sites=['  for(unsigned pass=0;', '   for(unsigned i=0;i<(unsigned)mesh->Get_Vertex_Count();', '     if(m_BonesPerVertex==1)', '      '+newR]
   if order and where==3:
    core=core.replace('      '+newR+'\n      '+h.W,'      '+h.W+'\n'+declaration+'      '+newR)
   else:core=core.replace(sites[where],declaration+sites[where],1)
   src=src[:lo]+core+src[hi:]
   yield f'mapbase_{kind}{decl}{where}{one}{order}{readptr}',src
 if mode=='guardhelpers':
  for rt,pt,neg,which,wrap,spelling in itertools.product(range(4),range(4),range(2),range(3),range(3),range(3)):
   src=BASE;rett=['bool','int','unsigned','const void *'][rt]
   param=['const Vector3 *p','const Vector3 *const &p','const void *p','unsigned p'][pt]
   val='p==0' if neg else 'p!=0'
   if rt==3:val='(const void *)'+('!p' if neg else 'p')
   if spelling==1:val='('+val+') ? 1 : 0' if rt!=3 else val
   if spelling==2:val='!('+('p!=0' if neg else 'p==0')+')' if rt!=3 else val
   if wrap==0:helper=f'static __forceinline {rett} FxHasStream({param}) {{return {val};}}\n'
   elif wrap==1:helper=f'static inline {rett} FxHasStream({param}) throw() {{return {val};}}\n'
   else:
    paramtype=['const Vector3 *','const Vector3 *const &','const void *','unsigned'][pt]
    # Reference members initialize directly from the passed lvalue.
    helper=f'struct FxStreamTest {{ {paramtype} p; FxStreamTest({param}):p(p) {{}} operator {rett}() const {{return {val};}} }};\n'
   for idx,n in enumerate(['positions','normals']):
    if which!=2 and which!=idx:continue
    arg=f'{n}[1]' if pt!=3 else f'(unsigned){n}[1]'
    expr=('FxStreamTest' if wrap==2 else 'FxHasStream')+'('+arg+')'
    if neg:expr='!'+expr
    src=src.replace('if('+n+'[1])','if('+expr+')')
   src=src.replace('bool FXShaderGeometry::Init(',helper+'bool FXShaderGeometry::Init(')
   yield f'guardhelpers_{rt}{pt}{neg}{which}{wrap}{spelling}',src
 if mode=='cursoruse':
  for bits,typ,where,form in itertools.product(range(64),range(2),range(2),range(2)):
   src=BASE
   start=src.index('    else if(m_BonesPerVertex>0 && links[0])')
   end=src.index('    vertices+=',start)
   arm=src[start:end]
   decl=('const unsigned short *' if not typ else 'const unsigned short *const &')+'cursor=links[0]+i;\n'
   if where==0:arm=arm.replace('     if(m_BonesPerVertex==1)',decl+'     if(m_BonesPerVertex==1)')
   else:arm=arm.replace('else if(m_BonesPerVertex==2) {','else if(m_BonesPerVertex==2) {\n'+decl)
   lo=arm.index('else if(m_BonesPerVertex==2)') if where else 0
   a,b=arm[:lo],arm[lo:]
   old=['links[0][i]','links[1][i]','weights[0][i]','weights[1][i]','positions[1][i]','normals[1][i]']
   new=['*cursor','cursor[links[1]-links[0]]','cursor[weights[0]-links[0]]','cursor[weights[1]-links[0]]','positions[1][cursor-links[0]]','normals[1][cursor-links[0]]']
   if form:new=[v.replace('cursor[links[1]-links[0]]','*(cursor+(links[1]-links[0]))').replace('cursor[weights[0]-links[0]]','*(cursor+(weights[0]-links[0]))').replace('cursor[weights[1]-links[0]]','*(cursor+(weights[1]-links[0]))') for v in new]
   for i in range(6):
    if bits>>i&1:b=b.replace(old[i],new[i])
   src=src[:start]+a+b+src[end:]
   yield f'cursoruse_{bits:02x}{typ}{where}{form}',src
 if mode in ['posflags','posflagsscope']:
  sites=['  for(unsigned pass=0;', '   for(unsigned i=0;i<(unsigned)mesh->Get_Vertex_Count();', '    *(Vector3 *)(vertices+desc.Get_Location_Offset(0))', '     if(m_BonesPerVertex==1)', '      '+h.R, '      '+h.W, '      if(positions[1])']
  for ty,where,which,inv,read in itertools.product(range(4),range(7),range(3),range(2),range(2)):
   if mode=='posflagsscope' and where!=1:continue
   src=BASE;dec='';types=['bool','volatile bool','unsigned','volatile unsigned']
   for idx,n in enumerate(['positions','normals']):
    if which!=2 and which!=idx:continue
    expr=f'{n}[1]!=0' if not inv else f'{n}[1]==0'
    dec+=types[ty]+f' have_{n}=({expr});\n'
    src=src.replace(f'if({n}[1])',f'if({"!" if inv else ""}have_{n})')
    if read:src=src.replace(f'={n}[1][i];',f'=(*(const Vector3 *volatile *)&{n}[1])[i];')
   token=sites[where]
   if where==6:token=token.replace('if(positions[1])',f'if({"!" if inv else ""}have_positions)') if which in [0,2] else token
   assert token in src
   src=src.replace(token,dec+token,1)
   yield f'{mode}_{ty}{where}{which}{inv}{read}',src
 if mode=='postypes':
  for style,which,guard,access in itertools.product(range(8),range(3),range(6),range(2)):
   src=BASE
   for idx,n in enumerate(['positions','normals']):
    if which!=2 and which!=idx:continue
    init='mesh->Get_Vertex_Array(0)' if idx==0 else 'mesh->Get_Vertex_Normal_Array(false)'
    old=f'const Vector3 *{n}[2]={{{init},0}};'
    expr0=f'{n}[0]';expr1=f'{n}[1]'
    if style==0:pass
    elif style==1:src=src.replace(old,f'const Vector3 *volatile {n}[2]={{{init},0}};')
    elif style in [2,3,4,5]:
     second=['const Vector3 *volatile b','const Vector3 *b','unsigned b','volatile unsigned b'][style-2]
     src=src.replace(old,f'struct {{const Vector3 *a; {second};}} {n}={{{init},0}};')
     src=src.replace(f'{n}[0]',f'{n}.a').replace(f'{n}[1]',f'{n}.b')
     expr1=f'{n}.b'
     if style in [4,5]:
      getter='mesh->Get_Vertex_Array(1)' if idx==0 else 'mesh->Get_Vertex_Normal_Array(true)'
      src=src.replace(f'{n}.b={getter};',f'{n}.b=(unsigned){getter};')
      src=src.replace(f'={n}.b[i];',f'=((const Vector3 *){n}.b)[i];')
    elif style==6:
     src=src.replace(f'if({n}[1])',f'if(*(unsigned volatile *)&{n}[1])')
    elif style==7:
     src=src.replace(f'if({n}[1])',f'if(*(unsigned *)&{n}[1])')
    forms=[expr1,f'(unsigned){expr1}!=0',f'((unsigned){expr1}|(0-(unsigned){expr1}))>>31',f'((unsigned){expr1}>0)',f'((unsigned){expr1}-1)!=0xffffffff',f'((const void*){expr1})!=0']
    src=src.replace('if('+expr1+')','if('+forms[guard]+')')
    if access and style not in [4,5]:src=src.replace('='+expr1+'[i];','=(*(const Vector3 *volatile *)&'+expr1+')[i];')
   yield f'postypes_{style}{which}{guard}{access}',src
 if mode in ['aliases','aliasesscope']:
  sites=['  for(unsigned pass=0;', '   for(unsigned i=0;i<(unsigned)mesh->Get_Vertex_Count();', '    *(Vector3 *)(vertices+desc.Get_Location_Offset(0))', '     if(m_BonesPerVertex==1)']
  maps=['short *boneMap=remap.begin();','short *const &boneMap=remap.begin();','short *&boneMap=remap._M_start;']
  for mp,lp,mt,lt,use,high in itertools.product(range(4),range(4),range(3),range(2),range(3),range(2)):
   if mode=='aliasesscope' and mp!=1 and lp!=1:continue
   src=BASE
   lo=src.index('  for(unsigned pass=0;');hi=src.index(' m_IB=new',lo)
   core=src[lo:hi]
   if use!=2:core=core.replace('remap[links[0][i]];', 'boneMap[sourceLinks[i]];')
   if use!=1:
    core=core.replace('remap[links[0][i]]+(remap[links[1][i]]<<8)', 'boneMap[sourceLinks[i]]+(boneMap['+('sourceLinks[i+(links[1]-links[0])]' if high else 'links[1][i]')+']<<8)')
   src=src[:lo]+core+src[hi:]
   dec=maps[mt]+'\n'
   ld=('const unsigned short *const &sourceLinks=links[0];' if lt else 'const unsigned short *sourceLinks=links[0];')+'\n'
   src=src.replace(sites[mp],dec+sites[mp],1)
   src=src.replace(sites[lp],ld+sites[lp],1)
   yield f'{mode}_{mp}{lp}{mt}{lt}{use}{high}',src
 if mode=='guards':
  choices=[
   ('if(positions[1])','if(*(const Vector3 *volatile *)&positions[1])'),
   ('if(normals[1])','if(*(const Vector3 *volatile *)&normals[1])'),
   ('m_BonesPerVertex>0 && links[0]','m_BonesPerVertex>0 && *(const unsigned short *volatile *)&links[0]'),
   ('if(m_BonesPerVertex==2)','if(*(volatile int *)&m_BonesPerVertex==2)'),
   ('if(m_BonesPerVertex==1)','if(*(volatile int *)&m_BonesPerVertex==1)'),
   ('if(m_Passes>1) *(unsigned *)','if(*(volatile int *)&m_Passes>1) *(unsigned *)'),
   ('if(basis) {','if(*(volatile bool *)&basis) {'),
   ('m_BonesPerVertex>0 &&','*(volatile int *)&m_BonesPerVertex>0 &&')]
  for bits in range(256):
   src=BASE
   for k,(old,new) in enumerate(choices):
    if bits>>k&1:src=src.replace(old,new)
   yield f'guards_{bits:03x}',src

def main(mode):
 items=list(variants(mode));start=time.time();rs=[]
 with ThreadPoolExecutor(max_workers=8) as pool:
  for f in as_completed([pool.submit(h.run_one,item) for item in items]):
   r=f.result();rs.append(r)
   if len(rs)%64==0:print(mode,len(rs),'/',len(items),'best',sorted(rs,key=lambda x:(x.get('nd',99999),x.get('cost',9999)))[:2],flush=True)
 Path(f'build/fxinit-cost22/{mode}-results.json').write_text(json.dumps(rs,indent=2))
 print('DONE',len(rs),'seconds',time.time()-start,'best',sorted(rs,key=lambda x:(x.get('nd',99999),x.get('cost',9999)))[:8],flush=True)
if __name__=='__main__':main(sys.argv[1])
