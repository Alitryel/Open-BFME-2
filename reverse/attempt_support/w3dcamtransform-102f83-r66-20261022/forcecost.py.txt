"""Diagnostic only: force Spillcosts results for chosen camera components, then score.
usage: forcecost.py label src.cpp 'sourcePos.X=-1,targetPos.X=-1,...'"""
from pathlib import Path
import json, struct, sys
H = Path(__file__).resolve().parent
ROOT = H.parents[2]
sys.path.insert(0, str(ROOT / 'build/army-causal-20261021'))
sys.path.insert(0, str(ROOT / 'build/camera-causal-20261021/decode'))
sys.path.insert(0, str(ROOT / 'tools'))
import trace_pair as T
import symbol_decode as D
import build, permute

SYM = '?rva00102F83@W3DCamTransform@@QAEXPAUCameraTransformInput@@@Z'
label, rel, spec = sys.argv[1], sys.argv[2], sys.argv[3]
force = {k: int(v) for k, v in (x.split('=') for x in spec.split(',') if x)}
OUT = H / 'force' / label
OUT.mkdir(parents=True, exist_ok=True)
state = {'done': False, 'log': []}


def capture(t, info, ctx, record):
    if info['name'] != '@Spillcosts@12' or state['done']:
        t.records.pop(); return
    globs = [g for g in D.globs(t) if D.camera_component(g['symbol'])]
    if not globs:
        t.records.pop(); return
    state['done'] = True

    def returned(t, ctx):
        allg = '--all' in sys.argv
        best = {}
        for g in globs:
            comp = D.camera_component(g['symbol'])[0]
            refs = t.word(g['address'] + 36)
            if comp in force and (allg or refs > best.get(comp, (0, 0))[1]):
                best[(comp, g['address']) if allg else comp] = (g['address'], refs)
        for key, (addr, refs) in best.items():
            comp = key[0] if allg else key
            old = struct.unpack('<i', struct.pack('<I', t.word(addr + 64)))[0]
            t.write(addr + 64, struct.pack('<i', force[comp]))
            state['log'].append((comp, refs, old, force[comp]))
    t.on_return(ctx, returned, pop_bytes=12)


cpp = H / rel
obj = OUT / (label + '.obj')
cmd, env = T.TC.build.compiler_command(cpp, obj)
T.TC.OUT = T.LAB
t = T.Tracer(label, cmd, SYM, on_hit=capture, env=env)
T.TC.OUT = OUT
t.names = ['@Spillcosts@12']
t.run(timeout=240)
print('forced', state['log'])
row = {'name': SYM, 'target_rva': '0x00102F83', 'target_size': '4299', 'source': cpp.relative_to(ROOT).as_posix(), 'notes': ''}
got = build.compile_function(row, build.load_symbol_map(), obj)
print('fitness', permute.fitness(got['bytes'], got['target']), 'len', len(got['bytes']))
