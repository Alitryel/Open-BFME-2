from pathlib import Path
import subprocess,sys,re,json
out=Path('build/resume-eh-local/turn63-wall-returns');out.mkdir(exist_ok=True)
home=Path('Code/GameEngine/Source/GameLogic/SkirmishAI/AIWallBuilder')
base=Path('reverse/attempts/0x00596e15.cpp').read_text()
base=re.sub(r'#include "([^"]+)"',lambda m:'#include "'+(home/m[1]).resolve().as_posix()+'"',base)
base=base.replace('register int zero = 0;','').replace('m_startNodeID == zero','m_startNodeID == 0').replace('if (0) { AsciiString dummy; }','')
base=base.replace('int rva0055ADD8(int ignored);','int failBuild(int ignored);').replace('->rva0055ADD8((int)player)','->failBuild((int)player)')
tail='m_totalCost = result.m_pad04;\n \n return 0;'
assert tail in base
forms={'baseline':base}
for name,expr in {
 'comma':'return (m_totalCost = result.m_pad04, 0);',
 'assignment_compare':'return (m_totalCost = result.m_pad04) != result.m_pad04;',
 'cast_byte_zero':'m_totalCost = result.m_pad04; unsigned char success = 0; return success;',
 'bool_zero':'m_totalCost = result.m_pad04; bool success = false; return success;',
 'zero_after_quote_scope':'m_totalCost = result.m_pad04;\n }\n return 0;'
}.items():
 s=base.replace(tail,expr)
 if name=='zero_after_quote_scope':s=s.replace('Rva0039205C result;','{ Rva0039205C result;')
 forms[name]=s
# Forceinline bodies preserve real return values while exposing a distinct
# expression boundary to the return-value/EH lowering, without extra effects.
for typ in ['int','unsigned char','bool']:
 for mode in ['zero','all']:
  helper='__forceinline '+typ+' wallResult('+typ+' value) { return value; }\n'
  s=base.replace('int AIBuildableWallSegment::canMake(Player *player)',helper+'int AIBuildableWallSegment::canMake(Player *player)')
  s=s.replace('return 0;','return wallResult(0);')
  if mode=='all':
   if typ=='bool':continue
   s=s.replace('return 9;','return wallResult(9);').replace('return 2;','return wallResult(2);')
  forms['inline_'+typ.replace(' ','_')+'_'+mode]=s
name='?canMake@AIBuildableWallSegment@@UAEHPAVPlayer@@@Z'
results=[]
for key,s in forms.items():
 p=out/(key+'.cpp');p.write_bytes(s.encode())
 r=subprocess.run([sys.executable,'tools/explain_mismatch.py',name,'--rva','0x00596E15','--size','218','--source',p.as_posix()],stdout=subprocess.PIPE,stderr=subprocess.STDOUT,text=True)
 p.with_suffix('.txt').write_bytes(r.stdout.encode())
 d={'variant':key,'exit':r.returncode,'summary':[x for x in r.stdout.splitlines() if x.startswith(('compiled symbol size:','classification:','first diff:','unresolved:'))]}
 results.append(d);print(d,flush=True)
 (out/'results.json').write_bytes((json.dumps(results,indent=2)+'\n').encode())
