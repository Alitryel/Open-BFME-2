import json
import re
import subprocess
import sys
from pathlib import Path

root = Path(__file__).resolve().parents[2]
scratch = root / 'build/resume-eh-local'
bank = (root / 'reverse/attempts/0x005a9f70.cpp').read_text()
bank = bank.replace('// cl: ', '// cl: /I. ', 1)
bank = bank.replace('../../../Libraries/Include/Lib/Coord3D.h',
                    'Code/Libraries/Include/Lib/Coord3D.h')
can_run = re.search(r'bool AISimpleSiegeTactic::canRun\(void \*request\)\n\{.*?\n\}',
                    bank, re.S)
assert can_run
contexts = {'bank': bank, 'current-home': bank[:can_run.start()] + bank[can_run.end():]}
results = []
for context, source in contexts.items():
    for profile in ('default', 'vc71-sp1', 'vc71-ddk-4035'):
        label = context + '-' + profile
        trial = source if profile == 'default' else '// compiler: ' + profile + '\n' + source
        path = scratch / ('turn47-siege-' + label + '.cpp')
        path.write_text(trial)
        proc = subprocess.run([sys.executable, str(root / 'tools/explain_mismatch.py'),
            '?initializeTeamTemplate@AISimpleSiegeTactic@@UAE_NPAVTeamPrototype@@H@Z',
            '--source', path.relative_to(root).as_posix(), '--rva', '0x005A9F70', '--size', '214'],
            cwd=root, capture_output=True, text=True)
        log = scratch / ('turn47-siege-' + label + '.txt')
        log.write_text(proc.stdout + proc.stderr)
        size = re.search(r'compiled symbol size: (\d+)', proc.stdout)
        diffs = [s for s in proc.stdout.splitlines() if s.startswith('!=')]
        result = {'variant': label, 'exit': proc.returncode,
                  'compiled_size': int(size[1]) if size else None,
                  'differing_instruction_rows': len(diffs),
                  'exact': 'OK: bytes match' in proc.stdout,
                  'log': log.relative_to(root).as_posix()}
        results.append(result)
        print(json.dumps(result), flush=True)
(scratch / 'turn47-siege-results.json').write_text(json.dumps(results, indent=2) + '\n')
