from pathlib import Path
from itertools import product
W=Path(__file__).resolve().parent
out=W/'callbridge';out.mkdir(exist_ok=True)
for idx,(model,order,locals_,types,sites,bank) in enumerate(product(range(4),range(2),range(4),range(4),range(8),range(2))):
 s=(W/('diagnostic.cpp' if bank else 'base.cpp')).read_text()
 defs=''
 pk='const ContactVector *p' if types&1 else 'const ContactVector &p'
 nk='const ContactVector *n' if types&2 else 'const ContactVector &n'
 pv='*p' if types&1 else 'p';nv='*n' if types&2 else 'n'
 for which in range(3):
  name=['doTest','doResponse','doReverse'][which]
  ret='bool' if which==0 else 'void';method='test' if which==0 else 'respond'
  if model==0:
   head=['ContactData *a','ContactData *b'];a='a->source';b='b->source'
  elif model==1:
   head=['ContactNode *node'];a='node->'+('second' if which==2 else 'first')+'->source';b='node->'+('first' if which==2 else 'second')+'->source'
  elif model==2:
   head=['ContactData *b'];a='source';b='b->source'
  else:
   head=[];a=('second' if which==2 else 'first')+'->source';b=('first' if which==2 else 'second')+'->source'
  pre=''
  if locals_ in [1,2]:
   da='ContactSource *x='+a+';';db='ContactSource *y='+b+';'
   pre+=db+da if locals_==2 else da+db;a='x';b='y'
  prep='altPrepare('+b+','+pv+','+nv+')' if bank and which==2 else b+'->prepare('+pv+','+nv+')'
  if locals_==3:pre+='void *info='+prep+';';prep='info'
  expr=a+'->'+method+'('+prep+')'
  params=head+([nk,pk] if order else [pk,nk])
  defs+='__forceinline '+ret+' '+name+'('+','.join(params)+'){'+pre+('return ' if which==0 else '')+expr+';}\n'
 if model==0 or model==1:s=s.replace('class BfmeThingCGD',defs+'class BfmeThingCGD',1)
 elif model==2:s=s.replace('ContactSource *source; };','ContactSource *source; '+defs+' };')
 else:s=s.replace('ContactNode *next;','ContactNode *next;\n'+defs)
 for which in range(3):
  outer='second' if which==2 else 'first';inner='first' if which==2 else 'second'
  n='ContactVector(node->normal).negate()' if which==2 else 'node->normal'
  prep='altPrepare(node->'+inner+'->source,node->point,'+n+')' if bank and which==2 else 'node->'+inner+'->source->prepare(node->point,'+n+')'
  old='node->'+outer+'->source->'+('test' if which==0 else 'respond')+'('+prep+')'
  assert old in s,(idx,which)
  if not sites&(1<<which):continue
  vals={'p':'&(node->point)' if types&1 else 'node->point','n':'&('+n+')' if types&2 else n}
  args=([ 'node->'+outer,'node->'+inner] if model==0 else ['node'] if model==1 else ['node->'+inner] if model==2 else [])+[vals[k] for k in (['n','p'] if order else ['p','n'])]
  receiver='' if model in [0,1] else 'node->'+outer+'->' if model==2 else 'node->'
  new=receiver+['doTest','doResponse','doReverse'][which]+'('+','.join(args)+')'
  s=s.replace(old,new,1)
 (out/('v%04d.cpp'%idx)).write_text(s)
print('generated',idx+1)
