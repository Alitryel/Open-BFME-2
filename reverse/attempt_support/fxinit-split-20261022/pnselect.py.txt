import sys,itertools,json,time
from pathlib import Path
from concurrent.futures import ThreadPoolExecutor
sys.path.insert(0,'build/fxinit-seat22');import search as h
h.HERE=Path('build/fxinit-split22').resolve()
LOOP='   for(unsigned i=0;i<(unsigned)mesh->Get_Vertex_Count();++i) {'
def variants():
 for which,form,stage,scope,order,init in itertools.product(range(3),range(4),range(3),range(3),range(2),range(2)):
  src=h.BASE;arm=h.ARM;decl=[];sets=[];zeros=[];pn=[]
  selected=[0,1] if which==0 else [which-1]
  if order:selected=selected[::-1]
  for k in selected:
   name='chosen'+str(k);stream=['positions[1]','normals[1]'][k];offset=['desc.Get_Location_Offset(1)','desc.Get_Normal_Offset(1)'][k];old=[h.P,h.N][k]
   ty=['const Vector3 *','const Vector3 *','unsigned ','const Vector3 *volatile '][form]
   value=stream if form!=2 else '(unsigned)'+stream
   # The prebiased variant retains the original pointer-null test.
   if form==1:value=stream+' ? '+stream+'+i : 0'
   decl.append(ty+name+';');sets.append(name+'='+value+';');zeros.append(name+'=0;')
   rhs='*'+name if form==1 else ('((const Vector3 *)'+name+')[i]' if form==2 else name+'[i]')
   pn.append('if('+name+') *(Vector3 *)(vertices+'+offset+')='+rhs+';')
   arm=arm.replace(old,'')
  setcode=''.join(sets);zero=''.join(zeros)
  if stage==0:arm=arm.replace(h.R,setcode+h.R)
  if stage==1:arm=arm.replace(h.W,h.W+setcode)
  if stage==2:arm=arm.replace('else if(m_BonesPerVertex==2) {','else if(m_BonesPerVertex==2) {'+setcode)
  before=''.join(decl)
  if init==0:before+=zero
  else:
   # Init to the real pointer, explicitly null it on each skipped path.
   arm=arm.replace(setcode,'')
   before+=setcode
   arm=arm.replace('if(m_Passes>1) *(unsigned *)(vertices+desc.Get_Basis_End())=pass;','if(m_Passes>1) {'+zero+'*(unsigned *)(vertices+desc.Get_Basis_End())=pass;}')
   arm=arm.replace('if(m_BonesPerVertex==1) *(int *)(vertices+desc.Get_Basis_End())=remap[links[0][i]];','if(m_BonesPerVertex==1) {'+zero+'*(int *)(vertices+desc.Get_Basis_End())=remap[links[0][i]];}')
   at=arm.rfind('     }');arm=arm[:at]+arm[at:].replace('     }','     } else {'+zero+'}',1)
   arm=arm.rstrip()+' else {'+zero+'}\n'
  arm+=''.join(pn)+'\n'
  if scope==0:arm=before+arm
  src=src[:h.START]+arm+src[h.END:]
  if scope==1:src=src.replace(LOOP,LOOP+before)
  if scope==2:
   src=src.replace('  for(unsigned pass=0;', ''.join(decl)+'\n  for(unsigned pass=0;')
   src=src.replace(LOOP,LOOP+(zero if init==0 else setcode))
  yield f'pnselect_{which}{form}{stage}{scope}{order}{init}',src
if __name__=='__main__':
 t=time.time()
 with ThreadPoolExecutor(max_workers=8)as pool:r=list(pool.map(h.run_one,variants()))
 r.sort(key=lambda x:(x.get('nd',99999),x.get('cost',99999)))
 (h.HERE/'pnselect-results.json').write_text(json.dumps(r,indent=2))
 print(json.dumps({'count':len(r),'valid':sum('nd'in x and not x.get('unresolved')for x in r),'seconds':time.time()-t,'best':r[:8]},indent=2))
