from driver import *
def gen():
 yield 'order-control',BASE
 blocks=re.findall(r'(                int pixelSample0=.*?                int pixelSample8=.*?;\n)',BASE,re.S)
 assert len(blocks)==2
 for group,p,q in itertools.product(range(3),itertools.permutations(range(5)),itertools.permutations(range(5,9))):
  b=BASE
  order=p+q
  for j in ([0] if group==0 else ([1] if group==1 else [0,1])):
   lines=blocks[j].splitlines(True)
   assert len(lines)==9
   b=b.replace(blocks[j],''.join(lines[k] for k in order))
  yield 'order-%d-%s'%(group,''.join(map(str,order))),b
if __name__=='__main__':run(list(gen()),'loadorders')
