from search import *

def source(n):
    head,body=HEAD,BODY
    head=head.replace('struct MpGameSetupRules\n{','''struct MpGameSetupRules
{
    int getRule(int i) const { return ((const int *)this)[i]; }
    int getRule2(int i) const { return getRule(i); }
    const MpGameSetupRules &getRules() const { return *this; }
    MpGameSetupRules copy() const { return *this; }
''')
    head=head.replace('class Rva0057EE5C\n{','struct MpGameSetupRules;\nclass Rva0057EE5C\n{')
    head=head.replace('void rva0057F0AA();','''void rva0057F0AA();
    const MpGameSetupRules &getRules() const { return *(const MpGameSetupRules *)((const char *)this+0x8c); }
''')
    mode=n%8
    access=['m_rules','m_d0.getRules()','getRulesView()','getRulesCopy()',
            'm_rules.getRules()','m_d0.getRules().getRules()','m_rules.copy()','getRulesView().copy()'][mode]
    head=head.replace('\nprivate:\n\tunsigned char m_pad000','''
    const MpGameSetupRules &getRulesView() const { return m_rules; }
    MpGameSetupRules getRulesCopy() const { return m_rules; }
private:
    unsigned char m_pad000''')
    for i,f in enumerate(['m_00','m_04']):
        getter='getRule2' if n & (8<<i) else 'getRule'
        body=body.replace('m_rules.'+f,access+'.'+getter+'(%d)'%i)
    if n&32:
        head=head.replace('virtual bool v12();','virtual bool v12() const;').replace('virtual int v13();','virtual int v13() const;')
    if n&64:
        head=head.replace('virtual bool v12();','virtual bool v12() throw();').replace('virtual int v13();','virtual int v13() throw();')
        head=head.replace('virtual bool v12() const;','virtual bool v12() const throw();').replace('virtual int v13() const;','virtual int v13() const throw();')
    if n&128:
        head=head.replace('bool isObserver() const;','bool isObserver() const throw();').replace('bool isAI() const;','bool isAI() const throw();')
    return head+MARK+body

if __name__=='__main__':run([('rules-%03d'%n,source(n)) for n in range(256)],'rules')
