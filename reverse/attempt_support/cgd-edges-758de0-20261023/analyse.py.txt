"""Compact address definitions and verify completed read-only c2 traces."""
import hashlib,json,subprocess,sys
from pathlib import Path
import search
W=Path(__file__).resolve().parent
checks=[]
for tag,name in [('early-edge','v0787'),('early-uniform','v0786')]:
 snapshots=json.loads((W/(tag+'.json')).read_text());stages=[]
 for s in snapshots:
  nodes=s['nodes'];defs=[]
  for bb,bw in s['bbs'].items():
   p=bw[7];seen=set()
   while p and p not in seen:
    seen.add(p);v=nodes[str(p)]
    if (v[2]&255) in (13,18):
     src=nodes.get(str(v[10]),[]);dst=nodes.get(str(v[11]),[])
     if len(src)>13 and (src[2]&255)==5 and src[12] in (20,32) and len(dst)>7 and (dst[2]&255) in (1,2,4):
      ptr=dst[7] or dst[6];sw=s['syms'].get(str(ptr),[])
      reg=(ptr-0x108a97a4)//96 if 0x108a9804<=ptr<0x108a9b04 else None
      defs.append(dict(bb=bb,il=hex(p),offset=src[12],destination_symbol=hex(ptr),word7=sw[7] if len(sw)>7 else None,physical=reg))
    if p==bw[8]:break
    p=v[0]
  stages.append(dict(stage=s['stage'],definitions=defs))
 (W/(tag+'-address-stages.json')).write_text(json.dumps(stages,indent=1)+'\n')
 selected=[s for s in snapshots if s['stage'] in ('0x1077c0c8','0x107821a7','0x107b0037')]
 selectedpath=W/(tag+'-selected.json');selectedpath.write_text(json.dumps(selected))
 text=subprocess.check_output([sys.executable,str(W/'show_ir.py'),str(selectedpath)],text=True)
 (W/(tag+'-ir.txt')).write_text(text)
 a,r=search.build.read_object_symbol_bytes(W/(tag+'.obj'),search.symbol,code_only=True)
 b=(W/'edges'/(name+'.bin')).read_bytes()
 terminated='compiler terminated' in (W/(tag+'.out')).read_text()
 assert a==b and not r and terminated
 checks.append(dict(trace=tag,source='edges/'+name+'.cpp',size=len(a),relocations=len(r),sha256=hashlib.sha256(a).hexdigest(),equal_to_untraced=True,compiler_terminated=terminated,checkpoints=[s['stage'] for s in snapshots]))
(W/'trace-checks.json').write_text(json.dumps(checks,indent=2)+'\n')
bank=search.run(W/'base.cpp')
assert bank['size']==333 and bank['relocs']==0 and bank['sha']=='01e69aef7e003a734f1bc04dc4faf28fd44f3c0d6757d8db6f5fb935382ab90e'
(W/'bank-check.json').write_text(json.dumps(bank,indent=2)+'\n')
blobs={'retail':search.target,'bank':(W/'base.bin').read_bytes(),
 'edge-bridge':(W/'edges/v0787.bin').read_bytes(),
 'edge-direct':(W/'edges/v0785.bin').read_bytes(),
 'edge-uniform':(W/'edges/v0786.bin').read_bytes()}
lines=[]
for label,blob in blobs.items():
 lines.append(label+' '+hashlib.sha256(blob).hexdigest())
 for m,o,z,addr in search.insns(blob):
  if 0x90<=addr<0xe1 or 0x12c<=addr<0x142:lines.append('%03x %-7s %s'%(addr,m,o))
 lines.append('')
(W/'call-windows.txt').write_text('\n'.join(lines))
print(json.dumps(checks,indent=2))
