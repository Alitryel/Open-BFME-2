from wdbg import Dbg
from pathlib import Path
import sys,json,time
W=Path('build/locomotor-zero-cfg'); label=sys.argv[1] if len(sys.argv)>1 else 'baseline'
d=Dbg(str(W/(label+'.cpp')),str(W/(label+'-costwrites.obj')),str(W/(label+'-costwrites-debugger.log')))
fn='';cls=-1;rows=[];start=time.time();armed=False;active=False;steps=0
extra=[0x82692,0x84c3e,0x8281c,0x8cb73,0x82a4d,0x82273]
def zero(p):
 w=d.words(p,30)
 if len(w)<30:return
 sy=d.words(w[0],16)
 if len(sy)<11 or sy[:2]!=[0,0x100d]:return
 op=d.words(sy[10],8)
 if len(op)<7 or op[1]!=0x24b or op[4:7]!=[0,0,0]:return
 return dict(addr=p,w=w,sym=sy,op=op)
def tup(p):
 w=d.words(p,16); ops=[]
 q=w[11] if len(w)>11 else 0;seen=set()
 while q and q not in seen and len(ops)<16:
  seen.add(q);v=d.words(q,12);ops.append(dict(addr=q,w=v))
  q=v[0] if v else 0
 return dict(addr=p,w=w,ops=ops)
try:
 d.prompt();d.prompt('set $CanDeferOnBPByAddr = 1');d.prompt('break *0x10783acd')
 for step in range(15000):
  out=d.prompt('cont');steps+=1
  if 'terminated' in out:break
  if time.time()-start>180:raise TimeoutError('overall deadline')
  off=d.reg('eip')-0x10700000
  if off==0x83acd:
   ebp=d.reg('ebp');p=d.words(ebp,1)[0];fn=d.string(d.words(p+4,1)[0]);cls=d.reg('ebx')
   active='rva001E7ECA' in fn and cls==0
   if active and not armed:
    armed=True
    for v in extra:d.prompt('break *0x%x'%(0x10700000+v))
  if not active:continue
  if off==0x82273:
   active=False
   for i in range(2,8):d.prompt('delete '+str(i))
   print('finished',steps,'steps',len(rows),'events',flush=True)
   continue
  if off==0x82a4d:
   p=d.reg('eax');w=d.words(p,30);sy=d.words(w[0],16)
   rows.append(dict(phase='color',addr=p,w=w,sym=sy))
   continue
  if off==0x82692:p=d.reg('esi');new=d.reg('ebx');t=d.reg('ebp')
  elif off==0x84c3e:p=d.reg('ebp');new=d.reg('ebx');t=d.reg('esi')
  elif off==0x8281c:p=d.reg('ebp');new=d.reg('edx');t=d.reg('esi')
  elif off==0x8cb73:p=d.reg('esi');new=d.reg('ecx');t=d.words(d.reg('esp')+0x40,1)[0]
  else:continue
  z=zero(p)
  if z:
   rows.append(dict(phase=hex(off),old=z['w'][16],new=new,row=z,tuple=tup(t)))
 (W/(label+'-costwrites.json')).write_text(json.dumps(rows,indent=2))
 print('total',steps,'steps',len(rows),'events',flush=True)
finally:
 if d.p.poll() is None:
  try:d.prompt('quit')
  except Exception:d.p.kill()
 d.log.close()
