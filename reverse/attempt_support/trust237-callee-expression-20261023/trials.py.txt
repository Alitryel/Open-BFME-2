from pathlib import Path
import json,hashlib,subprocess,sys,struct
sys.path.insert(0,'tools')
import build
from pe_imports import read_imports,coff_import_names
root=Path('build/trust237-callee-expression')
root.mkdir(parents=True,exist_ok=True)
s=Path('reverse/attempt_support/trust237-patched-target-20261023/mixed-candidate.cpp.txt').read_text()
old='''    __asm {
        mov ebx,eax
    }
    LocalFree(argv);'''
variants={
'comma-void':('__forceinline void saveStatus(){ __asm mov ebx,eax }','(saveStatus(), LocalFree)(argv);'),
'comma-pointer':('__forceinline void saveStatus(){ __asm mov ebx,eax }','(saveStatus(), &LocalFree)(argv);'),
'callee-pointer':('typedef void *(__stdcall *FreeArgs)(void *);\n__forceinline FreeArgs saveStatus(){ __asm mov ebx,eax\n return LocalFree; }','saveStatus()(argv);'),
'callee-reference':('typedef void *(__stdcall FreeArgs)(void *);\n__forceinline FreeArgs &saveStatus(){ __asm mov ebx,eax\n return LocalFree; }','saveStatus()(argv);'),
'callee-pointer-nothrow':('typedef void *(__stdcall *FreeArgs)(void *);\n__forceinline FreeArgs saveStatus() throw(){ __asm mov ebx,eax\n return LocalFree; }','saveStatus()(argv);'),
'callee-template':('template<class T> __forceinline T saveStatus(T freeArgs){ __asm mov ebx,eax\n return freeArgs; }','saveStatus(LocalFree)(argv);'),
}
(root/'base.cpp').write_text(s)
assert old in s
for name,(helper,call) in variants.items():
    for internal in (False,True):
        h=helper
        if internal:h=h.replace('__forceinline','static __forceinline')
        candidate=s.replace('int Rva00237D3DCheck() {',h+'\nint Rva00237D3DCheck() {').replace(old,'    '+call)
        (root/(('static-' if internal else '')+name+'.cpp')).write_text(candidate)
for src in root.glob('*.cpp'):
    subprocess.run([sys.executable,'tools/variant.py',src.as_posix(),'Rva00237D3DCheck','0x237D3D','226','_memset=0x6291AE','_WinVerifyTrust@12=0x655652'],check=True)
sym='?Rva00237D3DCheck@@YAHXZ'
target=build.read_target_bytes(0x237D3D,226)
imports=read_imports(build.EXE.read_bytes())
symbols=build.load_symbol_map()
# Diagnostic only: this name's real import thunk is rowed as ji_00655652.
symbols['_WinVerifyTrust@12']=[0x655652]
base,base_reloc=build.read_object_symbol_bytes(Path('build/scratch/base.obj'),sym,code_only=True)
records=[]
for src in sorted(root.glob('*.cpp')):
    obj=Path('build/scratch')/(src.stem+'.obj')
    raw,reloc=build.read_object_symbol_bytes(obj,sym,code_only=True)
    row={'name':sym,'target_rva':'0x00237D3D','target_size':'226','source':src.as_posix(),'status':'matched','notes':''}
    patch=build.compile_function(row,symbols,obj)
    diffs=[i for i,(a,b) in enumerate(zip(target,patch['bytes'])) if a!=b]
    assert len(raw)==226 and raw==base and reloc==base_reloc
    assert not patch['unresolved'] and diffs==[0xC2,0xC3,0xC4]
    assert all(not name.startswith('?saveStatus') for _,kind,name in reloc if kind==0x14)
    checked=[]
    for offset,kind,name in reloc:
        if kind==6 and name.startswith('__imp_'):
            slot=(struct.unpack_from('<I',target,offset)[0]-struct.unpack_from('<I',raw,offset)[0])&0xFFFFFFFF
            entry=imports[slot]
            assert entry.name in coff_import_names(name)
            checked.append(entry.name)
    assert len(checked)==3
    records.append({'variant':src.stem,'size':len(raw),'source_sha256':hashlib.sha256(src.read_bytes()).hexdigest(),'raw_sha256':hashlib.sha256(raw).hexdigest(),'object_sha256':hashlib.sha256(obj.read_bytes()).hexdigest(),'differences':[hex(x) for x in diffs],'relocations':reloc,'checked_imports':checked,'defined_code_symbols':sorted(build.defined_code_symbols(obj))})
(root/'evidence.json').write_text(json.dumps({'trials':records},indent=2)+'\n')
print('Twelve variants plus baseline: identical full226B and relocations; C2..C4 remains.')
