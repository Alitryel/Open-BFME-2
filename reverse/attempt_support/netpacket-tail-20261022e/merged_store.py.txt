import itertools
from search import *
SIGN='NetCommandList *NetPacket::getCommandList()\n';H,B=BANK.split(SIGN,1)
SELECT=['m_packetLen & 1','m_packetLen > 0','m_packetLen > offset','relay & 1','commandType & 1','commandID & 1','i & 1','lastCommand != 0']
def dup(s,c,form):
 if form==0:return 'if ('+c+') { '+s+' } else { '+s+' }'
 if form==1:return 'switch (('+c+') ? 1 : 0) { case 0: { '+s+' } break; default: { '+s+' } break; }'
 if form==2:return 'if ('+c+') { '+s+' goto mergedStoreDone; } '+s+' mergedStoreDone:;'
 return 'if ('+c+') { Int next = offset; '+s.replace('i = offset;','i = next;')+' } else { Int next = offset; '+s.replace('i = offset;','i = next;')+' }'
def change(s,kind,c,form,label):
 end=s.index('i = offset;')+len('i = offset;')
 starts=['i = offset;','msg->detach();','lastCommand = new NetCommandRef(msg);','lastCommand = 0;\n\t\t\tlastCommand = new NetCommandRef(msg);']
 if kind==3:
  ref=s.index('lastCommand = new NetCommandRef(msg);')
  start=s[:ref].rfind('delete lastCommand;') if s[:ref].endswith('lastCommand = 0;\n\t\t\t') else s[:ref].rfind('if (lastCommand != 0)')
 else:start=s.index(starts[kind])
 old=s[start:end]
 return s[:start]+dup(old,c,form).replace('mergedStoreDone',label)+s[end:]
items=[]
for sel,kind,form,location in itertools.product(range(8),range(4),range(4),range(3)):
 d,z=B.split("} else if (m_packet[i] == 'Z')",1)
 if location in (1,2):d=change(d,kind,SELECT[sel],form,'dStoreDone')
 if location in (0,2):z=change(z,kind,SELECT[sel],form,'zStoreDone')
 items.append((f'mstore_{sel}{kind}{form}{location}',H+SIGN+d+"} else if (m_packet[i] == 'Z')"+z))
if __name__=='__main__':batch(items,'merged-store')
