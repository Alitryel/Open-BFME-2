import itertools,re,sys
from search import *
BASE=(HERE/'base.cpp').read_text()
def make(phase,keep):
 s=BASE
 for bit,helper in enumerate(['aptMatrixMulUVFirst','aptMatrixMulNamedY','aptMatrixMulNamedProducts']):
  if not phase&(1<<bit):continue
  h=re.search(r'static __forceinline void '+helper+r'\([^}]+\}',s)[0]
  h1=re.sub(r'out.X = ([^;]+);',lambda m:'out.X = ('+m[1]+');',h)
  s=s.replace(h,h1)
 def pad(m):
  n=int(m[1]);g=0 if n<5 else 1 if n<8 else 2;idx=n-[0,5,8][g]
  return m[0] if idx<keep[g] else ''
 s=re.sub(r'^\s*int unusedSlot(\d+) = 0;\n',pad,s,flags=re.M)
 return s
if __name__=='__main__':
 if len(sys.argv)>1 and sys.argv[1]=='control':batch([('phase_control',make(0,[5,3,5]))]);raise SystemExit
 batch([(f'phase_{phase}_{a}{b}{c}',make(phase,[a,b,c])) for phase,a,b,c in itertools.product(range(8),range(6),range(4),range(6))])
