import sys,os
sys.path.insert(0,'/work/Open-BFME-2/build/c2tr')
from wdbg import Dbg
src,obj=sys.argv[1:3]; WI=int(os.environ.get('WI','11')); WA=[int(x,16) for x in os.environ.get('WATCH','').split(',') if x]
d=Dbg(src,obj,obj+'.log')
d.prompt();d.prompt('set $CanDeferOnBPByAddr = 1')
B=0x10700000
bps=[0x83acd,0x82978,0x821a7]; N821=int(os.environ.get('N821','-1')); c821=0; added=set()
for b in bps:
    o=d.prompt('break *0x%x'%(B+b))
    if 'Breakpoint' in o and 'Unable' not in o: added.add(b)
def rec_of(i):
    p=d.words(0x108807b8+(i&0x3ff)*4,1)[0]
    while p:
        w=d.words(p,13)
        if len(w)<13: return None,None
        if w[7]==i: return p,w
        p=w[12]
    return None,None
on=False;k=0;nh=0
while True:
    out=d.prompt('cont')
    if 'terminated' in out: break
    eip=d.reg('eip')-B
    for b in bps:
        if b not in added and b!=eip: d.prompt('break *0x%x'%(B+b)); added.add(b)
    added.add(eip)
    if eip==0x821a7:
        c821+=1
        if c821==N821:
            for a in WA: print('arm',d.prompt('watch *(int*)0x%x'%a).strip()[-80:],flush=True)
            on=True; k=0
        continue
    if eip==0x83acd:
        if N821>0 and c821>=N821: continue
        ebp=d.reg('ebp'); f=d.words(ebp,1)[0]; curfn=d.string(d.words(f+4,1)[0]); on='updateMeleeAttack' in curfn and d.reg('ebx')==0; k=0
        if on: print('c821',c821,flush=True)
        if on and os.environ.get('BT'):
            e=d.reg('ebp'); esp=d.reg('esp'); print('esp-ret',hex(d.words(esp,1)[0]-B))
            st=d.words(esp,64); print('stack ras',[hex(x-B) for x in st if B<x<B+0x200000])
            for i in range(10):
                w=d.words(e,2)
                if len(w)<2: break
                print('ra',hex(w[1]-B)); e=w[0]
            break
        if on and WA:
            for a in WA: print(d.prompt('watch *(int*)0x%x'%a).strip()[-200:],flush=True)
        continue
    if not on: continue
    if eip==0x82978:
        k+=1
        if k==1:
            p,w=rec_of(WI)
            print('rec',hex(p) if p else None,[hex(x) for x in d.words(p,20)] if p else '',flush=True)
            if p:
                ch=d.words(p+0x20,1)[0]; print('w8hdr',hex(ch)); c=d.words(ch,1)[0]
                while c:
                    ww=d.words(c,3); print(' chunk',hex(c),[hex(x) for x in ww],flush=True); c=ww[1]
            if not WA: break
        continue
    # watch hit
    nh+=1
    if nh<60:
        bt=d.prompt('bt 6')
        fr=[l.strip()[:60] for l in bt.splitlines() if l.strip().startswith(('=>','0','1','2','3','4','5'))]
        print('hit eip=%x k=%d vals=%s'%(eip,k,[hex(d.words(a,1)[0]) for a in WA]),'|',' / '.join(fr[:6]),flush=True)
print('done',flush=True)
