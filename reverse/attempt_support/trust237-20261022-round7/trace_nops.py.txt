from pathlib import Path
import sys, json, importlib.util, shutil, re, hashlib
ROOT=Path.cwd(); OUT=ROOT/'build/trust237-round7'
original=(ROOT/'reverse/attempt_support/a9f98-c2keys-20261022/wdbg.py.txt').read_text()
(OUT/'wdbg.py').write_text(original.replace("ROOT=Path('/work/Open-BFME-2')", 'ROOT=Path.cwd()'))
sys.path.insert(0,str(OUT));from wdbg import Dbg
src=Path(sys.argv[1]).resolve();name=sys.argv[2]
obj=OUT/(name+'.obj')
if obj.exists():obj.unlink()
d=Dbg(str(src),str(obj),str(OUT/(name+'.debug.log')))
records=[];ended=False
try:
    d.prompt();d.prompt('set $CanDeferOnBPByAddr = 1')
    # Deferred breakpoints have no number until c2 loads. Set conditions only
    # after a known function-compilation breakpoint resolves and stops.
    d.prompt('break *0x10760539')
    first=d.prompt('cont',45)
    if 'terminated' in first:raise RuntimeError('compiler exited before c2 function setup')
    assert d.reg('eip')==0x10760539,first
    d.prompt('delete 1')
    response=d.prompt('break *0x107a7dd8')
    bp=int(re.search(r'Breakpoint (\d+)',response)[1])
    d.prompt('condition %d *(unsigned int *)($esp+4) == 0xa9'%bp)
    sites=[0x10734489,0x107a7ec9,0x107c4dba,0x1082f17a,0x107b0da2,0x1082eea8]
    if '--detail' in sys.argv:sites += [0x1082ef83,0x1082efa3]
    for pc in sites:
        d.prompt('break *0x%x'%pc)
    for i in range(256):
        out=d.prompt('cont',45)
        if 'terminated' in out or not out.strip(): ended=True;break
        pc=d.reg('eip');esp=d.reg('esp');eax=d.reg('eax')
        row={'pc':hex(pc),'stack':d.words(esp,8),'eax':hex(eax)}
        if pc in [0x107b0da2,0x1082eea8]:
            settings=d.words(0x10883370,1)[0]
            row['settings_1e8']=d.words(settings+0x1e8,3)
            row['guards_108b0258']=d.words(0x108b0258,2)
        elif pc in [0x1082ef83,0x1082efa3]:
            row['edi']=d.reg('edi')
            row['ebp_node']=d.words(d.reg('ebp'),14)
            row['esi_node']=d.words(d.reg('esi'),14)
        elif pc!=0x107a7dd8:row['node']=d.words(eax,14)
        records.append(row);print(name,json.dumps(row),flush=True)
    else:raise RuntimeError('breakpoint limit')
    assert obj.exists(),'compiler made no object'
finally:
    (OUT/(name+'.trace.json')).write_text(json.dumps(dict(ended=ended,source_sha256=hashlib.sha256(src.read_bytes()).hexdigest(),object_exists=obj.exists(),records=records),indent=2)+'\n')
    if d.p.poll() is None:
        try:d.prompt('quit')
        except Exception:d.p.kill()
    d.log.close()
print(name,'completed',ended,'hits',len(records),flush=True)
