"""Recombine the older ESI selector and new return-holder cleanup lifetime."""
from pathlib import Path
import concurrent.futures,itertools,json,re,runpy
OUT=Path('build/canmake-alias-20261023/lifetime-esi').resolve();OUT.mkdir(parents=True,exist_ok=True)
r=runpy.run_path('reverse/attempt_support/canmake-00596e15-20261022/replay.py.txt');r['run'].__globals__['OUT']=OUT
# Reconstruct the frozen 223 B ESI control without requiring a scratch input.
n=runpy.run_path('reverse/attempt_support/canmake-early-cleanup-20261022/replay.py.txt');e=n['context']();prefix=e['prefix'];body=e['win'].replace('if (money < result.m_pad04)\n return 2;','if (money < result.m_pad04) return (player->m_money < (unsigned int)result.m_pad04)?2:0;')
assert '?2:0;'in body
dtors=['','code=0;','code=~code;','if(code!=0)code=0;','code+=1;','code=code?2:9;']
tests=[]
for dtor,where,mask,conditional in itertools.product(range(6),range(3),range(16),range(2)):
 b=body;p='struct MakeCodeLife { int code; __forceinline MakeCodeLife():code(0){} __forceinline ~MakeCodeLife(){'+dtors[dtor]+'} };\n'+prefix
 decl='MakeCodeLife selected;\n'
 if where==0:b=decl+b
 elif where==1:b=b.replace('Rva0039205C result;',decl+'Rva0039205C result;')
 else:b=b.replace('if (failed)',decl+'if (failed)')
 count=[0]
 def replace(m):
  v=int(m[1]);site=3 if v==0 else count[0]
  if v:count[0]+=1
  return '{selected.code='+str(v)+';return selected.code;}'if mask&(1<<site)else m[0]
 b=re.sub(r'return (0|2|9);',replace,b)
 if conditional:b=b.replace('return (player->m_money < (unsigned int)result.m_pad04)?2:0;','{selected.code=(player->m_money < (unsigned int)result.m_pad04)?2:0;return selected.code;}')
 tests.append((f'esi-life-{dtor}{where}-{mask}-{conditional}',p+'\nint AIBuildableWallSegment::canMake(Player *player) {\n'+b+'\n}\n'))
results=[]
with concurrent.futures.ThreadPoolExecutor(max_workers=8)as pool:
 for i,d in enumerate(pool.map(r['run'],tests),1):
  results.append(d)
  if i%192==0 or d.get('exact_candidate'):print(json.dumps(dict(done=i,total=len(tests),best=sorted(results,key=lambda x:(x.get('nd',999),-x.get('score',0)))[:2])),flush=True)
(OUT/'results.json').write_text(json.dumps(results,indent=2)+'\n')
print(json.dumps(dict(trials=len(results),errors=sum('error'in x for x in results),unresolved=sum(bool(x.get('unresolved'))for x in results),exact=sum(x.get('exact_candidate',False)for x in results),bodies=len({x.get('masked_code_sha256')for x in results}))),flush=True)
