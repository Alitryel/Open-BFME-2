import sys,struct
def secs(obj):
    n=struct.unpack_from('<H',obj,2)[0]; opt=struct.unpack_from('<H',obj,16)[0]
    for i in range(n):
        o=20+opt+40*i; nm=obj[o:o+8].rstrip(b'\0'); sz,ptr=struct.unpack_from('<II',obj,o+16)
        yield nm,obj[ptr:ptr+sz]
def recs(d):
    p=4
    while p+8<=len(d):
        t,l=struct.unpack_from('<II',d,p)
        if l==0: l=len(d)-p-8
        b=d[p+8:p+8+l]; p+=8+l; p=(p+3)&~3
        if t!=0xf1: continue
        q=0
        while q+4<=len(b):
            rl,rt=struct.unpack_from('<HH',b,q); r=b[q+4:q+2+rl]; q+=2+rl
            nm=r.rstrip(b'\0').split(b'\0')[-1] if b'\0' in r else b''
            # name is last cstring
            if rt==0x110b: off=struct.unpack_from('<i',r,0)[0]; yield 'BPREL %d %s'%(off,r[8:].split(b'\0')[0].decode())
            elif rt==0x1108: yield 'UDT '+r[4:].split(b'\0')[0].decode()
            elif rt==0x1105: yield 'LABEL '+r[7:].split(b'\0')[0].decode()
            elif rt==0x110c: yield 'LDATA '+r[10:].split(b'\0')[0].decode()
            elif rt==0x1110: yield 'PROC '+r[r.index(b'\x01',30)+1:].split(b'\0')[0].decode() if b'\x01' in r[30:] else 'PROC ?'
            elif rt==0x1103: yield 'BLOCK'
            elif rt==0x6: yield 'END'
            else: yield 'rec%04x'%rt
obj=open(sys.argv[1],'rb').read()
on=False
for nm,d in secs(obj):
    if nm==b'.debug$S':
        for r in recs(d):
            if r.startswith('PROC'): on=sys.argv[2] in r
            if on: print(r)
