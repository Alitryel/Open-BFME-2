from pathlib import Path
import itertools,sys
ROOT=Path.cwd();OUT=ROOT/'build/trust237-round14';OUT.mkdir(exist_ok=True)
sys.path.insert(0,str(ROOT/'build/trust237-round9'));import frontends as f
f.OUT=OUT
# Follow round13's 227-byte region winner: its return suffix is one byte late.
# Unlike the earlier constant-mask type matrix, vary a status-dependent bound.
# For unsigned values (lo & bound) <= bound is true at every bit pattern.
bounds=['u|1u','u^1u','u>>1','0u-u','(u<<1)|1u',
        '(unsigned char)u|1u','(unsigned char)u^1u','~u']
values=['2u','3u','0xfffffffeu','0xffffffffu']
jobs=[]
for bound,val,form,shape,cpu in itertools.product(bounds,values,range(4),range(2),[6,7]):
    helper='__forceinline void releaseArgs(unsigned short **p) throw(){LocalFree(p);}\n'
    pre='';mid='';expr='(u&('+bound+'))<=('+bound+')'
    if form in [1,2]:
        decl='const unsigned long &bound=('+bound+');'
        if form==1:mid=decl
        else:pre=decl
        expr='(u&bound)<=bound'
    elif form==3:
        helper+='__forceinline int rangePredicate(unsigned long u){const unsigned long &bound=('+bound+');return (u&bound)<=bound;}\n'
        expr='rangePredicate(u)'
    if shape==0:
        tail='if(u=='+val+'){releaseArgs(argv);'+mid+'r=(int)('+expr+');}else{releaseArgs(argv);'+mid+'r=(int)('+expr+');}return r;'
    else:
        # Separate lexical blocks permit declarations on goto destinations.
        tail='if(u=='+val+')goto left;goto right;right:{releaseArgs(argv);'+mid+'r=(int)('+expr+');goto done;}left:{releaseArgs(argv);'+mid+'r=(int)('+expr+');goto done;}done:return r;'
    src=f.base.replace(f.marker,'long status=WinVerifyTrust(0,&action,&data);unsigned long u=(unsigned long)status;int r=0;'+pre+tail)
    src=src.replace('int Rva00237D3DCheck() {',helper+'int Rva00237D3DCheck() {').replace('/O1 /G7',f'/O1 /G{cpu}')
    jobs.append((f'b{len(jobs):04}',src,dict(bound=bound,value=val,form=form,shape=shape,cpu=cpu)))
if __name__=='__main__':f.execute('bounds',jobs)
