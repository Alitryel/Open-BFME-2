from pathlib import Path
import itertools,re
r=Path(__file__).parent;out=r/'update';out.mkdir(exist_ok=True)
bases=[(r/'bank.cpp').read_text(),(r/'native_split.cpp').read_text()]
methods='''
 __forceinline void addStatusInView(unsigned mask){m_status|=mask;}
 __forceinline void assignStatusInView(unsigned value){m_status=value;}
 __forceinline unsigned statusInView() const{return m_status;}
 __forceinline unsigned &statusRefInView(){return m_status;}
 template<unsigned mask> __forceinline void addStatusConstantInView(){m_status|=mask;}
 __forceinline void addStatusRefInView(const unsigned &mask){m_status|=mask;}
 __forceinline void addStatusCompoundInView(unsigned mask){statusRefInView()|=mask;}
'''
forms=['m_status |= MASK;','m_status = m_status | MASK;','m_status = MASK | m_status;',
 'addStatusInView(MASK);','assignStatusInView(m_status | MASK);',
 'assignStatusInView(MASK | statusInView());','statusRefInView() |= MASK;',
 'statusRefInView() = statusInView() | MASK;','addStatusConstantInView<128>();',
 'addStatusRefInView(MASK);','addStatusCompoundInView(MASK);',
 '{ unsigned &flags=m_status;flags|=MASK; }',
 '{ unsigned flags=m_status;flags|=MASK;m_status=flags; }',
 'm_status |= (unsigned char)MASK;','m_status |= (unsigned long)MASK;',
 'm_status |= (unsigned short)MASK;']
for bi,a,b,shared in itertools.product(range(2),range(16),range(16),range(2)):
    s=bases[bi].replace('Object *rva00272AD5();','Object *rva00272AD5();\n'+methods,1)
    pos=[m.span()for m in re.finditer(r'm_status\s*\|=\s*mask80;',s)];assert len(pos)==2
    for (start,end),v in reversed(list(zip(pos,[a,b]))):s=s[:start]+forms[v].replace('MASK','mask80')+s[end:]
    if shared:
        s=re.sub(r'int mask80\s*=\s*0x80;','',s)
        s=s.replace('void Drawable::onComingOutOfShroud()\n{','void Drawable::onComingOutOfShroud()\n{\n unsigned int mask80=128;',1)
    (out/f'b{bi}a{a:02d}c{b:02d}s{shared}.cpp').write_text(s)
print(len(list(out.glob('*.cpp'))))
