import sys,itertools,re
from pathlib import Path
sys.path.insert(0,'build/netpacket-20261022c');import search
search.WORK=Path.cwd()/'build/netpacket-20261022c'
sig='NetCommandList *NetPacket::getCommandList()\n';H,B=search.BANK.split(sig,1)
items=[]
for carrier,place,refresh,restore,si in itertools.product(range(3),range(2),range(4),range(3),range(2)):
 b=B;h=H
 if si:
  b=b.replace('\tInt offset;\n','').replace('\t\t\toffset = i;\n','').replace('\t\t\ti = offset;\n','')
  b=re.sub(r'\boffset\b','i',b)
 decl=['const Int *limit = &m_packetLen;', 'const Int &limit = m_packetLen;', 'Int limit = m_packetLen;'][carrier]
 expr=['*limit','limit','limit'][carrier]
 initpoint='\tInt i = 0;' if place else '\tUnsignedByte commandType = 0;'
 b=b.replace(initpoint,'\t'+decl+'\n'+initpoint,1)
 b=b.replace('while (i < m_packetLen)', 'while (i < '+expr+')')
 update=['limit = &m_packetLen;', '', 'limit = m_packetLen;'][carrier]
 # Scalar value snapshots are refreshed at every loop edge, preserving current length.
 if carrier==2:
  b=b.replace('\n\t}\n\n\tif (lastCommand != 0)', '\n\t\t'+update+'\n\t}\n\n\tif (lastCommand != 0)')
 if update and refresh:
  z=b.index("} else if (m_packet[i] == 'Z')")
  d=b[:z];tail=b[z:]
  for which in range(2):
   if (refresh==1 and which==0) or (refresh==2 and which==1):continue
   part=d if which==0 else tail
   loc='\t\t\tmsg->detach();' if restore==0 else '\t\t\tmsg = 0;' if restore==1 else '\t\t\ti = offset;'
   if si and restore==2:loc='\t\t\tmsg = 0;'
   part=part.replace(loc,'\t\t\t'+update+'\n'+loc)
   if which==0:d=part
   else:tail=part
  b=d+tail
 if carrier==1 and refresh:
  # A dependent lvalue reference is introduced only at the completion point.
  # It has no lifetime escape and remains an ordinary nonvolatile bound read.
  z=b.index("} else if (m_packet[i] == 'Z')")
  part=b[z:]
  if not si:part=part.replace('i = offset;', '{ const Int &cursor = offset; i = cursor; }' if refresh==1 else '{ const Int *cursor = &offset; i = *cursor; }' if refresh==2 else '{ Int &cursor = offset; i = cursor; }')
  b=b[:z]+part
 items.append((f'bound_{carrier}{place}{refresh}{restore}{si}',h+sig+b))
search.batch(items,'bound-carrier')
