import json,sys
from pathlib import Path
from wdbg import Dbg
W=Path(__file__).resolve().parent
tag=sys.argv[1];src=W/sys.argv[2]
d=Dbg(src,W/(tag+'.obj'),W/(tag+'.log'))
data=[];active=False
def snapshot(stage):
 g=d.words(0x108a8054,1)[0];gw=d.words(g,8)
 nodes={};edges={};il={};syms={}
 def operand(p,depth=0):
  if p<0x10000 or p in il or depth>6:return
  w=d.words(p,20);il[p]=w
  if len(w)<16:return
  k=w[2]&255
  if k in [1,2,4]:
   for q in [w[6],w[7]]:
    if q>0x10000 and q not in syms:syms[q]=d.words(q,24)
  if k in [5,6]:operand(w[13],depth+1)
  elif k in [13,18]:
   operand(w[10],depth+1);operand(w[11],depth+1)
 n=gw[0]
 while n and n not in nodes and len(nodes)<150:
  w=d.words(n,20);nodes[n]=w
  if len(w)<16:break
  operand(w[7])
  e=w[3]
  while e and e not in edges and len(edges)<1000:
   ew=d.words(e,8);edges[e]=ew;e=ew[0]
  if n==gw[1]:break
  n=w[0]
 return dict(stage=hex(stage),window=d.words(0x108a804c,3),g=g,gw=gw,graph=nodes,edges=edges,nodes=il,syms=syms)
try:
 d.prompt();d.prompt('set $CanDeferOnBPByAddr = 1');d.prompt('break *0x107b299b')
 for _ in range(300):
  o=d.prompt('cont',60)
  if 'terminated' in o:break
  pc=d.reg('eip')
  if pc==0x107b299b:
   f=d.reg('edi');fn=d.string(d.words(d.words(f,1)[0]+4,1)[0])
   if '?bfmeTwoCGD@' not in fn:continue
   active=True
   for a in [0x107b4e16,0x107b4e20,0x107b4e2a,0x107b29e0]:d.prompt('break *0x%x'%a)
   continue
  if active:
   if pc==0x107b29e0:break
   s=snapshot(pc);data.append(s)
   print(hex(pc),'window',s['window'],'graph',len(s['graph']),'edges',len(s['edges']),flush=True)
 for _ in range(200):
  o=d.prompt('cont',60)
  if 'terminated' in o:
   print('compiler terminated',flush=True);break
 else:raise RuntimeError('compiler did not terminate')
finally:
 (W/(tag+'.json')).write_text(json.dumps(data,indent=1))
 if d.p.poll() is None:
  try:d.prompt('quit')
  except Exception:d.p.kill()
 d.log.close()
