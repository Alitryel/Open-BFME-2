from pathlib import Path
import sys,itertools,concurrent.futures,json
sys.path.insert(0,str(Path(__file__).parent));import trials
trials.scratch=Path(__file__).resolve().parent
base=(trials.scratch/'bank.cpp').read_text();trials.symbol=base.splitlines()[0][3:]
names=['minus','plus','space','zero','point','sep']
specs=[]
for mask,policy,negative in itertools.product(range(64),[0,1],[0,1]):
 s=base
 for i,name in enumerate(names):
  if mask&(1<<i):
   needle='  char_type __'+name
   assert needle in s;s=s.replace(needle,'  const char_type& __'+name,1)
 if policy:s=s.replace('  _MoneySignPolicy __sign_policy;','  const _MoneySignPolicy __sign_policy = _MoneySignPolicy();')
 if negative:s=s.replace('  bool __is_negative;','  unsigned char __is_negative;')
 specs.append((f'refs-m{mask:02x}-p{policy}-n{negative}',s))
with concurrent.futures.ThreadPoolExecutor(max_workers=8) as pool:r=list(pool.map(trials.run,specs))
(trials.scratch/'scalar_refs.json').write_text(json.dumps(r,indent=2))
