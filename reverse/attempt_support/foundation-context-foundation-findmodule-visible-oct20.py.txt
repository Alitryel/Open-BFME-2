from pathlib import Path
import subprocess,sys,json
root=Path(__file__).resolve().parents[1];out=root/'build/foundation-bitset-oct20';tests={}
body='''
inline Module *Object::findModule(NameKeyType key) const {
    Module *found=0;
    for(Module **at=m_modules;*at;++at) {
        if((*at)->getModuleNameKey()==key) {found=*at;break;}
    }
    return found;
}
'''
for key in ['baseline','headers-only-budget100','model591-budget100']:
    s=(out/(key+'.cpp')).read_text()
    s=s.replace('virtual void m00();','virtual void m00();\n virtual void m01(); virtual void m02(); virtual void m03();\n virtual NameKeyType getModuleNameKey() const=0;',1)
    s=s.replace('Module *findModule(NameKeyType key) const;','__declspec(noinline) Module *findModule(NameKeyType key) const;',1)
    s=s.replace('unsigned char m_pad158[0x254 - 0x158];','unsigned char m_pad158[0x244 - 0x158];\n Module **m_modules;\n unsigned char m_pad248[0x254 - 0x248];',1)
    assert 'Module **m_modules' in s
    tests[key]=s+body
results=[]
for key,s in tests.items():
    p=out/('visible-find-'+key+'.cpp');p.write_bytes(s.encode())
    for name,rva,size in [('?construct@FoundationAIUpdate@@UAEPAVObject@@PBVThingTemplate@@PBUCoord3D@@MPAVPlayer@@H_N@Z','0x0045569D','1174'),('?findModule@Object@@IBEPAVModule@@W4NameKeyType@@@Z','0x0028B6D6','43')]:
        r=subprocess.run([sys.executable,'tools/explain_mismatch.py',name,'--rva',rva,'--size',size,'--source',p.relative_to(root).as_posix()],cwd=root,capture_output=True,text=True,encoding='utf8',errors='replace',timeout=240)
        log=r.stdout+r.stderr;p.with_suffix('.'+rva+'.log').write_bytes(log.encode())
        d={'key':key,'name':name,'exit':r.returncode,'exact':'OK: bytes match' in log,'summary':[x for x in log.splitlines() if x.startswith(('compiled symbol size:','classification:','unresolved:','first diff:'))]}
        if r.returncode:d['error']=log[-700:]
        results.append(d);(out/'findmodule-results.json').write_text(json.dumps(results,indent=2));print(json.dumps(d),flush=True)
