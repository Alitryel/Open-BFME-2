from pathlib import Path
import subprocess,sys,json
out=Path('build/resume-eh-local/turn64-camera-planar');out.mkdir(exist_ok=True)
base=Path('build/resume-eh-local/turn64-camera-state/control.cpp').read_text()
needle='const Coord3D*start=from;Coord2D from2,to2,delta;from2.x=start->x;from2.y=start->y;to2.x=to->x;to2.y=to->y;delta.x=to2.x-from2.x;delta.y=to2.y-from2.y;'
assert needle in base
view='''class CameraPlanar : public Coord2D {
public:
 CameraPlanar() {}
 CameraPlanar(float px,float py) {x=px;y=py;}
 CameraPlanar(const Coord3D &p) {x=p.x;y=p.y;}
 CameraPlanar(const Coord2D &p) {x=p.x;y=p.y;}
 CameraPlanar(const CameraPlanar &p) {x=p.x;y=p.y;}
 CameraPlanar &operator-=(const Coord2D &p) {x-=p.x;y-=p.y;return *this;}
 float operator*(const Coord2D &p) const {return x*p.x+y*p.y;}
 DTOR
};
'''
variants={}
for dtor in [False,True]:
 head=view.replace('DTOR','~CameraPlanar() {}' if dtor else '')
 for ctor in ['floats','points']:
  init='CameraPlanar from2(start->x,start->y),to2(to->x,to->y);' if ctor=='floats' else 'CameraPlanar from2(*start),to2(*to);'
  for dot in [False,True]:
   s=base.replace('bool PolygonTrigger::adjustCameraMovement(',head+'bool PolygonTrigger::adjustCameraMovement(',1)
   s=s.replace(needle,'const Coord3D*start=from;'+init+'CameraPlanar delta(to2);delta-=from2;')
   if dot:s=s.replace('along.x*delta.x+along.y*delta.y','CameraPlanar(along.x,along.y)*delta')
   variants[ctor+('_dtor' if dtor else '')+('_dot' if dot else '')]=s
sym='?adjustCameraMovement@PolygonTrigger@@QAE_NPBUCoord3D@@0PAU2@_N@Z';results=[]
for key,s in variants.items():
 p=out/(key+'.cpp');p.write_bytes(s.encode())
 r=subprocess.run([sys.executable,'tools/explain_mismatch.py',sym,'--rva','0x002E3AA8','--size','723','--source',p.as_posix()],stdout=subprocess.PIPE,stderr=subprocess.STDOUT,text=True)
 p.with_suffix('.txt').write_bytes(r.stdout.encode());d={'variant':key,'exit':r.returncode,'summary':[x for x in r.stdout.splitlines() if x.startswith(('compiled symbol size:','classification:','first diff:','unresolved:'))]}
 if r.returncode:d['tail']=r.stdout[-900:]
 results.append(d);print(d,flush=True)
 (out/'results.json').write_bytes((json.dumps(results,indent=2)+'\n').encode())
