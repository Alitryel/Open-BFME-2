from pathlib import Path
import csv, subprocess, sys
root=Path(__file__).resolve().parents[2]
sys.path.insert(0,str(root/'tools'))
import build
folder=Path(__file__).parent
record='''
// Native 500839 copies the existing 24-byte holder through 50055D.
class Rva0050055D {
public: Rva0050055D(const Rva0050055D &);
private: unsigned char m_storage[24];
};
typedef _STL::pair<const int, Rva0050055D> HolderPair;
'''
factory='Code/Libraries/Source/WWVegas/WWLib/stlport_rb_tree_copy_0052D25D.cpp'
body=(root/factory).read_text()
body=body[:body.index('// The rowed _M_create_node (above)')]
body+=record+'''
namespace _STL { template <> void _Construct<HolderPair, HolderPair>(HolderPair *, const HolderPair &); }
typedef _STL::_Rb_tree<int, HolderPair, _STL::_Select1st<HolderPair>, _STL::less<int>, _STL::allocator<HolderPair> > HolderTree;
template HolderTree::_Link_type HolderTree::_M_create_node(const HolderPair &);
'''
construct='Code/GameEngine/Source/Common/Rva00566695Assign.cpp'
other=(root/construct).read_text()
old=next(line for line in other.splitlines() if line.startswith('template void _STL::_Construct<'))
other=other.replace(old,record+'''
namespace _STL { template <> pair<const int,Rva0050055D>::pair(const pair<const int,Rva0050055D> &); }
template void _STL::_Construct<HolderPair, HolderPair>(HolderPair *, const HolderPair &);
''')
rows=[r for r in csv.DictReader((root/'reverse/functions.csv').open()) if r['target_rva'] in ('0x00500CFE','0x0050116B')]
symbols=build.load_symbol_map()
for row in rows: symbols[row['name'].replace('AsciiString','Rva0050055D')]=[int(row['target_rva'],16)]
for label,text,rva in [('factory',body,'0x0050116B'),('construct',other,'0x00500CFE')]:
    # Scratch paths do not inherit the production source's flag-default entry.
    text=text.replace('// cl:', '// cl: /O1 /arch:SSE /G7', 1)
    src=folder/(label+'.cpp');obj=folder/(label+'.obj');src.write_text(text)
    command,env=build.compiler_command(src,obj)
    done=subprocess.run(command,cwd=root,env=env,capture_output=True,text=True)
    (folder/(label+'.compile.txt')).write_text(done.stdout+done.stderr)
    if done.returncode: print(label,done.stdout+done.stderr);continue
    row=next(r for r in rows if r['target_rva']==rva)
    name=row['name'].replace('AsciiString','Rva0050055D')
    got=build.compile_function(dict(row,name=name),symbols,obj)
    raw,_=build.read_object_symbol_bytes(obj,name)
    print(label,len(raw),got['bytes']==got['target'],got['unresolved'],flush=True)
