from search import *

def source(n):
    marker=MARK
    body=BODY
    if n&1:marker=marker.replace('bool enable','volatile bool enable')
    for bit,old,new in [
        (2,'bool observer =','volatile bool observer ='),
        (4,'bool locked =','volatile bool locked ='),
        (8,'bool teamEnable;','volatile bool teamEnable;'),
        (16,'bool found =','volatile bool found ='),
        (32,'int i =','volatile int i ='),
    ]:
        if n&bit:body=body.replace(old,new)
    return HEAD+marker+body

if __name__=='__main__':run([('bool-%03d'%n,source(n)) for n in range(64)],'bool')
