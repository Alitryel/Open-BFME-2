# Different equivalent spellings across a dynamic branch: test whether a
# join survives into split/reload reconstruction but merges before emission.
import sys,itertools,json,time
from pathlib import Path
from concurrent.futures import ThreadPoolExecutor
sys.path.insert(0,'build/fxinit-seat22');import search as h
h.HERE=Path('build/fxinit-split22').resolve()
DEST='*(int *)(vertices+desc.Get_Basis_End())='
a='remap[links[0][i]]';b='remap[links[1][i]]'
RS=[h.R,DEST+'('+b+'<<8)+'+a+';',
 'int mapped0='+a+';int mapped1='+b+';'+DEST+'mapped0+(mapped1<<8);',
 'const short &mapped0='+a+';const short &mapped1='+b+';'+DEST+'mapped0+(mapped1<<8);',
 DEST+'MergePack('+a+','+b+');']
WS=[h.W,'Vector2 weightValue(weights[0][i]/100.0f,weights[1][i]/100.0f);*(Vector2 *)(vertices+desc.Get_Extension_Data_Offset())=weightValue;',
 'float weightFirst=weights[0][i]/100.0f;float weightSecond=weights[1][i]/100.0f;*(Vector2 *)(vertices+desc.Get_Extension_Data_Offset())=Vector2(weightFirst,weightSecond);',
 'MergeWeights((Vector2 *)(vertices+desc.Get_Extension_Data_Offset()),weights[0][i],weights[1][i]);']
def variants():
 for cond,rform,wform,reverse,scope in itertools.product(['tangents','binormals','positions[1]','normals[1]','basis','m_FVF&0x10'],range(5),range(4),range(2),range(3)):
  src=h.BASE;common=[h.R,h.W,h.R+'\n'+h.W][scope];alt=[RS[rform],WS[wform],RS[rform]+'\n'+WS[wform]][scope]
  x,y=(alt,common)if reverse else(common,alt)
  body='if('+cond+') {'+x+'}else{'+y+'}'
  if scope==0:body+='\n'+h.W
  if scope==1:body=h.R+'\n'+body
  old=h.R+'\n      '+h.W;assert old in src;src=src.replace(old,body)
  helpers=''
  if rform==4 and scope!=1:helpers+='static __forceinline int MergePack(int a,int b){return a+(b<<8);}\n'
  if wform==3 and scope!=0:helpers+='static __forceinline void MergeWeights(Vector2 *p,unsigned a,unsigned b){*p=Vector2(a/100.0f,b/100.0f);}\n'
  src=src.replace('bool FXShaderGeometry::Init(',helpers+'bool FXShaderGeometry::Init(')
  yield f'branchmerge_{cond.replace("[","").replace("]","").replace("&","_")}_{rform}{wform}{reverse}{scope}',src
if __name__=='__main__':
 t=time.time()
 with ThreadPoolExecutor(max_workers=8)as pool:r=list(pool.map(h.run_one,variants()))
 r.sort(key=lambda x:(x.get('nd',99999),x.get('cost',99999)))
 (h.HERE/'branchmerge-results.json').write_text(json.dumps(r,indent=2))
 print(json.dumps({'count':len(r),'valid':sum('nd'in x and not x.get('unresolved')for x in r),'seconds':time.time()-t,'best':r[:8]},indent=2))
