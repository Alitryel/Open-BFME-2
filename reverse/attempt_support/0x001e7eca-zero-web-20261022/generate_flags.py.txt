from search import *
import itertools
s=(W/'base.cpp').read_text();vs=[]
choices=list(itertools.product(['/O1','/O2','/Ox','/Od /Og /Os'],['','/Og-'],['/Ob0','/Ob1','/Ob2'],['','/Z7','/Zd'],['','/Oy-'],['','/Op']))
for i,flags in enumerate(choices):
 line=next(l for l in s.splitlines() if l.startswith('// cl:'))
 o=s.replace(line,line+' '+' '.join(flags))
 vs.append((f'flags_{i:03d}',o))
(W/'flags-choices.json').write_text(json.dumps(choices))
run(vs,'flags288')
