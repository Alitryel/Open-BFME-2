from trials import *
import itertools,re,collections
names=['fuel','resistance','burnRate','maxBurn','decay']
decl='int fuel, resistance, burnRate, maxBurn, decay;'
call='TheTriggerManager->rva002858E5(&world.x, &fuel, &burnRate, &decay, &resistance, &maxBurn)'
char='template <> void StringBase<char>::releaseBuffer() throw();\n'
items=[]
for mask,kind,release in itertools.product(range(1,32),range(4),range(2)):
 s=BASE if release else BASE.replace(char,'')
 selected=[n for i,n in enumerate(names) if mask>>i&1]
 types=['volatile int','unsigned int','volatile unsigned int','long']
 ds=' '.join((types[kind] if n in selected else 'int')+' '+n+';' for n in names)
 cc=call
 for n in selected:cc=cc.replace('&'+n,'(int *)&'+n)
 s=s.replace(decl,ds).replace(call,cc)
 # Signedness passed to C varargs does not alter its representation or values
 # for this diagnostic; the query itself owns values and still writes int.
 assert collections.Counter(re.findall(r'"(?:\\.|[^"\\])*"',s))==collections.Counter(re.findall(r'"(?:\\.|[^"\\])*"',BASE))
 items.append(('types%02d%d%d'%(mask,kind,release),s))
batch(items,'output-types')
