from pathlib import Path
import subprocess,sys,json
out=Path('build/resume-eh-local/turn62-phase-byte-copy');out.mkdir(exist_ok=True)
base=Path('build/resume-eh-local/turn62-phase-clean/minimal_pun.cpp').read_text()
forms={
'object_byte':'bool result; *(unsigned char *)&result = (unsigned char)allowed; return result;',
'object_char':'bool result; *(char *)&result = (char)allowed; return result;',
'reference_byte':'bool result; unsigned char &storage = reinterpret_cast<unsigned char &>(result); storage = (unsigned char)allowed; return result;',
'memcpy':'bool result; memcpy(&result, &allowed, sizeof(result)); return result;',
'memcpy_byte':'unsigned char byte = (unsigned char)allowed; bool result; memcpy(&result, &byte, sizeof(result)); return result;'}
results=[]
for key,body in forms.items():
 source=base.replace('return *(bool*)&allowed;',body)
 if key.startswith('memcpy'):source='extern "C" void *__cdecl memcpy(void *,const void *,unsigned int);\n#pragma intrinsic(memcpy)\n'+source
 p=out/(key+'.cpp');p.write_bytes(source.encode())
 r=subprocess.run([sys.executable,'tools/explain_mismatch.py','?rva002B5EB5@Rva002B5EB5@@QAE_NXZ','--rva','0x002B5EB5','--size','213','--source',p.as_posix()],stdout=subprocess.PIPE,stderr=subprocess.STDOUT,text=True)
 (out/(key+'.txt')).write_bytes(r.stdout.encode());d={'variant':key,'exit':r.returncode}
 for f in ['compiled symbol size','classification','first diff:','unresolved']:
  lines=[s for s in r.stdout.splitlines() if s.lower().startswith(f)]
  if lines:d[f]=lines[0]
 results.append(d);print(d,flush=True)
 (out/'results.json').write_bytes((json.dumps(results,indent=2)+'\n').encode())
