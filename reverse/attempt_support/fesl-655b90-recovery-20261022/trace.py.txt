import sys, json
from pathlib import Path
sys.path.insert(0,str(Path(__file__).resolve().parent))
sys.path.insert(0,'build/4404ac-mergeweb')
from wdbg import Dbg

tag=sys.argv[1]
out=Path('build/655b90-mergeweights')
d=Dbg(str(out/(tag+'.cpp')),str(out/(tag+'-trace.obj')),str(out/(tag+'-trace.log')))
records=[];active=False
try:
 d.prompt();d.prompt('set $CanDeferOnBPByAddr = 1');d.prompt('break *0x10760539')
 for i in range(1000):
  o=d.prompt('cont',30)
  if 'terminated' in o or not o.strip():break
  if '0x10760539' in o:
   w=d.words(d.reg('eax'),1);sy=d.words(w[0],2);fn=d.string(sy[1])
   if active:break
   if '?Rva007E8B00@@' in fn:
    active=True;d.prompt('break *0x10782b26');print('active',flush=True)
   continue
  if active and '0x10782b26' in o:
   addr=d.reg('eax');w=d.words(addr,28)
   r={'addr':addr,'words':w,'stack':d.words(d.reg('esp'),12)}
   if len(w)>17:
    r['priority']=w[3] if w[3]<0x80000000 else w[3]-0x100000000
    r['id']=w[17]
    sw=d.words(w[0],20);r['symbol']=sw
    if len(sw)>7:r['symbol_id']=sw[7]
    if len(sw)>1 and 0x10000<sw[1]<0x10000000:r['name']=d.string(sw[1])
    if sw[0] and sw[0]<0x10000000:
     fw=d.words(sw[0],16);r['front']=fw
     if len(fw)>1 and 0x10000<fw[1]<0x10000000:r['front_name']=d.string(fw[1])
   records.append(r)
finally:
 (out/(tag+'-trace.json')).write_text(json.dumps(records,indent=1))
 if d.p.poll() is None:
  try:d.prompt('quit')
  except Exception:d.p.kill()
 d.log.close()
print([(r.get('symbol_id'),r.get('priority'),r.get('front_name',''),hex(r['stack'][0])) for r in records],flush=True)
