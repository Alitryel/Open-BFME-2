from pathlib import Path
import sys,subprocess,json
root=Path(__file__).resolve().parents[1];sys.path.insert(0,str(root/'tools'))
out=root/'build/string29530-inline-sites-oct20';out.mkdir(exist_ok=True)
base=(root/'reverse/attempts/0x00029530.cpp').read_text()
symbol='?bfmeDoInsertQX@BfmeThingQX@@QAEXPADID@Z'
fill='''inline void fill(char *__first,char *__last,const char &__val) {
 char __tmp=__val;
 memset(__first,(unsigned char)__tmp,__last-__first);
}
'''
decl='void fill(char *__first, char *__last, const char &__val);'
assert base.count(decl)==1
visible=base.replace(decl,fill)
variants={'declaration':base,'visible':visible}
call='\tfill(__first, __first + __n, __val);'
assert visible.count(call)==1
variants['site-fill']=visible.replace(call,'#pragma inline_depth(0)\n'+call+'\n#pragma inline_depth(255)')
call='\treturn fill_n(__first, __n, __x);';assert visible.count(call)==1
variants['site-fill-n']=visible.replace(call,'#pragma inline_depth(0)\n'+call+'\n#pragma inline_depth(255)')
call='\treturn __uninitialized_fill_n(__first, __n, __x, __true_type());';assert visible.count(call)==1
variants['site-uninitialized']=visible.replace(call,'#pragma inline_depth(0)\n'+call+'\n#pragma inline_depth(255)')
for depth in [1,2,3,4,8]:
    variants['function-depth-'+str(depth)]=visible+'\n#pragma inline_depth('+str(depth)+')\n'
    # A nonzero state at the function's closing brace is what VC7 consumes.
    variants['function-depth-'+str(depth)]=visible.rsplit('}',1)[0]+'#pragma inline_depth('+str(depth)+')\n}\n'
import build,capstone
from locate import object_functions
rows=[]
for key,text in variants.items():
    src=out/(key+'.cpp');src.write_text(text)
    p=subprocess.run([sys.executable,'tools/explain_mismatch.py',symbol,'--rva','0x29530','--size','662','--source',str(src)],cwd=root,capture_output=True,text=True,encoding='utf8',errors='replace',timeout=300)
    log=p.stdout+p.stderr;src.with_suffix('.mismatch.txt').write_text(log)
    obj=build.obj_path(src)
    if not obj.exists():print(key,'COMPILE ERROR',log[-800:],flush=True);continue
    fs=list(object_functions(obj));parent=next(b for n,b,r in fs if n==symbol)
    q=subprocess.run([sys.executable,'tools/align_diff.py',str(src),symbol,'0x29530','662'],cwd=root,capture_output=True,text=True,encoding='utf8',errors='replace',timeout=120)
    src.with_suffix('.align.txt').write_text(q.stdout+q.stderr)
    md=capstone.Cs(capstone.CS_ARCH_X86,capstone.CS_MODE_32)
    frame=next((i.op_str for i in md.disasm(parent,0) if i.mnemonic=='sub' and i.op_str.startswith('esp,')),None)
    summary=[l for l in log.splitlines() if l.startswith(('classification:','first diff:','OK:','unresolved:'))]
    print(key,len(parent),frame,summary,flush=True)
    rows.append(dict(key=key,bytes=len(parent),frame=frame,summary=summary))
(out/'results.json').write_text(json.dumps(rows,indent=2))
