"""New 218-byte lifetime context x four failure-return source choices."""
from pathlib import Path
import concurrent.futures,itertools,json,re,runpy
OUT=Path('build/canmake-alias-20261023/lifetime-interactions').resolve();OUT.mkdir(parents=True,exist_ok=True)
r=runpy.run_path('reverse/attempt_support/canmake-00596e15-20261022/replay.py.txt');r['run'].__globals__['OUT']=OUT
base=runpy.run_path('reverse/attempt_support/canmake-alias-20261023/context.py.txt')['bank_text']();prefix,body=base.rsplit('int AIBuildableWallSegment::canMake(Player *player)',1)
# Every predicate is true at its respective original failure return.
conditions=['failed','result.m_20==7','result.m_20!=0 && result.m_20!=7','player->m_money < (unsigned int)result.m_pad04']
values=[9,2,9,2]
pattern=r'\{selected.code=(2|9);return selected.code;\}'
assert len(re.findall(pattern,body))==4

def form(shape,c,v):
 return [f'{{selected.code={v};return selected.code;}}',
 f'{{selected.code=({c})?{v}:0;return selected.code;}}',
 f'{{selected.code=0;if({c})selected.code={v};return selected.code;}}',
 f'{{selected.code={v};if(!({c}))selected.code=0;return selected.code;}}',
 f'{{selected.code=({c})?{v}:{11-v};return selected.code;}}',
 f'{{selected.code={v};return ({c})?selected.code:0;}}'][shape]
tests=[]
for shapes in itertools.product(range(6),repeat=4):
 counter=iter(range(4))
 def replace(m):
  i=next(counter);assert int(m[1])==values[i]
  return form(shapes[i],conditions[i],values[i])
 b=re.sub(pattern,replace,body)
 tests.append(('interaction-'+''.join(map(str,shapes)),prefix+'\nint AIBuildableWallSegment::canMake(Player *player)'+b))
results=[]
with concurrent.futures.ThreadPoolExecutor(max_workers=8)as pool:
 for i,d in enumerate(pool.map(r['run'],tests),1):
  results.append(d)
  if i%216==0 or d.get('exact_candidate'):print(json.dumps(dict(done=i,total=len(tests),best=sorted(results,key=lambda x:(x.get('nd',999),-x.get('score',0)))[:2])),flush=True)
(OUT/'results.json').write_text(json.dumps(results,indent=2)+'\n')
print(json.dumps(dict(trials=len(results),errors=sum('error'in x for x in results),unresolved=sum(bool(x.get('unresolved'))for x in results),exact=sum(x.get('exact_candidate',False)for x in results),bodies=len({x.get('masked_code_sha256')for x in results}))),flush=True)
