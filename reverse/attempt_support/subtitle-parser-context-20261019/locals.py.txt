import json
import re
import subprocess
import sys
from pathlib import Path

root = Path(__file__).resolve().parents[2]
scratch = root / 'build/resume-eh-local'
base = (scratch / 'turn47-subtitle-separate-and-callback.cpp').read_text()
block = '''	UnicodeString text;
	unsigned int color = 0;
	int style = 0x7fffffff;
	int align = 0x7fffffff;
	int line = 0x7fffffff;
	int startFrame = 0x7fffffff;
	int endFrame = 0x7fffffff;'''
assert block in base
declarations = block.splitlines()[1:]
variants = {}
for position in range(1, 7):
    lines = declarations[:]
    lines.insert(position, '\tUnicodeString text;')
    variants['text-position-' + str(position)] = base.replace(block, '\n'.join(lines))
variants['copy-init-text'] = base.replace('\tUnicodeString text;', '\tUnicodeString text = UnicodeString();')
variants['value-init-text'] = base.replace('\tUnicodeString text;', '\tUnicodeString text((UnicodeString()));')
results = []
for label, source in variants.items():
    path = scratch / ('turn47-subtitle-' + label + '.cpp')
    path.write_text(source)
    proc = subprocess.run([sys.executable, str(root / 'tools/explain_mismatch.py'),
        '?parseSubtitleText@@YAXPAVINI@@PAX1PBX@Z', '--source', path.relative_to(root).as_posix(),
        '--rva', '0x00688B10', '--size', '1578'], cwd=root, capture_output=True, text=True)
    log = scratch / ('turn47-subtitle-' + label + '.txt')
    log.write_text(proc.stdout + proc.stderr)
    size = re.search(r'compiled symbol size: (\d+)', proc.stdout)
    result = {'variant': label, 'exit': proc.returncode,
        'compiled_size': int(size[1]) if size else None,
        'exact': 'OK: bytes match' in proc.stdout,
        'differing_instruction_rows': sum(s.startswith('!=') for s in proc.stdout.splitlines()),
        'log': log.relative_to(root).as_posix()}
    print(json.dumps(result), flush=True)
    results.append(result)
(scratch / 'turn47-subtitle-local-results.json').write_text(json.dumps(results, indent=2) + '\n')
