from wdbg import *
W=Path('build/vp6-1c5e30-20261023')
tag=sys.argv[1]
regs=len(sys.argv)>2 and sys.argv[2]=='regs'
src=W/(tag+'.cpp')
d=Dbg(str(src),str(W/(tag+'-trace.obj')),str(W/(tag+'-trace.log')))
active=False
records=[]
try:
 d.prompt();d.prompt('set $CanDeferOnBPByAddr = 1');d.prompt('break *0x10760539')
 for i in range(1000):
  o=d.prompt('cont',30)
  if 'terminated' in o or not o.strip():break
  if '0x10760539' in o:
   w=d.words(d.reg('eax'),1);sy=d.words(w[0],2);fn=d.string(sy[1])
   if active:break
   if '?Rva009B5530Prepare@@' in fn:
    active=True;d.prompt('break *'+('0x10782b26' if regs else '0x10790b19'))
   continue
  if active and regs and '0x10782b26' in o:
   addr=d.reg('eax');w=d.words(addr,28)
   r={'addr':addr,'words':w,'stack':d.words(d.reg('esp'),12),'priority':w[3],'id':w[17]}
   sw=d.words(w[0],20);r['symbol']=sw
   if len(sw)>1 and 0x10000<sw[1]<0x10000000:r['name']=d.string(sw[1])
   if sw[0] and sw[0]<0x10000000:
    fw=d.words(sw[0],16);r['front']=fw
    if len(fw)>1 and 0x10000<fw[1]<0x10000000:r['front_name']=d.string(fw[1])
   records.append(r)
   continue
  if active and not regs and '0x10790b19' in o:
   addr=d.words(0x108a78a4,1)[0]
   for j in range(500):
    if not addr:break
    w=d.words(addr,24)
    r={'addr':addr,'words':w,'weight':w[14],'size':w[8],'index':w[15]}
    if 0x10000<w[0]<0x10000000:
     sw=d.words(w[0],20);r['symbol']=sw
     if len(sw)>1 and 0x10000<sw[1]<0x10000000:r['name']=d.string(sw[1])
    records.append(r)
    addr=w[12]
   print(json.dumps(records),flush=True)
   d.prompt('delete 2')
finally:
 (W/(tag+('-regs' if regs else '-trace')+'.json')).write_text(json.dumps(records,indent=1))
 if d.p.poll() is None:
  try:d.prompt('quit')
  except Exception:d.p.kill()
 d.log.close()
