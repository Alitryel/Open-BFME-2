from pathlib import Path
import sys,subprocess,json,hashlib,concurrent.futures
ROOT=Path.cwd();OUT=ROOT/'build/trust237-round11';sys.path.insert(0,str(ROOT/'tools'));import build,capstone
base=(OUT/'swap-control.cpp').read_text()
def run(cpu):
 src=OUT/f'control-g{cpu}.cpp';obj=src.with_suffix('.obj');src.write_text(base.replace('/G6',f'/G{cpu}'))
 cmd,env=build.compiler_command(src,obj);cp=subprocess.run(cmd,env=env,capture_output=True,text=True)
 row=dict(cpu=cpu,returncode=cp.returncode,warnings=(cp.stdout+cp.stderr)[-1200:],functions={})
 if cp.returncode:return row
 for name in ['swap16','swap32','swap64','rot64','same64']:
  raw,rel=build.read_object_symbol_bytes(obj,'_'+name,code_only=True);md=capstone.Cs(capstone.CS_ARCH_X86,capstone.CS_MODE_32)
  row['functions'][name]=dict(size=len(raw),sha256=hashlib.sha256(raw).hexdigest(),raw=raw.hex(),relocations=rel,instructions=[dict(offset=i.address,raw=i.bytes.hex(),mnemonic=i.mnemonic,operands=i.op_str) for i in md.disasm(raw,0)])
 return row
with concurrent.futures.ThreadPoolExecutor(max_workers=5) as pool:rows=list(pool.map(run,[3,4,5,6,7]))
(OUT/'controls.json').write_text(json.dumps(rows,indent=2)+'\n')
for r in rows:
 print('G',r['cpu'],'returncode',r['returncode'])
 for k,v in r['functions'].items():print(k,v['size'],' '.join(i['mnemonic'] for i in v['instructions']))
