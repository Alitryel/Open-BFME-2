from pathlib import Path
from itertools import product
W=Path(__file__).resolve().parent
out=W/'memberbridge';out.mkdir(exist_ok=True)
for idx,(model,types,order,body,const,sites,bank) in enumerate(product(range(4),range(4),range(2),range(4),range(2),range(8),range(2))):
 s=(W/('diagnostic.cpp' if bank else 'base.cpp')).read_text()
 defs=''
 pv='*p' if types&1 else 'p';nv='*n' if types&2 else 'n'
 args={'p':('const ContactVector *p' if types&1 else 'const ContactVector &p'),'n':('const ContactVector *n' if types&2 else 'const ContactVector &n')}
 keys=['n','p'] if order else ['p','n']
 if model==2:keys=['n']
 for name,alt in [('bridge',False),('reverseBridge',bool(bank))]:
  params=([('ContactData *s')] if model in [1,2,3] else [])+[args[k] for k in keys]
  source='source' if model==0 else 's->source'
  point='point' if model==2 else pv;normal=nv
  pre=''
  if body&1:pre+='ContactSource *q='+source+';';source='q'
  if body&2:
   d={'p':'const ContactVector &a='+point+';','n':'const ContactVector &b='+normal+';'}
   pre+=''.join(d[k] for k in (['n','p'] if order else ['p','n']));point='a';normal='b'
  expr=('altPrepare('+source+','+point+','+normal+')') if alt else source+'->prepare('+point+','+normal+')'
  defs+=('__forceinline static ' if model==1 else '__forceinline ')+'void *'+name+'('+','.join(params)+')'+(' const' if const and model!=1 else '')+'{'+pre+'return '+expr+';}\n'
 if model in [0,1]:s=s.replace('ContactSource *source; };','ContactSource *source; '+defs+' };')
 else:s=s.replace('ContactNode *next;','ContactNode *next;\n'+defs)
 pos=0
 for i,(side,n) in enumerate([('second','node->normal'),('second','node->normal'),('first','ContactVector(node->normal).negate()')]):
  source='node->'+side+'->source'
  old=('altPrepare('+source+',node->point,'+n+')') if bank and i==2 else source+'->prepare(node->point,'+n+')'
  at=s.find(old,pos);assert at>=0,(idx,i)
  if sites&(1<<i):
   vals={'p':'&(node->point)' if types&1 else 'node->point','n':'&('+n+')' if types&2 else n}
   name='reverseBridge' if i==2 else 'bridge'
   values=(['node->'+side] if model in [1,2,3] else [])+[vals[k] for k in keys]
   receiver=('node->'+side+'->') if model==0 else ('ContactData::' if model==1 else 'node->')
   new=receiver+name+'('+','.join(values)+')'
  else:new=old
  s=s[:at]+new+s[at+len(old):];pos=at+len(new)
 (out/('v%04d.cpp'%idx)).write_text(s)
print('generated',idx+1)
