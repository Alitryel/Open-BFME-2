from pathlib import Path
W=Path(__file__).resolve().parent
D=W/'hybrid';D.mkdir(exist_ok=True)
i=0
for family,call in [('targetcopy','prepareCopy0'),('slotaddr','prepareSlot0')]:
 for n in range(128):
  s=(W/family/('v%04d.cpp'%n)).read_text()
  old=call+'(node->second->source,node->point,node->normal)'
  assert old in s
  s=s.replace(old,'node->second->source->prepare(node->point,node->normal)',1)
  (D/('v%04d.cpp'%i)).write_text(s);i+=1
print('generated',i)
