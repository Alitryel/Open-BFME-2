from pathlib import Path
from itertools import product
W=Path(__file__).resolve().parent;out=W/'nodehandle';out.mkdir(exist_ok=True)
banks=[((W/'control.cpp')).read_text(),(W/'base.cpp').read_text()]
for idx,(flags,ret,create,cv,bank) in enumerate(product(range(16),range(4),range(4),range(2),range(2))):
 b=banks[bank]
 ctorcc='__fastcall ' if flags&1 else ''
 defs='struct NodeHandle { ContactNode *p; '+ctorcc+'NodeHandle(ContactNode *q):p(q){} '
 if flags&2:defs+='NodeHandle(const NodeHandle &q):p(q.p){} '
 if flags&4:defs+='~NodeHandle(){} '
 rt=['ContactNode *','ContactNode *const &','ContactNode *','ContactNode *const &'][ret]
 rb='ContactNode *q=p;return q;' if ret==2 else 'return p;'
 # The reference-returning form binds to the member, never to a local.
 defs+='__forceinline '+rt+'operator->() const {'+rb+'} '
 if flags&8:defs+='bool operator==(int q) const { return p==0; } '
 defs+='operator ContactNode *() const {return p;} };\n'
 defs+='__forceinline NodeHandle makeNode(ContactNode *p) { '+('NodeHandle n(p);return n;' if ret==3 else 'return NodeHandle(p);')+' }\n'
 b=b.replace('class BfmeThingCGD {',defs+'class BfmeThingCGD {')
 decl=['NodeHandle node(current);','NodeHandle node=NodeHandle(current);','NodeHandle node=makeNode(current);','const NodeHandle &node=makeNode(current);'][create]
 if cv and create!=3:decl='const '+decl
 b=b.replace('ContactNode *node=current;',decl)
 (out/('v%04d.cpp'%idx)).write_text(b)
print('generated',idx+1)
