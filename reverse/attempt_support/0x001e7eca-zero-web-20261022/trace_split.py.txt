from wdbg import Dbg
from pathlib import Path
import sys,json,re,time
W=Path('build/locomotor-zero-cfg');label=sys.argv[1] if len(sys.argv)>1 else 'baseline'
d=Dbg(str(W/(label+'.cpp')),str(W/(label+'-split.obj')),str(W/(label+'-split-debugger.log')))
fn='';cls=-1;snaps=[];start=time.time();armed=False
phases={0x82219:'setup0',0x82225:'setup1',0x82232:'setup2',0x82239:'setup3',0x8448d:'int0',0x84493:'int1',0x83aac:'graph0',0x83acd:'cost1',0x83add:'graph1',0x83b64:'order0',0x83b6b:'order1',0x8226d:'rewrite0',0x82273:'finished'}
def snapshot(phase):
 rows=[]
 for head in d.words(0x108807b8,1024):
  p=head
  while p:
   words=d.words(p,30)
   if len(words)<30:break
   sw=d.words(words[0],32)
   if len(sw)>10 and sw[:2]==[0,0x100d]:
    rows.append(dict(addr=p,w=words,sym=sw,operand=d.words(sw[10],16) if sw[10]>0x100000 else []))
   p=words[12]
 snaps.append(dict(phase=phase,cls=cls,rows=rows))
 print(phase,cls,[(r['w'][7],r['w'][3],r['w'][9],r['w'][16]) for r in rows],flush=True)
try:
 d.prompt();d.prompt('set $CanDeferOnBPByAddr = 1');d.prompt('break *0x10782219')
 for step in range(700):
  out=d.prompt('cont')
  if 'terminated' in out:break
  if time.time()-start>90:raise TimeoutError('overall deadline')
  off=d.reg('eip')-0x10700000
  if off==0x82219:
   ebp=d.reg('ebp');p=d.words(ebp,1)[0];fn=d.string(d.words(p+4,1)[0]);cls=-1
   if 'rva001E7ECA' in fn and not armed:
    armed=True
    for addr in phases:
     if addr!=0x82219:d.prompt('break *0x%x'%(0x10700000+addr))
  if 'rva001E7ECA' not in fn:continue
  if off in [0x83aac,0x83acd,0x83add,0x83b64,0x83b6b,0x8448d,0x84493]:cls=d.reg('ebx')
  if off in phases and (cls<=0 or off in [0x8226d,0x82273]):snapshot(phases[off])
 (W/(label+'-split.json')).write_text(json.dumps(snaps,indent=2))
finally:
 if d.p.poll() is None:
  try:d.prompt('quit')
  except Exception:d.p.kill()
 d.log.close()
