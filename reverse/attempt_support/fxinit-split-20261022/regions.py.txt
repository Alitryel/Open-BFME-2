import json
from pathlib import Path
H=Path('build/fxinit-split22');out={}
for p in sorted(H.glob('*-results.json')):
 if p.name=='regional-results.json':continue
 rows=json.loads(p.read_text());scores=[]
 for r in rows:
  f=H/'trials'/(r['id']+'.json')
  if not f.exists() or r.get('unresolved'):continue
  d=json.loads(f.read_text())['diff'];n=0
  for x in d:
   if 'region' in x:
    a=x['retail'][0]
    n+=sum(0x707<=i[0]<0x722 for i in a)
   elif 0x707<=x['retail'][0]<0x722:n+=1
  scores.append({'remap':n,**r})
 scores.sort(key=lambda x:(x['remap'],x['cost'],x['nd']))
 out[p.stem]=scores[:4]
 print(p.stem,[(x['id'],x['remap'],x['cost'],x['nd'],x['size'])for x in scores[:2]])
(H/'regional-results.json').write_text(json.dumps(out,indent=2)+'\n')
