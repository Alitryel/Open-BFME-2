"""Run from the repo root; isolated ordinary compiles, no ledger/cache writes.

Clean-Room: reverse/vp6_cleanroom/specs/001c2b80.md
The equal-arm historical source is a negative control, never recovery code.
"""
from pathlib import Path
import hashlib
import json
import subprocess
import sys

root = Path.cwd().resolve()
sys.path.insert(0, str(root / 'tools'))
from locate import object_functions
from variant_search import Scorer, diff_text, fitness
import build

evidence = root / 'reverse/attempt_support/diagonal-1c2b80-20261022'
output = root / 'build/diagonal-controls-20261022'
output.mkdir(parents=True, exist_ok=True)
scorer = Scorer('_DiagonalBlur', 0x001C2B80, 916, output)
old = subprocess.check_output([
    'git', 'show', '4ff1498bbd:reverse/attempts/0x001c2b80.cpp'
], cwd=root).decode('utf-8')
controls = [
    ('historical-equal-arm-negative-control', old),
    ('clean-pointer-value-boundary', (evidence / 'cursor-control.cpp.txt').read_text()),
    ('clean-real-samples-bank', (root / 'reverse/attempts/0x001c2b80.cpp').read_text()),
    ('unsigned-address-alternative', (evidence / 'unsigned-address.cpp.txt').read_text()),
]
records = []
for name, source in controls:
    result = scorer(source)
    if 'compiled' not in result:
        raise RuntimeError((name, result))
    raw = next(
        data.rstrip(b'\xCC')
        for symbol, data, relocations in object_functions(output / f'c{scorer.n}.obj')
        if symbol == '_DiagonalBlur'
    )
    assert raw == result['compiled'], 'Unexpected relocation/extent masking'
    assert len(raw) == len(result['target']) == 916
    differences = [
        {'offset': f'0x{i:X}', 'compiled': f'{a:02X}', 'retail': f'{b:02X}'}
        for i, (a, b) in enumerate(zip(raw, result['target'])) if a != b
    ]
    if name == 'clean-real-samples-bank':
        assert differences == [
            {'offset': '0xB5', 'compiled': '0B', 'retail': '19'},
            {'offset': '0x138', 'compiled': '0B', 'retail': '19'},
        ]
    records.append({
        'control': name,
        'whole_extent': len(raw),
        'exact': result['exact'],
        'fitness': fitness(raw, result['target']),
        'compiled_sha256': hashlib.sha256(raw).hexdigest(),
        'retail_sha256': hashlib.sha256(result['target']).hexdigest(),
        'byte_differences': differences,
        'instruction_diff': diff_text(raw, result['target'], 120),
    })
    print(name, len(raw), 'differing bytes:', len(differences), flush=True)

command, environment = build.compiler_command(output / 'c1.cpp', output / 'probe.obj')
c2 = Path(command[0]).with_name('c2.dll')
c2_hash = hashlib.sha256(c2.read_bytes()).hexdigest()
assert c2_hash == 'bcd28f39b1798b07989c30a1df471cf87fc20e3802d130901a08f83800e6b81e', 'Different backend: the observations are version-specific'
proof = {
    'symbol': '_DiagonalBlur', 'rva': '0x001C2B80', 'size': 916,
    'compiler': 'MSVC 13.10.3077', 'flags': '/O2 /G6 /DNDEBUG /MD',
    'c2_sha256': c2_hash,
    'status': 'partial; none of these controls is a landing',
    'controls': records,
}
(output / 'proof.json').write_text(json.dumps(proof, indent=2) + '\n', newline='\n')
print(output / 'proof.json')
