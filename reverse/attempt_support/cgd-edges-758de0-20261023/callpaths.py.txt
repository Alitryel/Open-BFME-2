from pathlib import Path
import itertools,json
W=Path(__file__).resolve().parent;b=(W/'base.cpp').read_text();D=W/'callpaths';D.mkdir(exist_ok=True)
state='node->state=node->first->source->test(node->second->source->prepare(node->point,node->normal))?1:2;'
first='node->first->source->respond(bridge(node->second,node->point,node->normal));'
last='node->second->source->respond(altPrepare(node->first->source,node->point,ContactVector(node->normal).negate()));'
preds=['node->state==1','node->state&1','node->key0&1','node->key0==node->key1',
 'node->first==node->second','index&1','node->key1<node->key0','node->first->source==node->second->source']
def wrap(s,shape,p,tag):
 e=s.rstrip(';');ready=tag+'_ready';done=tag+'_done'
 return [f'if({p}) {{ {s} }} else {{ {s} }}',
 f'({p}) ? (void)({e}) : (void)({e});',
 f'if({p}) goto {ready}; {s} goto {done}; {ready}: {s} {done}:;',
 f'if({p}) {{ {ready}: {s} }} else {{ goto {ready}; }}',
 f'if({p}) {{ {s} goto {done}; }} {s} {done}:;',
 f'if(!({p})) {{ {s} goto {done}; }} {s} {done}:;',
 f'if({p}) {{ {s} }} else {{ for(;;) {{ {s} break; }} }}',
 f'if({p}) {{ do {{ {s} }} while(0); }} else {{ {s} }}'][shape]
rows=[]
for n,(shape,pred,sites,bridge,mixed) in enumerate(itertools.product(range(8),range(8),range(8),range(2),range(2))):
 p=preds[pred];s=b
 f=first if bridge else first.replace('bridge(node->second,','node->second->source->prepare(')
 r=last if mixed else last.replace('altPrepare(node->first->source,','node->first->source->prepare(')
 st=wrap(state,shape,p,'tested') if sites&1 else state
 s=s.replace('if(node->state==0)\n   '+state,'if(node->state==0) { '+st+' }')
 s=s.replace(first,wrap(f,shape,p,'first') if sites&2 else f)
 s=s.replace(last,wrap(r,shape,p,'reverse') if sites&4 else r)
 (D/('v%04d.cpp'%n)).write_text(s)
 rows.append(dict(name='v%04d'%n,shape=shape,pred=pred,sites=sites,bridge=bridge,mixed=mixed))
(W/'callpaths-models.json').write_text(json.dumps(rows,indent=1));print(len(rows),'duplicate-call path models')
