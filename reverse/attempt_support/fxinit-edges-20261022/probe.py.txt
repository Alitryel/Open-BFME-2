import sys,itertools,json,time,subprocess,re
from pathlib import Path
from concurrent.futures import ThreadPoolExecutor
sys.path.insert(0,'build/fxinit-seat22');import search as h
HERE=Path('build/fxinit-edges22/probes').resolve();HERE.mkdir(exist_ok=True)
HEAD=h.BASE[:h.BASE.index('class FXShaderGeometry {')]
PARAMS='char *vertices, BfmeFVFDescriptor &desc, const Vector3 **positions, const Vector3 **normals, const Vector3 *tangents, const Vector3 *binormals, const unsigned short **links, const unsigned short **weights, _STL::vector<short> &remap, unsigned m_FVF, unsigned m_Passes, int m_BonesPerVertex, bool basis, unsigned pass, unsigned count, MeshModelClass *mesh, int textureCount, const unsigned *colors'
PREFIX=[
'*(Vector3 *)(vertices+desc.Get_Location_Offset(0))=positions[0][i];',
'if(m_FVF&0x10) *(Vector3 *)(vertices+desc.Get_Normal_Offset(0))=normals[0][i];',
'if(m_FVF&0x40) {if(colors) *(unsigned *)(vertices+desc.Get_Diffuse_Offset())=colors[i];else *(unsigned *)(vertices+desc.Get_Diffuse_Offset())=0xFFFFFFFF;}',
'for(int t=0;t<textureCount;++t) {const Vector2 *uv=mesh->Get_UV_Array_By_Index(t);if(uv) *(Vector2 *)(vertices+desc.Get_Tex_Offset(t))=uv[i];}',
'if(basis) {if(tangents) *(Vector3 *)(vertices+desc.Get_Basis_Offset())=tangents[i];if(binormals) *(Vector3 *)(vertices+desc.Get_Basis_Second_Offset())=binormals[i];}']
def source(mask,mode):
 arm=h.ARM
 if mode:
  arm=arm.replace(h.P,'').replace(h.N,'')
  if mode==1:arm+=h.P+'\n'+h.N
  if mode==2:arm+=f'if(m_Passes<=1 && m_BonesPerVertex==2 && links[0]) {{ {h.P} {h.N} }}'
  if mode==3:
   arm='bool copySecond=false;'+arm.replace('else if(m_BonesPerVertex==2) {','else if(m_BonesPerVertex==2) {copySecond=true;')+f'if(copySecond) {{ {h.P} {h.N} }}'
 return HEAD+'\nextern "C" void FXProbe('+PARAMS+') {for(unsigned i=0;i<count;++i) {\n'+''.join(x for k,x in enumerate(PREFIX)if mask>>k&1)+arm+'vertices+=desc.Get_FVF_Size();}}\n'
def run(item):
 mask,mode=item;label=f'probe_{mask:02x}_{mode}';p=HERE/(label+'.cpp');o=p.with_suffix('.obj');p.write_text(source(mask,mode))
 cmd,env=h.build.compiler_command(p,o)
 r=subprocess.run(cmd,env=env,cwd=h.ROOT,capture_output=True,text=True)
 if r.returncode:return {'id':label,'error':(r.stdout+r.stderr)[-1800:]}
 raw,rel=h.build.read_object_symbol_bytes(o,'_FXProbe',code_only=True);ii=h.insns(raw,rel)
 contexts=[]
 for k,x in enumerate(ii):
  if x[1]=='shl'and x[2].endswith(', 8'):contexts.append(ii[max(0,k-9):k+5])
 p.with_suffix('.dis').write_text('\n'.join(f'{a:04x} {m} {v}'for a,m,v,s in ii))
 return {'id':label,'size':len(raw),'context':contexts}
if __name__=='__main__':
 with ThreadPoolExecutor(max_workers=8)as pool:r=list(pool.map(run,itertools.product(range(32),range(4))))
 (HERE/'results.json').write_text(json.dumps(r,indent=2))
 for x in r:
  if x['id'].startswith(('probe_00','probe_10','probe_1f')):print(json.dumps(x))
 print('done',len(r),'errors',sum('error'in x for x in r))
