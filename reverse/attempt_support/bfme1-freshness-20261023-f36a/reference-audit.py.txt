from pathlib import Path
import datetime,hashlib,json,subprocess,sys
ROOT=Path.cwd();OUT=ROOT/'build/trust237-round15';SUB=ROOT/'reference/open-bfme-1'
sys.path.insert(0,str(ROOT/'tools'));import build
OLD='f890413e75e671748215ba2ef3304aa7953993e9';NEW='f36a4db066aa61edeee90085a31d628eb829001d'
def git(*args,cwd=ROOT):return subprocess.check_output(['git',*args],cwd=cwd)
changed=git('diff','--name-only',OLD,NEW,cwd=SUB).decode().splitlines()
changed_paths={str(Path('reference/open-bfme-1')/p) for p in changed}
receipts=[];malformed=[];affected=[]
for folder in ['build/match','build/spike_zh']:
 for path in (ROOT/folder).glob('*.deps.json'):
  try:
   receipt=json.loads(path.read_text());deps=receipt['deps'];assert isinstance(deps,dict)
  except Exception as e:malformed.append(dict(path=str(path.relative_to(ROOT)),error=str(e)));continue
  receipts.append(str(path.relative_to(ROOT)))
  hits=sorted(changed_paths & (set(deps)|{receipt.get('path','')}))
  if hits:affected.append(dict(receipt=str(path.relative_to(ROOT)),paths=hits))
intended=json.loads((ROOT/'reverse/attempt_support/bfme1-freshness-20261022-f890/input-audit.json').read_text())['intended_inputs']
for p,v in intended.items():
 data=(SUB/p).read_bytes();before=git('show',OLD+':'+p,cwd=SUB);after=git('show',NEW+':'+p,cwd=SUB)
 assert data==before==after
 assert hashlib.sha256(data).hexdigest()==v['sha256']
layouts={k:dict(resolved=build.bfme1_subtree(k),exists=build.bfme1_path(k).is_dir()) for k in ['reference','toolchains','baselines','vendor','game','ledger']}

print({k:v for k,v in result.items() if k not in ['changed_files','intended_inputs','layouts']},flush=True)
