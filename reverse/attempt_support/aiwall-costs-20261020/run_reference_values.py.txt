from pathlib import Path
import sys,json,itertools,re
ROOT=Path(__file__).resolve().parents[2]
sys.path.insert(0,str(ROOT/'tools'))
from variant_search import Scorer
out=Path(__file__).parent/'reference-values'
s=Scorer('?distSqToSegment@@YAMPBVCoord2D@@00@Z',0x4eacf5,197,out)
base=(ROOT/'reverse/attempts/0x004eadba.cpp').read_text().replace('../../Code/',str(ROOT/'Code').replace('\\','/')+'/')
rows=[]
for a,b,group in itertools.product(range(4),range(4),range(2)):
 text=base
 for var,mode in (('apx',a),('apy',b)):
  if mode==0:continue
  text=text.replace('float '+var+' = ','const float '+('&' if mode==1 else '')+var+' = ')
  if mode==2:
   text=text.replace('const float '+var+' = ', 'float '+var+'Storage = ')
   expr='pt->x - lineStart->x' if var=='apx' else 'pt->y - lineStart->y'
   text=text.replace('float '+var+'Storage = '+expr+';','float '+var+'Storage = '+expr+';\n\tconst float &'+var+' = '+var+'Storage;')
  if mode==3:
   expr='pt->x - lineStart->x' if var=='apx' else 'pt->y - lineStart->y'
   text=text.replace('const float '+var+' = '+expr+';','const float &'+var+' = ('+expr+');')
 if group:
  text=text.replace('float dot = apy * dy;\n\tdot += apx * dx;','float dot = apx * dx + apy * dy;')
 r=s(text);got=r.get('compiled',b'');target=r.get('target',b'')
 row={'forms':[a,b,group],'file':f'c{s.n}.cpp','size':len(got),'diff':sum(a!=b for a,b in zip(got,target))+abs(len(got)-len(target)),'exact':r['exact'],'fitness':r['fitness'],'class':r['class']}
 rows.append(row)
 if s.n==1 or r['exact'] or row['diff']<15:print(json.dumps(row),flush=True)
 if r['exact']:(out/'win.cpp').write_bytes(text.encode());break
(out/'results.json').write_text(json.dumps(rows,indent=2))
print('completed',len(rows),'best',min(rows,key=lambda r:(not r['exact'],-r['fitness'])),flush=True)
