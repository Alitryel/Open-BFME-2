for i,(a,b,pos,ret,c) in enumerate(product(range(16),range(16),range(2),range(2),range(2))):
 s=control if c else bank

  defs+='__forceinline PrepareMethod *slotAddress%d(PrepareMethod *q){'%site+ids[n]+'}\n'

   expr='&((DispatchTable *)slotAddress%d((PrepareMethod *)*(DispatchTable **)s))->prepare'%site
