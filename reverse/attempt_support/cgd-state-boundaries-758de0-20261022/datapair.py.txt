from pathlib import Path
from itertools import product
W=Path(__file__).resolve().parent
out=W/'datapair';out.mkdir(exist_ok=True)
sources=sorted((W/'memberbridge').glob('*.cpp'))[::2]
for idx,(source,kind,direct) in enumerate(product(sources,range(2),range(2))):
 s=source.read_text()
 ty='const ContactVector &' if kind else 'const ContactVector *'
 pair='struct ContactPair {'+ty+'point;'+ty+'normal;ContactPair(const ContactVector &p,const ContactVector &n):point('+('p' if kind else '&p')+'),normal('+('n' if kind else '&n')+'){} };\n'
 s=s.replace('struct ContactSource {',pair+'struct ContactSource {',1)
 s=s.replace('virtual void *prepare(const ContactVector &point, const ContactVector &normal)=0;', 'virtual void *__fastcall preparePair(ContactPair args)=0;\n __forceinline void *prepare(const ContactVector &p,const ContactVector &n) { return preparePair(ContactPair(p,n)); }')
 if direct:
  at=0
  while True:
   at=s.find('->prepare(',at)
   if at<0:break
   begin=at+len('->prepare(');end=begin;depth=1
   while depth:
    if s[end]=='(':depth+=1
    elif s[end]==')':depth-=1
    end+=1
   inside=s[begin:end-1]
   new='->preparePair(ContactPair('+inside+'))'
   s=s[:at]+new+s[end:];at+=len(new)
 (out/('v%04d.cpp'%idx)).write_text(s)
print('generated',idx+1)
