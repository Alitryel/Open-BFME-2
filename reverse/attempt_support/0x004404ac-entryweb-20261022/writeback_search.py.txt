from search import *
import itertools,re

def source(flags):
 place, init, dtor, scope = flags
 marks=['\n\tGameInfo *game','\n\tif (!game)','\n\tif (slot == -1 || slot >= 8)','\n\tbool observer']
 p=BODY.index(marks[place]);before=BODY[:p];tail=BODY[p:]
 dtors=[
  'original=value;',
  'if (original!=value) original=value;',
  'if (active) original=value;',
  'if (active && original!=value) original=value;',
  'int v=value; original=v;',
  'if (!active) return; original=value;',
  'if (active) {int &r=original;r=value;}',
  'release();',
 ]
 cl='struct Index {int &original; int value; bool active; Index(int &r):original(r),value(r),active(false) {} ~Index() {'+dtors[dtor]+'} void release(){original=value;} };'
 decl=cl+' Index index(slot);'
 if init in (1,2):
  decl=decl.replace('value(r)','value(0)')
  marker='\n\tif (slot == -1 || slot >= 8)' if place<3 else '\n\tbool observer'
  # This assignment is on the post-null path. Initialization timing and
  # cleanup scope interact with the entry candidates.
  q=tail.index(marker)
  tail=tail[:q]+'\n index.value=slot;'+tail[q:]
 if init==2:
  decl=decl.replace('value(0),','')
 if init==3:decl=decl.replace('value(r)','value((int)(unsigned int)r)')
 # Activate only after the null-game guard, while every original use is still
 # represented by the slot parameter. The reference binds before that guard.
 activate='\n\tif (slot == -1 || slot >= 8)'if place<3 else'\n\tbool observer'
 q=tail.index(activate);tail=tail[:q]+'\n index.active=true;'+tail[q:]
 tail=re.sub(r'\bslot\b','index.value',tail)
 # Initial post-null copy still refers to the original argument home.
 tail=tail.replace('index.value=index.value;','index.value=slot;')
 if scope:
  before+='\n {'
  q=tail.rfind('\n}');tail=tail[:q]+'\n }'+tail[q:]
 return HEAD+MARK+before+'\n '+decl+tail

if __name__=='__main__':
 cases=[f for f in itertools.product(range(4),range(4),range(8),range(2)) if not (f[0]<2 and f[1]==2 and f[2] in (0,1,4,7))]
 run([('writeback-%03d'%n,source(f))for n,f in enumerate(cases)], 'writeback')
