import re, subprocess, sys, os, json, concurrent.futures as cf
HERE = os.path.dirname(os.path.abspath(__file__))
ROOT = os.path.abspath(os.path.join(HERE, '../..'))
base = open(os.path.join(HERE, os.environ.get('A9_BASE', 'f0.cpp'))).read()


def _row(out, x, y, m0, m1, t, form, tag):
    P0 = f"{x} * m.{m0}"; P1 = f"{y} * m.{m1}"
    if form == 0: return f"{out} = {P0} + {P1} + m.{t};"
    if form == 1: return f"float {tag}0 = {P0};\n\tfloat {tag}1 = {P1};\n\t{out} = {tag}0 + {tag}1 + m.{t};"
    if form == 2: return f"float {tag}1 = {P1};\n\tfloat {tag}0 = {P0};\n\t{out} = {tag}0 + {tag}1 + m.{t};"
    if form == 3: return f"float {tag}0 = {P0};\n\t{out} = {tag}0 + {P1} + m.{t};"
    if form == 4: return f"float {tag}1 = {P1};\n\t{out} = {P0} + {tag}1 + m.{t};"
    if form == 5: return f"float {tag}s = {P0} + {P1};\n\t{out} = {tag}s + m.{t};"
    if form == 6: return f"{out} = m.{m0} * {x} + m.{m1} * {y} + m.{t};"


PT = {0: ("", "in.X", "in.Y"),
      1: ("const float *q = &in.X;\n\t", "q[0]", "q[1]"),
      2: ("float x = in.X, y = in.Y;\n\t", "x", "y")}
H = {}
for pa in range(3):
    for fx in range(7):
        for fy in range(7):
            pre, x, y = PT[pa]
            H[pa * 49 + fx * 7 + fy] = pre + _row("out.X", x, y, "a", "c", "tx", fx, "X") + "\n\t" + _row("out.Y", x, y, "b", "d", "ty", fy, "Y")

helpers = "".join(
    f"static __forceinline void aptMM{k}(V2 &out, const V2 &in, const AptRenderTransform &transform)\n{{\n\tconst AptMatrix &m = (const AptMatrix &)transform;\n\t{b}\n}}\n"
    for k, b in H.items())
anchor = "void AptShapeContainer::render()"
i = base.index(anchor)
head, body = base[:i] + helpers, base[i:]
sites = [m.start() for m in re.finditer(r'aptMatrixMul\(', body)]
N = len(sites)


def make(cfg):
    out = []; last = 0
    for s, c in zip(sites, cfg):
        out.append(body[last:s]); out.append(f"aptMM{c}("); last = s + len("aptMatrixMul(")
    out.append(body[last:])
    return head + "".join(out)


def score(cfg, tag):
    p = f"build/a9f98/s_{tag}.cpp"
    open(os.path.join(ROOT, p), 'w').write(make(cfg))
    r = subprocess.run([sys.executable, 'tools/explain_mismatch.py', '?render@AptShapeContainer@@QAEXXZ', '--rva', '0x000A9F98',
                        '--size', '1471', '--source', p], cwd=ROOT, capture_output=True, text=True)
    txt = r.stdout + r.stderr
    m = re.search(r'compiled symbol size: (\d+)', txt)
    size = int(m.group(1)) if m else -1
    n = 0
    for l in txt.splitlines():
        if ' | ' in l and '\t' in l:
            a, b = l.split(' | ', 1); pa = a.split('\t'); pb = b.split('\t')
            if len(pa) >= 3 and len(pb) >= 3 and pa[2].strip() != pb[2].strip(): n += 1
    if size != 1471: n += 1000
    return n


if __name__ == '__main__':
    cfg = json.loads(sys.argv[1]) if len(sys.argv) > 1 else [0] * N
    print('sites', N, 'variants', len(H)); best = score(cfg, 'base'); print('start', cfg, best, flush=True)
    improved = True
    while improved:
        improved = False
        for si in range(N):
            cands = []
            for v in H:
                if v == cfg[si]: continue
                c = list(cfg); c[si] = v; cands.append(c)
            with cf.ThreadPoolExecutor(6) as ex:
                res = list(ex.map(lambda ct: (score(ct[1], f"w{ct[0]}"), ct[1]), enumerate(cands)))
            for sc, c in res:
                if sc < best: best, cfg = sc, c; improved = True
            print('site', si, 'best', best, json.dumps(cfg), flush=True)
    print('FINAL', best, json.dumps(cfg))
