"""single reached-goal exit variants from k0c0pf"""
from pathlib import Path
H = Path('/work3/Open-BFME-2/build/h72'); O = Path('/work3/Open-BFME-2/build/h73/v1')
for f in O.glob('*'): f.unlink()
b = (H / 'k0c0pf.cpp').read_text(encoding='latin-1')
lines = b.split('\n')
def rep(x, old, new, n=1):
    c = x.count(old); assert c == n, (old[:60], c); return x.replace(old, new)
EXIT1 = '\n'.join(lines[183:189]) + '\n'   # lines 184..189 (inner if(reachedGoal){..} incl closing)
EXIT2 = '\n'.join(lines[227:233]) + '\n'   # lines 228..233
assert EXIT1.strip().startswith('if(reachedGoal){') and EXIT2.strip().startswith('if(reachedGoal){'), (EXIT1, EXIT2)
PORTAL = '  if(m_zoneManager.rva005315C8(parentCell->getXIndex(),parentCell->getYIndex())) {\n'
BODY = EXIT2.split('\n',1)[1].rsplit('  }\n',1)[0]   # exit body statements
V = {}
# a: goto found in both places, label at function end
x = rep(b, EXIT1, '   if(reachedGoal)goto found;\n')
x = rep(x, EXIT2, '  if(reachedGoal)goto found;\n')
x = x.rstrip('\n'); assert x.endswith('}')
x = x[:-1] + ' found:\n' + BODY + '}\n'
V['a_goto'] = x
# b: nested mineshaft only; portal guarded by !reachedGoal; single exit after portal
x = rep(b, EXIT1, '')
x = rep(x, PORTAL, '  if(!reachedGoal && m_zoneManager.rva005315C8(parentCell->getXIndex(),parentCell->getYIndex())) {\n')
V['b_guard'] = x
# c: if(!reachedGoal){ if(portal) {...} }
x = rep(b, EXIT1, '')
x = rep(x, PORTAL, '  if(!reachedGoal)\n' + PORTAL)
V['c_guard2'] = x
# d: goto with label placed just before the second exit copy (inside loop)
x = rep(b, EXIT1, '   if(reachedGoal)goto found;\n')
x = rep(x, EXIT2, '  if(reachedGoal){\n found:\n' + BODY + '  }\n')
V['d_gotoin'] = x
# e: sequential (not nested) mineshaft, then exit test, portal, second test w/ goto to first
x = rep(b, EXIT1, '')
x = rep(x, '  Bool wasMineshaft=useMineshaft;\n  if(reachedGoal) {\n', '  Bool wasMineshaft=useMineshaft;\n  if(reachedGoal) {\n')
x = rep(x, PORTAL, '  if(reachedGoal) {\n found:\n' + BODY + '  }\n' + PORTAL)
x = rep(x, EXIT2, '  if(reachedGoal)goto found;\n')
V['e_first'] = x
for k, v in V.items(): (O / f'{k}.cpp').write_text(v, encoding='latin-1', newline='')
print(len(V))
