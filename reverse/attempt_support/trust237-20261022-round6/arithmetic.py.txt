from pathlib import Path
import concurrent.futures, hashlib, itertools, json, random, subprocess, sys
sys.path.insert(0, 'tools')
import build, capstone
sys.path.insert(0, 'build/trust237-round4/venv/lib/python3.12/site-packages')
from unicorn import Uc, UC_ARCH_X86, UC_MODE_32
from unicorn.x86_const import UC_X86_REG_ESP, UC_X86_REG_EAX

ROOT=Path.cwd(); OUT=ROOT/'build/trust237-round6'; OUT.mkdir(exist_ok=True)
NAME='?ArithmeticPredicate@@YAHK@Z'
predicates=[]
def add(family, expr): predicates.append(dict(family=family,expression=expr))
for terms,bits in [(2,1),(3,1),(4,3),(5,3),(6,4),(8,7)]:
    product='*'.join('(u+%du)'%i for i in range(terms))
    add('consecutive_product',f'(({product})&{bits}u)==0u')
for divisor in [3,5,7,10,31,257,65535,0x80000001]:
    add('division_recomposition',f'(u/{divisor}u)*{divisor}u+(u%{divisor}u)==u')
    add('division_bound',f'(u/{divisor}u)*{divisor}u<=u')
for shift in [1,3,7,15]:
    rotate=f'((u<<{shift})|(u>>{32-shift}))'
    add('rotate_inverse',f'(({rotate}>>{shift})|({rotate}<<{32-shift}))==u')
    add('distributivity',f'u*{(1<<shift)+1}u==((u<<{shift})+u)')
for expr in [
    '(u&(0u-u))<=u',
    '(u&(u-1u))<=u',
    '(u|(u+1u))>=u',
    '((u&(0u-u))&((u&(0u-u))-1u))==0u',
    '(u/(u|1u))<=1u',
    '(u%(u|1u))<=u',
    '((u*u-u)&1u)==0u',
    '((u*u*u-u)&1u)==0u',
    '((u+0x7fffffffu)^0x80000000u)==(u-1u)',
    '((u^0x80000000u)+0x80000000u)==u',
    '((u*0xaaaaaaabu)*3u)==u',
    '((u*0xcccccccdu)*5u)==u',
    '((u|(u>>1))&(u&~(u>>1)))==(u&~(u>>1))',
    '((u^0xffffffffu)+u)==0xffffffffu',
    '((u&0xffffu)+(u&0xffff0000u))==u',
    '((u>>16)+(u<<16))==((u>>16)|(u<<16))',
]: add('bit_ring',expr)

md=capstone.Cs(capstone.CS_ARCH_X86,capstone.CS_MODE_32)
samples=[0,1,2,3,4,7,8,15,16,31,32,63,64,127,128,255,256,
         0x7ffffffe,0x7fffffff,0x80000000,0x80000001,0xfffffffd,0xfffffffe,0xffffffff]
rng=random.Random(0x237d3d)
samples += [rng.randrange(1<<32) for _ in range(128)]

def run(job):
    n,pred,opt,cpu=job
    src=OUT/f'a{n:04}.cpp';obj=src.with_suffix('.obj')
    src.write_text(f'// cl: {opt} /G{cpu} /arch:SSE /MD /EHsc\n'
                   'int ArithmeticPredicate(unsigned long u) {return '+pred['expression']+';}\n')
    cmd,env=build.compiler_command(src,obj)
    cp=subprocess.run(cmd,cwd=ROOT,env=env,capture_output=True,text=True)
    result=dict(name=src.stem,predicate=pred,opt=opt,cpu=cpu,returncode=cp.returncode)
    if cp.returncode:
        result['error']=(cp.stdout+cp.stderr)[-1500:];return result
    raw,rel=build.read_object_symbol_bytes(obj,NAME,code_only=True)
    result.update(size=len(raw),sha256=hashlib.sha256(raw).hexdigest(),raw=raw.hex(),
                  relocations=[str(x) for x in rel],nops=sum(i.mnemonic=='nop' for i in capstone.Cs(capstone.CS_ARCH_X86,capstone.CS_MODE_32).disasm(raw,0)),
                  nop16=b'\x90'*16 in raw)
    if rel:
        result['emulation']='not attempted: relocations';return result
    uc=Uc(UC_ARCH_X86,UC_MODE_32);uc.mem_map(0x1000000,0x1000);uc.mem_write(0x1000000,raw)
    uc.mem_map(0x2000000,0x2000);stack=0x2001000;end=0x1000800
    failed=[]
    for sample in samples:
        uc.mem_write(stack,end.to_bytes(4,'little')+sample.to_bytes(4,'little'))
        uc.reg_write(UC_X86_REG_ESP,stack);uc.reg_write(UC_X86_REG_EAX,0xeeeeeeee)
        uc.emu_start(0x1000000,end,count=2048)
        if uc.reg_read(UC_X86_REG_EAX)!=1:failed.append(sample)
    result.update(emulation_samples=len(samples),emulation_failures=failed)
    return result

if __name__=='__main__':
    jobs=[(n,*v) for n,v in enumerate(itertools.product(predicates,['/O1','/O2'],[6,7]))]
    with concurrent.futures.ThreadPoolExecutor(max_workers=8) as pool:results=list(pool.map(run,jobs))
    (OUT/'arithmetic.json').write_text(json.dumps(dict(predicates=predicates,samples=samples,results=results),indent=2))
    good=[r for r in results if 'error' not in r]
    print('PROBES',len(results),'GOOD',len(good),'UNIQUE',len({r['sha256'] for r in good}),
          'NOP16',sum(r['nop16'] for r in good),'NOP',sum(r['nops']>0 for r in good),
          'EMU_FAILURES',sum(bool(r.get('emulation_failures')) for r in good),flush=True)
    for r in sorted({r['sha256']:r for r in good}.values(),key=lambda r:r['size'])[:12]:
        print({k:v for k,v in r.items() if k not in ['raw','relocations']},flush=True)
