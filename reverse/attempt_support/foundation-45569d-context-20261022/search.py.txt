from pathlib import Path
import sys, subprocess, json, re, difflib, concurrent.futures, hashlib, time
ROOT=Path(__file__).resolve().parents[2]
sys.path.insert(0,str(ROOT/'tools'))
import build, capstone
OUT=Path(__file__).resolve().parent
NAME='?construct@FoundationAIUpdate@@UAEPAVObject@@PBVThingTemplate@@PBUCoord3D@@MPAVPlayer@@H_N@Z'
RVA=0x45569d
MD=capstone.Cs(capstone.CS_ARCH_X86,capstone.CS_MODE_32)
TARGET=build.read_target_bytes(RVA,1174)
def disasm(b):
    md=capstone.Cs(capstone.CS_ARCH_X86,capstone.CS_MODE_32)
    md.detail=True
    return list(md.disasm(b,0))
T=disasm(TARGET)
def score(raw,relocs):
    c=disasm(raw)
    sm=difflib.SequenceMatcher(None,[i.mnemonic for i in c],[i.mnemonic for i in T],autojunk=False)
    roffs={r[0] for r in relocs}
    bad=[];regions=[];opdiff=[]
    for tag,a,b,x,y in sm.get_opcodes():
        if tag!='equal':
            regions.append([tag,hex(T[x].address) if x<len(T) else 'end',b-a,y-x])
            bad.extend([1]*max(b-a,y-x))
            continue
        for p,q in zip(c[a:b],T[x:y]):
            # Calls/data relocations are checked by normal byte gates for a winner.
            reloc=any(p.address <= r < p.address+p.size for r in roffs)
            if reloc or p.mnemonic.startswith('j'):continue
            if p.bytes!=q.bytes:
                bad.append(1);opdiff.append([hex(q.address),q.mnemonic,q.op_str,p.op_str])
    return dict(size=len(raw),bad=len(bad),regions=regions,operands=opdiff)
def compile_one(item):
    key,source=item
    p=OUT/(key+'.cpp'); obj=OUT/(key+'.obj'); log=OUT/(key+'.log');result=OUT/(key+'.json')
    h=hashlib.sha256(source.encode()).hexdigest()
    if p.exists() and result.exists() and p.read_text()==source:
        r=json.loads(result.read_text())
        if r.get('hash')==h:return r
    p.write_text(source)
    try:
        command,env=build.compiler_command(p,obj)
        run=subprocess.run(command,cwd=ROOT,env=env,capture_output=True,text=True,timeout=90)
        log.write_text(run.stdout+run.stderr)
        if run.returncode: return dict(key=key,error=(run.stdout+run.stderr)[-500:])
        raw,relocs=build.read_object_symbol_bytes(obj,NAME,code_only=True)
        raw=raw.rstrip(b'\xcc')
        r=score(raw,relocs);r.update(key=key,hash=h,relocs=relocs)
        result.write_text(json.dumps(r,indent=2))
        return r
    except Exception as e:return dict(key=key,error=str(e))
def run(tests,label):
    # Prime toolchain/reference freshness once, before parallel scratch compiles.
    p=OUT/'baseline.cpp'
    p.write_text(BASE)
    build.compiler_command(p,OUT/'prime.obj')
    start=time.time()
    with concurrent.futures.ThreadPoolExecutor(max_workers=8) as pool:
        results=list(pool.map(compile_one,tests.items()))
    (OUT/(label+'-results.json')).write_text(json.dumps(results,indent=2))
    good=[r for r in results if 'error' not in r]
    from collections import Counter
    print(label,len(results),'%.1fs'%(time.time()-start),'distribution',Counter((r['size'],r['bad'],len(r['regions'])) for r in good))
    for r in sorted(good,key=lambda r:(r['bad'],abs(r['size']-1174)))[:15]:
        print(r['key'],r['size'],r['bad'],r['regions'])
    for r in results:
        if 'error' in r:print('ERROR',r['key'],r['error'][-200:])
    return results
BASE=(ROOT/'reverse/attempts/0x0045569d.cpp').read_text()
BASE=re.sub(r'#include "(?:\.\./)+(Code/[^\"]+)"',r'#include "\1"',BASE)
BASE=BASE.replace('/GX','/GX /I.',1)
if __name__=='__main__':run({'baseline':BASE},'baseline')
