from pathlib import Path
import subprocess,sys,json
root=Path(__file__).resolve().parents[1];out=root/'build/activebody-kind-bitset-oct20'
base=(out/'boundary-list-boundaries.cpp').read_text()
base=base.replace('Bool testKindOf(Int k) const { return m_kindOf.test(k); }','Bool testKindCheckedInView(Int k) const { return m_kindOf.test(k); }\n\tUnsignedInt testKindOf(Int k) const { return ((const UnsignedInt *)&m_kindOf)[k >> 5] & (1U << (k & 31)); }')
start=base.index('void ActiveBody::attemptDamage(DamageInfo *damageInfo)');prefix,body=base[:start],base[start:]
groups=((55,),(9,8),(25,89),(45,));name='?attemptDamage@ActiveBody@@UAEXPAVDamageInfo@@@Z';results=[]
for mask in range(16):
    s=body
    for i,group in enumerate(groups):

            for value in group:s=s.replace('->testKindOf('+str(value)+')','->testKindCheckedInView('+str(value)+')')
    p=out/('sites-%02d.cpp'%mask);p.write_text(prefix+s)
    r=subprocess.run([sys.executable,'tools/explain_mismatch.py',name,'--rva','0x004BFE07','--size','2107','--source',p.relative_to(root).as_posix()],cwd=root,capture_output=True,text=True,encoding='utf8',errors='replace',timeout=240)
    log=r.stdout+r.stderr;p.with_suffix('.log').write_text(log)
    d={'mask':mask,'exit':r.returncode,'exact':'OK: bytes match' in log,'summary':[x for x in log.splitlines() if x.startswith(('compiled symbol size:','classification:','unresolved:','first diff:'))]}

    results.append(d);(out/'sites-results.json').write_text(json.dumps(results,indent=2));print(json.dumps(d),flush=True)
