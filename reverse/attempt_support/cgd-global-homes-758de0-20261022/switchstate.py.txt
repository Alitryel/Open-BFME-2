from pathlib import Path
from itertools import product
W=Path(__file__).resolve().parent;out=W/'switchstate';out.mkdir(exist_ok=True)
banks=[(W/'control.cpp').read_text(),(W/'base.cpp').read_text(),(W/'pair.cpp').read_text()]
for idx,(bank,assign,pre,post,guard) in enumerate(product(range(3),range(4),range(5),range(4),range(2))):
 s=banks[bank];a=s.index('  if(node->state==0)');z=s.index('  if(node->state==2) continue;',a)
 expr=s[a:z].strip();test=expr.rsplit('node->state=',1)[1].rsplit('?1:2;',1)[0]
 statement=['node->state='+test+'?1:2;','if('+test+')node->state=1;else node->state=2;','bool hit='+test+';node->state=hit?1:2;','node->state=2-('+test+'?1:0);'][assign]
 pres=[
  'if(node->state==0) {'+statement+'}',
  'switch(node->state){case 0:{'+statement+'}break;default:break;}',
  'switch(node->state){default:break;case 0:{'+statement+'}break;}',
  'switch(node->state){case 0:{'+statement+'}break;case 2:break;default:break;}',
  'switch(node->state){case 2:break;default:break;case 0:{'+statement+'}break;}'
 ][pre]
 posts=['if(node->state==2) continue;', 'switch(node->state){case 2:continue;default:break;}', 'switch(node->state){default:break;case 2:continue;}', 'switch(node->state){case 0:break;case 1:break;case 2:continue;default:break;}'][post]
 s=s[:a]+'  '+pres+'\n  '+posts+s[z+len('  if(node->state==2) continue;'):]
 if guard:
  s=s.replace('if(node->first->source==0 || node->second->source==0) continue;', 'if(node->first->source==0)continue; if(node->second->source==0)continue;')
 (out/('v%04d.cpp'%idx)).write_text(s)
print('generated',idx+1)
