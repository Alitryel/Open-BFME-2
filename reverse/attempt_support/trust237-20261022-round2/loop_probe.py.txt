from pathlib import Path
import itertools,sys,subprocess,json,concurrent.futures
sys.path.insert(0,str(Path('tools').resolve()))
import build,capstone
OUT=Path('build/trust237-round2').resolve();ROOT=Path.cwd()
def run(job):
 n,cpu,opt,shape,k=job
 flags=f'// cl: /{opt} /G{cpu} /arch:SSE /DNDEBUG /MD /EHsc\n'
 pre='n=n*3u+1u; '*k
 loops=[
  'for(unsigned i=0;i<n;++i) p[i]=(int)i;',
  'while(n) {*p++=(int)n;--n;}',
  'for(unsigned i=0;i<n;++i) p[i]+=(int)(n-i);',
  'for(unsigned i=0;i<n;++i) {p[i]=(int)i; if(p[(i+1)%n]) p[i]+=1;}',
 ]
 src=OUT/f'loop{n:03}.cpp';obj=src.with_suffix('.obj')
 src.write_text(flags+'int alignmentProbe(unsigned n,int *p){'+pre+loops[shape]+' return (int)n;}')
 cmd,env=build.compiler_command(src,obj);proc=subprocess.run(cmd,cwd=ROOT,env=env,capture_output=True,text=True)
 if proc.returncode:return {'n':n,'error':(proc.stdout+proc.stderr)[-1200:]}
 name=next(s for s in build.defined_code_symbols(obj) if s.startswith('?alignmentProbe@'))
 raw,rel=build.read_object_symbol_bytes(obj,name,code_only=True)
 md=capstone.Cs(capstone.CS_ARCH_X86,capstone.CS_MODE_32)
 ins=list(md.disasm(raw,0))
 return dict(n=n,cpu=cpu,opt=opt,shape=shape,k=k,size=len(raw),nop16=b'\x90'*16 in raw,nops=[dict(off=i.address,size=i.size,bytes=i.bytes.hex()) for i in ins if i.mnemonic=='nop'],same_reg_leas=[dict(off=i.address,asm=i.mnemonic+' '+i.op_str,bytes=i.bytes.hex()) for i in ins if i.mnemonic=='lea' and i.op_str.startswith(('ecx, [ecx','esi, [esi','edi, [edi','ebx, [ebx'))],raw=raw.hex())
jobs=[(i,*j) for i,j in enumerate(itertools.product([5,6,7],['O1','O2'],range(4),[0,1,2,3]))]
with concurrent.futures.ThreadPoolExecutor(max_workers=8) as pool:rows=list(pool.map(run,jobs))
(OUT/'loop_probe.json').write_text(json.dumps(rows,indent=2))
good=[r for r in rows if 'error' not in r]
print('Probes',len(rows),'compiled',len(good),'nop16',sum(r['nop16'] for r in good),'single-NOP',sum(bool(r['nops']) for r in good),'LEA-like',sum(bool(r['same_reg_leas']) for r in good))
for r in [r for r in good if r['nops'] or r['same_reg_leas']][:6]:
 print({k:v for k,v in r.items() if k!='raw'})
