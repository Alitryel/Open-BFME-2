from search import *
import itertools
s=(W/'base.cpp').read_text();vs=[]
forms=[('{v}-{k}',''),('{v}^{k}',''),('{v}=={k}?0:{v}',''),('{v}!={k}',''),
 ('z','int z={v};z-={k};'),('z','int z={v};z^={k};'),
 ('z','int z={v};if(z=={k})z=0;'),('z','int z;switch({v}){case {k}:z=0;break;default:z={v};break;}'),
 ('{v}-{k}','__assume({v}=={k});'),('{v}^{k}','__assume({v}=={k});'),
 ('z','int z={v}-{k};__assume(z==0);'),('{v}=={k}?0:{v}','__assume({v}=={k});')]
for form,site,cached,where in itertools.product(range(len(forms)),range(3),range(2),range(2)):
 o=s
 a=s.index('  case 1:');b=s.index('  case 4:',a);c=s.index('  case 6:');d=s.index('  case 7:',c)
 def region(blk,k):
  expr,dec=forms[form]
  expr=expr.replace('{v}','savedMode' if cached else 'mode').replace('{k}',str(k))
  dec=dec.replace('{v}','savedMode' if cached else 'mode').replace('{k}',str(k))
  if where==0:blk=blk.replace('   requires=',dec+'   requires=',1)
  elif k==1:blk=blk.replace('   else pos.z=','   else {'+dec+'pos.z=',1).replace('object->rva0028B511(),0,true);','object->rva0028B511(),0,true);}',1)
  else:blk=blk.replace('   pos.z=','   {'+dec+'pos.z=',1).replace('->ground(pos.x,pos.y,0);','->ground(pos.x,pos.y,0);}',1)
  if k==1:blk=blk.replace('object->rva0028B511(),0,true)','object->rva0028B511(),(Coord3D*)('+expr+'),true)')
  else:blk=blk.replace('->ground(pos.x,pos.y,0)','->ground(pos.x,pos.y,(Coord3D*)('+expr+'))')
  return blk
 if site in [0,2]:o=o[:c]+region(o[c:d],6)+o[d:]
 if site in [0,1]:o=o[:a]+region(o[a:b],1)+o[b:]
 if cached:o=o.replace(' switch(mode)',' const int savedMode=mode;\n switch(mode)')
 vs.append((f'casefacts_{form:02d}_{site}{cached}{where}',o))
run(vs,'casefacts144')
