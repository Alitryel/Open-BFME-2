from pathlib import Path
import sys,itertools,concurrent.futures,json
sys.path.insert(0,'build/df20-scalars-20261020');import trials
trials.scratch=Path.cwd()/'build/df20-scalars-20261020'
base=Path('build/df20-scalars-20261020/bank.cpp').read_text();trials.symbol=base.splitlines()[0][3:]
hdr=Path('vendor/stlport/stl/_ostreambuf_iterator.h').read_text()
specs=[];seen=set()
for ctor,assign,opaque,nt,force in itertools.product(range(3),range(3),[0,1],[0,1],[0,1]):
 if ctor==assign==0:continue
 attr='__forceinline' if force else 'inline';exc=' throw()' if nt else ''
 decl=''
 if ctor:
  body=' : _M_buf(s._M_buf), _M_ok(s._M_ok) {}' if ctor==1 else ' { ::memcpy(this,&s,sizeof(*this)); }'
  decl+=f'  {attr} ostreambuf_iterator(const ostreambuf_iterator& s){exc}'+body+'\n'
 if assign:
  body='_M_buf=s._M_buf; _M_ok=s._M_ok;' if assign==1 else '::memcpy(this,&s,sizeof(*this));'
  decl+=f'  {attr} ostreambuf_iterator& operator=(const ostreambuf_iterator& s){exc} {{ {body} return *this; }}\n'
 label=f'iter-c{ctor}-a{assign}-o{opaque}-n{nt}-f{force}'
 d=trials.scratch/label/'stl';d.mkdir(parents=True,exist_ok=True)
 (d/'_ostreambuf_iterator.h').write_text(hdr.replace('  typedef output_iterator_tag',decl+'\n  typedef output_iterator_tag',1))
 s=base.replace('// cl:','// cl: /I'+str(d.parent.relative_to(Path.cwd())),1)
 if not opaque:s=s.replace('template <> _String_base<wchar_t, allocator<wchar_t> >::~_String_base();','')
 specs.append((label,s))
with concurrent.futures.ThreadPoolExecutor(max_workers=8) as p:r=list(p.map(trials.run,specs))
(trials.scratch/'iterators.json').write_text(json.dumps(r,indent=2))
