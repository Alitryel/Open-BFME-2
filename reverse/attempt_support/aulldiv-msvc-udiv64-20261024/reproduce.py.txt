from pathlib import Path
import sys,json,hashlib,subprocess,os
sys.path.insert(0,'tools');import build
ROOT=build.ROOT
bank=ROOT/'reverse/attempt_support/aulldiv-msvc-udiv64-20261024'
proof=json.loads((bank/'input-proof.json').read_text());groups={}
for row in proof['trials']:
 label=row['label'];template={'atom':'cross-first','frame-control':'mul-order'}.get(label,label)
 text=(bank/(template+'.cpp.txt')).read_text()
 flags=row.get('flags')
 if flags:
  key='/O1 /G6' if label=='full' else '/O1'
  text=text.replace(key,'/'+flags.replace('-',' /'))
 text=text.rstrip()+'\n'*row['trailing_newlines']
 source=ROOT/row['command'][-1];source.parent.mkdir(parents=True,exist_ok=True);source.write_text(text)
 assert hashlib.sha256(source.read_bytes()).hexdigest()==row['source_sha256'],str(source)
 compiler=Path(row['command'][0])
 for name,key in (('cl.exe','compiler_sha256'),('c1xx.dll','c1xx_sha256'),('c2.dll','c2_sha256')):
  assert hashlib.sha256((compiler.parent/name).read_bytes()).hexdigest()==row[key],name
 output=ROOT/next(arg[3:] for arg in row['command'] if arg.startswith('-Fo'))
 command,env=build.compiler_command(source,output);command[0]=str(compiler)
 assert command==row['command'],(command,row['command'])
 env['PATH']=str(compiler.parent)+os.pathsep+env.get('PATH','')
 result=subprocess.run(command,cwd=ROOT,env=env,stdout=subprocess.PIPE,stderr=subprocess.STDOUT,text=True)
 assert result.returncode==0,result.stdout
 symbol='_probe@16' if label=='probe' else '__aulldiv@16'
 raw,rel,info=build.read_object_symbol_bytes(output,symbol,10000,code_only=True,detail=True)
 assert hashlib.sha256(raw).hexdigest()==row['object_body_sha256'],str(output)
 assert not rel,rel
 groups.setdefault(label,[]).append(dict(row,bytes=raw.hex()))
for label,rows in groups.items():
 (ROOT/'build/aulldiv-modern'/(label+'-results.json')).write_text(json.dumps(rows,indent=2)+'\n')
print('Reproduced',sum(map(len,groups.values())),'private compiler trials; all source/toolchain/body hashes and complete relocations match.')
