import sys,os
sys.path.insert(0,'/work/Open-BFME-2/build/c2tr')
from wdbg import Dbg
# t50.py SRC OBJ : per round, at each coloring (82a0d) print rec, reg, members of esi set and eax return after 8368e
src,obj=sys.argv[1:3]; KMAX=int(os.environ.get('KMAX','3')); CLR=os.environ.get('CLR','')  # CLR=R:recname:victimname
d=Dbg(src,obj,obj+'.log')
d.prompt();d.prompt('set $CanDeferOnBPByAddr = 1')
B=0x10700000
bps=[0x83acd,0x82978,0x82a0d,0x82a12]; added=set()
for b in bps:
    o=d.prompt('break *0x%x'%(B+b))
    if 'Breakpoint' in o and 'Unable' not in o: added.add(b)
names={}
def nm(sym):
    if sym in names: return names[sym]
    sw=d.words(sym,1); pw=d.words(sw[0],2) if sw and sw[0]>0x100000 else []
    n=d.string(pw[1]).strip() if len(pw)>1 and pw[1]>0x100000 else '?'
    n=n.strip('"').lstrip('_'); names[sym]=n; return n
def rec_of(i):
    p=d.words(0x108807b8+(i&0x3ff)*4,1)[0]
    while p:
        w=d.words(p,13)
        if len(w)<13: return None,None
        if w[7]==i: return p,w
        p=w[12]
    return None,None
def chunks(hdr):
    out=[]; ch=d.words(hdr,1)[0] if hdr else 0; n=0
    while ch and n<200:
        ww=d.words(ch,3)
        if len(ww)<3: break
        out.append((ch,)+tuple(ww)); ch=ww[1]; n+=1
    return out
def members(hdr):
    m=[]
    for ch,base,nx,bits in chunks(hdr): m+=[base+j for j in range(32) if bits>>j&1]
    return m
def lab(i):
    p,w=rec_of(i); return '%s:%x'%(nm(w[0]),i) if w else '%x'%i
on=False;k=0;cur=None
CR,CN,CV=(CLR.split(':')+['','',''])[:3] if CLR else (None,None,None)
while True:
    out=d.prompt('cont')
    if 'terminated' in out: break
    eip=d.reg('eip')-B
    for b in bps:
        if b not in added and b!=eip: d.prompt('break *0x%x'%(B+b)); added.add(b)
    added.add(eip)
    if eip==0x83acd:
        ebp=d.reg('ebp'); f=d.words(ebp,1)[0]; curfn=d.string(d.words(f+4,1)[0]); on='updateMeleeAttack' in curfn and d.reg('ebx')==0; k=0; continue
    if not on: continue
    if eip==0x82978:
        k+=1
        if k>KMAX: break
        continue
    if eip==0x82a0d:
        rec=d.reg('ebp'); w=d.words(rec,13); cur=(rec,w,d.reg('ebx'),d.reg('esi')); continue
    if eip==0x82a12 and cur:
        rec,w,reg,es=cur; ea=d.reg('eax')
        mi=members(es); 
        try: me=members(ea) if ea>0x100000 else ea
        except Exception: me='?'
        print('R%d color %s:%x reg=%d esi=%s eax=%s'%(k,nm(w[0]),w[7],reg,[lab(i) for i in mi],[lab(i) for i in me] if isinstance(me,list) else me),flush=True)
        if CR and int(CR)==k and nm(w[0])==CN:
            for hdr in (es,ea):
                if not (isinstance(hdr,int) and hdr>0x100000): continue
                for ch,base,nx,bits in chunks(hdr):
                    for j in range(32):
                        if bits>>j&1 and lab(base+j).split(':')[0]==CV:
                            nb=bits&~(1<<j); d.prompt('set *(int*)0x%x = %d'%(ch+8,nb-(1<<32) if nb>=1<<31 else nb)); bits=nb
                            print('cleared',CV,hex(base+j),'in',hex(hdr),flush=True)
        cur=None
print('done',flush=True)
