from pathlib import Path
import sys,subprocess,json,hashlib
root=Path(__file__).resolve().parents[1];sys.path.insert(0,str(root/'tools'));import build
out=root/'build/zh-sdk-followups-oct20';out.mkdir(exist_ok=True)
inputs=json.loads((root/'build/zh-sdk-repair-oct20/inputs.json').read_text())
results=[]
for task in inputs:
    name=Path(task['donor']).name
    if name not in ('wwmouse.cpp','W3DShaderManager.cpp'):continue
    prefix='#include <windows.h>\n#include <mmsystem.h>\n' if name=='wwmouse.cpp' else '#include <new>\n#define _OPERATOR_NEW_DEFINED_\nclass PersistFactoryClass;\n'
    text=(root/task['trial']).read_text();at=text.index('#include ');text=text[:at]+prefix+text[at:]
    trial=out/Path(task['trial']).name;trial.write_text(text)
    assert build.source_extra_flags(trial)==task['flags']
    p=subprocess.run([sys.executable,'tools/place_bodies.py','--min','8',trial.relative_to(root).as_posix()],cwd=root,capture_output=True,text=True,encoding='utf8',errors='replace',timeout=240)
    output=p.stdout+p.stderr;trial.with_suffix('.log').write_text(output)
    r={**task,'new_trial':trial.relative_to(root).as_posix(),'prefix':prefix,'compile_failed':p.returncode!=0 or 'COMPILE FAILED' in output,'errors':[s for s in output.splitlines() if 'error ' in s][:5],'leads':[s for s in output.splitlines() if s.startswith(('BODY\t','PIN\t'))],'log_sha256':hashlib.sha256(output.encode()).hexdigest()}
    results.append(r);(out/'results.json').write_text(json.dumps(results,indent=2)+'\n');print(name,task['profile'],r['compile_failed'],r['leads'],r['errors'][:1],flush=True)
