"""fr.py file.cpp... : nd (instruction diff), slot mismatches (aligned insns differing only in ebp offsets),
and the slot of adjustTo / options / goalBlockNdx."""
import sys, re, difflib
sys.path.insert(0, '/work3/Open-BFME-2/build/h72')
from score import compile_, md, norm
from concurrent.futures import ThreadPoolExecutor
EBP = re.compile(r'\[ebp ([-+]) (0x[0-9a-f]+|\d+)\]')


def key(n):
    return (n[0], EBP.sub('[ebp]', n[1]))


def one(src):
    try:
        r, err = compile_(src)
    except BaseException as e:
        return f'{src}: ERR {e!r}'[:200]
    if r is None:
        return f'{src.split("/")[-1]}: FAIL ' + ' | '.join(l for l in err.splitlines() if 'error' in l)[:200]
    got, raw = r
    T = [norm(i, 9999) for i in md.disasm(bytes(got['target']), 0)]
    full = bytes(raw).rstrip(b'\xcc') if raw else got['bytes']
    O = [norm(i, 9999) for i in md.disasm(full, 0)]
    sm = difflib.SequenceMatcher(None, T, O, autojunk=False)
    nd = sum(max(i2 - i1, j2 - j1) for tg, i1, i2, j1, j2 in sm.get_opcodes() if tg != 'equal')
    sm2 = difflib.SequenceMatcher(None, [key(x) for x in T], [key(x) for x in O], autojunk=False)
    slot = 0
    for tg, i1, i2, j1, j2 in sm2.get_opcodes():
        if tg == 'equal':
            slot += sum(1 for k in range(i2 - i1) if T[i1 + k] != O[j1 + k])
    nd2 = sum(max(i2 - i1, j2 - j1) for tg, i1, i2, j1, j2 in sm2.get_opcodes() if tg != 'equal')
    L = list(md.disasm(full, 0))
    st = [i.op_str for i in L if i.mnemonic == 'movss' and i.op_str.startswith('dword ptr [ebp')][:1]
    g = [i.op_str for i in L if i.mnemonic == 'or' and '0xffffffff' in i.op_str][:1]
    sur = [i.op_str for i in L if 0x380 < i.address < 0x3c0 and i.mnemonic == 'mov' and i.op_str.startswith('dword ptr [ebp') and 'ecx' in i.op_str][:1]
    f = lambda s: (EBP.search(s[0]).group(0) if s and EBP.search(s[0]) else '?').replace('ebp ', '').replace(' ', '')
    return f'{src.split("/")[-1]}: len={len(full)} nd={nd} nslot={nd2} slotd={slot} pt{f(st)} op{f(sur)} gb{f(g)}'


if __name__ == '__main__':
    with ThreadPoolExecutor(8) as ex:
        for line in ex.map(one, sys.argv[1:]):
            print(line, flush=True)
