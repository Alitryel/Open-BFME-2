"""Scratch driver for ordinary bounded verification commands; no gate overrides."""
from pathlib import Path
import hashlib
import json
import os
import subprocess
import sys
import time

ROOT = Path(__file__).resolve().parents[2]
OUT = Path(__file__).resolve().parent
step = sys.argv[1]
if step in ("prepare-1", "prepare-2"):
    batch = step.rsplit("-", 1)[1]
    args = ["--prepare-current", "--paths-from", str(OUT / f"sources-{batch}.txt")]
elif step == "preview":
    args = ["--paths-from", str(OUT / "sources.txt")]
elif step == "byte-gate":
    args = (OUT / "sources.txt").read_text(encoding="utf-8").splitlines()
else:
    raise SystemExit("Unknown step")
command = [sys.executable, "-u", "tools/build.py" if step == "byte-gate" else "tools/link_check.py", *args]
env = dict(os.environ, BUILD_POOL="4", PYTHONIOENCODING="utf-8")
started = time.time()
log_path = OUT / f"{step}.log"
print(f"Starting {step}: ordinary verification command, BUILD_POOL=4", flush=True)
with log_path.open("wb") as log:
    result = subprocess.run(command, cwd=ROOT, env=env, stdout=log, stderr=subprocess.STDOUT)
record = {"command": command, "returncode": result.returncode,
          "elapsed_seconds": round(time.time() - started, 3),
          "log_sha256": hashlib.sha256(log_path.read_bytes()).hexdigest()}
(OUT / f"{step}-result.json").write_text(json.dumps(record, indent=2) + "\n", encoding="utf-8")
print(json.dumps(record, indent=2))
print("\n".join(log_path.read_text(encoding="utf-8", errors="replace").splitlines()[-18:]))
raise SystemExit(result.returncode)
