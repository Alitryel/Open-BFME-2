from itertools import product
from pathlib import Path
W=Path(__file__).resolve().parent
bank=(W/'bank.cpp').read_text();uniform=(W/'uniform-control.cpp').read_text()
D=W/'dispatch';D.mkdir(exist_ok=True)
for idx,opts in enumerate(product(range(4),repeat=6)):
 for control in range(2):
  s=uniform if control else bank
  defs='''struct __single_inheritance ContactSource;
typedef void *(ContactSource::*PrepareMethod)(const ContactVector &,const ContactVector &);
typedef bool (ContactSource::*TestMethod)(void *);
typedef void (ContactSource::*RespondMethod)(void *);
struct DispatchTable { void *pad0[3]; PrepareMethod prepare; void *pad1[4]; TestMethod test; RespondMethod respond; };
typedef char wordcheck[sizeof(PrepareMethod)==4?1:-1];
'''
  sources=['node->second->source','node->first->source','node->second->source','node->first->source','node->first->source','node->second->source']
  methods=['prepare','test','prepare','respond','prepare','respond']
  # All replacements refer to the bank/control's original source text.
  inner0='node->second->source->prepare(node->point,node->normal)'
  inner1='node->second->source->prepare(node->point,node->normal)' if control else 'bridge(node->second,node->point,node->normal)'
  inner2='node->first->source->prepare(node->point,ContactVector(node->normal).negate())' if control else 'altPrepare(node->first->source,node->point,ContactVector(node->normal).negate())'
  inners=[inner0,inner1,inner2]
  for j in range(3):
   outer=sources[j*2+1];method=methods[j*2+1]
   old=outer+'->'+method+'('+inners[j]+')'
   newinner=inners[j]
   for n in range(j*2,j*2+2):
    opt=opts[n]
    if not opt:continue
    meth=methods[n];name='dispatch'+str(n)
    tabtype='volatile DispatchTable *' if opt==3 else 'DispatchTable *'
    readtype='DispatchTable * volatile *' if opt==2 else 'DispatchTable **'
    tab='(('+tabtype+')*('+readtype+')s)'
    if meth=='prepare':
     decl='void *'+name+'(ContactSource *s,const ContactVector &p,const ContactVector &n)'
     ret='return (s->*'+tab+'->prepare)(p,n);'
     args='node->point,'+('ContactVector(node->normal).negate()' if j==2 else 'node->normal')
     newinner=name+'('+sources[n]+','+args+')'
    else:
     decl=('bool ' if meth=='test' else 'void ')+name+'(ContactSource *s,void *v)'
     ret=('return ' if meth=='test' else '')+'(s->*'+tab+'->'+meth+')(v);'
    defs+='__forceinline '+decl+' {'+ret+'}\n'
   new=(('dispatch'+str(j*2+1)+'('+outer+','+newinner+')') if opts[j*2+1] else outer+'->'+method+'('+newinner+')')
   assert old in s,(idx,control,j,old)
   s=s.replace(old,new,1)
  a=s.index('struct ContactData {')
  s=s[:a]+defs+s[a:]
  (D/('v%05d.cpp'%(idx*2+control))).write_text(s)
print('generated',(idx+1)*2)
