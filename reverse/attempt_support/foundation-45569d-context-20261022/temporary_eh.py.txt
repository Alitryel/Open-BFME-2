from search import *
from context import getters,mutators
import itertools

def temporaries(s,mask,style):
 # Empty destructors are an ABI experiment only. The helpers' visible bodies
 # come from matched providers and write only the supplied 76-byte bitset.
 if mask&1:s=s.replace('#define COORD3D_VIEW_INLINE','#define COORD3D_VIEW_DTOR\n#define COORD3D_VIEW_INLINE',1)
 if mask&2:s=s.replace('struct CreateMask\n{','struct CreateMask\n{\n ~CreateMask() {}',1)
 if mask&4:s=s.replace('class Rva001E4912\n{\npublic:','class Rva001E4912\n{\npublic:\n ~Rva001E4912() {}',1)
 if mask&8:s=s.replace('class Rva0028F59A\n{\npublic:','class Rva0028F59A\n{\npublic:\n ~Rva0028F59A() {}',1)
 if mask&16:s=s.replace('struct CreateMask\n{','struct CreateMask\n{\n CreateMask() {}',1)
 if mask&32:s=s.replace('Rva001E4912 *rva001E4912(int unused, unsigned bit1, unsigned bit2);','Rva001E4912() {}\n Rva001E4912 *rva001E4912(int unused, unsigned bit1, unsigned bit2);',1)
 spec=['',' throw()',''][style]
 attr='__declspec(nothrow) ' if style==2 else ''
 if mask&64:
  s=s.replace('Rva0028F59A(int unused, int bit);',attr+'__declspec(noinline) Rva0028F59A(int unused, int bit)'+spec+' {memset(this,0,0x4c);m_words[(unsigned)bit>>5] |= 1u<<(bit&31);}',1)
  # The declaration precedes these inline bodies just as the real header does.
  s=s.replace('// Rowed bit-set helpers', 'extern "C" void *__cdecl memset(void *,int,unsigned int);\n// Rowed bit-set helpers',1)
 if mask&128:
  s=s.replace('Rva001E4912 *rva001E4912(int unused, unsigned bit1, unsigned bit2);',attr+'__declspec(noinline) Rva001E4912 *rva001E4912(int unused, unsigned bit1, unsigned bit2)'+spec+' {memset(this,0,0x4c);m_words[bit1>>5] |= 1u<<(bit1&31);m_words[bit2>>5] |= 1u<<(bit2&31);return this;}',1)
  s=s.replace('// Rowed bit-set helpers', 'extern "C" void *__cdecl memset(void *,int,unsigned int);\n// Rowed bit-set helpers',1)
 if mask&256:
  s=s.replace('Real normalizeAngle(Real angle);',attr+'Real normalizeAngle(Real angle)'+spec+';',1)
 if mask&512:s=s.replace('Real Get_Z_Rotation() const;',attr+'Real Get_Z_Rotation() const'+spec+';',1)
 return s

tests={}
for style,mask in itertools.product(range(3),range(1024)):
 tests[f'temporary-{style}-{mask:04d}']=temporaries(BASE,mask,style)
run(tests,'temporary-eh')
