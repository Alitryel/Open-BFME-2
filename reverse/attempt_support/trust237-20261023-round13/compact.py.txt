from pathlib import Path
import sys,itertools
ROOT=Path.cwd();OUT=ROOT/'build/trust237-round13'
sys.path.insert(0,str(ROOT/'build/trust237-round9'));import frontends as f
f.OUT=OUT
# Follow the observed f0180 region winner: full-width immediate equality
# folds one arm's range result; imm8 constants may shorten its retained test.
values=['0u','1u','2u','3u','127u','0xffffff80u','0xfffffffeu','0xffffffffu']
predicates=['(u&1u)<=1u','1u>=(1u&u)','(u%2u)<=1u',
 '(unsigned char)(u&1u)<=1u','(u&bound)<=bound',
 '(u&0x7fu)<=0x7fu','(u|1u)!=0u','(u>>31)<=1u']
jobs=[]
for val,pred,shape,ref,reverse in itertools.product(values,predicates,[0,1],[False,True],[False,True]):
    decl='const unsigned long &bound=1u;' if ref else 'unsigned long bound=1u;'
    cond=('u!=' if reverse else 'u==')+val
    if shape==0:
        tail='if(P){releaseArgs(argv);r=(int)(R);}else{releaseArgs(argv);r=(int)(R);}return r;'
    else:
        tail='if(P)goto left;goto right;right:releaseArgs(argv);r=(int)(R);goto done;left:releaseArgs(argv);r=(int)(R);goto done;done:return r;'
    src=f.base.replace(f.marker,'long status=WinVerifyTrust(0,&action,&data);unsigned long u=(unsigned long)status;int r=0;'+decl+tail.replace('P',cond).replace('R',pred))
    src=src.replace('int Rva00237D3DCheck() {','__forceinline void releaseArgs(unsigned short **p) throw(){LocalFree(p);}\nint Rva00237D3DCheck() {').replace('/O1 /G7','/O1 /G6')
    jobs.append((f'k{len(jobs):04}',src,dict(value=val,predicate=pred,shape=shape,bound_reference=ref,reverse=reverse)))
if __name__=='__main__':f.execute('compact',jobs)
