from pathlib import Path
import itertools
r=Path(__file__).parent;out=r/'leaf_context';out.mkdir(exist_ok=True)
bases=[(r/'bank.cpp').read_text(),(r/'native_split.cpp').read_text(),(r/'native_or.cpp').read_text()]
team='''

'''

 Module *found=0;for(Module **at=m_modules;*at;++at){if((*at)->getModuleNameKey()==key){found=*at;break;}}return found;}
'''
for bi,mask,order,nothrow in itertools.product(range(3),range(16),range(2),range(2)):
 s=bases[bi]
 s=s.replace('class Object\n{',team+'\nclass Object\n{',1)
 s=s.replace('char m_pad44[0x438 - 0x44];','''char m_pad44[0x84-0x44];
 Drawable *m_drawable;

 Module **m_modules;

 if mask&8:
  s=s.replace('void *m_vtable;\n\tconst ModuleData *m_moduleData;', '''virtual void dtorInView()=0;
 virtual void slot01InView()=0;
 virtual void slot02InView()=0;
 virtual void slot03InView()=0;
 virtual NameKeyType getModuleNameKey()const=0;
 const ModuleData *m_moduleData;''',1)
 suffix='throw()'if nothrow else''
 if nothrow:
  s=s.replace('Drawable *getDrawable() const;','Drawable *getDrawable() const throw();')
  s=s.replace('const Coord3D *getPosition() const {','const Coord3D *getPosition() const throw() {')
 add=(owner if mask&4 else'')+(ctrl if mask&2 else'')+(draw.replace('SUFFIX',suffix)if mask&1 else'')+(lookup if mask&8 else'')
 if order:s+='\n'+add
 else:s=s.replace('void Drawable::onComingOutOfShroud()',add+'\nvoid Drawable::onComingOutOfShroud()',1)
 (out/f'b{bi}m{mask:02d}o{order}t{nothrow}.cpp').write_text(s)
