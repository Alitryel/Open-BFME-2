import runpy,itertools,concurrent.futures,json

def expr(c,v,w,k):
 c='(unsigned int)('+c+')'
 return [f'{c}?{v}:{w}',f'{w}+({v}-{w})*{c}',f'({v}&(0u-{c}))|({w}&(0u-!{c}))',f'({v}*{c})+({w}*!{c})',f'{w}^(({w}^{v})&(0u-{c}))',f'{v}+({w}-{v})*!{c}',f'({c}*{v})|(!{c}*{w})'][k]
tests=[]
for q,s,f,guard,local,scope in itertools.product(range(8),range(8),range(8),range(2),range(2),range(2)):
 b=body
 for which,k in enumerate([q,s,f]):
  if not k:continue
  var='chosen'+str(which) if local else 'selected.code'
  decl='unsigned int '+var+';' if local else ''
  c=['failed','result.m_20==7','money<(unsigned int)result.m_pad04'][which]
  v=[9,2,2][which];w=[0,9,0][which]
  assign=decl+var+'='+expr(c,v,w,k-1)+';\n'
  ret='{'+('selected.code='+var+';return selected.code;' if local else 'return selected.code;')+'}'
  if which==0:
   b=b.replace('if (failed)\n {selected.code=9;return selected.code;}',assign+'if ('+(var if guard else 'failed')+')\n'+ret)
  elif which==1:
   a=b.index('switch (result.m_20)');z=b.index('unsigned int money')
   sw=b[a:z].replace('{selected.code=2;return selected.code;}',ret).replace('{selected.code=9;return selected.code;}',ret)
   b=b[:a]+assign+sw+b[z:]
  else:
   b=b.replace('if (money < result.m_pad04)',assign+'if ('+(var if guard else 'money < result.m_pad04')+')')
   # Only the guarded funds arm uses the selected value. The false arm is unreachable by the enclosing guard.
   b=b.replace('{selected.code=2;return selected.code;} }while(0);',ret+' }while(0);')
 if scope:b=b.replace('MakeCodeLife selected;','').replace('Rva0039205C result;','MakeCodeLife selected;\nRva0039205C result;')
 tests.append((f'pre-{q}{s}{f}{guard}{local}{scope}',prefix+'\nint AIBuildableWallSegment::canMake(Player *player)'+b))

if case:tests=[x for x in tests if x[0]==case];assert tests,case

 for i,d in enumerate(pool.map(r['run'],tests),1):
  results.append(d)
  if i%512==0 or d.get('exact_candidate'):print(json.dumps(dict(done=i,total=len(tests),best=sorted(results,key=lambda x:(x.get('nd',999),-x.get('score',0)))[:2])),flush=True)
