"""Read-only current-object audit after the ordinary gates and link preview."""
from pathlib import Path
import collections
import csv
import hashlib
import json
import subprocess
import sys

ROOT = Path(__file__).resolve().parents[2]
OUT = Path(__file__).resolve().parent
sys.path.insert(0, str(ROOT / "tools"))
import build
import link_check
import link_census

def sha(path):
    return hashlib.sha256(path.read_bytes()).hexdigest()

def git(*args, cwd=ROOT):
    return subprocess.check_output(["git", *args], cwd=cwd, text=True).strip()

symbol = "??_EWeightInfoStruct@NamedPivotMapClass@@QAEPAXI@Z"
historical = json.loads((OUT / "historical.json").read_text(encoding="utf-8"))
sources = (OUT / "sources.txt").read_text(encoding="utf-8").splitlines()
assert len(sources) == 34 and set(sources) == set(historical["entries"])
gate_result = json.loads((OUT / "byte-gate-result.json").read_text(encoding="utf-8"))
assert gate_result["returncode"] == 0, gate_result
preparation = json.loads((OUT / "prepare-1-result.json").read_text(encoding="utf-8"))
preview_result = json.loads((OUT / "preview-result.json").read_text(encoding="utf-8"))
assert preview_result["returncode"] in (0, 1)
assert git("status", "--porcelain", "--untracked-files=no") == "", "Input worktree changed"
assert git("rev-parse", "HEAD", cwd=build.BFME1_ROOT) == git("rev-parse", "HEAD:reference/open-bfme-1")
build.ensure_reference_current()
objects = [build.obj_path(ROOT / source) for source in sources]
strict = {}
inventory_cache = {}
for source, obj in zip(sources, objects):
    assert build.compile_is_current(ROOT / source, obj), source
    strict[source] = link_census.object_current(ROOT / source, obj, allow_fresh=False,
                                               inventory_cache=inventory_cache)
assert build._inventory_cache_still_current(inventory_cache), "Include inventory changed"
index_hash = sha(link_check.INDEX)
index = link_check.load_index()
rows = link_census.ledger()
truth = link_census.RetailTruth(rows)
link_check.refresh(index, objects, truth)
weights = link_check.source_bytes(set(sources))
positions = {name:i for i,name in enumerate(index["objects"])}
copies = index["comdat"].get(symbol, [])
definers = {i for i,_,_ in copies} | set(index["strong"].get(symbol, []))
holder = index["selection"]["exceptions"].get(symbol, min(definers))
if holder is not None and holder not in definers:
    holder = min(definers)
selection = link_census.judge_selected(
    index["objects"][holder] if holder is not None else None,
    {index["objects"][i]:(d,v) for i,d,v in copies},
    {index["objects"][i] for i in definers},
    {index["objects"][i] for i in index["strong"].get(symbol, [])},
    index["selection"]["owners"].get(symbol, set()))
audited = []
fields = ("unresolved", "duplicates", "comdat", "addresses", "selected")
for source, obj in zip(sources, objects):
    result = link_check.check_object(obj, index, truth, source)
    result = {key:sorted(result[key]) for key in fields}
    violations = [key for key in fields if any(symbol == item or
                  (isinstance(item, (list, tuple)) and symbol == item[0])
                  for item in result[key])]
    old = historical["entries"][source]
    old_names = {(kind, name) for kind in ("unresolved", "duplicates", "losers", "wrong_selected")
                 for name in old.get(kind, ())}
    sole = old_names == {("wrong_selected", symbol)} and not old.get("addresses")
    audited.append({"source":source, "object":obj.name, "source_sha256":sha(ROOT/source),
                    "object_sha256":sha(obj), "receipt_sha256":sha(obj.with_suffix(".deps.json")),
                    "ordinary_receipt_current":True, "strict_receipt_current":strict[source],
                    "historical_sole_blocker":sole,
                    "historical_authored_bytes":historical["bytes"].get(source, 0),
                    "current_authored_nonpadding_bytes":weights.get(source, 0),
                    "defines_destructor":positions[obj.name] in {i for i,_,_ in copies},
                    "destructor_blockers":violations,
                    "preview_links":not any(result.values()), "remaining":result})
assert sha(link_check.INDEX) == index_hash, "Published index changed"
assert all(not row["destructor_blockers"] for row in audited)
assert copies and all(v == "retail" for _,_,v in copies), copies
assert selection == "ok", selection
counts = collections.Counter()
raw = 0
with (ROOT / "reverse/functions.csv").open(encoding="utf-8", newline="") as handle:
    for row in csv.DictReader(handle):
        if row["source"] in sources and row["status"] == "matched":
            counts[row["source"]] += 1
            raw += int(row["target_size"] or 0, 0)
report = {"scope":"Bounded ordinary byte verification and legacy link preview; strict preparation refusal retained; no census, link.exe or closed-strict result",
          "revision":git("rev-parse", "HEAD"), "reference_revision":git("rev-parse", "HEAD", cwd=build.BFME1_ROOT),
          "symbol":symbol, "published_index":index["meta"], "published_index_sha256":index_hash,
          "current_providers":len(copies), "provider_verdicts":dict(collections.Counter(v for _,_,v in copies)),
          "provider_digest_kinds":len({d for _,d,_ in copies}),
          "selected_object":index["objects"][holder] if holder is not None else None,
          "selection_verdict":selection,
          "providers":[{"object":index["objects"][i], "digest":d, "verdict":v} for i,d,v in copies],
          "verified_units":len(audited), "verified_matched_rows":sum(counts.values()),
          "strict_reusable_receipt_units":sum(strict.values()),
          "strict_preparation_returncode":preparation["returncode"],
          "matched_raw_row_bytes":raw, "preview_clean_units":sum(x["preview_links"] for x in audited),
          "preview_clean_authored_nonpadding_bytes":sum(x["current_authored_nonpadding_bytes"] for x in audited if x["preview_links"]),
          "historical_sole_blocker_units":sum(x["historical_sole_blocker"] for x in audited),
          "units":audited}
(OUT / "audit.json").write_text(json.dumps(report, indent=2) + "\n", encoding="utf-8")
print(json.dumps({k:v for k,v in report.items() if k not in ("units", "providers")}, indent=2))
for row in audited:
    if not row["preview_links"]:
        print(row["source"], {k:len(v) for k,v in row["remaining"].items() if v})
