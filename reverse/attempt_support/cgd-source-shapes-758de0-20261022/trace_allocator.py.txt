"""Read-only RTM c2 allocation snapshot; run from a build/ scratch copy."""
import json
import sys
from pathlib import Path

from wdbg import Dbg

work = Path(__file__).resolve().parent
source = Path(sys.argv[1]) if len(sys.argv) > 1 else work / 'base.cpp'
debugger = Dbg(source, work / 'allocator.obj', work / 'allocator.log')
records = []
found = False
try:
    debugger.prompt()
    debugger.prompt('set $CanDeferOnBPByAddr = 1')
    debugger.prompt('break *0x10783acd')
    for _ in range(100):
        output = debugger.prompt('cont', 60)
        if 'terminated' in output or not output.strip():
            break
        if debugger.reg('ebx') != 0:
            continue
        function = debugger.words(debugger.reg('ebp'), 1)[0]
        name = debugger.string(debugger.words(function + 4, 1)[0])
        if '?bfmeTwoCGD@' not in name:
            continue
        found = True
        buckets = debugger.words(0x108807b8, 1024)
        seen = set()
        for address in buckets:
            while address and address not in seen:
                seen.add(address)
                words = debugger.words(address, 18)
                records.append({'ptr': address, 'words': words,
                                'sym': debugger.words(words[0], 24)})
                address = words[12]
        break
finally:
    if debugger.p.poll() is None:
        try:
            debugger.prompt('quit')
        except Exception:
            debugger.p.kill()
    debugger.log.close()
if not found:
    raise SystemExit('Target allocation pass was not captured')
(work / 'allocator-reproduced.json').write_text(json.dumps(records, indent=1))
print('Captured', len(records), 'raw allocation records')
