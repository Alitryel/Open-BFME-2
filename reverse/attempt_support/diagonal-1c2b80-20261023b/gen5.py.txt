import itertools,sys,os
bank=open('bank.cpp').read()
a=bank.index('            {\n                int pixelSample0=src[(col+0)];')
b=bank.index('        }\n        dst += pitch;\n        src = src + (pitch);')
c=bank.index('    for (row = 0; row < 8; row++)\n    {\n        for (col = 0; col < 8; col++)')
d=bank.index('extern "C" void DiagonalBlur')
head,mid,tail=bank[:a],bank[b:c],bank[d:]
anchor='    unsigned char *source=cursor.source;\n'
T=['src[(col+%d)]','src[(col+%d) - pitch - 1]','src[(col+%d) + pitch - 1]','src[(col+%d) - pitch + 1]','src[(col+%d) + pitch + 1]',
   'src[(col+%d) - 2 * pitch - 2]','src[(col+%d) + 2 * pitch + 2]','src[(col+%d) + 2 * pitch - 2]','src[(col+%d) - 2 * pitch + 2]']
def pix(j,named):
    t=[x%j for x in T]
    if named:
        s='            {\n'+''.join('                int pixelSample%d=%s;\n'%(i,t[i]) for i in range(9))
        e=['pixelSample%d'%i for i in range(9)]
    else:
        s='            {\n'; e=t
    s+='                dst[(col+%d)] = (unsigned char)((16 + 8 * %s\n                + 4 * (%s + %s + %s + %s)\n                + 2 * (%s + %s + %s + %s)) >> 5);\n            }\n'%((j,)+tuple(e))
    return s
C=['if (value < 0)\n                value = 0;\n            else if (value > 255)\n                value = 255;\n',
   'if (value < 0)\n                value = 0;\n            if (value > 255)\n                value = 255;\n']
NAT=bank[c:d]
def unroll(ci,vs,ix):
    def blk(j):
        cc=('(col+%d)'%j) if ix==0 else ('col + %d'%j if j else 'col')
        f='(1 + 6 * src[%s] - src[%s - 1] - src[%s - pitch] - src[%s + pitch] - src[%s + 1]) >> 1'%(cc,cc,cc,cc,cc)
        s='            {\n            int value = %s;\n'%f if vs==0 else '            value = %s;\n'%f
        s+='            '+C[ci]+'            dst[%s] = (unsigned char)value;\n'%cc
        if vs==0: s+='            }\n'
        return s
    p2='    for (row = 0; row < 8; row++)\n    {\n'
    if vs: p2+='        int value;\n'
    p2+='        for (col = 0; col < 8; col += 4)\n        {\n'+''.join(blk(j) for j in range(4))
    p2+='        }\n        dst += pitch;\n        src += pitch;\n    }\n}\n'
    return p2
P2={'nat':NAT,'u110':unroll(1,1,0),'u010':unroll(0,1,0),'u111':unroll(1,1,1),'u100':unroll(1,0,0)}
out=sys.argv[1]; os.makedirs(out,exist_ok=True)
ks=[int(x) for x in sys.argv[2].split(',')]
for pm in range(16):
    p1=''.join(pix(j,(pm>>j)&1) for j in range(4))
    for pn,p2 in P2.items():
        for k in ks:
            pad=''.join('    int pad%d = %d;\n'%(i,i) for i in range(k))
            s=(head+p1+mid+p2+tail).replace(anchor,pad+anchor)
            open('%s/p%02d_%s_k%03d.cpp'%(out,pm,pn,k),'w').write(s)
