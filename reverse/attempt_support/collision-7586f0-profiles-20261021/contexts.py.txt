from pathlib import Path
import sys,json
root=Path.cwd();sys.path.insert(0,str(root/'tools'))
from variant_search import Scorer,diff_text
from locate import object_functions
out=root/'build/collision-finish';src=(out/'banked.cpp').read_text()
sym='?rva007586F0@CollisionManagerImpl@@QAEXXZ'
variants=[]
# Bind the input run pair through a small aggregate, preserving four contiguous 12-byte records.
for decl in ['struct Pair { SortList first, second; };', 'struct Pair { SortList items[2]; };']:
 for ptr in ['(Pair *)&lists[src]', '(Pair *)(lists + src)']:
  A='&input->first' if 'first' in decl else '&input->items[0]'
  B='&input->second' if 'first' in decl else '&input->items[1]'
  code=src.replace('SortList lists[4];',decl+'\nSortList lists[4];')
  code=code.replace('SortList *A = &lists[src];\n\t\t\tSortList *B = A + 1;', 'Pair *input = '+ptr+'; SortList *A = '+A+'; SortList *B = '+B+';')
  variants.append(('aggregate'+str(len(variants)),code))
# Integrate against actual current owner definitions rather than the bank's older owner.
home=(root/'Code/Libraries/Source/collisionmanager/collisionmanager_impl.cpp').read_text()
extra=src[src.index('struct SortList'):]
variants.append(('current_home',home+'\n'+extra))
# Output queue helper transfers the source pair together through a reference, no opaque call.
for B in ['A + 1','&lists[src + 1]']:
 for ref in ['SortList *a','SortList &a']:
  ret='a+1' if '*' in ref else '&a+1'
  arg='A' if '*' in ref else '*A'
  code=src.replace('typedef Rva009A3AD0Endpoint Endpoint;', 'inline SortList *secondList('+ref+') { return '+ret+'; }\ntypedef Rva009A3AD0Endpoint Endpoint;')
  code=code.replace('SortList *B = A + 1;', 'SortList *B = secondList('+arg+');')
  variants.append(('refhelper'+str(len(variants)),code))
scorer=Scorer(sym,0x7586f0,460,out/'objects3')
results=[]
for name,s in variants:
 r=scorer(s);a,b=r.pop('compiled',b''),r.pop('target',b'')
 extent=None
 obj=scorer.dir/('c'+str(scorer.n)+'.obj')
 if obj.exists():
  extent=next((len(v) for n,v,rel in object_functions(obj) if n==sym),None)
 r.update(name=name,extent=extent,different_bytes=sum(x!=y for x,y in zip(a,b))+abs(len(a)-len(b)))
 (out/(name+'.cpp')).write_text(s)
 if a:(out/(name+'.diff.txt')).write_text(diff_text(a,b,200))
 results.append(r);(out/'results3.json').write_text(json.dumps(results,indent=2));print(json.dumps(r),flush=True)
