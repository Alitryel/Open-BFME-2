"""Test the documented MSVC local-number tie breaker on the AIWall helper."""
from concurrent.futures import ThreadPoolExecutor
import json
from pathlib import Path
import sys

ROOT = Path(__file__).resolve().parents[2]
sys.path.insert(0, str(ROOT / 'tools'))
import permute

BASE = (ROOT / 'reverse/attempts/0x004eadba.cpp').read_text()
OUT = ROOT / 'build/aiwall-probe'
SYM = '?distSqToSegment@@YAMPBVCoord2D@@00@Z'
START = '{\n\tfloat dx = lineEnd->x;'
ENTRY = '\tfloat apx = pt->x - lineStart->x;\n\tfloat apy = pt->y - lineStart->y;'


def variant(mode, count):
    source = BASE
    if mode & 1:
        source = source.replace(ENTRY, '\tCoord2D ap;\n\tap.x = pt->x - lineStart->x;\n\tap.y = pt->y - lineStart->y;')
        source = source.replace('apy * dy', 'ap.y * dy').replace('apx * dx', 'ap.x * dx')
    if mode & 2:
        source = source.replace('Lib/Coord2D.h', 'Lib/Coord2DCtorView.h')
    padding = ''.join(f'\tint unusedTie{i} = 0;\n' for i in range(count))
    assert START in source
    return source.replace(START, '{\n' + padding + '\tfloat dx = lineEnd->x;')


def run(shard):
    directory = ROOT / f'build/aiwall-local-count-{shard}'
    directory.mkdir(exist_ok=True)
    scorer = permute.Scorer('0x004EACF5', SYM, 197, directory)
    records = []
    best = 1000
    for mode in range(4):
        for count in range(81):
            if (mode * 81 + count) % 2 != shard:
                continue
            source = variant(mode, count)
            fit, exact = scorer.score(source)
            r = dict(mode=mode, count=count, fitness=fit, exact=exact)
            if fit >= 0 and scorer.last:
                a, b = scorer.last
                x, y = permute._instructions(a), permute._instructions(b)
                nd = sum(i != j for i, j in zip(x, y)) + abs(len(x)-len(y))
                r.update(size=len(a), differences=nd)
                if nd < best or exact:
                    best = nd
                    print(json.dumps(r), flush=True)
                    (OUT / f'local-count-best-{shard}.cpp').write_text(source)
                    (OUT / f'local-count-best-{shard}.bin').write_bytes(a)
                if exact:
                    (OUT / 'win.cpp').write_text(source)
                    records.append(r)
                    break
            records.append(r)
        if records[-1]['exact']:
            break
    (OUT / f'local-count-results-{shard}.json').write_text(json.dumps(records, indent=2))
    print(f'shard {shard} done: {len(records)} candidates, best {best}', flush=True)


if __name__ == '__main__':
    with ThreadPoolExecutor(2) as pool:
        list(pool.map(run, range(2)))
