from search import *
import itertools

split=BASE.index('void CollisionManagerImpl::rva007586F0()')
pre,body=BASE[:split],BASE[split:]
endpoint='Rva009A3AD0Endpoint'
extra='''
    Real value() const { return m_value; }
    const Real &valueRef() const { return m_value; }
    Real nestedValue() const { return value(); }
    Rva009A3AD0Endpoint *next() const { return m_next; }
    void next(Rva009A3AD0Endpoint *p) { m_next=p; }
    void previous(Rva009A3AD0Endpoint *p) { m_prev=p; }
    bool after(const Rva009A3AD0Endpoint *p) const { return value()>p->value(); }
'''
pre=pre.replace('bool isMin() const { return !m_isMax; }',extra+'\n bool isMin() const { return !m_isMax; }')
extra2='''
inline Real readValue(const Endpoint *p) { return p->m_value; }
inline bool afterValue(Real a, Real b) { return a>b; }
inline bool afterPoint(const Endpoint *a,const Endpoint *b) { return a->m_value>b->m_value; }
'''
pre+=extra2
items=[]
for n,(indexed,getter,local,compare,epops,parens) in enumerate(itertools.product(range(2),range(5),range(4),range(4),range(4),range(2))):
    s=body
    if indexed:s=s.replace('SortList *B = A + 1;','SortList *B = &lists[src+1];')
    a=['A->head->m_value','A->head->value()','A->head->valueRef()','A->head->nestedValue()','readValue(A->head)'][getter]
    b=['B->head->m_value','B->head->value()','B->head->valueRef()','B->head->nestedValue()','readValue(B->head)'][getter]
    decl=''
    prefix=pre
    if local==1:
        decl='const Endpoint *left=A->head; const Endpoint *right=B->head;'
        a=a.replace('A->head','left');b=b.replace('B->head','right')
    if local in [2,3]:
        typ='const Real &' if local==3 and getter!=4 else 'Real '
        # A reference cannot bind a getter rvalue except to const, which is intentional.
        decl=typ+'left='+a+'; '+typ+'right='+b+';'
        a,b='left','right'
    test=[a+' > '+b,'afterValue('+a+', '+b+')','A->head->after(B->head)','afterPoint(A->head, B->head)'][compare]
    if parens:test='(('+test+'))'
    original='if (b != 0 && B->count != 0 && A->head->m_value > B->head->m_value)'
    if local:
        # Keep pointer/value loads in the same guarded region as in retail.
        repl='if (b != 0 && B->count != 0 && [&](){'+decl+' return '+test+';}())'
        # VC7.1 has no lambdas: a local inline class with static helper spells that region.
        helper='struct CompareLocal { static bool apply(const Endpoint *a,const Endpoint *b) { '
        d=decl.replace('A->head','a').replace('B->head','b')
        t=test.replace('A->head','a').replace('B->head','b')
        helper+=d+' return '+t+'; } };'
        prefix+=helper+'\n'
        repl='if (b != 0 && B->count != 0 && CompareLocal::apply(A->head,B->head))'
    else:repl='if (b != 0 && B->count != 0 && '+test+')'
    s=s.replace(original,repl)
    if epops&1:
        s=s.replace('cur->m_next','cur->next()').replace('p->m_next','p->next()')
        s=s.replace('cur->next() = lists[i].head;','cur->next(lists[i].head);')
        s=s.replace('lists[src].tail->m_next = 0;','lists[src].tail->next(0);')
    if epops&2:
        s=s.replace('e->m_next','e->next()')
        s=s.replace('o->tail->m_next = e;','o->tail->next(e);')
        s=s.replace('p->next()->m_prev = p;','p->next()->previous(p);')
        s=s.replace('p->m_next->m_prev = p;','p->m_next->previous(p);')
    items.append(('access-%04d'%n,prefix+s))
batch(items,'accessors')
