import csv, concurrent.futures, difflib, hashlib, json, pathlib, re, subprocess, sys, time
ROOT=pathlib.Path.cwd(); sys.path.insert(0,str(ROOT/'tools'))
import build, permute
NAME='?getCommandList@NetPacket@@QAEPAVNetCommandList@@XZ'; RVA='0x00593F15'; SIZE=1475
WORK=ROOT/'build/netpacket-20261022c'; BANK=(ROOT/'reverse/attempts/0x00593f15.cpp').read_text(); SM=build.load_symbol_map()

def run(item):
 label,code=item; src=WORK/(label+'.cpp'); obj=src.with_suffix('.obj'); src.write_text(code)
 command,env=build.compiler_command(src,obj)
 p=subprocess.run(command,cwd=ROOT,env=env,capture_output=True,text=True)
 if p.returncode:
  src.with_suffix('.log').write_text(p.stdout+p.stderr); return {'label':label,'error':(p.stdout+p.stderr)[-800:]}
 row=dict(name=NAME,target_rva=RVA,target_size=str(SIZE),source=str(src.relative_to(ROOT)),notes='')
 try:
  got=build.compile_function(row,SM,obj)
  raw,rel=build.read_object_symbol_bytes(obj,NAME,code_only=True)
  a,b=got['bytes'],got['target']; src.with_suffix('.bin').write_bytes(a)
  diffs=[i for i,(x,y) in enumerate(zip(a,b)) if x!=y]
  return dict(label=label,size=len(raw),diff=len(diffs)+abs(len(a)-len(b)),first=diffs[0] if diffs else None,fit=permute.fitness(a,b),unresolved=got['unresolved'],tail=a[0x500:0x517].hex(),sha=hashlib.sha256(a).hexdigest())
 except (Exception,SystemExit) as e:return dict(label=label,error=str(e))

def batch(items,tag):
 start=time.monotonic(); out=[]
 with concurrent.futures.ThreadPoolExecutor(max_workers=8) as ex:
  for r in ex.map(run,items):
   out.append(r)
   if len(out)%64==0:print(f'{tag}: {len(out)} tested, {time.monotonic()-start:.0f}s',flush=True)
 (WORK/(tag+'.json')).write_text(json.dumps(out,indent=2))
 good=[r for r in out if 'error' not in r]; bad=[r for r in out if 'error' in r]
 print(json.dumps({'tag':tag,'count':len(out),'errors':len(bad),'unique':len(set(r['sha'] for r in good)),'best':sorted(good,key=lambda r:(r['diff'],-r['fit']))[:8],'first_error':bad[:1]},indent=2),flush=True)
 return out
