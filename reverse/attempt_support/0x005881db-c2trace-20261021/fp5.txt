import sys,os
sys.path.insert(0,'/work/Open-BFME-2/build/c2tr')
from wdbg import Dbg
# fp5.py SRC OBJ : at round KD entry list every record: name idx class(sym+0x10>>12) table value, w1, allowed
src,obj=sys.argv[1:3]; KD=int(os.environ.get('KD','2'))
d=Dbg(src,obj,obj+'.log')
d.prompt();d.prompt('set $CanDeferOnBPByAddr = 1')
B=0x10700000
bps=[0x83acd,0x82978]; added=set()
for b in bps:
    o=d.prompt('break *0x%x'%(B+b))
    if 'Breakpoint' in o and 'Unable' not in o: added.add(b)
names={}
def nm(sym):
    if sym in names: return names[sym]
    sw=d.words(sym,1); pw=d.words(sw[0],2) if sw and sw[0]>0x100000 else []
    n=d.string(pw[1]).strip() if len(pw)>1 and pw[1]>0x100000 else '?'
    n=n.strip('"').lstrip('_'); names[sym]=n; return n
def members(hdr):
    out=[]; ch=d.words(hdr,1)[0] if hdr else 0
    while ch:
        ww=d.words(ch,3)
        if len(ww)<3: break
        base,nx,bits=ww; out+=[base+j for j in range(32) if bits>>j&1]; ch=nx
    return out
on=False;k=0
while True:
    out=d.prompt('cont')
    if 'terminated' in out: break
    eip=d.reg('eip')-B
    for b in bps:
        if b not in added and b!=eip: d.prompt('break *0x%x'%(B+b)); added.add(b)
    added.add(eip)
    if eip==0x83acd:
        ebp=d.reg('ebp'); f=d.words(ebp,1)[0]; curfn=d.string(d.words(f+4,1)[0]); on='updateMeleeAttack' in curfn and d.reg('ebx')==0; k=0; continue
    if not on: continue
    if eip==0x82978:
        k+=1
        if k==KD:
            tab=d.words(0x108807b8,0x400)
            for h,p in enumerate(tab):
                while p:
                    w=d.words(p,13)
                    sw=d.words(w[0],5); cls=(sw[4]&0xffff)>>12 if len(sw)>4 else -1
                    tv=d.words(0x10869140+cls*4,1)[0] if cls>=0 else -1
                    print('%s:%x cls=%d tv=%x w1=%x w9=%d allowed=%s'%(nm(w[0]),w[7],cls,tv,w[1],w[9],members(w[8]) if w[8] else None),flush=True)
                    p=w[12]
            break
print('done',flush=True)
