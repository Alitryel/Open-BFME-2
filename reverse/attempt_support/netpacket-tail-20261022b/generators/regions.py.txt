import sys,collections,json,capstone,difflib
sys.path.insert(0,'build/netpacket-20261022b');from search import *
md=capstone.Cs(capstone.CS_ARCH_X86,capstone.CS_MODE_32)
patterns=collections.defaultdict(list)
for name in sys.argv[1:] or ['visible-memcpy','inline-boundary','loop-carrier','aggregate-state']:
 for r in json.loads((WORK/(name+'.json')).read_text()):
  if 'error' in r:continue
  a=(WORK/(r['label']+'.bin')).read_bytes();ins=list(md.disasm(a,0x593f15))
  calls=[i for i,x in enumerate(ins) if x.mnemonic=='call' and x.op_str=='0x4d55bc']
  if not calls:continue
  k=calls[-1]
  tail='; '.join(x.mnemonic+' '+x.op_str for x in ins[k+1:k+4])
  patterns[(len(calls),tail)].append((r['fit'],r['label'],r['diff']))
for tail,entries in sorted(patterns.items(),key=lambda kv:-max(e[0] for e in kv[1])):
 print(len(entries),max(entries),tail)
