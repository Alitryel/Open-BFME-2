import sys,re,collections
pat=re.compile(r'\[ebp ([-+]) (0x[0-9a-f]+|\d+)\]')
def offs(f):
    c=collections.Counter()
    for l in open(f):
        for m in pat.finditer(l):
            o=int(m.group(2),0)*(-1 if m.group(1)=='-' else 1); c[o]+=1
    return c
a=offs(sys.argv[1]); b=offs(sys.argv[2])
for k in sorted(set(a)|set(b)): print(f'{k:+5x} {a.get(k,0):3d} {b.get(k,0):3d}')
