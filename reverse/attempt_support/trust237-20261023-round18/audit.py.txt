def add_group(label,raw,rel,helpers,bank=False):
    key=sha(raw+json.dumps([rel,helpers,bank],sort_keys=True).encode())
    if key not in groups:groups[key]=dict(raw=raw,rel=rel,helpers=helpers,bank=bank,sources=[])
    groups[key]['sources'].append(label)

    assert all(r['exit']==0 for r in rows),family
    for row in rows:
        src=OUT/(row['name']+'.cpp'); obj=src.with_suffix('.obj')
        assert sha(src.read_bytes())==row['source_sha256']
        raw,rel=build.read_object_symbol_bytes(obj,NAME,code_only=True)
        assert sha(raw)==row['raw_sha256'] and len(raw)==row['size']
        assert json.loads(json.dumps(rel))==row['relocations']
        helpers=row.get('same_unit_helpers',{})
        for name,h in helpers.items():
            hb,hr=build.read_object_symbol_bytes(obj,name,code_only=True)
            assert hb.hex()==h['raw'] and json.loads(json.dumps(hr))==h['relocations']
        resolved=bytearray(raw)
        for off,kind,name in rel:
            assert raw[off:off+4]==b'\0'*4

            else:
                assert kind==20

        differences=[i for i,(a,b) in enumerate(itertools.zip_longest(resolved,target)) if a!=b]

        add_group(row['name'],raw,rel,helpers)

START=0x1000000; STOP=0x1001000; STUB=0x1100000; IAT=0x1200000
ARGV=0x1201000; PATH=0x1201100; STACK=0x3008000
symbols={'_memset':('memset',0),'_WinVerifyTrust@12':('WinVerifyTrust',12),
 '__imp__GetCommandLineW@0':('GetCommandLineW',0),
 '__imp__CommandLineToArgvW@8':('CommandLineToArgvW',8),'__imp__LocalFree@4':('LocalFree',4)}
locations={name:(STUB+n*0x100,IAT+n*4) for n,name in enumerate(symbols)}

 0x8009600f,0x80096011,0x800b00ff,0x800b0101,0x800b0111,0x800b0004]

for key,group in groups.items():
    uc=Uc(UC_ARCH_X86,UC_MODE_32)
    for addr,size in [(START,0x2000),(STUB,0x1000),(IAT,0x3000),(0x3000000,0x10000)]:uc.mem_map(addr,size)
    def relocate(blob,relocations,address):
        data=bytearray(blob)
        for off,kind,name in relocations:
            addend=struct.unpack_from('<I',data,off)[0]
            if name in locations:
                function,iat=locations[name]
                value=iat+addend if kind==6 else function-(address+off+4)+addend
            else:
                assert kind==20 and name in group['helpers']
                function=START+group['helpers'][name]['rva']-RVA
                value=function-(address+off+4)+addend
            assert kind in (6,20)
            struct.pack_into('<I',data,off,value&0xffffffff)
        return bytes(data)
    uc.mem_write(START,relocate(group['raw'],group['rel'],START))
    for name,h in group['helpers'].items():
        address=START+h['rva']-RVA
        uc.mem_write(address,relocate(bytes.fromhex(h['raw']),h['relocations'],address))
    uc.mem_write(ARGV,struct.pack('<II',PATH,0));uc.mem_write(PATH,'game.dat\0'.encode('utf-16le'))
    by_addr={}
    for name,(address,iat) in locations.items():
        uc.mem_write(iat,struct.pack('<I',address));uc.mem_write(address,b'\xc3');by_addr[address]=symbols[name]
    state={}
    def read32(addr):return struct.unpack('<I',uc.mem_read(addr,4))[0]
    def hook(uc,address,size,_):
        if address not in by_addr:return
        name,n=by_addr[address];sp=uc.reg_read(UC_X86_REG_ESP)
        args=[read32(sp+4+4*j) for j in range(3)];state['calls'].append(name)
        if name=='memset':
            dst,value,count=args;uc.mem_write(dst,bytes([value&255])*count);value=dst
        elif name=='GetCommandLineW':value=PATH
        elif name=='CommandLineToArgvW':
            assert args[0]==PATH;uc.mem_write(args[1],struct.pack('<I',state['argc']));value=ARGV
        elif name=='LocalFree':assert args[0]==ARGV;value=state['free']
        else:
            assert name=='WinVerifyTrust' and args[0]==0
            assert bytes(uc.mem_read(args[1],16)).hex()=='6bc5aa0044cdd0118cc200c04fc295ee'
            data=struct.unpack('<11I',uc.mem_read(args[2],44))
            assert data[:6]==(44,0,0,2,0,1) and data[7:]==(0,0,0,0x210)
            assert struct.unpack('<4I',uc.mem_read(data[6],16))==(16,PATH,0,0)
            value=state['status']
        uc.reg_write(UC_X86_REG_ECX,0xabababab);uc.reg_write(UC_X86_REG_EDX,0xcdcdcdcd)
        uc.reg_write(UC_X86_REG_EFLAGS,0x246);uc.reg_write(UC_X86_REG_EAX,value)
        uc.reg_write(UC_X86_REG_ESP,sp+4+n);uc.reg_write(UC_X86_REG_EIP,read32(sp))
    uc.hook_add(UC_HOOK_CODE,hook,begin=STUB,end=STUB+0xfff)
    for status,j in itertools.product(samples,range(4)):
        state.update(status=status,free=[0,ARGV,0x80000000,0xffffffff][j],argc=[0,1,2,0xffffffff][j],calls=[])
        uc.mem_write(STACK,struct.pack('<I',STOP));uc.reg_write(UC_X86_REG_ESP,STACK)
        saved=[(UC_X86_REG_EBX,0x11111111),(UC_X86_REG_ESI,0x22222222),(UC_X86_REG_EDI,0x33333333),(UC_X86_REG_EBP,0x44444444)]
