from search import *
import itertools
items=[]
for n,choices in enumerate(itertools.product(range(2),range(3),range(3),range(3),range(3),range(2),range(2),range(2),range(2))):
 independent,bc1,bc2,bh,fb,ah,fa,ac0,ac1=choices
 s=BASE
 if independent:s=s.replace('SortList *B = A + 1;','SortList *B = &lists[src+1];')

 s=s.replace('B->count != 0', ['B->count != 0','A[1].count != 0','lists[src+1].count != 0'][bc1],1)
 # Replace the second original occurrence without accidentally matching the first replacement.
 a=s.index('else if (b != 0')
 s=s[:a]+s[a:].replace('B->count != 0',['B->count != 0','A[1].count != 0','lists[src+1].count != 0'][bc2],1)
 s=s.replace('B->head->m_value',['B->head->m_value','A[1].head->m_value','lists[src+1].head->m_value'][bh])

 if fa:s=s.replace('from = A;','from = &lists[src];')
 if ac0:s=s.replace('while (A->count != 0)','while (lists[src].count != 0)')
 if ac1:s=s.replace('a != 0 && A->count != 0','a != 0 && lists[src].count != 0')
 items.append(('paths-%04d'%n,s))
batch(items,'accesspaths')
