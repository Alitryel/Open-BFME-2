from driver import *
def gen():
 yield 'ref-control',BASE
 taps=['',' - pitch - 1',' + pitch - 1',' - pitch + 1',' + pitch + 1',' - 2 * pitch - 2',' + 2 * pitch + 2',' + 2 * pitch - 2',' - 2 * pitch + 2']
 for group,kind,mask in itertools.product(range(3),range(2),range(1,512)):
  b=BASE
  for j in ([0,2] if group==0 else ([1,3] if group==1 else range(4))):
   prefix=''
   for k,tail in enumerate(taps):
    if mask&(1<<k):
     old='src[(col+%d)%s]'%(j,tail)
     index='tapIndex%d_%d'%(j,k)
     typ=['int','unsigned int'][kind]
     prefix+='const %s &%s=(col+%d)%s;\n'%(typ,index,j,tail)
     assert old in b,old
     b=b.replace(old,'src[%s]'%index)
   mark='int pixelSample0=' if j<2 else 'dst[(col+%d)] ='%j
   if j<2:
    start=b.index('src[(col+%d)'%j) if 'src[(col+%d)'%j in b else -1
    # Identify each pixel by the first named declaration's block, including when tap zero is replaced.
    nth=[m.start() for m in re.finditer(re.escape(mark),b)][j]
   else:nth=b.index(mark)
   b=b[:nth]+prefix+b[nth:]
  yield 'ref-%d-%d-%03d'%(group,kind,mask),b
if __name__=='__main__':run(list(gen()),'offsetrefs')
