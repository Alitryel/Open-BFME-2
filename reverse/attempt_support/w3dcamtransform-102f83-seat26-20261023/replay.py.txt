"""Recreate the two decisive ordinary compiles without editing production."""
import sys,subprocess,shutil,json
from pathlib import Path
ROOT=Path(subprocess.check_output(['git','rev-parse','--show-toplevel'],text=True).strip())
SUPPORT=Path(__file__).resolve().parent
WORK=ROOT/'build/camera-102f83-seat26-replay'
WORK.mkdir(exist_ok=True)
for name in ['search.py','schedule.py','wdbg.py','probe_shapes.py','probe.cpp']:
    shutil.copyfile(SUPPORT/(name+'.txt'),WORK/name)
sys.path.insert(0,str(WORK))
import search as h
for label,patch in [('pre_final','pre-final-source.patch'),('winner','winning-source.patch')]:
    source=WORK/'camera.cpp'
    source.write_text(h.BASE)
    subprocess.run(['git','apply','--unidiff-zero','-p0','--directory='+str(WORK.relative_to(ROOT)),
                    str(SUPPORT/patch)],cwd=ROOT,check=True)
    result=h.run_one((label,source.read_text()))
    print(json.dumps(result))
    assert result['size']==4299 and not result['unresolved']
    assert result['nd']==(0 if label=='winner' else 17)
