from pathlib import Path
import sys,itertools,concurrent.futures,json,re
sys.path.insert(0,'build/df20-scalars-20261020')
import trials
trials.scratch=Path.cwd()/'build/df20-scalars-20261020'
base=Path('build/df20-scalars-20261020/bank.cpp').read_text()
trials.symbol=base.splitlines()[0][3:]
choices=['const_chars','const_scalars','hoist_chars','unsigned_loops','int_negative','hoist_negative','split_postfix','named_return']
(trials.scratch/'choices.json').write_text(json.dumps(choices))
def source(bits):
 s=base
 if bits[0]: s=re.sub(r'  char_type (__(?:minus|plus|space|zero|point|sep))',r'  const char_type \1',s)
 if bits[1]:
  for ty,name in [('int','frac_digits'),('size_t','width'),('size_t','fill_amt'),('ios_base::fmtflags','fill_pos'),('bool','generate_curr')]:
   s=s.replace('  '+ty+' __'+name,'  const '+ty+' __'+name)
 if bits[2]:
  decl=[]
  for name in ['minus','plus','space','zero','point','sep']:
   pat=r'  (?:const )?char_type __'+name+r'\s*='
   m=re.search(pat,s); assert m,name
   # Hoisted consts must remain initialized; use nonconst for this crossed case.
   s=s[:m.start()]+'  __'+name+' ='+s[m.end():]
   decl.append('  char_type __'+name+';')
  s=s.replace('  locale __loc =', '\n'.join(decl)+'\n  locale __loc =')
 if bits[3]: s=s.replace('for (int __i = 0;', 'for (unsigned int __i = 0;')
 if bits[4]: s=s.replace('  bool __is_negative;', '  int __is_negative;')
 if bits[5]:
  m=re.search(r'  (?:bool|int) __is_negative;\n',s); assert m
  decl=m[0];s=s[:m.start()]+s[m.end():]
  s=s.replace('  locale __loc =', decl+'  locale __loc =')
 if bits[6]:
  s=s.replace('*__s++ = __space;', '*__s = __space; ++__s;')
  s=s.replace('if (__sign.size() != 0)\n        *__s++ = __sign[0];', 'if (__sign.size() != 0) {\n        *__s = __sign[0]; ++__s; }')
 if bits[7]: s=s.replace('return __s;', '{ const _MoneyOut __returned = __s; return __returned; }')
 return s
specs=[('scalar-'+''.join(map(str,b)),source(b)) for b in itertools.product(range(2),repeat=8)]
with concurrent.futures.ThreadPoolExecutor(max_workers=8) as pool:
 out=list(pool.map(trials.run,specs))
(trials.scratch/'trials.json').write_text(json.dumps(out,indent=2))
