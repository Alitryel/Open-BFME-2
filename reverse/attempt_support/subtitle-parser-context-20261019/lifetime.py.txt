import json

import subprocess
import sys
from pathlib import Path

root = Path(__file__).resolve().parents[2]
scratch = root / 'build/resume-eh-local'
base = (scratch / 'turn47-subtitle-separate-and-callback.cpp').read_text()
helper = '''
#include <new>
// Explicit lifetime in the dead local parameter storage; no caller memory
// is overwritten. The original pointer has already been copied and tested.
class ScopedSubtitleText {
public:
 explicit ScopedSubtitleText(void *&slot):text(*new(&slot) UnicodeString()) {}
 ~ScopedSubtitleText() { text.~UnicodeString(); }
 UnicodeString &text;
};
'''
source = base.replace('void parseSubtitleText( INI *ini', helper + '\nvoid parseSubtitleText( INI *ini')
source = source.replace('\tUnicodeString text;',
    '\tScopedSubtitleText textLifetime(store);\n\tUnicodeString &text = textLifetime.text;')
path = scratch / 'turn47-subtitle-parameter-lifetime.cpp'
path.write_text(source)
proc = subprocess.run([sys.executable, str(root / 'tools/explain_mismatch.py'),
    '?parseSubtitleText@@YAXPAVINI@@PAX1PBX@Z', '--source', path.relative_to(root).as_posix(),
    '--rva', '0x00688B10', '--size', '1578'], cwd=root, capture_output=True, text=True)
log = scratch / 'turn47-subtitle-parameter-lifetime.txt'
log.write_text(proc.stdout + proc.stderr)
print('\n'.join((proc.stdout + proc.stderr).splitlines()[:20]))
print('differing_instruction_rows', sum(s.startswith('!=') for s in proc.stdout.splitlines()))
