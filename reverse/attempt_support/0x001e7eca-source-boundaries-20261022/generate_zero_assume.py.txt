from search import *
import itertools
s=(W/'base.cpp').read_text();vs=[]
forms=[('union N {float f; Coord3D *p;} n;n.f=0.0f;','n.p'),
       ('float f=0.0f;','*(Coord3D**)&f'),
       ('const float f=0.0f;','*(Coord3D*const*)&f'),
       ('union N {double f; Coord3D *p;} n;n.f=0.0;','n.p')]
for form,assume,place,site,kind in itertools.product(range(4),range(4),range(2),range(3),range(2)):
 dec,expr=forms[form]
 test=['__assume(('+expr+').p==0);','__assume('+expr+'==0);','if('+expr+')__assume(0);','__assume(!'+expr+');'][assume]
 # First form uses a named pointer; remaining forms apply directly.
 if assume==0:
  dec+='Coord3D *const z='+expr+';';expr='z';test='__assume(z==0);'
 init=dec+test
 if kind:
  pre='__forceinline Coord3D *assumedZero(){'+init+'return '+expr+';}\n';init='';expr='assumedZero()'
 else:pre=''
 a=s.index('  case 1:');b=s.index('  case 4:',a);c=s.index('  case 6:');d=s.index('  case 7:',c)
 sea=s[a:b];ground=s[c:d]
 if site in [0,1]:
  if place==0:sea=sea.replace('   float water;','   float water;'+init)
  else:sea=sea.replace('   else pos.z=', '   else {'+init+'pos.z=',1).replace('object->rva0028B511(),0,true);','object->rva0028B511(),0,true);}',1)
  sea=sea.replace('object->rva0028B511(),0,true)','object->rva0028B511(),'+expr+',true)')
 if site in [0,2]:
  if place==0:ground=ground.replace('   pos.z=',init+'   pos.z=',1)
  else:ground=ground.replace('   pos.z=','   {'+init+'pos.z=',1).replace('->ground(pos.x,pos.y,0);','->ground(pos.x,pos.y,0);}',1)
  ground=ground.replace('->ground(pos.x,pos.y,0)','->ground(pos.x,pos.y,'+expr+')')
 o=s[:c]+ground+s[d:];o=o[:a]+sea+o[b:]
 o=o.replace('bool Locomotor::rva001E7ECA',pre+'bool Locomotor::rva001E7ECA')
 vs.append((f'zeroassume_{form}{assume}{place}{site}{kind}',o))
run(vs,'zeroassume192')
