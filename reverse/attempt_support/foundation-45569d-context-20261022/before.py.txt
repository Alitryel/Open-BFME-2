from search import *
from context import funcs
from ehcontext import event_context
import itertools

def before(s):
 end=s.index('\n\treturn obj;\n}',s.index('Object *FoundationAIUpdate::construct'))+len('\n\treturn obj;\n}')
 extra=s[end:]
 s=s[:end]
 pos=s.index('Object *FoundationAIUpdate::construct')
 return s[:pos]+extra+'\n'+s[pos:]

tests={}
for mask in range(128):
 s=BASE
 for i,f in enumerate(funcs):
  if mask&(1<<i):s=f(s)
 tests[f'before-context-{mask:03d}']=before(s)
for mask,opts in itertools.product(range(32),range(8)):
 s=event_context(BASE,mask)
 for i,f in enumerate([funcs[3],funcs[4],funcs[6]]):
  if opts&(1<<i):s=f(s)
 tests[f'before-eh-{mask:02d}-{opts}']=before(s)
run(tests,'before')
