from search import *
home=(ROOT/'Code/Libraries/Source/collisionmanager/collisionmanager_impl.cpp').read_text()
best=(OUT/'qual-0321.cpp').read_text()
tail=best[best.index('struct SortList'):]
helper='inline unsigned int readCount(const SortList *p) { return *(const volatile unsigned int *)&p->count; }'
items=[('home-original',home+'\n'+tail)]
for name,new in [
 ('static','static inline unsigned int readCount(const SortList *p) { return *(const volatile unsigned int *)&p->count; }'),
 ('pointee','inline unsigned int readCount(const volatile SortList *p) { return p->count; }'),
 ('static-pointee','static inline unsigned int readCount(const volatile SortList *p) { return p->count; }'),
 ('reference','inline unsigned int readCount(const SortList &p) { return *(const volatile unsigned int *)&p.count; }'),
 ('volatile-ref','inline unsigned int readCount(const volatile SortList &p) { return p.count; }'),
 ('forceinline','__forceinline unsigned int readCount(const SortList *p) { return *(const volatile unsigned int *)&p->count; }'),
 ('direct-member',''),
 ('cast-member',''),
]:
 s=tail.replace(helper,new)
 if name=='reference' or name=='volatile-ref':s=s.replace('readCount(A)','readCount(*A)')
 if name.endswith('member'):
  m='unsigned int getCount() const { return *(const volatile unsigned int *)&count; }'
  if name=='direct-member':m='unsigned int getCount() const volatile { return count; }'
  s=s.replace('unsigned int count;','unsigned int count; '+m)
  s=s.replace('readCount(A)','A->getCount()')
 items.append(('home-'+name,home+'\n'+s))
batch(items,'landing')
