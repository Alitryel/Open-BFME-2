from pathlib import Path
from itertools import product
W=Path(__file__).resolve().parent
out=W/'fieldmodels';out.mkdir(exist_ok=True)
for idx,(state,source,node,access,cc,copy,bank) in enumerate(product(range(4),range(4),range(4),range(2),range(2),range(2),range(2))):
 s=(W/('diagnostic.cpp' if bank else 'base.cpp')).read_text()
 call='__fastcall ' if cc else ''
 defs=''
 for name,typ,style in [('SP','ContactSource',source),('DP','ContactData',node)]:
  if not style:continue
  defs+='struct '+typ+'; struct '+name+' { '+typ+' *p; '
  ret=typ+' *const &' if style==2 else typ+' *'
  defs+=ret+' '+call+'operator->() const { return p; } '
  defs+='operator '+typ+' *() const { return p; } '
  if style==3:defs+='bool '+call+'operator==(int x) const { return p==0; } '
  if copy:defs+=name+'(const '+name+' &a):p(a.p) {} '
  defs+='};\n'
 if state:
  defs+='struct State { unsigned v; operator unsigned() const { return v; } '
  if state==1:defs+='State &'+call+'operator=(unsigned x) { v=x; return *this; } '
  elif state==2:defs+='void '+call+'operator=(unsigned x) { v=x; } bool '+call+'operator==(unsigned x) const { return v==x; } '
  else:defs+='unsigned &'+call+'get() { return v; } State &'+call+'operator=(unsigned x) { v=x; return *this; } '
  if copy:defs+='State(const State &a):v(a.v) {} '
  defs+='};\n'
 s=s.replace('struct ContactData',defs+'struct ContactData',1)
 if source:s=s.replace('ContactSource *source;', 'SP source;')
 if node:s=s.replace('ContactData *first,*second;', 'DP first,second;')
 if state:s=s.replace('unsigned key0,key1,state;', 'unsigned key0,key1; State state;')
 if state==3:s=s.replace('node->state','node->state.get()')
 if access:
  s=s.replace(' ContactNode *next;', ' ContactNode *next;\n const ContactVector &'+call+'getPoint() const {return point;} const ContactVector &'+call+'getNormal() const {return normal;}')
  s=s.replace('node->point','node->getPoint()').replace('node->normal','node->getNormal()')
 (out/('v%04d.cpp'%idx)).write_text(s)
print('generated',idx+1)
