from search import *
items=[]
for mask in range(2048):
 s=BASE
 # Reference aliases are the new independent choice; factorial includes the old two attractors as controls.
 mode=mask%8
 if mode:
  alias=['','const int &source = src;','int &source = src;','const int source = src;','const unsigned int &source = reinterpret_cast<const unsigned int &>(src);','const SortList &source = lists[src];','SortList &source = lists[src];','SortList *const source = &lists[src];'][mode]
  pos='while (lists[src + 1].count != 0)'
  if mask&8:
   s=s.replace('unsigned int runLen = 1;','unsigned int runLen = 1;\n'+alias)
  else:s=s.replace('SortList *A = &lists[src];',alias+'\nSortList *A = &lists[src];')
  if mode<=4:
   s=s.replace('SortList *A = &lists[src];','SortList *A = &lists[source];')
   B='&lists[source + 1]'
  elif mode in [5,6]:B='const_cast<SortList *>(&source)+1';s=s.replace('SortList *A = &lists[src];','SortList *A = const_cast<SortList *>(&source);')
  else:B='source+1';s=s.replace('SortList *A = &lists[src];','SortList *A = source;')
 else:B='&lists[src + 1]'
 if mask&16:s=s.replace('SortList *B = A + 1;','SortList *B = '+B+';')
 # Whole-array ownership changes aggregate alias and struct-piece numbering.
 wrap=(mask>>5)&3
 if wrap:
  decl=['','struct SortBuffer { SortList data[4]; }; SortBuffer buffer; SortList (&lists)[4] = buffer.data;','union SortBuffer { SortList data[4]; }; SortBuffer buffer; SortList *lists = buffer.data;','SortList storage[4]; SortList (&lists)[4] = storage;'][wrap]
  s=s.replace('SortList lists[4];',decl)
 if mask&128:s=s.replace('SortList *A =','SortList *const A =').replace('SortList *B =','SortList *const B =')
 if mask&256:s=s.replace('int out = src ^ 2;','const int &output = src;\nint out = output ^ 2;')
 if mask&512:s=s.replace('unsigned int b = runLen;','const unsigned int &run = runLen;\nunsigned int b = run;').replace('unsigned int a = runLen;','unsigned int a = run;')
 if mask&1024:s=s.replace('for (unsigned int k = 0; k < m_axisCount; k++)','unsigned int k;\nfor (k = 0; k < m_axisCount; k++)')
 # Mode3 aliases must track src at every pass. Skip invalid semantics of a cached copied index.
 if mode==3 and mask&8:continue
 if mode>=5 and mask&8:continue
 items.append(('alias-%04d'%mask,s))
batch(items,'aliases')
