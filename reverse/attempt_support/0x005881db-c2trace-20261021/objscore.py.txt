import sys,re,difflib
sys.path.insert(0,'/work2/Open-BFME-2/tools')
import build, variant_search as vs
from pathlib import Path
SYM='?updateMeleeAttack@HordeMeleeAmoeba@@UAEXPAVObject@@@Z'
obj=Path(sys.argv[1]).resolve(); src=sys.argv[2]
row={"name":SYM,"target_rva":"0x005881DB","target_size":"2199","source":src,"notes":""}
got=build.compile_function(row,build.load_symbol_map(),obj)
c,t=got['bytes'],got['target']
A=[f"{m} {o}" for m,o in vs.instructions(c)]; B=[f"{m} {o}" for m,o in vs.instructions(t)]
J=re.compile(r'^(j\w+|call) 0x[0-9a-f]+'); n=lambda x:J.sub(r'\1 T',x)
sm=difflib.SequenceMatcher(None,[n(x) for x in A],[n(x) for x in B],autojunk=False)
nd=sum(max(i2-i1,j2-j1) for tag,i1,i2,j1,j2 in sm.get_opcodes() if tag!='equal')
print(len(c),nd,'exact' if c==t and not got['unresolved'] else vs.non_exact_fitness(c,t))
if len(sys.argv)>3:
    for tag,i1,i2,j1,j2 in sm.get_opcodes():
        if tag!='equal': print(tag,A[i1:i2],B[j1:j2])
