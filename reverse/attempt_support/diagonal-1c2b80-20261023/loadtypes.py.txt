from driver import *
def gen():
 yield 'load-control',BASE
 taps=['',' - pitch - 1',' + pitch - 1',' - pitch + 1',' + pitch + 1',' - 2 * pitch - 2',' + 2 * pitch + 2',' + 2 * pitch - 2',' - 2 * pitch + 2']
 for group,kind,mask in itertools.product(range(3),range(2),range(1,512)):
  b=BASE
  for j in ([0,2] if group==0 else ([1,3] if group==1 else range(4))):
   for k,tail in enumerate(taps):
    if mask&(1<<k):
     old='src[(col+%d)%s]'%(j,tail)
     cast=['signed char','char'][kind]
     new='(unsigned char)((%s*)src)[(col+%d)%s]'%(cast,j,tail)
     assert old in b,old
     b=b.replace(old,new)
  yield 'load-%d-%d-%03d'%(group,kind,mask),b
if __name__=='__main__':run(list(gen()),'loadtypes')
