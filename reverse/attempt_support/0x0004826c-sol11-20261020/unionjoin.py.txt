from headmerge import *
# New representation hypothesis: same four bytes observed as pointer/integer
# fields can defer a null fact without introducing padding or another read.
# MSVC's union extension is intentional; every nonzero 32-bit pointer stays true.
checks=['bits.u != 0','bits.u > 0','bits.s != 0','bits.v != 0']
loads=['p','bits.p','(Drawable*)bits.u','(Drawable*)bits.v']
assigns=['p','bits.p','(Drawable*)bits.u','(Drawable*)bits.v']
helper='union DrawBits {Drawable *p; void *v; unsigned u; int s;};\n'
items=[]
for f,c,load,assign,nul,contract in itertools.product([8,9],range(4),range(4),range(4),range(2),range(2)):
 form=forms[f].replace('obj=0','obj='+( '(Object*)p' if nul else '0'))
 form=form.replace('p?p->m_object:0','p?p->m_object:'+( '(Object*)p' if nul else '0'))
 form=form.replace('p->m_object', '('+loads[load]+')->m_object')
 form=form.replace('if(p)','if('+checks[c]+')').replace('obj=p?','obj=('+checks[c]+')?')
 form=form.replace('draw=p;', 'draw='+assigns[assign]+';')
 src=BASE[:lo]+decl+'DrawBits bits; bits.p=p;\n'+form+'\n'+BASE[hi:]
 src=src.replace('void W3DDisplay::gatherDebugStats(void)',helper+'void W3DDisplay::gatherDebugStats(void)')
 if not contract:src=src.replace('template <> void StringBase<char>::releaseBuffer() throw();\n','')
 items.append(('uj%02d%d%d%d%d%d'%(f,c,load,assign,nul,contract),src))
if __name__=='__main__':batch(items,'unionjoin')
