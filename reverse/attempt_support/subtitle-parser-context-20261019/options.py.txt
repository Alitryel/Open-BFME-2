import json
import re
import subprocess
import sys
from pathlib import Path

root = Path(__file__).resolve().parents[2]
scratch = root / 'build/resume-eh-local'
base = (scratch / 'turn47-subtitle-separate-and-callback.cpp').read_text()
options = {'forscope': '/Zc:forScope', 'old-forscope': '/Zc:forScope-',
           'no-global-opt': '/Og-', 'alias-independent': '/Oa',
           'alias-limited': '/Ow', 'inline-all': '/Ob2'}
results = []
for label, option in options.items():
    lines = base.splitlines()
    index = next(i for i, line in enumerate(lines) if line.startswith('// cl:'))
    lines[index] += ' ' + option
    path = scratch / ('turn47-subtitle-' + label + '.cpp')
    path.write_text('\n'.join(lines) + '\n')
    proc = subprocess.run([sys.executable, str(root / 'tools/explain_mismatch.py'),
        '?parseSubtitleText@@YAXPAVINI@@PAX1PBX@Z', '--source', path.relative_to(root).as_posix(),
        '--rva', '0x00688B10', '--size', '1578'], cwd=root, capture_output=True, text=True)
    log = scratch / ('turn47-subtitle-' + label + '.txt')
    log.write_text(proc.stdout + proc.stderr)
    size = re.search(r'compiled symbol size: (\d+)', proc.stdout)
    result = {'variant': label, 'option': option, 'exit': proc.returncode,
        'compiled_size': int(size[1]) if size else None,
        'exact': 'OK: bytes match' in proc.stdout,
        'differing_instruction_rows': sum(s.startswith('!=') for s in proc.stdout.splitlines()),
        'log': log.relative_to(root).as_posix()}
    print(json.dumps(result), flush=True)
    results.append(result)
(scratch / 'turn47-subtitle-option-results.json').write_text(json.dumps(results, indent=2) + '\n')
