# dis.py OBJ [start end] -> disassemble queueSupplyTruck from an obj (offsets relative)
import sys; sys.path.insert(0,'/work2/Open-BFME-2/build/qAo')
from bh import fbytes
import capstone
r=fbytes(sys.argv[1]); md=capstone.Cs(capstone.CS_ARCH_X86,capstone.CS_MODE_32)
a=int(sys.argv[2],16) if len(sys.argv)>2 else 0; b=int(sys.argv[3],16) if len(sys.argv)>3 else len(r)
for i in md.disasm(r,0):
    if a<=i.address<b: print('%04x: %-8s %s'%(i.address,i.mnemonic,i.op_str))
