from pathlib import Path
import os
import subprocess
import sys

sys.path.insert(0,'tools')
import build

p=Path('build/w7-73c450')
if Path('build/STOP_WORKER').exists():
    sys.exit('STOP_WORKER exists')
native=build.read_target_bytes(0x73C450,874)
(p/'retail.inc').write_text('static const unsigned char retailCode[]={'+','.join(str(x) for x in native)+'};\n',encoding='utf-8')
template=Path('reverse/attempt_support/73bfe0-20261019/native-trace-harness.cpp.txt').read_text(encoding='utf-8')
template=template.replace('Rva0073BFE0Raster','Rva0073C450Fill').replace('rva0073BAA0','rva0073BB40')
for trial in sys.argv[1:]:
    if Path('build/STOP_WORKER').exists():
        sys.exit('STOP_WORKER exists')
    source=template.replace('Rva0073BAA0Base','Rva0073BB40') if trial=='before' else template
    if trial=='bankready':
        source=source.replace('Rva0073C450Fill','Rva0073C450Raster')
    src=build.ROOT/p/f'trace-{trial}.cpp'
    src.write_text(source,encoding='utf-8',newline='\n')
    obj=src.with_suffix('.obj')
    cmd,env=build.compiler_command(src,obj)
    env['BFME_OBJSTORE']='off'
    with (p/f'trace-{trial}-compile.log').open('w',encoding='utf-8') as log:
        log.write(repr(cmd)+'\n'); log.flush()
        r=subprocess.run(cmd,env=env,stdout=log,stderr=subprocess.STDOUT)
        log.write(f'process_exit={r.returncode}\n')
    if r.returncode:
        print(trial,'compile failure',r.returncode); continue
    exe=p/f'trace-{trial}.exe'
    cmd=[str(build.vc71_root()/'Vc7/bin/link.exe'),'-nologo','-subsystem:console',f'-out:{exe}',str(obj),str(build.obj_path(build.ROOT/p/f'{trial}.cpp')),'kernel32.lib']
    with (p/f'trace-{trial}-link.log').open('w',encoding='utf-8') as log:
        log.write(repr(cmd)+'\n'); log.flush()
        r=subprocess.run(cmd,env=env,stdout=log,stderr=subprocess.STDOUT)
        log.write(f'process_exit={r.returncode}\n')
    if r.returncode:
        print(trial,'link failure',r.returncode); continue
    with (p/f'trace-{trial}.log').open('w',encoding='utf-8') as log:
        cmd=[str(exe)]+(['stop-after-three'] if trial=='before' else [])
        log.write(repr(cmd)+'\n'); log.flush()
        try:
            r=subprocess.run(cmd,env=env,stdout=log,stderr=subprocess.STDOUT,timeout=30)
            log.write(f'process_exit={r.returncode}\n')
        except subprocess.TimeoutExpired:
            log.write('process timeout after 30 seconds; child terminated by subprocess.run\n')
    print((p/f'trace-{trial}.log').read_text(encoding='utf-8'))
