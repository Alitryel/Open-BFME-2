from pathlib import Path
import subprocess,sys,json
root=Path(__file__).resolve().parents[1];out=root/'build/string-insert29530-helpers-oct20'
caller=(root/'Code/GameEngine/Source/Common/BfmeThreeHundredSix.cpp').read_text()
body=caller[caller.index('BfmeThingQX *BfmeThingQX::bfmeInsertQX'):].replace('m_bfmeEnd','m_finish').replace('m_bfmeAt','m_start')
declarations='BfmeThingQX *bfmeInsertQX(unsigned int where, unsigned int many, unsigned char what);\n\tvoid bfmeRangeErrorQX();\n\tvoid bfmeLengthErrorQX();\n\t'
results=[]
for shape in ('landed_sibling_declaration','inline'):
    text=(out/(shape+'.cpp')).read_text()
    text=text.replace('void bfmeDoInsertQX(char *__position, unsigned int __n, char __c);',declarations+'void bfmeDoInsertQX(char *__position, unsigned int __n, char __c);')
    assert declarations in text
    marker='void BfmeThingQX::bfmeDoInsertQX(char *__position, unsigned int __n, char __c)'
    text=text.replace(marker,body+'\n'+marker)
    path=out/('caller_'+shape+'.cpp');path.write_text(text)
    p=subprocess.run([sys.executable,'tools/explain_mismatch.py','?bfmeDoInsertQX@BfmeThingQX@@QAEXPADID@Z','--rva','0x00029530','--size','662','--source',path.relative_to(root).as_posix()],cwd=root,capture_output=True,text=True,encoding='utf8',errors='replace')
    output=p.stdout+p.stderr;path.with_suffix('.log').write_text(output)
    summary=[s for s in output.splitlines() if s.startswith(('compiled symbol size:','first diff:','unresolved:','OK:','classification:'))]
    results.append({'shape':shape,'exit':p.returncode,'summary':summary});(out/'caller_results.json').write_text(json.dumps(results,indent=2)+'\n');print(shape,p.returncode,summary,flush=True)
