import sys,os
sys.path.insert(0,'/work/Open-BFME-2/build/c2tr')
from wdbg import Dbg
# keep.py SRC OBJ ROUND:NAME[,NAME...] : at round ROUND's first a8f54 call keep only NAMEs in the split set
src,obj,tgt=sys.argv[1:4]
TR,TN,OFF,VAL=tgt.split(':'); TR=int(TR); TN=TN.split(','); OFF=int(OFF,0); VAL=int(VAL,0)
d=Dbg(src,obj,obj+'.log')
d.prompt();d.prompt('set $CanDeferOnBPByAddr = 1')
B=0x10700000
bps=[0x83acd,0x82978,0xa8f54]; added=set()
for b in bps:
    o=d.prompt('break *0x%x'%(B+b))
    if 'Breakpoint' in o and 'Unable' not in o: added.add(b)
names={}
def nm(sym):
    if sym in names: return names[sym]
    sw=d.words(sym,1); pw=d.words(sw[0],2) if sw and sw[0]>0x100000 else []
    n=d.string(pw[1]).strip() if len(pw)>1 and pw[1]>0x100000 else '?'
    n=n.strip('"').lstrip('_'); names[sym]=n; return n
def rec_of(i):
    p=d.words(0x108807b8+(i&0x3ff)*4,1)[0]
    while p:
        w=d.words(p,13)
        if w[7]==i: return p,w
        p=w[12]
    return None,None
def s32(v): return v-(1<<32) if v>=1<<31 else v
on=False;k=0;calls={};nrounds=0
while True:
    out=d.prompt('cont')
    if 'terminated' in out: break
    eip=d.reg('eip')-B
    for b in bps:
        if b not in added and b!=eip: d.prompt('break *0x%x'%(B+b)); added.add(b)
    added.add(eip)
    if eip==0x83acd:
        ebp=d.reg('ebp'); f=d.words(ebp,1)[0]; curfn=d.string(d.words(f+4,1)[0]); on='updateMeleeAttack' in curfn and d.reg('ebx')==0; k=0; continue
    if not on: continue
    if eip==0x82978:
        k+=1
        if k==TR:
            for b in d.words(0x108807b8,1024):
                q=b
                while q:
                    w=d.words(q,18)
                    if nm(w[0]) in TN and (w[9]&0xffff)>=2:
                        print('set',nm(w[0]),hex(w[7]),'off',hex(OFF),'old',d.words(q+OFF,1)[0],'->',VAL,'w3',w[3],'cost',d.words(q+0x40,1)[0],flush=True)
                        d.prompt('set *(int*)0x%x = %d'%(q+OFF,VAL))
                    q=w[12]
        continue
    if eip==0xa8f54:
        esp=d.reg('esp'); st=d.words(esp,4); ra=st[0]-B
        c=calls.get(k,0); calls[k]=c+1
        ch=d.words(st[2],1)[0]; items=[]
        while ch:
            base,nx,bits=d.words(ch,3)
            for j in range(32):
                if bits>>j&1:
                    i=base+j; p,w=rec_of(i); items.append((ch,base,i,w))
            ch=nx
        print('R%d call%d n=%d :: %s'%(k,c,len(items),' '.join('%s:%x'%(nm(w[0]),i) if w else '%x'%i for _,_,i,w in items)),flush=True)
        if k>40: print('too many rounds');break
        continue
print('done',flush=True)
