import sys,itertools
from pathlib import Path
sys.path.insert(0,str(Path('build/trust237-round2').resolve()))
import search
loops=[
 'for(unsigned i=u&1u;i<2u;i+=2u) ++r;',
 'for(unsigned i=u&3u;i<4u;i+=4u) ++r;',
 'for(unsigned i=u&255u;i<256u;i+=256u) ++r;',
 'for(unsigned i=u&65535u;i<65536u;i+=65536u) ++r;',
 'for(unsigned i=u|1u;i;i=0u) ++r;',
 'do {++r;} while((u&1u)>1u);',
 'do {++r;} while((u&255u)>255u);',
 'for(unsigned i=u&1u;i<=1u;i=2u) ++r;',
 'for(unsigned i=u&255u;i<=255u;i=256u) ++r;',
 'do {++r;if((u|1u)!=0u)break;}while(!r);',
 'do {++r;if((u&1u)==0u)break;}while(!r);',
 'for(unsigned i=0;i<(bool)(u|1u);++i) ++r;',
 'for(unsigned i=0;i<(unsigned char)u/256u+1u;++i) ++r;',
 'for(unsigned i=(u&1u);i<2u;++i){++r;++i;}',
 'for(unsigned i=1u+(u&1u);i<3u;i+=2u) ++r;',
 'for(unsigned i=0u;i<=((u&1u)/2u);++i) ++r;',
]
jobs=[]
for li,opt,cpu,early,scope in itertools.product(range(len(loops)),['O1','O2'],[6,7],range(4),range(2)):
 region=loops[li]
 if scope:region='{'+region+'}'
 tail='LocalFree(argv); int r=0; '+region+' return r;'
 text=search.base.replace(search.marker,'    long status=WinVerifyTrust(0,&action,&data); unsigned long u=(unsigned long)status; '+tail)
 if early==1:text=text.replace('LocalFree(argv); int r=0;','int r=0; LocalFree(argv);')
 elif early==2:text=text.replace('    TrustFile file;', '    int r=0;\n    TrustFile file;').replace('LocalFree(argv); int r=0;','LocalFree(argv);')
 elif early==3:text=text.replace('    TrustFile file;', '    long status;\n    TrustFile file;').replace('long status=WinVerifyTrust','status=WinVerifyTrust')
 text=text.replace('/O1 /G7',f'/{opt} /G{cpu}')
 jobs.append((f'l{len(jobs):04}',text,dict(loop=li,opt=opt,cpu=cpu,early=early,scope=scope)))
search.execute('loops',jobs)
