import csv, io, os, sys

NAME = '?rva002E9680ParseModelConditionEvent@LuaScriptEngine@@QAEXPAVBfmeLexEAN@@@Z'
raw = Path('reverse/functions.csv').read_text(encoding='utf-8')
lines = raw.split('\n')

assert len(kept) == len(lines) - 1

    kept.pop()
kept.append(f'{NAME},,0x00336C5D,595,{SRC},matched,preview')

scratch.write_text('\n'.join(kept) + '\n', encoding='utf-8', newline='\n')
