# Copy the tracked isolated harness to build/<dir>/trial.py, then this file
# to the same directory as continue.py. Run from the repository root:
# python3 build/<dir>/continue.py [all|visibility|flag-interactions|inheritance|expression-scope|lookup-expression]
# Inheritance mi=1/virt=0 are eight INVALID layout controls; exclude them.
from trial import *
import itertools
FAMILY=sys.argv[1] if len(sys.argv)>1 else "all"

if FAMILY in ("all", 'visibility'):
    items=[]
    for enum,cv,visible,selectinline in itertools.product(range(2),repeat=4):
     s=BASE
     if enum:
      s=s.replace('class Drawable;','class Drawable;\nenum DrawableID { INVALID_DRAWABLE_ID=0, FORCE_DRAWABLEID_TO_LONG_SIZE=0x7ffffff };',1)
      s=s.replace('virtual int getMousedOverDrawableID();','virtual DrawableID getMousedOverDrawableID();')
      s=s.replace('findDrawableByID(int id)','findDrawableByID(const DrawableID id)')
     if cv:
      s=s.replace('getMousedOverDrawableID();','getMousedOverDrawableID() const;')
     if visible:
      ty='DrawableID' if enum else 'int'
      decl='struct Rva00239CC9DrawableHashNode { Rva00239CC9DrawableHashNode *m_next; '+ty+' m_id; Drawable *m_drawable; };\nclass Rva00239CC9DrawableHash { public: Rva00239CC9DrawableHashNode *rva002888D4(const '+ty+' &id) const; char bytes[0x14]; };\n'
      s=s.replace('struct GameClient\n{',decl+'struct GameClient\n{')
      s=s.replace('virtual Drawable *findDrawableByID('+('const DrawableID id' if enum else 'int id')+');','virtual Drawable *findDrawableByID('+('const DrawableID id' if enum else 'int id')+') { Rva00239CC9DrawableHashNode *node = ((Rva00239CC9DrawableHash *)((char *)this+0x18))->rva002888D4(id); if(!node) return 0; return node->m_drawable; }')
     if selectinline:
      s=s.replace('virtual int getSelectCount();','virtual int getSelectCount() { return *(int *)((char *)this+0x568); }')
     items.append(('vis%d%d%d%d'%(enum,cv,visible,selectinline),s))
    batch(items,'visibility')

if FAMILY in ("all", 'flag-interactions'):
    flags=['/Oa','/Ow','/Op','/Ot','/Oy-','/Og','/Ob2','/Oi']
    items=[]
    for arch,eh,bits in itertools.product(['/G5','/G6','/G7'],range(2),itertools.product(range(2),repeat=8)):
     # Single switches were already disproved; test only interactions here.
     if sum(bits)+eh+(arch!='/G7') < 2: continue
     opts=[arch]+(['/EHs','/EHc-'] if eh else [])+[f for f,b in zip(flags,bits) if b]
     tag='flags'+arch[2:]+str(eh)+''.join(map(str,bits))
     items.append((tag,BASE,opts))
    batch(items,'flag-interactions')

if FAMILY in ("all", 'inheritance'):
    items=[]
    for mi,virt,access,moused,cv in itertools.product(range(2),repeat=5):
     s=BASE
     prefix=0x74 if mi else 8
     thing=('class Thing { public: '+('virtual ~Thing(); ' if virt else 'void *m_vtbl; ')+'ThingTemplate *m_template; '+('char m_thingPad[0x70-8];' if mi else '')+' ThingTemplate *getTemplate() const { return m_template; } };\n')
     snapshot='class Snapshot { public: virtual ~Snapshot(); };\n' if mi else ''
     s=s.replace('class Object\n{',thing+snapshot+'class Object : public Thing'+(', public Snapshot' if mi else '')+'\n{')
     s=s.replace('void *m_vtbl; ThingTemplate *m_template; char m_pad8[0x74 - 8]; int m_id;',('int m_id;' if mi else 'char m_pad8[0x74 - 8]; int m_id;'))
     s=s.replace('class Drawable\n{','class Drawable : public Thing'+(', public Snapshot' if mi else '')+'\n{')
     s=s.replace('char m_pad0[0xFC]; Object *m_object;','char m_pad0[0xFC - '+str(prefix)+']; Object *m_object;')
     if access:
      s=s.replace('((Object *)draw)->m_template->getNameText()', 'draw->getTemplate()->getNameText()')
      s=s.replace('obj->m_template->','obj->getTemplate()->')
     if moused:
      s=s.replace('Object *obj;\n\t\tif (draw && (obj = draw->m_object) != 0)', 'Object *obj = 0;\n\t\tif(draw) obj=draw->m_object;\n\t\tif(obj)')
     if cv:
      s=s.replace('Object *m_object;', 'Object *const m_object;')
     items.append(('inherit'+''.join(map(str,(mi,virt,access,moused,cv))),s))
    batch(items,'inheritance')

if FAMILY in ("all", 'expression-scope'):
    find='TheGameClient->findDrawableByID(TheInGameUI->getMousedOverDrawableID())'
    get='TheInGameUI->getFirstSelectedDrawable()'
    cnt='TheInGameUI->getSelectCount()'
    decl='Drawable *draw = '+find+';'
    select='if (TheInGameUI->getSelectCount() == 1)\n\t\t\tdraw = '+get+';'
    anchor='void W3DDisplay::gatherDebugStats(void)\n{'
    update='UnicodeString unibuffer, unibuffer2;'
    forms=[
     '('+cnt+'!=1) || ((draw='+get+'),true);',
     '('+cnt+'==1) && ((draw='+get+'),true);',
     cnt+'==1 ? (void)(draw='+get+') : (void)0;',
     'if(!('+cnt+'!=1)) { Drawable *selected='+get+'; draw=selected; }',
     'switch('+cnt+') { case 1: draw='+get+'; break; default: break; }',
     'if (bool selected = '+cnt+'==1) draw='+get+';',
     'if('+cnt+'!=1) {} else draw='+get+';',
     '(void)(('+cnt+'==1 && (draw='+get+')) || draw);'
    ]
    items=[]
    for form,op,dp,reuse in itertools.product(range(8),range(4),range(4),range(2)):
     s=BASE
     if reuse:
      s=s.replace('Object *selObj;','').replace('selObj','obj')
     s=s.replace(select,forms[form])
     # Donor's object lifetime begins before the release find call; cross with
     # the other declaration birth points and with reuse in selection.
     s=s.replace('Object *obj;\n\t\tif (draw && (obj = draw->m_object) != 0)', 'if (draw) obj = draw->m_object;\n\t\tif (obj)')
     if op==0: s=s.replace(decl,'Object *obj=0;\n\t\t'+decl)
     if op==1: s=s.replace(decl,decl+'\n\t\tObject *obj=0;')
     if op==2: s=s.replace(update,'Object *obj=0;\n\t\t'+update)
     if op==3: s=s.replace(anchor,anchor+'\n\tObject *obj=0;')
     if dp:
      s=s.replace(decl,'draw = '+find+';')
      if dp==1: s=s.replace(update,'Drawable *draw;\n\t\t'+update)
      if dp==2: s=s.replace(anchor,anchor+'\n\tDrawable *draw;')
      if dp==3: s=s.replace(anchor,anchor+'\n\tDrawable *draw=0;')
     items.append(('expr%d%d%d%d'%(form,op,dp,reuse),s))
    batch(items,'expression-scope')

if FAMILY in ("all", 'lookup-expression'):
    find='TheGameClient->findDrawableByID(TheInGameUI->getMousedOverDrawableID())'
    lookup='Drawable *draw = '+find+';\n\t\tObject *obj;\n\t\tif (draw && (obj = draw->m_object) != 0)'
    forms=[
     'Drawable *draw; Object *obj; if ((draw='+find+') && (obj=draw->m_object)!=0)',
     'Drawable *draw=0; Object *obj; if ((draw='+find+') && (obj=draw->m_object)!=0)',
     'Drawable *draw; Object *obj=0; if(draw='+find+') obj=draw->m_object; if(obj)',
     'Object *obj=0; Drawable *draw; if(draw='+find+') obj=draw->m_object; if(obj)',
     'Drawable *draw; Object *obj; if (((draw='+find+'),draw) && (obj=draw->m_object)!=0)',
     'Drawable *draw; Object *obj; bool found=(draw='+find+')!=0; if(found && (obj=draw->m_object)!=0)',
     'Drawable *draw; Object *obj; bool missing=(draw='+find+')==0; if(!missing && (obj=draw->m_object)!=0)',
     'Drawable *draw; Object *obj; if ((draw='+find+'), (!draw ? false : (obj=draw->m_object)!=0))',
     'Drawable *draw; Object *obj=(draw='+find+') ? draw->m_object : 0; if(obj)',
     'Drawable *draw; Object *obj; if(draw='+find+') { obj=draw->m_object; } else { obj=0; } if(obj)',
     'Drawable *draw; Object *obj; bool found; if(found=((draw='+find+')!=0)) obj=draw->m_object; else obj=0; if(obj)',
     'Drawable *draw; if(Object *obj = (draw='+find+') ? draw->m_object : 0)',
     'Drawable *draw; Object *obj; if (!( (draw='+find+')==0 || (obj=draw->m_object)==0))',
     'Drawable *draw; Object *obj; if ((draw='+find+') == 0 ? false : ((obj=draw->m_object)!=0))',
     'Drawable *draw; Object *obj; if (((draw='+find+')!=0) ? ((obj=draw->m_object)!=0) : false)',
     'Drawable *draw; Object *obj; if (0!=(draw='+find+') && 0!=(obj=draw->m_object))'
    ]
    items=[]
    for f,bits in itertools.product(range(len(forms)),itertools.product(range(2),repeat=4)):
     a,b,c,d=bits; s=BASE.replace(lookup,forms[f]); assert s!=BASE
     if a:
      s=s.replace('virtual int getMousedOverDrawableID();','virtual int getMousedOverDrawableID() const;')
     if b:
      s=s.replace('virtual int getSelectCount();','virtual int getSelectCount() { return *(int *)((char *)this+0x568); }')
     if c:
      s=s.replace('draw = TheInGameUI->getFirstSelectedDrawable();','draw = drawRef(TheInGameUI->getFirstSelectedDrawable());')
      s=s.replace('void W3DDisplay::gatherDebugStats(void)','__forceinline Drawable *drawRef(Drawable *const &p) { return p; }\nvoid W3DDisplay::gatherDebugStats(void)')
     opts=['/Ob2'] if d else []
     items.append(('look%02d'%f+''.join(map(str,bits)),s,opts))
    batch(items,'lookup-expression')
