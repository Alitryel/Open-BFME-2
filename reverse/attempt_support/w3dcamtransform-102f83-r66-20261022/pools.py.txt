"""Named-local allocation order (diagnostic). usage: pools.py label src.cpp"""
from pathlib import Path
import sys,json
H=Path(__file__).resolve().parent;BASE=H.parent;ROOT=BASE.parents[1]
sys.path.insert(0,str(ROOT/'build/agX'));import canon_trace as C
sys.path.insert(0,str(BASE/'decode'));from symbol_decode import decoded_symbol
SYM='?rva00102F83@W3DCamTransform@@QAEXPAUCameraTransformInput@@@Z'
label,rel=sys.argv[1],sys.argv[2];OUT=H/'pools';OUT.mkdir(exist_ok=True)
alloc=[];prev=[]
def pending(t):
    while prev:
        a=prev.pop(0);d=decoded_symbol(t,a['address']);a['name']=d.get('name','') if isinstance(d,dict) else ''
def hit(t,info,c,r):
    if r['function']!=SYM:return
    pending(t)
    if info['name']=='@SymAlloc@4':
        a={'kind':c.Eax&255,'ret':r['return_rva']}
        def done(t,c):a['address']=c.Eax;a['id']=t.word(c.Eax+28);alloc.append(a);prev.append(a)
        t.on_return(c,done)
    elif info['name']=='@SymFree@4':
        a={'kind':-1,'ret':r['return_rva'],'address':c.Eax,'id':t.word(c.Eax+28),'fkind':t.read(c.Eax+4,1)[0]}
        d=decoded_symbol(t,c.Eax);a['name']=d.get('name','') if isinstance(d,dict) else ''
        alloc.append(a)
cpp=H/rel;cmd,env=C.build.compiler_command(cpp,OUT/(label+'.obj'))
C.trace_compiler.OUT=C.LAB;t=C.Tracer(label,cmd,SYM,on_hit=hit,env=env)
t.maps['@SymAlloc@4']=0x5A0C5
t.names=['@SymAlloc@4','@SymFree@4','@dop2_PreLowerGlobalOpts@4','@dop2_Codegen@4']
C.trace_compiler.OUT=OUT;t.run(timeout=240)
rows=[{k:a.get(k) for k in ('id','kind','name','ret','fkind')} for a in alloc]
(OUT/(label+'.json')).write_text(json.dumps(rows))
named=[r for r in rows if r['kind'] in (4,5)]
print(label,'alloc',len(rows),'named',len(named))
