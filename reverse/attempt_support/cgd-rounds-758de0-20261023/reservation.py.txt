from pathlib import Path
from itertools import product
import json
W=Path(__file__).resolve().parent
banks=[(W/'bank.cpp').read_text(),Path('reverse/attempt_support/cgd-final-schedule-758de0-20261022/control.cpp.txt').read_text()]
D=W/'reservation';D.mkdir(exist_ok=True)
rows=[]
for idx,(bank,dummy,sites,shape,cc) in enumerate(product(range(2),range(8),range(8),range(4),range(2))):
 s=banks[bank]
 dv=['*(void **)q','q','(void *)&n->point','(void *)&n->normal','n','n->first','n->second','(void *)n->state'][dummy]
 decl='typedef void * (__fastcall *RegisterPrepare)(ContactSource *,void *,const ContactVector &,const ContactVector &);\nstruct RegisterTable { void *slots[3]; RegisterPrepare prepare; };\n'
 body='ContactSource *q=data->source;'
 if shape==0:body+='return ((RegisterPrepare *)(*(void **)q))[3](q,'+dv+',p,normal);'
 elif shape==1:body+='RegisterTable *t=*(RegisterTable **)q;return t->prepare(q,'+dv+',p,normal);'
 elif shape==2:body+='RegisterPrepare f=((RegisterPrepare *)(*(void **)q))[3];return f(q,'+dv+',p,normal);'
 else:body+='RegisterPrepare &f=((RegisterPrepare *)(*(void **)q))[3];return f(q,'+dv+',p,normal);'
 decl+='__forceinline void * '+('__fastcall ' if cc else '')+'registerPrepare(ContactNode *n,ContactData *data,const ContactVector &p,const ContactVector &normal){'+body+'}\n'
 s=s.replace('class BfmeThingCGD {',decl+'class BfmeThingCGD {')
 originals=[('node->second->source->prepare(node->point,node->normal)','second','node->normal'),(('bridge(node->second,node->point,node->normal)' if bank==0 else 'node->second->source->prepare(node->point,node->normal)'),'second','node->normal'),(('altPrepare(node->first->source,node->point,ContactVector(node->normal).negate())' if bank==0 else 'node->first->source->prepare(node->point,ContactVector(node->normal).negate())'),'first','ContactVector(node->normal).negate()')]
 h,t=s.split('void BfmeThingCGD::bfmeTwoCGD() {',1)
 pos=0
 for i,(old,side,normal) in enumerate(originals):
  at=t.find(old,pos);assert at>=0,(idx,i,old)
  new='registerPrepare(node,node->'+side+',node->point,'+normal+')' if sites>>i&1 else old
  t=t[:at]+new+t[at+len(old):];pos=at+len(new)
 (D/('v%04d.cpp'%idx)).write_text(h+'void BfmeThingCGD::bfmeTwoCGD() {'+t)
 rows.append(dict(name='v%04d'%idx,bank=bank,dummy=dummy,sites=sites,shape=shape,fastcall_helper=cc))
(W/'reservation-models.json').write_text(json.dumps(rows,indent=1));print(len(rows))
