from pathlib import Path
import sys, json, subprocess, hashlib, concurrent.futures
root=Path(__file__).resolve().parents[1]
sys.path.insert(0,str(root/'tools'))
import build
out=root/'build/zh-sdk-repair-oct20'
out.mkdir(exist_ok=True)
sdk='reference/open-bfme-1/inputs/toolchains/vs2003/Program Files/Microsoft Visual Studio .NET 2003/Vc7/PlatformSDK/Include'
dx='reference/open-bfme-1/inputs/toolchains/dx81/include'
zh=build.ZH_REFERENCE_ROOT
audio=zh/'Libraries/Source/WWVegas/WWAudio'
benchmark=zh/'Libraries/Source/Benchmark'
assert (root/sdk/'windows.h').is_file()
assert (root/dx/'d3d8.h').is_file()
assert (audio/'WWAudio.h').is_file()
assert (benchmark/'benchmark.h').is_file()
names={'mono.cpp','mpu.cpp','rawfile.cpp','registry.cpp','WWCOMUtil.cpp','verchk.cpp','wwmouse.cpp','shdhwshader.cpp','animatedsoundmgr.cpp','W3DShaderManager.cpp'}
failures=json.loads((root/'build/zh-compile-failures-after-loader-oct20.json').read_text())
donors=[root/r['source'] for r in failures if Path(r['source']).name in names]
assert len(donors)==len(names),(len(donors),names)
revision=subprocess.check_output(['git','-C',str(root/'reference/open-bfme-1'),'rev-parse','HEAD'],text=True).strip()
tasks=[]
for n,donor in enumerate(donors):
    assert not any(s in donor.as_posix().lower() for s in ('vp6','on2','winamp','ffdshow','mfnode','bink','theora','video'))
    flags=build._ZH_BASE_FLAGS+['-I'+sdk,'-I'+dx,'-I'+audio.relative_to(root).as_posix(),'-I'+benchmark.relative_to(root).as_posix()]+build._reference_include_dirs(zh)
    for profile,extra in [('original',[]),('retail',['-O1','-G7','-arch:SSE2'])]:
        selected=flags+extra
        for flag in selected:
            if flag.startswith('-I'): assert (root/flag[2:]).is_dir(),flag
        def render(flag):return '-I"'+flag[2:]+'"' if flag.startswith('-I') and ' ' in flag else flag
        text='// cl: '+' '.join(map(render,selected))+'\n'
        if build.source_needs_stlport(donor):text+='// stlport\n'
        text+='#include "'+donor.as_posix()+'"\n'
        trial=out/(f'{n:02d}_{profile}_{donor.name}')
        trial.write_text(text)
        assert build.source_extra_flags(trial)==selected
        tasks.append({'donor':donor.relative_to(root).as_posix(),'sha256':hashlib.sha256(donor.read_bytes()).hexdigest(),'revision':revision,'profile':profile,'flags':selected,'trial':trial.relative_to(root).as_posix()})
(out/'inputs.json').write_text(json.dumps(tasks,indent=2)+'\n')
def run(task):
    p=subprocess.run([sys.executable,'tools/place_bodies.py','--min','8',task['trial']],cwd=root,capture_output=True,text=True,encoding='utf8',errors='replace',timeout=240)
    output=p.stdout+p.stderr
    (root/task['trial']).with_suffix('.log').write_text(output)
    return {**task,'returncode':p.returncode,'compile_failed':p.returncode!=0 or 'COMPILE FAILED' in output,'errors':[s for s in output.splitlines() if 'error ' in s][:5],'leads':[s for s in output.splitlines() if s.startswith(('BODY\t','PIN\t'))],'log_sha256':hashlib.sha256(output.encode()).hexdigest()}
results=[]
with concurrent.futures.ThreadPoolExecutor(max_workers=3) as pool:
    for r in pool.map(run,tasks):
        results.append(r)
        (out/'results.json').write_text(json.dumps(results,indent=2)+'\n')
        print(Path(r['donor']).name,r['profile'],'failed',r['compile_failed'],'leads',r['leads'],'errors',r['errors'][:1],flush=True)
