// compiler:clang23-x86
// cl: /O1 /G6 /GS- /MD
// CRT helper protocol: two 64-bit stack arguments, EDX:EAX result, RET 16.
// The compiler's native external spelling has no stdcall suffix.
extern "C" unsigned __int64 __stdcall _allmul(unsigned __int64 a,
    unsigned __int64 b) __asm__("__allmul");

unsigned __int64 __stdcall _allmul(unsigned __int64 a, unsigned __int64 b)
{
    unsigned a_low = (unsigned)a;
    unsigned b_low = (unsigned)b;
    unsigned a_high = (unsigned)(a >> 32);
    unsigned b_high = (unsigned)(b >> 32);
    if ((a_high | b_high) == 0)
        return (unsigned __int64)a_low * b_low;

    unsigned cross = a_high * b_low;
    cross += a_low * b_high;
    unsigned __int64 low_product = (unsigned __int64)a_low * b_low;
    return low_product + ((unsigned __int64)cross << 32);
}
