from headmerge import *
# New hypothesis: a redundant null guard after the two-definition join makes
# the p==0 fact available to jump threading, with the packed weight still 3.
# Object is deliberately uninitialized only when the short-circuit p guard
# guarantees it is not evaluated on that path.
first=['p','(unsigned)p>0','(unsigned)p>=1','(unsigned)p!=0']
nulls=['obj=0;', 'obj=(Object*)p;', '']
finals=['p && obj','obj && p','p ? obj!=0 : false',
        'p ? bool(obj) : false','!(!p || !obj)',
        '(unsigned)p>0 && obj','obj && (unsigned)p>0',
        'p!=0 && obj!=0','(p && obj) ? true : false',
        'p ? (obj ? true : false) : false',
        '(p!=0) & (obj!=0)', 'obj']
items=[]
for test,nul,fin,order,neg,contract in itertools.product(range(4),range(3),range(12),range(2),range(2),range(2)):
 if nul==2 and fin in [1,6,10,11]:continue
 yesfirst='draw=p; obj=p->m_object;' if not order else 'obj=p->m_object; draw=p;'
 arms='if('+first[test]+') {'+yesfirst+'} else {draw=p; '+nulls[nul]+'}'
 tail=('if(!('+finals[fin]+')) {'+terrain+'} else {'+body+'\n}' if neg else
       'if('+finals[fin]+') {'+body+'\n} else {'+terrain+'}')
 src=BASE[:lo]+decl+arms+tail+'\n'+BASE[hi:]
 if not contract:src=src.replace('template <> void StringBase<char>::releaseBuffer() throw();\n','')
 items.append(('fj%d%d%02d%d%d%d'%(test,nul,fin,order,neg,contract),src))
if __name__=='__main__':batch(items,'factjoin')
