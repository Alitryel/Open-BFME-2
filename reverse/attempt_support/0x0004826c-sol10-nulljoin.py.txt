from headmerge import *

# Reuse the two-arm-copy contexts from the preceding grid. Null values carried
# as pointers/aggregate values may stay as copies until jump threading, rather
# than becoming the zero-register source before head merging.
nulls=[
 ('','0'),
 ('','(Object*)p'),
 ('','reinterpret_cast<Object*>(p)'),
 ('static __forceinline Object *nullObject(Drawable *q) {return (Object*)q;}\n','nullObject(p)'),
 ('struct ObjValue { Object *v; ObjValue(Object *q):v(q){} operator Object*() const {return v;} };\n','ObjValue((Object*)p)'),
 ('static __forceinline Object *nullObject(Object *const &q) {return q;}\n','nullObject((Object*)p)'),
]
items=[]
select='if (TheInGameUI->getSelectCount() == 1)\n\t\t\tdraw = TheInGameUI->getFirstSelectedDrawable();'
for f,n,cond,sel,contract in itertools.product([0,1,8,9,10,11,12,15],range(6),range(2),range(3),range(2)):
 helper,null=nulls[n]
 form=forms[f].replace('obj=0','obj='+null)
 local=decl
 if cond:
  local=local.replace('Object *obj;','Object *obj=(Object*)p;')
  # Only the first test, before getObject writes obj, sees the pointer alias.
  form=form.replace('if(p)', 'if(obj)',1).replace('p?p->m_object','obj?p->m_object',1)
 src=BASE[:lo]+local+form+'\n'+BASE[hi:]
 if sel==1:
  src=src.replace(select,'draw = TheInGameUI->getSelectCount()==1 ? TheInGameUI->getFirstSelectedDrawable() : p;')
 elif sel==2:
  src=src.replace(select,'if (TheInGameUI->getSelectCount() == 1) draw = TheInGameUI->getFirstSelectedDrawable(); else draw=p;')
 if not contract:src=src.replace('template <> void StringBase<char>::releaseBuffer() throw();\n','')
 src=src.replace('void W3DDisplay::gatherDebugStats(void)',helper+'void W3DDisplay::gatherDebugStats(void)')
 items.append(('nj%02d%d%d%d%d'%(f,n,cond,sel,contract),src))

if __name__=='__main__':batch(items,'nulljoin')
