import sys,json
from pathlib import Path
sys.path.insert(0,str(Path(__file__).parent))
from wdbg import Dbg
src,obj=sys.argv[1:3];d=Dbg(src,obj,obj+'.debug.log')
d.prompt();d.prompt('set $CanDeferOnBPByAddr = 1')
BASE=0x10700000;focus=False;roundno=0;armed=False
d.prompt('break *0x%x'%(BASE+0x83ac8))
while True:
    out=d.prompt('cont')
    if 'terminated' in out:break
    ip=d.reg('eip')-BASE
    if ip==0x83ac8:
        f=d.words(d.reg('ebp'),1)[0];fn=d.string(d.words(f+4,1)[0]);focus='onComingOutOfShroud' in fn and d.reg('ebx')==0;roundno=0
        if focus and not armed:
            for off in [0x82978,0x8292a,0x83a56,0x83a7d,0x82273]:d.prompt('break *0x%x'%(BASE+off))
            armed=True
        continue
    if not focus:continue
    if ip==0x82978:roundno+=1
    elif ip==0x82273:focus=False
    else:
        p=d.reg('eax');w=d.words(p,18)
        if w[7] not in [5,7]:continue
        new=d.reg('ecx' if ip==0x83a7d else 'ebx')
        print(json.dumps(dict(phase=hex(ip),round=roundno,record=p,index=w[7],old=w[3],new=new,refs=w[9],cost=w[16],words=w,edx=d.reg('edx'),stack=d.words(d.reg('esp'),13))),flush=True)
