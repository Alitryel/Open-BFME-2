from search import *
import itertools
s=(W/'base.cpp').read_text();vs=[]
# A zero represented as floating storage has a different IR type/alias class
# from the already-refuted integer constants. No value-changing probes.
forms=[('','0')]
for ty,val in [('float','0.0f'),('double','0.0'),('unsigned __int64','0ui64'),('float','-0.0f')]:
 for style in range(6):
  if style==0:dec='union N { '+ty+' bits; Coord3D *ptr; } n; n.bits='+val+';';expr='n.ptr'
  elif style==1:dec=ty+' z='+val+';';expr='*(Coord3D**)&z'
  elif style==2:dec='const '+ty+' z='+val+';';expr='*(Coord3D*const*)&z'
  elif style==3:dec='union N { '+ty+' bits; unsigned raw; } n; n.bits='+val+';';expr='(Coord3D*)n.raw'
  elif style==4:dec=ty+' z='+val+'; Coord3D *n; memcpy(&n,&z,sizeof(n));';expr='n'
  else:dec='union N { '+ty+' bits; Coord3D *ptr; N('+ty+' b):bits(b){} }; N n('+val+');';expr='n.ptr'
  if val=='-0.0f':
   # Preserve null bits: flip the known sign bit instead of passing -0 bits.
   expr='(Coord3D*)((unsigned)('+expr+')^0x80000000U)'
  forms.append((dec,expr))
for i,(dec,expr) in enumerate(forms):
 for sites,where,helper in itertools.product([1,2,3],range(2),range(2)):
  pre='#include <string.h>\n';o=s
  if helper:
   pre+='__forceinline Coord3D* bitZeroNormal(){'+dec+'return '+expr+';}\n';value='bitZeroNormal()';init=''
  else:value=expr;init=dec
  a=o.index('  case 1:');b=o.index('  case 4:',a);c=o.index('  case 6:');d=o.index('  case 7:',c)
  sea=o[a:b];ground=o[c:d]
  if sites&1:
   sea=sea.replace('   float water;','   float water;'+init) if where==0 else sea.replace('   requires=false;',init+'   requires=false;',1)
   sea=sea.replace('object->rva0028B511(),0,true)','object->rva0028B511(),'+value+',true)')
  if sites&2:
   ground=ground.replace('   pos.z=',init+'   pos.z=',1) if where==0 else ground.replace('   requires=true;',init+'   requires=true;',1)
   ground=ground.replace('->ground(pos.x,pos.y,0)','->ground(pos.x,pos.y,'+value+')')
  o=o[:c]+ground+o[d:];o=o[:a]+sea+o[b:]
  o=o.replace('bool Locomotor::rva001E7ECA',pre+'bool Locomotor::rva001E7ECA')
  vs.append((f'zerobits_{i:02d}_{sites}{where}{helper}',o))
run(vs,'zerobits300')
