import sys,itertools,re
sys.path.insert(0,'build/netpacket-20261022'); from search import *
SIGN='NetCommandList *NetPacket::getCommandList()\n'
H,B=BANK.split(SIGN,1)
items=[]
for kind,pos,uses,indextype in itertools.product(range(4),range(4),range(1,8),range(3)):
 decl=['NetPacket *packet = this;','NetPacket *const packet = this;','NetPacket &packet = *this;','NetPacket *const &packet = this;'][kind]
 prefix='packet.' if kind==2 else 'packet->'
 code=B
 if uses&1:code=code.replace('m_packetLen',prefix+'m_packetLen')
 if uses&2:code=re.sub(r'\bm_packet\b',prefix+'m_packet',code)
 if uses&4:code=code.replace('rva0058E57B();',prefix+'rva0058E57B();')
 if indextype==1:code=code.replace('Int offset;','UnsignedInt offset;').replace('m_packet, offset)', 'm_packet, (Int &)offset)')
 if indextype==2:code=code.replace('Int i = 0;', 'UnsignedInt i = 0;').replace('i < '+prefix+'m_packetLen','(Int)i < '+prefix+'m_packetLen').replace('i < m_packetLen','(Int)i < m_packetLen')
 markers=['\n\tNetCommandList *retval','\n\tUnsignedByte commandType','\n\tInt offset' if indextype!=1 else '\n\tUnsignedInt offset','\n\twhile (i <' if indextype!=2 else '\n\twhile ((Int)i <']
 marker=markers[pos]; assert marker in code
 code=code.replace(marker,'\n\t'+decl+marker,1)
 items.append((f'self_{kind}_{pos}_{uses}_{indextype}',H+SIGN+code))
batch(items,'self-aliases')
