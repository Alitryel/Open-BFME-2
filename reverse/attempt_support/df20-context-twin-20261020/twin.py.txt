from pathlib import Path
import sys,json,collections
import pefile,capstone
from capstone.x86_const import X86_OP_IMM,X86_OP_MEM
sys.path.insert(0,'tools');import build
root=Path.cwd(); dest=root/'build/df20-context-20261020'
wb=pefile.PE(str(root/'baselines/bfme2/workshop-vanilla-1.06/files/worldbuilder.exe'))
wa=0x6e71a0;ga=0x40df20;size=2610
wraw=wb.get_data(wa-wb.OPTIONAL_HEADER.ImageBase,size);graw=build.read_target_bytes(ga-0x400000,size)
md=capstone.Cs(capstone.CS_ARCH_X86,capstone.CS_MODE_32);md.detail=True
wi=list(md.disasm(wraw,wa));gi=list(md.disasm(graw,ga));changes=[];bad=[]
assert len(wi)==len(gi)
for w,g in zip(wi,gi):
 off=g.address-ga
 if w.address-wa!=off or w.size!=g.size or w.mnemonic!=g.mnemonic:bad.append(['shape',off]);continue
 wm=bytearray(w.bytes);gm=bytearray(g.bytes)
 if w.mnemonic in ('call','jmp') and w.operands and w.operands[0].type==X86_OP_IMM:
  wt=w.operands[0].imm;gt=g.operands[0].imm
  if not wa<=wt<wa+size and not ga<=gt<ga+size:
   n=w.encoding.imm_size;o=w.encoding.imm_offset
   assert n==4
   wm[o:o+n]=gm[o:o+n]=bytes(n);changes.append({'offset':hex(off),'kind':'external-branch','game_va':hex(gt),'wb_va':hex(wt)})
 for idx,(wo,go) in enumerate(zip(w.operands,g.operands)):
  if w.mnemonic in ('call','jmp') and wo.type==X86_OP_IMM:continue
  if wo.type==go.type==X86_OP_IMM and 0x400000<=wo.imm<0x400000+wb.OPTIONAL_HEADER.SizeOfImage and 0x400000<=go.imm<0x1900000:
   o=w.encoding.imm_offset;n=w.encoding.imm_size
   wm[o:o+n]=gm[o:o+n]=bytes(n);changes.append({'offset':hex(off),'kind':'image-immediate','game_va':hex(go.imm),'wb_va':hex(wo.imm)})
 if wm!=gm:bad.append({'offset':hex(off),'game':f'{g.mnemonic} {g.op_str}','wb':f'{w.mnemonic} {w.op_str}'})
result={'game_va':hex(ga),'wb_va':hex(wa),'size':size,'instructions':len(gi),'masked_sites':len(changes),'remaining_differences':bad,'sites':changes}
(dest/'twin.json').write_text(json.dumps(result,indent=2)+'\n')
print(json.dumps({k:v for k,v in result.items() if k!='sites'}))
