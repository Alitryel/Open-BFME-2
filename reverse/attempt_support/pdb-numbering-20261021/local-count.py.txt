from pathlib import Path
import subprocess,sys,json,concurrent.futures
out=Path('build/resume-eh-local/turn58-pdb-locals');out.mkdir(exist_ok=True)
base=Path('reverse/attempt_support/pdb-numbering-20261021/candidate.cpp.txt').read_text()
name='?_RTC_GetSrcLine@@YAHKPADHPAHPAPAD@Z'
anchor='int _RTC_GetSrcLine(unsigned long address,char *source,int sourcelen,int *pline,char **moduleName) {'
assert base.count(anchor)==1
def run(k):
 locals='\n'+''.join(' int unused%d = 0;\n'%i for i in range(k))
 p=out/('pdb_%03d.cpp'%k);p.write_bytes(base.replace(anchor,anchor+locals).encode())
 r=subprocess.run([sys.executable,'tools/explain_mismatch.py',name,'--rva','0x0075B177','--size','847','--source',p.as_posix()],stdout=subprocess.PIPE,stderr=subprocess.STDOUT,text=True)
 (out/('pdb_%03d.txt'%k)).write_bytes(r.stdout.encode())
 info={'k':k,'exit':r.returncode,'source':p.as_posix()}
 for field in ['compiled symbol size','classification','first diff:','unresolved']:
  lines=[x for x in r.stdout.splitlines() if x.lower().startswith(field)]
  if lines:info[field]=lines[0]
 return info
results=[]
with concurrent.futures.ThreadPoolExecutor(max_workers=2) as pool:
 for result in pool.map(run,range(129)):
  results.append(result)
  if result['k']%16==0 or result.get('classification')=='classification: exact match':print(result,flush=True)
  (out/'results.json').write_bytes(json.dumps(results,indent=2).encode())
print('classes',sorted(set(x.get('classification','ERROR') for x in results)),flush=True)
