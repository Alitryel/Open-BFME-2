from pathlib import Path
import subprocess,sys,json
root=Path(__file__).resolve().parents[1];out=root/'build/activebody-kind-bitset-oct20';base=(out/'kind218-ob1-budget100.cpp').read_text()
declarations='''
class Drawable;
namespace _STL {
template<> __declspec(noinline) _List_base<Drawable *, allocator<Drawable *> >::_List_base(const allocator<Drawable *> &);
template<> __declspec(noinline) list<Drawable *, allocator<Drawable *> >::iterator list<Drawable *, allocator<Drawable *> >::insert(iterator, Drawable *const &);
}
'''
tests={'list-boundaries':base.replace('#include <list>','#include <list>\n'+declarations,1)}
plain=(out/'kind218-default.cpp').read_text()
for kind,body in [('checked','m_kindOf.test(k)'),('const-index','m_kindOf[k]'),('unchecked','m_kindOf._Unchecked_test(k)')]:
    for force in (False,True):
        s=plain.replace('return m_kindOf.test(k);','return '+body+';',1)
        if force:s=s.replace('Bool testKindOf(Int k)','__forceinline Bool testKindOf(Int k)',1)
        tests[kind+('-force' if force else '')]=s
for key,s in list(tests.items()):
    if key!='list-boundaries':tests[key+'-boundaries']=s.replace('#include <list>','#include <list>\n'+declarations,1).replace('/O1 /I.','/O1 /Ob1 /d2inlS100 /I.',1)
name='?attemptDamage@ActiveBody@@UAEXPAVDamageInfo@@@Z';results=[]
for key,s in tests.items():
    p=out/('boundary-'+key+'.cpp');p.write_text(s)
    if '--generate-only' in sys.argv:
        print('Generated scratch source only: '+p.relative_to(root).as_posix(),flush=True)
        continue
    r=subprocess.run([sys.executable,'tools/explain_mismatch.py',name,'--rva','0x004BFE07','--size','2107','--source',p.relative_to(root).as_posix()],cwd=root,capture_output=True,text=True,encoding='utf8',errors='replace',timeout=240)
    log=r.stdout+r.stderr;p.with_suffix('.log').write_text(log)
    d={'key':key,'exit':r.returncode,'exact':'OK: bytes match' in log,'summary':[x for x in log.splitlines() if x.startswith(('compiled symbol size:','classification:','unresolved:','first diff:'))]}
    if r.returncode:d['error']=log[-700:]
    results.append(d);(out/'boundary-results.json').write_text(json.dumps(results,indent=2));print(json.dumps(d),flush=True)
