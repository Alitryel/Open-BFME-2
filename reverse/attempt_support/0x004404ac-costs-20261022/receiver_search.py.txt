from search import *
import re

def source(n):
    kind=n%8;pos=(n//8)%8;use=n//64
    marks=['\n\tGameInfo *game','\n\tif (!game)', '\n\tif (slot == -1 || slot >= 8)',
           '\n\tbool observer', '\n\tGameWindow *hero','\n\tGameWindow **team',
           '\n\tif (slot == game->v13())','\n\t\tbool found']
    idx=BODY.index(marks[pos]);tail=BODY[idx:]
    decl=[
        'AptMpGameSetup *self = this;',
        'AptMpGameSetup *const self = this;',
        'AptMpGameSetup *const &self = this;',
        'AptMpGameSetup &self = *this;',
        'AptMpGameSetup *volatile self = this;',
        'struct Receiver { AptMpGameSetup *p; }; Receiver self = {this};',
        'AptMpGameSetup *self[1] = {this};',
        'union Receiver { AptMpGameSetup *p; unsigned long u; }; Receiver self = {this};',
    ][kind]
    prefix=['self->','self->','self->','self.','self->','self.p->','self[0]->','self.p->'][kind]
    if use in (0,1):tail=re.sub(r'(?<![.>\w])m_[a-zA-Z0-9_]+\b',lambda m:prefix+m.group(0),tail)
    if use in (0,2):tail=re.sub(r'\b(HandleHeroSelection|UpdateHeroDisplay)\(',lambda m:prefix+m.group(1)+'(',tail)
    if use==3:tail=tail.replace('m_60',prefix+'m_60')
    return HEAD+MARK+BODY[:idx]+'\n '+decl+tail

if __name__=='__main__':run([('recv-%03d'%n,source(n)) for n in range(256)],'receiver')
