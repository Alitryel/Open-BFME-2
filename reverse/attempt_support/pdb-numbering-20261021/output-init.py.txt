from pathlib import Path
import subprocess,sys,json,itertools,concurrent.futures
out=Path('build/resume-eh-local/turn58-pdb-output-init');out.mkdir(exist_ok=True)
base=Path('reverse/attempt_support/pdb-numbering-20261021/candidate.cpp.txt').read_text()
name='?_RTC_GetSrcLine@@YAHKPADHPAHPAPAD@Z'
anchor='int _RTC_GetSrcLine(unsigned long address,char *source,int sourcelen,int *pline,char **moduleName) {'
types={'source':'char *source','pline':'int *pline','moduleName':'char **moduleName','res':'int &res'}
items=[]
for order in itertools.permutations(types):
 helper='static __forceinline void ResetLookupOutputs('+','.join(types[x] for x in order)+') { res=0; *pline=0; source[0]=0; *moduleName=0; }\n'
 s=base.replace(anchor,helper+anchor).replace('int res=0;ImageInfo *info;','int res;ImageInfo *info;').replace('*pline=0;source[0]=0;*moduleName=0;','ResetLookupOutputs('+','.join(order)+');')
 items.append(('-'.join(order),s))
for order in itertools.permutations(['source','pline','moduleName']):
 helper='static __forceinline int ResetLookupOutputs('+','.join(types[x] for x in order)+') { *pline=0; source[0]=0; *moduleName=0; return 0; }\n'
 s=base.replace(anchor,helper+anchor).replace('int res=0;ImageInfo *info;','int res;ImageInfo *info;').replace('*pline=0;source[0]=0;*moduleName=0;','res=ResetLookupOutputs('+','.join(order)+');')
 items.append(('return-'+'-'.join(order),s))
def run(item):
 label,s=item;p=out/(label+'.cpp');p.write_bytes(s.encode())
 r=subprocess.run([sys.executable,'tools/explain_mismatch.py',name,'--rva','0x0075B177','--size','847','--source',p.as_posix()],stdout=subprocess.PIPE,stderr=subprocess.STDOUT,text=True)
 (out/(label+'.txt')).write_bytes(r.stdout.encode())
 info={'trial':label,'exit':r.returncode,'source':p.as_posix()}
 for field in ['compiled symbol size','classification','first diff:','unresolved']:
  lines=[x for x in r.stdout.splitlines() if x.lower().startswith(field)]
  if lines:info[field]=lines[0]
 return info
with concurrent.futures.ThreadPoolExecutor(max_workers=2) as pool:results=list(pool.map(run,items))
(out/'results.json').write_bytes(json.dumps(results,indent=2).encode())
print(json.dumps(results,indent=2))
