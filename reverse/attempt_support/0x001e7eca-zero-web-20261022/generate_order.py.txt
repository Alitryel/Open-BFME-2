from search import *
import itertools
s=(W/'base.cpp').read_text()
keys=['8','0','1','4','6','7','2']
starts=[s.index('  case '+k+':') for k in keys]
end=s.index('\n }\n',starts[-1])
blocks={k:s[a:b] for k,a,b in zip(keys,starts,starts[1:]+[end])}
# Keep early returns and common exit while changing block insertion/numbering.
vs=[]
for i,order in enumerate(itertools.permutations(['0','1','4','6','7','2'])):
 o=s[:starts[0]]+blocks['8']+''.join(blocks[k] for k in order)+s[end:]
 vs.append((f'order_{i:03d}',o))
run(vs,'order720')
