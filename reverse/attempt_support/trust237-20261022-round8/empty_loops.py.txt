from pathlib import Path
import itertools,sys
ROOT=Path.cwd();sys.path.insert(0,str(ROOT/'build/trust237-round5'))
import search
search.OUT=ROOT/'build/trust237-round8'
# All loops terminate for every 32-bit u. Accumulated z is mathematically
# zero; unsigned consecutive products are even, including wraparound.
loops=[
 ('count_up','for(unsigned j=0;j<(u&31u);++j){}'),
 ('count_down','for(unsigned j=u&31u;j;--j){}'),
 ('one_step','for(unsigned j=u;j!=u+1u;++j){}'),
 ('sixteen_steps','for(unsigned j=u;j!=u+16u;++j){}'),
 ('clear_bits','for(unsigned j=u;j;j&=j-1u){}'),
 ('while_steps','unsigned j=u;while(j!=u+16u){++j;}'),
 ('even_product','for(unsigned j=0;j<(u&7u);++j){z+=(j*(j+1u))&1u;}'),
 ('xor_pairs','for(unsigned j=0;j<16u;++j){z^=u;}'),
]
tails=[
 'return 1+(int)z;',
 'int r=(int)z;if((u&1u)<=1u)goto yes;return r;yes:return r+1;',
]
jobs=[]
for (kind,loop),reps,nest,pos,tail,opt,cpu in itertools.product(loops,[1,16],range(2),range(2),range(2),['O1','O2'],[6,7]):
 if nest:
  region='for(unsigned repeat=0;repeat<'+str(reps)+'u;++repeat){'+loop+'}'
 else:region=('{'+loop+'}')*reps
 code='long status=WinVerifyTrust(0,&action,&data);unsigned long u=(unsigned long)status;unsigned z=0;'
 code+=region+'LocalFree(argv);' if pos==0 else 'LocalFree(argv);'+region
 code+=tails[tail]
 text=search.base.replace(search.marker,code).replace('/O1 /G7',f'/{opt} /G{cpu}')
 jobs.append((f'e{len(jobs):04}',text,dict(kind=kind,repetitions=reps,nested=nest,position=pos,tail=tail,opt=opt,cpu=cpu)))
if __name__=='__main__':search.execute('empty-loops',jobs)
