import itertools,sys
from pathlib import Path
src=Path('bank.cpp').read_text()
K=[("  if(!unit || (unit->flags&1)) continue;\n","  if(unit && !(unit->flags&1)) {\n"),
   ("  if(unit->aux && *(unsigned char *)((char *)unit->aux+0x5c)) continue;\n","  if(!unit->aux || !*(unsigned char *)((char *)unit->aux+0x5c)) {\n"),
   ("  if(!ai) continue;\n","  if(ai) {\n"),
   ("  if(index<0 || index>=attacks.size()) continue;\n","  if(index>=0 && index<attacks.size()) {\n"),
   ("  if(*(int *)((char *)ai+0x1fc)==4 && *(int *)((char *)ai+0x140)) { entry.needsPosition=true; entry.timer=0; continue; }\n","  if(*(int *)((char *)ai+0x1fc)==4 && *(int *)((char *)ai+0x140)) { entry.needsPosition=true; entry.timer=0; } else {\n"),
   ("  if(unit->testStatus(AMOEBA_STATUS_1C) || unit->testStatus(AMOEBA_STATUS_45)) continue;\n","  if(!unit->testStatus(AMOEBA_STATUS_1C) && !unit->testStatus(AMOEBA_STATUS_45)) {\n")]
END="  ++count;\n }\n _STL::sort"
out=Path(sys.argv[1]); out.mkdir(exist_ok=True)
for combo in itertools.product((0,1),repeat=len(K)):
    pre,s=src.split('void HordeMeleeAmoeba::updateMeleeAttack',1); s='void HordeMeleeAmoeba::updateMeleeAttack'+s
    for (a,b),c in zip(K,combo):
        assert a in s
        if c: s=s.replace(a,b,1)
    n=sum(combo); s=s.replace(END,"  ++count;\n"+"  }\n"*n+" }\n _STL::sort",1)
    s=pre+s
    (out/('n'+''.join(map(str,combo))+'.cpp')).write_text(s)
