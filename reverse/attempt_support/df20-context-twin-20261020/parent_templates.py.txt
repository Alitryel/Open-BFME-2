import trials,itertools,concurrent.futures,json
specs=[]
original='template <> _MoneyOut money_put<wchar_t, _MoneyOut>::do_put(_MoneyOut __s, bool __intl, ios_base& __str, char_type __fill, const string_type& __digits) const'
for generic,attr,emit,initializer,aligned in itertools.product([0,1],['','inline ','__forceinline '],[0,1],[0,1],[0,1]):
 s=trials.bank
 if generic:
  sig='template <class _BankChar, class _BankOut> '+attr+'_BankOut money_put<_BankChar, _BankOut>::do_put(_BankOut __s, bool __intl, ios_base& __str, char_type __fill, const string_type& __digits) const'
  s=s.replace(original,sig).replace('  typedef wchar_t _CharT;','  typedef wchar_t _CharT;\n  typedef basic_string<wchar_t> string_type;\n  typedef wchar_t char_type;')
 else:s=s.replace(original,original.replace('template <> _MoneyOut','template <> '+attr+'_MoneyOut'))
 if initializer:s=s.replace('  bool __is_negative;\n  __is_negative =','  bool __is_negative =')
 if not aligned:s=s.replace('__declspec(align(8)) money_base::pattern','money_base::pattern')
 if emit:
  emitter='''
namespace _STL {
struct _MoneyParentEmitter:money_put<wchar_t,_MoneyOut> {
#pragma inline_depth(0)
 static _MoneyOut invoke(_MoneyParentEmitter *p,_MoneyOut a,bool b,ios_base& s,wchar_t f,const basic_string<wchar_t>&d) {return p->money_put<wchar_t,_MoneyOut>::do_put(a,b,s,f,d);}
#pragma inline_depth(255)
};
_MoneyOut (*_emit_money)(_MoneyParentEmitter*,_MoneyOut,bool,ios_base&,wchar_t,const basic_string<wchar_t>&)=&_MoneyParentEmitter::invoke;
}
'''
  s+=emitter
 elif generic:s+='\nnamespace _STL {template _MoneyOut money_put<wchar_t,_MoneyOut>::do_put(_MoneyOut,bool,ios_base&,wchar_t,const basic_string<wchar_t>&) const;}\n'
 specs.append((f'parent-g{generic}-a{attr.strip() or "plain"}-e{emit}-b{initializer}-l{aligned}',s))
with concurrent.futures.ThreadPoolExecutor(max_workers=8) as p:r=list(p.map(trials.run,specs))
(trials.scratch/'parent-templates.json').write_text(json.dumps(r,indent=2)+'\n')
