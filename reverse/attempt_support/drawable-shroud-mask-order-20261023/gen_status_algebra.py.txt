from pathlib import Path
import itertools
r=Path(__file__).parent;d=r/'status_algebra';d.mkdir(exist_ok=True)
b=(Path(__file__).parent/'native_split.cpp').read_text()
# Each predicate is exactly !(v & 64) for every unsigned 32-bit v.
forms=[
 '!(V & 0x40U)',
 '((V | ~0x40U) != ~0U)',
 '((~V & 0x40U) != 0)',
 '(~(V | ~0x40U) != 0)',
 '((V & 0x40U) != 0x40U)',
 '((V & 0x40U) < 0x40U)',
 '((V & 0x40U) <= 0)',
 '((V & 0x40U) ^ 0x40U)',
 '(((V & 0x40U) - 0x40U) != 0)',
 '(((V >> 6) & 1U) == 0)',
 '(((unsigned char)V & 0x40U) == 0)',
 '(((unsigned char)(V << 1) & 0x80U) == 0)',
 '((int)(V << 25) >= 0)',
 '((V | 0x40U) != V)',
 '((V ^ 0x40U) > V)',
 '(((V ^ 0x40U) & 0x40U) != 0)',
 '(((V ^ 0x40U) & 0x40U) == 0x40U)',
 '(((V & 0x40U) ^ 0x40U) == 0x40U)',
 '((V & 0x40U) ? false : true)',
 '(((V | ~0x40U) + 1U) != 0)',
 '((V | ~0x40U) < ~0U)',
 '((((V & 0x40U) >> 6) - 1U) != 0)',
 '(((signed char)(V << 1)) >= 0)',
 '((V & 0x40U) / 0x40U == 0)',
]
for i,j in itertools.product(range(len(forms)),repeat=2):
 s=b.replace('!(m_status & 0x40)', forms[i].replace('V','m_status')).replace('!(castleDraw->m_status & 0x40)',forms[j].replace('V','castleDraw->m_status'))
 (d/f'a{i:02d}b{j:02d}.cpp').write_text(s)
print(len(list(d.glob('*.cpp'))))
