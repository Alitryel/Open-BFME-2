"""Private diagnostic trials: known numeric status, not Boolean-return arithmetic."""
from pathlib import Path
import concurrent.futures, hashlib, itertools, json, re, runpy

OUT=Path('build/canmake-alias-20261023/numeric-status').resolve()
OUT.mkdir(parents=True,exist_ok=True)
n=runpy.run_path('reverse/attempt_support/canmake-early-cleanup-20261022/replay.py.txt')
e=n['context'](); r=e['r']; r['run'].__globals__['OUT']=OUT
bank=__import__('subprocess').check_output(['git','cat-file','blob','20afdd90f70d238a5f035acc08f72ff23e28a844'],text=True).split('int AIBuildableWallSegment::canMake(Player *player)',1)[1]
bank=bank[bank.index('{')+1:bank.rindex('}')]
bases=[bank,n['early'](e),e['win']]
seven=['(v-5u)','(v/3u)','(v%5u)','(v&2u)','(v^5u)','((v>>1)-1u)','((v+1u)>>2)','(2u*v/7u)']
zero=['(v+2u)','(v|2u)','(v^2u)','(v?0u:2u)','(2u-v)','((v+1u)*2u)','(~v+3u)','((v<<1)+2u)']
for v in range(8):
    pass  # Formula assertions below use Python equivalents, with unsigned wrapping.
assert [7-5,7//3,7%5,7&2,7^5,(7>>1)-1,(7+1)>>2,2*7//7]==[2]*8
assert [0+2,0|2,0^2,2,2-0,(0+1)*2,((-1)+3)&0xffffffff,(0<<1)+2]==[2]*8
tests=[]; seen=set(); choices={}
for base,s,z,binding,placement in itertools.product(range(3),range(8),range(8),range(4),range(2)):
    b=bases[base]
    # The first literal two is status seven; the second is insufficient funds.
    positions=[m.start() for m in re.finditer(r'return 2;',b)]
    assert len(positions)==2,(base,positions)
    v=['((unsigned int)result.m_20)','status','status','((unsigned int)status)'][binding]
    forms=[seven[s].replace('v',v),zero[z].replace('v',v)]
    for pos,form in reversed(list(zip(positions,forms))):
        b=b[:pos]+'return '+form+';'+b[pos+9:]
    if binding:
        declaration=['','int status=result.m_20;','unsigned int status=result.m_20;','const int &status=result.m_20;'][binding]
        # Bind after the quote and either before or inside the successful quote region.
        if placement==0:
            at=b.index('if (failed)') if 'if (failed)' in b else b.index('if(failed)')
            b=b[:at]+declaration+'\n'+b[at:]
        else:
            marker='return 9;'
            at=b.index(marker)+len(marker)
            if b[at:].lstrip().startswith('}'):
                at=b.index('}',at)+1
            b=b[:at]+'\n'+declaration+'\n'+b[at:]
        # Conditions use the same saved value, allowing real def/use joins.
        for old,new in [('switch (result.m_20)','switch (status)'),('result.m_20==','status=='),('result.m_20!=','status!='),('result.m_20 ==','status =='),('result.m_20 !=','status !=')]:
            b=b.replace(old,new)
    source=e['prefix']+'\nint AIBuildableWallSegment::canMake(Player *player) {\n'+b+'\n}\n'
    digest=hashlib.sha256(source.encode()).hexdigest()
    if digest in seen:continue
    seen.add(digest)
    key=f'numeric-{base}-{s}-{z}-{binding}-{placement}'
    tests.append((key,source));choices[key]=dict(base=base,seven=s,zero=z,binding=binding,placement=placement)
(OUT/'choices.json').write_text(json.dumps(choices,indent=2)+'\n')
results=[]
with concurrent.futures.ThreadPoolExecutor(max_workers=8) as pool:
    for i,d in enumerate(pool.map(r['run'],tests),1):
        results.append(d)
        if i%128==0 or d.get('exact_candidate'):
            print(json.dumps(dict(done=i,total=len(tests),best=sorted(results,key=lambda x:(x.get('nd',999),-x.get('score',0)))[:2])),flush=True)
(OUT/'results.json').write_text(json.dumps(results,indent=2)+'\n')
print(json.dumps(dict(done=len(results),errors=sum('error'in x for x in results),exact=sum(x.get('exact_candidate',False)for x in results),unique_bodies=len({x.get('masked_code_sha256')for x in results if 'error'not in x}))),flush=True)
