#!/bin/bash
# objscore.sh OBJ -> score an already compiled object against retail
cd /work2/Open-BFME-2
o=$1; f=build/qAf/fx/_objscore_$$.cpp; echo "// placeholder" > $f
obj=$(python3 -c "import sys;sys.path.insert(0,'tools');import build;print(build.obj_path(build.ROOT/'$f'))")
cp "$o" "$obj"
out=$(BFME_OBJSTORE=off python3 tools/align_diff.py "$f" '?queueSupplyTruck@AIPlayer@@IAEXXZ' 0x004F37B0 1537 2>&1)
rm -f "$obj" "$f"
size=$(echo "$out" | rg -o "ours [0-9]+ B" | head -1 | rg -o "[0-9]+")
reg=$(echo "$out" | rg "^\s+[0-9]+\s+register" | awk '{print $1}')
mem=$(echo "$out" | rg "^\s+[0-9]+\s+memory" | awk '{print $1}')
regions=$(echo "$out" | rg -o "structural regions: [0-9]+" | rg -o "[0-9]+$")
echo "$o ${size} r=${regions:-0} m=${mem:-0} g=${reg:-0} score=$(python3 -c "print(round(1-(${regions:-0}+${mem:-0}+${reg:-0})/450,4))")"
[ -n "$KEEP" ] && echo "$out" > "${o%.obj}.align"
