from trials import *
import itertools
baseinc='#include "../../reference/shims/bfme2_ascii/string_base.h"\n'
uinc='#include "../../reference/shims/bfme2_ascii/unicode_string.h"\n'
ainc='#include "../../reference/shims/bfme2_ascii/ascii_string.h"\n'
items=[]
for b,a,eh,temp,reg in itertools.product(range(2),range(2),range(2),range(2),range(2)):
 s=BASE;lo=s.index('template <class T> class StringBase');hi=s.index('#define W(s)')
 ascii_class=s[s.index('class AsciiString : public'):s.index('class UnicodeString : public')]
 # Canonical StringBase + canonical Unicode, either simple or canonical ASCII.
 s=s[:lo]+baseinc+('template <> void StringBase<char>::releaseBuffer() throw();\n' if eh else '')+(ainc if a else ascii_class)+uinc+s[hi:]
 if b:s=s.replace('fontName->isEmpty()','fontName->isNotEmpty()').replace('!fontName->isNotEmpty()','fontName->isNotEmpty()')
 if temp:s=s.replace('AsciiString fixedSys("FixedSys");\n\t\t\tfont = TheFontLibrary->getFont(&fixedSys, 8.0f, false);','font = TheFontLibrary->getFont(&AsciiString("FixedSys"), 8.0f, false);')
 if reg:s=s.replace('GameFont *font;','register GameFont *font;')
 if a:
  s=s.replace('name.set(selObj->m_name);','name = selObj->m_name;').replace('name.set("No-Name");','name = "No-Name";')
 items.append(('canonical%d%d%d%d%d'%(b,a,eh,temp,reg),s))
batch(items,'canonical-complete')
