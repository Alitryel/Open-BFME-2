from trials import *
import itertools,re,collections
names=['fuel','resistance','burnRate','maxBurn','decay']
call='TheTriggerManager->rva002858E5(&world.x, &fuel, &burnRate, &decay, &resistance, &maxBurn)'
char='template <> void StringBase<char>::releaseBuffer() throw();\n'
family=sys.argv[1]
items=[]
literal=lambda s:collections.Counter(re.findall(r'"(?:\\.|[^"\\])*"',s))
if family=='query':
 for p,perm in enumerate(itertools.permutations(names)):
  for byref,release,conv in itertools.product(range(2),range(2),range(2)):
   s=BASE if release else BASE.replace(char,'')
   params=', '.join(('int &' if byref else 'int *')+n for n in perm)
   args=', '.join(('&' if byref else '')+n for n in ['fuel','burnRate','decay','resistance','maxBurn'])
   helper='static __forceinline bool '+('__fastcall ' if conv else '')+'queryFire(Rva002858E5 *manager, float *pos, '+params+') { return manager->rva002858E5(pos, '+args+'); }\n'
   s=s.replace('void W3DDisplay::gatherDebugStats(void)',helper+'void W3DDisplay::gatherDebugStats(void)')
   s=s.replace(call,'queryFire(TheTriggerManager, &world.x, '+', '.join(('' if byref else '&')+n for n in perm)+')')
   assert literal(s)==literal(BASE)
   items.append(('query%03d%d%d%d'%(p,byref,release,conv),s))
if family=='groups':
 # All non-singleton subsets put their outputs in one actual POD record.
 # Preserve each call and formatting argument's identity; no padding or unused fields.
 for mask in range(1,32):
  chosen=[n for i,n in enumerate(names) if mask>>i&1]
  if len(chosen)<2:continue
  for p,perm in enumerate(itertools.permutations(chosen)):
   for release in range(2):
    s=BASE if release else BASE.replace(char,'')
    decl='int fuel, resistance, burnRate, maxBurn, decay;'
    repl='struct FireOutputs { '+''.join('int '+n+'; ' for n in perm)+'} outputs; '
    rest=[n for n in names if n not in chosen]
    if rest:repl+='int '+', '.join(rest)+';'
    start=s.index(decl);end=s.index('\n\t\t}\n\t\telse',start)
    chunk=s[start+len(decl):end]
    # Substitute only C++ identifiers outside string literals.
    chunk=re.sub(r'"(?:\\.|[^"\\])*"|\b(?:'+ '|'.join(chosen)+r')\b',lambda m:m[0] if m[0].startswith('"') else 'outputs.'+m[0],chunk)
    s=s[:start]+repl+chunk+s[end:]
    assert literal(s)==literal(BASE)
    items.append(('group%02d%03d%d'%(mask,p,release),s))
batch(items,'fire-'+family)
