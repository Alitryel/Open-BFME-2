import sys,os
sys.path.insert(0,'/work/Open-BFME-2/build/c2tr')
from wdbg import Dbg
# fp1.py SRC OBJ : trace the ebp-availability flag and 8011c visits per round
src,obj=sys.argv[1:3]; WN=os.environ.get('WN','node,entry').split(',')
d=Dbg(src,obj,obj+'.log')
d.prompt();d.prompt('set $CanDeferOnBPByAddr = 1')
B=0x10700000
bps=[0x83acd,0x82978,0x84791,0xa8f54]; added=set()
for b in bps:
    o=d.prompt('break *0x%x'%(B+b))
    if 'Breakpoint' in o and 'Unable' not in o: added.add(b)
names={}
def nm(sym):
    if sym in names: return names[sym]
    sw=d.words(sym,1); pw=d.words(sw[0],2) if sw and sw[0]>0x100000 else []
    n=d.string(pw[1]).strip() if len(pw)>1 and pw[1]>0x100000 else '?'
    n=n.strip('"').lstrip('_'); names[sym]=n; return n
def members(hdr):
    out=[]; ch=d.words(hdr,1)[0] if hdr else 0
    while ch:
        ww=d.words(ch,3)
        if len(ww)<3: break
        base,nx,bits=ww; out+=[base+j for j in range(32) if bits>>j&1]; ch=nx
    return out
on=False;k=0;nv=0
while True:
    out=d.prompt('cont')
    if 'terminated' in out: break
    eip=d.reg('eip')-B
    for b in bps:
        if b not in added and b!=eip: d.prompt('break *0x%x'%(B+b)); added.add(b)
    added.add(eip)
    if eip==0x83acd:
        ebp=d.reg('ebp'); f=d.words(ebp,1)[0]; curfn=d.string(d.words(f+4,1)[0]); on='updateMeleeAttack' in curfn and d.reg('ebx')==0; k=0; continue
    if not on: continue
    if eip==0x82978:
        k+=1; f30,f28=d.words(0x108a7b28,3)[2],d.words(0x108a7b28,1)[0]
        print('R%d entry flag30=%x flag28=%x visits=%d'%(k,f30,f28,nv),flush=True); nv=0
        if k>12: break
        continue
    if eip==0x84791:
        nv+=1; esi=d.reg('esi'); w=d.words(esi,10); n=nm(w[0])
        if n in WN:
            f30=d.words(0x108a7b30,1)[0]
            print('  8011c R%d %s:%x fl=%x allowed=%s flag30=%x'%(k,n,w[7],w[1],members(w[8]) if w[8] else None,f30),flush=True)
        continue
    if eip==0xa8f54:
        print('  a8f54 R%d'%k,flush=True); continue
print('done',flush=True)
