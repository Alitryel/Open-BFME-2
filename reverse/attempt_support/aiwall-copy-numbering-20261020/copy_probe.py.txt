"""Isolated AIWall trials: non-trivial 2D copies and return-value inlining."""
from concurrent.futures import ThreadPoolExecutor
from itertools import product
import json
from pathlib import Path
import sys

ROOT = Path(__file__).resolve().parents[2]
sys.path.insert(0, str(ROOT / "tools"))
import permute

BASE = (ROOT / "reverse/attempts/0x004eadba.cpp").read_text()
OUT = ROOT / "build/aiwall-probe"
SYMBOL = "?distSqToSegment@@YAMPBVCoord2D@@00@Z"
ENTRY = "\tfloat apx = pt->x - lineStart->x;\n\tfloat apy = pt->y - lineStart->y;"
DOT = "\tfloat dot = apy * dy;\n\tdot += apx * dx;"


def source_for(copy, creation, call, expression, order):
    assignment = ("x = a.x-b.x; y = a.y-b.y;" if order == "xy" else
                  "y = a.y-b.y; x = a.x-b.x;")
    copy_ctor = "" if copy == "implicit" else (
        "PointDelta(const PointDelta &v) { " +
        ("x=v.x; y=v.y;" if copy == "xy" else "y=v.y; x=v.x;") + " }")
    dot_body = ("float q = a.y*dy; q += a.x*dx; return q;" if expression == "accum" else
                "return a.x*dx + a.y*dy;")
    definitions = """
struct PointDelta : Coord2D {
    PointDelta() {}
    COPY_CTOR
    PointDelta(const Coord2D &a, const Coord2D &b) { ASSIGN }
    __forceinline float dot(float dx, float dy) const { MEMBER_DOT }
};
__forceinline PointDelta difference(const Coord2D &a, const Coord2D &b) {
    PointDelta result; RESULT_ASSIGN return result;
}
__forceinline float pointDot(PARAM, float dx, float dy) { DOT_BODY }
""".replace("COPY_CTOR", copy_ctor).replace("ASSIGN", assignment)
    definitions = definitions.replace("MEMBER_DOT", dot_body.replace("a.", ""))
    definitions = definitions.replace("RESULT_" + assignment,
                                      assignment.replace("x =", "result.x =").replace("y =", "result.y ="))
    definitions = definitions.replace("DOT_BODY", dot_body).replace(
        "PARAM", "PointDelta a" if call == "value" else "const PointDelta &a")
    if creation == "assign":
        delta = "\tPointDelta ap; " + assignment.replace("a.", "pt->").replace("b.", "lineStart->").replace(
            "x =", "ap.x =").replace("y =", "ap.y =")
    elif creation == "ctor":
        delta = "\tPointDelta ap(*pt, *lineStart);"
    else:
        delta = "\tPointDelta ap = difference(*pt, *lineStart);"
    expression = "ap.dot(dx, dy)" if call == "member" else "pointDot(ap, dx, dy)"
    source = BASE.replace("static __declspec(noinline) float distSqToSegment", definitions +
                          "\nstatic __declspec(noinline) float distSqToSegment")
    assert ENTRY in source and DOT in source
    return source.replace(ENTRY, delta).replace(DOT, "\tfloat dot = " + expression + ";")


def run_shard(shard):
    directory = ROOT / f"build/aiwall-copy-probe-{shard}"
    directory.mkdir(exist_ok=True)
    scorer = permute.Scorer("0x004EACF5", SYMBOL, 197, directory)
    records = []
    baseline_fit, _ = scorer.score(BASE)
    print(f"shard {shard} baseline fitness={baseline_fit:.8f}", flush=True)
    best = 4
    variants = list(product(("xy", "yx", "implicit"), ("rvo", "ctor", "assign"),
                            ("value", "ref", "member"), ("accum", "sum"), ("xy", "yx")))
    for index, variant in enumerate(variants):
        if index % 2 != shard:
            continue
        text = source_for(*variant)
        fitness, exact = scorer.score(text)
        if fitness >= 0 and scorer.last is not None:
            compiled, target = scorer.last
            ours, theirs = permute._instructions(compiled), permute._instructions(target)
            differences = sum(a != b for a, b in zip(ours, theirs)) + abs(len(ours)-len(theirs))
            record = {"id": index, "variant": variant, "fitness": fitness, "exact": exact,
                      "size": len(compiled), "different_instructions": differences}
            if differences < best or exact:
                best = differences
                (OUT / f"copy-best-{shard}.cpp").write_text(text)
                (OUT / f"copy-best-{shard}.bin").write_bytes(compiled)
                print(json.dumps(record), flush=True)
            if exact:
                (OUT / "win.cpp").write_text(text)
                records.append(record)
                break
        else:
            record = {"id": index, "variant": variant, "fitness": fitness, "exact": exact,
                      "compile_failed": True}
        records.append(record)
    (OUT / f"copy-results-{shard}.json").write_text(json.dumps(records, indent=2))
    print(f"shard {shard}: {len(records)} trials, best instruction differences={best}", flush=True)


if __name__ == "__main__":
    with ThreadPoolExecutor(2) as pool:
        list(pool.map(run_shard, range(2)))
