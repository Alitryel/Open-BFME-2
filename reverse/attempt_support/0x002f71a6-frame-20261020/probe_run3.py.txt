"""Compile probes; print the lea slot immediately preceding each call to extp/extq (first occurrence)."""
import sys, re
from pathlib import Path
sys.path.insert(0, '/work3/Open-BFME-2/tools')
import build, capstone
from locate import object_functions
from concurrent.futures import ThreadPoolExecutor
md = capstone.Cs(capstone.CS_ARCH_X86, capstone.CS_MODE_32)
SYM = sys.argv[1]


def one(src):
    src = Path(src).resolve()
    obj = src.with_suffix('.obj')
    if obj.exists():
        obj.unlink()
    ok, text, code = build.try_compile_source(src, obj)
    if not ok:
        return src.name, 'FAIL'
    for n, span, rl in object_functions(obj):
        if n == SYM:
            names = {o: nm for o, t, nm in rl}
            res = {}
            last = None
            for i in md.disasm(bytes(span), 0):
                if i.mnemonic == 'lea':
                    m = re.search(r'\[(e[bs]p[^\]]*)\]', i.op_str)
                    last = m.group(1).replace(' ', '') if m else None
                if i.mnemonic == 'call':
                    nm = names.get(i.address + 1, '?')
                    if last and nm[:5] in ('?extp', '?extq'):
                        res.setdefault(nm[4], last)
                    last = None
            return src.name, ('q>p' if int(res['q'][4:], 16) < int(res['p'][4:], 16) else 'p>q') + f" p{res.get('p')} q{res.get('q')}"
    return src.name, 'nosym'


with ThreadPoolExecutor(8) as ex:
    for name, r in ex.map(one, sorted(sys.argv[2:])):
        print(name, r, flush=True)
