from pathlib import Path
import sys, json, subprocess, hashlib, concurrent.futures
root=Path(__file__).resolve().parents[1]
sys.path.insert(0,str(root/'tools'))
import build
out=root/'build/zh-prim-animation-realheaders-oct20'
out.mkdir(exist_ok=True)
zh=build.ZH_REFERENCE_ROOT
sdk=root/'reference/open-bfme-1/inputs/toolchains/vs2003/Program Files/Microsoft Visual Studio .NET 2003/Vc7/PlatformSDK/Include'
dx=root/'reference/open-bfme-1/inputs/toolchains/dx81/include'
assert (sdk/'windows.h').is_file() and (dx/'d3d8.h').is_file()
header=zh/'Libraries/Source/WWVegas/WW3D2/prim_anim.h'
old=header.read_text()
needle='const PrimitiveAnimationChannelClass<T>::KeyClass &PrimitiveAnimationChannelClass<T>::Get_Key'
assert old.count(needle)==1 and '#define __PRIM_ANIM_H' in old
overlay=out/'prim_anim_typename.h'
overlay.write_bytes(old.replace(needle,'const typename PrimitiveAnimationChannelClass<T>::KeyClass &PrimitiveAnimationChannelClass<T>::Get_Key').encode())
revision=subprocess.check_output(['git','-C',str(root/'reference/open-bfme-1'),'rev-parse','HEAD'],text=True).strip()
tasks=[]
for name in ('prim_anim.cpp','assetmgr.cpp','ringobj.cpp','sphereobj.cpp'):
    donor=header.parent/name
    reference_dirs=build._reference_include_dirs(zh)
    actual=[f for f in reference_dirs if 'shims/sweep' not in f.replace('\\','/')]
    shim=[f for f in reference_dirs if 'shims/sweep' in f.replace('\\','/')]
    flags=build._ZH_BASE_FLAGS+actual+['-I'+sdk.relative_to(root).as_posix(),'-I'+dx.relative_to(root).as_posix()]+shim
    for profile,extra in [('original',[]),('retail',['-O1','-G7','-arch:SSE2'])]:
        selected=flags+extra
        for flag in selected:
            if flag.startswith('-I'): assert (root/flag[2:]).is_dir(),flag
        def render(flag):return '-I"'+flag[2:]+'"' if flag.startswith('-I') and ' ' in flag else flag
        text='// cl: '+' '.join(map(render,selected))+'\n'
        if build.source_needs_stlport(donor):text+='// stlport\n'
        text+='#include <windows.h>\n#include "'+overlay.as_posix()+'"\n#include "'+donor.as_posix()+'"\n'
        trial=out/(profile+'_'+name)
        trial.write_bytes(text.encode())
        assert build.source_extra_flags(trial)==selected
        tasks.append({'donor':donor.relative_to(root).as_posix(),'donor_sha256':hashlib.sha256(donor.read_bytes()).hexdigest(),'revision':revision,'header':header.relative_to(root).as_posix(),'header_sha256':hashlib.sha256(header.read_bytes()).hexdigest(),'overlay_sha256':hashlib.sha256(overlay.read_bytes()).hexdigest(),'profile':profile,'flags':selected,'trial':trial.relative_to(root).as_posix()})
(out/'inputs.json').write_bytes((json.dumps(tasks,indent=2)+'\n').encode())
def run(task):
    p=subprocess.run([sys.executable,'tools/place_bodies.py','--min','8',task['trial']],cwd=root,capture_output=True,text=True,encoding='utf8',errors='replace',timeout=240)
    output=p.stdout+p.stderr
    (root/task['trial']).with_suffix('.log').write_bytes(output.encode())
    return {**task,'returncode':p.returncode,'compile_failed':p.returncode!=0 or 'COMPILE FAILED' in output,'errors':[s for s in output.splitlines() if 'error ' in s][:8],'leads':[s for s in output.splitlines() if s.startswith(('BODY\t','PIN\t'))]}
results=[]
with concurrent.futures.ThreadPoolExecutor(max_workers=3) as pool:
    for r in pool.map(run,tasks):
        results.append(r)
        (out/'results.json').write_bytes((json.dumps(results,indent=2)+'\n').encode())
        print(Path(r['donor']).name,r['profile'],'failed',r['compile_failed'],'leads',r['leads'],'errors',r['errors'][:2],flush=True)
