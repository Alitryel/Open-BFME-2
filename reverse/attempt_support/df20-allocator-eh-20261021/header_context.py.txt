from pathlib import Path
import sys,itertools,concurrent.futures,json,re
sys.path.insert(0,str(Path(__file__).parent));import trials
trials.scratch=Path(__file__).resolve().parent
base=(trials.scratch/'bank.cpp').read_text();trials.symbol=base.splitlines()[0][3:]
money=Path('vendor/stlport/stl/_monetary.h').read_text()
ios=Path('vendor/stlport/stl/_ios_base.h').read_text()
ctor='template <> _String_base<wchar_t,allocator<wchar_t> >::_String_base(const allocator<wchar_t>&);'
dtor='template <> _String_base<wchar_t, allocator<wchar_t> >::~_String_base();'
marker='locale::~locale() _STLP_NOTHROW { _M_impl->_M_decr(); }'
copy='''
inline locale::locale(const locale& that) {
 _M_impl=0; _Locale_impl* impl=that._M_impl; impl->_M_incr(); _M_impl=impl;
}
__declspec(noinline) inline locale ios_base::getloc() const { return _M_locale; }
'''
facet='''
__declspec(noinline) locale::facet* locale::_M_use_facet(const id& index) const {
 if(index._M_index < _M_impl->_M_count) return _M_impl->_M_facets[index._M_index];
 return 0;
}
'''
specs=[]
for cvsym,cvsign,loccopy,lookup,opaque,visible in itertools.product([0,1],repeat=6):
 label=f'header-c{cvsym}-s{cvsign}-l{loccopy}-f{lookup}-o{opaque}-v{visible}'
 s=base;h=money
 if cvsym:h=re.sub(r'\bstring_type(\s+curr_symbol\(\))',r'const string_type\1',h)
 if cvsign:h=re.sub(r'\bstring_type(\s+(?:positive|negative)_sign\(\))',r'const string_type\1',h)
 d=trials.scratch/label/'stl';d.mkdir(parents=True,exist_ok=True)
 (d/'_monetary.h').write_text(h)
 if loccopy:
  h=ios.replace('locale getloc() const { return _M_locale; }','__declspec(noinline) locale getloc() const;')
  assert h!=ios
  (d/'_ios_base.h').write_text(h)
  s=s.replace(marker,marker+'\n'+copy,1)
 if lookup:s=s.replace(marker,marker+'\n'+facet,1)
 if not opaque:s=s.replace(ctor,'')
 if visible:s=s.replace(dtor,'')
 s=s.replace('// cl:','// cl: /I'+str(d.parent.relative_to(Path.cwd())),1)
 specs.append((label,s))
with concurrent.futures.ThreadPoolExecutor(max_workers=8) as pool:r=list(pool.map(trials.run,specs))
(trials.scratch/'header_context.json').write_text(json.dumps(r,indent=2))
