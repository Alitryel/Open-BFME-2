from pathlib import Path
import capstone,hashlib,json,pefile,re
ROOT=Path.cwd();OUT=ROOT/'build/trust237-round8';OUT.mkdir(exist_ok=True)
p=ROOT/'reference/open-bfme-1/inputs/toolchains/vs2003/Program Files/Microsoft Visual Studio .NET 2003/Vc7/bin/c2.dll'
data=p.read_bytes();sha=hashlib.sha256(data).hexdigest()
assert sha=='bcd28f39b1798b07989c30a1df471cf87fc20e3802d130901a08f83800e6b81e'
pe=pefile.PE(data=data);base=pe.OPTIONAL_HEADER.ImageBase
md=capstone.Cs(capstone.CS_ARCH_X86,capstone.CS_MODE_32);md.detail=True
raw_sites=[];candidates=[]
# Raw immediate candidates are only leads. A decode can begin inside another
# instruction's displacement; the saved bounded contexts distinguish this.
for section in pe.sections:
    if not section.Characteristics&0x20000000:continue
    for match in re.finditer(re.escape(b'\xa9\0\0\0'),section.get_data()):
        va=base+section.VirtualAddress+match.start();raw_sites.append(hex(va))
        for start in range(va-10,va+1):
            ins=list(md.disasm(pe.get_data(start-base,va+4-start),start))
            if len(ins)!=1 or ins[0].address+ins[0].size!=va+4:continue
            i=ins[0]
            if any(o.type==capstone.x86.X86_OP_IMM and o.imm==0xa9 for o in i.operands):
                candidates.append(dict(va=hex(start),raw=i.bytes.hex(),mnemonic=i.mnemonic,operands=i.op_str))
contexts={}
for lo,hi in [(0x10736e5d,0x10736e7d),(0x107db0e9,0x107db101),
              (0x10801cd4,0x10801cf2),(0x1084d8e4,0x1084d901)]:
    contexts[hex(lo)]=[dict(va=hex(i.address),raw=i.bytes.hex(),mnemonic=i.mnemonic,operands=i.op_str)
                       for i in md.disasm(pe.get_data(lo-base,hi-lo),lo)]
compiler=dict(path=str(p.relative_to(ROOT)),sha256=sha,raw_a9_immediates=raw_sites,
              possible_decodings=candidates,bounded_contexts=contexts,
              diagnostic_path=pe.get_string_at_rva(0x1087b774-base).decode())
(OUT/'compiler-a9-leads.json').write_text(json.dumps(compiler,indent=2)+'\n')
images=[]
for path in [ROOT/'baselines/bfme2/workshop-vanilla-1.06/files/game.dat',
             ROOT/'reference/open-bfme-1/inputs/baselines/bfme1/workshop-vanilla-1.03/files/lotrbfme.exe',
             ROOT/'reference/open-bfme-1/inputs/baselines/bfme1/retail-1.03-unpacked/files/lotrbfme.exe']:
    data=path.read_bytes();pe=pefile.PE(data=data)
    row=dict(path=str(path.relative_to(ROOT)),sha256=hashlib.sha256(data).hexdigest(),
             base=hex(pe.OPTIONAL_HEADER.ImageBase),imports=[],guid_word_sites=[])
    for dll in getattr(pe,'DIRECTORY_ENTRY_IMPORT',[]):
        for imp in dll.imports:
            if imp.name not in [b'WinVerifyTrust',b'CommandLineToArgvW',b'GetCommandLineW',b'LocalFree']:continue
            sites=[]
            for opcode in [b'\xff\x15',b'\xff\x25']:
                pattern=opcode+imp.address.to_bytes(4,'little')
                for match in re.finditer(re.escape(pattern),data):
                    sites.append(dict(rva=hex(pe.get_rva_from_offset(match.start())),raw=pattern.hex()))
            row['imports'].append(dict(dll=dll.dll.decode(),name=imp.name.decode(),iat=hex(imp.address),sites=sites))
    for section in pe.sections:
        if not section.Characteristics&0x20000000:continue
        for match in re.finditer(re.escape(bytes.fromhex('6bc5aa00')),section.get_data()):
            rva=section.VirtualAddress+match.start()
            row['guid_word_sites'].append(dict(rva=hex(rva),context=pe.get_data(rva-8,44).hex()))
    images.append(row)
(OUT/'game-image-probe.json').write_text(json.dumps(images,indent=2)+'\n')
print('Compiler raw A9 patterns',len(raw_sites),'possible decodings',len(candidates))
for row in images:print(row['path'],'GUID-word sites',len(row['guid_word_sites']),
                        'matching imports',[i['name'] for i in row['imports']])
