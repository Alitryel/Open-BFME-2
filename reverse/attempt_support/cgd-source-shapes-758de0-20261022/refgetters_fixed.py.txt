from pathlib import Path
import sys,itertools
sys.path.insert(0,str(Path(__file__).resolve().parent))
import search
b=search.DIR.joinpath('base.cpp').read_text();D=search.DIR/'refgetters_fixed';D.mkdir(exist_ok=True)
for n,choices in enumerate(itertools.product(range(4),range(4),range(4),range(4),range(4))):
 ret,field,method,temps,cv=choices
 source=['ContactSource *','ContactSource *const &','ContactSource *&','ContactSource *volatile &'][ret]
 data=['ContactData *','ContactData *const &','ContactData *&','ContactData *volatile &'][ret]
 s=b.replace('ContactSource *source;','ContactSource *source; '+source+' getSource() { return source; }')
 s=s.replace('ContactData *first,*second;','ContactData *first,*second; '+data+' getFirst() { return first; } '+data+' getSecond() { return second; }')
 if cv&1:s=s.replace('getSource() {','getSource() const {').replace('getFirst() {','getFirst() const {').replace('getSecond() {','getSecond() const {')
 if cv&2:s=s.replace('getSource()','__fastcall getSource()').replace('getFirst()','__fastcall getFirst()').replace('getSecond()','__fastcall getSecond()')
 # replace calls only in body
 h,t=s.split('void BfmeThingCGD::bfmeTwoCGD() {')
 for member in ['first','second']:
  old='node->'+member+'->source';get='node->get'+member.capitalize()+'()->getSource()'
  if field==1:get='node->'+member+'->getSource()'
  if field==2:get='node->get'+member.capitalize()+'()->source'
  if field==3:get='node->get'+member.capitalize()+'()->getSource()'
  if field==0:get=old
  if method==1:t=t.replace(old+'->prepare',get+'->prepare')
  elif method==2:t=t.replace(old+'->respond',get+'->respond').replace(old+'->test',get+'->test')
  elif method==3:t=t.replace(old,get)
  else:t=t.replace(old+'==0',get+'==0')
 if temps&1:t=t.replace('ContactNode *node=current;','ContactNode *saved=current; ContactNode *const &node=saved;') # reference denotes an independent saved cursor value
 if temps&2:t=t.replace('node->point','*point').replace('node->normal','*normal').replace('if(node->state==0)','const ContactVector *point=&node->point,*normal=&node->normal;\n  if(node->state==0)')
 s=h+'void BfmeThingCGD::bfmeTwoCGD() {'+t
 (D/('g%04d.cpp'%n)).write_text(s)
