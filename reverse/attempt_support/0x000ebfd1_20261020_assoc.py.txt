"""Symbolically execute the world path through postMul/translate and print each sum's association."""
import sys, re
from pathlib import Path
ROOT = Path(__file__).resolve().parents[2]
sys.path.insert(0, str(ROOT / "tools"))
import build, capstone
from locate import object_functions
SYM = '?rva000EBFD1@Rva000EC9C6@@QAEXXZ'
md = capstone.Cs(capstone.CS_ARCH_X86, capstone.CS_MODE_32)
NAMES = {}
for k in range(12):
    NAMES[0x650 + 4 * k] = f"W{k // 4}{k % 4}"   # world matrix (merge input)
    NAMES[0x5d0 + 4 * k] = f"T{k // 4}{k % 4}"
NAMES[0x5cc] = "s"
OFF = {0x4455c: 'oX', 0x44560: 'oY', 0x44564: 'oZ'}

def run(data):
    insns = list(md.disasm(data, 0))
    # start at the world block: first 'movss xmm2, dword ptr [eax + 0x650]'
    start = next(i for i, x in enumerate(insns) if x.mnemonic == 'movss' and x.op_str.endswith('[eax + 0x650]'))
    reg = {'xmm1': 's'}
    mem = {}
    edi = None
    outs = []
    for x in insns[start:]:
        m, o = x.mnemonic, x.op_str
        if m == 'lea' and o.startswith('edi, [eax + '):
            edi = int(re.search(r'0x([0-9a-f]+)', o).group(1), 16); continue
        if m.startswith('j') or m in ('imul', 'add', 'mov', 'cmp', 'shl', 'and', 'or', 'dec', 'xor', 'lea', 'push', 'call', 'sar', 'inc'):
            if m == 'cmp' and o.startswith('edi, ebx'):
                break
            continue
        dst, src = [p.strip() for p in o.split(',', 1)]
        def val(p):
            if p.startswith('xmm'):
                return reg.get(p, '?' + p)
            mm = re.search(r'\[(\w+)(?: ([+-]) (0x[0-9a-f]+|\d+))?\]', p)
            base, sign, disp = mm.group(1), mm.group(2), mm.group(3)
            d = int(disp, 16) if disp and disp.startswith('0x') else int(disp or 0)
            if sign == '-': d = -d
            if base == 'eax': return NAMES.get(d, f"e{d:x}")
            if base == 'edi': return NAMES.get(edi + d, f"e{edi + d:x}")
            if base == 'ecx': return OFF.get(d, f"c{d:x}")
            if base == 'ebp': return mem.get(d, f"slot{d}")
            return p
        if m in ('movss', 'movaps'):
            if dst.startswith('xmm'):
                reg[dst] = val(src)
            else:
                mm = re.search(r'ebp - (0x[0-9a-f]+)', dst)
                if mm:
                    mem[-int(mm.group(1), 16)] = val(src)
        elif m in ('mulss', 'addss'):
            op = '*' if m == 'mulss' else '+'
            reg[dst] = f"({val(dst)}{op}{val(src)})"
    return reg, mem

def canon_products(e):
    return e

def report(tag, data):
    reg, mem = run(data)
    vals = sorted(set(list(reg.values()) + list(mem.values())), key=len, reverse=True)
    print(f"== {tag}")
    for v in vals[:16]:
        print("  ", v)

if __name__ == '__main__':
    report('retail', build.read_target_bytes(0xEBFD1, 1710))
    for p in sys.argv[1:]:
        obj = build.obj_path(ROOT / p)
        if not obj.exists():
            build.compile_source(ROOT / p, obj)
        em = {n: s for n, s, r in object_functions(obj)}
        report(Path(p).stem, em[SYM].rstrip(b"\xcc"))
