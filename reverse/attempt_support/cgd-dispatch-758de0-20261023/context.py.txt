from pathlib import Path
from itertools import product
W=Path(__file__).resolve().parent
b=(W/'bank.cpp').read_text()
D=W/'context';D.mkdir(exist_ok=True)
patterns=[
 'return x+y+z;',
 'return x*y+y*z+z*x;',
 'if(x) return *y; return z;',
 'return p[x]+p[y]+p[z];',
 'while(x--) { y=y*3+x; z^=y;} return y+z;',
 'return f(x)+f(y)+f(z);',
 'return f(x,f(y,z));',
 'return a->f(b->f(x,y),z);'
]
for i,(pat,count,where,inline) in enumerate(product(range(8),[1,2,3,4,7,8,16,31],range(2),range(2))):
 defs=''
 for k in range(count):
  name='ctx_'+str(k)
  prefix='__forceinline' if inline else '__declspec(noinline)'
  if pat<2:decl='int x,int y,int z'
  elif pat==2:decl='int x,int *y,int z'
  elif pat==3:decl='int *p,int x,int y,int z'
  elif pat==4:decl='int x,int y,int z'
  elif pat==5:decl='int (*f)(int),int x,int y,int z'
  elif pat==6:decl='int (*f)(int,int),int x,int y,int z'
  else:
   if k==0:defs+='struct CtxIface { virtual int f(int,int)=0; };\n'
   decl='CtxIface *a,CtxIface *b,int x,int y,int z'
  defs+=prefix+' int '+name+'('+decl+') {'+patterns[pat]+'}\n'
  args=['x,y,z','x,y,z','x,y,z','p,x,y,z','x,y,z','f,x,y,z','f,x,y,z','a,b,x,y,z'][pat]
  if inline:defs+='int use_'+name+'('+decl+') {return '+name+'('+args+');}\n'
 s=(defs+b) if where==0 else (b+defs)
 (D/('v%04d.cpp'%i)).write_text(s)
print('generated',i+1)
