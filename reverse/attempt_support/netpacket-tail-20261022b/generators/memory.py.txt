import sys,itertools,re
sys.path.insert(0,'build/netpacket-20261022b');from search import *
SIGN='NetCommandList *NetPacket::getCommandList()\n';H,B=BANK.split(SIGN,1)
def single(b):
 b=b.replace('\tInt offset;\n','').replace('\t\t\toffset = i;\n','').replace('\t\t\ti = offset;\n','')
 return re.sub(r'\boffset\b','i',b)
DECL='extern "C" void *__cdecl memcpy(void *dest, const void *src, unsigned int count);'
DEF='extern "C" __declspec(noinline) void *__cdecl memcpy(void *dest, const void *src, unsigned int count) SPEC { unsigned char *d=(unsigned char *)dest; const unsigned char *s=(const unsigned char *)src; for(unsigned int j=0;j<count;++j)d[j]=s[j];return dest; }\n'
items=[]
for si,vis,prag,flags,width in itertools.product(range(2),range(4),range(3),range(2),range(2)):
 h=H;b=single(B) if si else B
 if vis==1:h=h.replace(DECL,DECL.replace('void *__cdecl','__declspec(noinline) void *__cdecl'))
 if vis>=2:h=h.replace(DECL,DEF.replace('SPEC','throw()' if vis==3 else ''))
 h+=['','#pragma intrinsic(memcpy)\n','#pragma intrinsic(memcpy)\n#pragma function(memcpy)\n'][prag]
 if flags:h=h.replace('/O1 ', '/O1 /Oi ')
 if width:b=b.replace('UnsignedInt commandID = 0;','UnsignedShort commandID = 0;')
 items.append((f'mem_{si}{vis}{prag}{flags}{width}',h+SIGN+b))
batch(items,'visible-memcpy')
