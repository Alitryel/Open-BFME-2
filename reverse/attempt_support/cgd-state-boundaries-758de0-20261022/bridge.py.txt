from pathlib import Path
from itertools import product,permutations
W=Path(__file__).resolve().parent
out=W/'bridge';out.mkdir(exist_ok=True)
orders=list(permutations('spn'))
for idx,(receiver,order,types,cc,body,sites,bank) in enumerate(product(range(6),range(6),range(4),range(3),range(2),range(8),range(2))):
 s=(W/('diagnostic.cpp' if bank else 'base.cpp')).read_text()
 conv=['__cdecl ','__fastcall ','__stdcall '][cc]
 st=['ContactSource *s','ContactSource *const &s','ContactSource *const *s','ContactData *s','ContactData *const &s','ContactData *const *s'][receiver]
 src=['s','s','(*s)','s->source','s->source','(*s)->source'][receiver]
 params={'s':st,'p':('const ContactVector *p' if types&1 else 'const ContactVector &p'),'n':('const ContactVector *n' if types&2 else 'const ContactVector &n')}
 pv='*p' if types&1 else 'p';nv='*n' if types&2 else 'n'
 defs=''
 for name,alt in [('bridge',False),('reverseBridge',bool(bank))]:
  target=('altPrepare('+src+','+pv+','+nv+')') if alt else src+'->prepare('+pv+','+nv+')'
  content='void *v='+target+'; return v;' if body else 'return '+target+';'
  defs+='__forceinline void *'+conv+name+'('+','.join(params[k] for k in orders[order])+') { '+content+' }\n'
 # Helpers need the completed ContactData; place them after ContactNode.
 s=s.replace('class BfmeThingCGD',defs+'class BfmeThingCGD',1)
 calls=[('second','node->normal'),('second','node->normal'),('first','ContactVector(node->normal).negate()')]
 pos=0
 for i,(side,n) in enumerate(calls):
  source='node->'+side+'->source'
  old=('altPrepare('+source+',node->point,'+n+')') if bank and i==2 else source+'->prepare(node->point,'+n+')'
  at=s.find(old,pos);assert at>=0,(idx,i)
  if sites&(1<<i):
   sv=[source,source,'&('+source+')','node->'+side,'node->'+side,'&(node->'+side+')'][receiver]
   vals={'s':sv,'p':'&(node->point)' if types&1 else 'node->point','n':'&('+n+')' if types&2 else n}
   name='reverseBridge' if i==2 else 'bridge'
   new=name+'('+','.join(vals[k] for k in orders[order])+')'
  else:new=old
  s=s[:at]+new+s[at+len(old):];pos=at+len(new)
 (out/('v%05d.cpp'%idx)).write_text(s)
print('generated',idx+1)
