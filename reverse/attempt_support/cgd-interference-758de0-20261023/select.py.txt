import json,sys
from pathlib import Path
from wdbg import Dbg
W=Path(__file__).resolve().parent;tag=sys.argv[1]
d=Dbg(W/(tag+'.cpp'),W/(tag+'-select.obj'),W/(tag+'-select.log'))
events=[];active=False;finished=False

def members(s):
 out=[];seen=set();p=d.words(s,1)[0] if s else 0
 while p and p not in seen:
  seen.add(p);w=d.words(p,3)
  for i in range(32):
   if w[2]>>i&1:out.append(w[0]+i)
  p=w[1]
 return out

def row(p):
 w=d.words(p,18);sw=d.words(w[0],24)
 return dict(p=p,w=w,sym=sw,allowed=members(w[8]) if w[8] else [])
try:
 d.prompt();d.prompt('set $CanDeferOnBPByAddr = 1');d.prompt('break *0x107821a7')
 for _ in range(600):
  out=d.prompt('cont',60)
  if 'terminated' in out:finished=True;break
  e=d.reg('eip')
  if e==0x107821a7:
   f=d.words(d.reg('esp')+4,1)[0];fw=d.words(f,40);fn=d.string(d.words(fw[0]+4,1)[0])
   if '?bfmeTwoCGD@' not in fn:continue
   active=True;d.prompt('delete 1')
   for a in (0x107833f2,0x107834bd,0x107b0037):d.prompt('break *0x%x'%a)
   continue
  if not active:continue
  if e==0x107b0037:
   for i in (2,3,4):d.prompt('delete '+str(i))
   break
  if e==0x107833f2:
   p=d.words(d.reg('esp')+4,1)[0];r=row(p);ids=members(d.reg('edx'));neighbors=[]
   for idx in ids:
    a=d.words(0x108807b8+4*(idx&1023),1)[0]
    while a:
     w=d.words(a,18)
     if w[7]==idx:neighbors.append(row(a));break
     a=w[12]
   ev=dict(stage='entry',row=r,neighbors=neighbors,indices=ids);events.append(ev)
   print('entry',r['sym'][7],r['w'][7],r['allowed'],[(n['sym'][7],n['w'][7],hex(n['w'][4])) for n in neighbors],flush=True)
  elif e==0x107834bd:
   r=row(d.reg('eax'));ev=dict(stage='selected',row=r,reg=d.reg('ebx'),costs=d.words(0x10880680,16));events.append(ev)
   print('selected',r['sym'][7],r['w'][7],ev['reg'],ev['costs'][:9],flush=True)
 if not finished:
  for _ in range(300):
   out=d.prompt('cont',60)
   if 'terminated' in out:finished=True;print('compiler terminated',flush=True);break
  else:raise RuntimeError('did not finish')
finally:
 (W/(tag+'-select.json')).write_text(json.dumps(events,indent=1))
 if d.p.poll() is None:
  try:d.prompt('quit')
  except Exception:d.p.kill()
 d.log.close()
