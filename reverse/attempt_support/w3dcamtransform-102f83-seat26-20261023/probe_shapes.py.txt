from pathlib import Path
import search as h
import subprocess,json,hashlib
p=Path(__file__).parent
base=(p/'probe.cpp').read_text().split('static __forceinline')[0]
shapes={
 'direct':'s.X+=v->pos.x;s.Y+=v->pos.y;t.X+=v->pos.x;t.Y+=v->pos.y;',
 'loop':'for(int i=0;i<2;++i){float &a=i?s.Y:s.X;const float &b=i?v->pos.y:v->pos.x;a+=b;}for(int j=0;j<2;++j){float &a=j?t.Y:t.X;const float &b=j?v->pos.y:v->pos.x;a+=b;}',
 'loopcopy':'s=loopTranslation(s,v);t=loopTranslation(t,v);',
 'onefield':'s.X=sum1(s.X,v->pos.x).value;s.Y=sum1(s.Y,v->pos.y).value;t.X=sum1(t.X,v->pos.x).value;t.Y=sum1(t.Y,v->pos.y).value;',
 'onefieldcopy':'s.X=sum2(s.X,v->pos.x).value;s.Y=sum2(s.Y,v->pos.y).value;t.X=sum2(t.X,v->pos.x).value;t.Y=sum2(t.Y,v->pos.y).value;',
 'onefieldparam':'s.X=sum3(Scalar2(s.X),v->pos.x).value;s.Y=sum3(Scalar2(s.Y),v->pos.y).value;t.X=sum3(Scalar2(t.X),v->pos.x).value;t.Y=sum3(Scalar2(t.Y),v->pos.y).value;',
 'onefieldlocal':'Scalar2 value;value=sum3(Scalar2(s.X),v->pos.x);s.X=value.value;value=sum3(Scalar2(s.Y),v->pos.y);s.Y=value.value;value=sum3(Scalar2(t.X),v->pos.x);t.X=value.value;value=sum3(Scalar2(t.Y),v->pos.y);t.Y=value.value;',
}
base+='''
struct Scalar1 {float value;Scalar1(float x):value(x){}};
struct Scalar2 {float value;Scalar2(){}Scalar2(float x):value(x){}Scalar2(const Scalar2 &x){value=x.value;}Scalar2 &operator=(const Scalar2 &x){value=x.value;return *this;}};
static __forceinline Scalar1 sum1(float a,float b){return Scalar1(a+b);}
static __forceinline Scalar2 sum2(float a,float b){return Scalar2(a+b);}
static __forceinline Scalar2 sum3(Scalar2 a,float b){a.value+=b;return a;}
static __forceinline Vector3 loopTranslation(Vector3 p,const Context *v){for(int i=0;i<2;++i){float &a=i?p.Y:p.X;const float &b=i?v->pos.y:v->pos.x;a+=b;}return p;}
'''
for name,body in shapes.items():base+='extern "C" void probe_'+name+'(const Context *v){Vector3 s,t;touch(&s,&t);'+body+'touch(&s,&t);}\n'
cpp=p/'probe_shapes.cpp';obj=cpp.with_suffix('.obj');cpp.write_text(base);cmd,env=h.build.compiler_command(cpp.resolve(),obj.resolve());r=subprocess.run(cmd,cwd=h.ROOT,env=env,capture_output=True,text=True);assert r.returncode==0,r.stdout+r.stderr
out=[]
for name in shapes:
 raw,rels=h.build.read_object_symbol_bytes(obj,'_probe_'+name,code_only=True);ins=h.insns(raw,rels);out.append(dict(name=name,size=len(raw),sha256=hashlib.sha256(raw).hexdigest(),instructions=ins));(p/('probe_'+name+'.dis')).write_text('\n'.join(f'{a:04x} {m} {o}' for a,m,o,z in ins))
(p/'probe_shapes-results.json').write_text(json.dumps(out,indent=2));print([(r['name'],r['size'],r['sha256'][:12]) for r in out])
