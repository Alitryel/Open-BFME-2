from pathlib import Path
import datetime, hashlib, json, os, subprocess, sys, time
ROOT=Path.cwd()
OUT=ROOT/'build/resume-eh-local/turn71-reference'
OUT.mkdir(exist_ok=True)
SUB=ROOT/'reference/open-bfme-1'
OLD='f5140c1cff6a58e16276fd6de6788186286953d4'
NEW='1d5cd9725d113c0a58b58f09f282340d8e7b26a7'
sys.path.insert(0,str(ROOT/'tools'))
import build
def git(*args,cwd=ROOT):
    return subprocess.check_output(['git',*args],cwd=cwd).decode().strip()
def utc(): return datetime.datetime.now(datetime.timezone.utc).isoformat()
def fingerprint():
    return hashlib.sha256(git('ls-tree','-r','HEAD','Code','tools','.githooks','reverse/functions.csv','reverse/symbols.csv','reverse/data_rows.csv').encode()).hexdigest()
label=sys.argv[1]
assert label in ('old','new')
expected=OLD if label=='old' else NEW
assert git('rev-parse','HEAD',cwd=SUB)==expected
assert git('status','--porcelain',cwd=SUB)==''
changed=git('diff','--name-only',OLD,NEW,cwd=SUB).splitlines()
layouts={k:dict(path=build.bfme1_subtree(k),exists=build.bfme1_path(k).is_dir()) for k in ('game','reference','vendor','toolchains','baselines','ledger')}
assert all(v['exists'] for v in layouts.values())
paths=['inputs/reference/CnC_Generals_Zero_Hour/GeneralsMD/Code/Libraries/Source/WWVegas/WWMath/vector3.h','inputs/reference/CnC_Generals_Zero_Hour/GeneralsMD/Code/GameEngine/Include/Common/AsciiString.h','inputs/toolchains/vs2003/Program Files/Microsoft Visual Studio .NET 2003/Vc7/bin/c2.dll','tools/lift_lane.py']
inputs={}
for p in paths:
    content=(SUB/p).read_bytes()
    before=subprocess.check_output(['git','show',OLD+':'+p],cwd=SUB)
    after=subprocess.check_output(['git','show',NEW+':'+p],cwd=SUB)
    assert before==after,p
    assert content==before or (p.endswith(('.h','.py')) and content.replace(b'\r\n',b'\n')==before.replace(b'\r\n',b'\n')),p
    inputs[p]=dict(exists=True,checkout_sha256=hashlib.sha256(content).hexdigest(),git_blob_sha256=hashlib.sha256(before).hexdigest(),checkout_crlf_conversion=content!=before)
record=dict(started=utc(),bfme2=git('rev-parse','HEAD'),donor=expected,old=OLD,new=NEW,production_fingerprint=fingerprint(),changed_files=changed,layouts=layouts,intended_inputs=inputs,steps=[])
def save(): (OUT/(label+'-receipt.json')).write_text(json.dumps(record,indent=2)+'\n')
save()
commands=[('full',['C:/Program Files/Git/bin/bash.exe','./build.sh']),('eh',['py','-3','tools/eh_verify.py','--all','--json',str(OUT/(label+'-eh.json'))])]
for name,command in commands:
    log=OUT/(label+'-'+name+'.txt');start=time.monotonic()
    step=dict(name=name,started=utc(),command=command);record['steps'].append(step);save()
    with log.open('wb') as output:
        cp=subprocess.run(command,stdout=output,stderr=subprocess.STDOUT,env={**os.environ,'BUILD_POOL':'2','PYTHONUNBUFFERED':'1'})
    step.update(exit=cp.returncode,elapsed_seconds=time.monotonic()-start,finished=utc(),log_sha256=hashlib.sha256(log.read_bytes()).hexdigest())
    save()
    if cp.returncode: sys.exit(cp.returncode)
assert record['production_fingerprint']==fingerprint()
record['finished']=utc();save()
print(json.dumps(record),flush=True)
