import sys,itertools,json,time
from pathlib import Path
from concurrent.futures import ThreadPoolExecutor,as_completed
sys.path.insert(0,'build/fxinit-seat22');import search as h
h.HERE=Path('build/fxinit-edges22').resolve()
CONDS=['positions[1]','normals[1]','tangents','binormals','basis','m_FVF&0x10','m_BonesPerVertex==2','m_Passes==1']
def variants():
 for a,b,group,shape,invert in itertools.product(range(8),range(8),range(3),range(3),range(2)):
  c=CONDS[a];d=CONDS[b];part=[h.P,h.N,h.P+'\n'+h.N][group]
  if invert:c='!('+c+')';d='!('+d+')'
  if shape==0:code=f'if({c}) goto SharedCopy; if({d}) {{ SharedCopy: {part} }} else {{ {part} }}'
  elif shape==1:code=f'if({c}) goto SharedCopy; if({d}) {{ {part} }} else {{ SharedCopy: {part} }}'
  else:code=f'if({c}) {{ if({d}) goto SharedCopy; {part} }} else {{ SharedCopy: {part} }}'
  src=h.BASE.replace(part,code)
  yield f'sharedcopy_{a}{b}{group}{shape}{invert}',src
if __name__=='__main__':
 t=time.time();items=list(variants());r=[]
 with ThreadPoolExecutor(max_workers=8)as pool:
  for f in as_completed([pool.submit(h.run_one,x)for x in items]):
   r.append(f.result())
   if len(r)%256==0:print('tested',len(r),'best',min((x for x in r if 'nd'in x),key=lambda x:(x['nd'],x['cost'])),flush=True)
 r.sort(key=lambda x:(x.get('nd',99999),x.get('cost',99999)))
 (h.HERE/'sharedcopy-results.json').write_text(json.dumps(r,indent=2))
 print(json.dumps({'count':len(r),'valid':sum('nd'in x and not x.get('unresolved')for x in r),'seconds':time.time()-t,'best':r[:4]},indent=2))
