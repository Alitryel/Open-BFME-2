import itertools
# row-loop shapes per pass x row/col sharing x front split
H='// cl: /O2 /G6 /DNDEBUG /MD\nextern "C" void DiagonalBlur(void *instance, unsigned char *source, unsigned char *dest, int pitch)\n{\n'
NEAR='src[col - pitch - 1] + src[col + pitch - 1] + src[col - pitch + 1] + src[col + pitch + 1]'
FAR='src[col - 2 * pitch - 2] + src[col + 2 * pitch + 2] + src[col + 2 * pitch - 2] + src[col - 2 * pitch + 2]'
NB='src[col - 1] - src[col - pitch] - src[col + pitch] - src[col + 1]'
CL='\t\t\tif (value < 0)\n\t\t\t\tvalue = 0;\n\t\t\telse if (value > 255)\n\t\t\t\tvalue = 255;\n'
def rowloop(k,r,body,decl):
    d='int ' if decl else ''
    if k==0: return '\tfor (%s%s = 0; %s < 8; %s++) {\n%s\t}\n'%(d,r,r,r,body)
    if k==1: return '\tfor (%s%s = 8; %s > 0; %s--) {\n%s\t}\n'%(d,r,r,r,body)
    if k==2: return '\tfor (%s%s = 7; %s >= 0; %s--) {\n%s\t}\n'%(d,r,r,r,body)
    if k==3: return '\t%s%s = 8;\n\twhile (%s--) {\n%s\t}\n'%(d,r,r,body)
    if k==4: return '\t%s%s = 0;\n\tdo {\n%s\t} while (++%s < 8);\n'%(d,r,body,r)
    if k==5: return '\tfor (%s%s = 0; %s != 8; %s++) {\n%s\t}\n'%(d,r,r,r,body)
def gen(k1,k2,rs,cs,fr):
    rows=['row','row2'] if rs==1 else ['row','row']
    cols=['col','col2'] if cs==1 else ['col','col']
    fs=list(dict.fromkeys(rows+cols))
    s=H+'\tunsigned char *src = source;\n\tunsigned char *dst = dest;\n\tint %s;\n'%', '.join(fs)
    c=cols[0]
    b1='\t\tfor (%s = 0; %s < 8; %s++) {\n'%(c,c,c)
    if fr==2: b1+='\t\t\tint sum = (16 + 8 * src[col] + 4 * (%s) + 2 * (%s)) >> 5;\n\t\t\tdst[col] = (unsigned char)sum;\n'%(NEAR,FAR)
    else: b1+='\t\t\tdst[col] = (unsigned char)((16 + 8 * src[col] + 4 * (%s) + 2 * (%s)) >> 5);\n'%(NEAR,FAR)
    b1=b1.replace('col',c)+'\t\t}\n\t\tdst += pitch;\n\t\tsrc += pitch;\n'
    c=cols[1]
    b2='\t\tfor (col = 0; col < 8; col++) {\n'
    C='src[col]'
    if fr==1: b2+='\t\t\tint center = src[col];\n'; C='center'
    b2+='\t\t\tint value = (1 + 6 * %s - %s) >> 1;\n'%(C,NB)+CL+'\t\t\tdst[col] = (unsigned char)value;\n'
    b2=b2.replace('col',c)+'\t\t}\n\t\tdst += pitch;\n\t\tsrc += pitch;\n'
    s+=rowloop(k1,rows[0],b1,False)+rowloop(k2,rows[1],b2,False)+'}\n'
    return s
for o in itertools.product(range(6),range(6),range(2),range(2),range(3)):
    open('g13/r%d%d%d%d%d.cpp'%o,'w').write(gen(*o))
