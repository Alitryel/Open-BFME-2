from pathlib import Path
import sys,json,itertools
ROOT=Path(__file__).resolve().parents[2]
sys.path.insert(0,str(ROOT/'tools'))
from variant_search import Scorer
out=Path(__file__).parent/'scopes'
s=Scorer('?distSqToSegment@@YAMPBVCoord2D@@00@Z',0x4eacf5,197,out)
rows=[]
for baseidx,scope,init,early in itertools.product((18,30),range(2),range(2),range(2)):
 text=(Path(__file__).parent/f'partial-coords/c{baseidx}.cpp').read_text()
 text=text.replace('\tCoord2D ap;\n','')
 start=text.index('\tap.x = pt->x - lineStart->x;')
 end=text.index('\n\tif (dot <= 0.0f)',start)
 chunk=text[start:end]
 chunk=chunk.replace('float dot = ','dot = ')
 pre='\tfloat dot'+(' = 0.0f' if init else '')+';\n'
 prefix=('\t{\n' if scope else '')+'\tCoord2D ap;\n'
 post=('\n\t}' if scope else '')
 text=text[:start]+prefix+chunk+post+text[end:]
 if early:
  pos=text.index('\n{\n',text.index('float distSqToSegment'))+3
  text=text[:pos]+pre+text[pos:]
 else:
  pos=text.index(prefix,text.index('float distSqToSegment'))
  text=text[:pos]+pre+text[pos:]
 r=s(text);got=r.get('compiled',b'');target=r.get('target',b'')
 row={'forms':[baseidx,scope,init,early],'file':f'c{s.n}.cpp','size':len(got),'diff':sum(a!=b for a,b in zip(got,target))+abs(len(got)-len(target)),'exact':r['exact'],'fitness':r['fitness'],'class':r['class']}
 rows.append(row)
 if s.n==1 or r['exact'] or row['diff']<12:print(json.dumps(row),flush=True)
 if r['exact']:(out/'win.cpp').write_bytes(text.encode());break
(out/'results.json').write_text(json.dumps(rows,indent=2))
print('completed',len(rows),'best',min(rows,key=lambda r:(not r['exact'],-r['fitness'])),flush=True)
