"""Print retail (target) or ours disassembly in a hex range: tdis.py file.cpp lo hi [o]"""
import sys
sys.path.insert(0, '/work3/Open-BFME-2/build/h72')
from score import compile_, md, norm
f = sys.argv[1]; lo = int(sys.argv[2], 16); hi = int(sys.argv[3], 16)
(got, raw), _ = compile_(f)
b = got['bytes'] if len(sys.argv) > 4 else got['target']
for i in md.disasm(bytes(b), 0):
    if lo <= i.address < hi:
        print(f'{i.address:4x} {i.mnemonic} {i.op_str}')
