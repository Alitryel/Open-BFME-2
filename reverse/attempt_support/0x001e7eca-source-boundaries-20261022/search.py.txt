import os, sys, json, difflib, subprocess, concurrent.futures, hashlib, re
from pathlib import Path
sys.path.insert(0,str(Path('tools').resolve()))
import build,capstone
ROOT=Path.cwd(); W=ROOT/'build/locomotor-zero-cfg'
SYM='?rva001E7ECA@Locomotor@@QAE_NPAVObject@@PBUCoord3D@@@Z'
MD=capstone.Cs(capstone.CS_ARCH_X86,capstone.CS_MODE_32)
TARGET=(W/'retail.bin').read_bytes()[:1527]
B=[(x.address,x.mnemonic,x.op_str,x.size) for x in MD.disasm(TARGET,0)]
def evaluate(item):
 name,src=item
 obj=W/(name+'.obj')
 if src is not None:
  p=W/(name+'.cpp'); p.write_text(src); obj=W/(name+'.obj')
  cmd,env=build.compiler_command(p,obj)
  result=subprocess.run(cmd,cwd=ROOT,env=env,capture_output=True,text=True)
  if result.returncode:
   (W/(name+'.err')).write_text(result.stdout+result.stderr)
   return dict(name=name,error=(result.stdout+result.stderr)[-300:])
 raw,rel=build.read_object_symbol_bytes(obj,SYM,code_only=True)
 # Cut at ret 8; following bytes are jump table.
 A=[]
 for x in MD.disasm(raw,0):
  A.append((x.address,x.mnemonic,x.op_str,x.size))
  if x.mnemonic=='ret':break
 sm=difflib.SequenceMatcher(a=[x[1] for x in A],b=[x[1] for x in B],autojunk=False)
 counts=dict(struct=0,reg=0,mem=0,imm=0,case1=0,case6=0,other=0)
 for tag,a1,a2,b1,b2 in sm.get_opcodes():
  if tag!='equal':
   nd=max(a2-a1,b2-b1)
   counts['struct']+=nd
   at=B[b1][0] if b1<len(B) else 1527
   counts['case1' if 0x173<=at<0x22f else 'case6' if 0x27f<=at<0x35b else 'other']+=nd
   continue
  for a,b in zip(A[a1:a2],B[b1:b2]):
   if a[2]==b[2]:continue
   # Ignore all branch/call targets and large relocated absolute addresses.
   if a[1].startswith('j') or (a[1]=='call' and 'ptr' not in a[2]):continue
   aa=re.sub(r'0x[0-9a-f]{6,}', 'ADDR',a[2]); bb=re.sub(r'0x[0-9a-f]{6,}','ADDR',b[2])
   if any(a[0]<=off<a[0]+a[3] and typ==6 for off,typ,sym in rel):
    if 'ptr [' in aa: aa=re.sub(r'\[[^]]+\]','[ADDR]',aa); bb=re.sub(r'\[[^]]+\]','[ADDR]',bb)
    else: continue
   if aa==bb:continue
   counts['mem' if 'ptr' in a[2] and 'ptr' in b[2] else 'reg']+=1
   at=b[0]; counts['case1' if 0x173<=at<0x22f else 'case6' if 0x27f<=at<0x35b else 'other']+=1
 return dict(name=name,size=len(raw),hash=hashlib.sha256(raw).hexdigest()[:12],score=sum(counts[k] for k in ['struct','reg','mem','imm']),**counts)
def run(variants,label,jobs=8):
 out=[]
 with concurrent.futures.ThreadPoolExecutor(max_workers=jobs) as pool:
  for result in pool.map(evaluate,variants):
   out.append(result)
 (W/(label+'.json')).write_text(json.dumps(out,indent=2)+'\n')
 good=sorted((x for x in out if 'error' not in x),key=lambda x:(x['score'],abs(x['size']-1563)))
 print(label,len(out),'trials',len(out)-len(good),'errors',len(set(x['hash'] for x in good)),'hashes',flush=True)
 for x in good[:12]:print(json.dumps(x),flush=True)
 return out
if __name__=='__main__':
 run([('baseline',(W/'base.cpp').read_text())],'baseline',1)
