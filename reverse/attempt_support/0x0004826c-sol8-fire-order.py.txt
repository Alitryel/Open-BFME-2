from trials import *
import itertools
names=['fuel','resistance','burnRate','maxBurn','decay']
decl='int fuel, resistance, burnRate, maxBurn, decay;'
call='TheTriggerManager->rva002858E5(&world.x, &fuel, &burnRate, &decay, &resistance, &maxBurn)'
fmt='unibuffer.format(W("Fire logic: fuel=%d, resistance=%d, burn rate=%d, max=%d, decay=%d"), fuel, resistance, burnRate, maxBurn, decay);'
char='template <> void StringBase<char>::releaseBuffer() throw();\n'
family=sys.argv[1]
if family=='addresses':
 items=[]
 for mask,rev,pre,release in itertools.product(range(32),range(2),range(4),range(2)):
  s=BASE
  if not release:s=s.replace(char,'')
  order=names[::-1] if rev else names
  ds=' '.join('int *const '+n+'Ptr=&'+n+';' for n in order if mask>>names.index(n)&1)
  cc=call
  for i,n in enumerate(names):
   if mask>>i&1:cc=cc.replace('&'+n,n+'Ptr')
  s=s.replace(decl,decl+' '+ds).replace(call,cc)
  if pre:
   o=[names,['fuel','burnRate','resistance','maxBurn','decay'],names[::-1]][pre-1]
   prefix=' '.join('const int '+n+'Value='+n+';' for n in o)
   ff=fmt.split('), fuel,')[0]+'), '+', '.join(n+'Value' for n in names)+');'
   s=s.replace(fmt,'{ '+prefix+' '+ff+' }')
  items.append(('addr%02d%d%d%d'%(mask,rev,pre,release),s))
 batch(items,'fire-address-order')
if family=='helpers':
 items=[]
 for p,perm in enumerate(itertools.permutations(names)):
  for byref,release in itertools.product(range(2),repeat=2):
   s=BASE
   if not release:s=s.replace(char,'')
   params=', '.join(('const int &' if byref else 'int ')+n for n in perm)
   body=fmt.replace('unibuffer.','text->')
   helper='static __forceinline void formatFire(UnicodeString *text, '+params+') { '+body+' }\n'
   s=s.replace('void W3DDisplay::gatherDebugStats(void)',helper+'void W3DDisplay::gatherDebugStats(void)')
   s=s.replace(fmt,'formatFire(&unibuffer, '+', '.join(perm)+');')
   items.append(('helper%03d%d%d'%(p,byref,release),s))
 batch(items,'fire-format-order')
