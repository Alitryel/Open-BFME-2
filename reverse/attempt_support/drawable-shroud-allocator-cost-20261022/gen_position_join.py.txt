from pathlib import Path
import itertools

root=Path(__file__).parent;out=root/'position_join';out.mkdir(exist_ok=True)
base=Path('reverse/attempts/0x0027893e.cpp').read_text()
for capture, scope, representation, receiver, edge in itertools.product(range(5),range(2),range(2),range(2),range(2)):
    source=base
    decl='const Coord3D *notifyPosition;'
    pos='notifyPosition'
    if representation:
        decl='struct PositionForNotify { const Coord3D *value; }; PositionForNotify notifyPosition;'
        pos='notifyPosition.value'
    anchor='int notifyEvent;' if scope==0 else 'Player *localPlayer'
    source=source.replace(anchor,decl+'\n'+anchor,1)
    assignment=pos+'=obj->getPosition();'
    guard='if (!(m_status & mask80) && !obj->isEffectivelyDead()) {'
    if capture==0:source=source.replace(guard,assignment+' '+guard)
    if capture==1:source=source.replace(guard,guard+' '+assignment)
    if capture==2:source=source.replace('m_status |= mask80;','m_status |= mask80; '+assignment)
    if capture==3:
        for field in ['m_revealedEvent','m_shroudedEvent']:
            source=source.replace('notifyEvent = tmpl->'+field+';','notifyEvent = tmpl->'+field+'; '+assignment)
    if capture==4:source=source.replace('completeNotify:','completeNotify: '+assignment)
    at=source.index('completeNotify:')
    head,tail=source[:at],source[at:]
    if receiver:tail=tail.replace('((Drawable *)obj)','((Drawable *)((char *)'+pos+'-0x38))')
    tail=tail.replace('obj->getPosition()',pos) if capture!=4 else tail.replace('obj->getPosition()',pos,1).replace('obj->getPosition()',pos)
    # With capture at the join, do not rewrite its own address computation.
    if capture==4:tail=tail.replace(pos+'='+pos+';',assignment,1)
    if edge:tail=tail.replace('if (notifyEvent != -1) {','if(notifyEvent==-1)return; {')
    (out/f'c{capture}s{scope}p{representation}r{receiver}e{edge}.cpp').write_text(head+tail)
print(len(list(out.glob('*.cpp'))))
