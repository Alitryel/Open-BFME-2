import itertools, json, concurrent.futures
import search

# Counterfactual controls: these can return zero and are not candidate banks.
# Two full-width compares with short branches would occupy exactly sixteen bytes.
tails=[
    'return status==A || status==B;',
    'return status!=A && status!=B;',
    'int r=0;if(status==A || status==B)++r;return r;',
    'int r=0;if(status!=A && status!=B)++r;return r;',
    'int r=0;if(status==A)goto yes;if(status==B)goto yes;return r;yes:return r+1;',
    'int r=0;if(status==A)goto done;if(status==B)goto done;++r;done:return r;',
    'if(status==A)return 0;if(status==B)return 0;return 1;',
    'switch(status){case A:case B:return 0;default:return 1;}',
]
consts=[('0x800b0100L','0x80096010L'),('0x800b0111L','0x800b0004L'),('0x800b0100L','0x800b0004L')]
jobs=[]
for tail,constants,cpu,scope,resultkind,decl,barrier in itertools.product(range(len(tails)),range(len(consts)),[5,6,7],range(2),['int','bool'],range(2),range(2)):
    a,b=consts[constants];code=tails[tail].replace('A',a).replace('B',b).replace('int r=',resultkind+' r=')
    if scope:code='{'+code+'}'
    if barrier:code='_WriteBarrier();'+code
    d='unsigned long u=(unsigned long)status;' if decl else ''
    if decl:code=code.replace('status','u')
    text=search.base.replace(search.marker,'long status=WinVerifyTrust(0,&action,&data);'+d+'LocalFree(argv);'+code).replace('/G7',f'/G{cpu}')
    if barrier:text=text.replace('int Rva00237D3DCheck() {','extern "C" void _WriteBarrier();\n#pragma intrinsic(_WriteBarrier)\nint Rva00237D3DCheck() {')
    jobs.append((f'c{len(jobs):04}',text,dict(tail=tail,constants=constants,cpu=cpu,scope=scope,resultkind=resultkind,decl=decl,barrier=barrier,diagnostic_only=True)))
results=search.execute('two_codes',jobs)
good=[r for r in results if 'error' not in r]
for r in good:
    raw,rel=search.build.read_object_symbol_bytes(search.OUT/(r['name']+'.obj'),search.NAME,code_only=True)
    norm,un=search.resolve_bytes(raw,rel,search.TARGET+b'\0'*max(0,len(raw)-search.SIZE),search.RVA,len(raw),{'_memset':[0x6291ae],'_WinVerifyTrust@12':[0x655652]})
    r['outside_nop_differences']=[i for i,(a,b) in enumerate(itertools.zip_longest(norm,search.TARGET)) if a!=b and not 0xcd<=i<0xdd]
(search.OUT/'two_codes.json').write_text(json.dumps(results,indent=2))
print('BEST OUTSIDE NOP:')
for r in sorted(good,key=lambda r:len(r['outside_nop_differences']))[:5]:print(r)
