import itertools
from pathlib import Path
b = Path('sw.cpp').read_text(encoding='latin-1')
EB = " _ReadWriteBarrier();\n Coord3D *to=&adjustTo;"
D0 = "     HierarchyPoint delta;delta.x=way->location.x;delta.y=way->location.y;delta.z=way->location.z;\n"
TB = "  m_isTunneling=false;\n  _ReadWriteBarrier();\n  Coord3D closest;\n  *adjustedTo=*(Coord3D *)Rva002E8C4DCall((Int)&closest,1,(Rva002E8C23Pair *)candidate.cell->m_pathInfo,1);"
for s in (EB, D0, TB): assert b.count(s) == 1, s
DV = [D0, "     HierarchyPoint delta(way->location);\n", "     Coord3D delta=way->location;\n",
      "     const Coord3D *loc=&way->location;HierarchyPoint delta;delta.x=loc->x;delta.y=loc->y;delta.z=loc->z;\n"]
TV = [TB,
      "  m_isTunneling=false;\n  _ReadWriteBarrier();\n  Coord3D closest;\n  PathfinderGroundInfo *info=candidate.cell->m_pathInfo;\n  *adjustedTo=*(Coord3D *)Rva002E8C4DCall((Int)&closest,1,(Rva002E8C23Pair *)info,1);",
      "  m_isTunneling=false;\n  Coord3D closest;\n  *adjustedTo=*(Coord3D *)Rva002E8C4DCall((Int)&closest,1,(Rva002E8C23Pair *)candidate.cell->m_pathInfo,1);",
      "  m_isTunneling=false;\n  Coord3D closest;\n  PathfinderGroundInfo *info=candidate.cell->m_pathInfo;\n  *adjustedTo=*(Coord3D *)Rva002E8C4DCall((Int)&closest,1,(Rva002E8C23Pair *)info,1);"]
for e, d, t in itertools.product(range(2), range(4), range(4)):
    u = b
    if e: u = u.replace(EB, " Coord3D *to=&adjustTo;")
    u = u.replace(D0, DV[d]).replace(TB, TV[t])
    Path(f'v22/e{e}d{d}t{t}.cpp').write_text(u, encoding='latin-1', newline='')
