 ('gne',     'if ((c = *p++) != 0)'),

 ('gnul',    "if ((c = *p++) != '\\0')"),

 ('lne',    '{ do { c = *p++; } while (c != 0); }'),
 ('lsplit', '{ do { c = *p; ++p; } while (c); }'),
 ('lsplit2','{ do { c = *p; p += 1; } while (c); }'),
 ('lb',     '{ char b; do { b = *p++; } while (b); }'),
]
for gt, gx in GUARDS:
    if gt not in ('gplain', 'gne', 'gvol', 'gcomma', 'gnz'):
        continue
    for lt, lx in LATCHES:
        add('fa_%s_%s' % (gt, lt), NZ + FN + 'const char *p=s; char c; ' + gx + ' ' + lx + RET)
