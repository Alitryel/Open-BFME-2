from pathlib import Path
import sys,subprocess,json,concurrent.futures,time
ROOT=Path('/work7/Open-BFME-2');sys.path.insert(0,str(ROOT/'tools'))
import build,permute
OUT=ROOT/'build/collision-7586f0';SYMBOL='?rva007586F0@CollisionManagerImpl@@QAEXXZ'
BASE='// cl: /O2 /G6 /DNDEBUG /MD\n'+(ROOT/'reverse/attempts/0x007586f0.cpp').read_text()
SM=build.load_symbol_map()
def run(item):
 tag,s=item;src=OUT/(tag+'.cpp');obj=src.with_suffix('.obj');same=src.exists() and src.read_text()==s
 if not same:src.write_text(s)
 cmd,env=build.compiler_command(src,obj)
 p=subprocess.CompletedProcess(cmd,0,'','') if same and obj.exists() and obj.stat().st_mtime>=src.stat().st_mtime else subprocess.run(cmd,cwd=ROOT,env=env,capture_output=True,text=True)
 if p.returncode:
  src.with_suffix('.err').write_text(p.stdout+p.stderr);return {'tag':tag,'error':(p.stdout+p.stderr)[-200:]}
 try:
  raw,rel=build.read_object_symbol_bytes(obj,SYMBOL,code_only=True)
  row={'name':SYMBOL,'target_rva':'0x7586f0','target_size':'460','source':str(src.relative_to(ROOT)),'notes':''}
  got=build.compile_function(row,SM,obj);a=bytes(got['bytes']);b=bytes(got['target'])
  src.with_suffix('.bin').write_bytes(a)
  return {'tag':tag,'size':len(raw.rstrip(b'\xcc')),'nd':sum(x!=y for x,y in zip(a,b))+abs(len(a)-len(b)), 'idiff':sum(x!=y for x,y in zip(permute._instructions(a),permute._instructions(b)))+abs(len(permute._instructions(a))-len(permute._instructions(b))), 'fit':permute.fitness(a,b),'exact':a==b and len(raw.rstrip(b'\xcc'))==460 and not got['unresolved']}
 except Exception as e:return {'tag':tag,'error':str(e)}
def batch(items,label):
 start=time.time();res=[]
 with concurrent.futures.ThreadPoolExecutor(max_workers=8) as pool:
  for i,r in enumerate(pool.map(run,items)):
   res.append(r)
   if r.get('exact'):print('EXACT',r,flush=True)
   if (i+1)%256==0:print('progress',i+1,'sec',round(time.time()-start),flush=True)
 (OUT/(label+'.json')).write_text(json.dumps(res,indent=2))
 best=sorted(res,key=lambda r:(not r.get('exact',False),-r.get('fit',0),r.get('idiff',999),r.get('nd',999)))
 print('BEST',label,json.dumps(best[:12]),'errors',sum('error'in r for r in res),flush=True)
 return res
if __name__=='__main__':batch([('control',BASE)],'control')
