from pathlib import Path
import hashlib, json, struct
import pefile, capstone

ROOT=Path.cwd();OUT=ROOT/'build/trust237-round7';OUT.mkdir(parents=True,exist_ok=True)
path=ROOT/'reference/open-bfme-1/inputs/toolchains/vs2003/Program Files/Microsoft Visual Studio .NET 2003/Vc7/bin/c2.dll'
data=path.read_bytes();sha=hashlib.sha256(data).hexdigest()
assert sha=='bcd28f39b1798b07989c30a1df471cf87fc20e3802d130901a08f83800e6b81e'
pe=pefile.PE(data=data);base=pe.OPTIONAL_HEADER.ImageBase
def read(va,n):return pe.get_data(va-base,n)
def u32(va):return struct.unpack('<I',read(va,4))[0]
def string(va):return pe.get_string_at_rva(va-base).decode('ascii')
md=capstone.Cs(capstone.CS_ARCH_X86,capstone.CS_MODE_32)
def instructions(lo,hi):
    return [dict(va=hex(i.address),raw=i.bytes.hex(),mnemonic=i.mnemonic,operands=i.op_str)
            for i in md.disasm(read(lo,hi-lo),lo)]

# These are encoding-table rows containing 90, not a claim that each emits
# an unprefixed NOP. Operand/prefix/descriptor handling must be read as well.
rows=[]
for opcode in range(1,0x249):
    raw=read(0x108719c0+6*opcode,6)
    if 0x90 in raw:
        rows.append(dict(opcode=hex(opcode),name=string(u32(0x108743a8+4*opcode)),
                         encoding=raw.hex(),descriptor=read(0x10872780+6*opcode,6).hex(),
                         flags=hex(u32(0x10870dc0+4*opcode))))

# Start at independently identified instruction boundaries, stop before data.
ranges=[(0x10734471,0x1073448e),(0x10734092,0x107340b2),
        (0x107a7ead,0x107a7ece),(0x107c4d85,0x107c4dc5),
        (0x1082f161,0x1082f17f),
        (0x107b0d9d,0x107b0daa),(0x107c4a25,0x107c4a53),
        (0x1082ee91,0x1082f1bb),(0x1070b97a,0x1070b9ae)]
sites={f'{lo:08x}-{hi:08x}':instructions(lo,hi) for lo,hi in ranges}
descriptors=[]
for va in [0x108664f4,0x10866500]:
    values=struct.unpack('<III',read(va,12))
    descriptors.append(dict(va=hex(va),words=[hex(x) for x in values],name=string(values[0])))
result=dict(compiler=str(path.relative_to(ROOT)),sha256=sha,
            encoding_rows_containing_90=rows,option_descriptors=descriptors,
            instructions=sites)
(OUT/'compiler-nop-paths.json').write_text(json.dumps(result,indent=2)+'\n')
print('compiler',sha,'encoding rows containing 90',len(rows),'bounded disassemblies',len(sites))
for row in rows:print(row['opcode'],row['name'],row['encoding'],row['flags'])
for row in descriptors:print(row)
