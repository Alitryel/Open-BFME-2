from search import *
import itertools
s=(W/'mergedcalls_00_01.cpp').read_text()
u='((LocoTerrainView*)TheTerrainLogic)->underwater(pos.x,pos.y,&water,0,0)'
layer='((LocoTerrainView*)TheTerrainLogic)->layer(pos.x,pos.y,object->rva0028B511(),0,true)'
orig='if('+u+')pos.z=water;\n   else pos.z='+layer+';'
selectors=['object->disabled&1','object->disabled&2','object->definition!=0','data->behavior==1','mode==1','flags44&1','requires','object!=0']
vs=[]
for c,f,polarity,scope in itertools.product(selectors,range(8),range(2),range(2)):
 cond=('!('+c+')') if polarity else c
 if f==0:new='bool wet;if('+cond+')wet='+u+';else wet='+u+';if(wet)pos.z=water;else pos.z='+layer+';'
 elif f==1:new='bool wet=('+cond+')?'+u+':'+u+';if(wet)pos.z=water;else pos.z='+layer+';'
 elif f==2:new='if('+cond+'){'+orig+'}else{'+orig+'}'
 elif f==3:new='if(('+cond+')?'+u+':'+u+')pos.z=water;else pos.z='+layer+';'
 elif f==4:new='if('+cond+'){if('+u+')pos.z=water;else goto layer;}else{if('+u+')pos.z=water;else goto layer;}goto joined;layer:pos.z='+layer+';joined:;'
 elif f==5:new='bool wet;switch(('+cond+')?1:0){case 0:wet='+u+';break;default:wet='+u+';break;}if(wet)pos.z=water;else pos.z='+layer+';'
 elif f==6:new='float value;if('+cond+'){if('+u+')value=water;else value='+layer+';}else{if('+u+')value=water;else value='+layer+';}pos.z=value;'
 else:new='if('+cond+'){if('+u+')goto watery;else goto layer;}else{if('+u+')goto watery;else goto layer;}watery:pos.z=water;goto joined;layer:pos.z='+layer+';joined:;'
 o=s.replace(orig,'{'+new+'}' if scope else new)
 vs.append((f'mergedunder_{selectors.index(c)}_{f}_{polarity}{scope}',o))
run(vs,'mergedunder256')
