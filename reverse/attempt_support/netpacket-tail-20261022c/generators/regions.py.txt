import json,sys,difflib
from pathlib import Path
from collections import Counter
import capstone
sys.path.insert(0,'tools');import build,align_diff
W=Path('build/netpacket-20261022c');md=capstone.Cs(capstone.CS_ARCH_X86,capstone.CS_MODE_32)
retail=build.read_target_bytes(0x593f15,1475)
R=list(md.disasm(retail[:1351],0x593f15))
for f in sys.argv[1:]:
 rs=json.loads((W/(f+'.json')).read_text());out=[];seen=set()
 for r in rs:
  if 'error'in r or r['sha']in seen:continue
  seen.add(r['sha']);a=(W/(r['label']+'.bin')).read_bytes();A=list(md.disasm(a[:max(0,r['size']-124)],0x593f15))
  sm=difflib.SequenceMatcher(a=[i.mnemonic for i in A],b=[i.mnemonic for i in R],autojunk=False);c=Counter()
  for tag,x,y,z,v in sm.get_opcodes():
   if tag=='equal':
    for aa,bb in zip(A[x:y],R[z:v]):
     if aa.op_str!=bb.op_str:c[align_diff.classify(aa.op_str,bb.op_str)]+=1
   else:c['structural_regions']+=1;c['our_structural_insns']+=y-x;c['retail_structural_insns']+=v-z
  det=[j for j,i in enumerate(A) if i.mnemonic=='call' and i.op_str=='0x4d55ba']
  # Also preserve every near-end call's resolved name/address rather than guessing.
  tail=[f'{i.address:x}: {i.mnemonic} {i.op_str}' for i in A if i.address>0x594410 and i.address<0x59444a]
  out.append(dict(label=r['label'],size=r['size'],fit=r['fit'],diff=r['diff'],counts=dict(c),tail=tail))
 (W/(f+'-regions.json')).write_text(json.dumps(out,indent=2))
 print(f)
 for x in sorted(out,key=lambda x:-x['fit'])[:7]:print(x['label'],x['size'],round(x['fit'],5),x['counts'])
