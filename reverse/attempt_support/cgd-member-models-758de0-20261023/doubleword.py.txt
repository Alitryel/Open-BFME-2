from pathlib import Path
from itertools import product
W=Path(__file__).resolve().parent
base=(W/'uniform-control.cpp').read_text()
out=W/'doubleword';out.mkdir(exist_ok=True)
forms=[
 'return (Word)(unsigned)p | ((Word)(unsigned)n<<32);',
 'return (Word)(unsigned)p + ((Word)(unsigned)n<<32);',
 'union U { Word q; const ContactVector *v[2]; }; U a; a.v[0]=p; a.v[1]=n; return a.q;',
 'union U { Word q; struct { const ContactVector *p,*n; } v; }; U a; a.v.n=n; a.v.p=p; return a.q;',
 'Word q=(unsigned)p; q|=(Word)(unsigned)n<<32; return q;',
 'Word q=(Word)(unsigned)n<<32; q|=(unsigned)p; return q;',
 'Word q; ((unsigned *)&q)[0]=(unsigned)p; ((unsigned *)&q)[1]=(unsigned)n; return q;',
 'Word q=(unsigned)n; q*=0x100000000ui64; q+=(unsigned)p; return q;',
]
for idx,(kind,form,fast,sites,order,local) in enumerate(product([2],range(8),range(2),range(8),range(2),range(2))):
 s=base
 defs='typedef unsigned __int64 Word;\n'
 ty=['Word','__int64','double'][kind]
 params='const ContactVector *n,const ContactVector *p' if order else 'const ContactVector *p,const ContactVector *n'
 defs+='__forceinline Word pack('+params+') {'+forms[form]+'}\n'
 args=lambda p,n: ','.join(['&('+n+')','&('+p+')'] if order else ['&('+p+')','&('+n+')'])
 def call(p,n):
  expr='pack('+args(p,n)+')'
  return 'toDouble('+expr+')' if kind==2 else '('+ty+')'+expr
 if kind==2:defs+='__forceinline double toDouble(Word v) { union U {Word q;double d;}; U u;u.q=v;return u.d;}\n'
 s=s.replace('struct ContactSource {',defs+'struct ContactSource {')
 adapt=(ty+' v='+call('p','n')+';return prepareWord(v);') if local else 'return prepareWord('+call('p','n')+');'
 s=s.replace('virtual void *prepare(const ContactVector &point, const ContactVector &normal)=0;',
  'virtual void *'+('__fastcall ' if fast else '')+'prepareWord('+ty+' args)=0;\n __forceinline void *prepare(const ContactVector &p,const ContactVector &n) {'+adapt+'}')
 pos=0
 for i,(r,n) in enumerate([('node->second->source','node->normal'),('node->second->source','node->normal'),('node->first->source','ContactVector(node->normal).negate()')]):
  old=r+'->prepare(node->point,'+n+')';at=s.find(old,pos);assert at>=0
  new=r+'->prepareWord('+call('node->point',n)+')' if sites&(1<<i) else old
  s=s[:at]+new+s[at+len(old):];pos=at+len(new)
 (out/('v%04d.cpp'%(idx+1024))).write_text(s)
print('generated',idx+1)
