import sys,json,itertools,time
from pathlib import Path
from concurrent.futures import ThreadPoolExecutor,as_completed
sys.path.insert(0,'build/fxinit-seat22');import search as h
h.HERE=Path('build/fxinit-admit23').resolve()
def variants():
 for shape,order,cc,ty,view,where in itertools.product(range(3),range(6),range(3),range(2),range(3),range(2)):
  # Caller/inline-member boundary, raw map input vs short-vector receiver,
  # signedness of uint16-valued indices, and argument order are crossed.
  t='unsigned' if ty else 'int';conv=['__cdecl','__stdcall','__fastcall'][cc]
  if shape==0:args=[('const short *map','remap.begin()'),(t+' a','links[0][i]'),(t+' b','links[1][i]')];calc='map[a]+(map[b]<<8)'
  elif shape==1:args=[('const short *map','remap.begin()'),('const unsigned short *a','links[0]+i'),('const unsigned short *b','links[1]+i')];calc='map[*a]+(map[*b]<<8)'
  else:args=[('_STL::vector<short> &map','remap'),(t+' a','links[0][i]'),(t+' b','links[1][i]')];calc='map[a]+(map[b]<<8)'
  perm=list(itertools.permutations(range(3)))[order];decl=','.join(args[j][0]for j in perm);vals=','.join(args[j][1]for j in perm)
  if view==0:helper='static __forceinline int '+conv+' RemapPair('+decl+'){return '+calc+';}\n';call='RemapPair('+vals+')';prefix=''
  elif view==1:helper='struct RemapPairView{ static __forceinline int '+conv+' Pack('+decl+'){return '+calc+';}};\n';call='RemapPairView::Pack('+vals+')';prefix=''
  else:
   # Real value view, no address overlay or fabricated object lifetime.
   keep=[j for j in perm if j!=0];pars=','.join(args[j][0]for j in keep);vs=','.join(args[j][1]for j in keep)
   field='_STL::vector<short> &map;' if shape==2 else 'const short *map;'
   init='_STL::vector<short> &m' if shape==2 else 'const short *m'
   helper='struct RemapPairView{'+field+' __forceinline RemapPairView('+init+'):map(m){} __forceinline int '+conv+' Pack('+pars+') const{return '+calc+';}};\n'
   prefix='RemapPairView remapView('+args[0][1]+');';call='remapView.Pack('+vs+')'
  # where 1 gives the helper result a named scalar before its store.
  # Both remain before W, matching the original effects.
  core=prefix+('int packedBones='+call+';' if where else '')+'*(int *)(vertices+desc.Get_Basis_End())='+('packedBones' if where else call)+';'
  s=h.BASE.replace(h.R,'{'+core+'}').replace('bool FXShaderGeometry::Init(',helper+'bool FXShaderGeometry::Init(')
  yield f'remapabi_{shape}{order}{cc}{ty}{view}{where}',s
if __name__=='__main__':
 t=time.time();rs=[]
 with ThreadPoolExecutor(max_workers=8)as ex:
  for f in as_completed([ex.submit(h.run_one,x)for x in variants()]):
   rs.append(f.result())
   if len(rs)%128==0:print('tested',len(rs),'best',min(rs,key=lambda z:(z.get('nd',99999),z.get('cost',99999))),flush=True)
 rs.sort(key=lambda z:(z.get('nd',99999),z.get('cost',99999)))
 (h.HERE/'remapabi-results.json').write_text(json.dumps(rs,indent=2))
 print(json.dumps({'count':len(rs),'seconds':time.time()-t,'errors':sum('error'in z for z in rs),'best':rs[:6]},indent=2))
