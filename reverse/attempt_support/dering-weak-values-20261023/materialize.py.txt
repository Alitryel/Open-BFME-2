"""Name consumed expression values; never add unused locals or unroll loops."""
from pathlib import Path
import json,re

source=Path('build/local-vm/dering-weak-original.cpp.txt').read_bytes().decode('latin1')
tree=json.loads(Path('build/local-vm/dering-weak-ast.json').read_text())
def bounds(n):
 r=n['range'];return r['begin']['offset'],r['end']['offset']+r['end']['tokLen']
def find(n):
 if n.get('kind')=='FunctionDecl' and n.get('name')=='DeringBlockWeak_C':return n
 for c in n.get('inner',[]):
  result=find(c)
  if result:return result
fn=find(tree)
body=next(n for n in fn['inner'] if n['kind']=='CompoundStmt')
arith={'+':'Sum','-':'Difference','*':'Product','/':'Quotient','%':'Remainder',
       '<<':'Shift','>>':'Shift','&':'Mask','|':'Bits','^':'Bits'}

def generate(mode,limit=None):
 edits=[];count=0;serial=0
 def rewrite(n,context,decls,root=False,dest_type=None,lvalue=False):
  nonlocal count,serial
  lo,hi=bounds(n);kind=n['kind'];children=n.get('inner',[])
  if kind=='ImplicitCastExpr' and children:
   text=rewrite(children[0],context,decls,root,dest_type,lvalue)
  else:
   pieces=[];cursor=lo
   for i,ch in enumerate(children):
    if 'range' not in ch or 'offset' not in ch['range'].get('begin',{}):continue
    a,b=bounds(ch)
    if a<cursor or b>hi:continue
    child_lvalue=lvalue and not(kind=='ArraySubscriptExpr' and i==1)
    child_root=root if kind in ('ParenExpr','CStyleCastExpr') else False
    text_ch=rewrite(ch,context,decls,child_root,dest_type,child_lvalue)
    pieces.extend([source[cursor:a],text_ch]);cursor=b
   pieces.append(source[cursor:hi]);text=''.join(pieces)
  typ=n.get('type',{}).get('qualType','')
  label=None
  if kind=='BinaryOperator' and n.get('opcode') in arith and not root:
   label=arith[n['opcode']]
  elif mode in ('loads','promotions') and kind=='ArraySubscriptExpr' and not lvalue:
   # A narrow load followed by integer promotion is a genuine separate value.
   if not root or typ.replace('const ','') != (dest_type or '').replace('const ',''):
    label='Sample'
  elif mode=='promotions' and kind=='ImplicitCastExpr' and n.get('castKind')=='IntegralCast':
   if not root:label='Promoted'
  if label and (limit is None or count<limit):
   serial+=1;name=context+label+('%03d'%serial)
   value_type=re.sub(r'^const\s+','',typ) if '*' not in typ else typ
   decls.append(value_type+' '+name+' = '+text+';')
   count+=1;text=name
  return text

 def statement(n):
  kind=n['kind']
  if kind in ('CompoundStmt','ForStmt','WhileStmt','DoStmt','IfStmt'):
   for c in n.get('inner',[]):
    if c.get('kind') in ('CompoundStmt','ForStmt','WhileStmt','DoStmt','IfStmt'):
     statement(c)
    elif kind=='CompoundStmt':statement(c)
   return
  decls=[];patches=[]
  if kind=='DeclStmt':
   for var in n.get('inner',[]):
    init=var.get('inner',[])
    if not init:continue
    expr=init[-1]
    a,b=bounds(expr)
    text=rewrite(expr,var.get('name','value'),decls,True,var.get('type',{}).get('qualType'))
    patches.append((a,b,text))
  elif kind=='BinaryOperator' and n.get('opcode')=='=':
   left,right=n['inner'];a,b=bounds(left)
   patches.append((a,b,rewrite(left,'outputIndex',decls,True,None,True)))
   a,b=bounds(right)
   patches.append((a,b,rewrite(right,'outputValue',decls,True,left.get('type',{}).get('qualType'))))
  else:return
  if not decls:return
  lo,hi=bounds(n);text=source[lo:hi]
  for a,b,value in sorted(patches,reverse=True):text=text[:a-lo]+value+text[b-lo:]
  line=source.rfind('\n',0,lo)+1;indent=source[line:lo]
  if indent.strip():indent='\t'
  text=('\n'+indent).join(decls)+'\n'+indent+text
  edits.append((lo,hi,text))
 statement(body)
 result=source
 for a,b,text in sorted(edits,reverse=True):result=result[:a]+text+result[b:]
 return result,count

out=Path('build/local-vm/dering-materialized');out.mkdir(exist_ok=True)
manifest=[]
for mode in ['arithmetic','loads','promotions']:
 for limit in [None,64,80,96,112]:
  text,count=generate(mode,limit)
  if any(r['mode']==mode and r['count']==count for r in manifest):continue
  name=mode+'-'+str(count);(out/(name+'.cpp')).write_text(text)
  manifest.append(dict(name=name,mode=mode,count=count))
(out/'manifest.json').write_text(json.dumps(manifest,indent=2)+'\n')
print(json.dumps(manifest,indent=2))
