from itertools import product
from pathlib import Path
W=Path(__file__).resolve().parent;base=(W/'base.cpp').read_text()
out=W/'callconv';out.mkdir(exist_ok=True)
for index,(cc,pf,inf,copy,ctorcc,prepconst) in enumerate(product(range(8),range(4),range(4),range(2),range(2),range(2))):
 s=base
 ct='__fastcall ' if ctorcc else ''
 pty=['const ContactVector *','const ContactVector &','unsigned','const void *'][pf]
 pv=['&p','p','(unsigned)&p','&p'][pf]
 it=['void *','unsigned','const void *','char *'][inf]
 iv=['p','(unsigned)p','p','(char *)p'][inf]
 defs='struct PArg { '+pty+' p; '+ct+'PArg(const ContactVector &p):p('+pv+') {}'
 if copy:defs+=' '+ct+'PArg(const PArg &v):p(v.p) {}'
 defs+=' };\nstruct IArg { '+it+' p; '+ct+'IArg(void *p):p('+iv+') {}'
 if copy:defs+=' '+ct+'IArg(const IArg &v):p(v.p) {}'
 defs+=' };\n'
 s=s.replace('struct ContactSource {',defs+'struct ContactSource {')
 s=s.replace('virtual void *prepare(const ContactVector &point, const ContactVector &normal)=0;',
  'virtual void *'+('__fastcall ' if cc&1 else '')+'prepare(PArg point,PArg normal)'+(' const' if prepconst else '')+'=0;')
 s=s.replace('virtual bool test(void *info)=0;', 'virtual bool '+('__fastcall ' if cc&2 else '')+'test(IArg info)=0;')
 s=s.replace('virtual void respond(void *info)=0;', 'virtual void '+('__fastcall ' if cc&4 else '')+'respond(IArg info)=0;')
 (out/('v%04d.cpp'%index)).write_text(s)
print('generated',index+1)
