## build/sk21/pr.py: TPA product operand signature vs retail, usage: python3 build/sk21/pr.py X.cpp get_deformed 0x700
# pr.py probe.cpp [sym-substring] -> emitted product signature of first TPA loop
import sys,subprocess,re
from pathlib import Path
ROOT=Path('/work/Open-BFME-2'); sys.path.insert(0,str(ROOT/'tools'))
import build,capstone
def run(src,symsub='probe'):
    src=Path(src).resolve()
    out=ROOT/'build'/'sk21'/'obj'/(src.stem+'.obj'); out.parent.mkdir(parents=True,exist_ok=True)
    if out.exists(): out.unlink()
    cmd,env=build.compiler_command(src,out)
    p=subprocess.run(cmd,cwd=ROOT,env=env,capture_output=True,text=True)
    if p.returncode: return 'ERR '+(p.stdout+p.stderr)[-300:].replace('\n',' ')
    import subprocess as sp
    syms=build.list_object_symbols(out) if hasattr(build,'list_object_symbols') else None
    return out
def sig(code):
    md=capstone.Cs(capstone.CS_ARCH_X86,capstone.CS_MODE_32)
    ins=[(i.mnemonic,i.op_str) for i in md.disasm(code,0)]
    # find movss/mulss pairs with memory operands
    prods=[]
    for k in range(len(ins)-1):
        a,b=ins[k],ins[k+1]
        if a[0]=='movss' and b[0]=='mulss':
            ma=re.search(r'\[(\w+)(?: ([+-]) (0x[0-9a-f]+|\d+))?\]',a[1]); mb=re.search(r'\[(\w+)(?: ([+-]) (0x[0-9a-f]+|\d+))?\]',b[1])
            if ma and mb:
                f=lambda m:(m.group(1),(int(m.group(3),0)*(-1 if m.group(2)=='-' else 1)) if m.group(3) else 0)
                prods.append((f(ma),f(mb)))
    return prods[:12]
def names(prods):
    bases={}
    for p in prods:
        for r,d in p: bases.setdefault(r,set()).add(d)
    if len(bases)!=2: return None
    rb=max(bases,key=lambda r:(len(bases[r]),max(bases[r])-min(bases[r])))
    vb=[r for r in bases if r!=rb][0]; vmin=min(bases[vb])
    def nm(o):
        r,d=o
        return ('R.'+'XYZW'[d//4]) if r==rb else ('v.'+'XYZ'[(d-vmin)//4])
    return ['%s*%s'%(nm(a),nm(b)) for a,b in prods]
RET='v.Y*R.Z R.Y*v.Z v.Z*R.X R.Z*v.X v.Z*R.Z R.W*v.X R.Y*v.X R.W*v.Y v.Y*R.X R.W*v.Z v.X*R.X R.Y*v.Y'.split()
if __name__=='__main__':
    src=sys.argv[1]; symsub=sys.argv[2] if len(sys.argv)>2 else 'probe'
    out=run(src)
    if isinstance(out,str): print(Path(src).stem,out); sys.exit()
    syms=[s for s in build.defined_code_symbols(out) if symsub in s]
    code,rel=build.read_object_symbol_bytes(out,syms[0],code_only=True)
    code=code[int(sys.argv[3],0) if len(sys.argv)>3 else 0:]
    n=names(sig(code))
    if n is None: print(Path(src).stem,'nosig'); sys.exit()
    # match ignoring order: per product operand order agreement
    ro={frozenset(x.split('*')):x for x in RET}
    flips=[x for x in n if ro.get(frozenset(x.split('*')))!=x]
    order_ok=[frozenset(x.split('*')) for x in n]==[frozenset(x.split('*')) for x in RET]
    print(Path(src).stem,'flips',len(flips),'order',int(order_ok),' '.join(n))
