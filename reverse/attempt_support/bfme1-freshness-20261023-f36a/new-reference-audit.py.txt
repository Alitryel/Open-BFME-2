from pathlib import Path
import collections,datetime,hashlib,json,subprocess,sys
R=Path.cwd();O=R/'build/trust237-round15';S=R/'reference/open-bfme-1'
sys.path.insert(0,str(R/'tools'));import build,zh_sweep
old='f890413e75e671748215ba2ef3304aa7953993e9';new='f36a4db066aa61edeee90085a31d628eb829001d'
def git(*args):return subprocess.check_output(['git',*args],text=True).strip()
assert git('-C',str(S),'rev-parse','HEAD')==new
assert not git('-C',str(S),'status','--porcelain')
audit=json.loads((O/'reference-audit.json').read_text())
for p,v in audit['intended_inputs'].items():assert hashlib.sha256((S/p).read_bytes()).hexdigest()==v['sha256']
trees={p:{rev:git('-C',str(S),'rev-parse',rev+':'+p) for rev in (old,new)} for p in ['inputs/reference','inputs/toolchains','inputs/vendor','inputs/baselines']}
assert all(v[old]==v[new] for v in trees.values())
counts=collections.Counter();exceptions=[];objects=[]
for row in json.loads((O/'zh-cache-refreshed-old.json').read_text())['rows']:
 src=R/row['source'];obj=zh_sweep.OBJ_DIR/(zh_sweep.object_stem(src.relative_to(R))+('.stl' if build.source_needs_stlport(src) else '')+'.obj')
 status='missing' if not obj.exists() else ('current' if build.compile_is_current(src,obj) else 'uncached-built')
 assert status==row['status'],(row['source'],status,row['status'])
 counts[status]+=1
 if status!='current':exceptions.append(dict(source=row['source'],status=status))
 if obj.exists():objects.append(dict(path=str(obj.relative_to(R)),sha256=hashlib.sha256(obj.read_bytes()).hexdigest()))
result=dict(recorded_utc=datetime.datetime.now(datetime.timezone.utc).isoformat(),donor=new,bfme2_revision=git('rev-parse','HEAD'),donor_status='',unchanged_input_trees=trees,intended_inputs_verified=len(audit['intended_inputs']),counts=dict(counts),exceptions=exceptions,object_count=len(objects),object_set_sha256=hashlib.sha256(json.dumps(objects,sort_keys=True).encode()).hexdigest(),object_rows=objects)
(O/'reference-new-audit.json').write_text(json.dumps(result,indent=2)+'\n')
print(json.dumps({k:v for k,v in result.items() if k not in ['exceptions','object_rows']}),flush=True)
