Usage: python3 reverse/attempt_support/water-vector-escape-20261020/replay.py.txt

# prepare.py
from pathlib import Path
import hashlib
import json
import re

BANK = ROOT / "reverse/attempts/0x0008304a.cpp"
LOADER = ROOT / "Code/GameEngine/Source/Common/Rva00240000Load308050.cpp"
CONTAINS = ROOT / "Code/GameEngine/Source/Common/Rva0030D111ContainsPoint.cpp"

bank = BANK.read_text()
loader = LOADER.read_text()
contains = CONTAINS.read_text()

# Preserve the loader's complete visibility graph and import-inline policies.
vector_start = loader.index("namespace _STL { extern template char *__uninitialized_copy")
vector_end = loader.index("\nclass File\n", vector_start)
vector = loader[vector_start:vector_end]
old_vector = """namespace _STL {
 template<> class vector<char,allocator<char> > {
  public: char *start,*finish,*end;
  void resize(unsigned int,char);
 };
}"""
assert bank.count(old_vector) == 1
full = bank.replace(old_vector, vector)

old_local = """ _STL::_Vector_base<BfmeE12,_STL::allocator<BfmeE12> > flags((_STL::allocator<BfmeE12>()));
 reinterpret_cast<_STL::vector<char>*>(&flags)->resize(total,0);"""
new_local = """ _STL::vector<char> flags;
 flags.resize(total,0);"""
assert full.count(old_local) == 1
typed = full.replace(old_local, new_local)
assert typed.count("(*reinterpret_cast<char**>(&flags))[y*nx+x]") == 2
typed = typed.replace("(*reinterpret_cast<char**>(&flags))[y*nx+x]", "flags[y*nx+x]")

contains_start = contains.index('#include "../../../Libraries/Include/Lib/Coord2D.h"')
contains_end = contains.index("// Native30D241", contains_start)
contains_body = contains[contains_start:contains_end]
# Resolve the source's relative includes against its actual owner directory.
contains_body = re.sub(
    r'#include "([^"]+)"',
    lambda m: '#include "' + (CONTAINS.parent / m[1]).resolve().as_posix() + '"',
    contains_body,
)
shape = "class Rva0030D111Shape {public:virtual int count();};"
prototype = "bool Rva0030D111Contains(Rva0030D111Shape*,const Coord2D*);"
parent_start = "void RenderableStandingWaterArea::AllocateAndFillBuffers() {"

def with_contains(source, own_profile):
    assert source.count(shape) == 1
    assert source.count(prototype) == 1
    source = source.replace(shape, "").replace(prototype, "")
    body = contains_body
    if own_profile:
        # The existing provider uses /O1 /Oy- /arch:SSE /G7. The parent has
        # /O1 /arch:SSE /G7 already, so only frame-pointer policy differs.
        body = '#pragma optimize("y", off)\n' + body + '#pragma optimize("y", on)\n'
    assert source.count(parent_start) == 1
    return source.replace(parent_start, body + "\n" + parent_start)

variants = {
    "00-bank.cpp": bank,
    "01-full-vector-e12.cpp": full,
    "02-full-vector-char.cpp": typed,
    "03-full-vector-e12-contains.cpp": with_contains(full, False),
    "04-full-vector-char-contains.cpp": with_contains(typed, False),
    "05-full-vector-char-contains-profile.cpp": with_contains(typed, True),
}
for name, source in variants.items():
    (OUT / name).write_text(source)
receipts = {
    "inputs": {
        str(p.relative_to(ROOT)): hashlib.sha256(p.read_bytes()).hexdigest()
        for p in (BANK, LOADER, CONTAINS)
    },
    "variants": {
        name: hashlib.sha256(source.encode()).hexdigest()
        for name, source in variants.items()
    },
}
(OUT / "inputs.json").write_text(json.dumps(receipts, indent=2))
print("Prepared", len(variants), "controls in", OUT)

from pathlib import Path
import hashlib
import json

loader = (ROOT / "Code/GameEngine/Source/Common/Rva00240000Load308050.cpp").read_text()
preamble = loader[loader.index("#define fill bfmeLoaderHeaderFill"):loader.index('#include "ascii_string.h"')]

def faithful(source):
    assert source.count("#include <vector>") >= 1
    return source.replace("#include <vector>", preamble.rstrip(), 1)

e12 = faithful((OUT / "01-full-vector-e12.cpp").read_text())

def first_pointer(source, is_e12=False):
    begin = "(*reinterpret_cast<char**>(&flags))" if is_e12 else "flags.begin()"
    lookup = "(*reinterpret_cast<char**>(&flags))[y*nx+x]" if is_e12 else "flags[y*nx+x]"
    old = """ for(y=0;y<ny;++y) {
  Coord2D pos;pos.y=y*spacing+center.y;
  for(int x=0;x<nx;++x){"""
    new = """ char *row = BEGIN;
 for(y=0;y<ny;++y,row+=nx) {
  Coord2D pos;pos.y=y*spacing+center.y;
  char *cell = row;
  for(int x=0;x<nx;++x,++cell){""".replace("BEGIN", begin)
    assert source.count(old) == 1
    source = source.replace(old, new)
    assert source.count(lookup + "=in;") == 1
    return source.replace(lookup + "=in;", "*cell=in;")

def second_pointer(source, is_e12=False):
    begin = "(*reinterpret_cast<char**>(&flags))" if is_e12 else "flags.begin()"
    lookup = "(*reinterpret_cast<char**>(&flags))[y*nx+x]" if is_e12 else "flags[y*nx+x]"
    old = """   for(y=0;y<ny;++y){

    for(int x=0;x<nx;++x){
     if(LOOKUP){""".replace("LOOKUP", lookup)
    new = """   for(y=0;y<ny;++y){
    char *cell = BEGIN + y*nx;
    for(int x=0;x<nx;++x,++cell){
     if(*cell){""".replace("BEGIN", begin)
    assert source.count(old) == 1
    return source.replace(old, new)

variants = {

    "07-preamble-char.cpp": char,

from pathlib import Path
import hashlib
import json

e12 = (OUT / "06-preamble-e12.cpp").read_text()
char = (OUT / "07-preamble-char.cpp").read_text()

def ordinary_policy(source):
    start = source.index("namespace _STL { extern template char *__uninitialized_copy")
    end = source.index("// game.dat's STLport passes vector<char>::resize", start)
    return source[:start] + source[end:]

def declaration_only(source, signature):
    start = source.index(signature)
    opening = source.index("{", start)
    depth = 1
    closing = opening + 1
    while depth:
        if source[closing] == "{":
            depth += 1
        elif source[closing] == "}":
            depth -= 1
        closing += 1
    return source[:opening] + ";" + source[closing:]

fill_start = e12.index("inline void vector<char, allocator<char> >::_M_fill_insert")
fill_open = e12.index("{", fill_start)
depth, fill_end = 1, fill_open + 1
while depth:
    if e12[fill_end] == "{":
        depth += 1
    elif e12[fill_end] == "}":
        depth -= 1
    fill_end += 1

variants = {
    "12-e12-header-policy.cpp": ordinary_policy(e12),
    "13-char-header-policy.cpp": ordinary_policy(char),
    "14-e12-opaque-erase.cpp": declaration_only(e12, "\tchar *erase(char *first, char *last)"),
    "15-e12-opaque-fill.cpp": e12[:fill_start] + e12[fill_end:],
    "16-e12-opaque-overflow.cpp": declaration_only(e12, "\tvoid _M_insert_overflow(char *position"),
}
for name, source in variants.items():
    (OUT / name).write_text(source)
(OUT / "round3-inputs.json").write_text(json.dumps({

# Normal compiler entry points followed by read-only object checks.
from pathlib import Path
import json
import os
import subprocess
import sys

TEMP = OUT / "temp"
TEMP.mkdir(exist_ok=True)
os.environ.update(BFME_OBJSTORE="off", TEMP=str(TEMP), TMP=str(TEMP))
sys.path.insert(0, str(ROOT / "tools"))
import build
import eh_verify
import permute

SYMBOL = "?AllocateAndFillBuffers@RenderableStandingWaterArea@@QAEXXZ"
CONTAINS = "?Rva0030D111Contains@@YA_NPAVRva0030D111Shape@@PBVCoord2D@@@Z"
records = json.loads((OUT / "results.json").read_text()) if (OUT / "results.json").exists() else []

def command_to_log(command, name):
    result = subprocess.run(
        command, cwd=ROOT, stdout=subprocess.PIPE, stderr=subprocess.STDOUT,
        text=True, encoding="utf-8", errors="replace",
    )
    (OUT / name).write_text(result.stdout, encoding="utf-8")
    if any(s in result.stdout for s in (
        "PermissionError", "WinError", "Access is denied", "Access denied",
    )):
        print(result.stdout[-2000:], flush=True)
        raise SystemExit("Permission blocker; report exact command to root.")
    return result

for source in sorted(OUT.glob("[0-9][0-9]-*.cpp")):
    if len(sys.argv) > 1 and source.stem[:2] not in sys.argv[1:]:
        continue
    rel = source.relative_to(ROOT).as_posix()
    records = [record for record in records if record["source"] != rel]
    cmd = [
        sys.executable, "tools/explain_mismatch.py", SYMBOL,
        "--rva", "0x8304A", "--size", "1156", "--source", rel,
        "--context", "5",
    ]
    p = command_to_log(cmd, source.stem + ".explain.txt")
    record = {"source": rel, "explain_exit": p.returncode}
    records.append(record)
    if p.returncode != 0 or "Traceback" in p.stdout or "fatal error" in p.stdout:
        record["error"] = p.stdout[-1800:]
        (OUT / "results.json").write_text(json.dumps(records, indent=2))
        print(json.dumps(record), flush=True)
        continue
    obj = build.obj_path(source)
    assert obj.exists(), obj
    stamp = obj.stat().st_mtime_ns
    q = command_to_log(
        [sys.executable, "tools/align_diff.py", rel, SYMBOL, "0x8304A", "1156"],
        source.stem + ".align.txt",
    )
    record["align_exit"] = q.returncode
    row = dict(name=SYMBOL, target_rva="0x0008304A", target_size="1156",
               source=rel, notes="")
    got = build.compile_function(row, build.load_symbol_map(), obj)
    assert obj.stat().st_mtime_ns == stamp, "Read-only measurement changed object"
    raw, unused_relocs = build.read_object_symbol_bytes(obj, SYMBOL)
    record.update(
        emitted=len(raw),
        score=permute.fitness(got["bytes"], got["target"]),
        unresolved=got["unresolved"],
        eh=eh_verify.verify_row(eh_verify.Image(), eh_verify.Obj(obj), SYMBOL, 0x8304A),
        align_summary=[line for line in q.stdout.splitlines()
                       if line.startswith(("ours ", "== structural", "total size"))],
    )
    if "contains" in source.stem:
        c = command_to_log(
            [sys.executable, "tools/explain_mismatch.py", CONTAINS,
             "--rva", "0x30D111", "--size", "304", "--source", rel, "--context", "3"],
            source.stem + ".contains.txt",
        )
        record["contains_exit"] = c.returncode
        record["contains_summary"] = [
            line for line in c.stdout.splitlines()
            if any(key in line for key in (
                "compiled symbol size:", "first diff:", "classification:", "unresolved:"
            ))
        ]
    (OUT / "results.json").write_text(json.dumps(records, indent=2))
    print(json.dumps(record), flush=True)


subprocess.run([sys.executable, str(SUPPORT / "audit.py.txt"),
                *sys.argv[1:]], cwd=ROOT, check=True)
