from headmerge import *

checks=[
 ('','p'),('','p!=0'),('','!(p==0)'),('','(bool)p'),('','!!p'),
 ('','(unsigned)p'),('','(unsigned)p>0'),('','(unsigned)p>=1'),
 ('','((unsigned)p-1)<0xffffffffu'),
 ('','((unsigned)p+0x80000000u)!=0x80000000u'),
 ('','((unsigned)p^0x80000000u)!=0x80000000u'),
 ('','(int)p<0 || (int)p>0'),('','!((int)p>=0 && (int)p<=0)'),
 ('','(p?true:false)'),('','(p?1:0)'),('','(p?0:1)==0'),
 ('','(p==0)==false'),('','(p!=0)==true'),
 ('static __forceinline bool validDraw(Drawable *v) {return v!=0;}\n','validDraw(p)'),
 ('static __forceinline bool validDraw(Drawable *const &v) {return v!=0;}\n','validDraw(p)'),
 ('static __forceinline int validDraw(Drawable *v) {return v?1:0;}\n','validDraw(p)'),
 ('static __forceinline bool validDraw(Drawable *v) {if(v)return true;return false;}\n','validDraw(p)'),
 ('static __forceinline bool validDraw(Drawable *v) {if(!v)return false;return true;}\n','validDraw(p)'),
 ('static __forceinline bool validDraw(Drawable *v) {return (unsigned)v>0;}\n','validDraw(p)'),
]
for dtor,member,conversion in itertools.product(range(3),range(2),range(3)):
 dt=['','~DrawTruth(){}','~DrawTruth() throw() {}'][dtor]
 ty='bool' if member else 'Drawable *'
 value='v!=0' if member else 'v'
 expression=['value!=0','value?true:false','!(!value)'][conversion]
 helper='struct DrawTruth { '+ty+' value; DrawTruth(Drawable *v):value('+value+'){} '+dt+' operator bool() const {return '+expression+';} };\n'
 checks.append((helper,'DrawTruth(p)'))

items=[]
family=sys.argv[1] if len(sys.argv)>1 else 'nullcheck'
topologies=[1,2,3,4,5,6,7,13,14,16,17,18,19] if family=='nested' else [0,8,9]
for f,c,order,contract in itertools.product(topologies,range(len(checks)),range(1) if family=='nested' else range(2),range(2)):
 helper,cond=checks[c]
 form=forms[f].replace('if(p)', 'if('+cond+')',1).replace('p?p->m_object','('+cond+')?p->m_object',1)
 form=form.replace('if(!p)','if(!('+cond+'))',1).replace('switch(p!=0)','switch(bool('+cond+'))',1).replace('while(p)','while('+cond+')',1).replace('for(;p;)','for(;'+cond+';)',1)
 if order:form=form.replace('draw=p; obj=p->m_object;', 'obj=p->m_object; draw=p;')
 src=BASE[:lo]+decl+form+'\n'+BASE[hi:]
 if not contract:src=src.replace('template <> void StringBase<char>::releaseBuffer() throw();\n','')
 src=src.replace('void W3DDisplay::gatherDebugStats(void)',helper+'void W3DDisplay::gatherDebugStats(void)')
 items.append(('nc%02d%02d%d%d'%(f,c,order,contract),src))
if __name__=='__main__':batch(items,family)
