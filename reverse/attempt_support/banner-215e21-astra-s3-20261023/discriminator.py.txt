import sys, subprocess
sys.path.insert(0, 'tools')
import build
from pathlib import Path

root = Path.cwd()
out = root / 'build/s3-banner-review'
orig = root / 'Code/GameEngine/Source/GameClient/GUI/InGame/Tactical/BannerUI.cpp'
text = orig.read_text().replace(
    '// cl: ', '// cl: /I' + orig.parent.relative_to(root).as_posix() + ' ', 1)

FIELDS = '''	unsigned int f0;
	int f4;
	const void *f8;
	int f0C;
	int f10;
	int f14;
	bool f18;
'''
STORES = '''	f0 = (unsigned int)-1;
	f4 = 0;
	f8 = 0;
	f0C = 0;
	f10 = 0;
	f14 = 0;
	f18 = false;
'''

VARIANTS = {
    'd_ctor': f'struct RvaDiscCtor {{ {FIELDS} RvaDiscCtor(); }};\n'
              f'RvaDiscCtor::RvaDiscCtor() {{ {STORES} }}\n',
    'd_void': f'struct RvaDiscVoid {{ {FIELDS} void clear(); }};\n'
              f'void RvaDiscVoid::clear() {{ {STORES} }}\n',
    'd_retthis': f'struct RvaDiscRet {{ {FIELDS} RvaDiscRet *ret(); }};\n'
                 f'RvaDiscRet *RvaDiscRet::ret() {{ {STORES} return this; }}\n',
}

retail26 = build.read_target_bytes(0x215E21, 26)
anchor = 'int BannerUI::CreateBanner(int bannerID)'
for name, snippet in VARIANTS.items():
    data = text.replace(anchor, snippet + '\n' + anchor, 1)
    src = out / (name + '.cpp')
    obj = out / (name + '.obj')
    src.write_text(data)
    cmd, env = build.compiler_command(src, obj)
    p = subprocess.run(cmd, env=env, capture_output=True, text=True)
    if p.returncode:
        print(name, 'COMPILE FAIL'); print(p.stdout + p.stderr); continue
    for sym in build.defined_code_symbols(obj):
        if 'RvaDisc' in sym:
            b, rs = build.read_object_symbol_bytes(obj, sym, code_only=True)
            b = bytes(b)
            print(f'{name}: {sym} size {len(b)} ==retail26 {b == retail26} relocs {rs}')
            if b != retail26:
                print('   ', b.hex())

print()
print('=== retail boundary bytes ===')
md_bytes = build.read_target_bytes(0x215E00, 0x40)
from capstone import Cs, CS_ARCH_X86, CS_MODE_32
md = Cs(CS_ARCH_X86, CS_MODE_32)
for i in md.disasm(md_bytes, 0x215E00):
    mark = ''
    if i.address == 0x215E20: mark = '  <== end of INI fn (row 278B: 0x215D0B..0x215E21)'
    if i.address == 0x215E21: mark = '  <== 26B body start (CANDIDATE)'
    if i.address == 0x215E26: mark = '  <== old 21B row start (INTERIOR)'
    if i.address == 0x215E3A: mark = '  <== 26B ret'
    if i.address == 0x215E3B: mark = '  <== next rowed entry (7B getter)'
    print(f'  {i.address:08X}: {i.bytes.hex():<16} {i.mnemonic} {i.op_str}{mark}')
