import json,sys
from pathlib import Path
from wdbg import Dbg
W=Path(__file__).resolve().parent
mode=sys.argv[1];src=Path(sys.argv[2]).resolve() if len(sys.argv)>2 else W/'clean-base.cpp';tag='choose-'+mode
d=Dbg(src,W/(tag+'.obj'),W/(tag+'.log'))
active=False;data=[];current=None
try:
 d.prompt();d.prompt('set $CanDeferOnBPByAddr = 1');d.prompt('break *0x107b0010')
 for _ in range(1500):
  out=d.prompt('cont',60)
  if 'terminated' in out:print('terminated',flush=True);break
  e=d.reg('eip')
  if e==0x107b0010:
   f=d.reg('esi');fn=d.string(d.words(d.words(f,1)[0]+4,1)[0])
   if '?adjustCameraMovement@PolygonTrigger@@' not in fn:continue
   active=True;d.prompt('break *0x1076fe13');d.prompt('break *0x1076fe56');d.prompt('break *0x107b0029')
  elif active and e==0x1076fe13:
   t=d.reg('ecx');w=d.words(t,16);current=None
   if w[1]!=0x204:continue
   p=w[10];q=d.words(p,20);r=d.words(q[0],20)
   def arg(x):
    s=x[7] or x[6];return dict(p=s,words=d.words(s,24)) if x[2]&255 in [1,2,4] else dict(words=x)
   current=dict(t=t,w=w,a=arg(q),b=arg(r),operands=[p,q[0]],nodes=[q,r]);data.append(current)
   print('add',len(data),current['a']['words'][7],current['b']['words'][7],flush=True)
   if mode.startswith('flip') and len(data)>(3 if mode=='flip2' else 2):
    d.prompt('set *(unsigned int*)0x%x=0x%x'%(t+40,q[0]));d.prompt('set *(unsigned int*)0x%x=0x%x'%(q[0],p));d.prompt('set *(unsigned int*)0x%x=0x%x'%(p,r[0]))
  elif active and e==0x1076fe56:
   if current:current['resultflags']=d.words(d.reg('esp')+24,6);print('flags',current['resultflags'],flush=True);current=None
  elif e==0x107b0029:
   # Existing breakpoints only matter during this parent, no further lowering.
   active=False
 else:raise RuntimeError('too many events')
finally:
 (W/(tag+'.json')).write_text(json.dumps(data,indent=1))
 if d.p.poll() is None:
  try:d.prompt('quit')
  except Exception:d.p.kill()
 d.log.close()
