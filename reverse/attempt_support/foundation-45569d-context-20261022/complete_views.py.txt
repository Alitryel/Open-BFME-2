from search import *
from context import getters,mutators,amount_local
import itertools

def views(s,mask,matrix):
 if matrix:
  a=s.index('class Matrix3D\n{');b=s.index('\n};',a)+3
  if matrix==1:decl='class Matrix3D {public: Real Get_Z_Rotation() const; float m[3][4];};'
  else:decl='#include "matrix3d.h"'
  s=s[:a]+decl+s[b:]
  s=s.replace('unsigned char m_pad009[0x38 - 0x09];','',1)
  if matrix==2:s=s.replace('/GX /I.','/GX /I. /Ireference/open-bfme-1/game/Libraries/Source/WWVegas/WWMath /Ireference/open-bfme-1/game/Libraries/Source/WWVegas/WWLib',1)
 if mask&1:
  s=s.replace('class Thing\n{\npublic:','class Thing\n{\npublic:\n virtual ~Thing();',1)
  s=s.replace('\tvoid *m_vptr;\n\tconst ThingTemplate *m_template;','\tconst ThingTemplate *m_template;',1)
 if mask&2:
  s=s.replace('class ThingTemplate\n{\npublic:','class ThingTemplate\n{\npublic:\n virtual ~ThingTemplate();',1)
  s=s.replace('unsigned char m_pad000[0x108];','unsigned char m_pad004[0x108-4];',1)
 if mask&4:
  s=s.replace('class Rva003B0D7C\n{\npublic:','class Rva003B0D7C\n{\npublic:\n int m_header,m_balance,m_playerIndex;',1)
  s=s.replace('unsigned char m_pad091[0x2EC - 0x91];','unsigned char m_pad09c[0x2EC - 0x9c];',1)
 if mask&8:
  s=s.replace('class Rva0039BAD2\n{\npublic:','class Rva0039BAD2\n{\npublic:\n virtual ~Rva0039BAD2();',1)
 if mask&16:
  s=s.replace('UnsignedInt m_kindOf[6];','UnsignedInt m_kindOf[7];',1)
  s=s.replace('unsigned char m_pad120[0x4D0 - 0x120];','unsigned char m_pad124[0x4D0 - 0x124];',1)
 if mask&32:
  s=s.replace('class AI\n{\npublic:','class AI\n{\npublic:\n virtual ~AI();',1)
  s=s.replace('unsigned char m_pad00[0x10];','unsigned char m_pad04[0x10-4];',1)
 if mask&64:s=getters(s)
 if mask&128:s=mutators(s)
 if mask&256:s=amount_local(s)
 return s

tests={}
for matrix,mask in itertools.product(range(3),range(512)):
 tests[f'views-{matrix}-{mask:03d}']=views(BASE,mask,matrix)
run(tests,'complete-views')
