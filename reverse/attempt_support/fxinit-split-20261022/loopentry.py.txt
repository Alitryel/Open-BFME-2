import sys,itertools,json,time
from pathlib import Path
from concurrent.futures import ThreadPoolExecutor
sys.path.insert(0,'build/fxinit-seat22');import search as h
h.HERE=Path('build/fxinit-split22').resolve()
SPECS=[('t','int','textureCount'),('i','unsigned','(unsigned)mesh->Get_Vertex_Count()'),('pass','unsigned','(unsigned)m_Passes')]
def variants():
 for forms,scope,neg,post in itertools.product(itertools.product(range(6),repeat=3),range(2),range(2),range(2)):
  src=h.BASE
  for (var,typ,bound),form in zip(SPECS,forms):
   token=f'for({typ} {var}=0;{var}<{bound};++{var}) {{';a=src.index(token);start=a+len(token);depth=1;p=start
   while depth:
    if src[p]=='{':depth+=1
    if src[p]=='}':depth-=1
    p+=1
   body=src[start:p-1];cond=f'!({var}>={bound})' if neg else f'{var}<{bound}'
   update=f'{var}++;' if post else f'++{var};'
   init=f'{typ} {var}=0;'
   if scope:
    src=src[:a]+src[a:];init=f'{var}=0;'
   if form==0:code=f'for({init}{cond};{update[:-1]}) {{{body}}}'
   if form==1:code=init+f'while({cond}) {{{body}{update}}}'
   if form==2:code=init+f'if({cond}) do {{{body}{update}}}while({cond});'
   if form==3:code=init+f'for(;;) {{if(!({cond}))break;{body}{update}}}'
   if form==4:code=init+f'for(;{cond};) {{{body}{update}}}'
   if form==5:code=init+f'goto Check_{var};Again_{var}:{{{body}{update}}}Check_{var}:if({cond})goto Again_{var};'
   src=src[:a]+'{'+code+'}'+src[p:]
  if scope:
   src=src.replace('bool FXShaderGeometry::Init(MeshModelClass *mesh)\n{','bool FXShaderGeometry::Init(MeshModelClass *mesh)\n{\nint t;unsigned i,pass;')
  yield 'loopentry_'+''.join(map(str,forms))+f'{scope}{neg}{post}',src
if __name__=='__main__':
 t=time.time()
 with ThreadPoolExecutor(max_workers=8)as pool:r=list(pool.map(h.run_one,variants()))
 r.sort(key=lambda x:(x.get('nd',99999),x.get('cost',99999)))
 (h.HERE/'loopentry-results.json').write_text(json.dumps(r,indent=2))
 print(json.dumps({'count':len(r),'valid':sum('nd'in x and not x.get('unresolved')for x in r),'seconds':time.time()-t,'best':r[:8]},indent=2))
