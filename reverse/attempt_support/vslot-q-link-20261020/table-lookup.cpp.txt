class Rva00669410

Rva00669410Entry *Rva00669410::rva00669410(const Rva00669410Arg *arg)
{
	Int id = arg->m_1C;
	if (id == -1)
		return 0;
	for (Rva00669410Entry **it = g_rva00669410Table; *it; ++it)
	{
		if ((*it)->m_08 == id)

}
