import sys,json,itertools,time
from pathlib import Path
from concurrent.futures import ThreadPoolExecutor,as_completed
sys.path.insert(0,'build/fxinit-seat22');import search as h
h.HERE=Path('build/fxinit-admit23').resolve()
def variants():
 for shape,order,ctor,byref,local,placement,sgn in itertools.product(range(3),range(6),range(4),range(2),range(2),range(2),range(2)):
  ti='unsigned' if sgn else 'int'
  if shape==0:
   fields=[('const short *','map','remap.begin()'),('const unsigned short *','a','links[0]+i'),('const unsigned short *','b','links[1]+i')];calc='v.map[*v.a]+(v.map[*v.b]<<8)'
  elif shape==1:
   fields=[('const unsigned short *','base','links[0]'),(ti,'delta','links[1]-links[0]'),(ti,'index','i')];calc='map[v.base[v.index]]+(map[v.base[v.delta+v.index]]<<8)'
  else:
   fields=[('const short *','map','remap.begin()'),('const unsigned short *','base','links[0]'),(ti,'delta','links[1]-links[0]')];calc='v.map[v.base[i]]+(v.map[v.base[v.delta+i]]<<8)'
  fs=[fields[j]for j in list(itertools.permutations(range(3)))[order]]
  decl=''.join(t+' '+n+';' for t,n,a in fs)
  pars=','.join(t+' p'+n for t,n,a in fs);args=','.join(a for t,n,a in fs)
  ct=''
  if ctor:ct='__forceinline PackInput('+pars+'):'+','.join(n+'(p'+n+')' for t,n,a in fs)+'{}'
  if ctor==2:ct+='__forceinline PackInput(const PackInput &q):'+','.join(n+'(q.'+n+')' for t,n,a in fs)+'{}'
  if ctor==3:ct+='__forceinline PackInput(const PackInput &q){'+''.join(n+'=q.'+n+';' for t,n,a in fs)+'}'
  # Every field is used; no inert padding or fabricated object overlay.
  definition='struct PackInput{'+decl+ct+'};\n'
  extra=',const short *map' if shape==1 else ',unsigned i' if shape==2 else ''
  extargs=',remap.begin()' if shape==1 else ',i' if shape==2 else ''
  definition+='static __forceinline int PackInputValue('+('const PackInput &' if byref else 'PackInput ')+'v'+extra+'){return '+calc+';}\n'
  if not ctor:
   pre='PackInput packInput={'+args+'};';val='packInput'
  elif local:
   pre='PackInput packInput('+args+');';val='packInput'
  else:pre='';val='PackInput('+args+')'
  call='*(int *)(vertices+desc.Get_Basis_End())=PackInputValue('+val+extargs+');'
  s=h.BASE
  if placement and pre:
   # Hoist only loop-invariant views; index-bearing views stay at iteration start.
   where='  for(unsigned pass=0;' if shape==2 else '    *(Vector3 *)(vertices+desc.Get_Location_Offset(0))=positions[0][i];'
   s=s.replace(where,pre+'\n'+where)
   s=s.replace(h.R,call)
  else:s=s.replace(h.R,'{'+pre+call+'}')
  s=s.replace('bool FXShaderGeometry::Init(',definition+'bool FXShaderGeometry::Init(')
  yield f'valuebundle_{shape}{order}{ctor}{byref}{local}{placement}{sgn}',s
if __name__=='__main__':
 t=time.time();rs=[]
 with ThreadPoolExecutor(max_workers=8)as ex:
  for f in as_completed([ex.submit(h.run_one,x)for x in variants()]):
   rs.append(f.result())
   if len(rs)%256==0:print('tested',len(rs),'best',min(rs,key=lambda z:(z.get('nd',99999),z.get('cost',99999))),flush=True)
 rs.sort(key=lambda z:(z.get('nd',99999),z.get('cost',99999)))
 (h.HERE/'valuebundle-results.json').write_text(json.dumps(rs,indent=2))
 print(json.dumps({'count':len(rs),'seconds':time.time()-t,'errors':sum('error'in z for z in rs),'best':rs[:6]},indent=2))
