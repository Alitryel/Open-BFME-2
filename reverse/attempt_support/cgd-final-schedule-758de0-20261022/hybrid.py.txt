from itertools import product
from pathlib import Path
W=Path(__file__).resolve().parent;base=(W/'base.cpp').read_text();out=W/'hybrid';out.mkdir(exist_ok=True)
for idx,(pack,ref,copy,ctorfast,site0,site1,site2,itype,scope) in enumerate(product(range(2),repeat=9)):
 cc='__fastcall ' if ctorfast else ''; fld='const ContactVector &' if ref else 'const ContactVector *';val='p' if ref else '&p'
 defs='struct PArg { '+fld+' p; '+cc+'PArg(const ContactVector &p):p('+val+'){}'
 if copy:defs+=' '+cc+'PArg(const PArg &a):p(a.p){}'
 defs+=' };\n'
 if pack:
  defs+='struct PPair { PArg point,normal; '+cc+'PPair(const ContactVector &p,const ContactVector &n):point(p),normal(n){}'
  if copy:defs+=' '+cc+'PPair(const PPair &a):point(a.point),normal(a.normal){}'
  defs+=' };\n'
 rt='unsigned' if itype else 'void *';at='PPair args' if pack else 'PArg point,PArg normal';pv='PPair(point,normal)' if pack else 'point,normal'
 defs+='struct RawPrepareTable { void *unused[3]; '+rt+'(__fastcall *prepare)(ContactSource *self,'+at+'); };\n'
 if scope:
  defs+='inline void *rawPrepare(ContactSource *s,const ContactVector &point,const ContactVector &normal) { RawPrepareTable *table=*(RawPrepareTable **)s; return (void *)table->prepare(s,'+pv+'); }\n'
 else:defs+='inline void *rawPrepare(ContactSource *s,const ContactVector &point,const ContactVector &normal) { return (void *)(*(RawPrepareTable **)s)->prepare(s,'+pv+'); }\n'
 s=base.replace('struct ContactData',defs+'struct ContactData',1)
 p='node->second->source->prepare(node->point,node->normal)'
 for i,use in reversed(list(enumerate([site0,site1]))):
  if use:
   starts=[];n=0
   while 1:
    n=s.find(p,n)
    if n<0:break
    starts.append(n);n+=len(p)
   j=starts[i];s=s[:j]+'rawPrepare(node->second->source,node->point,node->normal)'+s[j+len(p):]
 if site2:s=s.replace('node->first->source->prepare(node->point,ContactVector(node->normal).negate())','rawPrepare(node->first->source,node->point,ContactVector(node->normal).negate())')
 (out/('v%04d.cpp'%idx)).write_text(s)
print('generated',idx+1)
