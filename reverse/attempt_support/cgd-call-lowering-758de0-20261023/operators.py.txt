from itertools import product
from pathlib import Path
import re
W=Path(__file__).resolve().parent
D=W/'operators';D.mkdir(exist_ok=True)
for i,(mask,syntax,c) in enumerate(product(range(8),range(8),range(2))):
 s=(W/('uniform-control.cpp' if c else 'bank.cpp')).read_text()
 s=s.replace('struct ContactSource {','struct TestInfo;\nstruct ContactSourcePrepare {',1)
 s=s.replace(' virtual void slot4()=0;', '};\nstruct ContactSourceTest:ContactSourcePrepare {\n virtual void slot4()=0;',1)
 s=s.replace(' virtual void respond(void *info)=0;', '};\nstruct ContactSource:ContactSourceTest {\n'+(' using ContactSourceTest::operator();\n' if mask&3 else '')+' virtual void respond(void *info)=0;',1)
 s=s.replace('virtual void *prepare(', 'virtual TestInfo *prepare(')
 s=s.replace('virtual bool test(void *info)', ('using ContactSourcePrepare::operator();\n virtual bool test(TestInfo *info)' if mask&1 else 'virtual bool test(TestInfo *info)'))
 s=s.replace('virtual void * __fastcall preparePair(', 'virtual TestInfo * __fastcall preparePair(')
 s=s.replace('inline void *altPrepare(', 'inline TestInfo *altPrepare(')
 s=s.replace('__forceinline void *__cdecl bridge(', '__forceinline TestInfo *__cdecl bridge(')
 for k,name in enumerate(['prepare','test','respond']):
  if mask&(1<<k):
   s=s.replace(('*' if name=='prepare' else ' ')+name+'(', ('*' if name=='prepare' else ' ')+'operator()(')
  pattern=re.compile(r'([A-Za-z_]\w*(?:->\w+)*)->'+name+r'\(')
  pos=0
  while True:
   m=pattern.search(s,pos)
   if not m:break
   a=m.end();depth=1;b=a
   while depth:
    if s[b]=='(':depth+=1
    elif s[b]==')':depth-=1
    b+=1
   arg=s[a:b-1]
   if name=='respond':arg='(void *)('+arg+')'
   receiver=m.group(1)
   receiver='(('+['ContactSourcePrepare','ContactSourceTest','ContactSource'][k]+' *)'+receiver+')'
   if mask&(1<<k):
    call=(('(*'+receiver+')(') if syntax&(1<<k) else (receiver+'->operator()('))+arg+')'
   else:call=receiver+'->'+name+'('+arg+')'
   s=s[:m.start()]+call+s[b:]
   pos=m.start()+len(call)
 (D/('v%03d.cpp'%i)).write_text(s)
print('generated',i+1)
