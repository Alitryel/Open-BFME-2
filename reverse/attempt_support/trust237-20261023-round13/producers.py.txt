from pathlib import Path
import hashlib, json, struct
import capstone, pefile

ROOT=Path.cwd(); OUT=ROOT/'build/trust237-round13'
TOOL=ROOT/'reference/open-bfme-1/inputs/toolchains/vs2003/Program Files/Microsoft Visual Studio .NET 2003/Vc7/bin'
md=capstone.Cs(capstone.CS_ARCH_X86,capstone.CS_MODE_32); md.detail=True
records=[]
for name in ['c2.dll','c1xx.dll','c1.dll','link.exe']:
    path=TOOL/name; pe=pefile.PE(str(path)); base=pe.OPTIONAL_HEADER.ImageBase
    hits={}
    for section in pe.sections:
        if not section.Characteristics&0x20000000: continue
        data=section.get_data(); start=base+section.VirtualAddress
        for p,b in enumerate(data):
            if b!=0x90: continue
            for q in range(max(0,p-14),p+1):
                i=next(md.disasm(data[q:q+15],start+q,count=1),None)
                if i is None or not q<=p<q+i.size: continue
                if not i.imm_size or not q+i.imm_offset<=p<q+i.imm_offset+i.imm_size: continue
                imm=[o.imm&0xffffffff for o in i.operands if o.type==capstone.CS_OP_IMM]
                if not any(v in [0x90,0x9090,0x90909090] for v in imm): continue
                if i.mnemonic not in ['mov','push','cmp']: continue
                hits[i.address]=dict(va=hex(i.address),raw=i.bytes.hex(),mnemonic=i.mnemonic,operands=i.op_str)
    rows=sorted(hits.values(),key=lambda r:int(r['va'],16))
    records.append(dict(file=str(path.relative_to(ROOT)),sha256=hashlib.sha256(path.read_bytes()).hexdigest(),candidates=rows))
    print(name,'overlapping candidate decodes',len(rows),flush=True)
    for r in rows: print(r['va'],r['mnemonic'],r['operands'],flush=True)
(OUT/'direct-nop-sites-overlap.json').write_text(json.dumps(records,indent=2)+'\n')

if __name__=='__main__':
    path=TOOL/'c2.dll'; pe=pefile.PE(str(path)); base=pe.OPTIONAL_HEADER.ImageBase
    # Bounded surrounding disassembly is only a classification aid. Its
    # starting address may itself be unaligned; table decodes remain possible.
    for row in records[0]['candidates']:
        va=int(row['va'],16)
        print('C2 context',hex(va),flush=True)
        for i in md.disasm(pe.get_data(va-base-12,48),va-12):
            print(hex(i.address),i.mnemonic,i.op_str,flush=True)
