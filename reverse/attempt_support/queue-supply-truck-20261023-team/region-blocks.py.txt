from pathlib import Path
import json
out=Path(__file__).resolve().parent
for label in ('baseline','gh','allguards_01','looseforce_02'):
 directory=(out.parent/'observer'/label) if label in ('baseline','gh') else out/label
 if not (directory/'regions.json').exists():continue
 regions=json.loads((directory/'regions.json').read_text())
 r=next(r for r in regions if r['round']==3 and r['role']=='BASE_VTABLE')
 print('\n',label,'round3')
 lines=[]
 for i,b in enumerate(r['blocks']):
  names=sorted({o['name'] for t in b['tuples'] for o in t['operands'] if o.get('name')})
  origins=[(n,t['opcode'],t['words'][8:12]) for n,t in enumerate(b['tuples']) if t['origin_refs']]
  membership=''.join('Y' if r['index'] in b['sets'][k] else '-' for k in ('0x4c','0x50'))
  s=f'{i:02d} {membership} bb={b["address"]:08x} n={len(b["tuples"])} origin={origins} names={names}'
  print(s);lines.append(s)
 (out/(label+'-round3-blocks.txt')).write_text('\n'.join(lines)+'\n')
