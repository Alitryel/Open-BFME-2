from pathlib import Path
import hashlib,json,sys,itertools,collections
ROOT=Path.cwd();OUT=ROOT/'build/trust237-round9';sys.path.insert(0,str(ROOT/'tools'))
import build,capstone
from variant import resolve_bytes
TARGET=build.read_target_bytes(0x237d3d,226)
md=capstone.Cs(capstone.CS_ARCH_X86,capstone.CS_MODE_32)
mn=[i.mnemonic for i in md.disasm(TARGET,0)]
sets={k:json.loads((OUT/(k+'.json')).read_text()) for k in ['frontends','mixed','hints']}
# Recompute all opcode inventories sequentially from retained raw objects.
for label,rows in sets.items():
 for r in rows:
  assert 'error' not in r,r
  raw=bytes.fromhex(r['raw']);ins=list(md.disasm(raw,0))
  assert len(raw)==r['size'] and hashlib.sha256(raw).hexdigest()==r['sha256']
  assert [i.address for i in ins if i.mnemonic=='nop']==r['nops']
  assert [i.address for i in ins if i.mnemonic=='pause']==r['pauses']
  assert not r['unresolved'],(label,r['name'],r['unresolved'])
summary={}
for label,rows in sets.items():
 summary[label]=dict(attempts=len(rows),successful=len(rows),distinct_raw=len({r['sha256'] for r in rows}),with_nop=sum(bool(r['nops']) for r in rows),with_nop16=sum(r['nop16'] for r in rows),unresolved_rel32=0)
 if label=='frontends':
  by=collections.defaultdict(dict)
  for r in rows:
   m=r['meta'];by[tuple(m[k] for k in ['predicate','tail','hoist','cpu','opt'])][m['frontend']]=r
  equal=sum(len({hashlib.sha256(bytes.fromhex(r['raw'])+json.dumps(r['relocations']).encode()).hexdigest() for r in group.values()})==1 for group in by.values())
  summary[label]['paired_three_frontend_groups']=len(by)
  summary[label]['identical_code_and_relocations_across_all_three']=equal
 if label=='mixed':
  malformed=[r for r in rows if r['meta']['extra']]
  assert all(bytes.fromhex(r['raw'])[-1]==1 for r in malformed)
  summary[label]['hidden_mode_ineligible_malformed_return']=len(malformed)
  summary[label]['unsupported_managed_push_syntax']=sum('C4081' in r['warnings'] for r in rows)
  by=collections.defaultdict(dict)
  for r in rows:
   m=r['meta'];by[tuple(m[k] for k in ['kind','opt','cpu','extra'])][m['pragma']]=r
  equal=sum(len({r['sha256'] for r in group.values()})==1 for group in by.values())
  summary[label]['four_pragma_groups']=len(by);summary[label]['all_four_pragma_bytes_equal']=equal
 if label=='hints':
  pause=[r for r in rows if r['meta']['hint']=='_mm_pause']
  assert all(len(r['pauses'])==r['meta']['count'] and bytes.fromhex(r['raw']).count(b'\xf3\x90')==r['meta']['count'] for r in pause)
  summary[label]['pause_variants']=len(pause);summary[label]['all_pauses_keep_f3_prefix']=True
 summary[label]['representatives']=[{k:r[k] for k in ['name','meta','size','prefix','diff','eq','sha256','ret']} for r in sorted({r['sha256']:r for r in rows}.values(),key=lambda r:(r['diff'],-r['prefix']))[:3]]
compiler=Path(build.vc71_root())/'Vc7/bin'
summary['input']={'bfme2':__import__('subprocess').check_output(['git','rev-parse','HEAD'],text=True).strip(),'bfme1':__import__('subprocess').check_output(['git','-C','reference/open-bfme-1','rev-parse','HEAD'],text=True).strip(),'model':'gpt-6.1-sol','retail_extent':226,'retail_sha256':hashlib.sha256(TARGET).hexdigest(),'compiler_files':{f:hashlib.sha256((compiler/f).read_bytes()).hexdigest() for f in ['cl.exe','c1.dll','c1xx.dll','c2.dll']},'bank_source_sha256':hashlib.sha256(Path('reverse/attempts/0x00237d3d.cpp').read_bytes()).hexdigest()}
raw,rel=build.read_object_symbol_bytes(Path('build/scratch/0x00237d3d.obj'),'?Rva00237D3DCheck@@YAHXZ',code_only=True)
resolved,unres=resolve_bytes(raw,rel,TARGET,0x237d3d,len(raw),{'_memset':[0x6291ae],'_WinVerifyTrust@12':[0x655652]})
assert not unres
assert hashlib.sha256(raw).hexdigest()=='23875f743d6816184d590366a5617a42b8e18aedc479c4d75864e0f6d61de0c9'
summary['retained_bank']=dict(size=len(raw),masked_prefix=next(i for i,(a,b) in enumerate(zip(resolved,TARGET)) if a!=b),diagnostic_differences=sum(a!=b for a,b in itertools.zip_longest(resolved,TARGET)),raw_sha256=hashlib.sha256(raw).hexdigest(),score=69/86)
em=json.loads((OUT/'frontends-emulation.json').read_text());summary['emulation']={k:v for k,v in em.items() if k!='rows'}
summary['setup_correction']='First 96 plain-C++ trials had C++ linkage on API declarations; corrected all to extern C and recompiled exactly those 96. Final 480 rows have no unresolved REL32. Totals exclude superseded trials.'
summary['checks']=['all raw hashes/instruction inventories checked sequentially','independent variant.py bank compilation','symbol-relocated emulation of 305 eligible objects in 28 distinct code/relocation groups']
(OUT/'summary.json').write_text(json.dumps(summary,indent=2)+'\n')
for label in sets:print(label,{k:v for k,v in summary[label].items() if k!='representatives'})
print('bank',summary['retained_bank']);print('emulation',summary['emulation'])
