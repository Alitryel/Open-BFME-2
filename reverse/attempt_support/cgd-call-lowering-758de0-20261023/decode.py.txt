import json,sys
from pathlib import Path
D=json.loads(Path(sys.argv[1]).read_text())
for s in D:
 nodes=s['nodes'];syms=s['syms']
 def sym(p):
  sw=syms.get(str(p),[])
  if 0x108a9804<=p<0x108a9b04:return 'R'+str((p-0x108a97a4)//96)
  if len(sw)>7:return 'H'+str(sw[7])
  return hex(p)
 def operand(p,seen=()):
  if not p:return 'null'
  if p in seen:return '<cycle:%x>'%p
  w=nodes.get(str(p),[])
  if len(w)<16:return '<uncaptured:%x>'%p
  k=w[2]&255;seen=seen+(p,)
  if k in [1,2,4]:return sym(w[6] if w[7] in [0,0x108a92e4] else w[7])
  if k in [5,6]:return ('addr' if k==5 else 'mem')+'('+operand(w[13],seen)+',off='+hex(w[12])+',f='+hex(w[9])+')'
  if k==7:return '#'+hex(w[6])
  if k==12:return 'meta('+hex(w[4])+','+hex(w[6])+')'
  return 'k%d@%x'%(k,p)
 def operands(p):
  r=[];seen=set()
  while p and p not in seen:
   seen.add(p);r.append(operand(p));w=nodes.get(str(p),[])
   if not w:break
   p=w[0]
  return '; '.join(r)
 print('\nSTAGE',s['stage'])
 for key,bw in s['bbs'].items():
  n=bw[7];lines=[];seen=set();calls=0
  while n and n not in seen:
   seen.add(n);w=nodes[str(n)];k=w[2]&255
   if k==15:calls+=1
   if k in [13,18]:args=operands(w[10])+' -> '+operands(w[11])
   elif k==15:args='target/args '+operands(w[10])+'; result '+operands(w[11])+' flags '+hex(w[14])
   elif k==17:args=operands(w[10])+' ; '+operands(w[11])
   else:args='k%d'%k
   lines.append('%08x op%04x %s'%(n,w[1],args))
   if n==bw[8]:break
   n=w[0]
  if calls:print('BB',key+'\n'+'\n'.join(lines))
