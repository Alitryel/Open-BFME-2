import sys
from pathlib import Path
sys.path.insert(0, str(Path(__file__).resolve().parents[2] / 'tools'))
import build
import capstone

md = capstone.Cs(capstone.CS_ARCH_X86, capstone.CS_MODE_32)
a = build.read_target_bytes(0x73BFE0, 874)
b = build.read_target_bytes(0x73C450, 874)
masked = set()
for rva in (0x73BFE0, 0x73C450):
    raw = build.read_target_bytes(rva - 8, 890)
    print(f'\nRVA {rva:#x}: before={raw[:8].hex()} after={raw[882:].hex()}')
    for ins in md.disasm(raw[8:882], rva + 0x400000):
        print(f'{ins.address:08X} {ins.bytes.hex():24s} {ins.mnemonic:8s} {ins.op_str}')
        if ins.mnemonic == 'call':
            at = ins.address - (rva + 0x400000)
            masked.update(range(at + 1, at + 5))
diffs = [(i, x, y) for i, (x, y) in enumerate(zip(a, b)) if x != y]
print('\nAll twin byte differences:', diffs)
print('Non-call-displacement differences:', [v for v in diffs if v[0] not in masked])
for rva, count in ((0x73BB40,159),(0x73CCC0,362)):
    print(f'\nABI evidence {rva:#x}, {count} bytes')
    for ins in md.disasm(build.read_target_bytes(rva,count),rva+0x400000):
        print(f'{ins.address:08X} {ins.bytes.hex():24s} {ins.mnemonic:8s} {ins.op_str}')
