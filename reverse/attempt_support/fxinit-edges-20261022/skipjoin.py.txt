import sys,itertools,json,time
from pathlib import Path
from concurrent.futures import ThreadPoolExecutor,as_completed
sys.path.insert(0,'build/fxinit-seat22');import search as h
h.HERE=Path('build/fxinit-edges22').resolve()
PASS='*(unsigned *)(vertices+desc.Get_Basis_End())=pass;'
ONE='*(int *)(vertices+desc.Get_Basis_End())=remap[links[0][i]];'
def variants():
 for mask,target,shape,scope,wrap in itertools.product(range(1,16),range(3),range(4),range(2),range(2)):
  cond=['!positions[1] && !normals[1]','!normals[1]','!positions[1]'][target]
  label=['CopyBoth','CopyNormal','CopyPosition'][target]
  # CopyPosition here is the P-only block; it must skip N on these edges.
  # Thus the P-only target is copied in a separate fragment shared with the native P arm.
  if target==2:
   skip='if(!positions[1]) goto FinishSkin;' # replaced below by target inside native P null arm
   pn='if(positions[1]) {'+h.P[len('if(positions[1]) '):]+'} else {CopyPosition: goto PositionDone;}\nPositionDone:\n'+h.N
   # A non-skin edge must bypass the normal copy too: require both pointers null.
   cond='!positions[1] && !normals[1]'
  elif target==1:pn=h.P+'\nCopyNormal:\n'+h.N
  else:pn='CopyBoth:\n'+h.P+'\n'+h.N
  jump='goto '+label+';'
  if shape==0:entry=f'if({cond}) {jump}'
  elif shape==1:entry=f'if(!({cond})) goto FinishSkin; {jump}'
  elif shape==2:entry=f'if({cond}) {{if(m_BonesPerVertex>0) {jump} else {jump}}}'
  else:entry=f'if({cond}) {{if(m_Passes>1) {jump} else {jump}}}'
  if wrap:
   # Extra duplicate live edge within the copy successor, as in the landed NetPacket recipe.
   pn=pn.replace(label+':', 'if(tangents) { '+label+': ; } else { ; }')
  paths=[]
  for bit,store in enumerate([PASS,'',ONE,'']):
   e=entry if mask>>bit&1 else ''
   paths.append((e+store if scope else store+e)+'goto FinishSkin;')
  arm='if(m_Passes>1) {'+paths[0]+'}\nelse if(m_BonesPerVertex>0 && links[0]) {\nif(m_BonesPerVertex==1) {'+paths[2]+'}\nelse if(m_BonesPerVertex==2) {\n'+h.R+'\n'+h.W+'\n'+pn+'\n}\nelse {'+paths[3]+'}\n}\nelse {'+paths[1]+'}\nFinishSkin:;\n'
  # Entry before a real pass/one-bone store would skip it: excluded rather than testing changed semantics.
  if scope and mask&5:continue
  yield f'skipjoin_{mask:02x}{target}{shape}{scope}{wrap}',h.BASE[:h.START]+arm+h.BASE[h.END:]
if __name__=='__main__':
 t=time.time();items=list(variants());r=[]
 with ThreadPoolExecutor(max_workers=8)as pool:
  for f in as_completed([pool.submit(h.run_one,x)for x in items]):
   r.append(f.result())
   if len(r)%256==0:print('tested',len(r),'best',min((x for x in r if 'nd'in x),key=lambda x:(x['nd'],x['cost'])),flush=True)
 r.sort(key=lambda x:(x.get('nd',99999),x.get('cost',99999)))
 (h.HERE/'skipjoin-results.json').write_text(json.dumps(r,indent=2))
 print(json.dumps({'count':len(r),'valid':sum('nd'in x and not x.get('unresolved')for x in r),'seconds':time.time()-t,'best':r[:4]},indent=2))
