from headmerge import *
# New timing hypothesis: pointer extension/truncation or wide predicates are
# lowered at a different stage from ordinary pointer null facts. These are
# equivalent for every 32-bit pointer, including high-bit addresses.
checks=[
 '(unsigned __int64)(unsigned)p != 0',
 '(unsigned __int64)(unsigned)p > 0',
 '(__int64)(int)p != 0',
 '!((__int64)(int)p == 0)',
 '(unsigned __int64)(unsigned)p >= 1',
 '(unsigned __int64)(unsigned)p - 1 < 0xffffffffffffffffui64',
 '((unsigned __int64)(unsigned)p << 32) != 0',
 '((unsigned __int64)(unsigned)p * 0x100000001ui64) != 0',
 '((unsigned __int64)(unsigned)p + 0x100000000ui64) > 0x100000000ui64',
 '((unsigned __int64)(unsigned)p + 0x80000000ui64) != 0x80000000ui64',
 '((__int64)(int)p < 0 || (__int64)(int)p > 0)',
 '((unsigned __int64)(unsigned)p | ((unsigned __int64)(unsigned)p << 32)) != 0',
]
values=['p','(Drawable*)(unsigned)(__int64)(int)p',
        '(Drawable*)(unsigned)((unsigned __int64)(unsigned)p * 0x100000001ui64)']
items=[]
for f,c,val,nul,contract in itertools.product([8,9,10,12],range(len(checks)),range(len(values)),range(2),range(2)):
 form=forms[f].replace('obj=0','obj='+( '(Object*)p' if nul else '0'))
 form=form.replace('p?p->m_object:0','p?p->m_object:'+( '(Object*)p' if nul else '0'))
 form=form.replace('if(p)','if('+checks[c]+')').replace('obj=p?','obj=('+checks[c]+')?')
 form=form.replace('draw=p;','draw='+values[val]+';')
 src=BASE[:lo]+decl+form+'\n'+BASE[hi:]
 if not contract:src=src.replace('template <> void StringBase<char>::releaseBuffer() throw();\n','')
 items.append(('wj%02d%02d%d%d%d'%(f,c,val,nul,contract),src))
if __name__=='__main__':batch(items,'widejoin')
