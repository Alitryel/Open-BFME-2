from pathlib import Path
import json,capstone,collections
from capstone.x86 import X86_OP_REG,X86_OP_MEM
p=Path('build/trust237-round3');rows=json.loads((p/'padding.json').read_text())
md=capstone.Cs(capstone.CS_ARCH_X86,capstone.CS_MODE_32);md.detail=True
recipes=collections.defaultdict(list)
def noop(i):
 if i.mnemonic=='nop':return True
 if i.mnemonic=='mov' and len(i.operands)==2 and all(o.type==X86_OP_REG for o in i.operands):return i.operands[0].reg==i.operands[1].reg
 if i.mnemonic=='lea' and len(i.operands)==2 and i.operands[0].type==X86_OP_REG and i.operands[1].type==X86_OP_MEM:
  m=i.operands[1].mem
  return m.base==i.operands[0].reg and not m.index and m.disp==0
 return False
for r in rows:
 if 'error' in r:continue
 raw=bytes.fromhex(r['raw']);ins=list(md.disasm(raw,0));starts={i.address:n for n,i in enumerate(ins)}
 for edge in r['backedges']:
  head=edge['to'];idx=starts[head];first=idx
  while first and noop(ins[first-1]):first-=1
  if first==idx:continue
  blob=raw[ins[first].address:head]
  key=(r['cpu'],r['opt'],len(blob),blob.hex())
  recipes[key].append(r['n'])
summary=[]
for (cpu,opt,size,raw),cases in sorted(recipes.items()):
 summary.append(dict(cpu=cpu,opt=opt,size=size,bytes=raw,cases=len(cases),example=cases[0],disasm=[i.mnemonic+' '+i.op_str for i in md.disasm(bytes.fromhex(raw),0)]))
(p/'padding_recipes.json').write_text(json.dumps(summary,indent=2))
print('Recognized contiguous identity-instruction padding recipes',len(summary),'maximum size',max(r['size'] for r in summary))
for r in summary:print(r)
