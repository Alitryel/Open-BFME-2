extern "C" unsigned __cdecl _udiv64(unsigned __int64,unsigned,unsigned *);
#pragma intrinsic(_udiv64)
extern "C" unsigned __int64 __stdcall _aulldiv(unsigned __int64 n,unsigned __int64 d);
unsigned __int64 __stdcall _aulldiv(unsigned __int64 n,unsigned __int64 d) {
 unsigned nh=(unsigned)(n>>32),nl=(unsigned)n,dh=(unsigned)(d>>32),dl=(unsigned)d;
 if(!dh) {
  unsigned qh=nh/dl,rem=nh%dl;
  unsigned ignored;
  unsigned ql=_udiv64(((unsigned __int64)rem<<32)|nl,dl,&ignored);
  return ((unsigned __int64)qh<<32)|ql;
 }
 unsigned __int64 nn=n,dd=d;
 do { nn>>=1;dd>>=1; } while(dd>>32);
 unsigned ignored;
 unsigned q=_udiv64(nn,(unsigned)dd,&ignored);
 unsigned __int64 prod=(unsigned __int64)q*dl;
 unsigned cross=q*dh;
 unsigned high=(unsigned)(prod>>32)+cross;
 if(high<cross || high>nh || (high==nh && (unsigned)prod>nl))--q;
 return q;
}
