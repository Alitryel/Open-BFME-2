from pathlib import Path
import json,hashlib,sys
ROOT=Path.cwd();OUT=ROOT/'build/trust237-round10';sys.path.insert(0,str(ROOT/'tools'));import build
NAME='?Rva00237D3DCheck@@YAHXZ'
rows=json.loads((OUT/'exchanges.json').read_text());good=[r for r in rows if 'error' not in r]
# Audit hashes and instruction inventories sequentially, independent of the
# parallel scoring driver. A decoder is private to every driver invocation.
import capstone
for r in good:
 raw,rel=build.read_object_symbol_bytes(OUT/(r['name']+'.obj'),NAME,code_only=True)
 assert hashlib.sha256(raw).hexdigest()==r['sha256']
 decoder=capstone.Cs(capstone.CS_ARCH_X86,capstone.CS_MODE_32)
 assert [i.address for i in decoder.disasm(raw,0) if i.mnemonic=='nop']==r['nops']
 assert rel==[tuple(x) for x in r['relocations']]
raw,rel=build.read_object_symbol_bytes(ROOT/'build/scratch/0x00237d3d.obj',NAME,code_only=True)
assert len(raw)==219 and hashlib.sha256(raw).hexdigest()=='23875f743d6816184d590366a5617a42b8e18aedc479c4d75864e0f6d61de0c9'
unique={r['sha256']:r for r in good}
best=sorted(unique.values(),key=lambda r:(r['diff'],-r['prefix']))[:4]
emulation=json.loads((OUT/'exchanges-emulation.json').read_text());emulation.pop('rows');emulation['detailed_receipt_sha256']=hashlib.sha256((OUT/'exchanges-emulation.json').read_bytes()).hexdigest()
result=dict(attempts=len(rows),successful=len(good),distinct_raw=len(unique),with_nop=sum(bool(r['nops']) for r in good),with_nop16=sum(r['nop16'] for r in good),unresolved_rel32=sum(bool(r['unresolved']) for r in good),best_diagnostic_cases=[{k:r[k] for k in ['name','meta','size','prefix','diff','eq','sha256']} for r in best],bank=dict(size=len(raw),sha256=hashlib.sha256(raw).hexdigest(),score='69/86',prefix=205,differences=21),emulation=emulation,compiler=json.loads((OUT/'exchange-compiler.json').read_text())['compiler_sha256'],trace=json.loads((OUT/'exchange.trace.json').read_text()),control=json.loads((OUT/'exchange-control-object.json').read_text()))
(OUT/'summary.json').write_text(json.dumps(result,indent=2)+'\n')
print('verified inventory:',len(rows),'compiles,',len(unique),'raw bodies, no NOP; emulation',result['emulation']['executions'],'passed; bank unchanged')
