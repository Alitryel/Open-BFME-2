from pathlib import Path
import sys,json,hashlib,subprocess,concurrent.futures
root=Path(__file__).resolve().parents[1];sys.path.insert(0,str(root/'tools'));import build
out=root/'build/zh-api-followups-oct20';out.mkdir(exist_ok=True)
zh=build.ZH_REFERENCE_ROOT
previous=json.loads((root/'build/zh-remaining-headers-oct20/inputs.json').read_text())
template=previous[0]['flags'];base=template[:]
display=zh/'GameEngine/Include/GameClient/Display.h'
overlay=out/'include/GameClient/Display.h';overlay.parent.mkdir(parents=True,exist_ok=True)
old=display.read_text();needle='typedef void (DebugDisplayCallback)( DebugDisplayInterface *debugDisplay, void *userData, FILE *fp = NULL );';assert old.count(needle)==1
overlay.write_bytes(old.replace(needle,needle.replace(' = NULL','')).encode())
# Keep the existing sweep PCH's engine headers, but move this isolated copy
# away from the Windows stand-ins and remove that directory from its search.
(out/'include/PreRTS.h').write_bytes((root/'reference/shims/sweep/PreRTS.h').read_bytes())
tasks=[]
for task in previous:
    donor=root/task['donor'];name=donor.name
    if name not in ('IMEManager.cpp','urllaunch.cpp','Debug.cpp'):continue
    flags=task['flags'][:]
    if name=='IMEManager.cpp':flags=['-I'+(out/'include').relative_to(root).as_posix(),'-I'+display.parent.relative_to(root).as_posix()]+[f for f in flags if 'shims/sweep' not in f.replace('\\','/')]
    prefix='#include <windows.h>\n'
    if name=='IMEManager.cpp':prefix+='#include <imm.h>\n'
    if name=='urllaunch.cpp':prefix+='#include <stdio.h>\n#include <wchar.h>\n#include <tchar.h>\n'
    if name=='Debug.cpp':prefix+='#include "Common/AsciiString.h"\n#include "Common/StackDump.h"\n'
    tasks.append({**task,'flags':flags,'prefix':prefix,'included':donor.as_posix(),'adaptation':'Display typedef default argument removed' if name=='IMEManager.cpp' else 'real missing header declarations'})
for name in ('dazzle.cpp','textdraw.cpp'):
    donor=zh/'Libraries/Source/WWVegas/WW3D2'/name
    original=donor.read_text()
    variants=[('texture-no-arg','font->Peek_Texture( ch )','font->Peek_Texture()')] if name=='textdraw.cpp' else [('pow-double','pow(params->ic.history_weight,time_ms)','pow((double)params->ic.history_weight,(double)time_ms)'),('pow-float','pow(params->ic.history_weight,time_ms)','pow(params->ic.history_weight,(float)time_ms)')]
    for kind,needle,replacement in variants:
        assert original.count(needle)==1
        copy=out/(kind+'_'+name);adapted=original.replace(needle,replacement)
        if name=='textdraw.cpp':
            assert adapted.count("font->Peek_Texture('A')")==1
            adapted=adapted.replace("font->Peek_Texture('A')",'font->Peek_Texture()')
        copy.write_bytes(adapted.encode())
        for profile,extra in [('original',[]),('retail',['-O1','-G7','-arch:SSE2'])]:
            tasks.append({'donor':donor.relative_to(root).as_posix(),'sha256':hashlib.sha256(donor.read_bytes()).hexdigest(),'revision':previous[0]['revision'],'profile':profile+'-'+kind,'flags':base+extra,'prefix':'#include <windows.h>\n','included':copy.as_posix(),'adaptation':kind})
for t in tasks:
    def render(f):return '-I"'+f[2:]+'"' if f.startswith('-I') and ' ' in f else f
    text='// cl: '+' '.join(map(render,t['flags']))+'\n'
    if build.source_needs_stlport(root/t['donor']):text+='// stlport\n'
    text+=t['prefix']+'#include "'+t['included']+'"\n'
    trial=out/(t['profile']+'_wrapper_'+Path(t['donor']).name);trial.write_bytes(text.encode());t['trial']=trial.relative_to(root).as_posix()
    assert build.source_extra_flags(trial)==t['flags']
(out/'inputs.json').write_bytes((json.dumps(tasks,indent=2)+'\n').encode())
def run(t):
    p=subprocess.run([sys.executable,'tools/place_bodies.py','--min','8',t['trial']],cwd=root,capture_output=True,text=True,encoding='utf8',errors='replace',timeout=240)
    output=p.stdout+p.stderr;(root/t['trial']).with_suffix('.log').write_bytes(output.encode())
    return {**t,'compile_failed':p.returncode!=0 or 'COMPILE FAILED' in output,'errors':[s for s in output.splitlines() if 'error ' in s][:6],'leads':[s for s in output.splitlines() if s.startswith(('BODY\t','PIN\t'))]}
results=[]
with concurrent.futures.ThreadPoolExecutor(max_workers=3) as pool:
    for r in pool.map(run,tasks):
        results.append(r);(out/'results.json').write_bytes((json.dumps(results,indent=2)+'\n').encode())
        print(Path(r['donor']).name,r['profile'],'failed',r['compile_failed'],'leads',r['leads'],'errors',r['errors'][:2],flush=True)
