tests=[]
for paired,order,ctor,dtor,passing,ret,scope in itertools.product(range(2),range(2),range(2),range(2),range(4),range(3),range(2)):

 helper='struct MakeCodeLife { '+fields+' __forceinline MakeCodeLife('+ty+'p)'+(':'+init+'{}'if ctor==0 else '{code=0;'+name+'=p;}')+' __forceinline ~MakeCodeLife(){'+('code=0;'if dtor else '')+'} };\n'
 arg=['MakeCodeLife value','const MakeCodeLife value','const MakeCodeLife&value','MakeCodeLife&value'][passing]
