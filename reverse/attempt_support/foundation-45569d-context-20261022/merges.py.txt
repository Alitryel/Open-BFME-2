 if n==1:return lookups+'if (!newMember || !oldMember) { '+tail+' }\n'+copies+'if (isRebuild) { '+tail+' }\n'+cost+tail

def map_duplicate(s):

 e=s.index('\n\tBfmeItemE63 *newMember =',a)
 rest=s[e:s.rfind('}')]
 return s[:a]+'if (!obj->getTemplate()->isKindOf(KINDOF_3C)) { TheAI->pathfinder()->AddObjectToPathfindMap(obj);\n'+rest+' } else { '+rest+' }\n}\n'
def rebuild_duplicate(s):
 a=s.index('\tif (isRebuild)\n\t{\n\t\tobj->clearModelConditionState(69);')
 e=s.index('\n\tobj->rva00293E64(NULL);',a)
 block=s[a:e]
 rest=s[e:s.rfind('}')]
 # Both paths perform the same remaining work and return exactly once.
 return s[:a]+block[:-2]+rest+' } else { '+rest+' }\n}\n'
