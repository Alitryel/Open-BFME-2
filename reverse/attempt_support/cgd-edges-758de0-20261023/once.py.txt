from pathlib import Path
import itertools,json
W=Path(__file__).resolve().parent;b=(W/'base.cpp').read_text();D=W/'once';D.mkdir(exist_ok=True)
state='  if(node->state==0)\n   node->state=node->first->source->test(node->second->source->prepare(node->point,node->normal))?1:2;'
store=state.split('\n   ')[1]
first='  node->first->source->respond(bridge(node->second,node->point,node->normal));'
last='  node->second->source->respond(altPrepare(node->first->source,node->point,ContactVector(node->normal).negate()));'
def wrap(s,k,tag):
 return [s,f'do {{ {s} }} while(0);',
  f'for(int {tag}=0;{tag}<1;++{tag}) {{ {s} }}',
  f'for(int {tag}=1;{tag};--{tag}) {{ {s} }}',
  f'for(bool {tag}=true;{tag};{tag}=false) {{ {s} }}',
  f'{{ int {tag}=1; while({tag}) {{ {s} {tag}=0; }} }}',
  f'{{ bool {tag}=true; do {{ {s} {tag}=false; }} while({tag}); }}',
  f'for(;;) {{ {s} break; }}'][k]
states=[state, 'while(node->state==0) { '+store+' }',
 'while(node->state==0) { '+store+' break; }',
 'for(;node->state==0;) { '+store+' break; }',
 'if(node->state==0) do { '+store+' } while(node->state==0);',
 'if(node->state==0) for(;;) { '+store+' if(node->state!=0) break; }',
 'for(;;) { if(node->state!=0) break; '+store+' break; }',
 wrap(state,4,'testOnce')]
rows=[]
for n,(a,c,d,bridge,mixed) in enumerate(itertools.product(range(8),range(8),range(8),range(2),range(2))):
 f=first if bridge else first.replace('bridge(node->second,','node->second->source->prepare(')
 r=last if mixed else last.replace('altPrepare(node->first->source,','node->first->source->prepare(')
 s=b.replace(state,states[a]).replace(first,wrap(f,c,'firstOnce')).replace(last,wrap(r,d,'reverseOnce'))
 (D/('v%04d.cpp'%n)).write_text(s)
 rows.append(dict(name='v%04d'%n,state=a,first=c,reverse=d,bridge=bridge,mixed=mixed))
(W/'once-models.json').write_text(json.dumps(rows,indent=1));print(len(rows),'single-iteration loop models')
