from search import *
from context import getters,mutators,amount_local
import itertools

def event_context(s,mask):
 snap='struct Snapshot { Snapshot() {} virtual ~Snapshot() '+('{}' if mask&8 else ';')+' virtual void loadPostProcess(); virtual void crc(); virtual void xfer(); };\n'
 string='class AsciiString {public: AsciiString():data(0) {} ~AsciiString(); private: char *data;};\n'
 s=s.replace('struct BfmeDispatchDelayedLuaEvent\n{\n\tvoid *m_vtable;',snap+string+'struct EventParameter : Snapshot\n{\n EventParameter();\n virtual ~EventParameter();',1)
 s=s.replace('UnsignedInt m_string;','AsciiString m_string;',1)
 s=s.replace('struct DelayedLuaEventList\n{','struct DelayedLuaEventList : Snapshot\n{',1)
 s=s.replace('BfmeDispatchDelayedLuaEvent m_events[3];','EventParameter m_events[3];',1)
 if mask&1:
  s=s.replace('DelayedLuaEventList();','__declspec(noinline) DelayedLuaEventList();',1)
  s+='\nDelayedLuaEventList::DelayedLuaEventList() {}\n'
 if mask&2:
  s=s.replace('virtual ~DelayedLuaEventList();','__declspec(noinline) virtual ~DelayedLuaEventList();',1)
  s+='\nDelayedLuaEventList::~DelayedLuaEventList() {}\n'
 if mask&4:
  s+='\nEventParameter::EventParameter():m_number(0.0f),m_boolean(false),m_objectID(0),m_type(0) {}\n'
 if mask&16:
  s+='\nEventParameter::~EventParameter() {}\n'
 return s

if __name__=='__main__':
 tests={}
 for mask,context in itertools.product(range(32),range(8)):
  s=event_context(BASE,mask)
  if context&1:s=getters(s)
  if context&2:s=mutators(s)
  if context&4:s=amount_local(s)
  tests[f'ehcontext-{mask:02d}-{context}']=s
 run(tests,'ehcontext')
