"""Copy to build/truck-20261023/verify-wb.py; run from repository root."""
from pathlib import Path
import hashlib
import struct
import sys

repo = Path.cwd()
assert (repo / 'tools/build.py').is_file(), 'Run from repository root'
sys.path.insert(0, str(repo / 'tools'))
import build
import capstone

path = repo / 'baselines/bfme2/workshop-vanilla-1.06/files/worldbuilder.exe'
data = path.read_bytes()
sections = build.pe_sections(data)

def read(va, size):
    return build.read_pe_bytes(data, sections, va - 0x400000, size)

md = capstone.Cs(capstone.CS_ARCH_X86, capstone.CS_MODE_32)
print('worldbuilder.exe SHA256:', hashlib.sha256(data).hexdigest())
handler = 0x01CC9BE9
assert read(handler, 1) == b'\xb8'
info = struct.unpack_from('<I', read(handler, 5), 1)[0]
magic, count, unwind = struct.unpack('<III', read(info, 12))
assert info == 0x020F6364 and magic == 0x19930520 and count == 9
print(f'Handler {handler:#x}; FuncInfo {info:#x}; states {count}')
for state in range(count):
    previous, action = struct.unpack('<iI', read(unwind + state * 8, 8))
    print(f'State {state}, previous {previous}, action {action:#x}')
    for ins in md.disasm(read(action, 40), action):
        print(f'  {ins.address:#010x}: {ins.mnemonic} {ins.op_str}'.rstrip())
        if ins.mnemonic in ('ret', 'jmp'):
            break

thunk = list(md.disasm(read(0x0040BA05, 5), 0x0040BA05))[0]
assert thunk.mnemonic == 'jmp' and thunk.op_str == '0x419060'
print(f'  {thunk.address:#010x}: {thunk.mnemonic} {thunk.op_str}')
for ins in md.disasm(read(0x00419060, 32), 0x00419060):
    print(f'  {ins.address:#010x}: {ins.mnemonic} {ins.op_str}'.rstrip())
    if ins.mnemonic == 'ret':
        break

print('Coordinate-copy instructions in the WorldBuilder parent:')
instructions = list(md.disasm(read(0x010F6160, 3853), 0x010F6160))
for i, ins in enumerate(instructions):
    if ins.mnemonic == 'mov' and ins.op_str.startswith('dword ptr [ebp - 0x80],'):
        for near in instructions[max(0, i - 2):i + 8]:
            print(f'  {near.address:#010x}: {near.mnemonic} {near.op_str}')
