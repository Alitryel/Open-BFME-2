import sys,itertools

 loop=mask&3
 head='index=0; current=buckets[0];\n'+['for(;;) {','again: {','do {','while(1) {'][loop]
 skip='goto again;' if loop==1 else 'continue;'

 state='if(node->state==0) node->state='+test+'?1:2;'
 if mask&256:state='if(node->state==0) { if('+test+') node->state=1; else node->state=2; }'

 else:work='if(node->state==2) '+skip+'\n'+work
 if mask&2048:work=work.replace(guard,'if(node->first->source!=0 && node->second->source!=0) goto present; '+skip+'\npresent:;')
 tail=['}','} goto again;','} while(1);','}'][loop]
 s=b+'void BfmeThingCGD::bfmeTwoCGD() {\n'+head+'\n'+scan+'\n'+cap+'\n'+valid+'\n'+guard+'\n'+state+'\n'+work+'\n'+tail+'\ndone:;\n}\n'
 (D/('f%04d.cpp'%mask)).write_text(s)
