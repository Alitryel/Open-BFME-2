import sys,struct
obj=open(sys.argv[1],'rb').read()
symptr,nsym=struct.unpack_from('<II',obj,8); st=symptr+18*nsym; i=0
while i<nsym:
    o=symptr+18*i; nm=obj[o:o+8]
    if nm[:4]==b'\0\0\0\0':
        off=struct.unpack_from('<I',nm,4)[0]; s=obj[st+off:obj.index(b'\0',st+off)].decode('latin1')
    else: s=nm.rstrip(b'\0').decode('latin1')
    val,sec,typ,cls,aux=struct.unpack_from('<IhHBB',obj,o+8)
    if s.startswith('$') or 'GetSrcLine' in s or 'PDBOK' in s: print(i,s,hex(val),sec,cls)
    i+=1+aux
