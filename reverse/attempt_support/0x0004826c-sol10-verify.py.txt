import ast,hashlib,importlib.util,json,re,sys
from pathlib import Path
D=Path(__file__).resolve().parent
sys.path.insert(0,str(D))
import trials

families=[
 ('headmerge','headmerge',480),('nulljoin','nulljoin',576),
 ('nullcheck','nullcheck',504),('nullcheck','nested',1092),
 ('conjunction','conjunction',224),('nearflags','nearflags',192),
 ('targetpair','targetpair',96),('conditioncopy','conditioncopy',132),
 ('nullcarry','nullcarry',320),('idtypes','idtypes',192),
 ('getter_interaction','getter-interaction',512),
]
literal=lambda s:set(re.findall(r'"(?:\\.|[^"\\])*"',s))
summary=[]
for script,label,count in families:
 path=D/(script+'.py');ast.parse(path.read_text())
 spec=importlib.util.spec_from_file_location('verify_'+label,path)
 module=importlib.util.module_from_spec(spec)
 old=sys.argv;sys.argv=['verify',label]
 try:spec.loader.exec_module(module)
 finally:sys.argv=old
 results=json.loads((D/(label+'-results.json')).read_text())
 assert len(module.items)==len(results)==count,(label,len(module.items),len(results))
 for item,result in zip(module.items,results):
  tag,source,*flags=item
  assert result['tag']==tag and 'error' not in result,tag
  assert (D/(tag+'.cpp')).read_text()==source,tag
  # Duplicate source arms may repeat TERRAIN; no literal may disappear/change.
  assert literal(source)==literal(trials.BASE),tag
 best=min(results,key=lambda r:(r['nd'],r['regions']))
 summary.append(dict(family=label,successful=count,best={k:best[k] for k in ['tag','size','nd','regions','slots','others']}))
 print(label,count,'sources regenerated, successful, literal set intact')
assert sum(r['successful'] for r in summary)==4320
(D/'verified-summary.json').write_text(json.dumps(summary,indent=2)+'\n')

native={}
for tag in ['hm09000','carry081200']:
 capture=json.loads((D/(tag+'-before-pack.json')).read_text())
 rows=next(r for r in capture if isinstance(r,list) and any(x['name'].startswith('_draw\n') for x in r))
 def bits(nodes):return sorted({n[0]+b for n in nodes for b in range(32) if n[2]>>b&1})
 table=[dict(name=x['name'].splitlines()[0],ordinal=x['data'][7],size=x['data'][8],type=x['data'][11],weight=x['data'][14],index=x['data'][15]) for x in rows]
 draw=next(x for x in table if x['name']=='_draw')
 assert len(table)==77 and draw['weight']==3 and draw['index']==18
 native[tag]=dict(rows=table,conflicts={str(x['index']):bits(x['nodes']) for x in capture[-1]['conflicts']})
 print(tag,'native: 77 records; draw weight 3, index 18')
(D/'native-normalized.json').write_text(json.dumps(native,indent=2)+'\n')
kept_names={'_draw','_fontName','_burnRate','_resistance','_fuel'}
compact={}
for tag,full in native.items():
 kept=[x for x in full['rows'] if x['name'] in kept_names]
 compact[tag]=dict(source_sha256=hashlib.sha256((D/(tag+'.cpp')).read_bytes()).hexdigest(),
                  record_count=len(full['rows']),
                  records_sha256=hashlib.sha256(json.dumps(full['rows'],sort_keys=True).encode()).hexdigest(),
                  rows=kept,conflicts={str(x['index']):full['conflicts'][str(x['index'])] for x in kept})
(D/'native-evidence.json').write_text(json.dumps(compact,indent=2)+'\n')

assert (D/'carry081200.json').exists()
near=json.loads((D/'carry081200.json').read_text())
assert (near['size'],near['nd'],near['regions'],near['slots'],near['others'])==(4101,11,1,0,1)
raw,rel=trials.build.read_object_symbol_bytes(D/'carry081200.obj',trials.NAME)
mask={i for off,kind,name in rel for i in range(off,off+4)}
diff=[dict(offset=i,target=t,ours=o) for i,(t,o) in enumerate(zip(trials.TARGET,raw)) if t!=o and i not in mask]
assert len(raw)==4101 and len(diff)==11 and all(0x912<=x['offset']<0x920 for x in diff)
window=lambda data:[dict(offset=i.address,mnemonic=i.mnemonic,operands=i.op_str) for i in trials.MD.disasm(data,0) if 0x90d<=i.address<0x926]
(D/'near-diagnostic.json').write_text(json.dumps(dict(score=near,differences=diff,target=window(trials.TARGET),ours=window(raw)),indent=2)+'\n')
print('Best alternate: 4101 B, 11 diagnostic differences, one region, no slot differences')
