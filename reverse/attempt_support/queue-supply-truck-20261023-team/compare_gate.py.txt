from pathlib import Path
import sys,json
sys.path.insert(0,str(Path('tools').resolve()))
import build
from locate import object_functions
D=Path(__file__).resolve().parent
symbols=build.load_symbol_map()
reports=[]
for p in [D/'official-bank.obj',D/'canonical.obj',D.parent/'home/home_base.obj',D.parent/'home/home_gh.obj',D.parent/'cfg-refine/a_expr_scope_shortagegoto.obj',D.parent/'hoist-iterators/clean_headrefresh0.obj']:
 row=dict(name='?queueSupplyTruck@AIPlayer@@IAEXXZ',source=str(p.with_suffix('.cpp').relative_to(build.ROOT)),target_rva='0x004F37B0',target_size='1537',notes='',status='matched')
 # Read-only invocation of the normal comparator with an explicit private object.
 # This does not compile, add a row, pin a name, or bypass any acceptance check.
 r=build.compile_function(row,symbols,p,allow_weak=False)
 body=next(b for n,b,_ in object_functions(p) if n==row['name']).rstrip(b'\xcc')
 same=sum(a==b for a,b in zip(r['bytes'],r['target']))
 facts=dict(path=str(p.relative_to(build.ROOT)),emitted_bytes=len(body),target_bytes=len(r['target']),positional_diff_bytes=len(r['target'])-same,unresolved=r['unresolved'],prefix_equal=r['bytes']==r['target'],full_extent_equal=len(body)==len(r['target']),note='Read-only byte comparator; DIR32 copied at positional sites per normal comparator. Mismatched bodies do not establish relocation identity, data, strings, extent or EH acceptance.')
 reports.append(facts)
 print(json.dumps(facts),flush=True)
(D/'normal-comparator.json').write_text(json.dumps(reports,indent=2))
