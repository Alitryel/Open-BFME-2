import sys, os, re, json, subprocess, concurrent.futures as cf
import combo
from sig import sig
cfg=json.loads(sys.argv[1]); base={int(k):v for k,v in json.loads(sys.argv[2]).items()}
A=[3408,3409,3410,3441,3468,3469]
cands=[]
for dtmp in range(8):
    for dU in range(8):
        for dtT in range(8):
            dtS=dU; dg=dU; dseg=dtT
            d=[dU,dtT-dU,dtS-dtT,dg-dtS,dseg-dg,dtmp-dseg]
            p=dict(base)
            for a,x in zip(A,d): p[a]=(p.get(a,0)+x)%8
            p={x:y for x,y in p.items() if y}
            cands.append(((dtmp,dU,dtT),p))
def ev(t):
    i,(w,p)=t; return w,sig((cfg,p),f'd{i}')
with cf.ThreadPoolExecutor(int(sys.argv[3])) as ex:
    fs=[ex.submit(ev,t) for t in enumerate(cands)]
    for f in cf.as_completed(fs):
        w,(n,offs)=f.result(); print(json.dumps([w,n,offs]),flush=True)
