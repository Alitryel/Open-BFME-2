import sys, json, subprocess, concurrent.futures, hashlib, itertools, time
from pathlib import Path
sys.path.insert(0,"tools")
import build
ROOT=Path.cwd()
OUT=ROOT/"build/s3-strlen"
target=build.read_target_bytes(0x7b9cca,24)
forms=[('goto', "// Best new candidate (astra-s3 helper seat, 2026-10-23): 24 bytes, 5 differing\n// byte positions vs retail 0x7B9CCA (parent's best was 6).\n// Emits: 8b4c2404 8a11 84d2 8d4101 7407 8a10 40 84d2 75f9 2bc1 48c3\n// Retail: 8b4c2404 8a11 8d4101 84d2 7407 8a10 40 84d2 75f9 2bc1 48c3\n// The only delta: guard block order (test/lea swapped). The je+7 and jne-latch\n// bytes are exact without volatile, ja/jbe or unsigned casts.\nint rva007B9CCAStrLen(const char *s) {\n    const char *p = s;\n    char c;\n    if ((c = *p++)) goto loop;\n    goto out;\nloop:\n    while (c) { c = *p++; }\nout:\n    return (int)(p - s - 1);\n}\n"), ('plain', 'int rva007B9CCAStrLen(const char*s){const char*p=s; while(*p++); return p-s-1;}'), ('split', 'int rva007B9CCAStrLen(const char*s){const char*p=s+1; char c=*s; while(c){c=*p++;}return p-s-1;}')]
flags=['/Od /Os /Oy /G5', '/Od /Os /Oy /G6', '/Od /Os /Oy /G7', '/Od /Ot /Oy /G5', '/Od /Ot /Oy /G6', '/Od /Ot /Oy /G7', '/O1 /Og- /G5', '/O1 /Og- /G6', '/O1 /Og- /G7', '/O2 /Og- /G5', '/O2 /Og- /G6', '/O2 /Og- /G7', '/O1 /Ot /Og- /G5', '/O1 /Ot /Og- /G6', '/O1 /Ot /Og- /G7', '/Od /Oy /G5', '/Od /Oy /G6', '/Od /Oy /G7', '/Od /Os /Oy /Ob2 /G5', '/Od /Os /Oy /Ob2 /G6', '/Od /Os /Oy /Ob2 /G7', '/O1 /GL- /G5', '/O1 /GL- /G6', '/O1 /GL- /G7']
def trial(arg):
 idx, (tag,body), flag, profile=arg
 ext="cpp"
 src=OUT/("round12_%04d.%s"%(idx,ext)); obj=src.with_suffix(".obj")
 src.write_text(("// compiler: "+profile+"\n" if profile else "")+"// cl: "+flag+" /MD\n"+body+"\n")
 command,env=build.compiler_command(src,obj)
 p=subprocess.run(command,cwd=ROOT,env=env,capture_output=True,text=True,timeout=90)
 rec=dict(index=idx,tag=tag,flags=flag,profile=profile,ext=ext,returncode=p.returncode,diagnostic=(p.stdout+p.stderr).strip())
 if p.returncode==0:
  symbols=build.defined_code_symbols(obj)
  sym=next(s for s in symbols if "rva007B9CCAStrLen" in s)
  buf,relocs=build.read_object_symbol_bytes(obj,sym,code_only=True)
  rec.update(size=len(buf),hex=buf.hex(),relocs=relocs,exact=buf==target,diff=sum(a!=b for a,b in itertools.zip_longest(buf,target)))
 return rec
jobs=[(i,form,flag,ext) for i,(form,flag,ext) in enumerate(itertools.product(forms,flags,[""]))]
build.ensure_reference_current()
start=time.time()
with (OUT/"round12.jsonl").open("w") as fh, concurrent.futures.ThreadPoolExecutor(max_workers=2) as pool:
 for i,rec in enumerate(pool.map(trial,jobs)):
  fh.write(json.dumps(rec)+"\n"); fh.flush()
  if rec.get("exact") or i%32==0: print(json.dumps(rec),flush=True)
print("DONE",len(jobs),"seconds",time.time()-start,flush=True)
