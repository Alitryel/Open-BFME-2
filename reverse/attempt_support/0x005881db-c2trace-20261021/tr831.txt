import sys
sys.path.insert(0,'/work/Open-BFME-2/build/c2tr')
from wdbg import Dbg
# tr831.py SRC OBJ ROUND WATCHIDX : log each 831f7 (color-assign) call at round ROUND: colored rec, register, and WATCHIDX's allowed set when it is a neighbour
src,obj,R=sys.argv[1],sys.argv[2],int(sys.argv[3]); WI=int(sys.argv[4],16)
d=Dbg(src,obj,obj+'.log')
d.prompt();d.prompt('set $CanDeferOnBPByAddr = 1')
B=0x10700000
bps=[0x83acd,0x82978,0x831f7]; added=set()
for b in bps:
    o=d.prompt('break *0x%x'%(B+b))
    if 'Breakpoint' in o and 'Unable' not in o: added.add(b)
names={}
def nm(sym):
    if sym in names: return names[sym]
    sw=d.words(sym,1); pw=d.words(sw[0],2) if sw and sw[0]>0x100000 else []
    n=d.string(pw[1]).strip() if len(pw)>1 and pw[1]>0x100000 else '?'
    n=n.strip('"').lstrip('_'); names[sym]=n; return n
def members(hdr):
    out=[]; ch=d.words(hdr,1)[0]
    while ch:
        ww=d.words(ch,3)
        if len(ww)<3: break
        base,nx,bits=ww
        out+=[base+j for j in range(32) if bits>>j&1]; ch=nx
    return out
def rec_of(i):
    p=d.words(0x108807b8+(i&0x3ff)*4,1)[0]
    while p:
        w=d.words(p,13)
        if w[7]==i: return p,w
        p=w[12]
    return None,None
on=False;k=0;n=0
while True:
    out=d.prompt('cont')
    if 'terminated' in out: break
    eip=d.reg('eip')-B
    for b in bps:
        if b not in added and b!=eip: d.prompt('break *0x%x'%(B+b)); added.add(b)
    added.add(eip)
    if eip==0x83acd:
        ebp=d.reg('ebp'); f=d.words(ebp,1)[0]; curfn=d.string(d.words(f+4,1)[0]); on='updateMeleeAttack' in curfn and d.reg('ebx')==0; k=0; continue
    if not on: continue
    if eip==0x82978:
        k+=1
        if k>R: break
        continue
    if eip==0x831f7 and k==R:
        esp=d.reg('esp'); a0,a1,a2=d.words(esp+4,3); w=d.words(a1,18)
        regrec=w[4]; reg=d.words(regrec+0x1c,1)[0] if regrec else -1
        nb=members(a2) if a2 else []
        s='%d col %s:%x w3=%d cost=%d reg=%x nb=%d'%(n,nm(w[0]),w[7],w[3],w[16],reg,len(nb))
        if WI in nb:
            p,ww=rec_of(WI)
            s+=' | %x allowed=%s'%(WI,members(ww[8]) if ww and ww[8] else None)
        print(s,flush=True); n+=1
d.p.kill()
