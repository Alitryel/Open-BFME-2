from search import *
import itertools
MERGED=(OUT/'merged-control.cpp').read_text();H,B=MERGED.split(MARK)

def source(f):
 qualifiers,readmode,tail,style=f
 h=H;b=B
 # Hints on existing locals only; no new accesses.
 for bit,decl in enumerate(['bool observer','GameWindow *hero','GameWindow **team','bool locked','bool teamEnable']):
  if qualifiers>>bit&1:b=b.replace(decl,'register '+decl)
 read=['m_rules.m_04','*(volatile int *)&m_rules.m_04','*(const volatile int *)&m_rules.m_04','((volatile Rules0043671B *)&m_rules)->m_04'][readmode]
 # The last spelling is replaced with a named reference to the existing field,
 # avoiding an invented rules-owner type.
 if readmode==3:
  b=b.replace('bool locked = m_rules.m_04 == 1;', 'const volatile int &teamRule=m_rules.m_04; bool locked=teamRule==1;')
 else:b=b.replace('m_rules.m_04 == 1',read+' == 1')
 if tail==1:b=b.replace('(*team)->winEnable(teamEnable);','(*team)->winEnable((bool)teamEnable);')
 if tail==2:b=b.replace('(*team)->winEnable(teamEnable);','(*team)->winEnable(!!teamEnable);')
 if tail==3:b=b.replace('(*team)->winEnable(teamEnable);','bool callEnable=teamEnable; (*team)->winEnable(callEnable);')
 if style:b=b.replace('bool teamEnable;', 'bool teamEnable=false;')
 return h+MARK+b
if __name__=='__main__':
 run([('localcontracts-%04d'%n,source(f))for n,f in enumerate(itertools.product(range(32),range(4),range(4),range(2)))], 'local_contracts')
