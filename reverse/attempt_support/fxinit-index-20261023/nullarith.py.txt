import sys,json,itertools,time
from pathlib import Path
from concurrent.futures import ThreadPoolExecutor
sys.path.insert(0,'build/fxinit-seat22');import search as h
h.HERE=Path('build/fxinit-index22').resolve()
join=Path('build/fxinit-edges22/trials/skipjoin_020000.cpp').read_text()
full=(h.HERE/'trials/realtree_00.cpp').read_text();isrc=h.BASE.index('bool FXShaderGeometry::Init(')
fullprefix=full[:full.index('bool FXShaderGeometry::Init(')]
# Equivalent nonzero predicates over target's 32-bit unsigned pointer value.
# Their arithmetic forms probe a different canonicalization/branch kind than
# the already-tested direct equality and unsigned <1 / <=0 predicates.
def pred(p,k):
 p='((unsigned)('+p+'))'
 return [p+'>0',p+'>=1','('+p+'-1U)!=~0U','('+p+'+~0U)<~0U','~'+p+'<~0U','(('+p+'|(0U-'+p+'))>>31)!=0','((0U-'+p+')>>31)|('+p+'>>31)','('+p+'/2U+('+p+'&1U))!=0'][k]
def variants():
 for pc,nc,flow,context in itertools.product(range(8),range(8),range(3),range(2)):
  pp=pred('positions[1]',pc);nn=pred('normals[1]',nc)
  x=h.BASE if flow==0 else join
  x=x.replace('if(positions[1])','if('+pp+')').replace('if(normals[1])','if('+nn+')')
  if flow:
   terms=['!('+pp+')','!('+nn+')']
   if flow==2:terms.reverse()
   x=x.replace('if(!positions[1] && !normals[1])','if('+terms[0]+' && '+terms[1]+')')
  if context:x=fullprefix+x[x.index('bool FXShaderGeometry::Init('):]
  yield f'nullarith_{pc}{nc}{flow}{context}',x
if __name__=='__main__':
 t=time.time()
 with ThreadPoolExecutor(max_workers=8)as ex:r=list(ex.map(h.run_one,variants()))
 r.sort(key=lambda x:(x.get('nd',99999),x.get('cost',99999)))
 (h.HERE/'nullarith-results.json').write_text(json.dumps(r,indent=2))
 print(json.dumps({'count':len(r),'seconds':time.time()-t,'errors':sum('error'in x for x in r),'best':r[:5]},indent=2))
