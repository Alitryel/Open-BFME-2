import sys,json,time,importlib.util
from pathlib import Path
from concurrent.futures import ThreadPoolExecutor
sys.path.insert(0,'build/fxinit-seat22')
import search as h
spec=importlib.util.spec_from_file_location('livegen','build/fxinit-live22/search.py');g=importlib.util.module_from_spec(spec);spec.loader.exec_module(g)
h.HERE=Path('build/fxinit-live22').resolve()
def variants():
 for form in range(3):
  if form==0:R='int primary=links[1][i]; int secondary=links[0][i];const short *pivot=&remap[primary];*(int *)(vertices+desc.Get_Basis_End())=pivot[(int)(secondary)-(int)(primary)]+(*pivot<<8);'
  else:
   k=form-1;R='int stagedLink=links['+str(k)+'][i];'+h.R.replace('links['+str(k)+'][i]','stagedLink')
  g.BASE=h.BASE.replace(h.R,R)
  for label,src in g.variants('byteivs'):
   if label.split('_')[1] in ['064','127']:yield 'recombine_'+str(form)+'_'+label,src
if __name__=='__main__':
 start=time.time()
 with ThreadPoolExecutor(max_workers=8) as pool:r=list(pool.map(h.run_one,variants()))
 r.sort(key=lambda x:(x.get('nd',99999),x.get('cost',99999)))
 (h.HERE/'recombine-results.json').write_text(json.dumps(r,indent=2))
 print(json.dumps({'count':len(r),'valid':sum('nd'in x and not x.get('unresolved')for x in r),'seconds':time.time()-start,'best':r[:5]},indent=2))
