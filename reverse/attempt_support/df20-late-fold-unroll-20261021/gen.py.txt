import itertools, sys
src = open('../s26/a9.cpp').read().split('\n')
# a9: S spec lines 98-109 (1-based), phantom lines 100-108; sig pragmas at 235/242
TRAPS = {
 'ab': 'if (this == 0) abort();',
 'fr': 'if (this == 0) free(0);',
 'vo': 'if (!this) *(volatile int*)0 = 0;',
 'ex': 'if (this == 0) exit(1);',
}
def make(trap, n, kS, sig):
    L = list(src)
    ph = L[99]
    S = L[97:99] + [ph]*kS + [L[108]]
    W = ['template <> inline basic_string<unsigned short, char_traits<unsigned short>, allocator<unsigned short> >::~basic_string() {'] + ['  '+TRAPS[trap]]*n + ['  _Destroy(this->_M_start, this->_M_finish + 1);', '}']
    if not sig:
        L[234] = ''; L[241] = ''
    out = L[:97] + S + W + L[109:]
    return '\n'.join(out)
for trap in TRAPS:
    for n in [1,2,3,4,5,6,8]:
        for kS in [9,10,12]:
            for sig in [0,1]:
                open(f'v/{trap}_n{n}_k{kS}_s{sig}.cpp','w').write(make(trap,n,kS,sig))
