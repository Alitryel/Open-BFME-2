from regions import *
# Global constant propagation replaces a plain null-arm p with literal zero.
# Inverse intrinsic pairs are real word copies; test whether their machine
# lowering can retain the register copy through packing without extra code.
exprs=[
 '_byteswap_ulong(_byteswap_ulong(v))',
 '_rotr(_rotl(v,1),1)', '_rotr(_rotl(v,13),13)',
 '_rotl(_rotl(v,16),16)', '_rotr(_rotl(v,31),31)',
 '_rotl(_rotr(v,7),7)', '~_rotl(~_rotr(v,13),13)',
 '_byteswap_ulong(_byteswap_ulong(_byteswap_ulong(_byteswap_ulong(v))))',
]
preamble='''extern "C" unsigned long __cdecl _byteswap_ulong(unsigned long);
extern "C" unsigned int __cdecl _rotl(unsigned int,int);
extern "C" unsigned int __cdecl _rotr(unsigned int,int);
#pragma intrinsic(_byteswap_ulong,_rotl,_rotr)
'''
def generate():
 for ei,expr in enumerate(exprs):
  for bits in itertools.product(range(2),repeat=7):
   dy,dn,oy,on,test,guard,contract=bits
   def cp(var,bit,typ):return '('+typ+' *)copyWord((unsigned int)('+var+'))' if bit else var
   arms='if('+('(unsigned)p>0' if test else 'p')+') {draw='+cp('p',dy,'Drawable')+'; obj='+cp('p->m_object',oy,'Object')+';} else {draw='+cp('p',dn,'Drawable')+'; obj='+cp('(Object*)p',on,'Object')+';}'
   src=BASE[:lo]+decl+arms+' if('+('p && obj' if guard else 'obj')+') {'+body+'\n} else {'+terrain+'}\n'+BASE[hi:]
   if not contract:src=src.replace('template <> void StringBase<char>::releaseBuffer() throw();\n','')
   src=src.replace('#define W(s)',preamble+'#define W(s)')
   src=src.replace('void W3DDisplay::gatherDebugStats(void)','static __forceinline unsigned int copyWord(unsigned int v) {return '+expr+';}\nvoid W3DDisplay::gatherDebugStats(void)')
   yield ('li'+str(ei)+''.join(map(str,bits)),src)
if __name__=='__main__':batch(list(generate()),'lateintrinsic')
