import json,itertools,concurrent.futures as cf
from pathlib import Path
from check import BASE,OUT,run
prefix='''
struct CameraValueSlot { Coord3D value; };
struct CameraArraySlot { Coord3D value[1]; };
struct CameraNestedSlot { CameraValueSlot value; };
struct CameraBaseSlot : public Coord3D { CameraBaseSlot(){} CameraBaseSlot(float x,float y,float z):Coord3D(x,y,z){} CameraBaseSlot(const Coord3D &p):Coord3D(p){} };
'''
BASE=BASE.replace('bool PolygonTrigger::adjustCameraMovement(',prefix+'bool PolygonTrigger::adjustCameraMovement(')
# Wrappers give the addressable field roots different C2 ownership without
# adding accesses, volatile reads or barriers. Arrays have one element.
def decl(name,init,mode):
 if mode==0:return 'Coord3D '+name+init+';'
 if mode==4:return 'CameraBaseSlot '+name+init+';'
 typ,path={1:('CameraValueSlot','value'),2:('CameraArraySlot','value[0]'),3:('CameraNestedSlot','value.value')}[mode]
 return typ+' '+name+'Slot;Coord3D &'+name+'='+name+'Slot.'+path+';'+ (name+'=Coord3D'+init+';' if init else '')
def gen():
 for c,a,u,n in itertools.product(range(5),repeat=4):
  t=BASE.replace('Coord3D candidate;',decl('candidate','',c))
  t=t.replace('Coord3D along(previous.x,previous.y,0.0f);',decl('along','(previous.x,previous.y,0.0f)',a))
  t=t.replace('Coord3D used(candidate);',decl('used','(candidate)',u))
  t=t.replace('Coord3D next(candidate);',decl('next','(candidate)',n))
  yield ('home-%d%d%d%d'%(c,a,u,n),t)
rows=[]
with cf.ThreadPoolExecutor(max_workers=8)as ex:
 for r in ex.map(run,gen()):
  rows.append(r)
  if r.get('exact'):print('EXACT',json.dumps(r),flush=True)
(OUT/'homes.json').write_text(json.dumps(rows,indent=1))
print('trials',len(rows),'errors',sum('error'in r for r in rows),'distinct',len({r.get('sha')for r in rows}),flush=True)
for r in sorted(rows,key=lambda r:r.get('nd',999))[:12]:print(json.dumps(r),flush=True)
