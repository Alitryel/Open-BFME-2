from pathlib import Path
import sys,json,hashlib,itertools,difflib,collections
ROOT=Path.cwd();OUT=ROOT/'build/trust237-round14';sys.path.insert(0,str(ROOT/'tools'))
import build,capstone
from variant import resolve_bytes
NAME='?Rva00237D3DCheck@@YAHXZ';RVA=0x237d3d;SIZE=226
target=build.read_target_bytes(RVA,SIZE);decoder=capstone.Cs(capstone.CS_ARCH_X86,capstone.CS_MODE_32)
target_ins=[i.mnemonic for i in decoder.disasm(target,0)]
records=[]
for family in ['bounds','zero_lifetime','narrow_dynamic']:
    rows=json.loads((OUT/(family+'.json')).read_text())
    for r in rows:
        assert 'error' not in r,r
        raw,rel=build.read_object_symbol_bytes(OUT/(r['name']+'.obj'),NAME,code_only=True)
        assert raw.hex()==r['raw'] and rel==[tuple(x) for x in r['relocations']],r['name']
        assert hashlib.sha256(raw).hexdigest()==r['sha256']
        ins=list(decoder.disasm(raw,0));assert sum(i.size for i in ins)==len(raw)
        nops=[i.address for i in ins if i.mnemonic=='nop'];assert nops==r['nops']
        res,unres=resolve_bytes(raw,rel,target+b'\0'*max(0,len(raw)-SIZE),RVA,len(raw),{'_memset':[0x6291ae],'_WinVerifyTrust@12':[0x655652]})
        assert unres==r['unresolved']==[]
        diffs=[i for i,(a,b) in enumerate(itertools.zip_longest(res,target)) if a!=b]
        assert len(diffs)==r['diff']
        sm=difflib.SequenceMatcher(a=[i.mnemonic for i in ins],b=target_ins,autojunk=False)
        eq=sum(c-b for op,b,c,d,e in sm.get_opcodes() if op=='equal');assert eq==r['eq']
        prefix=next((i for i,(a,b) in enumerate(zip(res,target)) if a!=b),min(len(raw),SIZE));assert prefix==r['prefix']
        records.append(dict(name=r['name'],family=family,size=len(raw),sha256=r['sha256'],eq=eq,prefix=prefix,diff=len(diffs),nops=nops,exact_return_suffix=raw.endswith(target[-5:]),outside_nop_differences=sum(i<205 or i>=221 for i in diffs)))
    good=[r for r in records if r['family']==family]
    print(family,len(good),'raw',len({r['sha256'] for r in good}),'exact suffix',sum(r['exact_return_suffix'] for r in good),'nops',sum(bool(r['nops']) for r in good),flush=True)
raw,rel=build.read_object_symbol_bytes(ROOT/'build/scratch/0x00237d3d.obj',NAME,code_only=True)
assert len(raw)==219 and hashlib.sha256(raw).hexdigest()=='23875f743d6816184d590366a5617a42b8e18aedc479c4d75864e0f6d61de0c9'
best=sorted({r['sha256']:r for r in records}.values(),key=lambda r:(r['outside_nop_differences'],r['diff'],-r['prefix']))[:8]
result=dict(objects_audited=len(records),distinct_raw=len({r['sha256'] for r in records}),families={k:sum(r['family']==k for r in records) for k in ['bounds','zero_lifetime','narrow_dynamic']},nop_objects=sum(bool(r['nops']) for r in records),exact_return_suffix_objects=sum(r['exact_return_suffix'] for r in records),region_winners=best,old_bank=dict(size=len(raw),sha256=hashlib.sha256(raw).hexdigest()),records_sha256=hashlib.sha256(json.dumps(records,sort_keys=True).encode()).hexdigest())
(OUT/'audit.json').write_text(json.dumps(result,indent=2)+'\n')
for r in best:print(r)

import pefile,struct
pe=pefile.PE(str(ROOT/'baselines/bfme2/workshop-vanilla-1.06/files/game.dat'))
base=pe.OPTIONAL_HEADER.ImageBase
imports={item.address:item.name.decode() for library in pe.DIRECTORY_ENTRY_IMPORT for item in library.imports if item.name}
api_names={'__imp__GetCommandLineW@0':'GetCommandLineW','__imp__CommandLineToArgvW@8':'CommandLineToArgvW','__imp__LocalFree@4':'LocalFree'}
checks=[]
for off,typ,name in rel:
    if typ==6:
        address=struct.unpack('<I',target[off:off+4])[0]
        assert imports[address]==api_names[name],(name,hex(address),imports.get(address))
        checks.append(dict(symbol=name,kind='target-IAT-name',address=hex(address),target_name=imports[address]))
    else:
        destination=RVA+off+4+struct.unpack('<i',target[off:off+4])[0]
        assert destination=={'_memset':0x6291ae,'_WinVerifyTrust@12':0x655652}[name]
        check=dict(symbol=name,kind='target-REL32',destination_rva=hex(destination))
        if name in ['_memset','_WinVerifyTrust@12']:
            thunk=pe.get_data(destination,6);assert thunk[:2]==b'\xff\x25'
            address=struct.unpack('<I',thunk[2:])[0];assert imports[address]=={'_memset':'memset','_WinVerifyTrust@12':'WinVerifyTrust'}[name]
            check['import_name']=imports[address]
        checks.append(check)
result['selected']['target_api_relocation_checks']=checks
(OUT/'audit.json').write_text(json.dumps(result,indent=2)+'\n')
print('TARGET_API_RELOCATIONS',checks)
