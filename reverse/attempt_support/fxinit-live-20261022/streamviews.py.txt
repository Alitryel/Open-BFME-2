import sys,itertools,json,time
from pathlib import Path
from concurrent.futures import ThreadPoolExecutor
sys.path.insert(0,'build/fxinit-seat22')
import search as h
h.HERE=Path('build/fxinit-live22').resolve()
STREAMS=[('positions[0]','desc.Get_Location_Offset(0)'),('normals[0]','desc.Get_Normal_Offset(0)'),('tangents','desc.Get_Basis_Offset()'),('binormals','desc.Get_Basis_Second_Offset()'),('positions[1]','desc.Get_Location_Offset(1)'),('normals[1]','desc.Get_Normal_Offset(1)')]
def variants():
 for forms in itertools.product(range(4),repeat=6):
  src=h.BASE
  for k,((stream,offset),form) in enumerate(zip(STREAMS,forms)):
   old='*(Vector3 *)(vertices+'+offset+')='+stream+'[i];'
   if form==0:continue
   if form==1:new='{const float *sp=&'+stream+'[i].X;float *dp=(float *)(vertices+'+offset+');dp[0]=sp[0];dp[1]=sp[1];dp[2]=sp[2];}'
   if form==2:new='ScalarCopy((Vector3 *)(vertices+'+offset+'),'+stream+'[i].X,'+stream+'[i].Y,'+stream+'[i].Z);'
   if form==3:new='{const Vector3 &sp='+stream+'[i]; Vector3 &dp=*(Vector3 *)(vertices+'+offset+');dp.X=sp.X;dp.Y=sp.Y;dp.Z=sp.Z;}'
   assert old in src;src=src.replace(old,new)
  if 2 in forms:src=src.replace('bool FXShaderGeometry::Init(', 'static __forceinline void ScalarCopy(Vector3 *d,float x,float y,float z) {d->X=x;d->Y=y;d->Z=z;}\nbool FXShaderGeometry::Init(')
  yield 'streamviews_'+''.join(map(str,forms)),src
if __name__=='__main__':
 start=time.time()
 with ThreadPoolExecutor(max_workers=8) as pool:r=list(pool.map(h.run_one,variants()))
 r.sort(key=lambda x:(x.get('nd',99999),x.get('cost',99999)))
 (h.HERE/'streamviews-results.json').write_text(json.dumps(r,indent=2))
 print(json.dumps({'count':len(r),'valid':sum('nd'in x and not x.get('unresolved')for x in r),'seconds':time.time()-start,'best':r[:5]},indent=2))
