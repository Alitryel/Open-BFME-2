ROOT = Path(__file__).resolve().parents[2]

OUT = Path(__file__).resolve().parent
NAME = '?Rva009B5530Prepare@@YAXPAURva009B5830State@@PAXH@Z'
RVA, SIZE = 0x1c5e30, 768
BASE = subprocess.check_output(['git','show','3eaf822ff05b788b143fed982c482a90129e4cc3:reverse/attempts/0x001c5e30.cpp'],cwd=ROOT,text=True)
TARGET = build.read_target_bytes(RVA,SIZE)
ADDR = {'?g_bfmeVp6SelectorMap@@3PAEA':0xBD8D00,
        '?g_bfmeSlotB54@@3PAXA':0xE22D1C,
        '?g_bfmeSlotB50@@3PAXA':0xE22D18,
        '_VP6_PredictFiltered':0x5D8DD0,
        '_Var16Point':0x5D8A30}

def resolved(obj):
    raw, relocs = build.read_object_symbol_bytes(obj, NAME, code_only=True)
    code = bytearray(raw)
    for off,kind,name in relocs:
        dest=ADDR[name]
        addend=struct.unpack_from('<I',code,off)[0]
        value=dest+addend
        if kind==20: value-=0x400000+RVA+off+4
        elif kind!=6: raise ValueError(kind)
        struct.pack_into('<I',code,off,value&0xffffffff)
    return bytes(code)

def seq(code):
    md=capstone.Cs(capstone.CS_ARCH_X86,capstone.CS_MODE_32)
    md.detail=True
    inst=list(md.disasm(code,RVA))
    norm=[]
    for i in inst:
        op=i.op_str
        if i.group(capstone.CS_GRP_JUMP): op='BRANCH'
        norm.append((i.mnemonic,op))
    return inst,norm

TI,TN=seq(TARGET)

def metrics(obj):
    code=resolved(obj)
    inst,norm=seq(code)
    sm=difflib.SequenceMatcher(None,norm,TN,autojunk=False)
    nd=sum(max(a2-a1,b2-b1) for tag,a1,a2,b1,b2 in sm.get_opcodes() if tag!='equal')
    nb=sum(a!=b for a,b in zip(code,TARGET))+abs(len(code)-SIZE)
    return dict(size=len(code),nd=nd,nb=nb,fit=sm.ratio(),exact=code==TARGET,
                sha=hashlib.sha256(code).hexdigest())

def run(item):
    label,text=item
    src,obj=OUT/(label+'.cpp'),OUT/(label+'.obj')
    src.write_text(text)
    command,env=build.compiler_command(src,obj)
    p=subprocess.run(command,cwd=ROOT,env=env,capture_output=True,text=True)
    if p.returncode:
        return dict(id=label,error=(p.stdout+p.stderr)[-600:])
    return dict(id=label,**metrics(obj))

def duplicate_once(text,needle,pred,style=0):
    assert text.count(needle)==1,(needle,text.count(needle))
    if style==0:
        repl='if ('+pred+') { '+needle+' } else { '+needle+' }'
    elif style==1:
        repl='if ('+pred+') goto use_'+str(abs(hash(needle))%1000000)+'; '+needle+' goto done_'+str(abs(hash(needle))%1000000)+'; use_'+str(abs(hash(needle))%1000000)+': '+needle+' done_'+str(abs(hash(needle))%1000000)+':;'
    return text.replace(needle,repl)

def arm(text,kind):
    if kind==0:return text
    start=text.index('\t} else {\n\t\tmvx =')
    end=text.index('\n\t}\n\n\tif (xPhase',start)+3
    old=text[start:end]
    new=old
    if kind==1:
        new=new.replace('\t\txPhase = mvx & mask;\n','').replace('\t\tx = mvx;','\t\txPhase = mvx & mask;\n\t\tx = mvx;')
    elif kind==2:
        new=new.replace('\t\txPhase = mvx & mask;\n','').replace('\t\tstride = pbi->reconStride;','\t\txPhase = mvx & mask;\n\t\tstride = pbi->reconStride;')
    return text[:start]+new+text[end:]

def dup_family():
    preds=['block','mask','source != 0','pbi->profile','pbi->loopFilterMode','pbi->mode == 2']
    # Each site is a real operation on already initialised values.
    for regime in range(3):
        head=arm(BASE,regime)
        start=head.index('\t} else {\n\t\tmvx =')
        end=head.index('\n\t}\n\n\tif (xPhase',start)+3
        for region in ['filtered','unfiltered']:
            lo,hi=(head.index('\t\tstride = 16;'),start) if region=='filtered' else (start,end)
            part=head[lo:hi]
            sites=['mvx = pbi->vectors[block].x;', 'x = mvx;',
                   'mvy = pbi->vectors[block].y;', 'y = mvy;',
                   'xPhase = x & pbi->vectorMask;' if region=='filtered' else 'xPhase = mvx & mask;',
                   'yPhase = y & pbi->vectorMask;' if region=='filtered' else 'yPhase = y & mask;']
            for site,needle in enumerate(sites):
                for pi,pred in enumerate(preds):
                    for style in range(2):
                        changed=duplicate_once(part,needle,pred,style)
                        yield f'dup-r{regime}-{region}-s{site}-p{pi}-t{style}',head[:lo]+changed+head[hi:]

def shared_family():
    # Hand-off's untried shared phases: reload mask after the builder, then
    # calculate one or both phases at the join rather than in each arm.
    import itertools
    for bits in itertools.product(range(2),repeat=6):
        sx,sy,mx,my,xi,yi=bits
        if not(sx or sy):continue
        text=BASE
        if sx:
            text=text.replace('\t\txPhase = x & pbi->vectorMask;\n','').replace('\t\txPhase = mvx & mask;\n','')
        if sy:
            text=text.replace('\t\tyPhase = y & pbi->vectorMask;\n','').replace('\t\tyPhase = y & mask;\n','')
        text=text.replace('\t\tstride = 16;','\t\tmask = pbi->vectorMask;\n\t\tstride = 16;')
        if mx:text=text.replace('x & pbi->vectorMask','x & mask')
        if my:text=text.replace('y & pbi->vectorMask','y & mask')
        add=[]
        if sx:add.append('xPhase = '+('mvx' if xi else 'x')+' & mask;')
        if sy:add.append('yPhase = '+('mvy' if yi else 'y')+' & mask;')
        text=text.replace('\n\tif (xPhase != 0)','\n\t'+'\n\t'.join(add)+'\n\tif (xPhase != 0)')
        yield 'shared-'+''.join(map(str,bits)),text

def factorial_family():
    import itertools
    for regime,pred in itertools.product(range(3),['mask','block','pbi->profile']):
        head=arm(BASE,regime)
        sites=['mvx = pbi->vectors[block].x;','x = mvx;',
               'mvy = pbi->vectors[block].y;','y = mvy;',
               'xPhase = x & pbi->vectorMask;','yPhase = y & pbi->vectorMask;',
               'xPhase = mvx & mask;','yPhase = y & mask;',
               'base = pbi->loopFilteredBlock;',
               'limit = (1UL << (pbi->sizeThreshold - 1)) << 2;']
        # First four choices duplicate the same real assignment in both arms.
        # The remaining choices independently affect each arm and the tail.
        for bits in range(1<<len(sites)):
            text=head
            for bit,needle in enumerate(sites):
                if bits&(1<<bit):
                    repl='if ('+pred+') { '+needle+' } else { '+needle+' }'
                    text=text.replace(needle,repl)
            yield f'fac-r{regime}-p{["mask","block","pbi->profile"].index(pred)}-{bits:04x}',text

def qualified_family():
    for regime in range(3):
        head=arm(BASE,regime)
        for bits in range(256):
            text=head
            for n,needle in enumerate(['pbi->vectors[block].x','pbi->vectors[block].y','pbi->vectorMask','pbi->lastFrame']):
                parts=text.split(needle)
                replacements=[]
                for i in range(len(parts)-1):
                    if n==0: bit=i
                    elif n==1:bit=2+i
                    elif n==2:bit=4+i
                    else:bit=7
                    # The builder's argument reads are left alone; the raw
                    # short captures after it and in the other arm vary.
                    if n<2 and i==0: bit=99
                    elif n<2:bit=(0 if n==0 else 2)+(i-1)
                    q='(*(const volatile '+('short' if n<2 else 'int')+' *)&'+needle+')' if n<3 else '(*(unsigned char * const volatile *)&'+needle+')'
                    replacements.append(q if bits&(1<<bit) else needle)
                text=''.join(a+b for a,b in zip(parts,replacements))+parts[-1]
            yield f'qual-r{regime}-{bits:02x}',text

def helpers_family():
    import itertools
    for regime,passing in itertools.product(range(3),range(3)):
        head=arm(BASE,regime)
        ptype=['short','const short &','int'][passing]
        helpers='''
__forceinline short CaptureX(const Rva009B5830MotionVector &v) { return v.x; }
__forceinline short CaptureY(const Rva009B5830MotionVector &v) { return v.y; }
__forceinline int PromoteX(PTYPE v) { return v; }
__forceinline int PromoteY(PTYPE v) { return v; }
__forceinline int Phase(int value,int mask) { return value & mask; }
__forceinline int WholePel(int value,int mask,int shift) {
    return (value + ((value >> 31) & mask)) >> shift;
}
'''.replace('PTYPE',ptype)
        head=head.replace('void __cdecl Rva009B5530Prepare(',helpers+'\nvoid __cdecl Rva009B5530Prepare(')
        for bits in range(128):
            text=head
            if bits&1:text=text.replace('pbi->vectors[block].x','CaptureX(pbi->vectors[block])')
            if bits&2:text=text.replace('pbi->vectors[block].y','CaptureY(pbi->vectors[block])')
            if bits&4:text=text.replace('x = mvx;','x = PromoteX(mvx);')
            if bits&8:text=text.replace('y = mvy;','y = PromoteY(mvy);')
            if bits&16:
                for value,mask in itertools.product(['x','mvx','y'],['mask','pbi->vectorMask']):
                    text=re.sub(r'\b'+re.escape(value+' & '+mask+';'), 'Phase('+value+','+mask+');',text)
            if bits&32:text=text.replace('((y + ((y >> 31) & mask)) >> pbi->vectorShift)','WholePel(y,mask,pbi->vectorShift)')
            if bits&64:text=text.replace('((x + ((x >> 31) & mask)) >> pbi->vectorShift)','WholePel(x,mask,pbi->vectorShift)')
            yield f'helper-r{regime}-p{passing}-{bits:02x}',text

def prologue_family():
    import itertools
    for regime,pi in itertools.product(range(2),range(2)):
        head=arm(BASE,regime)
        pred=['block','pbi->profile'][pi]
        a=head.index('\t} else {\n\t\tmvx =');b=head.index('\n\t}\n\n\tif (xPhase',a)+3
        sites=['source = pbi->lastFrame;', 'source = pbi->goldenFrame;',
               'mask = pbi->vectorMask;',
               'VP6_PredictFiltered(pbi, source + pbi->reconIndex,\n\t\t\tpbi->vectors[block].x, pbi->vectors[block].y);',
               'stride = 16;','base = pbi->loopFilteredBlock;']
        for bits in range(1024):
            front=head[:a];armtext=head[a:b];tail=head[b:]
            for k,needle in enumerate(sites):
                if bits&(1<<k):front=duplicate_once(front,needle,pred)
            for k,needle in enumerate(['mvx = pbi->vectors[block].x;', 'mvy = pbi->vectors[block].y;','x = mvx;','y = mvy;']):
                if bits&(1<<(6+k)):armtext=duplicate_once(armtext,needle,pred)
            yield f'pro-r{regime}-p{pi}-{bits:03x}',front+armtext+tail

def main():
    mode=sys.argv[1]
    if mode=='diff':
        for name in sys.argv[2:]:
            path=Path(name)
            code=resolved(path.with_suffix('.obj'))
            inst,norm=seq(code)
            sm=difflib.SequenceMatcher(None,TN,norm,autojunk=False)
            print(name,metrics(path.with_suffix('.obj')))
            for tag,a,b,c,d in sm.get_opcodes():
                if tag=='equal': continue
                print(tag)
                for i in TI[max(0,a-2):min(len(TI),b+2)]: print(' R',hex(i.address-RVA),i.mnemonic,i.op_str)
                for i in inst[max(0,c-2):min(len(inst),d+2)]: print(' C',hex(i.address-RVA),i.mnemonic,i.op_str)
        return
    items=[('baseline',BASE)]
    if mode=='dup':items+=list(dup_family())
    elif mode=='shared':items+=list(shared_family())
    elif mode=='fac':items+=list(factorial_family())
    elif mode=='qual':items+=list(qualified_family())
    elif mode=='helpers':items+=list(helpers_family())
    elif mode=='retry':
        for prior,generator in [('qual',qualified_family),('helpers',helpers_family)]:
            failures={r['id'] for r in map(json.loads,(OUT/(prior+'.jsonl')).read_text().splitlines()) if 'error' in r}
            items += [('fixed-'+label,text) for label,text in generator() if label in failures]
    elif mode=='prologue':items+=list(prologue_family())
    elif mode!='base':raise ValueError(mode)
    t=time.time()
    results=[]
    with (OUT/(mode+'.jsonl')).open('w') as log,cf.ThreadPoolExecutor(max_workers=8) as pool:
        for r in pool.map(run,items):
            results.append(r)
            log.write(json.dumps(r)+'\n');log.flush()
            if r.get('exact'): print('EXACT',r['id'],flush=True)
    good=sorted([r for r in results if 'error' not in r],key=lambda r:(r['nd'],r['nb']))
    print('count',len(results),'errors',len(results)-len(good),'distinct',len({r['sha'] for r in good}),'seconds',round(time.time()-t,1))
    for r in good[:20]:print(json.dumps(r))

if __name__=='__main__':main()
