from pathlib import Path
import sys,itertools,concurrent.futures,json
sys.path.insert(0,'build/df20-scalars-20261020');import trials
trials.scratch=Path.cwd()/'build/df20-scalars-20261020'
base=Path('build/df20-scalars-20261020/bank.cpp').read_text();trials.symbol=base.splitlines()[0][3:]
hdr=Path('vendor/stlport/stl/_string.h').read_text()
ctor='template <> template <> __forceinline basic_string<wchar_t>::basic_string(const wchar_t* first,const wchar_t* last,const allocator<wchar_t>& a) : _String_base<wchar_t,allocator<wchar_t> >(a) { _M_range_initialize(first,last,forward_iterator_tag()); }'
specs=[]
for method,base_decl,form in itertools.product(['decl-noinline','decl-explicit','defined','header'],[0,1],['primary','specialized','named']):
 label=f'rangefix-{method}-b{base_decl}-{form}';s=base
 decl='namespace _STL {\n'
 if base_decl:decl+='template <> _String_base<wchar_t,allocator<wchar_t> >::_String_base(const allocator<wchar_t>&);\n'
 sig='template <> template <> __declspec(noinline) void basic_string<wchar_t>::_M_range_initialize'+('<const wchar_t*>' if method=='decl-explicit' else '')+'(const wchar_t* first,const wchar_t* last,const forward_iterator_tag&)'
 if method=='defined':
  decl+=sig+' { ptrdiff_t n=last-first; _M_allocate_block(n+1); this->_M_finish=uninitialized_copy(first,last,this->_M_start); _M_terminate_string(); }\n'
 elif method.startswith('decl'):decl+=sig+';\n'
 elif method=='header':
  d=trials.scratch/label/'stl';d.mkdir(parents=True,exist_ok=True)
  text=hdr.replace('template <class _ForwardIter> void _M_range_initialize','template <class _ForwardIter> __declspec(noinline) void _M_range_initialize')
  assert text!=hdr
  (d/'_string.h').write_text(text)
  s=s.replace('// cl:','// cl: /I'+str(d.parent.relative_to(Path.cwd())),1)
 decl+='}\n'
 s=s.replace('#include <locale>',decl+'#include <locale>',1)
 if form=='primary':s=s.replace(ctor,'')
 elif form=='named':s=s.replace(ctor,ctor.replace('{ _M_range_initialize(first,last,forward_iterator_tag()); }','{ forward_iterator_tag tag; _M_range_initialize(first,last,tag); }'))
 specs.append((label,s))
with concurrent.futures.ThreadPoolExecutor(max_workers=8) as p:r=list(p.map(trials.run,specs))
(trials.scratch/'range_fixed.json').write_text(json.dumps(r,indent=2))
