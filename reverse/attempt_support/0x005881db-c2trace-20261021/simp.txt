import sys,os
sys.path.insert(0,'/work/Open-BFME-2/build/c2tr')
from wdbg import Dbg
# simp.py SRC OBJ ROUNDS [VAL] : at a8f54 entry in the listed rounds (comma list or 'all'), set the IR block count
# [0x108817b8] to VAL (default 50) and restore it at a8f54's return
src,obj,rs=sys.argv[1:4]; VAL=int(sys.argv[4]) if len(sys.argv)>4 else 50
ROUNDS=None if rs=='all' else set(int(x) for x in rs.split(','))
d=Dbg(src,obj,obj+'.log')
d.prompt();d.prompt('set $CanDeferOnBPByAddr = 1')
B=0x10700000; G=0x108817b8
bps=[0x83acd,0x82978,0xa8f54]; added=set()
for b in bps:
    o=d.prompt('break *0x%x'%(B+b))
    if 'Breakpoint' in o and 'Unable' not in o: added.add(b)
on=False;k=0;pend=[]
while True:
    out=d.prompt('cont')
    if 'terminated' in out: break
    eip=d.reg('eip')-B
    for b in bps:
        if b not in added and b!=eip: d.prompt('break *0x%x'%(B+b)); added.add(b)
    added.add(eip)
    if eip==0x83acd:
        ebp=d.reg('ebp'); f=d.words(ebp,1)[0]; curfn=d.string(d.words(f+4,1)[0]); on='updateMeleeAttack' in curfn and d.reg('ebx')==0; k=0; continue
    if not on: continue
    if eip==0x82978: k+=1; continue
    if eip==0xa8f54:
        esp=d.reg('esp'); st=d.words(esp,1); ra=st[0]-B
        if ROUNDS is None or k in ROUNDS:
            old=d.words(G,1)[0]
            d.prompt('set *(int*)0x%x = %d'%(G,VAL))
            pend.append((ra,esp+16,old)); print('R%d set blocks %d->%d'%(k,old,VAL),flush=True)
            if ra not in added: d.prompt('break *0x%x'%(B+ra)); added.add(ra)
        continue
    if pend and eip==pend[-1][0] and d.reg('esp')==pend[-1][1]:
        ra,e,old=pend.pop(); d.prompt('set *(int*)0x%x = %d'%(G,old)); print('restored',old,flush=True)
        continue
print('done',flush=True)
