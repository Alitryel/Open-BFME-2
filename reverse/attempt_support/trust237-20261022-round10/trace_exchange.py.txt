from pathlib import Path
import sys,json,re,hashlib
ROOT=Path.cwd();OUT=ROOT/'build/trust237-round10'
sys.path.insert(0,str(ROOT/'build/trust237-round7'));from wdbg import Dbg
src=Path(sys.argv[1]).resolve();label=sys.argv[2];obj=OUT/(label+'.trace.obj')
if obj.exists():obj.unlink()
d=Dbg(str(src),str(obj),str(OUT/(label+'.debug.log')));records=[];ended=False
try:
 d.prompt();d.prompt('set $CanDeferOnBPByAddr = 1');d.prompt('break *0x10760539')
 out=d.prompt('cont',45);assert d.reg('eip')==0x10760539,out;d.prompt('delete 1')
 out=d.prompt('break *0x107b2311');bp=int(re.search(r'Breakpoint (\d+)',out)[1]);d.prompt('condition %d *(unsigned int *)($ebx+4) == 9'%bp)
 out=d.prompt('break *0x107a7dd8');bp=int(re.search(r'Breakpoint (\d+)',out)[1]);d.prompt('condition %d *(unsigned int *)($esp+4) == 9'%bp)
 for n in range(160):
  out=d.prompt('cont',45)
  if 'terminated' in out or not out.strip():ended=True;break
  pc=d.reg('eip');sp=d.reg('esp');row=dict(pc=hex(pc),stack=d.words(sp,16))
  if pc==0x107b2311:
   node=d.reg('ebx');row['node_address']=hex(node);row['node']=d.words(node,14)
   row['operand_words']={hex(x):d.words(x,14) for x in row['node'][10:12] if x}
  records.append(row);print(label,json.dumps(row),flush=True)
 else:raise RuntimeError('trace breakpoint limit')
 assert obj.exists(),'no object emitted'
finally:
 result=dict(source=str(src.relative_to(ROOT)),source_sha256=hashlib.sha256(src.read_bytes()).hexdigest(),object_exists=obj.exists(),ended=ended,records=records)
 (OUT/(label+'.trace.json')).write_text(json.dumps(result,indent=2)+'\n')
 if d.p.poll() is None:
  try:d.prompt('quit')
  except Exception:d.p.kill()
 d.log.close()
print('trace ended',ended,'hits',len(records),'object',obj.exists(),flush=True)
