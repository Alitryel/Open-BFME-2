// 8x8 byte block copy (retail 0x001D7630): two groups of four 64-bit loads and
// stores, source and destination advancing by the same stride. The MMX state is
// left in use for the caller to clear.

// RVA 0x001D7630
#include <mmintrin.h>

struct CopyBlockRows
{
	unsigned __int64 first;
	unsigned __int64 second;
	unsigned __int64 third;
	unsigned __int64 fourth;
};

extern "C" void __cdecl CopyBlockMMX(const void *source, void *destination, int stride)
{
	const unsigned char *src = static_cast<const unsigned char *>(source);
	unsigned char *dst = static_cast<unsigned char *>(destination);
	CopyBlockRows rows;
	reinterpret_cast<__m64 &>(rows.first) = *reinterpret_cast<const __m64 *>(src);
	reinterpret_cast<__m64 &>(rows.second) = *reinterpret_cast<const __m64 *>(src + stride);
	reinterpret_cast<__m64 &>(rows.third) = *reinterpret_cast<const __m64 *>(src + stride * 2);
	reinterpret_cast<__m64 &>(rows.fourth) = *reinterpret_cast<const __m64 *>(src + stride * 3);
	src += stride * 4;
	*reinterpret_cast<__m64 *>(dst) = reinterpret_cast<const __m64 &>(rows.first);
	*reinterpret_cast<__m64 *>(dst + stride) = reinterpret_cast<const __m64 &>(rows.second);
	*reinterpret_cast<__m64 *>(dst + stride * 2) = reinterpret_cast<const __m64 &>(rows.third);
	*reinterpret_cast<__m64 *>(dst + stride * 3) = reinterpret_cast<const __m64 &>(rows.fourth);
	dst += stride * 4;
	reinterpret_cast<__m64 &>(rows.first) = *reinterpret_cast<const __m64 *>(src);
	reinterpret_cast<__m64 &>(rows.second) = *reinterpret_cast<const __m64 *>(src + stride);
	reinterpret_cast<__m64 &>(rows.third) = *reinterpret_cast<const __m64 *>(src + stride * 2);
	reinterpret_cast<__m64 &>(rows.fourth) = *reinterpret_cast<const __m64 *>(src + stride * 3);
	*reinterpret_cast<__m64 *>(dst) = reinterpret_cast<const __m64 &>(rows.first);
	*reinterpret_cast<__m64 *>(dst + stride) = reinterpret_cast<const __m64 &>(rows.second);
	*reinterpret_cast<__m64 *>(dst + stride * 2) = reinterpret_cast<const __m64 &>(rows.third);
	*reinterpret_cast<__m64 *>(dst + stride * 3) = reinterpret_cast<const __m64 &>(rows.fourth);
}
