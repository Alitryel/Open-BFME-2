from pathlib import Path
import subprocess,sys,json
root=Path(__file__).resolve().parents[1];out=root/'build/foundation-bitset-oct20';base=(out/'baseline.cpp').read_text()
start=base.index('\tBfmeItemE63 *newMember =');end=base.index('\n\tObject *self = getObject();',start)
tail=base[start:end];tests={}
for typename in ('Bool','Int'):
    for shape in ('nested','combined','separate'):
        s=tail.replace('\tif (newMember && oldMember)', '\t'+typename+' haveMembers = newMember && oldMember;\n\tif (haveMembers)',1)
        if shape=='combined':
            copies='\t\tnewMember->m_keepID = oldMember->m_keepID;\n\t\tnewMember->m_castleID = oldMember->m_castleID;'
            s=s.replace(copies+'\n\t\tif (!isRebuild)\n\t\t{', copies+'\n\t}\n\tif (haveMembers && !isRebuild)\n\t{',1)
            s=s[:-3] if s.endswith('\t}\n') else s
        elif shape=='separate':
            s=s.replace('\t\tif (!isRebuild)', '\t\tif (haveMembers && !isRebuild)',1)
        tests[typename+'-'+shape]=base[:start]+s+base[end:]
for mask in (1,2,3):
    s=tail
    for i,n in enumerate(('newMember','oldMember')):
        if mask&(1<<i):
            s=s.replace('BfmeItemE63 *'+n+' = (BfmeItemE63 *)','Module *'+n+' = ',1)
            s=s.replace(n+'->m_', '((BfmeItemE63 *)'+n+')->m_')
    tests['module-carrier-'+str(mask)]=base[:start]+s+base[end:]
for anchor_key,anchor in [('start','\tconst FoundationAIUpdateModuleData *md ='),('member','\tstatic NameKeyType memberKey ='),('event','\towningPlayer->onStructureCreated')]:
    for order in (('newMember','oldMember'),('oldMember','newMember')):
        s=base
        for n in order:s=s.replace('\tBfmeItemE63 *'+n+' =','\t'+n+' =',1)
        decl=''.join('\tBfmeItemE63 *'+n+';\n' for n in order)
        s=s.replace(anchor,decl+anchor,1)
        tests['declaration-'+anchor_key+'-'+order[0]]=s
name='?construct@FoundationAIUpdate@@UAEPAVObject@@PBVThingTemplate@@PBUCoord3D@@MPAVPlayer@@H_N@Z';results=[]
for key,s in tests.items():
    p=out/('flow-'+key+'.cpp');p.write_text(s)
    r=subprocess.run([sys.executable,'tools/explain_mismatch.py',name,'--rva','0x0045569D','--size','1174','--source',p.relative_to(root).as_posix()],cwd=root,capture_output=True,text=True,encoding='utf8',errors='replace',timeout=240)
    log=r.stdout+r.stderr;p.with_suffix('.log').write_text(log)
    d={'key':key,'exit':r.returncode,'exact':'OK: bytes match' in log,'summary':[x for x in log.splitlines() if x.startswith(('compiled symbol size:','classification:','unresolved:','first diff:'))]}
    if r.returncode:d['error']=log[-700:]
    results.append(d);(out/'flow-results.json').write_text(json.dumps(results,indent=2));print(json.dumps(d),flush=True)
