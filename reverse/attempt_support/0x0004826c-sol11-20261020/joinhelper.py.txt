from headmerge import *
# Earlier helper trials extracted the ordinary bank, not the new two-definition
# CFG. Carry that CFG through actual inline formal/return joins and cross the
# predicate/null variants so pass timing, rather than an isolated getter, varies.
items=[]
for shape,ref,lookup,force,test,nul,order,contract in itertools.product(range(4),range(2),range(2),range(2),range(3),range(2),range(2),range(2)):
 cond=['p','(unsigned)p>0','(unsigned __int64)(unsigned)p>0'][test]
 null='(Object*)p' if nul else '0'
 assignments='draw=p; obj=p->m_object;' if not order else 'obj=p->m_object; draw=p;'
 arms='if('+cond+') {'+assignments+'} else {draw=p; obj='+null+';}'
 formal=('Drawable *const &p' if ref else 'Drawable *p')+', '
 pre=''
 arg='p, '
 outer='Drawable *p='+find+'; '
 if lookup:formal=''; pre='Drawable *p='+find+'; ';arg='';outer=''
 qualifier='static __forceinline ' if force else 'static inline '
 if shape==0:
  helper=qualifier+'Object *mouseJoin('+formal+'Drawable *&draw) {'+pre+'Object *obj; '+arms+'return obj;}\n'
  call='obj=mouseJoin('+arg+'draw); if(obj)'
 elif shape==1:
  helper=qualifier+'bool mouseJoin('+formal+'Drawable *&draw, Object *&obj) {'+pre+arms+'return obj!=0;}\n'
  call='if(mouseJoin('+arg+'draw,obj))'
 elif shape==2:
  helper=qualifier+'void mouseJoin('+formal+'Drawable *&draw, Object *&obj) {'+pre+arms+'}\n'
  call='mouseJoin('+arg+'draw,obj); if(obj)'
 else:
  helper=qualifier+'Object *mouseJoin('+formal+'Drawable *&draw) {'+pre+'if('+cond+') {draw=p; return p->m_object;} else {draw=p;return '+null+';}}\n'
  call='obj=mouseJoin('+arg+'draw); if(obj)'
 src=BASE[:lo]+'Drawable *draw; Object *obj; '+outer+call+' {'+body+'\n} else {'+terrain+'}\n'+BASE[hi:]
 src=src.replace('void W3DDisplay::gatherDebugStats(void)',helper+'void W3DDisplay::gatherDebugStats(void)')
 if not contract:src=src.replace('template <> void StringBase<char>::releaseBuffer() throw();\n','')
 items.append(('jh%d%d%d%d%d%d%d%d'%(shape,ref,lookup,force,test,nul,order,contract),src))
if __name__=='__main__':batch(items,'joinhelper')
