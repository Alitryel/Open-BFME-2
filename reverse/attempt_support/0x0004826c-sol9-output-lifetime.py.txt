from trials import *
import itertools,re,collections
names=['fuel','resistance','burnRate','maxBurn','decay']
decl='int fuel, resistance, burnRate, maxBurn, decay;'
char='template <> void StringBase<char>::releaseBuffer() throw();\n'
items=[]
for shape,release in itertools.product(itertools.product(range(4),repeat=5),range(2)):
 if not any(shape):continue
 s=BASE if release else BASE.replace(char,'')
 ds=[];chosen=[]
 for n,form in zip(names,shape):
  if form:
   chosen.append(n)
   contract=['','',' throw()',''][form]
   ctor=' Output_'+n+'() throw() {}' if form==3 else ''
   ds.append('struct Output_'+n+' { int value;'+ctor+' ~Output_'+n+'()'+contract+' {} } '+n+';')
  else:ds.append('int '+n+';')
 start=s.index(decl);end=s.index('\n\t\t}\n\t\telse',start)
 chunk=s[start+len(decl):end]
 chunk=re.sub(r'"(?:\\.|[^"\\])*"|\b(?:'+ '|'.join(chosen)+r')\b',lambda m:m[0] if m[0].startswith('"') else m[0]+'.value',chunk)
 s=s[:start]+' '.join(ds)+chunk+s[end:]
 assert collections.Counter(re.findall(r'"(?:\\.|[^"\\])*"',s))==collections.Counter(re.findall(r'"(?:\\.|[^"\\])*"',BASE))
 items.append(('life'+''.join(map(str,shape))+str(release),s))
batch(items,'output-lifetime')
