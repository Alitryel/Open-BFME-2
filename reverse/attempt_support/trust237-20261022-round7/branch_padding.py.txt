from pathlib import Path
import sys,itertools,json,subprocess,concurrent.futures,hashlib
ROOT=Path.cwd();OUT=ROOT/'build/trust237-round7';sys.path.insert(0,str(ROOT/'tools'))
import build,capstone
def run(job):
    n,typ,k,cpu,op,adjust=job
    start={'byte':'(unsigned char)*p','short':'*(const unsigned short *)p','signedbyte':'(signed char)*p','signedshort':'*(const short *)p'}[typ]
    text=f'// cl: /O1 /G{cpu} /arch:SSE /DNDEBUG /MD /EHsc\n'
    text+='extern "C" int boundaryProbe(unsigned char *p,const unsigned *q,unsigned n,unsigned *out){unsigned a='+start+';'
    for j in range(k):text+=f'a {op}=q[{j}];'
    text+=adjust+'if(a==n)*out=a+1;return a;}\n'
    src=OUT/(f'{mode}{n:04}.cpp');obj=src.with_suffix('.obj');src.write_text(text)
    cmd,env=build.compiler_command(src,obj);r=subprocess.run(cmd,env=env,cwd=ROOT,capture_output=True,text=True)
    if r.returncode:return dict(name=src.stem,error=(r.stdout+r.stderr)[-800:])
    raw,rel=build.read_object_symbol_bytes(obj,'_boundaryProbe',code_only=True)
    ins=list(capstone.Cs(capstone.CS_ARCH_X86,capstone.CS_MODE_32).disasm(raw,0))
    assert sum(i.size for i in ins)==len(raw)
    return dict(name=src.stem,typ=typ,k=k,cpu=cpu,op=op,adjust=adjust,size=len(raw),sha256=hashlib.sha256(raw).hexdigest(),
                relocations=[list(x) for x in rel],raw=raw.hex(),nops=[i.address for i in ins if i.mnemonic=='nop'])
mode='p'
if len(sys.argv)>1 and sys.argv[1]=='adjust':
    mode='q';choices=itertools.product(['byte','short'],range(8,16),[5,6,7],['+','^'],['++a;','--a;','a^=1;','a|=1;','a=~a;','a=-a;'])
else:
    choices=((*x,'') for x in itertools.product(['byte','short','signedbyte','signedshort'],range(32),[6,7],['+','^']))
jobs=[(n,*x) for n,x in enumerate(choices)]
with concurrent.futures.ThreadPoolExecutor(max_workers=8) as pool:rows=list(pool.map(run,jobs))
(OUT/('branch-padding-'+mode+'.json')).write_text(json.dumps(rows,indent=2)+'\n')
good=[r for r in rows if 'error' not in r];nops=[r for r in good if r['nops']]
print('jobs',len(rows),'good',len(good),'unique',len({r['sha256'] for r in good}),'with NOP',len(nops))
for r in nops[:12]:print(json.dumps(r))
for r in [r for r in rows if 'error' in r][:3]:print(json.dumps(r))
