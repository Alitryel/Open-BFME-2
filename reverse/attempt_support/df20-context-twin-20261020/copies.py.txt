import trials,itertools,concurrent.futures,json
specs=[]
for symref,extref,localtag,exprtag,point,direct in itertools.product([0,1],repeat=6):
 src=trials.bank
 label=f'copy-s{symref}-e{extref}-l{localtag}-t{exprtag}-p{point}-d{direct}'
 if symref:src=src.replace('_MoneyOut out,const _MoneySignPolicy& policy)', 'const _MoneyOut& out,const _MoneySignPolicy& policy)')
 if extref:src=src.replace('const wchar_t* last, _MoneyOut out)', 'const wchar_t* last, const _MoneyOut& out)')
 if localtag:
  src=src.replace(',const _MoneySignPolicy& policy) { return __copy(first,last,out,policy,',') { _MoneySignPolicy policy; return __copy(first,last,out,policy,')
  src=src.replace('  _MoneySignPolicy __sign_policy;\n','')
  src=src.replace(', __s, __sign_policy);', ', __s);')
 if exprtag:
  src=src.replace('  random_access_iterator_tag tag;\n  return __copy(first, last, result, tag,','  return __copy(first, last, result, random_access_iterator_tag(),')
  src=src.replace('  random_access_iterator_tag tag;\n  return __copy(first, last, out, tag,','  return __copy(first, last, out, random_access_iterator_tag(),')
 if point:src=src.replace('__money_write_point(__s,__point);','__s=__point;')
 if direct:
  # Work at the innermost selected-copy level, with begin/end evaluated before
  # disabling inlining. No wrapper-created output-iterator argument survives.
  src=src.replace('if (__generate_curr)\n        __s = __money_copy_symbol(static_cast<const string_type&>(__curr_sym).begin(), static_cast<const string_type&>(__curr_sym).end(), __s'+('' if localtag else ', __sign_policy')+');', '''if (__generate_curr) {
        const wchar_t* __first=__curr_sym.begin();
        const wchar_t* __last=__curr_sym.end();
        random_access_iterator_tag __tag;
#pragma inline_depth(0)
        __s = __copy(__first,__last,__s,__tag,(ptrdiff_t*)0);
#pragma inline_depth(255)
      }''')
  src=src.replace('if (__frac_digits == 0)\n        __s = __money_copy_external(__digits_first, __digits_last, __s);', '''if (__frac_digits == 0) {
        random_access_iterator_tag __tag;
#pragma inline_depth(0)
        __s = __copy(__digits_first,__digits_last,__s,__tag,(ptrdiff_t*)0);
#pragma inline_depth(255)
      }''')
 specs.append((label,src))
with concurrent.futures.ThreadPoolExecutor(max_workers=8) as p:r=list(p.map(trials.run,specs))
(trials.scratch/'copies.json').write_text(json.dumps(r,indent=2)+'\n')
