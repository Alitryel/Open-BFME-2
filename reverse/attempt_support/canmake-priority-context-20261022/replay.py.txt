"""Private MSVC diagnostics, not byte/identity acceptance gates. Run at repo root."""
from pathlib import Path
import runpy,sys

FAMILIES=('inline','late-weight','loop-weight','temp-context','read-priority','shared-code')
if len(sys.argv)!=2 or sys.argv[1] not in FAMILIES:
    print('families: '+', '.join(FAMILIES))
    raise SystemExit(0 if len(sys.argv)==2 and sys.argv[1]=='--list' else 2)
runpy.run_path(str(Path(__file__).with_name(sys.argv[1]+'.py.txt')),run_name='__main__')
