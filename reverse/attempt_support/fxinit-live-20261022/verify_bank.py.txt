import json,sys
from pathlib import Path
sys.path.insert(0,'build/fxinit-seat22')
import search as h
import eh_verify
h.HERE=Path('build/fxinit-live22').resolve()
r=h.run_one(('bank_reverified',Path('reverse/attempts/0x0018c84e.cpp').read_text()))
r['eh']=eh_verify.verify_row(eh_verify.Image(),eh_verify.Obj(h.HERE/'trials/bank_reverified.obj'),h.NAME,h.RVA)
old=h.build.read_object_symbol_bytes(Path('build/fxinit-cost22/trials/guards_000.obj'),h.NAME,code_only=True)
new=h.build.read_object_symbol_bytes(h.HERE/'trials/bank_reverified.obj',h.NAME,code_only=True)
r['unchanged_bank_bytes']=old[0]==new[0];r['unchanged_bank_relocations']=old[1]==new[1]
r['traces']={}
for label,control in [('baseline-base','build/fxinit-cost22/trials/guards_000.obj'),('successor-named','build/fxinit-seat22/trials/pn_300000.obj')]:
 a=h.build.read_object_symbol_bytes(h.HERE/(label+'.obj'),h.NAME,code_only=True)
 b=h.build.read_object_symbol_bytes(Path(control),h.NAME,code_only=True)
 r['traces'][label]={'bytes_equal':a[0]==b[0],'relocations_equal':a[1]==b[1]}
 assert a==b
(h.HERE/'bank-verification.json').write_text(json.dumps(r,indent=2)+'\n')
print(json.dumps(r,indent=2))
assert r['nd']==9 and r['size']==2333 and not r['unresolved']
assert r['eh'][0]=='EXACT' and r['unchanged_bank_bytes'] and r['unchanged_bank_relocations']
