from zerocompare import exprs
from trials import COMMAND,ENV,DIR,ROOT
import subprocess
src='''typedef unsigned int uint32_t;
#include <stdio.h>
unsigned _rotl(unsigned v,int n) {unsigned s=(unsigned)n&31u;return s?((v<<s)|(v>>(32u-s))):v;}
unsigned _rotr(unsigned v,int n) {return _rotl(v,-(int)((unsigned)n&31u));}
unsigned _byteswap_ulong(unsigned v) {return ((v<<24)|((v<<8)&0xff0000u)|((v>>8)&0xff00u)|(v>>24));}
unsigned zero(unsigned v,unsigned i) {switch(i) {
'''+''.join('case %d: return %s;\n'%(i,e) for i,e in enumerate(exprs))+'''} return 1;}
int main() {uint32_t v=0; unsigned n=0; for(unsigned k=0;k<100000;k++) {v=k<1024?k:(v*1664525u+1013904223u);for(unsigned i=0;i<32;i++) {if(zero(v,i)) {printf("bad %u %08x\\n",i,v);return 1;}n++;}} printf("%u computed-zero checks passed\\n",n);}

p=DIR/'zero-semantics.cpp';p.write_text(src)
rel=lambda p:p.relative_to(ROOT).as_posix()

r=subprocess.run(c,cwd=ROOT,env=ENV,stdout=subprocess.PIPE,stderr=subprocess.STDOUT,text=True);print(r.stdout);r.check_returncode()
r=subprocess.run([COMMAND[0],rel(DIR/'zero-semantics.exe')],cwd=ROOT,env=ENV,stdout=subprocess.PIPE,stderr=subprocess.STDOUT,text=True);print(r.stdout);r.check_returncode()
assert r.stdout.strip()=='3200000 computed-zero checks passed'
(DIR/'zero-semantics.txt').write_text(r.stdout)
