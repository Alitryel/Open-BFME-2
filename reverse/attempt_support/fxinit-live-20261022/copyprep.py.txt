import sys,itertools,json,time
from pathlib import Path
from concurrent.futures import ThreadPoolExecutor
sys.path.insert(0,'build/fxinit-seat22')
import search as h
h.HERE=Path('build/fxinit-live22').resolve()
S=[('tangents','desc.Get_Basis_Offset()'),('binormals','desc.Get_Basis_Second_Offset()'),('positions[1]','desc.Get_Location_Offset(1)'),('normals[1]','desc.Get_Normal_Offset(1)')]
LOOP='   for(unsigned i=0;i<(unsigned)mesh->Get_Vertex_Count();++i) {'
def variants():
 for mask,stage,typ in itertools.product(range(1,256),range(5),range(2)):
  if stage>1 and mask&15:continue
  src=h.BASE;decl=[]
  for k,(stream,offset) in enumerate(S):
   old='*(Vector3 *)(vertices+'+offset+')='+stream+'[i];'
   dst='*(Vector3 *)(vertices+'+offset+')';rhs=stream+'[i]'
   if mask>>(k*2)&1:
    n='copySource'+str(k);expr='(unsigned)'+stream+'+i*sizeof(Vector3)'
    decl.append(('unsigned '+n+'='+expr+';') if typ==0 else ('const Vector3 *'+n+'=(const Vector3 *)('+expr+');'))
    rhs='*(const Vector3 *)'+n if typ==0 else '*'+n
   if mask>>(k*2+1)&1:
    n='copyDest'+str(k);expr='(unsigned)vertices+'+offset
    decl.append(('unsigned '+n+'='+expr+';')if typ==0 else ('Vector3 *'+n+'=(Vector3 *)('+expr+');'))
    dst='*(Vector3 *)'+n if typ==0 else '*'+n
   src=src.replace(old,dst+'='+rhs+';')
  ds=''.join(decl)
  if stage==0:src=src.replace(LOOP,LOOP+'\n'+ds)
  if stage==1:src=src.replace('    if(basis) {',ds+'\n    if(basis) {')
  if stage==2:src=src.replace('    if(m_Passes>1) *(unsigned *)',ds+'\n    if(m_Passes>1) *(unsigned *)')
  if stage==3:src=src.replace(h.R,ds+'\n'+h.R)
  if stage==4:src=src.replace(h.W,h.W+'\n'+ds)
  yield f'copyprep_{mask:03}_{stage}{typ}',src
if __name__=='__main__':
 start=time.time()
 with ThreadPoolExecutor(max_workers=8) as pool:r=list(pool.map(h.run_one,variants()))
 r.sort(key=lambda x:(x.get('nd',99999),x.get('cost',99999)))
 (h.HERE/'copyprep-results.json').write_text(json.dumps(r,indent=2))
 print(json.dumps({'count':len(r),'valid':sum('nd'in x and not x.get('unresolved')for x in r),'seconds':time.time()-start,'best':r[:5]},indent=2))
