import sys,json,itertools,time
from pathlib import Path
from concurrent.futures import ThreadPoolExecutor,as_completed
sys.path.insert(0,'build/fxinit-seat22');import search as h
h.HERE=Path('build/fxinit-thread22').resolve()
p=Path('Code/Libraries/Source/WWVegas/WW3D2/VertexBufferLockClasses.cpp').read_text()
decl=p[p.index('// upstream layout: d3d9.h'):p.index('// Both constructors work through the parameter:')]
decl=decl.replace('~WriteLockClass();','~WriteLockClass();\n        void *Get_Vertex_Array() {return Vertices;}')
a=p.index('VertexBufferClass::WriteLockClass::WriteLockClass(');b=p.index('VertexBufferClass::WriteLockClass::~WriteLockClass()',a);c=p.index('VertexBufferClass::AppendLockClass::AppendLockClass(',b)
ctors=p[a:b];dtors=p[b:c]
start=h.BASE.index('class VertexBufferClass {');end=h.BASE.index('class BfmeDynamicNativeVB:',start)
base=h.BASE[:start]+'extern void DX8_Assert();\nextern void Log_DX8_ErrorCode(unsigned);\n'+decl+h.BASE[end:]
base=base.replace('extern void BFME_DX8_Thread_Assert(void);','extern bool BFME_DX8_Thread_Assert(void);')
base=base.replace(' char m_pad00[0x20];\n};\nclass IndexBufferClass',' char m_pad00[4];\n};\nclass IndexBufferClass')
base=base.replace('class IndexBufferClass {','typedef char vb_size_check[sizeof(BfmeDynamicNativeVB)==32?1:-1];\nclass IndexBufferClass {')
def variants():
 for loc,noinline,mask in itertools.product(range(2),range(2),range(4)):
  s=base
  at='inline '+('__declspec(noinline) ' if noinline else '')
  definitions=(at+ctors if mask&1 else '')+(at+dtors if mask&2 else '')
  if loc:s+='\n'+definitions
  else:s=s.replace('bool FXShaderGeometry::Init(',definitions+'\nbool FXShaderGeometry::Init(')
  yield f'lockvis_{loc}{noinline}{mask}',s
r=[];t=time.time()
with ThreadPoolExecutor(max_workers=8)as pool:
 for f in as_completed([pool.submit(h.run_one,x)for x in variants()]):r.append(f.result())
r.sort(key=lambda x:(x.get('nd',99999),x.get('cost',99999)))
(h.HERE/'lockvis-results.json').write_text(json.dumps(r,indent=2));print(json.dumps({'count':len(r),'seconds':time.time()-t,'results':r},indent=2))
