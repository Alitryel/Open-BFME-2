from search import *
tests={}
uses=['owningPlayer->onStructureCreated(getObject(), obj);','(BfmeItemE63 *)getObject()->findModule(memberKey);','(Int)getObject(), -1);','Object *self = getObject();']
anchors=['const FoundationAIUpdateModuleData *md = getFoundationAIUpdateModuleData();','owningPlayer->onStructureCreated','if (md->m_mode)\n\t{\n\t\tDelayedLuaEventList list;']
for form in range(5):
 for loc,anchor in enumerate(anchors):
  for mask in range(1,16):
   s=BASE
   if form==0:decl='Object *const &ownerObject = m_object;'; expr='ownerObject'
   elif form==1:decl='const BehaviorModule &ownerModule = *this;';expr='ownerModule.getObject()'
   elif form==2:decl='BehaviorModule &ownerModule = *this;';expr='ownerModule.getObject()'
   elif form==3:decl='';expr='getObjectRef()';s=s.replace('Object *getObject() const { return m_object; }','Object *getObject() const { return m_object; }\n Object *const &getObjectRef() const { return m_object; }')
   else:decl='Object *const *ownerAddress = &m_object;';expr='(*ownerAddress)'
   for i,u in enumerate(uses):
    if mask&(1<<i):s=s.replace(u,u.replace('getObject()',expr))
   if decl:
    # The first anchor's text is unique in the function definition.
    actual=anchor.replace('getObject()',expr) if mask&1 else anchor
    if actual not in s:actual=anchor
    s=s.replace(actual,decl+'\n\t'+actual,1)
   tests[f'refweb-{form}-{loc}-{mask:02d}']=s
run(tests,'refweb')
