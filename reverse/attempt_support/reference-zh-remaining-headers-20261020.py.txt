from pathlib import Path
import sys, json, subprocess, hashlib, concurrent.futures
root=Path(__file__).resolve().parents[1]
sys.path.insert(0,str(root/'tools'))
import build
out=root/'build/zh-remaining-headers-oct20';out.mkdir(exist_ok=True)
zh=build.ZH_REFERENCE_ROOT
sdk=root/'reference/open-bfme-1/inputs/toolchains/vs2003/Program Files/Microsoft Visual Studio .NET 2003/Vc7/PlatformSDK/Include'
dx=root/'reference/open-bfme-1/inputs/toolchains/dx81/include'
assert (sdk/'windows.h').is_file() and (sdk/'imm.h').is_file() and (dx/'d3d8.h').is_file()
names={'w3d_dep.cpp','IMEManager.cpp','urllaunch.cpp','GameSpeech.cpp','simpleplayer.cpp','Debug.cpp'}
failures=json.loads((root/'build/zh-compile-failures-after-loader-oct20.json').read_text())
donors=[root/r['source'] for r in failures if Path(r['source']).name in names]
assert len(donors)==len(names)
revision=subprocess.check_output(['git','-C',str(root/'reference/open-bfme-1'),'rev-parse','HEAD'],text=True).strip()
reference_dirs=build._reference_include_dirs(zh)
actual=[f for f in reference_dirs if 'shims/sweep' not in f.replace('\\','/')]
shim=[f for f in reference_dirs if 'shims/sweep' in f.replace('\\','/')]
flags=build._ZH_BASE_FLAGS+actual+['-I'+sdk.relative_to(root).as_posix(),'-I'+dx.relative_to(root).as_posix()]+shim
tasks=[]
for donor in donors:
    prefix='#include <windows.h>\n'
    if donor.name=='IMEManager.cpp':prefix+='#include <imm.h>\n'
    if donor.name=='w3d_dep.cpp':prefix+='#include <new>\n#define _OPERATOR_NEW_DEFINED_\n'
    for profile,extra in [('original',[]),('retail',['-O1','-G7','-arch:SSE2'])]:
        selected=flags+extra
        for f in selected:
            if f.startswith('-I'):assert (root/f[2:]).is_dir(),f
        def render(f):return '-I"'+f[2:]+'"' if f.startswith('-I') and ' ' in f else f
        text='// cl: '+' '.join(map(render,selected))+'\n'
        if build.source_needs_stlport(donor):text+='// stlport\n'
        text+=prefix+'#include "'+donor.as_posix()+'"\n'
        trial=out/(profile+'_'+donor.name);trial.write_bytes(text.encode())
        assert build.source_extra_flags(trial)==selected
        tasks.append({'donor':donor.relative_to(root).as_posix(),'sha256':hashlib.sha256(donor.read_bytes()).hexdigest(),'revision':revision,'profile':profile,'flags':selected,'prefix':prefix,'trial':trial.relative_to(root).as_posix()})
(out/'inputs.json').write_bytes((json.dumps(tasks,indent=2)+'\n').encode())
def run(t):
    p=subprocess.run([sys.executable,'tools/place_bodies.py','--min','8',t['trial']],cwd=root,capture_output=True,text=True,encoding='utf8',errors='replace',timeout=240)
    output=p.stdout+p.stderr;(root/t['trial']).with_suffix('.log').write_bytes(output.encode())
    return {**t,'compile_failed':p.returncode!=0 or 'COMPILE FAILED' in output,'errors':[s for s in output.splitlines() if 'error ' in s][:6],'leads':[s for s in output.splitlines() if s.startswith(('BODY\t','PIN\t'))]}
results=[]
with concurrent.futures.ThreadPoolExecutor(max_workers=3) as pool:
    for r in pool.map(run,tasks):
        results.append(r);(out/'results.json').write_bytes((json.dumps(results,indent=2)+'\n').encode())
        print(Path(r['donor']).name,r['profile'],'failed',r['compile_failed'],'leads',r['leads'],'errors',r['errors'][:2],flush=True)
