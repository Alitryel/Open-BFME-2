import itertools,re
from search import *
SIGN='NetCommandList *NetPacket::getCommandList()\n'
H,B=BANK.split(SIGN,1)
SELECT=['relay & 1','commandType & 1','commandID & 1','timestamp & 1','frame & 1','playerID & 1','i & 1','m_packetLen & 1']
def base(mode):
 b=B
 if mode==1:
  b=b.replace('\tInt offset;\n','').replace('\t\t\toffset = i;\n','').replace('\t\t\ti = offset;\n','')
  b=re.sub(r'\boffset\b','i',b)
 elif mode==2:
  d,z=b.split("} else if (m_packet[i] == 'Z')",1)
  z=z.replace('\t\t\toffset = i;\n','').replace('\t\t\ti = offset;\n','')
  b=d+"} else if (m_packet[i] == 'Z')"+z
 return b
def wrap(s,c,f):
 if f==0:return 'if ('+c+') {\n'+s+'\n} else {\n'+s+'\n}'
 return 'switch (('+c+') ? 1 : 0) { case 0: {\n'+s+'\n} break; default: {\n'+s+'\n} break; }'
def change(s,kind,c,form):
 if kind==0:old='msg->detach();'
 elif kind==1:old='lastCommand = new NetCommandRef(msg);'
 elif kind==2:
  start=s.index('\t\t\tmsg->setTiming')
  end=s.index('\t\t\tmsg = 0;',start)+len('\t\t\tmsg = 0;')
  if s[end:].startswith('\n\t\t\ti = offset;'):end+=len('\n\t\t\ti = offset;')
  old=s[start:end]
 else:
  old='DoesCommandRequireACommandID((NetCommandType)commandType)'
  new='requiresID'
  # The call result is shared after two identical call arms.
  prefix='Bool requiresID; '+wrap('requiresID = '+old+';',c,form)+'\n'
  pos=s.index('\t\t\tif ('+old+')')
  return s[:pos]+prefix+s[pos:].replace(old,new,1)
 assert s.count(old)==1,(kind,s.count(old))
 return s.replace(old,wrap(old,c,form),1)
items=[]
for mode,sel,kind,form,location in itertools.product(range(3),range(8),range(4),range(2),range(3)):
 b=base(mode);d,z=b.split("} else if (m_packet[i] == 'Z')",1)
 if location in (1,2):d=change(d,kind,SELECT[sel],form)
 if location in (0,2):z=change(z,kind,SELECT[sel],form)
 items.append((f'merge_{mode}{sel}{kind}{form}{location}',H+SIGN+d+"} else if (m_packet[i] == 'Z')"+z))
batch(items,'merged-calls')
