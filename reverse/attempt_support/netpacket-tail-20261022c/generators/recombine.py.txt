import sys,itertools,re
from pathlib import Path
sys.path.insert(0,'build/netpacket-20261022c');import search
search.WORK=Path.cwd()/'build/netpacket-20261022c';W=search.WORK
sig='NetCommandList *NetPacket::getCommandList()\n'
items=[]
for context,mode,bits in itertools.product(range(2),range(2),range(64)):
 base=(W/('fresh_100000.cpp' if context==0 else 'home_100.cpp')).read_text();h,b=base.rsplit(sig,1)
 arms=[('commandType','UnsignedByte'),('timestamp','UnsignedInt'),('frame','UnsignedInt'),('playerID','UnsignedByte'),('relay','UnsignedByte'),('commandID','UnsignedShort')]
 for n,(var,typ) in enumerate(arms):
  if not bits>>n&1:continue
  old='++i;\n\t\t\tmemcpy(&'+var+', m_packet + i, sizeof('+typ+'));\n\t\t\ti += sizeof('+typ+');'
  if mode==0:new='Int nextIndex = i + 1;\n\t\t\tmemcpy(&'+var+', m_packet + nextIndex, sizeof('+typ+'));\n\t\t\ti = nextIndex + sizeof('+typ+');'
  else:new='Int nextIndex = i + 1 + sizeof('+typ+');\n\t\t\tmemcpy(&'+var+', m_packet + nextIndex - sizeof('+typ+'), sizeof('+typ+'));\n\t\t\ti = nextIndex;'
  assert old in b,(context,mode,bits,n)
  b=b.replace(old,new,1)
 items.append((f'recombine_{context}{mode}{bits:02x}',h+sig+b))
search.batch(items,'cursor-copy-interactions')
