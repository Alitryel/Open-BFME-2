def make_source(n, sdk, early, unsigned, nested, throwing, helper, flags, tail):
 prefix=SDK if sdk else structs
 if sdk==2:
  prefix=prefix.replace('typedef WINTRUST_DATA TrustData;', 'struct TrustData { unsigned long size; void *policy,*sip; unsigned long ui,revocation,choice; TrustFile *file; unsigned long state; void *stateData; const unsigned short *url; unsigned long flags; };')
 if throwing and not sdk:
  prefix=prefix.replace('long __stdcall WinVerifyTrust(void *,TrustGuid *,TrustData *);','long __stdcall WinVerifyTrust(void *,TrustGuid *,TrustData *) throw();')
  prefix=prefix.replace('void *__stdcall LocalFree(void *);','void *__stdcall LocalFree(void *) throw();')
 typ='unsigned long' if unsigned else 'long'
 b=body
 if nested: b=b.replace('    TrustData data;','    {\n    TrustData data;')
 b=b.replace('    WinVerifyTrust(0,&action,&data);\n    LocalFree(argv);\n    return 1;',('    status=WinVerifyTrust(0,&action,&data);' if early else '    '+typ+' status=WinVerifyTrust(0,&action,&data);')+ ('\n    }' if nested else '')+'\n    '+TAILS[tail])
 if early or nested: b=b.replace('    TrustFile file;','    '+typ+' status;\n    TrustFile file;')
 if sdk:
  for a,v in fieldmap.items():
   if sdk!=2 or a.startswith('file.'):b=b.replace(a,v)
  b=b.replace('unsigned short **argv','LPWSTR *argv')
  b=b.replace('0x210;', 'WTD_HASH_ONLY_FLAG | WTD_REVOCATION_CHECK_NONE;')
 helpers='''__forceinline int finish(unsigned long status) { if(status==0) return 1; return 1; }\n'''
 if helper:
  helpers+='''__forceinline void releaseArgs(void *p) { LocalFree(p); }\n'''
  b=b.replace('LocalFree(argv)','releaseArgs(argv)')
 # nested lifetime and early status need the outer declaration rather than an inner shadow.
 if nested and not early: b=b.replace('    '+typ+' status=WinVerifyTrust','    status=WinVerifyTrust')
 sdkflag=' /I"reference/open-bfme-1/inputs/toolchains/vs2003/Program Files/Microsoft Visual Studio .NET 2003/Vc7/PlatformSDK/Include"' if sdk else ''
 eh='' if '/EHs-' in flags else ' /EHsc'
 return '// cl: '+flags+sdkflag+' /DNDEBUG /MD'+eh+'\n'+prefix+helpers+b

def compile_one(job):
 n,params=job
 try:
  src=OUT/f'v{n:04}.cpp'; obj=OUT/f'v{n:04}.obj'
  src.write_text(make_source(n,*params))
  command,env=build.compiler_command(src,obj)
  proc=subprocess.run(command,cwd=ROOT,env=env,capture_output=True,text=True)
  if proc.returncode:return {'n':n,'params':params,'error':(proc.stdout+proc.stderr)[-1800:]}
  raw,relocs=build.read_object_symbol_bytes(obj,NAME,code_only=True)
  resolved,unres=resolve_bytes(raw,relocs,TARGET+b'\0'*max(0,len(raw)-SIZE),RVA,len(raw),{'_memset':[0x6291ae],'_WinVerifyTrust@12':[0x655652]})
  ins=[(i.mnemonic,i.op_str) for i in capstone.Cs(capstone.CS_ARCH_X86,capstone.CS_MODE_32).disasm(raw,0)]
  sm=difflib.SequenceMatcher(a=[x[0] for x in ins],b=[x[0] for x in retins],autojunk=False)
  regions=sum(x[0]!='equal' for x in sm.get_opcodes())
  eq=sum(min(x[2]-x[1],x[4]-x[3]) for x in sm.get_opcodes() if x[0]=='equal')
  diff=sum(a!=b for a,b in itertools.zip_longest(resolved,TARGET))
  prefix=next((i for i,(a,b) in enumerate(zip(resolved,TARGET)) if a!=b),min(len(raw),SIZE))
  nop=max((len(x) for x in raw.split(b'\x90') if False),default=0)
  return {'n':n,'params':params,'size':len(raw),'diff':diff,'prefix':prefix,'regions':regions,'eq':eq,'nop16':b'\x90'*16 in raw,'tail':raw[-42:].hex(),'unresolved':unres}
 except BaseException as e:return {'n':n,'params':params,'error':str(e)}
