import sys,itertools,re
sys.path.insert(0,'build/netpacket-20261022'); from search import *
SIGN='NetCommandList *NetPacket::getCommandList()\n'; H,B=BANK.split(SIGN,1)
items=[]
classes=['NetAckStage1CommandMsg','NetAckStage2CommandMsg','NetAckBothCommandMsg']
for modes,helper,inherit in itertools.product(itertools.product(range(4),repeat=3),range(2),range(2)):
 h,b=H,B;extras=''
 if inherit:
  for cl in classes:
   h=h.replace('class '+cl+'\n','class '+cl+' : public NetCommandMsg\n')
   # Remove only the ACK flat base fields; all three classes have the same block.
  h=h.replace('private:\n\tchar m_pad[0x1C];\npublic:\n\tunsigned short m_1c;', 'public:\n\tunsigned short m_1c;')
 for cl,mode in zip(classes,modes):
  if mode==0:continue
  p=b.index('\t\t\t\tmsg = (NetCommandMsg *)new '+cl+'();')
  e=b.index('\n\t\t\t}',p)
  arm=b[p:e]
  arm=arm.replace('msg = (NetCommandMsg *)new '+cl+'();',cl+' *ack = new '+cl+'();').replace('(('+cl+' *)msg)->','ack->')
  if mode==1:arm+='\n\t\t\t\tmsg = (NetCommandMsg *)ack;'
  elif mode==2:arm=arm.replace(cl+' *ack = new '+cl+'();',cl+' *ack = new '+cl+'();\n\t\t\t\tmsg = (NetCommandMsg *)ack;')
  elif mode==3:
   # Typed last is constructed before the allocation, and survives across it.
   last=cl+' *last = ('+cl+' *)ncrCommand(lastCommand);'
   arm=arm.replace('\n\t\t\t\t'+last,'')
   arm='\t\t\t\t'+last+'\n'+arm+'\n\t\t\t\tmsg = (NetCommandMsg *)ack;'
  if helper:
   name='npCopy'+cl
   arm=arm.replace('msg = (NetCommandMsg *)ack;','return (NetCommandMsg *)ack;') if mode!=2 else arm.replace('\n\t\t\t\tmsg = (NetCommandMsg *)ack;','')+'\n\treturn (NetCommandMsg *)ack;'
   extras+='static __forceinline NetCommandMsg *'+name+'(NetCommandRef *lastCommand) {\n'+arm+'\n}\n'
   arm='\t\t\t\tmsg = '+name+'(lastCommand);'
  b=b[:p]+arm+b[e:]
 items.append(('typed_'+''.join(map(str,modes))+f'_{helper}_{inherit}',h+extras+SIGN+b))
batch(items,'typed-acks')
