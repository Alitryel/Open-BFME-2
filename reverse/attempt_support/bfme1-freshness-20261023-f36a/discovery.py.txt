from pathlib import Path
import concurrent.futures,datetime,hashlib,json,shutil,subprocess,sys,time
ROOT=Path.cwd();OUT=ROOT/'build/trust237-round15'
assert subprocess.check_output(['git','-C','reference/open-bfme-1','rev-parse','HEAD'],text=True).strip()=='f36a4db066aa61edeee90085a31d628eb829001d'
def run(name,command):
 start=datetime.datetime.now(datetime.timezone.utc).isoformat();tick=time.monotonic();log=OUT/(name+'.log')
 with log.open('wb') as output:cp=subprocess.run(command,cwd=ROOT,stdout=output,stderr=subprocess.STDOUT)
 row=dict(name=name,command=command,started_utc=start,elapsed_seconds=time.monotonic()-tick,exit=cp.returncode,log_sha256=hashlib.sha256(log.read_bytes()).hexdigest(),tail=log.read_text(errors='replace').splitlines()[-4:])
 print(json.dumps(row),flush=True)
 return row
def bfme1():
 rows=[run('bfme1-scan',['python3','tools/bfme1_sweep.py','scan'])]
 if rows[-1]['exit']==0:
  for name in ['ranked','near']:rows.append(run('bfme1-'+name,['python3','tools/bfme1_sweep.py',name]))
 return rows
def manifest(root):return {str(p.relative_to(root)):hashlib.sha256(p.read_bytes()).hexdigest() for p in root.rglob('*') if p.is_file()}
def zh():
 original=ROOT/'reverse/zh_sweep';saved=OUT/'zh-sweep-preserved'

  finally:
   generated=OUT/'zh-sweep-regenerated';assert not generated.exists()

 return rows
