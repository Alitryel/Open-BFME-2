from pathlib import Path
import hashlib,json,time
import pefile
from unicorn import UcError,UC_PROT_READ,UC_PROT_WRITE,UC_QUERY_TIMEOUT
from unicorn.x86_const import UC_X86_REG_EIP,UC_X86_REG_ESP
from emu_unpack import Unpacker,APIS

root=Path(__file__).parent
allocs=[]
mutexes=[]
def global_alloc(u,arg):
    flags,size=arg(0),arg(1)
    allocs.append(dict(flags=hex(flags),size=size))
    assert not flags&2, 'GMEM_MOVEABLE requires actual handle semantics'
    return u.virtual_alloc(0,size)
APIS['GlobalAlloc']=(2,global_alloc)
APIS['GlobalFree']=(1,lambda u,arg: 0)
def create_mutex(u,arg):
    mutexes.append(dict(attributes=arg(0),owner=arg(1),name=u._cstr(arg(2)) if arg(2) else None))
    return 0x72000000+4*len(mutexes)
APIS['CreateMutexA']=(3,create_mutex)
APIS['GetLastError']=(0,lambda u,arg: 0)
APIS['CloseHandle']=(1,lambda u,arg: 1)
APIS['WaitForSingleObject']=(2,lambda u,arg: 0)
APIS['ReleaseMutex']=(1,lambda u,arg: 1)
u=Unpacker(str(root/'game-patch.dat'))
u.map()
# Fence only the original .text, permitting both protection sections and its
# extra blank section. A fence hit is a candidate transfer, not proof of OEP.
s=u.pe.sections[0]
u.uc.mem_protect(u.base+s.VirtualAddress, (max(s.Misc_VirtualSize,s.SizeOfRawData)+4095)&~4095, UC_PROT_READ|UC_PROT_WRITE)
before=bytes(u.uc.mem_read(u.base+0x237d3d,226))
t=time.monotonic();error=None
try:
    u.uc.emu_start(u.entry,0xffffffff,timeout=10_000_000,count=10_000_000)
except UcError as e:
    error=str(e)
after=bytes(u.uc.mem_read(u.base+0x237d3d,226))
result=dict(input_sha256=hashlib.sha256((root/'game-patch.dat').read_bytes()).hexdigest(),
    emulator_source='sp00nznet/pcrecomp@e3c9e025a9093242487c96b47eec321792dc0928:tools/drm/emu_unpack.py',
    timeout_us=10_000_000,instruction_limit=10_000_000,elapsed_seconds=time.monotonic()-t,
    eip=hex(u.uc.reg_read(UC_X86_REG_EIP)),esp=hex(u.uc.reg_read(UC_X86_REG_ESP)),
    stopped=u.stopped,error=error,candidate_transfer=None if u.oep is None else hex(u.oep),
    timeout=bool(u.uc.query(UC_QUERY_TIMEOUT)),thunk_count=len(u.thunks),
    global_alloc=allocs,mutexes=mutexes,target_changed=before!=after,target_sha256_before=hashlib.sha256(before).hexdigest(),
    target_sha256_after=hashlib.sha256(after).hexdigest(),dump_written=False)
(root/'loader-probe.json').write_text(json.dumps(result,indent=2)+'\n')
print(json.dumps(result,indent=2))
