"""Inspect already emitted helper bodies; never invoke compilation."""
from pathlib import Path
import json
import os
import sys

ROOT = next(p for p in Path(__file__).resolve().parents if (p / "AGENTS.md").exists())
OUT = ROOT / "build/water-vector-escape-replay"
if "--directory" in sys.argv:
    at = sys.argv.index("--directory")
    OUT = ROOT / sys.argv[at + 1]
    del sys.argv[at:at + 2]
os.environ["BFME_OBJSTORE"] = "off"
sys.path.insert(0, str(ROOT / "tools"))
import build
import eh_verify

helpers = [
    ("cache", "??0Rva00082A53@@QAE@HHHMPAHH@Z", 0x82A53, 153),
    ("plus", "??H@YA?AVMathCoord2D@@V0@ABV0@@Z", 0x7DEC8, 39),
    ("resize", "?resize@?$vector@DV?$allocator@D@_STL@@@_STL@@QAEXID@Z", 0x826E3, 54),
]
symbols = build.load_symbol_map()
records = []
for source in sorted(OUT.glob("[0-9][0-9]-*.cpp")):
    if len(sys.argv) > 1 and source.stem[:2] not in sys.argv[1:]:
        continue
    obj = build.obj_path(source)
    if not obj.exists():
        continue
    before = obj.stat().st_mtime_ns
    names = {s.get("name") for s in build.read_object_symbols(obj.read_bytes())
             if s["section"] > 0}
    for label, symbol, rva, size in helpers:
        record = dict(source=source.name, helper=label)
        if symbol not in names:
            record["emitted"] = False
        else:
            row = dict(name=symbol, target_rva=hex(rva), target_size=str(size),
                       source=source.relative_to(ROOT).as_posix(), notes="")
            got = build.compile_function(row, symbols, obj)
            raw, relocs = build.read_object_symbol_bytes(obj, symbol)
            record.update(
                emitted=len(raw),
                exact=got["bytes"] == got["target"] and len(raw) == size
                      and not got["unresolved"],
                unresolved=got["unresolved"],
            )
            if label == "cache":
                record["eh"] = eh_verify.verify_row(
                    eh_verify.Image(), eh_verify.Obj(obj), symbol, rva)
        assert obj.stat().st_mtime_ns == before
        records.append(record)
        print(json.dumps(record), flush=True)
(OUT / "helpers.json").write_text(json.dumps(records, indent=2))
