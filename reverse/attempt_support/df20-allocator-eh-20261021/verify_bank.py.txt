from pathlib import Path
import sys,subprocess,json,os
sys.path.insert(0,'tools');import build,eh_verify
p=Path(__file__).resolve().parent;src=p/'bank.cpp';symbol=src.read_text().splitlines()[0][3:]
env={**os.environ,'BFME_OBJSTORE':'off'}
checks={}
for label,args in [('bank-align',['tools/align_diff.py',str(src.relative_to(Path.cwd())),symbol,'0xdf20','2610']),('bank-explain',['tools/explain_mismatch.py',symbol,'--rva','0xdf20','--size','2610','--source',str(src.relative_to(Path.cwd()))])]:
 r=subprocess.run([sys.executable,*args],env=env,capture_output=True,text=True)
 (p/(label+'.txt')).write_text(r.stdout+r.stderr)
 checks[label]={'exit':r.returncode}
 print(label,r.returncode)
obj=eh_verify.Obj(build.obj_path(src));img=eh_verify.Image();sy=obj.by_name[symbol]
checks['eh_verify']=eh_verify.verify_row(img,obj,symbol,0xdf20)
sections=obj.eh_sections(sy['section']);places,conflicts=eh_verify.section_placements(img,obj,dict(sy,retail_rva=0xdf20),sections)
checks['eh_conflicts']=conflicts
checks['eh_sections']=[{'name':s['name'],'size':len(s['data']),'rva':hex(places[s['index']])} for s in sections]
(p/'checks.json').write_text(json.dumps(checks,indent=2));print(checks)
