from pathlib import Path
import itertools
r=Path(__file__).parent;out=r/'prefix';out.mkdir(exist_ok=True)
bases=[(r/'bank.cpp').read_text(),(r/'native_split.cpp').read_text()]
for bi,kind,dead,data,ready in itertools.product(range(2),range(4),range(4),range(4),range(4)):
    s=bases[bi]
    if kind==1:
        s=s.replace('return m_kindOf.test(kind);','const std::bitset<218> &bits=m_kindOf; return bits.test(kind);')
    elif kind==2:
        s=s.replace('return m_kindOf.test(kind);','return kindFlagsInView().test(kind);')
        s=s.replace('char m_pad000[0x108];','__forceinline const std::bitset<218> &kindFlagsInView() const { return m_kindOf; }\n char m_pad000[0x108];',1)
    elif kind==3:
        s=s.replace('return m_kindOf.test(kind);','return checkedKindInView(kind);')
        s=s.replace('char m_pad000[0x108];','__forceinline bool checkedKindInView(unsigned kind) const { return m_kindOf.test(kind); }\n char m_pad000[0x108];',1)
    if dead==1:s=s.replace('return (m_privateStatus & 1) != 0;','const unsigned char &flags=m_privateStatus; return (flags & 1) != 0;')
    elif dead==2:
        s=s.replace('return (m_privateStatus & 1) != 0;','return (privateStatusInView() & 1) != 0;')
        s=s.replace('bool rva00294471', '__forceinline unsigned char privateStatusInView() const { return m_privateStatus; }\n bool rva00294471',1)
    elif dead==3:
        s=s.replace('return (m_privateStatus & 1) != 0;','return deadInView(1);')
        s=s.replace('bool rva00294471','__forceinline bool deadInView(unsigned mask) const { return (m_privateStatus & mask) != 0; }\n bool rva00294471',1)
    if data:
        at='struct SightedEventData\n{'
        methods=['','SightedEventData(Drawable *d,int e):drawable(d),event(e){}',
                 '__forceinline void setInView(Drawable *d,int e){drawable=d;event=e;}',
                 '__forceinline void setDrawableInView(Drawable *d){drawable=d;}\n __forceinline void setEventInView(int e){event=e;}'][data]
        s=s.replace(at,at+'\n '+methods)
        import re
        old=r'SightedEventData data;\s*data.drawable\s*=\s*this;\s*data.event\s*=\s*dataEvent;'
        text=['','SightedEventData data(this,dataEvent);','SightedEventData data; data.setInView(this,dataEvent);','SightedEventData data; data.setDrawableInView(this); data.setEventInView(dataEvent);'][data]
        s,n=re.subn(old,text,s);assert n==1
    if ready:
        s=s.replace('virtual bool isReady();','virtual bool isReady();\n __forceinline bool readyInView(){return isReady();}',1)
        parts=s.split('holder->isReady()');assert len(parts)==3
        s=parts[0]+('holder->readyInView()' if ready&1 else 'holder->isReady()')+parts[1]+('holder->readyInView()' if ready&2 else 'holder->isReady()')+parts[2]
    (out/f'b{bi}k{kind}d{dead}a{data}r{ready}.cpp').write_text(s)
print(len(list(out.glob('*.cpp'))))
