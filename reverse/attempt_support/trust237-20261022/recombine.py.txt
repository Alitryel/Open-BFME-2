from pathlib import Path
import sys,json,subprocess,concurrent.futures,itertools,difflib
sys.path.insert(0,str(Path('build/trust237-session').resolve()))
import search
from search import OUT,ROOT,NAME,RVA,SIZE,TARGET,build,resolve_bytes,capstone
exprs=['((unsigned long)status|1u)!=0','((unsigned long)status|1u)>0','((unsigned long)status|1u)==0','((unsigned long)status|1u)<=0','((unsigned long)status|1u)>=1']
templates=[
 'return EXPR;',
 'int result=0; if(EXPR) ++result; return result;',
 'int result=0; if(EXPR) result=1; return result;',
 'if(EXPR) return 1; return 0;',
 'int result=0; do { ++result; } while(!(EXPR)); return result;',
 'int result=0; while(!(EXPR)) { ++result; break; } return result+1;',
 'int result=0; for(;!(EXPR);) { ++result; break; } return result+1;',
 'int result=0; if(!(EXPR)) goto done; ++result; done: return result;',
 'int result=0; if(EXPR) goto yes; return 0; yes: ++result; return result;',
 'int result=1; if(!(EXPR)) result=0; return result;',
 'int result=0; if(EXPR) { LocalFree(argv); ++result; } else {LocalFree(argv);} return result;',
 'int result; if(EXPR) { LocalFree(argv); result=1; } else {LocalFree(argv);result=0;} return result;',
 'int result=1; if(EXPR) { LocalFree(argv); } else {LocalFree(argv); result=0;} return result;',
 'if(EXPR) { LocalFree(argv); return 1; } LocalFree(argv); return 0;',
]
B=[(i.mnemonic,i.op_str) for i in capstone.Cs(capstone.CS_ARCH_X86,capstone.CS_MODE_32).disasm(TARGET,0)]
def run(job):
 n,expr,template,early,flag=job
 src=OUT/f't{n:04}.cpp';obj=src.with_suffix('.obj')
 text=search.make_source(n,0,early,0,0,0,0,flag,0)
 tail=template.replace('EXPR',expr)
 if template.count('LocalFree')==0:tail='LocalFree(argv); '+tail
 if expr.endswith('==0') or expr.endswith('<=0'):tail=tail.replace('return EXPR','return !(EXPR)') if 'EXPR' in tail else tail
 # Negative-predicate cases are diagnostic only; mark below and do not bank them.
 text=text.replace('LocalFree(argv); return 1;',tail)
 src.write_text(text)
 cmd,env=build.compiler_command(src,obj);proc=subprocess.run(cmd,env=env,cwd=ROOT,capture_output=True,text=True)
 if proc.returncode:return {'n':n,'error':proc.stdout+proc.stderr}
 raw,rel=build.read_object_symbol_bytes(obj,NAME,code_only=True)
 blob,un=resolve_bytes(raw,rel,TARGET+b'\0'*max(0,len(raw)-SIZE),RVA,len(raw),{'_memset':[0x6291ae],'_WinVerifyTrust@12':[0x655652]})
 A=[(i.mnemonic,i.op_str) for i in capstone.Cs(capstone.CS_ARCH_X86,capstone.CS_MODE_32).disasm(raw,0)]
 sm=difflib.SequenceMatcher(a=[x[0] for x in A],b=[x[0] for x in B],autojunk=False)
 diff=sum(a!=b for a,b in itertools.zip_longest(blob,TARGET))
 prefix=next((i for i,(a,b) in enumerate(zip(blob,TARGET)) if a!=b),min(len(raw),SIZE))
 eq=sum(x[2]-x[1] for x in sm.get_opcodes() if x[0]=='equal')
 return {'n':n,'expr':expr,'template':template,'early':early,'flags':flag,'size':len(raw),'diff':diff,'prefix':prefix,'eq':eq,'regions':sum(x[0]!='equal' for x in sm.get_opcodes()),'nop16':b'\x90'*16 in raw,'tail':raw[-45:].hex(),'negative_diagnostic':expr.endswith(('==0','<=0'))}
jobs=[(n,*j) for n,j in enumerate(itertools.product(exprs,templates,range(2),['/O1 /G7 /arch:SSE','/O2 /G7 /arch:SSE']))]
with concurrent.futures.ThreadPoolExecutor(max_workers=8) as pool:rows=list(pool.map(run,jobs))
(OUT/'recombine.json').write_text(json.dumps(rows,indent=2))
good=[r for r in rows if 'error' not in r]
print('total',len(rows),'success',len(good),'nop16',sum(r['nop16'] for r in good))
for r in sorted(good,key=lambda r:(-r['eq'],r['regions'],r['diff']))[:10]:print(json.dumps(r))
