from pathlib import Path
import pefile,capstone,struct,sys
PATH=Path('reference/open-bfme-1/inputs/toolchains/vs2003/Program Files/Microsoft Visual Studio .NET 2003/Vc7/bin/c2.dll')
PE=pefile.PE(str(PATH));DATA=PATH.read_bytes();BASE=PE.OPTIONAL_HEADER.ImageBase
MD=capstone.Cs(capstone.CS_ARCH_X86,capstone.CS_MODE_32)
def dis(va,n):
 for i in MD.disasm(PE.get_data(va-BASE,n),va):print('%08x %-8s %s'%(i.address,i.mnemonic,i.op_str))
def callers(va):
 for sec in PE.sections:
  if not sec.Characteristics&0x20000000:continue
  data=sec.get_data();off=0
  while True:
   off=data.find(b'\xe8',off)
   if off<0:break
   if off+5<=len(data):
    call=BASE+sec.VirtualAddress+off
    dest=call+5+struct.unpack_from('<i',data,off+1)[0]
    if dest==va:print('call',hex(call))
   off+=1
 target=struct.pack('<I',va);off=0
 while True:
  off=DATA.find(target,off)
  if off<0:break
  print('literal',hex(BASE+PE.get_rva_from_offset(off)));off+=1
if __name__=='__main__':
 va=int(sys.argv[2],0)
 if sys.argv[1]=='callers':callers(va)
 else:dis(va,int(sys.argv[3],0))
