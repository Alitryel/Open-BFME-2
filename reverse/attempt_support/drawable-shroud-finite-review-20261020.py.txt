from pathlib import Path
import sys, subprocess, json, itertools, concurrent.futures
root=Path(__file__).resolve().parents[1]
out=root/'build/shroud-finite-review-oct20';out.mkdir(exist_ok=True)
base=subprocess.check_output(['git','show','50b6f372cc:reverse/attempts/0x0027893e.cpp'],cwd=root).decode()
decls={
 'tmpl':'const ThingTemplate *tmpl;',
 'obj':'Object *obj;',
 'holder':'Rva004A1828Holder *holder;',
 'player':'Player *localPlayer;',
}
body=base.replace('const ThingTemplate *tmpl = m_template;','tmpl = m_template;').replace('Object *obj = m_object;','obj = m_object;').replace('Player *localPlayer = ThePlayerList','localPlayer = ThePlayerList').replace('Rva004A1828Holder *holder = (Rva004A1828Holder *)Rva004A1828Get','holder = (Rva004A1828Holder *)Rva004A1828Get')
tests={'baseline':base}
for order in itertools.permutations(decls):
    name='declarations-'+'-'.join(order)
    tests[name]=body.replace('void Drawable::onComingOutOfShroud()\n{','void Drawable::onComingOutOfShroud()\n{\n'+'\n'.join(decls[x] for x in order),1)
def run(item):
    key,s=item;p=out/(key+'.cpp');p.write_bytes(s.encode())
    r=subprocess.run([sys.executable,'tools/explain_mismatch.py','?onComingOutOfShroud@Drawable@@AAEXXZ','--rva','0x0027893E','--size','747','--source',p.relative_to(root).as_posix()],cwd=root,capture_output=True,text=True,encoding='utf8',errors='replace',timeout=240)
    log=r.stdout+r.stderr;p.with_suffix('.log').write_bytes(log.encode())
    d={'key':key,'exit':r.returncode,'exact':'OK: bytes match' in log,'summary':[x for x in log.splitlines() if x.startswith(('compiled symbol size:','classification:','unresolved:','first diff:'))]}
    if r.returncode:d['error']=log[-600:]
    return d
results=[]
with concurrent.futures.ThreadPoolExecutor(max_workers=2) as pool:
    for d in pool.map(run,tests.items()):
        results.append(d);(out/'results.json').write_text(json.dumps(results,indent=2));print(json.dumps(d),flush=True)
