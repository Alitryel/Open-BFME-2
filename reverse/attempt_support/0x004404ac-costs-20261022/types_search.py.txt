from search import *

def source(n):
    head,body=HEAD,BODY
    slot_kind=n%8;obs_kind=(n//8)%4;enable_kind=(n//32)%4;pos=n//128
    typ=['int','unsigned char','signed char','unsigned int'][obs_kind]
    body=body.replace('bool observer =',typ+' observer =')
    if enable_kind:
        decl=['','unsigned char','signed char','int'][enable_kind]
        body=body.replace('enable','enabled')
        body='\n{\n %s enabled = enable;'%decl+body.split('{',1)[1]
    if slot_kind:
        # Keep the integer ABI and make only the local index an aggregate.
        place='\n\tif (slot == -1 || slot >= 8)' if pos==0 else '\n\tbool observer'
        if pos==1:place='\n\t'+typ+' observer'
        idx=body.index(place)
        tail=body[idx:]
        import re
        access='index.value'
        if slot_kind==1:decl='struct Index { int value; }; Index index = {slot};'
        if slot_kind==2:decl='union Index { int value; unsigned int u; }; Index index = {slot};'
        if slot_kind==3:decl='int index[1] = {slot};';access='index[0]'
        if slot_kind==4:
            decl='struct Index { int value; Index(int x):value(x){} operator int() const {return value;} Index &operator=(int x){value=x;return *this;} }; Index index(slot);';access='index'
        if slot_kind==5:
            decl='struct Index { int value; int &get(){return value;} }; Index index = {slot};';access='index.get()'
        if slot_kind==6:decl='long index = slot;';access='index'
        if slot_kind==7:decl='unsigned long index = slot;';access='(int&)index'
        tail=re.sub(r'\bslot\b',access,tail)
        body=body[:idx]+'\n\t'+decl+tail
    return head+MARK+body

if __name__=='__main__':run([('types-%03d'%n,source(n)) for n in range(256)],'types')
