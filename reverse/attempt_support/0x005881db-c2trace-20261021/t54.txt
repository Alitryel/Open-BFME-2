import sys,os
sys.path.insert(0,'/work/Open-BFME-2/build/c2tr')
from wdbg import Dbg
# t51.py SRC OBJ : during R2 coloring of entry:12 (8368e), log every 5a796/afb21/5a3d2 call (set ops) touching index VI
src,obj=sys.argv[1:3]; KR=int(os.environ.get('KR','2')); RN=os.environ.get('RN','entry'); VI=int(os.environ.get('VI','b'),16)
d=Dbg(src,obj,obj+'.log')
d.prompt();d.prompt('set $CanDeferOnBPByAddr = 1')
B=0x10700000
bps=[0x83acd,0x82978,0x82a0d]; added=set()
for b in bps:
    o=d.prompt('break *0x%x'%(B+b))
    if 'Breakpoint' in o and 'Unable' not in o: added.add(b)
names={}
def nm(sym):
    if sym in names: return names[sym]
    sw=d.words(sym,1); pw=d.words(sw[0],2) if sw and sw[0]>0x100000 else []
    n=d.string(pw[1]).strip() if len(pw)>1 and pw[1]>0x100000 else '?'
    n=n.strip('"').lstrip('_'); names[sym]=n; return n
def members(hdr):
    m=[]; ch=d.words(hdr,1)[0] if hdr else 0; n=0
    while ch and n<200:
        ww=d.words(ch,3)
        if len(ww)<3: break
        m+=[ww[0]+j for j in range(32) if ww[2]>>j&1]; ch=ww[1]; n+=1
    return m
on=False;k=0;act=False;fin=None
X=[0x95d97,0x82a12,0x8374a,0x82ba4,0x82be3]
while True:
    out=d.prompt('cont')
    if 'terminated' in out: break
    eip=d.reg('eip')-B
    for b in bps:
        if b not in added and b!=eip: d.prompt('break *0x%x'%(B+b)); added.add(b)
    added.add(eip)
    if eip==0x83acd:
        ebp=d.reg('ebp'); f=d.words(ebp,1)[0]; curfn=d.string(d.words(f+4,1)[0]); on='updateMeleeAttack' in curfn and d.reg('ebx')==0; k=0; continue
    if not on: continue
    if eip==0x82978:
        k+=1
        if k>KR: break
        continue
    if eip==0x82a0d and k==KR:
        w=d.words(d.reg('ebp'),13)
        if nm(w[0])==RN:
            act=True; es=d.reg('esi'); print('start',hex(es),flush=True)
            for x in X:
                if x not in added: d.prompt('break *0x%x'%(B+x)); added.add(x)
        continue
    if not act: continue
    if eip==0x82a12:
        print('end',[hex(i) for i in members(es)],flush=True); break
    if eip==0x8374a:
        print('8368e ebx=%x ebp=%x edi(blocks)=%x'%(d.reg('ebx'),d.reg('ebp'),d.reg('edi')),flush=True); continue
    if eip==0x82ba4:
        esp=d.reg('esp'); edi=d.reg('edi'); a90=d.words(esp+0x90,1)[0]; a8c=d.words(esp+0x8c,1)[0]
        print('blk=%x live4c=%s live44=%s live50=%s arg90=%x:%s reg=%x regset=%s'%(edi,[hex(i) for i in members(d.words(edi+0x4c,1)[0])],[hex(i) for i in members(d.words(edi+0x44,1)[0])],[hex(i) for i in members(d.words(edi+0x50,1)[0])],a90,[hex(i) for i in members(a90)],a8c,[hex(i) for i in members(d.words(0x108a7b40+a8c*4,1)[0])]),flush=True); continue
    if eip==0x82be3:
        esp=d.reg('esp'); print('after L=%s S2=%s'%([hex(i) for i in members(d.words(esp+0x10,1)[0])],[hex(i) for i in members(d.words(esp+0x2c,1)[0])]),flush=True); continue
    if eip==0x95d97:
        esp=d.reg('esp'); L=d.words(esp+0x10,1)[0]; blk=d.words(esp+0x44,1)[0]; tp=d.reg('ebp'); ea=d.reg('eax')
        print('def-add blk=%x tuple=%x tw=%s opnd=%x ow=%s L=%s'%(blk,tp,[hex(x) for x in d.words(tp,12)],ea,[hex(x) for x in d.words(ea,10)],[hex(i) for i in members(L)]),flush=True)
        continue
    if eip==0x5a796:
        if d.reg('ecx')==VI:
            esp=d.reg('esp'); ra=d.words(esp,1)[0]-B
            print('add5a796 set=%x ra=%x eax=%x'%(d.reg('eax'),ra,d.reg('eax')), 'es-has' , VI in members(es),flush=True)
            bt=d.prompt('bt 5'); print('  ',' / '.join(l.strip()[:50] for l in bt.splitlines()[1:6]),flush=True)
print('done',flush=True)
