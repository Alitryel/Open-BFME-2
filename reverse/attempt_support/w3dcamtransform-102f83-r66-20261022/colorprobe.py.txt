"""Dump glob words for camera components at OrderGlobregs and Color exits (first SSE round).
usage: colorprobe.py label src.cpp"""
from pathlib import Path
import json, struct, sys
H = Path(__file__).resolve().parent
ROOT = H.parents[2]
sys.path.insert(0, str(ROOT / 'build/army-causal-20261021'))
sys.path.insert(0, str(ROOT / 'build/camera-causal-20261021/decode'))
import trace_pair as T
import symbol_decode as D

SYM = '?rva00102F83@W3DCamTransform@@QAEXPAUCameraTransformInput@@@Z'
label, rel = sys.argv[1], sys.argv[2]
POPS = {'@OrderGlobregs@12': sys.argv[3], '@Color@4': sys.argv[4]}
OUT = H / 'color' / label
OUT.mkdir(parents=True, exist_ok=True)
snaps = []


def snap(t, tag):
    rows = []
    for g in D.globs(t):
        comp = D.camera_component(g['symbol'])
        w = [t.word(g['address'] + 4 * i) for i in range(28)]
        rows.append({'name': comp[0] if comp else None, 'addr': g['address'], 'w': w})
    snaps.append({'tag': tag, 'rows': rows})


def capture(t, info, ctx, record):
    name = info['name']
    record['function_hit'] = True
    cls = ctx.Ecx if name == '@Color@4' else t.word(ctx.Esp + 4)
    pops = int(POPS[name])
    t.on_return(ctx, lambda t, c: snap(t, f'{name}:{cls}'), pop_bytes=pops)


cmd, env = T.TC.build.compiler_command(H / rel, OUT / (label + '.obj'))
T.TC.OUT = T.LAB
t = T.Tracer(label, cmd, SYM, on_hit=capture, env=env)
T.TC.OUT = OUT
t.names = ['@OrderGlobregs@12', '@Color@4']
t.run(timeout=240)
print('counts', dict(t.counts), 'records', len(t.records))
(OUT / 'snaps.json').write_text(json.dumps(snaps))
print('snapshots', [s['tag'] for s in snaps])
