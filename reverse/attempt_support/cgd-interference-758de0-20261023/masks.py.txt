import json,sys
from pathlib import Path
from wdbg import Dbg

d=Dbg(W/(tag+'.cpp'),W/(tag+'-masks.obj'),W/(tag+'-masks.log'))
events=[];active=False;finished=False;watch_id=None;last=None
try:
 d.prompt();d.prompt('set $CanDeferOnBPByAddr = 1');d.prompt('break *0x107821a7')

   f=d.words(d.reg('esp')+4,1)[0];fw=d.words(f,40);fn=d.string(d.words(fw[0]+4,1)[0])
   if '?bfmeTwoCGD@' not in fn:continue
   active=True;d.prompt('delete 1')
   for a in (0x10782978,0x107834bd,0x107b0037):d.prompt('break *0x%x'%a)
   continue
  if not active:continue
  if e==0x107b0037:
   for i in (2,3,4):d.prompt('delete '+str(i))
   if watch_id:d.prompt('delete '+str(watch_id))
   break
  if e==0x10782978:
   if watch_id:continue
   p=d.words(0x108807b8+4*8,1)[0]
   while p:

     sw=d.words(w[0],24)

      q=d.words(w[8],1)[0];mask=d.words(q,3);watch_id=5
      msg=d.prompt('watch *(unsigned int*)0x%x'%(q+8))
      events.append(dict(stage='start',candidate=w,symbol=sw,setnode=q,mask=mask,msg=msg));print('start',hex(q),mask,flush=True)
     break
    p=w[12]
   continue
  if e==0x107834bd:
   w=d.words(d.reg('eax'),18);sw=d.words(w[0],24)

   events.append(last);continue
  if watch_id:
   rec=dict(stage='watch',eip=hex(e),last=last,mask=d.words(q,3),out=out,regs={n:d.reg(n) for n in ['eax','ebx','ecx','edx','esi','edi','ebp']},stack=d.words(d.reg('esp'),24),near=d.prompt('x/14i $eip-24'))
   events.append(rec);print('watch',hex(e),'after',last['symbol'],last['idx'],last['reg'],'mask',rec['mask'],flush=True)
 if not finished:
  for _ in range(300):
   out=d.prompt('cont',60)
   if 'terminated' in out:finished=True;print('compiler terminated',flush=True);break
  else:raise RuntimeError('did not finish')
finally:
 (W/(tag+'-masks.json')).write_text(json.dumps(events,indent=1))
 if d.p.poll() is None:
  try:d.prompt('quit')
  except Exception:d.p.kill()
 d.log.close()
