tests=[]
for bits,order,assign,member,dtor in itertools.product(range(1,8),range(2),range(3),range(2),range(2)):
 # One 64-bit scalar keeps both halves meaningful; it is not the prior two-field union.
 recv='(AIBuildableWallSegment*)(unsigned int)bits'if order==0 else '(AIBuildableWallSegment*)(unsigned int)(bits>>32)'
 val='(int)(bits>>32)'if order==0 else '(int)bits'
 initial='(unsigned int)p'if order==0 else '(unsigned __int64)(unsigned int)p<<32'
 mask='0x00000000ffffffffui64'if order==0 else '0xffffffff00000000ui64'
 code='(unsigned __int64)(unsigned int)c<<32'if order==0 else '(unsigned int)c'
 set_expr=['bits=(bits&'+mask+')|('+code+');','bits=(bits&'+mask+')+('+code+');','bits^=(bits&~'+mask+');bits|=('+code+');'][assign]
 helper='class AIBuildableWallSegment;struct MakeCodeLife{unsigned __int64 bits;__forceinline MakeCodeLife(AIBuildableWallSegment*p):bits('+initial+'){} __forceinline ~MakeCodeLife(){'+('bits=0;'if dtor else '')+'} __forceinline AIBuildableWallSegment*wall()const{return '+recv+';} __forceinline void set(int c){'+set_expr+'} __forceinline int get()const{return '+val+';} };\n'
 b=body.replace('MakeCodeLife selected;','MakeCodeLife selected(this);')
 for i,field in enumerate(['m_startNodeID','getWorldPosition()','m_totalCost']):
  if bits&(1<<i):b=re.sub(r'(?<![\w.>])'+re.escape(field),lambda m:'selected.wall()->'+m[0],b)
 b=re.sub(r'selected.code=([029]);',lambda m:'selected.set('+m[1]+');'if member else '{unsigned __int64 &bits=selected.bits;int c='+m[1]+';'+set_expr+'}',b)
 b=b.replace('return selected.code;','return selected.get();')
 tests.append((f'packed-{bits}{order}{assign}{member}{dtor}',helper+prefix+'\nint AIBuildableWallSegment::canMake(Player *player)'+b))

 for i,d in enumerate(pool.map(r['run'],tests),1):
  results.append(d)
  if i%84==0 or d.get('exact_candidate'):print(json.dumps(dict(done=i,total=len(tests),best=sorted(results,key=lambda x:(x.get('nd',999),-x.get('score',0)))[:1])),flush=True)
(OUT/'results.json').write_text(json.dumps(results,indent=2)+'\n')
print(json.dumps(dict(trials=len(results),errors=sum('error'in x for x in results),unresolved=sum(bool(x.get('unresolved'))for x in results),exact=sum(x.get('exact_candidate',False)for x in results),bodies=len({x.get('masked_code_sha256')for x in results}))),flush=True)
