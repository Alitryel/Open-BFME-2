from pathlib import Path
import sys,subprocess,json
root=Path(__file__).resolve().parents[1];sys.path.insert(0,str(root/'tools'))
out=root/'build/findpath-constructor-context-oct20';out.mkdir(exist_ok=True)
base=(root/'build/findpath-auto-ptr-oct20/baseline.cpp').read_text()
# Diagnostic only: the owner has two unrelated POD unions. Give their members
# separate storage so a non-POD header can be evaluated for FindPath. Their
# other functions are not byte-verified or proposed for landing by this trial.
assert 'union PathfinderCheckCoordinates {' in base
assert 'union { Coord3D start; Coord3D p3; } stage;' in base
base=base.replace('union PathfinderCheckCoordinates {','struct PathfinderCheckCoordinates {',1)
base=base.replace('union { Coord3D start; Coord3D p3; } stage;','struct { Coord3D start; Coord3D p3; } stage;',1)
header=root/'Code/Libraries/Include/Lib/Coord3DCtorView.h'
prefix='#include "'+header.as_posix()+'"\n'
pos=base.index('Path *Pathfinder::FindPath(');head=base[:pos];body=base[pos:]
variants={'actual-constructors':prefix+base,'actual-inline-mutators':'#define COORD3D_VIEW_INLINE\n'+prefix+base}
for key,old,new in [('destination','Coord3D destination; copyCoord3D(&destination, &adjusted);','Coord3D destination(adjusted);'),('delta','Coord3D delta; copyCoord3D(&delta, &adjusted);','Coord3D delta(adjusted);'),('adjusted','Coord3D adjusted; copyCoord3D(&adjusted, rawTo);','Coord3D adjusted(*rawTo);')]:
 assert old in body
 variants['copy-construct-'+key]=prefix+head+body.replace(old,new)
import build,capstone
from locate import object_functions
sym='?FindPath@Pathfinder@@QAEPAVPath@@PAVObject@@ABVLocomotorSet@@PBUCoord3D@@2PA_N@Z'
for key,s in variants.items():
 src=out/(key+'.cpp');src.write_text(s)
 p=subprocess.run([sys.executable,'tools/explain_mismatch.py',sym,'--rva','0x2FD344','--size','781','--source',str(src)],cwd=root,capture_output=True,text=True,encoding='utf8',errors='replace',timeout=300)
 log=p.stdout+p.stderr;src.with_suffix('.mismatch.txt').write_text(log)
 obj=build.obj_path(src)
 if not obj.exists():print(key,'ERROR',log[-1000:],flush=True);continue
 f=next(b for n,b,r in object_functions(obj) if n==sym)
 summary=[l for l in log.splitlines() if l.startswith(('classification:','first diff:','OK:','unresolved:'))]
 q=subprocess.run([sys.executable,'tools/align_diff.py',str(src),sym,'0x2FD344','781'],cwd=root,capture_output=True,text=True,encoding='utf8',errors='replace',timeout=150)
 src.with_suffix('.align.txt').write_text(q.stdout+q.stderr)
 print(key,len(f),summary,q.stdout[-100:],flush=True)
