import sys,struct,os
os.environ['WINEDEBUG']='-all'
sys.path.insert(0,'tools'); sys.path.insert(0,'build')
import link_census as lc, build
from pathlib import Path
src=sys.argv[1]; want=sys.argv[2:]  # substring(s) of symbol names
obj=build.obj_path(build.ROOT/src)
rows=lc.ledger(); truth=lc.RetailTruth(rows)
data=obj.read_bytes()
for symbol,body,relocs,digest,size in lc._comdat_sections(data):
    if not any(w in symbol['name'] for w in want): continue
    addrs=truth.addresses(symbol['name'],None)
    print('##',symbol['name'][:110],'size',size,'addrs',[hex(a) for a in addrs])
    for start in addrs:
        retail=truth._read(start,size)
        if retail is None: print('  no retail bytes'); continue
        mask=bytearray(body)
        for w,k,r in relocs:
            mask[w:w+4]=retail[w:w+4]
        if bytes(mask)!=retail:
            d=[i for i in range(size) if mask[i]!=retail[i]]
            print('  BYTES differ at',d[:12]); continue
        for w,k,r in relocs:
            add=struct.unpack_from('<i',body,w)[0]; val=struct.unpack_from('<i',retail,w)[0]
            if k==truth.REL32: tgt=start+w+4+val-add
            elif k==truth.DIR32: tgt=(val-add-lc.BASE)&0xFFFFFFFF
            else: continue
            if r['storage'] in (lc.EXTERNAL,lc.WEAK_EXTERNAL):
                exp=truth.addresses(r['name'],k)
                ok=truth._lands(tgt,exp) if exp else None
                print('  rel',hex(w),r['name'][:80],'target',hex(tgt),'expected',[hex(a) for a in exp],'OK' if ok else ('??' if ok is None else 'MISMATCH'))
