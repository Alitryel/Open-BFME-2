from search import *
import itertools

provider=(ROOT/'Code/GameEngine/Source/GameClient/GUI/MpGameSetupSlots.cpp').read_text()
def extract(mark):
 s=provider.index(mark); b=provider.index('{',s);level=1;e=b+1
 while level:
  level += (provider[e]=='{')-(provider[e]=='}');e+=1
 return provider[s:e]
DEFS=[extract('bool AptMpGameSetup::HandleHeroSelection(int index)'),extract('void AptMpGameSetup::UpdateHeroDisplay(int slot)')]
def source(flags,reverse):
 head=HEAD;before=[];after=[]
 for i,f in enumerate(flags):
  if not f:continue
  definition=DEFS[i]
  if f>=5:definition='__declspec(noinline) '+definition
  placement=(f-1)%4
  if placement in (0,2):before.append(definition)
  else:after.append(definition)
  if placement>=2:
   mark=('bool HandleHeroSelection(int index);','void UpdateHeroDisplay(int slot);')[i]
   head=head.replace(mark,'__declspec(noinline) '+mark)
 if reverse:before.reverse();after.reverse()
 return head+'\n'+'\n'.join(before)+'\n'+MARK+BODY+'\n'+'\n'.join(after)+'\n'
if __name__=='__main__':run([('ctx-%03d'%n,source(flags,rev))for n,(flags,rev)in enumerate(itertools.product(itertools.product(range(9),repeat=2),range(2)))],'context')
