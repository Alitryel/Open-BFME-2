===== build/sk20/ktr.py =====
# ktr.py file.k4 -> pieces counter, K4 signature hash before pieces, A count
import sys,hashlib
L=[l.split() for l in open(sys.argv[1]) if l.strip()]
ev=[(x[0],[int(a.split('=',1)[1],16) for a in x[1:]]) for x in L]
cs=[v[-1] for a,v in ev]
st=max(i for i in range(1,len(cs)) if cs[i]<cs[i-1])
seq=[]
for a,v in ev[st:]:
    if a=='73f83':
        fh,an,cur,end,line,c=v
        seq.append(('A',fh if fh else an,'!' if cur>=end else '',c))
    elif a=='9f44e': seq.append(('F',v[0],'',v[3]))
    else: seq.append(('C',v[0],'',v[3]))
out=['%s%d%s'%(k,x,e) for (k,x,e,c) in seq]
j=None
for i,(k,x,e,c) in enumerate(seq):
    if k=='A' and x==0 and e=='!' and c>=300: j=i;break
pre=' '.join(out[:j]) if j else ' '.join(out)
# id of first piece walk: A before j
print('PIECES@%d'%(seq[j][3] if j else -1), 'sig', hashlib.md5(pre.encode()).hexdigest()[:6], 'pre', out[j-2] if j else '')

import os,sys,itertools
base=open('d1.cpp').read()
a=base.index('  float cx=Rotation.Y*v->Z');b=base.index('  v++;out++;')
XF={'cx':['  float cx=Rotation.Y*v->Z-v->Y*Rotation.Z;\n','  float x=(Rotation.W*v->X+cx);\n'],
    'tw':['  float x=Rotation.W*v->X+(Rotation.Y*v->Z-v->Y*Rotation.Z);\n'],
    'tp':['  float x=(Rotation.W*v->X+(Rotation.Y*v->Z-v->Y*Rotation.Z));\n']}
Y='  float y=Rotation.W*v->Y-(Rotation.X*v->Z-v->X*Rotation.Z);\n'
Z='  float z=Rotation.W*v->Z+(Rotation.X*v->Y-v->X*Rotation.Y);\n'
WP=['Rotation.X*v->X','Rotation.Y*v->Y','Rotation.Z*v->Z']
OUT={'d1':['x*Rotation.W-w*Rotation.X+(z*Rotation.Y-y*Rotation.Z)+Translation.X','y*Rotation.W-w*Rotation.Y-(z*Rotation.X-x*Rotation.Z)+Translation.Y','z*Rotation.W-w*Rotation.Z+(y*Rotation.X-x*Rotation.Y)+Translation.Z'],
     'tw':['(x*Rotation.W-w*Rotation.X+(z*Rotation.Y-y*Rotation.Z))+Translation.X','(y*Rotation.W-w*Rotation.Y)-(z*Rotation.X-x*Rotation.Z)+Translation.Y','(z*Rotation.W-w*Rotation.Z)+(y*Rotation.X-x*Rotation.Y)+Translation.Z']}
def make(xf,wm,of,om,n,d,tag):
    s=''.join(XF[xf])+Y+Z
    s+='  float w=-('+'+'.join(('(%s)'%p if wm>>i&1 else p) for i,p in enumerate(WP))+');\n'
    s+=''.join('  int a%d=count;\n'%k for k in range(n))
    for i,c in enumerate('XYZ'):
        e=OUT[of][i]
        s+='  out->%s=%s;\n'%(c,'('+e+')' if om>>i&1 else e)
    src=base[:a]+s+base[b:]
    os.makedirs(d,exist_ok=True)
    p='%s/%s.cpp'%(d,tag); open(p,'w').write(src); return p
if __name__=='__main__':
    for xf in XF:
      for wm in range(8):
        for of in OUT:
          for om in range(8):
            for n in (4,5):
              print(make(xf,wm,of,om,n,'v6','%s_w%d_%s_o%d_n%d'%(xf,wm,of,om,n)))
