"""g21: mineshaft-block factorial from WB twin shapes"""
import itertools, sys
from pathlib import Path
base = Path('best.cpp').read_text(encoding='latin-1')
lines = base.split('\n')
s = next(i for i, l in enumerate(lines) if 'hasUsableNetwork' in l)
e = next(i for i, l in enumerate(lines) if l.startswith(' PathfindLayerEnum destinationLayer='))
pre, post = '\n'.join(lines[:s]), '\n'.join(lines[e:])
def block(sw, cell, ori, cur):
    W = 'way' if cur else 'candidate.waypoint'
    o = []
    o.append(" if(((MineshaftPortalNetworkManager *)g_Va00E01EDC)->hasUsableNetwork(obj->getControllingPlayer())) {")
    o.append("  float startDistance=0.0f,endDistance=0.0f;")
    if cur: o.append(f"  for(Waypoint *way=((HierarchyTerrain *)TheTerrainLogic)->getFirstWaypoint();way;way=way->next) {{")
    else: o.append(f"  for({W}=((HierarchyTerrain *)TheTerrainLogic)->getFirstWaypoint();{W};{W}={W}->next) {{")
    rest = f"(unsigned char)((Rva002E6ECA *){W})->get() && {W}->isValidUnitType(obj)"
    if sw: o.append(f"   switch({W}->type) {{ case 8: if({rest}) {{")
    else: o.append(f"   if({W}->type==8 && {rest}) {{")
    if cell == 0: o.append(f"    if(rva001E3647Pos(TheTerrainLogic->getLayerForDestination(0,&{W}->location),&{W}->location)) {{")
    elif cell == 1: o.append(f"    PathfindCell *cell=(PathfindCell *)rva001E3647Pos(TheTerrainLogic->getLayerForDestination(0,&{W}->location),&{W}->location);\n    if(cell) {{")
    else: o.append(f"    PathfindLayerEnum layer=TheTerrainLogic->getLayerForDestination(0,&{W}->location);\n    PathfindCell *cell=(PathfindCell *)rva001E3647Pos(layer,&{W}->location);\n    if(cell) {{")
    o.append(f"     HierarchyPoint delta;delta.x={W}->location.x;delta.y={W}->location.y;delta.z={W}->location.z;")
    o.append("     delta.x-=obj->position[0];delta.y-=obj->position[1];delta.z-=obj->position[2];")
    o.append("     float dist=delta.x*delta.x+delta.y*delta.y;")
    o.append(f"     if(!m_srcMineshaft || dist<startDistance) {{m_srcMineshaft=(PathfinderMineshaft *){W};startDistance=dist;}}")
    o.append(f"     delta={W}->location;")
    o.append("     delta.x-=rawTo->x;delta.y-=rawTo->y;delta.z-=rawTo->z;")
    o.append("     dist=delta.x*delta.x+delta.y*delta.y;")
    o.append(f"     if(!m_destMineshaft || dist<endDistance) {{m_destMineshaft=(PathfinderMineshaft *){W};endDistance=dist;}}")
    o.append("    }")
    o.append("   }" + (" break; }" if sw else ""))
    o.append("  }")
    if ori == 0:
        o.append("  if(m_srcMineshaft && m_destMineshaft && m_srcMineshaft!=m_destMineshaft &&\n     QuickDoesPathExist(obj,from,&m_srcMineshaft->m_position,0) &&\n     QuickDoesPathExist(obj,&m_destMineshaft->m_position,rawTo,0)) {")
        o.append("   useMineshaft=true;adjustTo=m_srcMineshaft->m_position;rva002E75A3(&clipFrom,&adjustTo);\n  } else {m_srcMineshaft=0;m_destMineshaft=0;}")
    else:
        o.append("  if(!m_srcMineshaft || !m_destMineshaft || m_srcMineshaft==m_destMineshaft ||\n     !QuickDoesPathExist(obj,from,&m_srcMineshaft->m_position,0) ||\n     !QuickDoesPathExist(obj,&m_destMineshaft->m_position,rawTo,0)) {m_srcMineshaft=0;m_destMineshaft=0;}")
        o.append("  else {useMineshaft=true;adjustTo=m_srcMineshaft->m_position;rva002E75A3(&clipFrom,&adjustTo);}")
    o.append(" }")
    return '\n'.join(o)
orig = '\n'.join(lines[s:e])
assert block(0,0,0,0) == orig, '\n'.join(l for l in __import__('difflib').unified_diff(orig.split('\n'), block(0,0,0,0).split('\n'), lineterm=''))
for sw, cell, ori, cur, dum in itertools.product((0,1),(0,1,2),(0,1),(0,1),(0,1)):
    t = pre + '\n' + block(sw, cell, ori, cur) + ('\n g_rv=((int*)rawTo)[0];' if dum else '') + '\n' + post
    if dum: t = t.replace("Path *Pathfinder::internal_findHierarchicalPath(", "extern volatile int g_rv;\nPath *Pathfinder::internal_findHierarchicalPath(", 1)
    Path(f'v21/s{sw}c{cell}o{ori}u{cur}d{dum}.cpp').write_text(t, encoding='latin-1', newline='')
