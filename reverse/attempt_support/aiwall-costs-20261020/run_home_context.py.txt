from pathlib import Path
import sys,json,re
ROOT=Path(__file__).resolve().parents[2]
sys.path.insert(0,str(ROOT/'tools'))
from variant_search import Scorer
home=ROOT/'Code/GameEngine/Source/GameLogic/SkirmishAI/AIWallBuilder/AIWall.cpp'
text=home.read_text()
text=re.sub(r'#include "(\.[^"]+)"',lambda m:'#include "'+str((home.parent/m[1]).resolve()).replace('\\','/')+'"',text)
text=text.replace('    void rva004EBB58();','    void rva004EBB58();\n    float rva004EADBA(const Coord2D *pt);')
out=Path(__file__).parent/'home-context';rows=[]
s=Scorer('?distSqToSegment@@YAMPBVCoord2D@@00@Z',0x4eacf5,197,out)
for label,path in [('scalar','carriers/c1.cpp'),('ap','partial-coords/c18.cpp'),('ap-dir','partial-coords/c30.cpp'),('dot-sum','reference-values/c2.cpp')]:

 r=s(source);got=r.get('compiled',b'');target=r.get('target',b'')

 rows.append(row);print(json.dumps(row),flush=True)

(out/'results.json').write_text(json.dumps(rows,indent=2))
