import sys,subprocess,itertools
sys.path.insert(0,'build/action41d60b-codex8');import trial
base=trial.seed
clean=subprocess.check_output(['git','show','125d7a88ce:reverse/attempts/0x0041d60b.cpp']).decode()
regions=[('  case 0x52:case 0x61:case 0x67:case 0x6a: {','geometry',False),('  case 0x55:case 0x56:','placement',False),('  case 0x41: {','closest65',True),('  case 0x8c: {','closest140',True),('  case 0x57: {','healing',False),('  case 0x63: {','terrain63',False),('  case 0x48:case 0x49:case 0x4a: {','terrain72',False)]
items=[]
for cleanbit,s in enumerate([base,clean]):
 for mask in range(1,128):
  t=s.replace(' protected: Module *findModule',' public: Module *findModule')
  helpers=[]
  for n,(tag,name,hasbreak) in enumerate(regions):
   if mask>>n&1:
    a=t.index(tag);b=t.index('{',a);depth=1;e=b+1
    while depth:
     if t[e]=='{':depth+=1
     elif t[e]=='}':depth-=1
     e+=1
    body=t[b:e]
    if hasbreak:body=body.replace('   break;','   return true;')
    helpers.append('static __forceinline bool region_'+name+'(const Object *obj,const Coord3D *loc,const SpecialPowerTemplate *spTemplate,SpecialPowerModuleInterface *mod,NameKeyType castleKey)\n'+body+'\n')
    call='region_'+name+'(obj,loc,spTemplate,mod,castleKey)'
    use='{ if(!'+call+') return false; break; }' if hasbreak else '{ return '+call+'; }'
    t=t[:b]+use+t[e:]
  start=t.index('bool ActionManager::canDoSpecialPowerAtLocation(')
  t=t[:start]+''.join(helpers)+t[start:]
  items.append(('regions%d_%03d'%(cleanbit,mask),t))
trial.batch(items)
