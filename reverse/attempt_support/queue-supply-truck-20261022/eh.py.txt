import sys, struct
sys.path.insert(0,'tools')
import retail_body as rb, capstone
md=capstone.Cs(capstone.CS_ARCH_X86, capstone.CS_MODE_32)
B=rb.IMAGE_BASE
def dis(rva,n):
    for i in md.disasm(rb.read_bytes(rva,n), rva+B):
        print(f'  {i.address:08x} {i.mnemonic} {i.op_str}')
        if i.mnemonic in ('jmp','ret'): break
h=int(sys.argv[1],16)-B
print('IMAGE_BASE',hex(B)); dis(h,16)
i=next(md.disasm(rb.read_bytes(h,5),h))
fi=int(i.op_str.split(', ')[1],16)-B
magic,maxstate,umap=struct.unpack('<III',rb.read_bytes(fi,12))
print('maxstate',maxstate)
for s in range(maxstate):
    prev,act=struct.unpack('<iI',rb.read_bytes(umap-B+8*s,8))
    print('state',s,'prev',prev,'action',hex(act))
    if act: dis(act-B,64)
