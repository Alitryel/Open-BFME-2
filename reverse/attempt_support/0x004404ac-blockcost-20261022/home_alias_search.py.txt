from search import *
import itertools,re

def source(flags):
 kind,pos,mode=flags
 head=HEAD;body=BODY
 marks=['\n\tGameInfo *game','\n\tif (!game)','\n\tif (slot == -1 || slot >= 8)','\n\tbool observer']
 p=body.index(marks[pos]);before=body[:p];tail=body[p:]
 decls=[
 'int *home = &slot;',
 'int *const home = &slot;',
 'int *const &home = &slot;',
 'void *home = &slot;',
 'unsigned char *home = (unsigned char *)&slot;',
 'unsigned long home = (unsigned long)&slot;',
 'struct Home {int *p;}; Home home={&slot};',
 'int *home[1]={&slot};',
 ]
 accesses=['*home','*home','*home','*(int *)home','*(int *)home','*(int *)home','*home.p','*home[0]']
 access='('+accesses[kind]+')'
 if mode==0:tail=re.sub(r'\bslot\b',access,tail)
 elif mode==1:
  b=tail.index('\n\tbool observer')if '\n\tbool observer'in tail else 0
  tail=tail[:b]+re.sub(r'\bslot\b',access,tail[b:])
 elif mode==2:
  # Take only the original value through the stable parameter address;
  # the correction and following body keep an ordinary value local.
  q=tail.index('\n\tif (slot == -1 || slot >= 8)')if '\n\tif (slot == -1 || slot >= 8)'in tail else 0
  tail=tail[:q]+'\n int index='+access+';'+re.sub(r'\bslot\b','index',tail[q:])
 else:
  h='__forceinline int &cost_home_value(int *p){return *p;}\n'
  head=head+h
  tail=re.sub(r'\bslot\b','cost_home_value(&'+access+')',tail)
 return head+MARK+before+'\n '+decls[kind]+tail
if __name__=='__main__':run([('homealias-%03d'%n,source(f))for n,f in enumerate(itertools.product(range(8),range(4),range(4)))],'home_alias')
