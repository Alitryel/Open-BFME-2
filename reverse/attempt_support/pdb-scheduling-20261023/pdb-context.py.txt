from pathlib import Path
import sys,re,json,hashlib
sys.path.insert(0,'tools')
from variant_search import Scorer
from zh_sweep import carve
import build
base=Path('build/pdb-context-input.cpp.txt').read_text()
start=base.index('int _RTC_GetSrcLine(')
head,body=base[:start],base[start:]
variants=[('baseline',base)]
for mapping in [
 {'address':'lookupAddress','moduleName':'moduleFileName'},
 {'address':'instructionAddress','moduleName':'imageName'},
 {'address':'programCounter','moduleName':'imageFileName'},
 {'address':'lookupAddress'}, {'moduleName':'moduleFileName'},
 {'res':'result'}, {'errname':'errorName'},
 {'info':'imageInfo','imageAddr':'imageOffset','secAddr':'sectionOffset'},
]:
 changed=re.sub(r'\b('+'|'.join(mapping)+r')\b',lambda m:mapping[m[0]],body)
 variants.append(('identifiers-'+'-'.join(mapping.values()),head+changed))
decls=' struct SSrcModuleHdr {unsigned short cFile,cSeg;};\n struct SStartEnd {unsigned long start,end;};\n'
variants.append(('table-types-file-scope',decls+head+body.replace(decls,'')))
for where in ['entry','after-alloca','after-reset']:
 for pointers in [('moduleName',),('source','pline','moduleName')]:
  assume=''.join('__assume('+p+' != 0);' for p in pointers)
  if where=='entry':changed=body.replace(' {\n',' {\n'+assume+'\n',1)
  elif where=='after-alloca':changed=body.replace('char *errname=(char *)_alloca(1024);','char *errname=(char *)_alloca(1024);'+assume)
  else:changed=body.replace('*pline=0;source[0]=0;*moduleName=0;','*pline=0;source[0]=0;*moduleName=0;'+assume)
  variants.append((where+'-nonnull-'+'-'.join(pointers),head+changed))
s=Scorer('?_RTC_GetSrcLine@@YAHKPADHPAHPAPAD@Z',0x75b177,847,Path('build/pdb-context-trials').resolve())
records=[]
print('input-sha256',hashlib.sha256(base.encode()).hexdigest(),flush=True)
for name,src in variants:
 r=s(src);obj=s.dir/('c%d.obj'%s.n)
 extent=next((len(b) for n,b,_ in carve(obj.read_bytes()) if n==s.symbol),None) if obj.exists() else None
 diffs=[i for i,(a,b) in enumerate(zip(r.get('compiled',b''),r.get('target',b''))) if a!=b]
 item=dict(trial=name,n=s.n,extent=extent,differences=diffs,**{k:v for k,v in r.items() if k not in ('compiled','target')})
 print(item,flush=True);records.append(item)
 (s.dir/'results.json').write_text(json.dumps(records,indent=2)+'\n')
 if r['exact'] and extent==847:
  (s.dir/'exact.cpp').write_text(src)
  break
