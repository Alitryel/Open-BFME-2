from pathlib import Path
import runpy,itertools,concurrent.futures,json,re
n=runpy.run_path('reverse/attempt_support/canmake-early-cleanup-20261022/replay.py.txt');e=n['context']();r=e['r']
OUT=Path('build/canmake-read-priority-20261022').resolve();OUT.mkdir(exist_ok=True);r['run'].__globals__['OUT']=OUT

zh=root/'reference/open-bfme-1/inputs/reference/CnC_Generals_Zero_Hour/GeneralsMD/Code'
real='// platformsdk\nclass INI;\n'+e['prefix']
real=real.replace('/EHsc','/EHsc /DWIN32 /DNDEBUG /I"'+str(zh/'GameEngine/Include')+'" /I"'+str(zh/'Libraries/Include')+'"',1)
real=real.replace('#include "'+str(root/'Code/Libraries/Include/Lib/Coord3D.h')+'"','#include "'+str(zh/'GameEngine/Include/Common/Money.h')+'"')

typedef char real_money_size[(sizeof(Money)==12)?1:-1];""",real,count=1)+'\n'

def tests():
 for base,money,flag,s1,s2,m1,m2,c1,c2 in itertools.product(range(2),repeat=9):
  p=[e['prefix'],real][money];b=[n['funds_after'](e),n['early'](e)][base]
  if flag:b=b.replace('bool failed','volatile bool failed')
  status='(*(const volatile int *)&result.m_20)'
  if s1:b=b.replace('result.m_20==7',status+'==7').replace('result.m_20)\n {','result.m_20)\n {')
  if s2:b=b.replace('result.m_20!=0',status+'!=0')
  # early() uses a switch: classify its single status load as s1.
  if s1:b=b.replace('switch (result.m_20)','switch ('+status+')')
  normal='player->getMoney()->countMoney()' if money else 'player->m_money'

  costs=iter(['(*(const volatile int *)&result.m_pad04)' if c1 else 'result.m_pad04',
              'result.m_pad04',
              '(*(const volatile int *)&result.m_pad04)' if c2 else 'result.m_pad04'])
  b=re.sub(r'result\.m_pad04',lambda m:next(costs),b)
  yield 'read-'+''.join(map(str,[base,money,flag,s1,s2,m1,m2,c1,c2])),p+'\nint AIBuildableWallSegment::canMake(Player *player){'+b+'}\n'

 print(json.dumps({'trials':len(ds),'errors':sum('error'in d for d in ds),'unresolved':sum(bool(d.get('unresolved'))for d in ds),'exact':sum(d.get('exact_candidate',False)for d in ds),'unique':len({d.get('masked_code_sha256')for d in ds})}),flush=True)
