from pathlib import Path
import itertools
P=Path(__file__).resolve().parent
B=(P/'base.cpp').read_text()
D=P/'accessors';D.mkdir(exist_ok=True)
# Each of the seven source expressions has a separate context. Compare
# direct members with two inline boundaries, with the helpers returning an
# independent named copy and the source getter using one named parameter.
for mask in range(1024):
    s=B
    cc='__fastcall ' if mask&512 else ''
    s=s.replace('struct ContactData { void *owner; ContactSource *source; };',
       'struct ContactData { void *owner; ContactSource *source; '+
       'inline ContactSource *'+cc+'getSource() { ContactSource *p=source; return p; } };')
    s=s.replace(' ContactNode *next;',
       ' ContactNode *next;\n inline ContactData *'+cc+'getFirst() { ContactData *p=first; return p; }\n'+
       ' inline ContactData *'+cc+'getSecond() { ContactData *p=second; return p; }')
    # Nine independent expression groups, not nine textual substitutions
    # chosen by occurrence count.
    groups=[
      'if(node->first->source==0 || node->second->source==0) continue;',
      'node->first->source->test(',
      'node->second->source->prepare(node->point,node->normal))?1:2;',
      'node->first->source->respond(',
      'node->second->source->prepare(node->point,node->normal));',
      'if(node->first->source==0 || node->second->source==0) continue;',
      'node->second->source->respond(',
      'node->first->source->prepare(node->point,ContactVector(node->normal).negate())',
    ]
    # Replace first null group before second by placeholder to avoid repeat.
    s=s.replace(groups[0],'@GUARD0@',1)
    s=s.replace(groups[5],'@GUARD1@',1)
    for i,g in enumerate(groups):
       old='@GUARD0@' if i==0 else '@GUARD1@' if i==5 else g
       new=g
       if mask&(1<<i):
          new=new.replace('node->first->source','node->getFirst()->getSource()').replace('node->second->source','node->getSecond()->getSource()')
       if mask&256:
          new=new.replace('node->getFirst()->getSource()','node->first->getSource()').replace('node->getSecond()->getSource()','node->second->getSource()')
       s=s.replace(old,new,1)
    (D/('a%04d.cpp'%mask)).write_text(s)
