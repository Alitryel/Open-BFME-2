from pathlib import Path
import sys,json,hashlib,subprocess,itertools,concurrent.futures,struct,collections
ROOT=Path.cwd();OUT=ROOT/'build/trust237-round18'
sys.path.insert(0,str(ROOT/'tools'));import build
sys.path.insert(0,str(ROOT/'build/trust237-round17'));import original_check as prior
NAME=prior.NAME;RVA=prior.RVA;target=prior.target
bank=(ROOT/'reverse/attempts/0x00237d3d.cpp').read_text();prefix=bank[:bank.index('long status=WinVerifyTrust')]
A='0x80096010L';B='0x800b0100L'
forms=[
'switch(status){case A:case B:break;default:++r;}return r;',
'switch(status){default:++r;break;case A:case B:break;}return r;',
'switch(status){case A:case B:goto done;default:break;}++r;done:return r;',
'switch(status){default:goto good;case A:case B:goto done;}good:++r;done:return r;',
'switch(status){case A:break;case B:break;default:r+=1;break;}return r;',
'switch(status){case A:case B:return r;default:break;}return r+1;',
'switch(status){case A:case B:goto done;default:++r;goto done;}done:return r;',
'switch(status){case A:case B:break;default:r=1;break;}return r;',
]
# Intentionally diagnostic: these controls can return zero for the two
# independently recovered error values; never admit them against the NOP image.
jobs=[]
for form,kind,declpos,cpu,status_unsigned,nothrow in itertools.product(range(8),['int','bool','unsigned char','unsigned short'],range(4),[6,7],[False,True],[False,True]):
 text=prefix.replace('/G6',f'/G{cpu}')
 if nothrow:text=text.replace('void *__stdcall LocalFree(void *);','void *__stdcall LocalFree(void *) throw();')
 init=kind+' r=0;'
 if declpos==0:text=text.replace('    TrustFile file;',init+'\n    TrustFile file;')
 if declpos==1:text+=init
 text+=('unsigned long' if status_unsigned else 'long')+' status=WinVerifyTrust(0,&action,&data);'
 if declpos==2:text+=init
 text+='releaseArgs(argv);'
 if declpos==3:text+=init
 text+=forms[form].replace('A',A).replace('B',B)+'\n}\n'
 jobs.append((f's{len(jobs):04}',text,dict(form=form,kind=kind,declpos=declpos,cpu=cpu,status_unsigned=status_unsigned,nothrow=nothrow)))
def run(job):
 name,text,config=job;src=OUT/(name+'.cpp');obj=OUT/(name+'.obj');src.write_text(text)
 cmd,env=build.compiler_command(src,obj);p=subprocess.run(cmd,cwd=ROOT,env=env,capture_output=True,text=True)
 r=dict(name=name,config=config,source_sha256=hashlib.sha256(src.read_bytes()).hexdigest(),exit=p.returncode)
 if p.returncode:r['error']=(p.stdout+p.stderr)[-1600:];return r
 raw,rel=build.read_object_symbol_bytes(obj,NAME,code_only=True)
 known={'_memset':0x6291ae,'_WinVerifyTrust@12':0x655652}
 helpers={}
 cursor=(len(raw)+15)&~15
 for off,typ,sym in rel:
  if typ==20 and sym not in known:
   hb,hr=build.read_object_symbol_bytes(obj,sym,code_only=True)
   helpers[sym]=dict(rva=RVA+cursor,raw=hb.hex(),relocations=hr)
   known[sym]=RVA+cursor;cursor=(cursor+len(hb)+15)&~15
 resolved=bytearray(raw);un=[]
 for off,typ,sym in rel:
  if typ==6:
   api={'__imp__GetCommandLineW@0':'GetCommandLineW','__imp__CommandLineToArgvW@8':'CommandLineToArgvW','__imp__LocalFree@4':'LocalFree'}[sym]
   value=struct.pack('<I',prior.imports[api])
  else:
   assert typ==20
   value=struct.pack('<i',known[sym]-(RVA+off+4))
  assert resolved[off:off+4]==b'\0'*4
  resolved[off:off+4]=value
 r['same_unit_helpers']=helpers
 resolved=bytes(resolved)
 r.update(size=len(raw),raw_sha256=hashlib.sha256(raw).hexdigest(),relocations=rel,unresolved=un,diff=[i for i,(a,b) in enumerate(itertools.zip_longest(resolved,target)) if a!=b],
 prefix=next((i for i,(a,b) in enumerate(zip(resolved,target)) if a!=b),min(len(resolved),len(target))))
 return r
if __name__=='__main__':
 with concurrent.futures.ThreadPoolExecutor(max_workers=8) as pool:
  rows=[]
  for r in pool.map(run,jobs):
   rows.append(r)
   if len(rows)%128==0:print('compiled',len(rows),'of',len(jobs),flush=True)
 (OUT/'switch-results.json').write_text(json.dumps(rows,indent=2)+'\n')
 good=[r for r in rows if r['exit']==0]
 print('SUMMARY',json.dumps(dict(compiles=len(rows),good=len(good),distinct_raw=len({r['raw_sha256'] for r in good}),exact=sum(not r['diff'] for r in good))))
 for r in sorted(good,key=lambda r:len(r['diff']))[:8]:print(json.dumps({k:v for k,v in r.items() if k!='relocations'}))
