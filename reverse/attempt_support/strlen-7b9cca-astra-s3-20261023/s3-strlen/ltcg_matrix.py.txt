from pathlib import Path
import sys,subprocess,json,shutil,hashlib,itertools,concurrent.futures
sys.path.insert(0,"tools");import build,compiler_profiles
ROOT=Path.cwd();OUT=ROOT/"build/s3-strlen";target=build.read_target_bytes(0x7b9cca,24)
profiles={"vc71-sp1":compiler_profiles.SP1_FILES,"vc71-ddk-4035":compiler_profiles.DDK4035_FILES}
tool_inputs={}
for profile,files in profiles.items():
 lane=OUT/("ltcg-"+profile);lane.mkdir(exist_ok=True)
 for name,expected in files.items():
  src=ROOT/"vendor"/profile/"bin"/name
  assert hashlib.sha256(src.read_bytes()).hexdigest()==expected
  shutil.copy2(src,lane/name)
 shutil.copy2(build.vc71_root()/"Vc7/bin/link.exe",lane/"link.exe")
 tool_inputs[profile]={name:hashlib.sha256((lane/name).read_bytes()).hexdigest() for name in list(files)+["link.exe"]}
forms=[
("goto","const char*p=s;char c;if((c=*p++))goto loop;goto out;loop:while(c)c=*p++;out:return p-s-1;"),
("post","const char*p=s;while(*p++);return p-s-1;"),
("split","const char*p=s+1;char c=*s;while(c)c=*p++;return p-s-1;"),
("do","const char*p=s;char c=*p++;if(c)do{c=*p++;}while(c);return p-s-1;")]
def trial(job):
 idx,profile,flags,form=job;name,body=form;src=OUT/("ltcg%03d.cpp"%idx);obj=src.with_suffix(".obj");dll=src.with_suffix(".dll")
 src.write_text("// compiler: "+profile+"\n// cl: "+flags+" /GL /MD\nextern \"C\" __declspec(dllexport) int rva007B9CCAStrLen(const char*s){"+body+"}\n")
 cmd,env=build.compiler_command(src,obj)
 cp=subprocess.run(cmd,env=env,capture_output=True,text=True,timeout=90)
 rec=dict(index=idx,profile=profile,flags=flags,form=name,compile_returncode=cp.returncode,compile_log=cp.stdout+cp.stderr)
 if cp.returncode:return rec
 cmd=["wine",str(OUT/("ltcg-"+profile)/"link.exe"),"/nologo","/DLL","/NOENTRY","/INCREMENTAL:NO","/LTCG","/NODEFAULTLIB","/OUT:"+str(dll.relative_to(ROOT)),str(obj.relative_to(ROOT))]
 lp=subprocess.run(cmd,env=env,capture_output=True,text=True,timeout=90);rec.update(link_command=cmd,link_returncode=lp.returncode,link_log=lp.stdout+lp.stderr)
 if lp.returncode:return rec
 data=dll.read_bytes();sections=build.pe_sections(data);pe=build.u32(data,0x3c);er=build.u32(data,pe+24+96);ed=build.read_pe_bytes(data,sections,er,40)
 assert build.u32(ed,20)==1 and build.u32(ed,24)==1
 table=build.read_pe_bytes(data,sections,build.u32(ed,28),4);rva=build.u32(table,0)
 text=next(s for s in sections if s["name"]==".text")
 raw=build.read_pe_bytes(data,sections,rva,text["rva"]+text["virtual_size"]-rva).rstrip(b"\xcc")
 rec.update(export_rva=hex(rva),size=len(raw),hex=raw.hex(),exact=raw==target,diff=sum(a!=b for a,b in itertools.zip_longest(raw,target)))
 return rec
jobs=[(i,*x) for i,x in enumerate(itertools.product(profiles,["/O1 /G6","/O1 /G5","/O2 /G6","/O2 /Os /G6"],forms))]
with (OUT/"ltcg-matrix.jsonl").open("w") as f,concurrent.futures.ThreadPoolExecutor(max_workers=2) as pool:
 for r in pool.map(trial,jobs):
  f.write(json.dumps(r)+"\n");f.flush()
  if r.get("exact") or r["index"]%8==0:print(json.dumps(r),flush=True)
(OUT/"ltcg-tool-inputs.json").write_text(json.dumps(tool_inputs,indent=2))
print("DONE",len(jobs),flush=True)
