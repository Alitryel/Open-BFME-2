from pathlib import Path
import sys, subprocess, re, json
root = next(p for p in Path(__file__).resolve().parents
            if (p / 'tools/explain_mismatch.py').exists())
sys.path.insert(0, str(root / 'tools'))
out = root / 'build/water-buffer-lifetimes-replay'
out.mkdir(exist_ok=True)
support = root / 'reverse/attempt_support/water-buffer-lifetimes-20261020'
home = root / 'Code/GameEngineDevice/Source/W3DDevice/GameClient/Water/W3DWater.cpp'
symbol = '?AllocateAndFillBuffers@RenderableStandingWaterArea@@QAEXXZ'
prior = (support / 'prior.cpp.txt').read_text()
prior = re.sub(r'#include "([.][.]/[^"]+)"',
               lambda m: '#include "' + str((home.parent / m[1]).resolve()).replace('\\', '/') + '"', prior)
prior = prior.replace('class Rva0030812E { public: void *rva0030812E(int index); };',
                      'class Rva0030812E { public: void *getTextureName(int index); };')
prior = prior.replace('m_area->rva0030812E(index)', 'm_area->getTextureName(index)')
old = out / 'prior.cpp'
old.write_text(prior)
bank = root / 'reverse/attempts/0x0008304a.cpp'
results = []
for label, source in [('prior', old), ('bank', bank)]:
    args = [sys.executable, 'tools/explain_mismatch.py', symbol,
            '--rva', '0x8304A', '--size', '1156', '--source', source.as_posix()]
    p = subprocess.run(args, cwd=root, capture_output=True, text=True,
                       encoding='utf8', errors='replace', timeout=300)
    (out / (label + '-mismatch.txt')).write_text(p.stdout + p.stderr)
    assert p.returncode == 0, p.stdout[-1000:] + p.stderr[-1000:]
    import build, permute, eh_verify
    obj = build.obj_path(source)
    assert obj.exists(), obj
    q = subprocess.run([sys.executable, 'tools/align_diff.py', source.as_posix(),
                        symbol, '0x8304A', '1156'], cwd=root, capture_output=True,
                       text=True, encoding='utf8', errors='replace', timeout=120)
    (out / (label + '-alignment.txt')).write_text(q.stdout + q.stderr)
    assert q.returncode == 0, q.stderr
    before = obj.stat().st_mtime_ns
    row = dict(name=symbol, target_rva='0x0008304A', target_size='1156',
               source=source.relative_to(root).as_posix(), notes='')
    got = build.compile_function(row, build.load_symbol_map(), obj)
    assert obj.stat().st_mtime_ns == before
    result = dict(source=str(source.relative_to(root)),
                  score=permute.fitness(got['bytes'], got['target']),
                  unresolved=got['unresolved'],
                  eh=eh_verify.verify_row(eh_verify.Image(), eh_verify.Obj(obj),
                                          symbol, 0x8304A))
    results.append(result)
    print(label, result, flush=True)
(out / 'results.json').write_text(json.dumps(results, indent=2))
