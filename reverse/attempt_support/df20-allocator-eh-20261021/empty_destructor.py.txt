from pathlib import Path
import sys,itertools,concurrent.futures,json
sys.path.insert(0,str(Path(__file__).parent));import trials
trials.scratch=Path(__file__).resolve().parent
base=(trials.scratch/'bank.cpp').read_text();trials.symbol=base.splitlines()[0][3:]
decl='template <> _String_base<wchar_t, allocator<wchar_t> >::~_String_base();'
specs=[]
for attr,wide,narrow,visible in itertools.product(['inline','__forceinline'],[0,1],[0,1],[0,1]):
 if not(wide or narrow):continue
 s=base.replace(decl,'')
 definition='namespace _STL {\n'
 if not visible:definition+=decl+'\n'
 # The scalar character range has no nontrivial elements to destroy.
 # An empty body is semantically the vendor's POD _Destroy specialization.
 if wide:definition+=f'template <> {attr} basic_string<wchar_t>::~basic_string() {{}}\n'
 if narrow:definition+=f'template <> {attr} basic_string<char>::~basic_string() {{}}\n'
 definition+='}\n'
 s=s.replace('#include <locale>',definition+'#include <locale>',1)
 specs.append((f'empty-fixed-a{attr}-w{wide}-n{narrow}-v{visible}',s))
with concurrent.futures.ThreadPoolExecutor(max_workers=8) as pool:r=list(pool.map(trials.run,specs))
(trials.scratch/'empty_destructor.json').write_text(json.dumps(r,indent=2))
