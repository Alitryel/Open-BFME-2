# Read-only IR snapshots before/after the first split and its rebuild.
import sys,json,re
from pathlib import Path
sys.path.insert(0,'build/netpacket-20261022e');from wdbg import Dbg
src,obj=sys.argv[1:3];wanted=int(sys.argv[3],0) if len(sys.argv)>3 else 0xc726e0
d=Dbg(src,obj,obj+'.debug.log');d.prompt();d.prompt('set $CanDeferOnBPByAddr = 1');d.prompt('break *0x10783aac')
on=False;bps=[];done=set();snaps=[]
def bits(h):
 p=d.words(h,1)[0] if h else 0;r=[];seen=set()
 while p and p not in seen:
  seen.add(p);a,n,b=d.words(p,3);r.extend(a+i for i in range(32)if b>>i&1);p=n
 return r
while True:
 out=d.prompt('cont')
 if 'terminated'in out:break
 ip=d.reg('eip')-0x10700000
 if ip==0x83aac:
  f=d.words(d.reg('ebp'),1)[0];name=d.string(d.words(f+4,1)[0]);yes='?Init@FXShaderGeometry'in name and d.reg('ebx')==0
  if yes and not on:
   on=True
   for addr in [0xa8f54,0x8ab54,0x8ab66]:
    s=d.prompt('break *0x%x'%(0x10700000+addr));bps.append(int(re.search(r'Breakpoint (\d+)',s)[1]))
  elif not yes and on:
   for n in bps:d.prompt('disable %d'%n)
   on=False
  continue
 if not on or ip in done:continue
 done.add(ip)
 snap={'ip':hex(ip),'wanted':hex(wanted),'block':d.words(wanted,48),'tuples':{},'operands':{},'symbols':{},'sets':{}}
 for off in [0x48,0x4c,0x50]:snap['sets'][hex(off)]=bits(d.words(wanted+off,1)[0])
 b=snap['block'];p=b[7];last=b[8];seen=set()
 while p and p not in seen and len(seen)<180:
  seen.add(p);w=d.words(p,16);snap['tuples'][p]=w
  if len(w)<16:break
  for off in [10,11]:
   a=w[off];ops=set()
   while 0x10000<a<0x10000000 and a not in ops and len(ops)<20:
    ops.add(a);aw=d.words(a,16);snap['operands'][a]=aw
    if len(aw)<16:break
    if aw[2]&255 in [1,2] and 0x10000<aw[6]<0x10000000:snap['symbols'][aw[6]]=d.words(aw[6],24)
    a=aw[0]
  if p==last:break
  p=w[0]
 snaps.append(snap);Path(obj+'.blocks.json').write_text(json.dumps(snaps))
 print(hex(ip),'tuples',len(snap['tuples']),'start/end',hex(b[7]),hex(b[8]),flush=True)
 if len(done)==3:
  for n in bps:d.prompt('disable %d'%n)
print('DONE',flush=True)
