from search import *
import itertools
old='''struct SortList
{
\tRva009A3AD0Endpoint *head;
\tRva009A3AD0Endpoint *tail;
\tunsigned int count;
};'''
assert old in BASE
items=[]
for n,(shape,qual,ctor,dtor,indexed) in enumerate(itertools.product(range(6),range(8),range(5),range(2),range(2))):
 h='Rva009A3AD0Endpoint *'+(' volatile' if qual&1 else '')+' head;'
 t='Rva009A3AD0Endpoint *'+(' volatile' if qual&2 else '')+' tail;'
 c=('volatile ' if qual&4 else '')+'unsigned int count;'
 members=[h,t,c]
 prefix=['','struct ListBase { '+h+' };','struct ListBase { '+h+t+' };','struct ListBase { '+h+t+c+' };','struct ListBase { '+h+' }; struct ListMiddle : ListBase { '+t+' };','struct ListBase { '+h+' };'][shape]
 parent=['', ' : ListBase', ' : ListBase', ' : ListBase', ' : ListMiddle', ' : ListBase'][shape]
 fields=[''.join(members),t+c,c,'',c,'struct { '+t+c+' };'][shape]
 method=['','SortList() {}','SortList() { count = 0; }','SortList() { head = 0; tail = 0; count = 0; }','SortList() { head = 0; }'][ctor]
 if dtor:method+=' ~SortList() {}'
 decl=prefix+' struct SortList'+parent+' { '+fields+method+' };'
 s=BASE.replace(old,decl)
 if indexed:s=s.replace('SortList *B = A + 1;','SortList *B = &lists[src+1];')
 items.append(('construct-%04d'%n,s))
batch(items,'construction')
