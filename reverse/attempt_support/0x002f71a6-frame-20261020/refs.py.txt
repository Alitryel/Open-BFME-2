"""refs.py file.cpp toff1,toff2.. ooff1,ooff2..: list instructions touching the given ebp offsets in retail (t) and ours (o)."""
import sys, re
sys.path.insert(0, '/work3/Open-BFME-2/build/h72')
from score import compile_, md
pat = re.compile(r'\[ebp ([-+]) (0x[0-9a-f]+|\d+)\]')
f = sys.argv[1]
T = [int(x, 0) for x in sys.argv[2].split(',')]
O = [int(x, 0) for x in sys.argv[3].split(',')]
(got, raw), _ = compile_(f)


def lst(b, offs):
    r = []
    for i in md.disasm(bytes(b), 0):
        for m in pat.finditer(i.op_str):
            off = int(m.group(2), 0) * (-1 if m.group(1) == '-' else 1)
            if off in offs:
                r.append(f'{i.address:4x} {i.mnemonic} {i.op_str}')
    return r


t = lst(got['target'], T); o = lst(got['bytes'], O)
for k in range(max(len(t), len(o))):
    print(f'{t[k] if k < len(t) else "":<44} | {o[k] if k < len(o) else ""}')
