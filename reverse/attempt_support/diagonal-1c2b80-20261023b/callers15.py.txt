import sys,os,collections
sys.path.insert(0,'/work/Open-BFME-2/build/c2tr')  # wdbg.py.txt from this directory
from wdbg import Dbg
src=sys.argv[1]; tag=sys.argv[2]
W='/work/Open-BFME-2/build/db/tr/'
d=Dbg(src,W+tag+'.cl.obj',W+tag+'.cl.log')
active=False; ev=[]
try:
    d.prompt();d.prompt('set $CanDeferOnBPByAddr = 1');d.prompt('break *0x10760539')
    while True:
        o=d.prompt('cont',300)
        if 'terminated' in o or not o.strip(): break
        if '0x10760539' in o:
            if active: continue
            w=d.words(d.reg('eax'),1);s=d.words(w[0],2);fn=d.string(s[1])
            if 'DiagonalBlur' in fn:
                active=True;d.prompt('break *0x1076045d');d.prompt('break *0x10760853')
            continue
        if active and ('0x1076045d' in o or '0x10760853' in o):
            k='15' if '0x1076045d' in o else '3'
            esp=d.reg('esp'); st=d.words(esp,64)
            ra=[x-0x10700000 for x in st if 0x10701000<=x<0x10870000][:5]
            ev.append((k,d.words(0x108a79d0,1)[0],ra))
finally:
    with open(W+tag+'.callers','w') as f:
        for k,g,ra in ev: f.write('%s %d %s\n'%(k,g,' '.join('%x'%r for r in ra)))
    os.path.exists(W+tag+'.cl.obj') and os.unlink(W+tag+'.cl.obj')
    if d.p.poll() is None:
        try: d.prompt('quit')
        except Exception: d.p.kill()
print('done',len(ev))
