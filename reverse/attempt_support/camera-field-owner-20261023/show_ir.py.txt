import json,sys
from pathlib import Path
D=json.loads(Path(sys.argv[1]).read_text())
for s in D:
 nodes=s['nodes'];syms=s['syms']
 def name(p):
  sw=syms.get(str(p),[])
  if len(sw)>7:
   if 0x108a9804<=p<0x108a9b04:return 'R'+str((p-0x108a97a4)//96)
   return 'H'+str(sw[7])
  return hex(p)
 def op(p,seen=()):
  if p in seen:return '<cycle>'
  w=nodes.get(str(p),[])
  if len(w)<16:return hex(p)
  k=w[2]&255; seen=seen+(p,)
  if k in [1,2,4]:return name(w[7] or w[6])
  if k in [5,6]:return ('&' if k==5 else '*')+'('+op(w[13],seen)+'+'+hex(w[12])+')'
  if k==7:return '#'+hex(w[6])
  if k==15:return 'CALL '+','.join(hex(x) for x in w[6:16])
  return 'k%d@%x'%(k,p)
 print('\nSTAGE',s['stage'])
 for key,bw in s['bbs'].items():
  n=bw[7];lines=[];seen=set();calls=False
  while n and n not in seen:
   seen.add(n);w=nodes[str(n)];k=w[2]&255
   if k==15:calls=True
   if k in [13,18]:
    srcs=[];p=w[10];ps=set()
    while p and p not in ps:
     ps.add(p);srcs.append(op(p));p=nodes.get(str(p),[0])[0]
    args=','.join(srcs)+' -> '+op(w[11])
   elif k==15:args=op(n)
   else:args='k%d'%k
   lines.append('%08x op%04x %s'%(n,w[1],args))
   if n==bw[8]:break
   n=w[0]
  if calls:print('BB',key+'\n'+'\n'.join(lines))
