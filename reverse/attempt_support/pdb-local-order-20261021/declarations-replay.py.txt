from pathlib import Path
import concurrent.futures,subprocess,sys,json
out=Path('build/resume-eh-local/turn62-pdb-declarations');out.mkdir(exist_ok=True)
base=Path('reverse/attempt_support/pdb-numbering-20261021/candidate.cpp.txt').read_text()
old=' char *errname=(char *)_alloca(1024);\n int res=0;ImageInfo *info;long imageAddr,secAddr;WORD sectionIndex;\n PDB *ppdb;DBI *pdbi;Mod *pmod;long err,offsetRes,size;DWORD fo,cb;char *pdbname;'
assert base.count(old)==1
parts=['char *errname;','int res;','ImageInfo *info;','long imageAddr;','long secAddr;','WORD sectionIndex;','PDB *ppdb;','DBI *pdbi;','Mod *pmod;','long err;','long offsetRes;','long size;','DWORD fo;','DWORD cb;','char *pdbname;']
variants=[('baseline',base)]
for reverse in [False,True]:
 for k in range(len(parts)):
  group=list(reversed(parts)) if reverse else parts[:]
  group=group[k:]+group[:k]
  body=' '+''.join(group)+'\n errname=(char *)_alloca(1024);\n res=0;'
  variants.append((f'order_{int(reverse)}_{k:02}',base.replace(old,body)))
# Place address-escaping PDB outputs in their owning control scope, after the unchanged operation order.
for group in ['PDB *ppdb;DBI *pdbi;Mod *pmod;','long err,offsetRes,size;DWORD fo,cb;char *pdbname;','PDB *ppdb;DBI *pdbi;Mod *pmod;long err,offsetRes,size;DWORD fo,cb;char *pdbname;']:
 variants.append((f'scope_{len(variants)}',base.replace(group,'').replace(' {WORD sectionIndexRes;',' {'+group+'WORD sectionIndexRes;')))
name='?_RTC_GetSrcLine@@YAHKPADHPAHPAPAD@Z'
def run(item):
 key,source=item;p=out/(key+'.cpp');p.write_bytes(source.encode())
 r=subprocess.run([sys.executable,'tools/explain_mismatch.py',name,'--rva','0x0075B177','--size','847','--source',p.as_posix()],stdout=subprocess.PIPE,stderr=subprocess.STDOUT,text=True)
 (out/(key+'.txt')).write_bytes(r.stdout.encode());d={'variant':key,'exit':r.returncode,'source':p.as_posix()}
 for field in ['compiled symbol size','classification','first diff:','unresolved']:
  lines=[s for s in r.stdout.splitlines() if s.lower().startswith(field)]
  if lines:d[field]=lines[0]
 return d
results=[]
with concurrent.futures.ThreadPoolExecutor(max_workers=2) as pool:
 for d in pool.map(run,variants):
  results.append(d)
  print(d,flush=True)
  (out/'results.json').write_bytes((json.dumps(results,indent=2)+'\n').encode())
