from search import *
import itertools

starts=[BODY.index('\n\tbool observer'),BODY.index('\n\tif (m_handicap'),BODY.index('\n\tGameWindow *hero'),BODY.index('\n\tif (m_playerTemplate'),BODY.index('\n\tGameWindow **team'),BODY.index('\n\tif (slot == game->v13())'),BODY.rfind('\n}')]
segments=[BODY[starts[k]:starts[k+1]]for k in range(6)]
def source(flags):
 mask,handler=flags
 catch=['catch(...) {throw;}','catch(int) {throw;}','catch(const char *) {throw;}'][handler]
 body=BODY
 for k in range(6):
  if mask&(1<<k):
   # observer must survive later segments, so the first region ends after its
   # declaration but covers the lifetime of the combo wrapper.
   text=segments[k]
   if k==0:
    p=text.index(';',text.index('\n\tMpGameSetupComboRef'))+1
    text=text[:p]+'\n try{'+text[p:]+'\n }'+catch
   else:text='\n try{'+text+'\n }'+catch
   body=body.replace(segments[k],text)
 return HEAD+MARK+body
if __name__=='__main__':
 run([('ehregion-%03d'%n,source(f))for n,f in enumerate(itertools.product(range(64),range(3)))], 'eh_regions')
