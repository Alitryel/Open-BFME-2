from search import *
DECLS=['bool HandleHeroSelection(int index);','void UpdateHeroDisplay(int slot);',
       'virtual bool v12();','virtual int v13();']
for d in DECLS:assert d in HEAD,d
def source(n):
    head=HEAD
    for i,d in enumerate(DECLS):
        if n&(1<<i):head=head.replace(d,d[:-1]+' throw();')
    return head+MARK+BODY
if __name__=='__main__':run([('mthrow-%02d'%n,source(n)) for n in range(16)],'member_throw')
