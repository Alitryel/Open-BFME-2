from pathlib import Path
import sys,itertools
ROOT=Path.cwd();OUT=ROOT/'build/trust237-round10'
sys.path.insert(0,str(ROOT/'build/trust237-round9'))
import frontends as f
f.OUT=OUT
bank=(ROOT/'reverse/attempts/0x00237d3d.cpp').read_text()
# Intrinsics are legal C++ atomic operations on private locals. These are
# producer controls, not claims about the target's original source.
prototypes='extern "C" long __cdecl _InterlockedExchange(long volatile *,long);\nextern "C" long __cdecl _InterlockedCompareExchange(long volatile *,long,long);\n#pragma intrinsic(_InterlockedExchange,_InterlockedCompareExchange)\n'
jobs=[]
for operation,count,volatile,init,scope,pos,cpu,opt in itertools.product(['exchange','compare'],[1,16],['','volatile '],['0','status'],['shared','per_op'],['before','after'],[6,7],['O1','O2']):
    call='_InterlockedExchange(&word,word);' if operation=='exchange' else '_InterlockedCompareExchange(&word,word,word);'
    decl=volatile+'long word='+init+';'
    body=decl+call*count if scope=='shared' else ('{'+decl+call+'}')*count
    # No external data changes; the bank's status and unconditional result survive.
    loc='LocalFree(argv);'
    text=bank.replace('/O1 /G6',f'/{opt} /G{cpu}')
    text=text.replace('int Rva00237D3DCheck() {',prototypes+'int Rva00237D3DCheck() {')
    text=text.replace(loc,body+loc if pos=='before' else loc+body)
    jobs.append((f'x{len(jobs):04}',text,dict(operation=operation,count=count,volatile=volatile,init=init,scope=scope,pos=pos,cpu=cpu,opt=opt)))
f.execute('exchanges',jobs)
# Small positive controls make the native producer easy to trace separately.
for operation in ['exchange','compare']:
    call='_InterlockedExchange(&word,word)' if operation=='exchange' else '_InterlockedCompareExchange(&word,word,word)'
    (OUT/(operation+'-control.cpp')).write_text('// cl: /O1 /G6 /MD /EHsc\n'+prototypes+'extern "C" int exchangeProbe(int x) {long word=x;'+call+'; return 1;}\n')
