from pathlib import Path
import re,sys,json,subprocess,hashlib
sys.path.insert(0,'tools')
import build,capstone

out=Path('build/resume-eh-local/turn68-mesh-addressing');out.mkdir(exist_ok=True)
base=Path('reverse/attempt_support/mesh-real-references-20261023/diagnostic-base.cpp.txt').read_text()
start=base.index('__forceinline void BoneTransform::TransformPositionArray')
end=base.index('\nclass SkinMatrix3D',start)
old=base[start:end]
shims=''.join('  int a%d=count;\n'%i for i in range(4))
assert shims in old
variants={}
for keep in (True,False):
    control=old if keep else old.replace(shims,'')
    prefix='bank_' if keep else 'real_'
    shapes={'control':control}
    shapes['output_index']=control.replace('out->','out[0].')
    shapes['input_index']=control.replace('v->','v[0].')
    shapes['output_ref']=control.replace(' while(count--) {',' while(count--) {\n  Vector3 &o=*out;').replace('out->','o.')
    shapes['input_ref']=control.replace(' while(count--) {',' while(count--) {\n  const Vector3 &in=*v;').replace('v->','in.')
    shapes['both_refs']=control.replace(' while(count--) {',' while(count--) {\n  const Vector3 &in=*v;\n  Vector3 &o=*out;').replace('v->','in.').replace('out->','o.')
    # Named scalar input values are real operands, not inert counter padding.
    shapes['input_scalars']=control.replace(' while(count--) {',' while(count--) {\n  float vx=v->X,vy=v->Y,vz=v->Z;')
    before,body=shapes['input_scalars'].split('  float cx=',1)
    shapes['input_scalars']=before+'  float cx='+body.replace('v->X','vx').replace('v->Y','vy').replace('v->Z','vz')
    shapes['input_copy']=control.replace(' while(count--) {',' while(count--) {\n  Vector3 in=*v;').replace('v->','in.')
    shapes['input_deref']=control.replace('v->','(*v).')
    shapes['output_deref']=control.replace('out->','(*out).')
    # Use the established vector setters/constructor with the same three values.
    expr=re.findall(r'  out->([XYZ])=(.*);',control)
    assert [k for k,v in expr]==list('XYZ')
    stores='\n'.join('  out->%s=%s;'%(k,v) for k,v in expr)
    shapes['output_set']=control.replace(stores,'  out->Set('+','.join(v for k,v in expr)+');')
    shapes['output_value']=control.replace(stores,'  *out=Vector3('+','.join(v for k,v in expr)+');')
    shapes['output_temp']=control.replace(stores,'  Vector3 result('+','.join(v for k,v in expr)+');\n  *out=result;')
    for key,shape in shapes.items():variants[prefix+key]=base[:start]+shape+base[end:]

sym='?get_deformed_vertices@MeshModelClass@@QAEXPAVVector3@@PBVHTreeClass@@@Z'
maps=build.load_symbol_map();reports=[]
for key,source in variants.items():
    p=out/(key+'.cpp');p.write_bytes(source.encode())
    r=subprocess.run([sys.executable,'tools/explain_mismatch.py',sym,'--rva','0x0016B680','--size','2237','--source',p.as_posix()],stdout=subprocess.PIPE,stderr=subprocess.STDOUT,text=True,errors='replace')
    p.with_suffix('.txt').write_text(r.stdout,encoding='utf-8')
    d={'variant':key,'exit':r.returncode,'source_sha256':hashlib.sha256(p.read_bytes()).hexdigest()}
    if r.returncode:d['error']=r.stdout[-1500:]
    else:
        obj=build.obj_path(p.resolve());raw,_=build.read_object_symbol_bytes(obj,sym)
        row={'name':sym,'target_rva':'0x0016B680','target_size':'2237','source':p.as_posix(),'status':'matched','notes':''}
        patch=build.compile_function(row,maps,obj)
        a,b=patch['target'],patch['bytes']
        d.update(extent=len(raw),unresolved=patch['unresolved'],region_byte_diffs=[sum(x!=y for x,y in zip(a[l:h],b[l:h])) for l,h in [(0,0x580),(0x580,0x700),(0x700,2237)]],byte_diffs=sum(x!=y for x,y in zip(a,b)),patched_sha256=hashlib.sha256(b).hexdigest(),exact=len(raw)==2237 and not patch['unresolved'] and a==b)
    reports.append(d)
    print(json.dumps(d),flush=True)
    (out/'results.json').write_text(json.dumps(reports,indent=2)+'\n')
