from headmerge import *

# The ZH source declares an enum DrawableID and a const mouse-query method.
# Cross these ABI-compatible declarations with the late-merge contexts;
# unsigned is a separate diagnostic hypothesis, not a recovered declaration.
enum='enum DrawableID { INVALID_DRAWABLE_ID=0, FORCE_DRAWABLEID_TO_LONG_SIZE=0x7ffffff };\n'
items=[]
for context,ty,cq,cf,cs,contract in itertools.product(range(4),range(3),range(2),range(2),range(2),range(2)):
 src=BASE if context==0 else BASE[:lo]+decl+forms[[8,9,1][context-1]]+'\n'+BASE[hi:]
 if ty:
  kind=['int','DrawableID','unsigned'][ty]
  if ty==1:src=src.replace('struct InGameUI',enum+'struct InGameUI',1)
  src=src.replace('virtual int getMousedOverDrawableID();','virtual '+kind+' getMousedOverDrawableID();')
  src=src.replace('findDrawableByID(int id)','findDrawableByID('+kind+' id)')
 if cq:src=src.replace('getMousedOverDrawableID();','getMousedOverDrawableID() const;')
 if cf:src=src.replace('findDrawableByID(int id);','findDrawableByID(int id) const;').replace('findDrawableByID(DrawableID id);','findDrawableByID(DrawableID id) const;').replace('findDrawableByID(unsigned id);','findDrawableByID(unsigned id) const;')
 if cs:src=src.replace('virtual Drawable *getFirstSelectedDrawable();','virtual Drawable *getFirstSelectedDrawable() const;')
 if not contract:src=src.replace('template <> void StringBase<char>::releaseBuffer() throw();\n','')
 items.append(('id%d%d%d%d%d%d'%(context,ty,cq,cf,cs,contract),src))
if __name__=='__main__':batch(items,'idtypes')
