from pathlib import Path
import sys,subprocess,itertools,json,concurrent.futures
sys.path.insert(0,str(Path('build/trust237-session').resolve()))
import search
from search import OUT,ROOT,NAME,RVA,SIZE,TARGET,build,resolve_bytes
profiles=['rtm','vc71-sp1','vc71-ddk-4035']
def run(job):
 n,profile,tail=job;src=OUT/f'c{n:03}.cpp';obj=src.with_suffix('.obj')
 text=search.make_source(n,0,0,0,0,0,0,'/O1 /G7 /arch:SSE',tail)
 if profile!='rtm':text='// compiler: '+profile+'\n'+text
 src.write_text(text)
 try:
  cmd,env=build.compiler_command(src,obj);p=subprocess.run(cmd,env=env,cwd=ROOT,capture_output=True,text=True)
  if p.returncode:return {'n':n,'profile':profile,'tail':tail,'error':(p.stdout+p.stderr)[-600:]}
  raw,rel=build.read_object_symbol_bytes(obj,NAME,code_only=True)
  blob,un=resolve_bytes(raw,rel,TARGET+b'\0'*max(0,len(raw)-SIZE),RVA,len(raw),{'_memset':[0x6291ae],'_WinVerifyTrust@12':[0x655652]})
  return {'n':n,'profile':profile,'tail':tail,'size':len(raw),'nop16':b'\x90'*16 in raw,'diff':sum(a!=b for a,b in itertools.zip_longest(blob,TARGET)),'prefix':next((i for i,(a,b) in enumerate(zip(blob,TARGET)) if a!=b),min(len(raw),SIZE)),'last':raw[-36:].hex()}
 except BaseException as e:return {'n':n,'profile':profile,'tail':tail,'error':str(e)}
jobs=[(n,*x) for n,x in enumerate(itertools.product(profiles,range(16)))]
with concurrent.futures.ThreadPoolExecutor(max_workers=8) as pool:rows=list(pool.map(run,jobs))
(OUT/'profiles.json').write_text(json.dumps(rows,indent=2));print('total',len(rows),'errors',sum('error' in r for r in rows))
for r in sorted(rows,key=lambda r:r.get('diff',999))[:4]:print(json.dumps(r))
