# usage: mk.py out.cpp [edit-file ...]; edit files hold "OLD\n=====\nNEW" blocks separated by "\n%%%%%\n"
import sys
from pathlib import Path
R = Path(__file__).resolve().parent
b = (R / "bank.cpp").read_text(encoding="utf-8")
import os
w2c = (R / "w2c.txt").read_text(encoding="utf-8")
if os.environ.get("NO_W2C"): w2c = w2c[:w2c.index("// 0x002E7875")]
anchor = "extern \"C\" int __cdecl abs( int n );\n"
assert b.count(anchor) == 1
b = b.replace(anchor, w2c + anchor)
for ef in sys.argv[2:]:
    for blk in (R / ef).read_text(encoding="utf-8").split("\n%%%%%\n"):
        if not blk.strip(): continue
        old, new = blk.split("\n=====\n")
        assert b.count(old) == 1, (ef, old[:80], b.count(old))
        b = b.replace(old, new)
(R / sys.argv[1]).write_text(b, encoding="utf-8", newline="\n")
