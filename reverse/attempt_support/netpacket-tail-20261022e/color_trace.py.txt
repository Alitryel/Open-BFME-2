"""Read-only MSVC 7.1 allocator observation; no compiler-state writes."""
import sys,json
from pathlib import Path
from wdbg import Dbg
src,obj=sys.argv[1:3];d=Dbg(src,obj,obj+'.debug.log')
d.prompt();d.prompt('set $CanDeferOnBPByAddr = 1')
BASE=0x10700000;bps=[0x83acd,0x82978,0x82a4d,0x831f7,0xa8f54];added=set();names={}
for b in bps:
 o=d.prompt('break *0x%x'%(BASE+b))
 if 'Breakpoint'in o and 'Unable'not in o:added.add(b)
def nm(sym):
 if sym in names:return names[sym]
 sw=d.words(sym,1);pw=d.words(sw[0],2) if sw and sw[0]>0x100000 else []
 n=d.string(pw[1]).strip().strip('"').lstrip('_') if len(pw)>1 and pw[1]>0x100000 else '?'
 names[sym]=n;return n
def members(h):
 p=d.words(h,1)[0] if h else 0;out=[];seen=set()
 while p and p not in seen:
  seen.add(p);a,n,b=d.words(p,3);out.extend(a+i for i in range(32) if b>>i&1);p=n
 return out
def rec(idx):
 p=d.words(0x108807b8+4*(idx&0x3ff),1)[0]
 while p:
  w=d.words(p,18)
  if w[7]==idx:return p,w
  p=w[12]
 return 0,[]
def info(p,w=None):
 w=w or d.words(p,18)
 return dict(address=p,name=nm(w[0]),idx=w[7],words=w,allowed=members(w[8]),register=d.words(w[4]+0x1c,1)[0] if w[4] else None)
events=[];on=False;k=0
while True:
 out=d.prompt('cont')
 if 'terminated'in out:break
 eip=d.reg('eip')-BASE
 for b in bps:
  if b not in added and b!=eip:d.prompt('break *0x%x'%(BASE+b));added.add(b)
 added.add(eip)
 if eip==0x83acd:
  f=d.words(d.reg('ebp'),1)[0];fn=d.string(d.words(f+4,1)[0]);on='getCommandList'in fn and d.reg('ebx')==0;k=0
  continue
 if not on:continue
 if eip==0x82978:k+=1;continue
 event=dict(round=k)
 if eip==0x82a4d:
  p=d.reg('eax');w=d.words(p,18)
  if nm(w[0])not in ['i','this']:continue
  event.update(kind='pop',record=info(p,w))
 elif eip==0x831f7:
  st=d.words(d.reg('esp'),4);p=st[2];w=d.words(p,18);neighbours=[]
  for idx in members(st[3]):
   q,v=rec(idx)
   if v and nm(v[0])in ['i','this']:neighbours.append(info(q,v))
  if nm(w[0])not in ['i','this'] and not neighbours:continue
  event.update(kind='colored',record=info(p,w),neighbours=neighbours)
 elif eip==0xa8f54:
  st=d.words(d.reg('esp'),4);rows=[]
  for idx in members(st[2]):
   p,w=rec(idx)
   if w:rows.append(dict(idx=idx,name=nm(w[0])))
  event.update(kind='split',records=rows)
 events.append(event);Path(obj+'.color.json').write_text(json.dumps(events,indent=2))
 if 'record'in event:print(k,event['kind'],event['record']['name'],event['record']['idx'],event['record']['allowed'],event['record']['register'],flush=True)
print('DONE',flush=True)
