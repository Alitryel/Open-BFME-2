md=capstone.Cs(capstone.CS_ARCH_X86,capstone.CS_MODE_32)
def norm(b,n):
    out=[]
    for i in md.disasm(bytes(b),0):
        if len(out)>=n: break
        o=i.op_str
        if i.mnemonic.startswith('j') or i.mnemonic=='call': o=''
        out.append(i.mnemonic+' '+o)
    return out
