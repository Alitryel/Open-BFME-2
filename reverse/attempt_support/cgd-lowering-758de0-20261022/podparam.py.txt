from itertools import product
from pathlib import Path
W=Path(__file__).resolve().parent;base=(W/'base.cpp').read_text()
out=W/'podparam';out.mkdir(exist_ok=True)
for index,(cc,group,site,decl,ret,scope) in enumerate(product(range(8),range(2),range(8),range(2),range(2),range(2))):
 s=base
 helpercc='__fastcall ' if scope else ''
 if group:
  defs='struct PArg { const ContactVector *p; const ContactVector *n; };\n'
  body='PArg a={&p,&n}; return a;' if decl else 'PArg a; a.p=&p; a.n=&n; return a;'
  defs+='inline PArg '+helpercc+'pack(const ContactVector &p,const ContactVector &n) { '+body+' }\n'
  args='PArg args';call='pack(p,n)'
  direct=lambda p,n:'pack('+p+','+n+')'
 else:
  defs='struct PArg { const ContactVector *p; };\n'
  body='PArg a={&p}; return a;' if decl else 'PArg a; a.p=&p; return a;'
  defs+='inline PArg '+helpercc+'pack(const ContactVector &p) { '+body+' }\n'
  args='PArg point,PArg normal';call='pack(p),pack(n)'
  direct=lambda p,n:'pack('+p+'),pack('+n+')'
 if ret:
  defs+='struct IArg { void *p; };\n'
  outerty='IArg';outerarg='IArg a={v}; '
 else:
  outerty='void *';outerarg='void *a=v; '
 s=s.replace('struct ContactSource {',defs+'struct ContactSource {')
 s=s.replace('virtual void *prepare(const ContactVector &point, const ContactVector &normal)=0;',
  'virtual void *'+('__fastcall ' if cc&1 else '')+'prepareRaw('+args+')=0;\n void *prepare(const ContactVector &p,const ContactVector &n) { return prepareRaw('+call+'); }')
 s=s.replace('virtual bool test(void *info)=0;',
  'virtual bool '+('__fastcall ' if cc&2 else '')+'testRaw('+outerty+' info)=0;\n bool test(void *v) { '+outerarg+'return testRaw(a); }')
 s=s.replace('virtual void respond(void *info)=0;',
  'virtual void '+('__fastcall ' if cc&4 else '')+'respondRaw('+outerty+' info)=0;\n void respond(void *v) { '+outerarg+'respondRaw(a); }')
 calls=[('node->second->source','node->point','node->normal'),('node->second->source','node->point','node->normal'),('node->first->source','node->point','ContactVector(node->normal).negate()')]
 pos=0
 for i,(recv,p,n) in enumerate(calls):
  old=recv+'->prepare('+p+','+n+')';at=s.find(old,pos)
  new=recv+'->prepareRaw('+direct(p,n)+')' if site&(1<<i) else old
  s=s[:at]+new+s[at+len(old):];pos=at+len(new)
 (out/('v%04d.cpp'%index)).write_text(s)
print('generated',index+1)
