import sys
sys.path.insert(0,'tools')
import build, capstone
src=build.ROOT/sys.argv[1]
obj=build.obj_path(src)
data=obj.read_bytes()
md=capstone.Cs(capstone.CS_ARCH_X86, capstone.CS_MODE_32)
syms=build.read_object_symbols(data)
n=build.u16(data,2)
strtab_off=build.u32(data,8)+18*build.u32(data,12)
for i in range(n):
    o=20+i*40
    name=data[o:o+8].rstrip(b'\0').decode()
    if name.startswith('/'):
        k=int(name[1:]); e=data.index(b'\0',strtab_off+k); name=data[strtab_off+k:e].decode()
    if name!='.text$x': continue
    size=build.u32(data,o+16); ptr=build.u32(data,o+20); rp=build.u32(data,o+24); rc=build.u16(data,o+32)
    rel={}
    for r in range(rc):
        ro=rp+r*10; rel[build.u32(data,ro)]=syms[build.u32(data,ro+4)]['name']
    for ins in md.disasm(data[ptr:ptr+size],0):
        rr=[v for k,v in rel.items() if ins.address<=k<ins.address+ins.size]
        print(f'  {ins.address:04x} {ins.mnemonic} {ins.op_str} {rr if rr else ""}')
