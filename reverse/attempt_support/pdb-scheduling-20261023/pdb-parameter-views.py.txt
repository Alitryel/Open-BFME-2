from pathlib import Path
import sys,re,json
sys.path.insert(0,'tools')
from variant_search import Scorer
from zh_sweep import carve
base=Path('build/pdb-context-input.cpp.txt').read_text()
start=base.index('int _RTC_GetSrcLine(');brace=base.index('{',start)
head,body=base[:brace+1],base[brace+1:]
module='char **moduleOutput=static_cast<char **volatile &>(moduleName);'
address='unsigned long instructionAddress=static_cast<volatile unsigned long &>(address);'
groups=[('module-only',{'moduleName':'moduleOutput'},module),
 ('both-module-first',{'moduleName':'moduleOutput','address':'instructionAddress'},module+address),
 ('both-address-first-control',{'moduleName':'moduleOutput','address':'instructionAddress'},address+module),
 ('address-only-control',{'address':'instructionAddress'},address)]
s=Scorer('?_RTC_GetSrcLine@@YAHKPADHPAHPAPAD@Z',0x75b177,847,Path('build/pdb-parameter-views').resolve())
records=[]
for name,mapping,captures in groups:
 for placement in ['after-alloca','late-error-binding']:
  changed=re.sub(r'\b('+'|'.join(mapping)+r')\b',lambda m:mapping[m[0]],body)
  old='char *errname=(char *)_alloca(1024);'
  replacement=old+captures if placement=='after-alloca' else 'char *allocation=(char *)_alloca(1024);'+captures+'char *errname=allocation;'
  if old not in changed:raise RuntimeError('allocation statement missing')
  src=head+changed.replace(old,replacement,1)
  r=s(src);obj=s.dir/('c%d.obj'%s.n)
  extent=next((len(b) for n,b,_ in carve(obj.read_bytes()) if n==s.symbol),None) if obj.exists() else None
  diffs=[i for i,(a,b) in enumerate(zip(r.get('compiled',b''),r.get('target',b''))) if a!=b]
  item=dict(trial=name,placement=placement,n=s.n,extent=extent,differences=diffs,**{k:v for k,v in r.items() if k not in ('compiled','target')})
  records.append(item);(s.dir/'results.json').write_text(json.dumps(records,indent=2)+'\n')
  print({k:v for k,v in item.items() if k!='differences'},'diff_count',len(diffs),'first',diffs[:18],flush=True)
  if r['exact'] and extent==847:
   (s.dir/'exact.cpp').write_text(src);raise SystemExit(0)
