"""Check the rotation ID conditions on a pools trace: cond.py label...
A (shaker X): tmp1 ends a named pool (tmp2 jumps), tmp2 < 2048, tmps above s/c.
B (final Rotate_Y): s,c < 2048 and tmp1,tmp2 >= 2048 (wrapped keys rank below trig)."""
import json, sys
from pathlib import Path
H = Path(__file__).resolve().parent


def groups(label):
    R = json.load(open(H / 'pools' / f'{label}.json'))
    seq = [r for r in R if r['kind'] in (4, 5)]
    out = []
    for n, r in enumerate(seq):
        if r['name'] == '_theta':
            g = {x['name'].lstrip('_'): x['id'] for x in seq[n - 1:n + 6] if x['name'] in ('_this', '_theta', '_s', '_c', '_tmp1', '_tmp2')}
            out.append(g)
    return out


for label in sys.argv[1:]:
    g = groups(label)
    if len(g) < 4:
        print(label, 'groups', len(g)); continue
    x, f = g[0], g[-1]
    A = x['tmp2'] - x['tmp1'] > 1 and x['tmp2'] < 2048 and min(x['tmp1'], x['tmp2']) > max(x['s'], x['c'])
    B = max(f['s'], f['c']) < 2048 and min(f['tmp1'], f['tmp2']) >= 2048
    print(f"{label}: A={'ok' if A else 'no'} B={'ok' if B else 'no'}  X={x}  final={f}")
