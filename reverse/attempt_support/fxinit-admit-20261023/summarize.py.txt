import json
from pathlib import Path
D=Path('build/fxinit-admit23');out={}
for label in ['latepack','setuppaths','remapabi','valuebundle','valuebundlesafe','volatileparam','recombineview','recombineuv']:
 r=json.loads((D/(label+'-results.json')).read_text());r=r['results'] if isinstance(r,dict) and 'results'in r else r
 if not isinstance(r,list):print(label,type(r));continue
 valid=[x for x in r if 'error'not in x and not x.get('unresolved')]
 ranked=[]
 for z in valid:
  j=json.loads((D/'trials'/(z['id']+'.json')).read_text());n=0
  for d in j['diff']:
   if 'region'in d:
    for ins in d.get('retail',[[]])[0]:n+=0x707<=ins[0]<0x723
   elif 0x707<=d['retail'][0]<0x723:n+=1
  ranked.append(dict(z,remap_lines=n))
 ranked.sort(key=lambda x:(x['remap_lines'],x['cost'],x['nd']))
 out[label]={'generated':len(r),'compiled':sum('error'not in x for x in r),'resolved':len(valid),'bank_score':sum(x.get('nd')==9 for x in r),'best':sorted(valid,key=lambda x:(x['nd'],x['cost']))[:2],'regional':ranked[:3]}
 print(label,out[label]['generated'],out[label]['compiled'],out[label]['resolved'],'best',out[label]['best'][0]['nd'],'region',[(x['id'],x['remap_lines'],x['nd'],x['cost'])for x in ranked[:3]])
(D/'regional-results.json').write_text(json.dumps(out,indent=2))
