import sys, os, re, json, subprocess, concurrent.futures as cf, random
HERE=os.path.dirname(os.path.abspath(__file__))
src=open(os.path.join(HERE,'f0.cpp')).read()
def _row(out, x, y, m0, m1, t, form, tag):
    P0 = f"{x} * m.{m0}"; P1 = f"{y} * m.{m1}"
    if form == 0: return f"{out} = {P0} + {P1} + m.{t};"
    if form == 1: return f"float {tag}0 = {P0};\n\tfloat {tag}1 = {P1};\n\t{out} = {tag}0 + {tag}1 + m.{t};"
    if form == 2: return f"float {tag}1 = {P1};\n\tfloat {tag}0 = {P0};\n\t{out} = {tag}0 + {tag}1 + m.{t};"
    if form == 3: return f"float {tag}0 = {P0};\n\t{out} = {tag}0 + {P1} + m.{t};"
    if form == 4: return f"float {tag}1 = {P1};\n\t{out} = {P0} + {tag}1 + m.{t};"
    if form == 5: return f"float {tag}s = {P0} + {P1};\n\t{out} = {tag}s + m.{t};"
    if form == 6: return f"{out} = m.{m0} * {x} + m.{m1} * {y} + m.{t};"
PT = {0: ("", "in.X", "in.Y"), 1: ("const float *q = &in.X;\n\t", "q[0]", "q[1]"), 2: ("float x = in.X, y = in.Y;\n\t", "x", "y")}
H = {}
for pa in range(3):
    for fx in range(7):
        for fy in range(7):
            pre, x, y = PT[pa]
            H[pa*49+fx*7+fy] = pre + _row("out.X", x, y, "a", "c", "tx", fx, "X") + "\n\t" + _row("out.Y", x, y, "b", "d", "ty", fy, "Y")
            H[147+pa*49+fx*7+fy] = pre + _row("float rx", x, y, "a", "c", "tx", fx, "X") + "\n\t" + _row("float ry", x, y, "b", "d", "ty", fy, "Y") + "\n\tout.X = rx;\n\tout.Y = ry;"
helpers = "".join(f"static __forceinline void aptMM{k}(V2 &out, const V2 &in, const AptRenderTransform &transform)\n{{\n\tconst AptMatrix &m = (const AptMatrix &)transform;\n\t{b}\n}}\n" for k, b in H.items())
i=src.index("void AptShapeContainer::render()")
base=(src[:i]+helpers+src[i:]).split('\n')
start=next(i for i,l in enumerate(base) if l.startswith('void AptShapeContainer::render()'))
end=next(i for i in range(start+2,len(base)) if base[i]=='}')
anch=[i for i in range(start+2,end) if not base[i].strip().startswith('switch') and (base[i].rstrip().endswith(';') or base[i].rstrip().endswith('{'))]
sites=[i for i in range(start,end) if 'aptMatrixMul(' in base[i]]
N=len(sites)
def make(st):
    cfg,pads=st; out=[]; n=0
    for i,l in enumerate(base):
        if i in sites: l=l.replace('aptMatrixMul(',f'aptMM{cfg[sites.index(i)]}(')
        out.append(l)
        k=pads.get(i,0)
        if k:
            ind='\t'*(len(l)-len(l.lstrip('\t'))+(1 if l.rstrip().endswith('{') else 0))
            for _ in range(k): out.append(f'{ind}int zpad{n} = 0;'); n+=1
    return '\n'.join(out)
def score(st,tag):
    p=os.path.join(HERE,'cw',f'{tag}.cpp'); open(p,'w').write(make(st))
    r=subprocess.run([sys.executable,os.path.join(HERE,'..','sc.py'),p],stdout=subprocess.PIPE,stderr=subprocess.DEVNULL,stdin=subprocess.DEVNULL,text=True,timeout=120)
    try: os.remove(p)
    except OSError: pass
    m=re.search(r'size=(\d+) nd=(\d+)',r.stdout)
    if not m: return 9999
    return int(m.group(2))+(0 if m.group(1)=='1471' else 1000)
def key(st): return json.dumps([st[0],sorted(st[1].items())])
if __name__=='__main__':
    os.makedirs(os.path.join(HERE,'cw'),exist_ok=True)
    st=(json.loads(sys.argv[1]), {int(k):v for k,v in json.loads(sys.argv[2]).items()})
    best=score(st,'base'); print('sites',N,'anchors',len(anch),'start',best,flush=True)
    ctr=[0]
    def run(cands):
        with cf.ThreadPoolExecutor(6) as ex:
            return list(ex.map(lambda t:(score(t[1],f'c{t[0]}'),t[1]),enumerate(cands)))
    improved=True
    while improved:
        improved=False
        # pad sweep
        cands=[]
        for a in anch:
            for k in range(8):
                if st[1].get(a,0)==k: continue
                p=dict(st[1]); p[a]=k
                if k==0: del p[a]
                cands.append((list(st[0]),p))
        res=min(run(cands),key=lambda r:r[0])
        if res[0]<best: best,st=res; improved=True
        print('pads',best,json.dumps(st[0]),json.dumps(st[1]),flush=True)
        for si in range(N):
            cands=[]
            for v in H:
                if v==st[0][si]: continue
                c=list(st[0]); c[si]=v; cands.append((c,dict(st[1])))
            res=min(run(cands),key=lambda r:r[0])
            if res[0]<best: best,st=res; improved=True
            print('site',si,best,json.dumps(st[0]),json.dumps(st[1]),flush=True)
        open(os.path.join(HERE,'combo_best.cpp'),'w').write(make(st))
    print('FINAL',best,json.dumps(st[0]),json.dumps(st[1]))
