"""Private parent/cleanup controls; run from the repository root after landing."""
from pathlib import Path
import concurrent.futures,json,re,runpy,sys
sys.path.insert(0,'tools')
import build,eh_verify
out=Path('build/canmake-joins-20261023/fold-controls').resolve()
out.mkdir(parents=True,exist_ok=True)
r=runpy.run_path('reverse/attempt_support/canmake-00596e15-20261022/replay.py.txt')
r['run'].__globals__['OUT']=out
home=Path('Code/GameEngine/Source/GameLogic/SkirmishAI/AIWallBuilder').resolve()
s=(home/'AIBuildableWallSegment.cpp').read_text()
s=re.sub(r'#include "([^"]+)"',lambda m:'#include "'+(home/m[1]).resolve().as_posix()+'"',s)
a=s.index('    // Same 26-byte body')
b=s.index('    int m_unk00;',a)
opaque=s[:a]+'    ~Rva0039205C();\n'+s[b:]
a=s.index('// Preserve the native unwind wrapper')
b=s.index('class BuildAssistant\n',a)
direct=(s[:a]+s[b:]).replace('WallQuoteResult result;','Rva0039205C result;')
tests=[('final',s),('opaque-anchor',opaque),('direct-cleanup',direct)]
with concurrent.futures.ThreadPoolExecutor(max_workers=8)as pool:
    results=list(pool.map(r['run'],tests))
for d in results:
    assert d.get('exact_candidate'),d
    o=eh_verify.Obj(out/(d['key']+'.obj'))
    sym=o.by_name[r['SYMBOL']]
    d['EH']=eh_verify.verify_row(eh_verify.Image(),o,r['SYMBOL'],0x596e15)
    d['cleanup_calls']=[o.symbols[i]['name']for sec in o.eh_sections(sym['section'])
                        for off,i,typ in sec['relocs']if typ==20]
    for name in ['??1WallQuoteResult@@QAE@XZ','??1Rva0039205C@@QAE@XZ']:
        if name not in o.by_name:continue
        raw,relocs=build.read_object_symbol_bytes(out/(d['key']+'.obj'),name,code_only=True)
        d[name]={'size':len(raw.rstrip(b'\xcc')),'relocations':relocs}
    assert d['EH']==('EXACT',''),d
    print(json.dumps(d),flush=True)
(out/'results.json').write_text(json.dumps(results,indent=2)+'\n')
