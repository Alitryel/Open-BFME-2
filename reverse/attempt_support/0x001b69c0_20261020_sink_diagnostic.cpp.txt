// cl: /O2 /G6 /DNDEBUG /MD
// DIAGNOSTIC ONLY (not landable): the global store makes the byte -dist[0] unconditional in the pixel loop.
#include <stdlib.h>
#include <math.h>

unsigned char g_sink;

extern "C" void __cdecl PlaneAddNoise_C(unsigned char *start, unsigned width, unsigned height, int pitch, int q)
{
	unsigned i;
	unsigned j;
	double sigma;
	double x;
	int next;
	signed char dist[300];
	unsigned char noise[2048];
	signed char first;

	sigma = 1 + 0.012698412698412698 * (63 - q);
	next = 0;
	for (x = -32; x < 32; x++) {
		int count = (int)(256 * (exp(-((x - 0) * (x - 0)) / (sigma * sigma + sigma * sigma)) / (sigma * sqrt(6.2831853))) + 0.5);
		if (count) {
			int k;
			for (k = 0; k < count; k++)
				dist[next + k] = (signed char)(int)x;
			next += k;
		}
	}
	for (; next < 256; next++)
		dist[next] = 0;
	for (i = 0; i < 2048; i++)
		noise[i] = dist[rand() & 0xff];
	first = dist[0];
	for (j = 0; j < height; j++) {
		unsigned char *noisePtr = &noise[rand() & 0xff];
		unsigned char *row = start + j * pitch;
		for (i = 0; i < width; i++) {
			int upper = 255 - first;
			g_sink = (unsigned char)-first;
			if (row[i] < -first)
				row[i] = -first;
			if (row[i] > upper)
				row[i] = 255 - first;
			row[i] += noisePtr[i];
		}
	}
}
