from trials import *
import itertools
names=['fuel','resistance','burnRate','maxBurn','decay']
fmt='unibuffer.format(W("Fire logic: fuel=%d, resistance=%d, burn rate=%d, max=%d, decay=%d"), fuel, resistance, burnRate, maxBurn, decay);'
char='template <> void StringBase<char>::releaseBuffer() throw();\n'
items=[]
family=sys.argv[1]
if family=='conventions':
 for p,perm in enumerate(itertools.permutations(names)):
  for cc,byref,release in itertools.product(['__fastcall','__stdcall'],range(2),range(2)):
   s=BASE if release else BASE.replace(char,'')
   params=', '.join(('const int &' if byref else 'int ')+n for n in perm)
   body=fmt.replace('unibuffer.','text->')
   helper='static __forceinline void '+cc+' formatFire(UnicodeString *text, '+params+') { '+body+' }\n'
   s=s.replace('void W3DDisplay::gatherDebugStats(void)',helper+'void W3DDisplay::gatherDebugStats(void)')
   s=s.replace(fmt,'formatFire(&unibuffer, '+', '.join(perm)+');')
   items.append(('cc%03d%s%d%d'%(p,cc[2:],byref,release),s))
if family=='values':
 defs=[
  ('static __forceinline int fireValue(int x) { return x; }','fireValue(%s)'),
  ('static __forceinline int fireValue(const int &x) { return x; }','fireValue(%s)'),
  ('static __forceinline int fireValue(const int *x) { return *x; }','fireValue(&%s)'),
  ('static __forceinline int __fastcall fireValue(int x) { return x; }','fireValue(%s)'),
  ('struct FireValue { const int &v; __forceinline FireValue(const int &x):v(x){} __forceinline operator int() const {return v;} };','int(FireValue(%s))'),
  ('struct FireValue { int v; __forceinline FireValue(int x):v(x){} __forceinline operator int() const {return v;} };','int(FireValue(%s))'),
  ('static __forceinline const int &fireValue(const int &x) { return x; }','fireValue(%s)'),
  ('static __forceinline int fireValue(const int &x) throw() { return x; }','fireValue(%s)')]
 for kind,(helper,expr) in enumerate(defs):
  for mask,release in itertools.product(range(1,32),range(2)):
   s=BASE if release else BASE.replace(char,'')
   s=s.replace('void W3DDisplay::gatherDebugStats(void)',helper+'\nvoid W3DDisplay::gatherDebugStats(void)')
   ff=fmt.split('), fuel,')[0]+'), '+', '.join(expr%n if mask>>i&1 else n for i,n in enumerate(names))+');'
   s=s.replace(fmt,ff)
   items.append(('value%d%02d%d'%(kind,mask,release),s))
batch(items,'fire-'+family)
