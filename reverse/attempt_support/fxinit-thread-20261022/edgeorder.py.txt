import sys,itertools,json,time
from pathlib import Path
from concurrent.futures import ThreadPoolExecutor,as_completed
sys.path.insert(0,'build/fxinit-seat22');import search as h
h.HERE=Path('build/fxinit-thread22').resolve()
PASS='*(unsigned *)(vertices+desc.Get_Basis_End())=pass;'
ONE='*(int *)(vertices+desc.Get_Basis_End())=remap[links[0][i]];'
def variants():
 for order,headform,guard,which,dispatch in itertools.product(itertools.permutations(range(4)),range(3),range(4),range(3),range(4)):
  labels=['PassStore','OneBone','TwoBone','SkipSkin']
  # New interaction: N-before-P skipped guards, so P agrees with the first
  # copy test, crossed with source order and equivalent signed-positive tests.
  cond=['!normals[1]&&!positions[1]','(!normals[1])&(!positions[1])','normals[1]==positions[1]&&!normals[1]','!(*(const Vector3 *volatile *)&normals[1])&&!positions[1]'][guard]
  bodies=[PASS+'goto FinishSkin;',ONE+'goto FinishSkin;',h.R+'\n'+h.W+'\nCopyBoth:\n'+h.P+'\n'+h.N+'\ngoto FinishSkin;','if('+cond+') goto CopyBoth;goto FinishSkin;']
  yes=['m_BonesPerVertex>0 && links[0]','links[0] && m_BonesPerVertex>0','m_BonesPerVertex>=1 && links[0]','((unsigned)m_BonesPerVertex-1)<0x7fffffffu && links[0]'][dispatch]
  sk1='SkipSkin' if which in [0,2]else'FinishSkin';sk2='SkipSkin' if which in [1,2]else'FinishSkin'
  if headform==0:head=f'if(m_Passes>1) goto PassStore;if(!({yes})) goto {sk1};if(m_BonesPerVertex==1) goto OneBone;if(m_BonesPerVertex==2) goto TwoBone;goto {sk2};'
  if headform==1:head=f'if(m_Passes>1) goto PassStore;else if({yes}) {{if(m_BonesPerVertex==1) goto OneBone;else if(m_BonesPerVertex==2) goto TwoBone;else goto {sk2};}}else goto {sk1};'
  if headform==2:head=f'if(m_Passes<=1) {{if({yes}) {{if(m_BonesPerVertex!=1) {{if(m_BonesPerVertex!=2)goto {sk2};goto TwoBone;}}goto OneBone;}}goto {sk1};}}goto PassStore;'
  arm=head+'\n'
  for k,j in enumerate(order):
   body=bodies[j];nxt=labels[order[k+1]] if k<3 else'FinishSkin'
   if body.endswith('goto '+nxt+';'):body=body[:-len('goto '+nxt+';')]
   arm+=labels[j]+':\n'+body+'\n'
  arm+='FinishSkin:;\n'
  yield 'edgeorder_'+''.join(map(str,order))+f'_{headform}{guard}{which}{dispatch}',h.BASE[:h.START]+arm+h.BASE[h.END:]
t=time.time();items=list(variants());r=[]
with ThreadPoolExecutor(max_workers=8)as pool:
 for f in as_completed([pool.submit(h.run_one,x)for x in items]):
  r.append(f.result())
  if len(r)%768==0:print('tested',len(r),'best',min((x for x in r if 'nd'in x),key=lambda x:(x['nd'],x['cost'])),flush=True)
r.sort(key=lambda x:(x.get('nd',99999),x.get('cost',99999)))
(h.HERE/'edgeorder-results.json').write_text(json.dumps(r,indent=2))
print(json.dumps({'count':len(r),'valid':sum('nd'in x and not x.get('unresolved')for x in r),'seconds':time.time()-t,'best':r[:4],'best_shape':sorted((x for x in r if 'cost'in x),key=lambda x:x['cost'])[:4]},indent=2))
