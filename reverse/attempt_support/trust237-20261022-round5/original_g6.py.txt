import itertools, json, concurrent.futures
import search

# These deliberately status-dependent programs are diagnostic controls only.
# They cannot be banked as semantically faithful replacements of retail.
tails=[
    'return status==0 || status==K;',
    'int r=0; if(status==0) r=1; else if(status==K) r=1; return r;',
    'int r=1; if(status!=0) {if(status!=K) r=0;} return r;',
    'int r=0; if(status==0) goto yes; if(status==K) goto yes; return r; yes:return r+1;',
    'int r=0; if(status!=0 && status!=K) goto done; ++r; done:return r;',
    'if(status==0) return 1; if(status==K) return 1; return 0;',
    'switch(status){case 0:case K:return 1;default:return 0;}',
    'int r=0; while(status!=0){if(status==K)break;return r;}return r+1;',
]
jobs=[]
for tail,cpu,opt,kind,barrier,scope in itertools.product(range(len(tails)),[5,6],['/O1','/O2'],['int','bool'],range(3),range(2)):
    code=tails[tail].replace('K','0x800b0100L').replace('int r=',kind+' r=')
    if scope: code='{'+code+'}'
    stmt=['','_WriteBarrier();','_ReadWriteBarrier();'][barrier]
    text=search.base.replace(search.marker,'long status=WinVerifyTrust(0,&action,&data);LocalFree(argv);'+stmt+code)
    if barrier:
        name=['','_WriteBarrier','_ReadWriteBarrier'][barrier]
        text=text.replace('int Rva00237D3DCheck() {','extern "C" void '+name+'();\n#pragma intrinsic('+name+')\nint Rva00237D3DCheck() {')
    text=text.replace('/O1 /G7',opt+f' /G{cpu}')
    jobs.append((f'o{len(jobs):04}',text,dict(tail=tail,cpu=cpu,opt=opt,kind=kind,barrier=barrier,scope=scope,diagnostic_only=True)))
results=search.execute('original_g6',jobs)
good=[r for r in results if 'error' not in r]
# Compute the diagnostic score outside the fixed retail NOP interval.
for r in good:
    raw,rel=search.build.read_object_symbol_bytes(search.OUT/(r['name']+'.obj'),search.NAME,code_only=True)
    norm,un=search.resolve_bytes(raw,rel,search.TARGET+b'\0'*max(0,len(raw)-search.SIZE),search.RVA,len(raw),{'_memset':[0x6291ae],'_WinVerifyTrust@12':[0x655652]})
    r['outside_nop_differences']=[i for i,(a,b) in enumerate(itertools.zip_longest(norm,search.TARGET)) if a!=b and not 0xcd<=i<0xdd]
(search.OUT/'original_g6.json').write_text(json.dumps(results,indent=2))
print('BEST OUTSIDE NOP:')
for r in sorted(good,key=lambda r:len(r['outside_nop_differences']))[:5]: print(r)
