import sys,itertools,json,time
from pathlib import Path
from concurrent.futures import ThreadPoolExecutor
sys.path.insert(0,'build/fxinit-seat22');import search as h
h.HERE=Path('build/fxinit-split22').resolve()
LOOP='   for(unsigned i=0;i<(unsigned)mesh->Get_Vertex_Count();++i) {'
T='if(tangents) *(Vector3 *)(vertices+desc.Get_Basis_Offset())=tangents[i];'
B='if(binormals) *(Vector3 *)(vertices+desc.Get_Basis_Second_Offset())=binormals[i];'
S=[('positions[1][i]','const Vector3','(unsigned)positions[1]+i*12','*(const Vector3 *)p0'),('normals[1][i]','const Vector3','(unsigned)normals[1]+i*12','*(const Vector3 *)p1'),('links[0][i]','const unsigned short','(unsigned)links[0]+i*2','*(const unsigned short *)p2'),('links[1][i]','const unsigned short','(unsigned)links[1]+i*2','*(const unsigned short *)p3'),('weights[0][i]','const unsigned short','(unsigned)weights[0]+i*2','*(const unsigned short *)p4'),('weights[1][i]','const unsigned short','(unsigned)weights[1]+i*2','*(const unsigned short *)p5'),('vertices+desc.Get_Location_Offset(1)','char','(unsigned)vertices+desc.Get_Location_Offset(1)','(char *)p6'),('vertices+desc.Get_Normal_Offset(1)','char','(unsigned)vertices+desc.Get_Normal_Offset(1)','(char *)p7')]
def variants():
 for mask,branch,where,typ,rev in itertools.product(range(1,256),range(2),range(2),range(2),range(2)):
  src=h.BASE;decl=[];assign=[];start=src.index(h.R);tail=src[start:]
  for k,(old,ty,expr,new) in enumerate(S):
   if mask>>k&1:
    decl.append(('unsigned ' if not typ else ty+' *')+f'p{k};')
    assign.append(f'p{k}='+('' if not typ else '('+ty+' *)')+'('+expr+');')
    tail=tail.replace(old,new)
  src=src[:start]+tail
  if rev:assign.reverse()
  ds=''.join(decl);a=''.join(assign)
  src=src.replace(LOOP,LOOP+ds)
  if branch==0:
   old='    if(basis) {\n     '+T+'\n     '+B+'\n    }'
   assert old in src
   core=T+B
   src=src.replace(old,'if(basis) {'+(a+core if where==0 else core+a)+'}else{'+a+'}')
  else:
   old='if(m_FVF&0x10) *(Vector3 *)(vertices+desc.Get_Normal_Offset(0))=normals[0][i];'
   core=old[old.index(') ')+2:]
   src=src.replace(old,'if(m_FVF&0x10) {'+(a+core if where==0 else core+a)+'}else{'+a+'}')
  yield f'branchprep_{mask:03}_{branch}{where}{typ}{rev}',src
if __name__=='__main__':
 t=time.time()
 with ThreadPoolExecutor(max_workers=8) as pool:r=list(pool.map(h.run_one,variants()))
 r.sort(key=lambda x:(x.get('nd',99999),x.get('cost',99999)))
 (h.HERE/'branchprep-results.json').write_text(json.dumps(r,indent=2))
 print(json.dumps({'count':len(r),'valid':sum('nd'in x and not x.get('unresolved')for x in r),'seconds':time.time()-t,'best':r[:8]},indent=2))
