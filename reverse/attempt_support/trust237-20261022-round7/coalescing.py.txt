from pathlib import Path
import sys,itertools
ROOT=Path.cwd();sys.path.insert(0,str(ROOT/'build/trust237-round5'))
import search
search.OUT=ROOT/'build/trust237-round7'
variants=[
 ('assign','u=u;',''),
 ('ref_assign','unsigned long &v=u;v=u;',''),
 ('local_swap','unsigned long v=u;unsigned long t=u;u=v;v=t;',''),
 ('xor_swap','unsigned long v=u;u^=v;v^=u;u^=v;',''),
 ('swap_ref','sameSwap(u,u);','__forceinline void sameSwap(unsigned long &a,unsigned long &b){unsigned long t=a;a=b;b=t;}\n'),
 ('copy_return','u=copyReturn(u);','__forceinline unsigned long copyReturn(unsigned long a){return a;}\n'),
 ('union_home','SameWord w;w.a=u;u=w.b;','union SameWord{unsigned long a,b;};\n'),
 ('bitcast_home','SameWord w;w.a=u;u=(unsigned long)w.b;','union SameWord{unsigned long a;long b;};\n'),
 ('member_copy','Word w(u);w=w;u=w.a;','struct Word{unsigned long a;__forceinline Word(unsigned long v):a(v){} __forceinline Word &operator=(const Word &w){a=w.a;return *this;}};\n'),
 ('member_swap','Word w(u);w.swap(w);u=w.a;','struct Word{unsigned long a;__forceinline Word(unsigned long v):a(v){} __forceinline void swap(Word &w){unsigned long t=a;a=w.a;w.a=t;}};\n'),
]
tails=[
 'return 1;',
 'int r=0;if((u&1u)<=1u)goto yes;return r;yes:return r+1;',
 'int r=0;if((unsigned char)u<=255u)++r;return r;',
]
jobs=[]
for (label,stmt,decl),count,pos,shape,cpu,opt in itertools.product(variants,[1,16],range(2),range(3),[6,7],['O1','O2']):
    # Each repeated operation lives in its own scope, permitting identical declarations.
    body=('{'+stmt+'}')*count
    code='long status=WinVerifyTrust(0,&action,&data);unsigned long u=(unsigned long)status;'
    code+= (body+'LocalFree(argv);' if pos==0 else 'LocalFree(argv);'+body)+tails[shape]
    source=search.base.replace(search.marker,code)
    source=source.replace('int Rva00237D3DCheck() {',decl+'int Rva00237D3DCheck() {').replace('/O1 /G7',f'/{opt} /G{cpu}')
    jobs.append((f'c{len(jobs):04}',source,dict(kind=label,count=count,position=pos,shape=shape,cpu=cpu,opt=opt)))
search.execute('coalescing',jobs)
