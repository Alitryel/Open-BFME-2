from search import *

def source(n):
 mask=n%32;kind=(n//32)%4;ret=n//128
 index=['int','const int &','unsigned int','const unsigned int &'][kind]
 rtyp='T &'
 expr=['elements[i]','*(elements+i)','*(T*)((char*)elements+i*sizeof(T))','*(&elements[0]+i)'][ret]
 arr='''
template<class T, int N> struct CostArray {
 T elements[N];
 __forceinline %s operator[](%s i) { return elements[i]; }
 __forceinline const T &operator[](%s i) const {return elements[i];}
};
'''%(rtyp,index,index)
 h=HEAD
 for k,(typ,name) in enumerate([('MpGameSetupComboRef','m_colorCombo'),('GameWindow *','m_handicap'),('GameWindow *','m_hero'),('GameWindow *','m_playerTemplate'),('GameWindow *','m_team')]):
  if not(mask&(1<<k)):continue
  use='CostArray<%s,8>'%typ
  if False:
   # Keep the wrapper's nontrivial copy boundary unchanged.
   h=h.replace(typ+' '+name+'[8]',typ+' '+name+'[8]')
   continue
  h=h.replace(typ+' '+name+'[8]',use+' '+name)
  # Most pointer fields put the star beside the name.
  if typ=='GameWindow *':h=h.replace('GameWindow *'+name+'[8]',use+' '+name)
 arr=arr.replace('return elements[i];','return '+expr+';')
 return arr+h+MARK+BODY
if __name__=='__main__':run([('array-%03d'%n,source(n)) for n in range(512)],'arrays')
