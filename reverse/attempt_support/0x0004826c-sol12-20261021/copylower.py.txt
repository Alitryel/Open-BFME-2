from regions import *
# Late lowering of a real 4-byte object-representation copy. No extra read,
# padding or observable operation; pointers are trivially copyable on this ABI.
def generate():
 for bits in itertools.product(range(2),repeat=9):
  dy,dn,oy,on,lookup,selection,guard,test,contract=bits
  init=('Drawable *found = '+find+'; Drawable *p; memcpy(&p,&found,sizeof p);' if lookup else 'Drawable *p = '+find+';')
  yesdraw='memcpy(&draw,&p,sizeof draw);' if dy else 'draw=p;'
  nodraw='memcpy(&draw,&p,sizeof draw);' if dn else 'draw=p;'
  yesobj='memcpy(&obj,&p->m_object,sizeof obj);' if oy else 'obj=p->m_object;'
  noobj='memcpy(&obj,&p,sizeof obj);' if on else 'obj=(Object*)p;'
  form='if('+('(unsigned)p>0' if test else 'p')+') {'+yesdraw+yesobj+'} else {'+nodraw+noobj+'} if('+('p && obj' if guard else 'obj')+') {'+body+'\n} else {'+terrain+'}'
  src=BASE[:lo]+'Drawable *draw; '+init+' Object *obj;\n'+form+'\n'+BASE[hi:]
  if not contract:src=src.replace('template <> void StringBase<char>::releaseBuffer() throw();\n','')
  if selection:src=src.replace('draw = TheInGameUI->getFirstSelectedDrawable();','{Drawable *selected=TheInGameUI->getFirstSelectedDrawable(); memcpy(&draw,&selected,sizeof draw);}')
  src=src.replace('#define W(s)', 'extern "C" void *__cdecl memcpy(void *, const void *, unsigned int);\n#pragma intrinsic(memcpy)\n#define W(s)')
  yield ('cp'+''.join(map(str,bits)),src)
if __name__=='__main__':batch(list(generate()),'copylower')
