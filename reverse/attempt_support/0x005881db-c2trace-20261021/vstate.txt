import sys,subprocess,capstone,os
from pathlib import Path
from concurrent.futures import ThreadPoolExecutor
sys.path.insert(0,'/work/Open-BFME-2/tools')
import build
# vstate.py DIR : compile every DIR/*.cpp (8 parallel) and report loop-1 node state: NODE (ebx), NEITHER (eax reload), other
SYM='?updateMeleeAttack@HordeMeleeAmoeba@@UAEXPAVObject@@@Z'
md=capstone.Cs(capstone.CS_ARCH_X86,capstone.CS_MODE_32)
def one(f):
    obj=Path(str(f)+'.obj')
    if obj.exists(): obj.unlink()
    cmd,env=build.compiler_command(f.resolve(),obj.resolve())
    subprocess.run(cmd,cwd=build.ROOT,env=env,capture_output=True)
    if not obj.exists(): return '%s ERR'%f.name
    data,rel=build.read_object_symbol_bytes(obj,SYM)
    ins=[(i.address,i.mnemonic,i.op_str) for i in md.disasm(bytes(data),0)]
    st='?'
    for k,(a,mn,o) in enumerate(ins):
        if a>0x60 and mn=='mov' and o.startswith('edi, dword ptr [') and o.endswith('+ 8]'):
            r=o.split('[')[1].split()[0]
            prev=ins[k-1]
            st={'ebx':'NODE','eax':'NEITHER'}.get(r,r)
            if r=='eax' and 'ebp' not in prev[2]: st='eax?'
            st+=' '+('/'.join('%s %s'%(x[1],x[2]) for x in ins[k-1:k]))
            break
    return '%s %d %s'%(f.name,len(data),st)
fs=sorted(Path(sys.argv[1]).glob('*.cpp'))
with ThreadPoolExecutor(int(os.environ.get('J','8'))) as ex:
    for r in ex.map(one,fs): print(r,flush=True)
