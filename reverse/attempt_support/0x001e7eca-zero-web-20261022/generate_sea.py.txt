from search import *
s=(W/'base.cpp').read_text(); a=s.index('  case 1:'); b=s.index('  case 4:',a)
blk=s[a:b]
start=blk.index('   float water;'); end=blk.index('   setTranslation(pos);')
body=blk[start:end]
vs=[]
for typ in range(4):
 for level in range(4):
  for kind in range(4):
   bd=body
   use=['p->','p.','p->','p.'][typ]
   bd=bd.replace('pos.',use)
   if level&1: bd=bd.replace('object->definition->flags11f&8','object->seaSpeedScale()')
   if level&2: bd=bd.replace('float speed=speed40;', 'float speed=seaSpeed();')
   rt='float' if kind&1 else 'void'
   bd+=('return '+use+'z;\n' if kind&1 else '')
   par=['Rva001E41D3Result *p','Rva001E41D3Result &p','Coord3D *p','Coord3D &p'][typ]
   helper='__forceinline '+rt+' seaBody(Object*object,'+par+'){\n'+bd+'\n}\n'
   out=s.replace(' float getMaxSpeed(Object*)const;', ' float getMaxSpeed(Object*)const;\n float seaSpeed()const{return speed40;}\n'+helper)
   out=out.replace(' bool isKindOf(KindOfType)const;', ' bool isKindOf(KindOfType)const;\n bool seaSpeedScale()const{return definition->flags11f&8;}')
   pararg=['&pos','pos','(Coord3D*)&pos','*(Coord3D*)&pos'][typ]
   call=('pos.z=' if kind&1 else '')+'seaBody(object,'+pararg+');'
   repl=blk[:start]+'   '+call+'\n'+blk[end:]
   if kind&2:repl=repl.replace('   requires=false;\n','').replace('   break;','   requires=false;\n   break;').replace('if(object->disabled&8)break;','if(object->disabled&8){requires=false;break;}')
   out=out.replace(blk,repl)
   vs.append((f'sea_{typ}_{level}_{kind}',out))
run(vs,'sea64')
