import sys, pathlib, itertools, json, collections, os
sys.path.insert(0, 'build/4404ac-mergeweb')
import search as s
s.OUT=pathlib.Path('build/4404ac-affinity').resolve()
BASE=s.OUT.joinpath('control.cpp').read_text()
H,B=BASE.split(s.MARK)
A=B.index('if (slot == 0)')
Z=B.index('\n\tif (slot == game->v13())', A)
REG=B[A:Z]
beg=REG.index('\tif (*team)')
mid=REG.index('} else {')
BLOCK=REG[beg:mid].rstrip()

def variant(k):
    t=BLOCK
    # Different IL graphs for the same flag and call; paired independently.
    if k&1:
        t=t.replace('bool locked = m_rules.m_04 == 1;', 'bool unlocked = m_rules.m_04 != 1;').replace('!locked','unlocked')
    if k&2:
        t=t.replace('bool teamEnable;', 'bool teamEnable = false;').replace('teamEnable = false;\n\t\t\tGadget', 'Gadget')
    if k&4:
        t=t.replace('bool locked =', 'int locked =').replace('bool unlocked =','int unlocked =')
    if k&8:
        flag='unlocked' if k&1 else '!locked'
        t=t.replace('enable && !observer && '+flag, '(enable ? (!observer ? '+flag+' : false) : false)')
    if k&16:
        t='GameWindow *window=*team;\n'+t.replace('if (*team)', 'if(window)').replace('GadgetComboBoxHideList(*team);','GadgetComboBoxHideList(window);')
    return t

def gen_asym():
    for i,j in itertools.product(range(32),repeat=2):
        yield ('asym-%04d'%(i*32+j), H+s.MARK+B[:A]+'if(slot==0){'+variant(i)+'}else{'+variant(j)+'}'+B[Z:])

def gen_graph():
    for i,(n,pred,form,mask) in enumerate(itertools.product([3,4,5,8,16],range(4),range(3),range(4))):
        blocks=[variant((k%2)*mask) for k in range(n)]
        if form==0:
            reg=''.join(('if' if k==0 else 'else if')+'(slot==%d){%s}'%(k,b) for k,b in enumerate(blocks[:-1]))+'else{'+blocks[-1]+'}'
        elif form==1:
            expr=['slot','slot&7','slot%8','(int)enable+2*(int)observer'][pred]
            reg='switch('+expr+'){'+''.join('case %d:{%s}break;'%(k,b) for k,b in enumerate(blocks[:-1]))+'default:{'+blocks[-1]+'}break;}'
        else:
            expr=['slot<4','slot&1','enable','observer'][pred]
            def tree(bs):
                if len(bs)==1:return bs[0]
                m=len(bs)//2
                return 'if('+expr+'){'+tree(bs[:m])+'}else{'+tree(bs[m:])+'}'
            reg=tree(blocks)
        yield ('graph-%04d'%i,H+s.MARK+B[:A]+reg+B[Z:])

def gen_helpers():
    for n,(kind,ret,callshape,ptrshape) in enumerate(itertools.product(range(8),range(4),range(4),range(2))):
        typ=['bool','char','unsigned short','int'][ret]
        args='GameWindow **ptr, const int &rule, '+typ+' &locked'
        if kind==0:
            code='static __forceinline GameWindow *PrepareTeam('+args+'){GameWindow *w=*ptr; if(w)locked=rule==1;return w;}'
            prep=typ+' locked; GameWindow *window=PrepareTeam(team,m_rules.m_04,locked);'
        elif kind==1:
            code='static __forceinline bool PrepareTeam('+args+', GameWindow *&w){w=*ptr; if(w)locked=rule==1;return w!=0;}'
            prep=typ+' locked; GameWindow *window; bool present=PrepareTeam(team,m_rules.m_04,locked,window);'
        elif kind==2:
            code='static __forceinline void PrepareTeam('+args+', GameWindow *&w){w=*ptr; if(w)locked=rule==1;}'
            prep=typ+' locked; GameWindow *window; PrepareTeam(team,m_rules.m_04,locked,window);'
        elif kind==3:
            code='struct TeamState {'+typ+' locked; GameWindow *window;}; static __forceinline TeamState PrepareTeam(GameWindow **p,const int &rule){TeamState v;v.window=*p;if(v.window)v.locked=rule==1;return v;}'
            prep='TeamState state=PrepareTeam(team,m_rules.m_04); '+typ+' locked=state.locked; GameWindow *window=state.window;'
        elif kind==4:
            code='struct TeamState {GameWindow *window; '+typ+' locked;}; static __forceinline TeamState PrepareTeam(GameWindow **p,const int &rule){TeamState v;v.window=*p;if(v.window)v.locked=rule==1;return v;}'
            prep='TeamState state=PrepareTeam(team,m_rules.m_04); '+typ+' locked=state.locked; GameWindow *window=state.window;'
        elif kind==5:
            code='static __forceinline GameWindow *PrepareTeam(GameWindow *w,const int &rule,'+typ+' &locked){if(w)locked=rule==1;return w;}'
            prep=typ+' locked; GameWindow *window=PrepareTeam(*team,m_rules.m_04,locked);'
        elif kind==6:
            code='static __forceinline GameWindow *PrepareTeam('+args+'){GameWindow *w=*ptr; if(!w)return 0;locked=rule==1;return w;}'
            prep=typ+' locked; GameWindow *window=PrepareTeam(team,m_rules.m_04,locked);'
        else:
            code='struct TeamState {GameWindow *window; '+typ+' locked; __forceinline TeamState(GameWindow **p,const int &rule):window(*p){if(window)locked=rule==1;} };'
            prep='TeamState state(team,m_rules.m_04); '+typ+' locked=state.locked; GameWindow *window=state.window;'
        # Avoid an indeterminate flag read in the null case of aggregate results.
        if kind in(3,4,7):
            prep=prep.replace(typ+' locked=state.locked;',typ+' locked; if(state.window)locked=state.locked;')
        block=BLOCK.replace('bool locked = m_rules.m_04 == 1;','').replace('if (*team)','if(present)'if kind==1 else 'if(window)')
        block=block.replace('GadgetComboBoxHideList(*team);','GadgetComboBoxHideList(window);')
        if ptrshape:
            # Existing post-call reload, with an explicit same-value definition on
            # the other path, retaining possible HideList mutation.
            block=block.replace('GadgetComboBoxHideList(window);','GadgetComboBoxHideList(window); window=*team;').replace('(*team)->winEnable(teamEnable);','window->winEnable(teamEnable);')
        if callshape==0:reg='if(slot==0){'+prep+block+'}else{'+prep+block+'}'
        elif callshape==1:reg=prep+'if(slot==0){'+block+'}else{'+block+'}'
        elif callshape==2:reg='if(slot==0)goto prepA; {'+prep+block+'}goto prepEnd;prepA:;{'+prep+block+'}prepEnd:;'
        else:reg=prep+block
        yield ('helpers-%04d'%n,H+code+'\n'+s.MARK+B[:A]+reg+B[Z:])

def gen_localgraph():
    predicates=['window==hero','window==combo.m_window','window!=0','(void*)window==(void*)this','(unsigned)window&4','slot==0','enable','observer']
    for n,(pred,shape,form,side) in enumerate(itertools.product(range(8),range(7),range(3),range(3))):
        cond=predicates[pred]
        def dup(t):
            if form==0:return 'if('+cond+'){'+t+'}else{'+t+'}'
            if form==1:return 'switch(bool('+cond+')){case true:{'+t+'}break;default:{'+t+'}break;}'
            return 'if(!('+cond+')){'+t+'}else{'+t+'}'
        lock='bool locked=m_rules.m_04==1;'
        calc='if(enable && !observer && !locked)teamEnable=true;else{teamEnable=false;GadgetComboBoxHideList(window);}'
        tail='(*team)->winEnable(teamEnable);'
        core=lock+'bool teamEnable;'+calc+tail
        if shape==0:core=dup(core)
        elif shape==1:core='bool teamEnable;'+dup(lock+calc)+tail
        elif shape==2:core=lock+'bool teamEnable;'+dup(calc)+tail
        elif shape==3:core=lock+'bool teamEnable;'+calc+dup(tail)
        elif shape==4:core='bool locked;'+dup('locked=m_rules.m_04==1;')+'bool teamEnable;'+calc+tail
        elif shape==5:core=lock+'bool teamEnable;if(enable && !observer && !locked)teamEnable=true;else{teamEnable=false;'+dup('GadgetComboBoxHideList(window);')+'}'+tail
        else:core=lock+'bool teamEnable;if(enable && !observer && !locked)teamEnable=true;else{'+dup('teamEnable=false;GadgetComboBoxHideList(window);')+'}'+tail
        modified='GameWindow *window=*team; if(window){'+core+'}'
        plain=variant(16)
        left=plain if side==2 else modified
        right=plain if side==1 else modified
        reg='if(slot==0){'+left+'}else{'+right+'}'
        yield ('localgraph-%04d'%n,H+s.MARK+B[:A]+reg+B[Z:])

if __name__=='__main__':
    s.run(list({'asym':gen_asym,'graph':gen_graph,'helpers':gen_helpers,'localgraph':gen_localgraph}[sys.argv[1]]()), sys.argv[1])
