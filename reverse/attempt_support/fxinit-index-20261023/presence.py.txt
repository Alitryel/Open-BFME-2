import sys,json,itertools,time
from pathlib import Path
from concurrent.futures import ThreadPoolExecutor,as_completed
sys.path.insert(0,'build/fxinit-seat22');import search as h
h.HERE=Path('build/fxinit-index22').resolve()
P=h.P.split(') ',1)[1];N=h.N.split(') ',1)[1]
def variants():
 # Combined presence state trades two independently live pointer predicates
 # for one encoded value. Complemented bits always map to the same copies.
 for scope,ty,combine,shape,invert,order in itertools.product(range(4),range(3),range(3),range(3),range(4),range(2)):
  p='(positions[1]'+('==0)'if invert&1 else'!=0)')
  n='(normals[1]'+('==0)'if invert&2 else'!=0)')
  terms=[f'((unsigned){p}|((unsigned){n}<<1))',f'((unsigned){p}+2*(unsigned){n})',f'({p}?({n}?3:1):({n}?2:0))']
  typ=['unsigned','unsigned char','int'][ty]
  init=f'{typ} presence={terms[combine]};\n'
  # Invert maps encoded bits back to P and N present, including case zero.
  cases=[]
  for value in range(4):
   bits=value^invert;cases.append((value,(P if bits&1 else'')+(N if bits&2 else'')))
  if order:cases.reverse()
  if shape==0:tail='switch(presence){'+''.join(f'case {v}:{{{body}}}break;'for v,body in cases)+'}'
  elif shape==1:tail=''.join(('if'if k==0 else'else if')+f'(presence=={v}){{{body}}}'for k,(v,body)in enumerate(cases))
  else:
   ps=f'((presence&1)'+('==0)'if invert&1 else'!=0)');ns=f'((presence&2)'+('==0)'if invert&2 else'!=0)')
   tail=f'if({ps}){{{P}}}if({ns}){{{N}}}'
  s=h.BASE
  a=s.index(h.P);b=s.index(h.N,a)+len(h.N);s=s[:a]+tail+s[b:]
  if scope==0:s=s.replace('  for(unsigned pass=0;',init+'  for(unsigned pass=0;')
  elif scope==1:s=s.replace('    *(Vector3 *)(vertices+desc.Get_Location_Offset(0))',init+'    *(Vector3 *)(vertices+desc.Get_Location_Offset(0))')
  elif scope==2:s=s.replace('    if(m_Passes>1) *(unsigned *)',init+'    if(m_Passes>1) *(unsigned *)')
  else:s=s.replace('     else if(m_BonesPerVertex==2) {','     else if(m_BonesPerVertex==2) {\n'+init)
  yield f'presence_{scope}{ty}{combine}{shape}{invert}{order}',s
if __name__=='__main__':
 items=list(variants());r=[];t=time.time()
 with ThreadPoolExecutor(max_workers=8)as pool:
  for f in as_completed([pool.submit(h.run_one,x)for x in items]):
   r.append(f.result())
   if len(r)%256==0:print('tested',len(r),'best',min((x for x in r if 'nd'in x),key=lambda x:(x['nd'],x['cost'])),flush=True)
 r.sort(key=lambda x:(x.get('nd',99999),x.get('cost',99999)))
 (h.HERE/'presence-results.json').write_text(json.dumps(r,indent=2))
 print(json.dumps({'count':len(r),'seconds':time.time()-t,'errors':sum('error'in x for x in r),'best':r[:4]},indent=2))
