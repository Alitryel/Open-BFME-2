from search import *
s=(W/'base.cpp').read_text()
a=s.index('  case 1:'); b=s.index('  case 4:',a)
original=s[a:b]
water='((LocoTerrainView*)TheTerrainLogic)->underwater(pos.x,pos.y,&water,0,0)'
layer='((LocoTerrainView*)TheTerrainLogic)->layer(pos.x,pos.y,object->rva0028B511(),0,true)'
tail=original[original.index('   if(object->definition'):original.rindex('   break;')]
# New factorial: duplicate shared case tail and requires stores, with independently
# scoped inline semantic wrappers. Repeated assignments can bound constant webs
# before the compiler's common-tail merge (recipes.md).
vs=[]
for mask in range(128):
 dup=bool(mask&1); req=bool(mask&2); helper=bool(mask&4); after=bool(mask&8); wlocal=bool(mask&16); sign=bool(mask&32); outer=bool(mask&64)
 head='  case 1:\n  {\n'
 if not req and not after:head+='   requires=false;\n'
 if wlocal:head+='   float water;\n'
 if not outer:
  head+='   if(object->disabled&8){'+('requires=false;' if req or after else '')+'break;}\n'
 else:head+='   if(!(object->disabled&8)){\n'
 if not wlocal:head+='   float water;\n'
 wt=water; lt=layer; prefix=''
 if helper:
  prefix='__forceinline bool seaQuery(Coord3D *p,float *w){return ((LocoTerrainView*)TheTerrainLogic)->underwater(p->x,p->y,w,0,0);}\n__forceinline float seaLayer(Coord3D *p,Object *o){return ((LocoTerrainView*)TheTerrainLogic)->layer(p->x,p->y,o->rva0028B511(),0,true);}\n'
  wt='seaQuery((Coord3D*)&pos,&water)'; lt='seaLayer((Coord3D*)&pos,object)'
 t=tail+('   requires=false;\n' if req else '')
 c1='pos.z=water;\n'+(t if dup else '')
 c2='pos.z='+lt+';\n'+(t if dup else '')
 if not sign:head+='   if('+wt+'){'+c1+'}else{'+c2+'}\n'
 else:head+='   if(!'+wt+'){'+c2+'}else{'+c1+'}\n'
 if not dup:head+=t
 if outer:head+='   }\n'
 if after:head+='   requires=false;\n'
 head+='   break;\n  }\n'
 out=s[:a]+head+s[b:]
 if prefix:out=out.replace('bool Locomotor::rva001E7ECA',prefix+'bool Locomotor::rva001E7ECA')
 vs.append((f'cfg_{mask:03d}',out))
run(vs,'cfg128')
