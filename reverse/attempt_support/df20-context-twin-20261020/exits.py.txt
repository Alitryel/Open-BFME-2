import trials,itertools,concurrent.futures,json
specs=[]
first='''  if (__digits.size() == 0) 
#pragma inline_depth(0)
    return __s;
#pragma inline_depth(255)'''
second='''  if (__cp == __digits_first)
#pragma inline_depth(1)
    return __s;
#pragma inline_depth(255)'''
assert first in trials.bank and second in trials.bank
for firstgoto,secondgoto,finalinside,localeinline in itertools.product([0,1],repeat=4):
 if not firstgoto and not secondgoto:continue
 s=trials.bank
 start=s.index('#pragma inline_depth(0)\n  string_type __curr_sym')
 s=s[:start]+'  {\n'+s[start:]
 if firstgoto:s=s.replace(first,first.replace('return __s','goto money_exit'))
 if secondgoto:s=s.replace(second,second.replace('return __s','goto money_exit'))
 pos=s.rfind('  return __s;\n}')
 s=s[:pos]+s[pos:].replace('  return __s;\n}', ('  return __s;\n' if finalinside else '')+'  }\nmoney_exit:\n  return __s;\n}',1)
 if localeinline:s=s.replace('locale::~locale()','inline locale::~locale()')
 specs.append((f'exit-f{firstgoto}-s{secondgoto}-i{finalinside}-l{localeinline}',s))
with concurrent.futures.ThreadPoolExecutor(max_workers=8) as p:r=list(p.map(trials.run,specs))
(trials.scratch/'exits.json').write_text(json.dumps(r,indent=2)+'\n')
