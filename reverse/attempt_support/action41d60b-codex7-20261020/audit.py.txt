import sys,json,difflib
sys.path.insert(0,'build/action41d60b-codex8');import trial
import build,variant_search
name='providers0_15'
p=trial.D/(name+'.cpp');o=p.with_suffix('.obj')
row={'name':trial.NAME,'target_rva':'0x0041D60B','target_size':'3709','source':str(p.relative_to(build.ROOT)),'notes':''}
got=build.compile_function(row,build.load_symbol_map(),o,allow_weak=False)
b,t=got['bytes'],got['target']
C=list(trial.md.disasm(b[:3521],0x41d60b));R=list(trial.md.disasm(t[:3521],0x41d60b))
calls=lambda seq:[(hex(i.address),i.op_str) for i in seq if i.mnemonic=='call' and i.op_str.startswith('0x')]
cc,rc=calls(C),calls(R)
out={'fitness':variant_search.non_exact_fitness(b,t),'exact':b==t and not got['unresolved'],'strict_unresolved':got['unresolved'],'masked':got['masked'],'retail_direct_calls':len(rc),'compiled_direct_calls':len(cc),'ordered_callees_equal':[v for a,v in cc]==[v for a,v in rc],'call_differences':[[n,rc[n],cc[n]] for n in range(min(len(rc),len(cc))) if rc[n][1]!=cc[n][1]],'object_symbol':trial.score(o),'relocation_counts':{str(k):sum(r[1]==k for r in got['relocs']) for k in sorted(set(r[1] for r in got['relocs']))}}
(trial.D/'final-audit.json').write_text(json.dumps(out,indent=2))
print(json.dumps({k:v for k,v in out.items() if k!='object_symbol'},indent=2))
