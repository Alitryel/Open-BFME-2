import sys,itertools,json,time
from pathlib import Path
from concurrent.futures import ThreadPoolExecutor,as_completed
sys.path.insert(0,'build/fxinit-seat22');import search as h
h.HERE=Path('build/fxinit-thread22').resolve()
def variants():
 for mode,bits in itertools.product(range(2),range(512)):
  s=h.BASE
  changes=[
   ('if(tangents)','if(*(const Vector3 *volatile *)&tangents)'),
   ('if(binormals)','if(*(const Vector3 *volatile *)&binormals)'),
   ('=tangents[i];','=(*(const Vector3 *volatile *)&tangents)[i];'),
   ('=binormals[i];','=(*(const Vector3 *volatile *)&binormals)[i];'),
   ('=positions[0][i];','=(*(const Vector3 *volatile *)&positions[0])[i];'),
   ('=normals[0][i];','=(*(const Vector3 *volatile *)&normals[0])[i];'),
   ('if(basis) {','if(*(volatile bool *)&basis) {'),
   ('if(positions[1])','if(*(const Vector3 *volatile *)&positions[1])'),
   ('if(normals[1])','if(*(const Vector3 *volatile *)&normals[1])')]
  # Mode one qualifies the whole T/B pointer objects instead of just their
  # guards; the source/read choices still distinguish all applied masks.
  if mode:
   changes[0]=('const Vector3 *tangents=', 'const Vector3 *volatile tangents=')
   changes[1]=('const Vector3 *binormals=', 'const Vector3 *volatile binormals=')
  for j,(a,b)in enumerate(changes):
   if bits>>j&1:assert a in s;s=s.replace(a,b)
  # Exclude old P/N/basis-only guard combinations: new T/B/first-stream
  # source access is the independent evidence-backed variable this session.
  if not bits&63:continue
  yield f'qual_{mode}_{bits:03x}',s
if __name__=='__main__':
 items=list(variants());r=[];t=time.time()
 with ThreadPoolExecutor(max_workers=8)as pool:
  for f in as_completed([pool.submit(h.run_one,x)for x in items]):r.append(f.result())
 r.sort(key=lambda x:(x.get('nd',99999),x.get('cost',99999)))
 (h.HERE/'qual-results.json').write_text(json.dumps(r,indent=2))
 print(json.dumps({'count':len(r),'resolved':sum('nd'in x and not x.get('unresolved')for x in r),'seconds':time.time()-t,'best':r[:6]},indent=2))
