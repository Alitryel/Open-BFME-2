import sys,subprocess,json,hashlib
from pathlib import Path
sys.path.insert(0,'tools');import build
from capstone import Cs,CS_ARCH_X86,CS_MODE_32
base=Path('build/vp6-abi-probe').resolve();results=[]
for attr in ('preserve_most','preserve_all','no_caller_saved_registers'):
 p=base/(attr+'.cpp');o=base/(attr+'.obj')
 p.write_text('// compiler: clang23-x86\nextern "C" __attribute__(('+attr+')) void probe(short *input, short *output) { unsigned i=0; do { output[i] = short(input[i]+input[(i+1)&7]); } while(++i != 8); }\n')
 cmd,env=build.compiler_command(p,o)
 cmd+=['-Werror=ignored-attributes']
 r=subprocess.run(cmd,env=env,cwd=build.ROOT,stdout=subprocess.PIPE,stderr=subprocess.STDOUT,text=True)
 entry={'attribute':attr,'command':cmd,'returncode':r.returncode,'output':r.stdout,'source_sha256':hashlib.sha256(p.read_bytes()).hexdigest(),'compiler_inputs':json.loads(env.get('BFME_COMPILER_INPUTS','{}'))}
 if r.returncode==0:
  raw,rel,detail=build.read_object_symbol_bytes(o,'_probe',10000,code_only=True,detail=True)
  entry['disassembly']=[hex(i.address)+' '+i.mnemonic+' '+i.op_str for i in Cs(CS_ARCH_X86,CS_MODE_32).disasm(raw,0)]
 results.append(entry)
proof={'head':subprocess.check_output(['git','rev-parse','HEAD'],text=True).strip(),'diagnostic_only':True,'landing_verified':False,'trials':results,'native':[]}
for rva,size in ((0x1d35e0,2374),(0x1d3f30,1724),(0x1d4650,1777),(0x1d4d50,1034),(0x1d5160,777)):
 raw=build.read_target_bytes(rva,size);proof['native'].append({'rva':hex(rva),'size':size,'sha256':hashlib.sha256(raw).hexdigest(),'entry':raw[:22].hex(),'tail':raw[-4:].hex()})
(base/'results.json').write_text(json.dumps(proof,indent=2))
for r in results: print(r['attribute'],r['returncode'],r['output'],r.get('disassembly'))
