import json,sys
from pathlib import Path
D=json.loads(Path(sys.argv[1]).read_text())
for si,s in enumerate(D):
 if s['stage']!='0x107b4e20':continue
 nodes=s['nodes'];syms=s['syms'];graph=s['graph'];edges=s['edges']
 if sum((nodes.get(str(w[7]),[0,0,0])[2]&255)==15 for w in graph.values())<2:continue
 def name(p):
  sw=syms.get(str(p),[])
  if len(sw)>7:
   if 0x108a9804<=p<0x108a9b04:return 'R'+str((p-0x108a97a4)//96)
   return 'H'+str(sw[7])
  return hex(p)
 def op(p,seen=()):
  if p in seen:return '<cycle>'
  w=nodes.get(str(p),[])
  if len(w)<16:return hex(p)
  k=w[2]&255;seen=seen+(p,)
  if k in [1,2,4]:return name(w[7] or w[6])
  if k in [5,6]:return ('&' if k==5 else '*')+'('+op(w[13],seen)+'+'+hex(w[12])+')'
  if k==7:return '#'+hex(w[6])
  return 'k%d'%k
 print('\nWINDOW',si//3, 'priority checkpoint',s['stage'])
 for p,w in graph.items():
  il=nodes.get(str(w[7]),[]);k=il[2]&255 if il else -1
  text=op(il[10])+' -> '+op(il[11]) if k in [13,18] else ('CALL' if k==15 else 'k'+str(k))
  e=w[3];ss=[]
  while e:
   ew=edges.get(str(e))
   if not ew:break
   tw=graph.get(str(ew[3]),[]);ss.append((tw[13]>>16 if len(tw)>13 else hex(ew[3]),ew[5]));e=ew[0]
  print('n%02d pri%02d il%08x op%04x %-26s -> %s'%(w[13]>>16,w[13]&65535,w[7],il[1] if il else 0,text,ss))
 for a in [D[si-1],D[si+1]]:
  ids={w[7]:w[13]>>16 for w in a['graph'].values() if w[7]};ns=a['nodes']
  prev={ns[str(p)][0] for p in ids};starts=[p for p in ids if p not in prev]
  for p in starts:
   seq=[]
   while p in ids and len(seq)<100:
    seq.append(ids[p]);p=ns[str(p)][0]
   print(a['stage'],'IL order',seq)
