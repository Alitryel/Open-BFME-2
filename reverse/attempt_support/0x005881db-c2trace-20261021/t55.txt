import sys,os
sys.path.insert(0,'/work/Open-BFME-2/build/c2tr')
from wdbg import Dbg
src,obj=sys.argv[1:3]; KMAX=int(os.environ.get('KMAX','3'))
d=Dbg(src,obj,obj+'.log')
d.prompt();d.prompt('set $CanDeferOnBPByAddr = 1')
B=0x10700000
bps=[0x83acd,0x82978]; added=set()
for b in bps:
    o=d.prompt('break *0x%x'%(B+b))
    if 'Breakpoint' in o and 'Unable' not in o: added.add(b)
names={}
def nm(sym):
    if sym in names: return names[sym]
    sw=d.words(sym,1); pw=d.words(sw[0],2) if sw and sw[0]>0x100000 else []
    n=d.string(pw[1]).strip() if len(pw)>1 and pw[1]>0x100000 else '?'
    n=n.strip('"').lstrip('_'); names[sym]=n; return n
def rec_of(i):
    p=d.words(0x108807b8+(i&0x3ff)*4,1)[0]
    while p:
        w=d.words(p,13)
        if len(w)<13: return None,None
        if w[7]==i: return p,w
        p=w[12]
    return None,None
def members(hdr):
    m=[]; ch=d.words(hdr,1)[0] if hdr else 0; n=0
    while ch and n<300:
        ww=d.words(ch,3)
        if len(ww)<3: break
        m+=[ww[0]+j for j in range(32) if ww[2]>>j&1]; ch=ww[1]; n+=1
    return m
def lab(i):
    p,w=rec_of(i); return '%s:%x/w3=%d'%(nm(w[0]),i,w[3]) if w else '%x'%i
on=False;k=0
while True:
    out=d.prompt('cont')
    if 'terminated' in out: break
    eip=d.reg('eip')-B
    for b in bps:
        if b not in added and b!=eip: d.prompt('break *0x%x'%(B+b)); added.add(b)
    added.add(eip)
    if eip==0x83acd:
        ebp=d.reg('ebp'); f=d.words(ebp,1)[0]; curfn=d.string(d.words(f+4,1)[0]); on='updateMeleeAttack' in curfn and d.reg('ebx')==0; k=0; continue
    if not on: continue
    if eip==0x82978:
        k+=1
        if k>KMAX: break
        esp=d.reg('esp'); a=d.words(esp,6); ra=a[0]-B
        print('R%d ra=%x args=%s'%(k,ra,[hex(x) for x in a[1:]]),flush=True)
        for j in range(1,5):
            if a[j]>0x100000:
                try:
                    m=members(a[j])
                    if m and len(m)<200: print('  arg%d %s'%(j,' '.join(lab(i) for i in m)),flush=True)
                except Exception as e: pass
        continue
print('done',flush=True)
