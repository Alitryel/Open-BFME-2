from pathlib import Path
from itertools import product
W=Path(__file__).resolve().parent;out=W/'statetype';out.mkdir(exist_ok=True)
banks=[(W/'control.cpp').read_text(),(W/'base.cpp').read_text(),(W/'pair.cpp').read_text()]
for idx,(bank,kind,assign,post,scope,guard) in enumerate(product(range(3),range(8),range(4),range(4),range(2),range(2))):
 s=banks[bank];s=s.replace('struct ContactNode {','enum StateCode { NoState=0,YesState=1,NotState=2 };\nstruct ContactNode {')
 typ=['unsigned','int','long','unsigned long','StateCode','StateCode','int','unsigned'][kind]
 decl=typ+' state'+(':32' if kind>=5 else '')+';'
 s=s.replace('unsigned key0,key1,state;','unsigned key0,key1; '+decl)
 a=s.index('  if(node->state==0)');z=s.index('  if(node->state==2) continue;',a)
 test=s[a:z].strip().rsplit('node->state=',1)[1].rsplit('?1:2;',1)[0]
 target='result' if scope else 'node->state'
 expr=test if assign<2 else 'hit'
 statement=[''+target+'='+expr+'?YesState:NotState;','if('+expr+')'+target+'=YesState;else '+target+'=NotState;','bool hit='+test+';'+target+'=hit?YesState:NotState;','bool hit='+test+';'+target+'=('+typ+')(2-(hit?1:0));'][assign]
 if scope:statement=typ+' result;'+statement+'node->state=result;'
 filt=['if(node->state==2)continue;','switch(node->state){case NotState:continue;default:break;}','switch(node->state){default:break;case NotState:continue;}','if(node->state!=NotState){}else continue;'][post]
 s=s[:a]+'if(node->state==NoState){'+statement+'}\n'+filt+s[z+len('  if(node->state==2) continue;'):]
 if guard:s=s.replace('if(node->first->source==0 || node->second->source==0) continue;', 'if(node->first->source==0)continue; if(node->second->source==0)continue;')
 (out/('v%04d.cpp'%idx)).write_text(s)
print('generated',idx+1)
