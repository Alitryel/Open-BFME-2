import sys,json,itertools
from pathlib import Path
from concurrent.futures import ThreadPoolExecutor,as_completed
sys.path.insert(0,'build/fxinit-seat22');import search as h
h.HERE=Path('build/fxinit-thread22').resolve()
controls={'bank':h.BASE,'skip':Path('build/fxinit-edges22/trials/skipjoin_020000.cpp').read_text(),'reverse':Path('build/fxinit-thread22/trials/revguard_00000.cpp').read_text(),'shared':Path('build/fxinit-edges22/trials/sharedcopy_00000.cpp').read_text()}
items=[]
for (name,s),flag in itertools.product(controls.items(),['/Os','/Ot']):
 s=s.replace('// cl: /I. /O1','// cl: '+flag+' /I. /O1') if False else s.replace('/MD /DNDEBUG /EHsc','/MD /DNDEBUG /EHsc '+flag,1)
 items.append(('speed_'+name+'_'+flag[2:],s))
r=[]
with ThreadPoolExecutor(max_workers=8)as p:
 for f in as_completed([p.submit(h.run_one,x)for x in items]):r.append(f.result())
(h.HERE/'speed-results.json').write_text(json.dumps(r,indent=2));print(json.dumps(r,indent=2))
