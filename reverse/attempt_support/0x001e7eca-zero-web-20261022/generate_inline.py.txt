from search import *
s=(W/'base.cpp').read_text(); a=s.index('  case 6:'); b=s.index('  case 7:',a)
blk=s[a:b]
# Enclose the complete building query in a forceinlined semantic helper, rather
# than merely wrapping its virtual ground call. This tests the original
# PartitionManager helper boundary suggested by the donor.
start=blk.index('   pos.z='); end=blk.index('   pos.z=height48+highest;')
body=blk[start:end]
vs=[]
for form in range(8):
 for separate in range(4):
  for tail in range(4):
   nm=f'inl_{form}_{separate}_{tail}'
   # original update position is explicitly in/out so the query still sees z.
   par=['Rva001E41D3Result *p','Rva001E41D3Result &p','Coord3D *p','Coord3D &p'][form%4]
   use='p->' if form%4 in (0,2) else 'p.'
   bd=body.replace('pos.',use).replace('(const Coord3D*)&pos','(const Coord3D*)'+('p' if '->' in use else '&p'))
   if separate&1:bd=bd.replace('float highest=0.0f;', 'float highest;').replace('   Object*entry;','   highest=0.0f;\n   Object*entry;')
   if separate&2:bd=bd.replace('   Object*entry;', '   Object*entry=result.next();').replace('while((entry=result.next())!=0)','for(;entry;entry=result.next())')
   if tail&1:bd=bd.replace('if(h>highest)highest=h;', 'highest=h>highest?h:highest;')
   if tail&2:bd=bd.replace('float h=((GeometryInfo*)((char*)entry+0xa8))->getMaxHeightAbovePosition();','float h; h=((GeometryInfo*)((char*)entry+0xa8))->getMaxHeightAbovePosition();')
   ret='return highest;' if form<4 else 'return '+use+'z+highest;'
   helper='__forceinline float buildQuery('+par+'){\n'+bd+'\n'+ret+'\n}\n'
   helper=helper.replace('ThePartitionManager->iterate','ThePartitionManager->iterate')
   call='buildQuery('+(['&pos','pos','(Coord3D*)&pos','*(Coord3D*)&pos'][form%4])+')'
   # forms 4..7: donor uses ground height + highest, but retail bank ignores
   # ground in final placement; subtract ground to preserve observed semantics.
   if form>=4:continue
   nb='  case 6:{ requires=true; float highest='+call+'; pos.z=height48+highest; setTranslation(pos); break; }\n'
   out=s[:a]+nb+s[b:]
   out=out.replace('bool Locomotor::rva001E7ECA',helper+'bool Locomotor::rva001E7ECA')
   vs.append((nm,out))
run(vs,'inline64')
