def make_cpp(out,hname,K=29,P=48,Q=1):
    L=list(SRC)
    i=[n for n,l in enumerate(L) if l.startswith('#define inline __forceinline')][0]
    L[i]=L[i]+'\n#include "build/s126/s44/h_%s/vector3.h"'%hname
    ku=[n for n,l in enumerate(L) if l.startswith('int unused0=0;')][0]
    L[ku]=''.join('int unused%d=0;'%j for j in range(K))
    pi=[n for n,l in enumerate(L) if l.startswith('__forceinline void padf(){')][0]
    L[pi]='__forceinline void padf(){%s}'%''.join('int a%d=0;'%j for j in range(P))
    qi=[n for n,l in enumerate(L) if l.startswith('__forceinline void padQ03(){')][0]
    L[qi]='__forceinline void padQ03(){%s}'%''.join('int q03_%d=0;'%j for j in range(Q))
    open(out,'w').write('\n'.join(L))
