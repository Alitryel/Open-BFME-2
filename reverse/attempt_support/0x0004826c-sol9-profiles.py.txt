import trials
from trials import *
char='template <> void StringBase<char>::releaseBuffer() throw();\n'
all_results=[]
for profile in ['rtm','vc71-ddk-4035','vc71-sp1']:
 s=BASE if profile=='rtm' else '// compiler: '+profile+'\n'+BASE
 src=DIR/('profile-'+profile+'.cpp');src.write_text(s)
 trials.COMMAND,trials.ENV=build.compiler_command(src,DIR/'unused.obj')
 results=batch([('profile-'+profile+str(release),s if release else s.replace(char,'')) for release in range(2)],'profile-'+profile)
 all_results.extend(results)
(DIR/'profiles-results.json').write_text(json.dumps(all_results,indent=2))
