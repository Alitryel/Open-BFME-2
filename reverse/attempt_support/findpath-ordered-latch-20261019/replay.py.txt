from pathlib import Path
import hashlib,json,re,subprocess
root=Path(__file__).resolve().parents[2]; out=Path(__file__).resolve().parent
owner=root/'Code/GameEngine/Source/GameLogic/Pathfinder/pathfinder.cpp'
text=owner.read_text(); bank=(root/'reverse/attempts/0x002fd344.cpp').read_text()
assert not subprocess.check_output(['git','diff','f51bbefda6a8b5ab3437d6028401cf990d39f75d','HEAD','--',owner.relative_to(root).as_posix()],cwd=root)
assert text.count('\tBool isDownhillOnly()const')==1
text=text.replace('\tBool isDownhillOnly()const','\tconst char *name() const { return m_nameBuffer ? m_nameBuffer+8 : ""; }\n\tBool isDownhillOnly()const')

decl='Path *FindPath(Object *,const LocomotorSet &,const Coord3D *,const Coord3D *,Bool *);'

assert text.count(needle)==1;text=text.replace(needle,needle+'\n\t'+decl)
helpers=bank[bank.index('static __forceinline void subCoord3D'):bank.index('// coord3d.cpp')]
handle=bank[bank.index('struct PathfinderSearchHandle'):bank.index('// Retail 2FD344')]
body=bank[bank.index('Path *Pathfinder::FindPath('):]
def absolute_include(match):
    path=(owner.parent/match.group(1)).resolve()
    if path.is_file():return '#include "'+path.as_posix()+'"'
    return match.group(0)
text=re.sub(r'#include\s+"([^"]+)"',absolute_include,text)
base=text+'\n'+helpers+'\n'+handle+'\n'+body
loop='''\t\tfor (Int i = 0; i < steps; i++) {
\t\t\tsubCoord3D(&adjusted, &delta);
\t\t\tif (QuickDoesPathExist(obj, from, &adjusted, 0)) break;
\t\t}'''
assert base.count(loop)==1
variants={'baseline':base}
variants['volatile-loop']=base.replace(loop,loop.replace('subCoord3D(&adjusted, &delta);','adjusted.y -= ((const volatile Coord3D *)&delta)->y;\n\t\t\tadjusted.x -= ((const volatile Coord3D *)&delta)->x;\n\t\t\tadjusted.z -= ((const volatile Coord3D *)&delta)->z;'))
for volatile in (False,True):
    view='((const volatile Coord3D *)&delta)->' if volatile else 'delta.'
    replacement='''\t\tfloat deltaX = delta.x, deltaY = delta.y, deltaZ = delta.z;
\t\tfor (Int i = 0; i < steps; i++) {
\t\t\tadjusted.x -= deltaX;
\t\t\tadjusted.y -= deltaY;
\t\t\tadjusted.z -= deltaZ;
\t\t\tif (QuickDoesPathExist(obj, from, &adjusted, 0)) break;

\t\t\tdeltaZ = VIEWz;

    variants['cached-'+('volatile' if volatile else 'ordinary')]=base.replace(loop,replacement)
records=[]
for name,source in variants.items():

    records.append({'name':name,'source':path.relative_to(root).as_posix(),'sha256':hashlib.sha256(path.read_bytes()).hexdigest()})
