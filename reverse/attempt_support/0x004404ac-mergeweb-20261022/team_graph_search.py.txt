def dup(s,cond,form):
 body=dup_inner(s,cond,form)
 return '{ '+body+' }'

def dup_inner(s,cond,form):
 if form==0:return 'if('+cond+') { '+s+' } else { '+s+' }'
 if form==1:return 'if('+cond+')goto teamA; {'+s+'} goto teamZ; teamA:; {'+s+'} teamZ:;'
 if form==2:return 'switch('+cond+'){case true: {'+s+'} break; default: {'+s+'} break;}'
 return 'if('+cond+'){'+s+' goto teamZ;} '+s+' teamZ:;'
