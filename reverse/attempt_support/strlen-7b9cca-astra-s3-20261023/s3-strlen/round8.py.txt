import sys, json, subprocess, concurrent.futures, hashlib, itertools, time
from pathlib import Path
sys.path.insert(0,"tools")
import build
ROOT=Path.cwd()
OUT=ROOT/"build/s3-strlen"
target=build.read_target_bytes(0x7b9cca,24)
forms=[('plain', 'const char*p=s;char c=*p++;if(c){do{c=*p++;}while(c);}else p=s+1;return(int)(p-s-1);'), ('bank', 'const char*p=s;char c=*p++;if(c){do{c=*p++;}while((unsigned char)c>0);}else p=s+1;return(int)(p-s-1);'), ('guard', 'const char*p=s;char c=*p++;if((unsigned char)c>0){do{c=*p++;}while(c);}else p=s+1;return(int)(p-s-1);'), ('simple', 'const char*p=s;if(*p++){do{}while(*p++);}return(int)(p-s-1);')]
flags=['/Od /Og /Os /Oy /G6', '/Od /Og /Ot /Oy /G6', '/Od /Og /Os /Oy /Ob0 /G6', '/Od /Og /Os /Oy /Ob2 /G6', '/Od /Og /Os /Oy /Gy- /G6', '/Od /Og /Os /Oy /Gf- /G6', '/Od /Og /Oy /G6', '/Od /Og /Os /Oy /Oi /G6', '/Od /Og /Os /Oy /Oi- /G6', '/Od /Og /Os /Oy /Op /G6', '/O1 /Og /Os /Oy /G6', '/O1 /Og /Ot /Oy /G6', '/O1 /Og /Os /Oy /Ob0 /G6', '/O1 /Og /Os /Oy /Ob2 /G6', '/O1 /Og /Os /Oy /Gy- /G6', '/O1 /Og /Os /Oy /Gf- /G6', '/O1 /Og /Oy /G6', '/O1 /Og /Os /Oy /Oi /G6', '/O1 /Og /Os /Oy /Oi- /G6', '/O1 /Og /Os /Oy /Op /G6', '/O2 /Og /Os /Oy /G6', '/O2 /Og /Ot /Oy /G6', '/O2 /Og /Os /Oy /Ob0 /G6', '/O2 /Og /Os /Oy /Ob2 /G6', '/O2 /Og /Os /Oy /Gy- /G6', '/O2 /Og /Os /Oy /Gf- /G6', '/O2 /Og /Oy /G6', '/O2 /Og /Os /Oy /Oi /G6', '/O2 /Og /Os /Oy /Oi- /G6', '/O2 /Og /Os /Oy /Op /G6', '/Od /Og /Os /Oy /G3', '/Od /Og /Os /Oy /G4', '/Od /Og /Os /Oy /G5', '/Od /Og /Os /Oy /G7', '/Od /Og /Os /Oy /GB', '/Od /Og /Os /Oy /Ob0 /G3', '/Od /Og /Os /Oy /Ob0 /G4', '/Od /Og /Os /Oy /Ob0 /G5', '/Od /Og /Os /Oy /Ob0 /G7', '/Od /Og /Os /Oy /Ob0 /GB', '/Od /Og /Os /Oy /Gy- /G3', '/Od /Og /Os /Oy /Gy- /G4', '/Od /Og /Os /Oy /Gy- /G5', '/Od /Og /Os /Oy /Gy- /G7', '/Od /Og /Os /Oy /Gy- /GB']
def trial(arg):
 idx, (tag,body), flag, profile=arg
 ext="cpp"
 src=OUT/("round8_%04d.%s"%(idx,ext)); obj=src.with_suffix(".obj")
 src.write_text(("// compiler: "+profile+"\n" if profile else "")+"// cl: "+flag+" /MD\nint rva007B9CCAStrLen(const char *s) { "+body+" }\n")
 command,env=build.compiler_command(src,obj)
 p=subprocess.run(command,cwd=ROOT,env=env,capture_output=True,text=True,timeout=90)
 rec=dict(index=idx,tag=tag,flags=flag,profile=profile,ext=ext,returncode=p.returncode,diagnostic=(p.stdout+p.stderr).strip())
 if p.returncode==0:
  symbols=build.defined_code_symbols(obj)
  sym=next(s for s in symbols if "rva007B9CCAStrLen" in s)
  buf,relocs=build.read_object_symbol_bytes(obj,sym,code_only=True)
  rec.update(size=len(buf),hex=buf.hex(),relocs=relocs,exact=buf==target,diff=sum(a!=b for a,b in itertools.zip_longest(buf,target)))
 return rec
jobs=[(i,form,flag,ext) for i,(form,flag,ext) in enumerate(itertools.product(forms,flags,[""]))]
build.ensure_reference_current()
start=time.time()
with (OUT/"round8.jsonl").open("w") as fh, concurrent.futures.ThreadPoolExecutor(max_workers=2) as pool:
 for i,rec in enumerate(pool.map(trial,jobs)):
  fh.write(json.dumps(rec)+"\n"); fh.flush()
  if rec.get("exact") or i%32==0: print(json.dumps(rec),flush=True)
print("DONE",len(jobs),"seconds",time.time()-start,flush=True)
