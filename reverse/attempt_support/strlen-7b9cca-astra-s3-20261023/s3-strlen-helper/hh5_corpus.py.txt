# astra-s3 helper seat round 5: guard OUTSIDE the loop region (goto out / early
# exit) + bottom-test latch, so the induction-init lea can stay in the preheader
# before the guard test. f5_v_gtw (diff=5) is the baseline to beat.

FN = 'int rva007B9CCAStrLen(const char *s) { '
RET = ' return (int)(p - s - 1); }'

VARIANTS = []
def add(tag, body, flags='/O1 /G6 /MD', profile='', ext='cpp'):
    VARIANTS.append((tag, flags, profile, ext, body))

VLOAD = 'char c = *(volatile const char *)s; '
PINIT  = 'const char *p = s + 1; '
NLOAD = 'char c = *s; '

# G1: guard "if (c == 0) goto out" + infinite loop with break latch
add('g1_v_brk',   FN + VLOAD + PINIT + 'if (c == 0) goto out; for (;;) { c = *p++; if (!c) break; } out:' + RET)
add('g1_v_brk_o', FN + PINIT + VLOAD + 'if (c == 0) goto out; for (;;) { c = *p++; if (!c) break; } out:' + RET)
add('g1_n_brk',   FN + NLOAD + PINIT + 'if (c == 0) goto out; for (;;) { c = *p++; if (!c) break; } out:' + RET)
add('g1_v_brk2',  FN + VLOAD + PINIT + 'if (!c) goto out; for (;;) { c = *p++; if (!c) break; } out:' + RET)
add('g1_v_w1',    FN + VLOAD + PINIT + 'if (c == 0) goto out; while (1) { c = *p++; if (!c) break; } out:' + RET)
add('g1_v_brkeq', FN + VLOAD + PINIT + 'if (c == 0) goto out; for (;;) { c = *p++; if (c == 0) break; } out:' + RET)
add('g1_v_brk_sb', FN + VLOAD + PINIT + 'if (c == 0) goto out; for (;;) { c = *p; p = p + 1; if (!c) break; } out:' + RET)

# G2: guard "if (c) ... else goto out" shapes around the infinite loop
add('g2_v_els',   FN + VLOAD + PINIT + 'if (c) { for (;;) { c = *p++; if (!c) break; } }' + RET)
add('g2_v_else',  FN + VLOAD + PINIT + 'if (c) { for (;;) { c = *p++; if (!c) break; } } else { }' + RET)
add('g2_v_elsg',  FN + VLOAD + PINIT + 'if (c) { for (;;) { c = *p++; if (!c) break; } } goto out; out:' + RET)

# G3: f5_v_gtw order/polarity/helping spellings (baseline diff=5)
add('g3_o',       FN + PINIT + VLOAD + 'if (c) goto loop; goto out; loop: while (c) { c = *p++; } out:' + RET)
add('g3_ne',      FN + VLOAD + PINIT + 'if (c != 0) goto loop; goto out; loop: while (c) { c = *p++; } out:' + RET)
add('g3_wn',      FN + VLOAD + PINIT + 'if (c) goto loop; goto out; loop: while (c != 0) { c = *p++; } out:' + RET)
add('g3_raw',     FN + VLOAD + PINIT + 'if (c) goto loop; goto out; loop: c = *p++; if (c) goto loop; out:' + RET)
add('g3_raw_o',   FN + PINIT + VLOAD + 'if (c) goto loop; goto out; loop: c = *p++; if (c) goto loop; out:' + RET)
add('g3_rawne',   FN + VLOAD + PINIT + 'if (c == 0) goto out; loop: c = *p++; if (c) goto loop; out:' + RET)
add('g3_rawne2',  FN + VLOAD + PINIT + 'if (!c) goto out; loop: c = *p++; if (c != 0) goto loop; out:' + RET)
add('g3_raw_sb',  FN + VLOAD + PINIT + 'if (c) goto loop; goto out; loop: c = *p; p = p + 1; if (c) goto loop; out:' + RET)

# G4: do-while at the label with je-polarity guard
add('g4_je',      FN + VLOAD + PINIT + 'if (c == 0) goto out; do { c = *p++; } while (c); out:' + RET)
add('g4_je_o',    FN + PINIT + VLOAD + 'if (c == 0) goto out; do { c = *p++; } while (c); out:' + RET)
add('g4_n',       FN + NLOAD + PINIT + 'if (c == 0) goto out; do { c = *p++; } while (c); out:' + RET)

# G5: p initialized in both arms (block-split probe)
add('g5_arms',    FN + VLOAD + 'const char *p; if (c == 0) goto out; p = s + 1; goto loop; out: p = s + 1; goto fin; loop: do { c = *p++; } while (c); fin:' + RET)
add('g5_arms2',   FN + VLOAD + 'const char *p; if (c) goto loop; p = s + 1; goto fin; loop: p = s + 1; do { c = *p++; } while (c); fin:' + RET)

# G6: ternary / comma guard spellings with goto arms
add('g6_ter',     FN + VLOAD + PINIT + 'if (c ? 1 : 0) goto loop; goto out; loop: while (c) { c = *p++; } out:' + RET)
add('g6_comma',   FN + VLOAD + PINIT + 'if ((c, c)) goto loop; goto out; loop: while (c) { c = *p++; } out:' + RET)

# G7: while at the label with split body and goto arms
add('g7_sb',      FN + VLOAD + PINIT + 'if (c) goto loop; goto out; loop: while (c) { c = *p; p = p + 1; } out:' + RET)
add('g7_sb_o',    FN + PINIT + VLOAD + 'if (c) goto loop; goto out; loop: while (c) { c = *p; p = p + 1; } out:' + RET)
