from pathlib import Path
import sys,json
sys.path.insert(0,'tools')
from variant_search import Scorer
from zh_sweep import carve
out=Path('build/local-vm/cookie-c-frontend')
s=Scorer('@__security_check_cookie@4',0x629bab,14,out.resolve())
records=[]
for flags in ['/O2 /TC /GS-','/O1 /TC /GS-','/O2 /TC /G7 /GS-']:
 src='// cl: '+flags+'\n'+'''#ifdef __cplusplus
extern "C" {
#endif
extern unsigned int __security_cookie;
void report_failure(void);
void __fastcall __security_check_cookie(unsigned int cookie) {
 if(cookie==__security_cookie) return;
 report_failure();
}
#ifdef __cplusplus
}
#endif
'''
 r=s(src);obj=out/('c%d.obj'%s.n)
 extent=next((len(b) for n,b,_ in carve(obj.read_bytes()) if n=='@__security_check_cookie@4'),None) if obj.exists() else None
 row=dict(flags=flags,extent=extent,**{k:v for k,v in r.items() if k not in ('compiled','target')})
 row['compiled']=r.get('compiled',b'').hex();records.append(row)
 print(row,flush=True);(out/'results.json').write_text(json.dumps(records,indent=2)+'\n')
 if r['exact'] and extent==14:(out/'exact.cpp').write_text(src);break
