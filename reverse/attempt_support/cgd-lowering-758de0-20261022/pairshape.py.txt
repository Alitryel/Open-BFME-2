from pathlib import Path
from itertools import product
W=Path(__file__).resolve().parent;base=(W/'base.cpp').read_text();out=W/'pairshape';out.mkdir(exist_ok=True)
for idx,(kind,ret,order,body,copy,ptr,local,cv,sites) in enumerate(product(range(4),range(2),range(2),range(2),range(2),range(2),range(2),range(2),range(8))):
 ty=['const ContactVector *','const ContactVector &','unsigned','const void *'][kind]
 aty='const ContactVector *' if ptr else 'const ContactVector &'
 pv=(['p','*p','(unsigned)p','p'] if ptr else ['&p','p','(unsigned)&p','&p'])[kind]
 nv=pv.replace('p','n')
 params=aty+' '+('n, '+aty+' p' if order else 'p, '+aty+' n')
 pair='struct ContactPair { '+ty+' point; '+ty+' normal; ContactPair('+params+')'
 # A reference member always needs initialization; body mode still introduces
 # a visible local value before returning the forwarded arguments below.
 if body and kind!=1:pair+=' { normal='+nv+'; point='+pv+'; }'
 else:pair+=' : '+('normal('+nv+'),point('+pv+')' if body else 'point('+pv+'),normal('+nv+')')+' {}'
 if copy:pair+=' ContactPair(const ContactPair &v) : '+('normal(v.normal),point(v.point)' if order else 'point(v.point),normal(v.normal)')+' {}'
 pair+=' };\n'
 def construct(p,n):
  a='&('+p+')' if ptr else p;b='&('+n+')' if ptr else n
  return 'ContactPair('+(','.join([b,a]) if order else ','.join([a,b]))+')'
 rty='unsigned' if ret else 'void *';cast='(void *)' if ret else ''
 expression=construct('p','n')
 forwarding='ContactPair args='+expression+'; return '+cast+'preparePair(args);' if local else 'return '+cast+'preparePair('+expression+');'
 s=base.replace('struct ContactSource {',pair+'struct ContactSource {')
 s=s.replace('virtual void *prepare(const ContactVector &point, const ContactVector &normal)=0;', 'virtual '+rty+' preparePair('+('const ' if cv else '')+'ContactPair args)=0;\n void *prepare(const ContactVector &p,const ContactVector &n) { '+forwarding+' }')
 calls=[('node->second->source','node->point','node->normal'),('node->second->source','node->point','node->normal'),('node->first->source','node->point','ContactVector(node->normal).negate()')]
 pos=0
 for i,(recv,p,n) in enumerate(calls):
  old=recv+'->prepare('+p+','+n+')';at=s.find(old,pos);assert at>=0
  new=cast+recv+'->preparePair('+construct(p,n)+')' if sites&(1<<i) else old
  s=s[:at]+new+s[at+len(old):];pos=at+len(new)
 (out/('v%04d.cpp'%idx)).write_text(s)
print('generated',idx+1)
