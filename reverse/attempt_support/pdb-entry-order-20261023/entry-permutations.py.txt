import subprocess,sys,itertools
from pathlib import Path
from concurrent.futures import ThreadPoolExecutor
base=Path('build/pdbtry/base.cpp').read_text()
old='*pline=0;source[0]=0;*moduleName=0;'
al=' char *errname=(char *)_alloca(1024);\n'
assert old in base and al in base and ' int res=0;ImageInfo' in base
b=base.replace(' int res=0;ImageInfo',' int res;ImageInfo')
st=['res=0;','*pline=0;','source[0]=0;','*moduleName=0;']
V={}
for perm in itertools.permutations(range(4)):
    seq=''.join(st[i] for i in perm)
    V['a'+''.join(map(str,perm))]=b.replace(old,seq)
    # alloca after the statements
    s2=b.replace(al,' char *errname;\n').replace(old,seq+'errname=(char *)_alloca(1024);')
    V['z'+''.join(map(str,perm))]=s2
def run(kv):
    k,s=kv
    p=Path(f'build/pdbtry/{k}.cpp'); p.write_text(s)
    r=subprocess.run([sys.executable,'tools/explain_mismatch.py','?_RTC_GetSrcLine@@YAHKPADHPAHPAPAD@Z','--rva','0x0075B177','--size','847','--source',str(p)],capture_output=True,text=True)
    o=r.stdout+r.stderr
    Path(f'build/pdbtry/{k}.txt').write_text(o)
    n=sum(1 for l in o.splitlines() if l.startswith('!='))
    return k,n,[l for l in o.splitlines() if l.startswith(('compiled symbol size','classification','first diff'))]
with ThreadPoolExecutor(4) as ex:
    for res in ex.map(run,V.items()): print(*res,flush=True)
