import ast,collections,json,re,sys
from pathlib import Path
D=Path(__file__).resolve().parent
sys.path.insert(0,str(D))
import trials
literal=lambda s:collections.Counter(re.findall(r'"(?:\\.|[^"\\])*"',s))
total=0
for script,family,label,n in [('query_order.py','query','fire-query',960),('query_order.py','groups','fire-groups',640),('output_types.py','types','output-types',248),('profiles.py','profiles','profiles',6),('output_lifetime.py','lifetime','output-lifetime',2046)]:
 captured=[]
 ns={k:v for k,v in vars(trials).items() if not k.startswith('__')}
 ns.update(__name__='verification',__file__=str(D/script))
 ns['batch']=lambda items,label:captured.extend(items) or []
 old=sys.argv;sys.argv=['verify',family]
 code=(D/script).read_text().replace('from trials import *','')
 ast.parse(code)
 exec(compile(code,str(D/script),'exec'),ns)
 sys.argv=old
 # profiles.py overwrites its own results when stubbed; source results are
 # retained separately by the normal driver for each profile.
 if script=='profiles.py':
  results=sum([json.loads((D/('profile-'+p+'-results.json')).read_text()) for p in ['rtm','vc71-ddk-4035','vc71-sp1']],[])
  (D/'profiles-results.json').write_text(json.dumps(results,indent=2))
 else:results=json.loads((D/(label+'-results.json')).read_text())
 assert len(captured)==len(results)==n
 for (tag,source),result in zip(captured,results):
  assert result['tag']==tag and 'error' not in result,tag
  assert source==(D/(tag+'.cpp')).read_text(),tag
  assert literal(source)==literal(trials.BASE),tag
  if script!='query_order.py' or family=='query':
   assert result['size']==4101 and result['regions']==0 and result['nd'] in (13,44)
  else:assert result['nd']>=72
 total+=n
 print(script,family,n,'sources regenerated, literals intact, successful compile')
assert total==3900
for release in range(2):
 raw,rel=trials.build.read_object_symbol_bytes(D/('profile-rtm'+str(release)+'.obj'),trials.NAME)
 for p in ['vc71-ddk-4035','vc71-sp1']:
  other,orel=trials.build.read_object_symbol_bytes(D/('profile-'+p+str(release)+'.obj'),trials.NAME)
  assert raw==other and rel==orel,(release,p)
print('All three compiler profiles: identical raw bodies and relocation tuples per contract')
native=json.loads((D/'query000000-before-pack.json').read_text())
rows=next(z for z in native if isinstance(z,list) and any(x['name'].startswith('_draw\n') for x in z))
prior=json.loads(Path('reverse/attempt_support/0x0004826c-sol6-allocator.json').read_text())['old']
keys=('index','size','weight','type')
rr=[dict(zip(keys,(x['data'][15],x['data'][8],x['data'][14],x['data'][11]))) for x in rows]
assert rr==[{k:r[k] for k in keys} for r in prior['rows']]
bs=lambda ns:sorted({n[0]+b for n in ns for b in range(32) if n[2]>>b&1})
c={str(x['index']):bs(x['nodes']) for x in native[-1]['conflicts']}
delta={k:dict(before=prior['conflicts'][k],after=v) for k,v in c.items() if v!=prior['conflicts'][k]}
assert delta=={'21':{'before':[0,1,5,9,21,70,73,74,75,76],'after':[21]}}
assert json.loads((D/'query-native-delta.json').read_text())==dict(parent='0x0004826c-sol6-allocator.json:old',ordered_records_equal=True,conflict_delta=delta)
print('Native query wrapper: unchanged 77 ordered records, canonical fontName delta only')
