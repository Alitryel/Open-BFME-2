import concurrent.futures, difflib, hashlib, json, subprocess, sys, time
from pathlib import Path
sys.path.insert(0, str(Path(__file__).resolve().parents[2] / 'tools'))
import build, capstone
ROOT=build.ROOT
DIR=Path(__file__).resolve().parent
target=build.read_target_bytes(0x758de0,333)
symbol='?bfmeTwoCGD@BfmeThingCGD@@QAEXXZ'

def insns(blob):
    md=capstone.Cs(capstone.CS_ARCH_X86,capstone.CS_MODE_32)
    return [(i.mnemonic,i.op_str,i.size,i.address) for i in md.disasm(blob,0)]
T=insns(target)

def run(src):
    obj=src.with_suffix('.obj')
    result={'name':src.stem}
    try:
        command,env=build.compiler_command(src,obj)
        p=subprocess.run(command,cwd=ROOT,env=env,capture_output=True,text=True,timeout=90)
        if p.returncode:
            src.with_suffix('.err').write_text(p.stdout+p.stderr)
            return dict(result,error='compile')
        data,relocs=build.read_object_symbol_bytes(obj,symbol,code_only=True)
        result.update(size=len(data),relocs=len(relocs),sha=hashlib.sha256(data).hexdigest())
        obj.with_suffix('.bin').write_bytes(data)
        A=insns(data)
        m=difflib.SequenceMatcher(None,[i[0] for i in A],[i[0] for i in T],autojunk=False)
        matched=0; regions=0; operands=0
        for tag,a,b,c,d in m.get_opcodes():
            if tag!='equal':regions+=1; continue
            for x,y in zip(A[a:b],T[c:d]):
                if x[:2]==y[:2]:matched+=y[2]
                else:operands+=1
        result.update(aligned=matched,regions=regions,operands=operands,
          positional=sum(a!=b for a,b in zip(data,target))+abs(len(data)-len(target)),
          chunks=[sum(a!=b for a,b in zip(data[lo:hi],target[lo:hi])) for lo,hi in [(0,148),(148,170),(190,225),(310,320)]],exact=data==target and not relocs)
    except BaseException as e:result['error']=str(e)
    return result

if __name__=='__main__':
    sources=sorted(DIR.glob(sys.argv[1]+'/*.cpp'))
    start=time.time();results=[]
    with concurrent.futures.ThreadPoolExecutor(max_workers=8) as pool:
        for i,r in enumerate(pool.map(run,sources)):
            results.append(r)
            if r.get('exact'):print('EXACT',r,flush=True)
            if (i+1)%128==0:print('done',i+1,'/',len(sources),'seconds',round(time.time()-start),flush=True)
    (DIR/(sys.argv[1]+'-results.json')).write_text(json.dumps(results,indent=2))
    good=[r for r in results if 'error' not in r]
    print('SUMMARY',len(sources),'compiled',len(good),'seconds',round(time.time()-start))
    for r in sorted(good,key=lambda r:(-r['aligned'],r['positional']))[:16]:print(json.dumps(r))
    print('distinct code',len({r['sha'] for r in good}))
