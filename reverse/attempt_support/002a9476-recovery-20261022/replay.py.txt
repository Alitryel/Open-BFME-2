"""Run from the repository root, copied under build/. Private ordinary compiles."""
from pathlib import Path
import json
import sys

sys.path.insert(0, str(Path('tools').resolve()))
from variant_search import Scorer
from locate import object_functions
import build

home = Path('Code/GameEngine/Source/GameLogic/SkirmishAI/SkirmishAIManager.cpp')
source = home.read_text()
# Scratch files are outside the managed compiler region: preserve effective flags.
source = source.replace('// cl: ', '// cl: /arch:SSE ', 1)
symbol = '?addNewAIForPlayer@Rva002A8F24@@QAEPAXPAVPlayer@@@Z'
directory = Path('build/addnewai-side-replay').resolve()
scorer = Scorer(symbol, 0x002A9476, 323, directory)
loop_use = '        side=&player->getSide();\n'
assert source.count(loop_use) == 2
common = source.replace(loop_use, '').replace('    const AsciiString *side;\n', '')
side_consumer = '    if (((ArmorStore *)TheArmyDefinitionManager)->'
common = common.replace(side_consumer,
                        '    const AsciiString *side=&player->getSide();\n' + side_consumer)
before_get = source.replace(loop_use, '', 1).replace(
    '            ai=((Rva004E9600 *)*it)->rva004E95D4(player);',
    '            side=&player->getSide();\n            ai=((Rva004E9600 *)*it)->rva004E95D4(player);')
after_get = source.replace(loop_use, '', 1).replace(
    '            ai=((Rva004E9600 *)*it)->rva004E95D4(player);',
    '            ai=((Rva004E9600 *)*it)->rva004E95D4(player);\n            side=&player->getSide();')
records = []
for label, text in [('common-only', common), ('found-before-get', before_get),
                    ('found-after-get', after_get), ('loop-body', source)]:
    result = scorer(text)
    output = directory / f'c{scorer.n}.obj'
    raw = next(body.rstrip(b'\xcc') for name, body, _ in object_functions(output)
               if name == symbol)
    row = dict(name=symbol, target_rva='0x002A9476', target_size='323',
               source=(directory / f'c{scorer.n}.cpp').relative_to(build.ROOT).as_posix(),
               notes='')
    strict = build.compile_function(row, scorer.symbol_map, output, allow_weak=False)
    record = dict(control=label, complete_extent=len(raw),
                  strict_exact=len(raw) == 323 and strict['bytes'] == strict['target']
                  and not strict['unresolved'], unresolved=strict['unresolved'])
    records.append(record)
    print(json.dumps(record), flush=True)
(directory / 'controls.json').write_bytes((json.dumps(records, indent=2) + '\n').encode())
assert records[-1]['strict_exact'], 'Replay is not a substitute for production verification'
