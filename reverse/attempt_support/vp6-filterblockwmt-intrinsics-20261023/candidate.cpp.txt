// Clean room: reverse/vp6_cleanroom/specs/001d7df0.md plus retail only.
// No decoder source was consulted. Native 1D7DF0..1D7FC0, cdecl, ebp frame
// with a 256-byte block buffer: picks the 2-tap (bilinear) or 4-tap (bicubic)
// sub-pixel filter from the distance between the two reference pointers. The
// bicubic path widens its byte block to int16 into the output with an SSE2
// inline asm loop; the bilinear path lets the callees write int16 directly.
// Filter banks: bilinear at 0x00DB84E0 (32 bytes per phase), bicubic at
// 0x00DB85E0 (64 bytes per phase), both inside the 768-byte table that
// Rva009C7380BinkSse.cpp owns.
union __declspec(intrin_type) __m128i
{
	unsigned __int64 words[2];
};

extern "C" __m128i _mm_loadu_si128(const __m128i *);
extern "C" __m128i _mm_unpacklo_epi8(__m128i, __m128i);
extern "C" __m128i _mm_setzero_si128();
extern "C" void _mm_storeu_si128(__m128i *, __m128i);

extern "C" const unsigned char g_vp6BilinearFilterWmt[];

#define BILINEAR_BANK(phase) (g_vp6BilinearFilterWmt + (phase) * 32)
#define BICUBIC_BANK(phase) (g_vp6BilinearFilterWmt + 0x100 + (phase) * 64)

extern "C" void __cdecl FilterBlock1d_h_wmt(const void *, void *, int, int, int, int, const void *);
extern "C" void __cdecl FilterBlock1d_v_wmt(const void *, void *, int, int, int, int, const void *);
extern "C" void __cdecl FilterBlock2d_wmt(const unsigned char *, unsigned char *, int, const void *, const void *);
extern "C" void __cdecl FilterUnpackBlock1d_hb8_wmt(const void *, void *, int, int, int, int, const void *);
extern "C" void __cdecl FilterUnpackBlock1d_vb8_wmt(const void *, void *, int, int, int, int, const void *);
extern "C" void __cdecl FilterUnpackBlock2dBil_wmt(const void *, void *, int, const void *, const void *);

extern "C" void __cdecl FilterBlock_wmt(const unsigned char *a, const unsigned char *b, short *out, int pixelsPerLine, int horizontalPhase, int verticalPhase, int bicubic)
{
	unsigned char block[256];
	int diff = b - a;
	if (diff < 0) {
		const unsigned char *t = a;
		a = b;
		b = t;
		diff = b - a;
	}
	if (diff == 0)
		return;
	if (bicubic) {
		if (diff == 1)
			FilterBlock1d_h_wmt(a, block, pixelsPerLine, 1, 8, 8, BICUBIC_BANK(horizontalPhase));
		else if (diff == pixelsPerLine)
			FilterBlock1d_v_wmt(a, block, pixelsPerLine, pixelsPerLine, 8, 8, BICUBIC_BANK(verticalPhase));
		else if (diff == pixelsPerLine - 1)
			FilterBlock2d_wmt(a - 1, block, pixelsPerLine, BICUBIC_BANK(horizontalPhase), BICUBIC_BANK(verticalPhase));
		else if (diff == pixelsPerLine + 1)
			FilterBlock2d_wmt(a, block, pixelsPerLine, BICUBIC_BANK(horizontalPhase), BICUBIC_BANK(verticalPhase));
		{
			const unsigned char *src = block;
			short *destinationRow = out;
			for (int rows = 8; rows; --rows)
			{
				_mm_storeu_si128(reinterpret_cast<__m128i *>(destinationRow),
					_mm_unpacklo_epi8(_mm_loadu_si128(reinterpret_cast<const __m128i *>(src)),
						_mm_setzero_si128()));
				src += 8;
				destinationRow += 8;
			}
		}
		return;
	}
	if (diff == 1)
		FilterUnpackBlock1d_hb8_wmt(a, out, pixelsPerLine, 1, 8, 16, BILINEAR_BANK(horizontalPhase));
	else if (diff == pixelsPerLine)
		FilterUnpackBlock1d_vb8_wmt(a, out, pixelsPerLine, pixelsPerLine, 8, 16, BILINEAR_BANK(verticalPhase));
	else if (diff == pixelsPerLine - 1)
		FilterUnpackBlock2dBil_wmt(a - 1, out, pixelsPerLine, BILINEAR_BANK(horizontalPhase), BILINEAR_BANK(verticalPhase));
	else if (diff == pixelsPerLine + 1)
		FilterUnpackBlock2dBil_wmt(a, out, pixelsPerLine, BILINEAR_BANK(horizontalPhase), BILINEAR_BANK(verticalPhase));
}
