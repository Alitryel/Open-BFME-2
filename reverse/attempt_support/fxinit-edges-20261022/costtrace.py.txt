import sys,json,re
from pathlib import Path
sys.path.insert(0,'build/netpacket-20261022e')
from wdbg import Dbg
src,obj=sys.argv[1:3];d=Dbg(src,obj,obj+'.debug.log')
d.prompt();d.prompt('set $CanDeferOnBPByAddr = 1');d.prompt('break *0x10783aac')
on=False;events=[];bps=[]
while True:
 out=d.prompt('cont')
 if 'terminated'in out:break
 eip=d.reg('eip')
 if eip==0x10783aac:
  f=d.words(d.reg('ebp'),1)[0];fn=d.string(d.words(f+4,1)[0])
  yes='?Init@FXShaderGeometry' in fn and d.reg('ebx')==0
  if yes and not on:
   on=True;print('ON',fn,flush=True)
   for ip in [0x10783a56,0x1078292a,0x10783a7d]:
    out=d.prompt('break *0x%x'%ip);print(out,flush=True)
    n=int(re.search(r'Breakpoint (\d+)',out)[1]);bps.append(n)
    print(d.prompt('condition %d (*(int *)($eax+0x1c)==51) || ((*(int *)($eax+0x18)==6) && (*(int *)($eax+0x44)>600))'%n),flush=True)
  elif not yes and on:
   for n in bps:d.prompt('disable %d'%n)
   on=False
  continue
 if not on:continue
 regs={n:d.reg(n) for n in ['eax','ebx','ecx','edx','esi','edi','esp','ebp']}
 w=d.words(regs['eax'],18)
 e={'eip':hex(eip),'regs':regs,'stack':d.words(regs['esp'],24),'record':w}
 block=d.words(regs['esp']+0x20,1)[0];e['blockptr']=hex(block);e['block']=d.words(block,48)
 if w[7]==51 or (w[6]==6 and w[17]>600 and eip==0x1078292a):
  lists={}
  for off in [0x10,0x1c]:
   p=d.words(regs['esp']+off,1)[0];lst=[];seen=set()
   while p and p not in seen:
    seen.add(p);rw=d.words(p,18);sy=d.words(rw[0],24);name='?'
    if sy and sy[0]>0x100000:
     sw=d.words(sy[0],2)
     if len(sw)>1 and sw[1]>0x100000:name=d.string(sw[1])
    lst.append({'address':p,'record':rw,'symbol':sy,'name':name});p=rw[5]
   lists[hex(off)]=lst
  e['lists']=lists
 events.append(e);Path(obj+'.cost.json').write_text(json.dumps(events,indent=2))
 print('COST',e['eip'],'idx',w[7],'serial',w[17],'old',w[3],'refs',w[6],'factor',regs['edx'],'new',regs['ebx'] if eip!=0x10783a7d else regs['ecx'],'block',hex(block),flush=True)
print('DONE',flush=True)
