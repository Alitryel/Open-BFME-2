import sys,itertools,json,time
from pathlib import Path
from concurrent.futures import ThreadPoolExecutor,as_completed
sys.path.insert(0,'build/fxinit-seat22');import search as h
h.HERE=Path('build/fxinit-edges22').resolve()
PASS='*(unsigned *)(vertices+desc.Get_Basis_End())=pass;'
ONE='*(int *)(vertices+desc.Get_Basis_End())=remap[links[0][i]];'
LOOP='   for(unsigned i=0;i<(unsigned)mesh->Get_Vertex_Count();++i) {'
def variants():
 for ty,stage,declscope,source,mask,order in itertools.product(range(3),range(4),range(3),range(2),[2,8,10],range(2)):
  if ty and source:continue
  typ=['const Vector3 *','bool','unsigned '][ty]
  streams=['positions[1]','normals[1]'];olds=[h.P,h.N];names=['selectedP','selectedN']
  decl='';sets=[];zero=[];pn=[]
  for k in range(2):
   decl+=typ+' '+names[k]+';'
   val=streams[k] if not ty else '('+streams[k]+'!=0)'
   sets.append(names[k]+'='+val+';');zero.append(names[k]+'=0;')
   line=olds[k].replace('if('+streams[k]+')','if('+names[k]+')')
   if source:line=line.replace('='+streams[k]+'[i]','='+names[k]+'[i]')
   pn.append(line)
  if order:sets.reverse();zero.reverse()
  sets=''.join(sets);zero=''.join(zero)
  rw=[sets+h.R+h.W,h.R+sets+h.W,h.R+h.W+sets,h.R+h.W][stage]
  entry=zero+'goto CopyBoth;'
  arm='if(m_Passes>1) {'+PASS+'goto FinishSkin;}\nelse if(m_BonesPerVertex>0 && links[0]) {\nif(m_BonesPerVertex==1) {'+ONE+'goto FinishSkin;}\nelse if(m_BonesPerVertex==2) {\n'+rw+'\nCopyBoth:\n'+''.join(pn)+'\n}\nelse {'+(entry if mask&8 else'goto FinishSkin;')+'}\n}\nelse {'+(entry if mask&2 else'goto FinishSkin;')+'}\nFinishSkin:;\n'
  if stage==3:arm=sets+arm
  if not declscope:arm=decl+arm
  src=h.BASE[:h.START]+arm+h.BASE[h.END:]
  if declscope==1:src=src.replace(LOOP,LOOP+decl)
  if declscope==2:src=src.replace('  for(unsigned pass=0;',decl+'\n  for(unsigned pass=0;')
  yield f'selectededge_{ty}{stage}{declscope}{source}{mask:02}{order}',src
if __name__=='__main__':
 t=time.time();items=list(variants());r=[]
 with ThreadPoolExecutor(max_workers=8)as pool:
  for f in as_completed([pool.submit(h.run_one,x)for x in items]):
   r.append(f.result())
   if len(r)%256==0:print('tested',len(r),'best',min((x for x in r if 'nd'in x),key=lambda x:(x['nd'],x['cost'])),flush=True)
 r.sort(key=lambda x:(x.get('nd',99999),x.get('cost',99999)))
 (h.HERE/'selectededge-results.json').write_text(json.dumps(r,indent=2))
 print(json.dumps({'count':len(r),'valid':sum('nd'in x and not x.get('unresolved')for x in r),'seconds':time.time()-t,'best':r[:4]},indent=2))
