import sys,itertools,json,time
from pathlib import Path
from concurrent.futures import ThreadPoolExecutor,as_completed
sys.path.insert(0,'build/fxinit-seat22');import search as h
h.HERE=Path('build/fxinit-index22').resolve()
def variants():
 changes=[
 ('char *vertices=', 'char *__restrict vertices='),
 ('const Vector3 *positions[2]', 'const Vector3 *__restrict positions[2]'),
 ('const Vector3 *normals[2]', 'const Vector3 *__restrict normals[2]'),
 ('const Vector3 *tangents=', 'const Vector3 *__restrict tangents='),
 ('const Vector3 *binormals=', 'const Vector3 *__restrict binormals='),
 ('const unsigned *colors=', 'const unsigned *__restrict colors='),
 ('const unsigned short *links[2]', 'const unsigned short *__restrict links[2]'),
 ('const unsigned short *weights[2]', 'const unsigned short *__restrict weights[2]'),
 ('const Vector2 *uv=', 'const Vector2 *__restrict uv=')]
 for bits in range(512):
  s=h.BASE
  for k,(a,b)in enumerate(changes):
   if bits>>k&1:assert a in s;s=s.replace(a,b)
  yield f'restrict_{bits:03x}',s
if __name__=='__main__':
 items=list(variants())
 if len(sys.argv)>1:items=[items[0],items[1],items[2],items[-1]]
 t=time.time()
 with ThreadPoolExecutor(max_workers=8)as pool:r=list(pool.map(h.run_one,items))
 r.sort(key=lambda x:(x.get('nd',99999),x.get('cost',99999)))
 (h.HERE/'restrict-results.json').write_text(json.dumps(r,indent=2));print(json.dumps({'count':len(r),'seconds':time.time()-t,'errors':sum('error'in x for x in r),'best':r[:4]},indent=2))
