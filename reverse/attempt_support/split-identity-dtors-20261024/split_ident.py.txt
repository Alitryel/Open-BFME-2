import re, csv, sys, subprocess
sys.path.insert(0,'build'); sys.path.insert(0,'tools')
import build as B, capstone
data,secs=B.exe_image()
def off(rva):
    for s in secs:
        if s['rva']<=rva<s['rva']+s['size']: return rva-s['rva']+s['raw_pointer']
md=capstone.Cs(capstone.CS_ARCH_X86,capstone.CS_MODE_32)
rows={}; byaddr={}
for r in csv.reader(open('reverse/functions.csv',encoding='utf-8',errors='replace')):
    if len(r)<6 or r[0]=='name': continue
    try: a=int(r[2],16); sz=int(r[3])
    except: continue
    rows[r[0]]=(a,sz,r[4]); byaddr.setdefault(a,[]).append(r[0])
txt=open(sys.argv[1]).read()
units={}
for b in re.split(r'\n(?=Code/|reference/)',txt):
    m=re.match(r'(\S+): does not link.*own bytes ([\d,]+)',b)
    if not m: continue
    bl=[l.strip() for l in b.split('\n')[1:] if re.match(r'\s*(unresolved|comdat|selected|duplicate|addresses)',l)]
    units[m.group(1)]=(int(m.group(2).replace(',','')),bl)
found={}
for u,(by,bl) in units.items():
    for l in bl:
        mm=re.match(r'unresolved\s+\?\?1(\w+)@@UAE@XZ',l)
        if not mm: continue
        cls=mm.group(1)
        ctors=[n for n in rows if n.startswith('??0%s@@'%cls)]
        for c in ctors:
            a,sz,_=rows[c]; o=off(a)
            vts=[]
            for ins in md.disasm(data[o:o+sz],a):
                if ins.mnemonic=='mov' and ins.op_str.startswith('dword ptr [') and re.search(r', 0x[bc][0-9a-f]{5}$',ins.op_str):
                    vts.append(int(ins.op_str.split(', ')[1],16))
            for vt in vts[-1:]:
                s0=int.from_bytes(data[off(vt-0x400000):off(vt-0x400000)+4],'little')-0x400000
                d=None
                for ins in md.disasm(data[off(s0):off(s0)+28],s0):
                    if ins.mnemonic in('call','jmp') and ins.op_str.startswith('0x'): d=int(ins.op_str,16); break
                if d is None: continue
                names=byaddr.get(d,[])
                found.setdefault(cls,set()).add((c,hex(vt),hex(d),tuple(names)))

tot=0
for cls,v in sorted(found.items()):
    us=[(u,units[u][0],len(units[u][1])) for u in units if any(('??1%s@@UAE@XZ'%cls) in l for l in units[u][1])]
    sole=sum(by for u,by,k in us if k==1); tot+=sole
    print(cls,'| dtor rows:',sorted({x[3] for x in v}),'| units:',[(u.split('/')[-1],by,k) for u,by,k in us])
print('sole-blocker bytes total',tot)
