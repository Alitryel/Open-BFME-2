from pathlib import Path
import sys,subprocess
root=Path(__file__).resolve().parents[1];sys.path.insert(0,str(root/'tools'))
out=root/'build/string-copy-canonical-oct20';out.mkdir(exist_ok=True)
src=out/'pod-copy.cpp'
src.write_text('''// cl: /O2 /Ob2 /EHsc /MD /D_STLP_USE_STATIC_LIB /D_STLP_USE_MALLOC
// stlport
#include <memory>
namespace _STL {
template<> char *uninitialized_copy<char*,char*>(char *first,char *last,char *result) {
 return last==first ? result :
  ((char*)memmove(result,first,last-first))+(last-first);
}
}
''')
tests=[(src,'??$uninitialized_copy@PADPAD@_STL@@YAPADPAD00@Z','0x179B0','41')]
consumer=out/'range-consumer.cpp'
consumer.write_text((root/'Code/Libraries/Source/STLport/Rva008312E0StringReplace.cpp').read_text().replace('Rva000179B0UninitializedCopy','uninitialized_copy'))
tests += [(consumer,'?bfmeReplaceAliasedRange@Rva008312E0String@@QAEAAV1@PAD000ABURva008312E0Tag@@@Z','0x2AE50','511'),(consumer,'?insertRange@Rva008312E0String@@QAEXPAD00ABURva008312E0Tag@@@Z','0x284D0','848')]
for source,sym,rva,size in tests:
 p=subprocess.run([sys.executable,'tools/explain_mismatch.py',sym,'--rva',rva,'--size',size,'--source',str(source)],cwd=root,capture_output=True,text=True,encoding='utf8',errors='replace',timeout=300)
 log=p.stdout+p.stderr;(out/(source.stem+'-'+rva+'.txt')).write_text(log)
 print(source.name,sym,log[:800],flush=True)
