from pathlib import Path
import subprocess,sys,json,concurrent.futures
out=Path('build/resume-eh-local/turn62-pdb-native-types');out.mkdir(exist_ok=True)
base=Path('reverse/attempt_support/pdb-numbering-20261021/candidate.cpp.txt').read_text()
variants=[('native_callback_chars',base.replace('const char *','char *')),('native_name_chars',base.replace('static const char *mspdbName','static char *mspdbName')),('native_pdb_input_chars',base.replace('const char *,const char *','char *,char *').replace('PDB *,const char *,const char *','PDB *,char *,char *'))]
rows=[('?_RTC_GetSrcLine@@YAHKPADHPAHPAPAD@Z','0x0075B177',847),('?GetImageInfo@@YAPAUImageInfo@@K@Z','0x0075ACEE',849),('?GetPdbDll@@YAPAUHINSTANCE__@@XZ','0x0075B03F',312)]
results=[]
for key,source in variants:
 p=out/(key+'.cpp');p.write_bytes(source.encode())
 for i,(name,rva,size) in enumerate(rows):
  r=subprocess.run([sys.executable,'tools/explain_mismatch.py',name,'--rva',rva,'--size',str(size),'--source',p.as_posix()],stdout=subprocess.PIPE,stderr=subprocess.STDOUT,text=True)
  (out/(key+f'_{i}.txt')).write_bytes(r.stdout.encode());d={'variant':key,'symbol':name,'exit':r.returncode}
  for field in ['compiled symbol size','classification','first diff:','unresolved']:
   lines=[s for s in r.stdout.splitlines() if s.lower().startswith(field)]
   if lines:d[field]=lines[0]
  results.append(d);print(d,flush=True)
  (out/'results.json').write_bytes((json.dumps(results,indent=2)+'\n').encode())
