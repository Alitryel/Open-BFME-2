import sys,capstone
P='/work/Open-BFME-2/reference/open-bfme-1/inputs/toolchains/vs2003/Program Files/Microsoft Visual Studio .NET 2003/Vc7/bin/c2.dll'
data=open(P,'rb').read()
B=0x10700000
lo=int(sys.argv[1],16); n=int(sys.argv[2]) if len(sys.argv)>2 else 80
md=capstone.Cs(capstone.CS_ARCH_X86,capstone.CS_MODE_32)
k=0
for i in md.disasm(data[lo:lo+0x2000],lo):  # .text rva == file offset
    print('%6x %s %s'%(i.address,i.mnemonic,i.op_str.replace('0x107','@')))
    k+=1
    if k>=n: break
