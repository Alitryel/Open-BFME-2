import trials,itertools,concurrent.futures,json
specs=[]
decl='template <> _String_base<wchar_t, allocator<wchar_t> >::~_String_base();'
for attribute,depth,position,visible in itertools.product(['','inline','__forceinline'],[0,1,255],['before','after'],[0,1]):
 label=f'whole-fixed-a{attribute or "plain"}-d{depth}-{position}-v{visible}'
 definition=f'\nnamespace _STL {{\n#pragma inline_depth({depth})\ntemplate <> {attribute} basic_string<wchar_t>::~basic_string() {{ _Destroy(this->_M_start, this->_M_finish+1); }}\n#pragma inline_depth(255)\n}}\n'
 src=trials.bank.replace(decl,'')
 declarations='namespace _STL {\n'+('' if visible else decl+'\n')+'template <> basic_string<wchar_t>::~basic_string();\n}\n'
 src=src.replace('#include <locale>',declarations+ ('\n'+definition if position=='before' else '') +'\n#include <locale>',1)
 if position=='after':src+=definition
 specs.append((label,src))
with concurrent.futures.ThreadPoolExecutor(max_workers=8) as p:r=list(p.map(trials.run,specs))
(trials.scratch/'whole-fixed.json').write_text(json.dumps(r,indent=2)+'\n')
