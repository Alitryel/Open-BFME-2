from pathlib import Path
import itertools,json
W=Path(__file__).resolve().parent
b=(W/'base.cpp').read_text();D=W/'edgepoint';D.mkdir(exist_ok=True)
decl='const ContactVector *p;'
preds=['node->state==1','node->state!=2','node->state&1','node->key0&1',
       'node->key1<node->key0','node->first==node->second','index&1',
       'node->first->source==node->second->source']
rows=[]
for n,(position,pred,shape,scope,rev,bridge,mixed) in enumerate(itertools.product(range(4),range(8),range(8),range(2),range(2),range(2),range(2))):
 P=preds[pred];A='p=&node->point;'
 # Preserve the address on both arms. Only the transfer/successor topology varies.
 E=[f'if({P}) {{ point_ready: {A} }} else {{ {A} }}',
    f'if({P}) goto point_ready; {A} goto point_done; point_ready: {A} point_done:;',
    f'if({P}) {{ {A} goto point_done; }} {A} point_done:;',
    f'if(!({P})) {{ {A} goto point_done; }} {A} point_done:;',
    f'if({P}) {{ point_ready: {A} }} else {{ goto point_ready; }}',
    f'p=({P})?&node->point:&node->point;',
    f'if({P}) {{ {A} }} else {{ {A} }}',
    f'{A} if({P}) goto point_done; {A} point_done:;'][shape]
 body=b.split('void BfmeThingCGD::bfmeTwoCGD() {',1)[1]
 if scope:body='\n'+decl+body
 else:body=body.replace('  ContactNode *node=current;',decl+'\n  ContactNode *node=current;')
 state='  if(node->state==0)\n   node->state=node->first->source->test(node->second->source->prepare(node->point,node->normal))?1:2;'
 guard='  if(node->state==2) continue;'
 first='  node->first->source->respond(bridge(node->second,node->point,node->normal));'
 last='  node->second->source->respond(altPrepare(node->first->source,node->point,ContactVector(node->normal).negate()));'
 if position==0:body=body.replace(state,E+'\n'+state);use_test=True;use_first=True
 elif position==1:body=body.replace(guard,E+'\n'+guard);use_test=False;use_first=True
 elif position==2:body=body.replace(guard,guard+'\n'+E);use_test=False;use_first=True
 else:body=body.replace(last,E+'\n'+last);use_test=False;use_first=False
 if use_test:body=body.replace(state,state.replace('node->point','*p'))
 f='bridge(node->second,*p,node->normal)' if use_first else 'bridge(node->second,node->point,node->normal)'
 if not bridge:f=f.replace('bridge(node->second,','node->second->source->prepare(')
 body=body.replace(first,'  node->first->source->respond('+f+');')
 rp='*p' if rev or position==3 else 'node->point'
 r=f'altPrepare(node->first->source,{rp},ContactVector(node->normal).negate())' if mixed else f'node->first->source->prepare({rp},ContactVector(node->normal).negate())'
 body=body.replace(last,'  node->second->source->respond('+r+');')
 (D/('v%04d.cpp'%n)).write_text(b.split('void BfmeThingCGD::bfmeTwoCGD() {',1)[0]+'void BfmeThingCGD::bfmeTwoCGD() {'+body)
 rows.append(dict(name='v%04d'%n,position=position,pred=pred,shape=shape,scope=scope,reverse=rev,bridge=bridge,mixed=mixed))
(W/'edgepoint-models.json').write_text(json.dumps(rows,indent=1));print(len(rows),'address-edge candidates')
