import sys,itertools,json,time
from pathlib import Path
from concurrent.futures import ThreadPoolExecutor
sys.path.insert(0,'build/fxinit-seat22');import search as h
h.HERE=Path('build/fxinit-split22').resolve()
def variants():
 base=(h.HERE/'trials/pnselect_010101.cpp').read_text()
 for move,refs,remap in itertools.product(range(4),repeat=3):
  s=base; copies=[]
  for k in range(2):
   if move>>k&1:
    stmt='if(chosen'+str(k)+') *(Vector3 *)(vertices+desc.Get_'+['Location','Normal'][k]+'_Offset(1))=*chosen'+str(k)+';'
    assert s.count(stmt)==1
    s=s.replace(stmt,'');copies.append(stmt)
   if refs>>k&1:
    name='Get_'+['Location','Normal'][k]+'_Offset'
    s=s.replace('unsigned '+name+'(','const unsigned &'+name+'(')
  s=s.replace(h.W,h.W+''.join(copies))
  forms=[h.R,
   '*(int *)(vertices+desc.Get_Basis_End())=(remap[links[1][i]]<<8)+remap[links[0][i]];',
   'int mapped0=remap[links[0][i]];int mapped1=remap[links[1][i]];*(int *)(vertices+desc.Get_Basis_End())=mapped0+(mapped1<<8);',
   'unsigned l0=links[0][i];unsigned l1=links[1][i];*(int *)(vertices+desc.Get_Basis_End())=remap[l0]+(remap[l1]<<8);']
  s=s.replace(h.R,forms[remap])
  yield f'recombine_{move}{refs}{remap}',s
if __name__=='__main__':
 t=time.time()
 with ThreadPoolExecutor(max_workers=8)as pool:r=list(pool.map(h.run_one,variants()))
 r.sort(key=lambda x:(x.get('nd',99999),x.get('cost',99999)))
 (h.HERE/'recombine-results.json').write_text(json.dumps(r,indent=2))
 print(json.dumps({'count':len(r),'valid':sum('nd'in x and not x.get('unresolved')for x in r),'seconds':time.time()-t,'best':r[:6]},indent=2))
