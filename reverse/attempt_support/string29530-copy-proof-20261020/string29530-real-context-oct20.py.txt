from pathlib import Path
import sys,subprocess,json
root=Path(__file__).resolve().parents[1];sys.path.insert(0,str(root/'tools'))
out=root/'build/string29530-real-context-oct20';out.mkdir(exist_ok=True)
base=(root/'reverse/attempts/0x00029530.cpp').read_text()
symbol='?bfmeDoInsertQX@BfmeThingQX@@QAEXPADID@Z'
alloc='''static char *allocate(unsigned int n,const void *hint) {
 unsigned int bytes=n; const void *hintCopy=hint;
 return (char *)__gameMemAllocatePtr(bytes,3,hintCopy);
}'''
assign='''static char *assign(char *s,unsigned int n,char c) {
 memset(s,c,n);return s;
}'''
free='''inline void free(void *block) {__gameMemFreePtr(block,3);}'''
globals='''extern "C" void *(__cdecl *__gameMemAllocatePtr)(unsigned int,int,const void *);
extern "C" void (__cdecl *__gameMemFreePtr)(void *,int);
'''
copy='''inline void *__copy_trivial(const void *first,const void *last,void *result) {
 if(first==last)return result;
 unsigned int n=(const char *)last-(const char *)first;
 return (char *)memmove(result,first,n)+n;
}'''
variants={}
for key,aa,ass,fr in [('allocate',True,False,False),('assign',False,True,False),('free',False,False,True),('all-leaves',True,True,True)]:
    s=base.replace('namespace _STL',globals+'\nnamespace _STL',1)
    if aa:s=s.replace('static char *allocate(unsigned int __n, const void *);',alloc)
    if ass:s=s.replace('static char *assign(char *__s, unsigned int __n, char __c);',assign)
    if fr:s=s.replace('void free(void *);',free)
    variants[key]=s
variants['full-copy']=base.replace('void *__copy_trivial(const void *__first, const void *__last, void *__result);',copy)
for depth in [0,1,2,3,4]:
    s=variants['full-copy']
    s=s.rsplit('}',1)[0]+'#pragma inline_depth('+str(depth)+')\n}\n'
    variants['full-copy-depth-'+str(depth)]=s
# Read-only proof trial: same complete copy/tag implementation under the
# optimized profile of native's already-owned 41-byte copy function.
s=variants['full-copy'].replace('/Od /Ob1','/O2 /Ob2')
s+='\nnamespace _STL {template char *Rva000179B0UninitializedCopy<char *,char *>(char *,char *,char *);}\n'
variants['copy-provider-O2']=s
import build,capstone
from locate import object_functions
rows=[]
for key,text in variants.items():
    src=out/(key+'.cpp');src.write_text(text)
    p=subprocess.run([sys.executable,'tools/explain_mismatch.py',symbol,'--rva','0x29530','--size','662','--source',str(src)],cwd=root,capture_output=True,text=True,encoding='utf8',errors='replace',timeout=300)
    log=p.stdout+p.stderr;src.with_suffix('.mismatch.txt').write_text(log)
    obj=build.obj_path(src)
    if not obj.exists():print(key,'COMPILE ERROR',log[-800:],flush=True);continue
    fs=list(object_functions(obj));parent=next(b for n,b,r in fs if n==symbol)
    helpers=[(n,len(b)) for n,b,r in fs if 'Rva000179B0UninitializedCopy' in n or n.startswith('?assign@') or n.startswith('?allocate@') or n.startswith('?free@')]
    q=subprocess.run([sys.executable,'tools/align_diff.py',str(src),symbol,'0x29530','662'],cwd=root,capture_output=True,text=True,encoding='utf8',errors='replace',timeout=120)
    src.with_suffix('.align.txt').write_text(q.stdout+q.stderr)
    md=capstone.Cs(capstone.CS_ARCH_X86,capstone.CS_MODE_32)
    frame=next((i.op_str for i in md.disasm(parent,0) if i.mnemonic=='sub' and i.op_str.startswith('esp,')),None)
    summary=[l for l in log.splitlines() if l.startswith(('classification:','first diff:','OK:','unresolved:'))]
    print(key,len(parent),frame,helpers,summary,flush=True)
    rows.append(dict(key=key,bytes=len(parent),frame=frame,helpers=helpers,summary=summary))
(out/'results.json').write_text(json.dumps(rows,indent=2))
