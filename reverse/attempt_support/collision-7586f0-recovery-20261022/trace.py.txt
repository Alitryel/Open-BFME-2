from wdbg import *
W=Path('build/collision-7586f0-accessors')
tag=sys.argv[1]
src=W/('trace-'+tag+'.cpp')
s=Path('build/collision-7586f0-accessors/trace-derived.cpp').read_text()
if tag=='winner':s=Path('Code/Libraries/Source/collisionmanager/collisionmanager_impl.cpp').read_text()
if tag=='indexed' or tag.startswith('diag'):s=s.replace('SortList *B = A + 1;','SortList *B = &lists[src+1];')
src.write_text(s)
d=Dbg(str(src),str(W/('trace-'+tag+'.obj')),str(W/('trace-'+tag+'.log')))
records=[];active=False
try:
 d.prompt();d.prompt('set $CanDeferOnBPByAddr = 1');d.prompt('break *0x10760539')
 for i in range(1000):
  o=d.prompt('cont',30)
  if 'terminated' in o or not o.strip():break
  if '0x10760539' in o:
   w=d.words(d.reg('eax'),1);sy=d.words(w[0],2);fn=d.string(sy[1])
   if active:break
   if '?rva007586F0@CollisionManagerImpl@@' in fn:
    active=True;d.prompt('break *0x10782b26');print('active',flush=True)
   continue
  if active and '0x10782b26' in o:
   addr=d.reg('eax');w=d.words(addr,28)
   r={'addr':addr,'words':w,'stack':d.words(d.reg('esp'),12)}
   if len(w)>17:
    r['priority']=w[3];r['id']=w[17]
    sw=d.words(w[0],20)
    r['symbol']=sw
    if len(sw)>7:r['symbol_id']=sw[7]
    if len(sw)>1 and 0x10000<sw[1]<0x10000000:r['name']=d.string(sw[1])
    if sw[0] and sw[0]<0x10000000:
     fw=d.words(sw[0],16);r['front']=fw
     if len(fw)>1 and 0x10000<fw[1]<0x10000000:r['front_name']=d.string(fw[1])
    # CAUSAL DIAGNOSTIC ONLY. Never copy these objects to production or admit them.
    selected=(r.get('symbol_id')==13 and w[7]==13 and tag in ['diag-a','diag-both']) or (r.get('symbol_id')==279 and w[7]==27 and tag in ['diag-h','diag-both'])
    if selected and r['stack'][0]==0x1078474f:
     value={13:32,279:240}[r['symbol_id']]
     d.prompt('set *(unsigned int *)0x%x = %d'%(addr+12,value))
     r['diagnostic_priority']=value
   records.append(r);print('entry',len(records),r.get('id'),r.get('priority'),flush=True)
finally:
 (W/('trace-'+tag+'.json')).write_text(json.dumps(records,indent=1))
 if d.p.poll() is None:
  try:d.prompt('quit')
  except Exception:d.p.kill()
 d.log.close()
