import switch_results as s
import itertools,json,concurrent.futures,re
# The success/default arms all return one. Explicit successful cases leave
# behavior unchanged but can alter SSA copy preferences, as in recipe 596E15.
forms=[
'switch(status){case K:r=1;break;case A:case B:r=0;break;default:r=1;break;}return r;',
'switch(status){case A:case B:r=0;break;case K:r=1;break;default:r=1;break;}return r;',
'switch(status){case K:goto good;case A:case B:goto bad;default:goto good;}bad:r=0;goto done;good:r=1;done:return r;',
'switch(status){case K:r=1;goto done;case A:case B:r=0;goto done;default:++r;goto done;}done:return r;',
'switch(status){case K:releaseArgs(argv);r=1;break;case A:case B:releaseArgs(argv);r=0;break;default:releaseArgs(argv);r=1;break;}return r;',
'switch(status){case K:releaseArgs(argv);return 1;case A:case B:releaseArgs(argv);return 0;default:releaseArgs(argv);return 1;}',
'switch(status){case A:case B:releaseArgs(argv);goto done;case K:releaseArgs(argv);++r;goto done;default:releaseArgs(argv);++r;goto done;}done:return r;',
'switch(status){case A:case B:releaseArgs(argv);break;default:releaseArgs(argv);++r;break;}return r;',
]
flagsets=['','/Og-','/Ob0','/Ob2','/Oi','/Os /Og-','/Ot','/O2','/O2 /Og-','/Ox','/Og /Oi /Os','/Ob0 /Og-','/Ob2 /Og-','/Oa','/Op','/Oy-']
jobs=[]
for fi,kind,flag in itertools.product(range(len(forms)),['int','bool','unsigned char','unsigned short'],flagsets):
 text=s.prefix.replace('/EHsc','/EHsc '+flag)
 text+=kind+' r=0;long status=WinVerifyTrust(0,&action,&data);'
 if fi<4:text+='releaseArgs(argv);'
 text+=re.sub(r'\b(A|B|K)\b',lambda m:{'A':s.A,'B':s.B,'K':'0L'}[m[0]],forms[fi])+'\n}\n'
 jobs.append((f'c{len(jobs):04}',text,dict(form=fi,kind=kind,flags=flag)))
if __name__=='__main__':
 old=json.loads((s.OUT/'success-cases.json').read_text()) if (s.OUT/'success-cases.json').exists() else []
 keep={r['name']:r for r in old if r['exit']==0}
 retry=[j for j in jobs if j[0] not in keep]
 with concurrent.futures.ThreadPoolExecutor(max_workers=8) as pool:updated=list(pool.map(s.run,retry))
 keep.update({r['name']:r for r in updated});rows=[keep[j[0]] for j in jobs]
 print('compiled',len(retry),'retained',len(rows)-len(updated))
 (s.OUT/'success-cases.json').write_text(json.dumps(rows,indent=2)+'\n')
 good=[r for r in rows if r['exit']==0]
 print('SUMMARY',json.dumps(dict(compiles=len(rows),good=len(good),distinct_raw=len({r['raw_sha256'] for r in good}),exact=sum(not r['diff'] for r in good))))
 for r in sorted(good,key=lambda r:len(r['diff']))[:5]:print(json.dumps({k:v for k,v in r.items() if k!='relocations'}))
