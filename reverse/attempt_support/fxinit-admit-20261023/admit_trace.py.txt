import sys,json,re
from pathlib import Path
sys.path.insert(0,'build/netpacket-20261022e')
from wdbg import Dbg

def capture_tuple(p):
 w=d.words(p,16);z={'ptr':p,'w':w,'args':[]}
 for ix in [10,11]:
  a=w[ix];ls=[];seen=set()
  while 0x10000<a<0x10000000 and a not in seen and len(seen)<20:

   if len(v)<12:break
   ls.append({'ptr':a,'w':v});a=v[0]
  z['args'].append(ls)
 return z
while True:

  f=d.words(d.reg('ebp'),1)[0];name=d.string(d.words(f+4,1)[0])
  yes='?Init@FXShaderGeometry' in name and d.reg('ebx')==0
  if yes and not on and not done:
   on=True;print('ON',name,flush=True)
   for site,reg in sites.items():
    r=d.prompt('break *0x%x'%(0x10700000+site));n=int(re.search(r'Breakpoint (\d+)',r)[1]);bps.append(n)
    cond='(*(int *)($%s+0x1c)==51)'%reg if site==0x8292a else '(*(int *)($%s+0x1c)==1)||(*(int *)($%s+0x1c)==29)'%(reg,reg)
    d.prompt('condition %d %s'%(n,cond))
  elif on and not yes:
   for n in bps:d.prompt('disable %d'%n)
   on=False;done=True

 if not on:continue

  e['tuple']={'ptr':stack[16],'w':d.words(stack[16],16)}

  q=bw[7];ls=[];seen=set()

  e['blocktuples']=ls
 events.append(e);Path(obj+'.events.json').write_text(json.dumps(events))
 print(hex(ip),'block',hex(block),'candidate',record[7],'flags',hex(record[1]),'tuple',hex(stack[16]),flush=True)
 if ip==0x8292a:
  for n in bps:d.prompt('disable %d'%n)
  on=False;done=True
print('DONE',flush=True)
