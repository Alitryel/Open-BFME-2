import sys,json,itertools,time
from pathlib import Path
from concurrent.futures import ThreadPoolExecutor,as_completed
sys.path.insert(0,'build/fxinit-seat22');import search as h
h.HERE=Path('build/fxinit-admit23').resolve()
sites=['links[1]=links[0]+mesh->Get_Vertex_Count();','weights[0]=links[1]+mesh->Get_Vertex_Count();','weights[1]=weights[0]+mesh->Get_Vertex_Count();','positions[1]=mesh->Get_Vertex_Array(1);','normals[1]=mesh->Get_Vertex_Normal_Array(true);','((Rva0018C7E5Vector *)&remap)->resize(*m_bones.rbegin()+1,0);','remap[*bone]=index;','const Vector2 *uv=mesh->Get_UV_Array_By_Index(t);']
def variants():
 for mask,cond,form in itertools.product(range(256),range(4),range(2)):
  if mask==0 and (cond or form):continue
  s=h.BASE;p=['mesh!=0','(m_FVF&0x10)!=0','m_BonesPerVertex==2','links[0]!=0'][cond]
  for i,old in enumerate(sites):
   if not mask>>i&1:continue
   code=old;pre=''
   if i==7:pre='const Vector2 *uv;';code='uv=mesh->Get_UV_Array_By_Index(t);'
   if form==0:new='if('+p+'){'+code+'}else{'+code+'}'
   else:new='if(!('+p+'))goto SetupElse'+str(i)+';'+code+'goto SetupDone'+str(i)+';SetupElse'+str(i)+':'+code+'SetupDone'+str(i)+':;'
   s=s.replace(old,'{'+pre+new+'}' if i!=7 else pre+new)
  yield f'setuppaths_{mask:03}_{cond}{form}',s
if __name__=='__main__':
 t=time.time();rs=[]
 with ThreadPoolExecutor(max_workers=8)as ex:
  for f in as_completed([ex.submit(h.run_one,x)for x in variants()]):
   rs.append(f.result())
   if len(rs)%256==0:print('tested',len(rs),'best',min(rs,key=lambda z:(z.get('nd',99999),z.get('cost',99999))),flush=True)
 rs.sort(key=lambda z:(z.get('nd',99999),z.get('cost',99999)))
 (h.HERE/'setuppaths-results.json').write_text(json.dumps(rs,indent=2))
 print(json.dumps({'count':len(rs),'seconds':time.time()-t,'errors':sum('error'in z for z in rs),'best':rs[:6]},indent=2))
