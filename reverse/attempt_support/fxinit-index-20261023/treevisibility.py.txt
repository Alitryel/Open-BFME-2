import sys,json,itertools,time
from pathlib import Path
from concurrent.futures import ThreadPoolExecutor
sys.path.insert(0,'build/fxinit-seat22');import search as h
h.HERE=Path('build/fxinit-index22').resolve()
base=(h.HERE/'trials/freevis_001.cpp').read_text()
base=base.replace('class Rva0018C262:','struct Rva0018C262Node {int m_00,m_04; Rva0018C262Node *m_08,*m_0C;unsigned short m_10;};\nclass Rva0018C262:')
base=base.replace(' void rva0018C316();',' void rva0018C316();\n void rva0018C262(Rva0018C262Node*);')
clear='''void Rva0018C262::rva0018C316() {
 if(_M_t._M_node_count==0) return;
 rva0018C262((Rva0018C262Node*)_M_t._M_header._M_data->_M_parent);
 _M_t._M_header._M_data->_M_left=_M_t._M_header._M_data;
 _M_t._M_header._M_data->_M_parent=0;
 _M_t._M_header._M_data->_M_right=_M_t._M_header._M_data;
 _M_t._M_node_count=0;
}
'''
erase='''void Rva0018C262::rva0018C262(Rva0018C262Node *p) {
 if(!p)return;
 do {rva0018C262(p->m_0C);Rva0018C262Node *next=p->m_08;free(p);p=next;}while(p);
}
'''
def variants():
 for a,b in itertools.product(range(5),repeat=2):
  pre='';post=''
  for body,c in [(clear,a),(erase,b)]:
   if c==1:pre+=body
   elif c==2:pre+='__declspec(noinline) '+body
   elif c==3:post+=body
   elif c==4:pre+='inline __declspec(noinline) '+body
  x=base.replace('Rva0018C262::~Rva0018C262()',pre+'Rva0018C262::~Rva0018C262()')+'\n'+post
  yield 'treevis_'+str(a)+str(b),x
if __name__=='__main__':
 with ThreadPoolExecutor(max_workers=8)as ex:r=list(ex.map(h.run_one,variants()))
 (h.HERE/'treevis-results.json').write_text(json.dumps(r,indent=2))
 print(json.dumps({'count':len(r),'errors':sum('error'in x for x in r),'best':sorted(r,key=lambda x:x.get('nd',99999))[:3]},indent=2))
