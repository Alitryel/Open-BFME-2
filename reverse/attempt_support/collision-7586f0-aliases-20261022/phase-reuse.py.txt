from search import *
import itertools,re
start=BASE.index('void CollisionManagerImpl::rva007586F0()')
pre,original=BASE[:start],BASE[start:]
items=[]
# Equal labels reuse one C++ local across disjoint phases. cur/next overlap and remain distinct.
partitions=[(0,1,x,y) for x in range(3) for y in range(4) if y<=max(0,1,x)+1]
for n,(iv,parts,hoist,indexed) in enumerate(itertools.product(range(6),partitions,range(2),range(2))):
 s=original
 if iv:
  target=['','src','runLen','out','a','b'][iv]
  dtype='int' if target in ['src','out'] else 'unsigned int'
  s=s.replace('int i = 0;',dtype+' '+target+' = 0;')
  s=re.sub(r'\bi\b',target,s)
  init={'src':'0','runLen':'1','out':'src ^ 2','a':'runLen','b':'runLen'}[target]
  # Remove the later declaration, preserving the early zero initialization.
  token=dtype+' '+target+' = '+init+';'
  if init=='0':
   p=s.index(token)+len(token)
   s=s[:p]+s[p:].replace(token,target+' = '+init+';',1)
  else:s=s.replace(token,target+' = '+init+';',1)
 names=['cur','next','e','p']; first={}
 for name,label in zip(names,parts):
  if label in first:
   rep=first[label]
   s=s.replace('Endpoint *'+name+' =',''+rep+' =')
   s=re.sub(r'\b'+name+r'\b',rep,s)
  else:first[label]=name
 if hoist:
  # Flatten endpoint declarations across phase boundaries, without retaining values.
  for name in first.values():
   s=s.replace('Endpoint *'+name+' =',name+' =')
  s=s.replace('SortList lists[4];','SortList lists[4]; Endpoint '+', '.join('*'+name for name in first.values())+';')
 if indexed:s=s.replace('SortList *B = A + 1;','SortList *B = &lists[src+1];')
 items.append(('reuse-%04d'%n,pre+s))
batch(items,'phase-reuse')
