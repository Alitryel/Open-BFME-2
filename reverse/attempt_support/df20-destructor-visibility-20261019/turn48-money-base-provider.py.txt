import json
import re
import subprocess
import sys
from pathlib import Path

root = Path(__file__).resolve().parents[2]
scratch = root / 'build/resume-eh-local'
sys.path.insert(0, str(root / 'tools'))
import build, eh_verify

source = (scratch / 'turn48-money-native-proxy-lifetime.cpp').read_text()
definition = '''namespace _STL {
template <> __declspec(noinline) _String_base<wchar_t, allocator<wchar_t> >::~_String_base() {
 _M_deallocate_block();
}
}
'''
source = source.replace('#include <locale>', definition + '\n#include <locale>')
path = scratch / 'turn48-money-base-provider.cpp'
path.write_text(source)
symbol = '??1?$_String_base@GV?$allocator@G@_STL@@@_STL@@QAE@XZ'
proc = subprocess.run([sys.executable, str(root / 'tools/explain_mismatch.py'), symbol,
    '--source', path.relative_to(root).as_posix(), '--rva', '0x0000B3C0', '--size', '63'],
    cwd=root, capture_output=True, text=True)
log = scratch / 'turn48-money-base-provider.txt'
log.write_text(proc.stdout + proc.stderr)
print('\n'.join((proc.stdout + proc.stderr).splitlines()[:18]))
if proc.returncode == 0:
    obj = eh_verify.Obj(build.obj_path(path))
    print('EH', eh_verify.verify_row(eh_verify.Image(), obj, symbol, 0xb3c0))
