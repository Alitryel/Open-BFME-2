from search import *
from context import getters,mutators,amount_local
import itertools

def scope(s,form,late):
 if form:
  # End setup-local lifetimes at a genuine phase boundary. A final object
  # carrier is the only value used outside these blocks.
  starts=['const FoundationAIUpdateModuleData *md =', 'Object *foundation =',
          'static NameKeyType memberKey =', 'CreateMask statusBits;']
  start=s.index(starts[(form-1)%4],s.index('Object *FoundationAIUpdate::construct'))
  # The first static key is also used later; leave its declaration outside
  # the nested preparation block when its original position would be inside.
  key='static NameKeyType memberKey = TheNameKeyGenerator->nameToKey("CastleMemberBehavior");'
  key_inside=start<=s.index(key)
  end=s.index('\n\towningPlayer->onStructureCreated',start)
  begin=s.index('Object *obj = TheThingFactory->newObject')
  if form<=4:
   s=s[:begin]+s[begin:].replace('Object *obj = TheThingFactory->newObject','obj = TheThingFactory->newObject',1)
   end=s.index('\n\towningPlayer->onStructureCreated',start)
   s=s[:end]+'\n }\n'+s[end:]
   s=s[:start]+'Object *obj;\n'+('NameKeyType *tailMemberKey;\n' if key_inside else '')+' {\n '+s[start:]
  else:
   region=s[start:end]
   region=re.sub(r'\bobj\b','preparedObject',region)
   s=s[:start]+'Object *obj;\n'+('NameKeyType *tailMemberKey;\n' if key_inside else '')+' {\n '+region+'\n obj=preparedObject;\n }\n'+s[end:]
  if key_inside:
   s=s.replace(key,key+'\n tailMemberKey=&memberKey;',1)
   end=s.index('\n\towningPlayer->onStructureCreated')
   s=s[:end]+s[end:].replace('findModule(memberKey)','findModule(*tailMemberKey)')
 if late:
  start=s.index('BfmeItemE63 *newMember =')
  end=s.index('\n\tObject *self = getObject();',start)
  if late&1:s=s[:end]+'\n }\n'+s[end:];s=s[:start]+'{\n '+s[start:]
  if late&2:
   start=s.index('Object *self = getObject();')
   end=s.index('\n\tif (isRebuild)\n\t\t((Rva0028CBFD *)obj)',start)
   s=s[:end]+'\n }\n'+s[end:];s=s[:start]+'{\n '+s[start:]
 return s

tests={}
for form,late,mask in itertools.product(range(9),range(4),range(32)):
 s=scope(BASE,form,late)
 if s is None:continue
 for i,fn in enumerate([getters,mutators,amount_local]):
  if mask&(1<<i):s=fn(s)
 if mask&8:
  s=s.replace('Real orientation = angle;','Real orientation;\n orientation = angle;',1)
 if mask&16:
  s=s.replace('BfmeItemE63 *newMember = (BfmeItemE63 *)obj->findModule(memberKey);','BfmeItemE63 *newMember; BfmeItemE63 *oldMember;\n newMember = (BfmeItemE63 *)obj->findModule(memberKey);',1)
  s=s.replace('BfmeItemE63 *oldMember = (BfmeItemE63 *)getObject()->findModule(memberKey);','oldMember = (BfmeItemE63 *)getObject()->findModule(memberKey);',1)
 tests[f'scopes-{form}-{late}-{mask:02d}']=s
run(tests,'scopes')
