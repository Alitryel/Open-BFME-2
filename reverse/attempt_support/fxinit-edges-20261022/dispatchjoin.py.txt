import sys,itertools,json,time
from pathlib import Path
from concurrent.futures import ThreadPoolExecutor,as_completed
sys.path.insert(0,'build/fxinit-seat22');import search as h
h.HERE=Path('build/fxinit-edges22').resolve()
PASS='*(unsigned *)(vertices+desc.Get_Basis_End())=pass;'
ONE='*(int *)(vertices+desc.Get_Basis_End())=remap[links[0][i]];'
def variants():
 for order,headform,guard,which in itertools.product(itertools.permutations(range(4)),range(3),range(3),range(3)):
  labels=['PassStore','OneBone','TwoBone','SkipSkin']
  cond=['!positions[1]&&!normals[1]','!((unsigned)positions[1]|(unsigned)normals[1])','positions[1]==normals[1]&&!positions[1]'][guard]
  bodies=[PASS+'goto FinishSkin;',ONE+'goto FinishSkin;',h.R+'\n'+h.W+'\nCopyBoth:\n'+h.P+'\n'+h.N+'\ngoto FinishSkin;','if('+cond+') goto CopyBoth;goto FinishSkin;']
  sk1='SkipSkin' if which in [0,2]else'FinishSkin';sk2='SkipSkin' if which in [1,2]else'FinishSkin'
  if headform==0:head=f'if(m_Passes>1) goto PassStore;if(m_BonesPerVertex<=0 || !links[0]) goto {sk1};if(m_BonesPerVertex==1) goto OneBone;if(m_BonesPerVertex==2) goto TwoBone;goto {sk2};'
  if headform==1:head=f'if(m_Passes>1) goto PassStore;else if(m_BonesPerVertex>0 && links[0]) {{if(m_BonesPerVertex==1) goto OneBone;else if(m_BonesPerVertex==2) goto TwoBone;else goto {sk2};}}else goto {sk1};'
  if headform==2:head=f'if(m_Passes<=1) {{if(m_BonesPerVertex>0 && links[0]) {{if(m_BonesPerVertex!=1) {{if(m_BonesPerVertex!=2)goto {sk2};goto TwoBone;}}goto OneBone;}}goto {sk1};}}goto PassStore;'
  arm=head+'\n'
  for k,j in enumerate(order):
   body=bodies[j];nxt=labels[order[k+1]] if k<3 else'FinishSkin'
   if body.endswith('goto '+nxt+';'):body=body[:-len('goto '+nxt+';')]
   arm+=labels[j]+':\n'+body+'\n'
  arm+='FinishSkin:;\n'
  yield 'dispatchjoin_'+''.join(map(str,order))+f'_{headform}{guard}{which}',h.BASE[:h.START]+arm+h.BASE[h.END:]
if __name__=='__main__':
 t=time.time();items=list(variants());r=[]
 with ThreadPoolExecutor(max_workers=8)as pool:
  for f in as_completed([pool.submit(h.run_one,x)for x in items]):
   r.append(f.result())
   if len(r)%512==0:print('tested',len(r),'best',min((x for x in r if 'nd'in x),key=lambda x:(x['nd'],x['cost'])),flush=True)
 r.sort(key=lambda x:(x.get('nd',99999),x.get('cost',99999)))
 (h.HERE/'dispatchjoin-results.json').write_text(json.dumps(r,indent=2))
 print(json.dumps({'count':len(r),'valid':sum('nd'in x and not x.get('unresolved')for x in r),'seconds':time.time()-t,'best':r[:4]},indent=2))
