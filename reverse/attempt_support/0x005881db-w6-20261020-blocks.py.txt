import sys; sys.path.insert(0,'tools')
from pathlib import Path
from locate import object_functions
import capstone
md=capstone.Cs(capstone.CS_ARCH_X86, capstone.CS_MODE_32)
SYM='?updateMeleeAttack@HordeMeleeAmoeba@@UAEXPAVObject@@@Z'
def body(n):
    p=Path('build/match/build_w6_5881db_v_%s.obj'%n)
    for x in object_functions(p):
        if x[0]==SYM: return x[1]
for n in sys.argv[1:]:
    c=body(n); ins=list(md.disasm(bytes(c),0))
    leaders={0}; nbr=0; ncall=0
    for k,i in enumerate(ins):
        if i.mnemonic.startswith('j'):
            nbr+=1
            try: leaders.add(int(i.op_str,16))
            except: pass
            if k+1<len(ins): leaders.add(ins[k+1].address)
        if i.mnemonic=='call': ncall+=1
    node='?'
    for i in ins:
        if 0x60<=i.address<0xa0 and i.mnemonic=='mov' and i.op_str.startswith(('edi, dword ptr [e','esi, dword ptr [e')) and i.op_str.endswith('+ 8]'): node=i.op_str; break
    print('%-14s blocks=%3d branches=%3d calls=%3d insns=%4d %s'%(n,len(leaders),nbr,ncall,len(ins),node))
