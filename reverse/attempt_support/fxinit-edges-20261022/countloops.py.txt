import sys,itertools,json,time
from pathlib import Path
from concurrent.futures import ThreadPoolExecutor,as_completed
sys.path.insert(0,'build/fxinit-seat22');import search as h
h.HERE=Path('build/fxinit-edges22').resolve()
streams=[('tangents','desc.Get_Basis_Offset()'),('binormals','desc.Get_Basis_Second_Offset()'),('positions[1]','desc.Get_Location_Offset(1)'),('normals[1]','desc.Get_Normal_Offset(1)')]
def variants():
 for context,a,b,c,d in itertools.product(range(2),range(3),range(3),range(8),range(8)):
  src=h.BASE if not context else (h.HERE/'trials/nulljoin_0020002.cpp').read_text()
  for k,((stream,off),form) in enumerate(zip(streams,[a,b,c,d])):
   old=f'if({stream}) *(Vector3 *)(vertices+{off})={stream}[i];'
   copy=f'*(Vector3 *)(vertices+{off})={stream}[i];';j='copyCount'+str(k)
   if form==0:new=old
   if form==1:new=f'for(unsigned {j}=0;{j}<({stream}!=0);++{j}) {{ {copy} }}'
   if form==2:new=f'{{int {j}={stream}?1:0;for(;{j};--{j}) {{ {copy} }} }}'
   if form==3:new=f'for(unsigned {j}=({stream}!=0);{j}>0;--{j}) {{ {copy} }}'
   if form==4:new=f'for(int {j}=0;{j}<=({stream}?0:-1);++{j}) {{ {copy} }}'
   if form==5:new=f'for(unsigned {j}=0;{j}!=({stream}!=0);++{j}) {{ {copy} }}'
   if form==6:new=f'{{int {j}=({stream}!=0);while({j}--) {{ {copy} }} }}'
   if form==7:new=f'for(int {j}={stream}?3:0;{j}>=3;{j}-=3) {{ {copy} }}'
   assert src.count(old)==1
   src=src.replace(old,new)
  yield f'countloops_{context}{a}{b}{c}{d}',src
if __name__=='__main__':
 t=time.time();items=list(variants());r=[]
 with ThreadPoolExecutor(max_workers=8)as pool:
  for f in as_completed([pool.submit(h.run_one,x)for x in items]):
   r.append(f.result())
   if len(r)%512==0:print('tested',len(r),'best',min((x for x in r if 'nd'in x),key=lambda x:(x['nd'],x['cost'])),flush=True)
 r.sort(key=lambda x:(x.get('nd',99999),x.get('cost',99999)))
 (h.HERE/'countloops-results.json').write_text(json.dumps(r,indent=2))
 print(json.dumps({'count':len(r),'valid':sum('nd'in x and not x.get('unresolved')for x in r),'seconds':time.time()-t,'best':r[:4]},indent=2))
