from search import *
import itertools,re

def source(flags):
    place, kind, dtor, access, storage = flags
    body=BODY
    marks=['\n\tGameInfo *game','\n\tif (!game)','\n\tif (slot == -1 || slot >= 8)','\n\tbool observer']
    p=body.index(marks[place])
    end=body[p:]
    payload='int value;'
    init='value(v)'
    if kind==1:payload='int *ptr;';init='ptr(&v)'
    if kind==2:payload='int *ptr; int value;';init='ptr(&v),value(v)'
    if kind==3:payload='const int *ptr; int value;';init='ptr(&v),value(v)'
    read='*ptr' if kind==1 else 'value'
    ctor='Index(int &v):'+init+'{}'
    dtors=['~Index() {}','~Index() throw() {}', '~Index() { value=0; }' if kind!=1 else '~Index() {ptr=0;}', '~Index() {release();} void release() {}']
    methods='operator int() const {return '+read+';} Index &operator=(int v){'+read+'=v;return *this;} int &get(){return '+read+';}'
    decl='struct Index {'+payload+ctor+dtors[dtor]+methods+'};'
    if storage==0:
        decl+=' Index index(slot);'
    else:
        decl+=' Index original(slot); Index index(original);'
    expr=['index', 'index.get()', ('(*index.ptr)' if kind==1 else 'index.value')][access]
    end=re.sub(r'\bslot\b',expr,end)
    return HEAD+MARK+body[:p]+'\n '+decl+end

if __name__=='__main__':
    run([('lifetime-%03d'%n,source(f)) for n,f in enumerate(itertools.product(range(4),range(4),range(4),range(3),range(2)))], 'lifetime')
