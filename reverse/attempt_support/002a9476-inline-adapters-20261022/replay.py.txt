from pathlib import Path
import json
import sys

sys.path.insert(0, str(Path('tools').resolve()))
from variant_search import Scorer
from locate import object_functions

source = Path('reverse/attempts/0x002a9476.cpp').read_text()
symbol = '?addNewAIForPlayer@Rva002A8F24@@QAEPAXPAVPlayer@@@Z'
directory = Path('build/addnewai-inline-replay').resolve()
scorer = Scorer(symbol, 0x002A9476, 323, directory)
for join, lookup in [(False, False), (True, False), (False, True), (True, True)]:
    text = source
    if not join:
        text = text.replace('joinTeam0((AIGameTeam *)*it,player)',
                            '((AIGameTeam *)*it)->addPlayer((const ModuleData *)player)')
    if not lookup:
        text = text.replace('getTeamAI0((Rva004E9600 *)*it,player)',
                            '((Rva004E9600 *)*it)->rva004E95D4(player)')
    result = scorer(text)
    record = {k: v for k, v in result.items() if k not in ('compiled', 'target')}
    record.update(join_adapter=join, lookup_adapter=lookup)
    obj = directory / f'c{scorer.n}.obj'
    if obj.exists():
        record['full_extent'] = next(len(body.rstrip(b'\xcc'))
                                    for name, body, relocs in object_functions(obj)
                                    if name == symbol)
    print(json.dumps(record))
