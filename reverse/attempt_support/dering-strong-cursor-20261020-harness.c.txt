/* Native differential check: banked body, candidate body and a spec model. */
#include <stdio.h>
#include <stdlib.h>
#include <string.h>

int Vp6FilterEdgeTagTable[64];
void __cdecl Bank_Strong(void *, const unsigned char *, unsigned char *, int, int, const int *);
void __cdecl Cand_Strong(void *, const unsigned char *, unsigned char *, int, int, const int *);

static unsigned int seed = 0x1C1AD0;
static unsigned int rnd(void) { seed = seed * 1103515245u + 12345u; return (seed >> 8) & 0xFFFFFF; }

static int weight(int a, int b, int quality, int cap, int sharpen)
{
	int d = a > b ? a - b : b - a;
	int w = quality - d + 32;
	if (w < -64) return sharpen;
	if (w < 0) return 0;
	return w > cap ? cap : w;
}

static void model(unsigned char *buf, int origin, int dest, int pitch, int q, const int *scale)
{
	int quality = scale[q], cap = 3 * quality > 32 ? 32 : 3 * quality, sharpen = Vp6FilterEdgeTagTable[q];
	int v[9][8], h[8][9], r, c;
	for (r = 0; r < 9; r++)
		for (c = 0; c < 8; c++)
			v[r][c] = weight(buf[origin + (r - 1) * pitch + c], buf[origin + r * pitch + c], quality, cap, sharpen);
	for (r = 0; r < 8; r++)
		for (c = 0; c < 9; c++)
			h[r][c] = weight(buf[origin + r * pitch + c - 1], buf[origin + r * pitch + c], quality, cap, sharpen);
	for (r = 0; r < 8; r++) {
		int row[8];
		for (c = 0; c < 8; c++) {
			int p = origin + r * pitch + c;
			int up = v[r][c], down = v[r + 1][c], left = h[r][c], right = h[r][c + 1];
			int value = ((128 - up - down - left - right) * buf[p] + up * buf[p - pitch] + down * buf[p + pitch]
				+ left * buf[p - 1] + right * buf[p + 1] + 64) >> 7;
			row[c] = value < 0 ? 0 : value > 255 ? 255 : value;
		}
		for (c = 0; c < 8; c++)
			buf[dest + r * pitch + c] = (unsigned char)row[c];
	}
}

int main(void)
{
	static unsigned char base[0x8000], a[0x8000], b[0x8000], m[0x8000];
	static const int low[8] = {-12,-11,-10,-10,-9,-9,-9,-9};
	static const int qual[8] = {0, 1, 2, 7, 15, 31, 64, 128};
	int scale[64], q, pattern, i, cases = 0, fail = 0, d;
	for (q = 0; q < 64; q++)
		Vp6FilterEdgeTagTable[q] = q < 8 ? low[q] : q < 16 ? -6 : q < 24 ? -4 : q < 40 ? -2 : 0;
	for (q = 0; q < 64; q++) {
		for (pattern = 0; pattern < 16; pattern++) {
			int pitch = (10 + ((q * 5 + pattern) % 23)) * (pattern & 8 ? -1 : 1);
			int origin = 0x4000, quality = qual[(q + pattern) % 8];
			int dests[5];
			dests[0] = 0x1000; dests[1] = origin; dests[2] = origin + 1; dests[3] = origin - 1; dests[4] = origin + pitch - 3;
			for (i = 0; i < 64; i++) scale[i] = quality + (int)(rnd() % 3);
			for (i = 0; i < (int)sizeof base; i++) {
				switch (pattern & 3) {
				case 0: base[i] = (unsigned char)(q * 4); break;
				case 1: base[i] = (unsigned char)(((i + i / (pitch < 0 ? -pitch : pitch)) & 1) ? 255 : 0); break;
				case 2: base[i] = (unsigned char)(i * 3 + i / 17); break;
				default: base[i] = (unsigned char)rnd(); break;
				}
			}
			for (d = 0; d < 5; d++) {
				memcpy(a, base, sizeof base); memcpy(b, base, sizeof base); memcpy(m, base, sizeof base);
				Bank_Strong(0, a + origin, a + dests[d], pitch, q, scale);
				Cand_Strong(0, b + origin, b + dests[d], pitch, q, scale);
				model(m, origin, dests[d], pitch, q, scale);
				cases++;
				if (memcmp(a, b, sizeof a) || memcmp(a, m, sizeof a)) { fail++; if (fail < 5) printf("mismatch q=%d pattern=%d dest=%d\n", q, pattern, d); }
			}
		}
	}
	printf("cases=%d failures=%d\n", cases, fail);
	return fail != 0;
}
