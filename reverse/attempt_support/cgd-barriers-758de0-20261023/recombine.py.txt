from pathlib import Path
from itertools import product
W=Path(__file__).resolve().parent
BASE=(W/'base.cpp').read_text()
EDGE=(W/'edge.cpp').read_text()

def source(edge,sites,model,cc,bits):
 s=EDGE if edge else BASE
 kind='_ReadWriteBarrier'
 decl='extern "C" void _ReadWriteBarrier(void);\n#pragma intrinsic(_ReadWriteBarrier)\n'
 s=s.replace('struct ContactVector {',decl+'struct ContactVector {')
 helper='\n__forceinline const ContactVector &'+('__fastcall ' if cc else '')+'pointBits(const ContactVector &v) { return *(const ContactVector *)(unsigned)&v; }\n'
 helper+='__forceinline const ContactVector &'+('__fastcall ' if cc else '')+'nodeBits(ContactNode *n) { unsigned u=(unsigned)&n->point;return *(const ContactVector *)u; }\n'
 helper+='__forceinline const ContactVector &bv(const ContactVector &v) { _ReadWriteBarrier();return v; }\n'
 helper+='__forceinline ContactSource *bs(ContactSource *p) { _ReadWriteBarrier();return p; }\n'
 helper+='__forceinline void *bi(void *p) { _ReadWriteBarrier();return p; }\n'
 s=s.replace('class BfmeThingCGD {',helper+'class BfmeThingCGD {')
 h,t=s.split('void BfmeThingCGD::bfmeTwoCGD() {',1)
 def cast(expr,which):
  if not sites>>which&1:return expr
  return 'pointBits('+expr+')' if model else 'nodeBits(node)'
 def bar(expr,on):return 'bv('+expr+')' if on else expr
 test_point=bar(cast('node->point',0),bits&1)
 test_normal=bar('node->normal',bits&2)
 t=t.replace('node->second->source->prepare(node->point,node->normal)',
  'node->second->source->prepare('+test_point+','+test_normal+')')
 old_point='*p' if edge else 'node->point'
 first_point=bar(cast(old_point,1),bits&4)
 first_normal=bar('node->normal',bits&8)
 old='bridge(node->second,'+old_point+',node->normal)'
 assert old in t
 call='bridge(node->second,'+first_point+','+first_normal+')'
 if bits&32:call='bi('+call+')'
 t=t.replace(old,call)
 t=t.replace('altPrepare(node->first->source,node->point,',
             'altPrepare(node->first->source,'+cast('node->point',2)+',')
 if bits&16:h=h.replace('return s->source->prepare(p,n);','return bs(s->source)->prepare(p,n);')
 # Helpers used by bridge must precede its definition.
 start=h.index('\n__forceinline const ContactVector &')
 end=h.index('class BfmeThingCGD {')
 helpers=h[start:end]
 h=h[:start]+h[end:]
 h=h.replace('__forceinline void *__cdecl bridge(',helpers+'__forceinline void *__cdecl bridge(')
 return h+'void BfmeThingCGD::bfmeTwoCGD() {'+t

if __name__=='__main__':
 out=W/'recombine';out.mkdir(exist_ok=True)
 args=list(product(range(2),range(8),range(2),range(2),range(64)))
 for idx,a in enumerate(args):(out/f'v{idx:04}.cpp').write_text(source(*a))
 print('generated',len(args))
