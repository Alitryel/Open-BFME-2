import sys,json,itertools
from pathlib import Path
from concurrent.futures import ThreadPoolExecutor
sys.path.insert(0,'build/fxinit-seat22');import search as h
h.HERE=Path('build/fxinit-phase22').resolve()
old='memcpy(indices,polys,mesh->Get_Polygon_Count()*sizeof(TriIndex));'
forms=[old,'(memcpy)(indices,polys,mesh->Get_Polygon_Count()*sizeof(TriIndex));','void *(__cdecl *copyIndices)(void *,const void *,unsigned)=memcpy;copyIndices(indices,polys,mesh->Get_Polygon_Count()*sizeof(TriIndex));','(*memcpy)(indices,polys,mesh->Get_Polygon_Count()*sizeof(TriIndex));']
def variants():
 for mode,form,change in itertools.product(range(3),range(4),range(2)):
  s=h.BASE.replace(old,forms[form]);helper=''
  if mode==0:s=s.replace('#include <string.h>','#include <string.h>\n#pragma intrinsic(memcpy)')
  else:
   helper='#pragma intrinsic(memcpy)\nstatic __forceinline void CopyVector12(void *dst,const void *src){memcpy(dst,src,12);}\n#pragma function(memcpy)\n'
   if mode==2:helper=helper.replace('#pragma function(memcpy)','')
  if change:
   expr='CopyVector12'if mode else'memcpy'
   args='vertices+desc.Get_Location_Offset(1),positions[1]+i'+(',12'if not mode else'')
   s=s.replace(h.P,'if(positions[1]) '+expr+'('+args+');')
  s=s.replace('bool FXShaderGeometry::Init(',helper+'bool FXShaderGeometry::Init(')
  yield f'intrinprobe_{mode}{form}{change}',s
with ThreadPoolExecutor(max_workers=8)as p:rs=list(p.map(h.run_one,variants()))
(h.HERE/'intrinprobe-results.json').write_text(json.dumps(rs,indent=2));print(json.dumps(rs,indent=2))
