import sys,itertools,json,time
from pathlib import Path
from concurrent.futures import ThreadPoolExecutor,as_completed
sys.path.insert(0,'build/fxinit-seat22');import search as h
h.HERE=Path('build/fxinit-edges22').resolve()
PASS='*(unsigned *)(vertices+desc.Get_Basis_End())=pass;'
ONE='*(int *)(vertices+desc.Get_Basis_End())=remap[links[0][i]];'
LOOP='   for(unsigned i=0;i<(unsigned)mesh->Get_Vertex_Count();++i) {'
def variants():
 for ty,loc,guard,form,order,mask in itertools.product(range(4),range(2),range(8),range(3),range(2),[2,8,10]):
  p,n='positions[1]','normals[1]';decl=''
  if ty:
   typ=['','bool','unsigned','const Vector3 *'][ty]
   vs=[p+'!=0',n+'!=0']if ty!=3 else[p,n]
   decl=f'{typ} hasP={vs[0]}; {typ} hasN={vs[1]};'
   p,n='hasP','hasN'
  cond=[f'!{p} && !{n}',f'!({p}||{n})',f'!((unsigned){p}|(unsigned){n})',f'{p}=={n} && !{p}',f'{p}=={n} && !{n}',f'({p} ? false : !{n})',f'(!{p} ? !{n} : false)',f'(!{p} & !{n})'][guard]
  # Pointer equal-null and boolean equal-false variants are the same condition.
  entry='if('+cond+') goto CopyBoth;goto FinishSkin;'
  pn=h.P.replace('if(positions[1])','if('+p+')')+'\n'+h.N.replace('if(normals[1])','if('+n+')')
  if order:pn=h.N.replace('if(normals[1])','if('+n+')')+'\n'+h.P.replace('if(positions[1])','if('+p+')')
  if form==1:pn='if('+p+'||'+n+') {'+pn+'}'
  if form==2:pn='if('+cond+') goto FinishSkin;'+pn
  arm='if(m_Passes>1) {'+PASS+'goto FinishSkin;}\nelse if(m_BonesPerVertex>0 && links[0]) {\nif(m_BonesPerVertex==1) {'+ONE+'goto FinishSkin;}\nelse if(m_BonesPerVertex==2) {\n'+h.R+'\n'+h.W+'\nCopyBoth:\n'+pn+'\n}\nelse {'+(entry if mask&8 else 'goto FinishSkin;')+'}\n}\nelse {'+(entry if mask&2 else 'goto FinishSkin;')+'}\nFinishSkin:;\n'
  if not loc:arm=decl+arm
  s=h.BASE[:h.START]+arm+h.BASE[h.END:]
  if loc:s=s.replace(LOOP,LOOP+decl)
  yield f'nulljoin_{ty}{loc}{guard}{form}{order}{mask:02}',s
if __name__=='__main__':
 t=time.time();items=list(variants());r=[]
 with ThreadPoolExecutor(max_workers=8)as pool:
  for f in as_completed([pool.submit(h.run_one,x)for x in items]):
   r.append(f.result())
   if len(r)%512==0:print('tested',len(r),'best',min((x for x in r if 'nd'in x),key=lambda x:(x['nd'],x['cost'])),flush=True)
 r.sort(key=lambda x:(x.get('nd',99999),x.get('cost',99999)))
 (h.HERE/'nulljoin-results.json').write_text(json.dumps(r,indent=2))
 print(json.dumps({'count':len(r),'valid':sum('nd'in x and not x.get('unresolved')for x in r),'seconds':time.time()-t,'best':r[:4]},indent=2))
