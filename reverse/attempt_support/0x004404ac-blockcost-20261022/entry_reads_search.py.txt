from search import *
import re,itertools

def source(flags):
 mask,kind=flags
 end=BODY.index('\n\tbool observer')
 prefix=BODY[:end];tail=BODY[end:]
 if kind==1:
  # The volatile load reads the existing parameter home only once.
  prefix=prefix.replace('slot == -1','(int)(unsigned int)slot == -1')
 if kind==2:
  prefix=prefix.replace('slot == -1','(slot ^ -1) == 0')
 if kind==3:
  prefix=prefix.replace('slot == -1','(~slot) == 0')
 j=0
 def read(m):
  nonlocal j
  # Do not qualify the clamp's write.
  if re.match(r'\s*=(?!=)',prefix[m.end():]):return 'slot'
  i=j;j+=1
  return '((const volatile int &)slot)'if mask&(1<<i)else 'slot'
 prefix=re.sub(r'\bslot\b',read,prefix)
 return HEAD+MARK+prefix+tail
if __name__=='__main__':run([('entryread-%03d'%n,source(f))for n,f in enumerate(itertools.product(range(16),range(4)))],'entry_reads')
