import json,itertools,time
from pathlib import Path
j=json.loads(Path('reverse/attempt_support/0x0004826c-sol6-allocator.json').read_text())
want={11:-0x24,12:-0x38,40:-0x18,37:-0x14,38:-0x24,39:-0x1c};weights={11:3,12:2,40:2,37:2,38:2,39:2}
def pack(rows,c):
 g=[]
 for r in rows:
  i,n=r['index'],r['size']
  for q in g[::-1]:
   if n<=2*q[1] and i not in q[3] and not c[i]&q[0]:
    q[0].add(i);q[1]=max(n,q[1]);q[2]+=r['weight'];q[3]|=c[i];break
  else:g.append([{i},n,r['weight'],c[i].copy()])
 return g
def score(g):
 g.sort(key=lambda q:-(1000*q[2]//q[1]));ties=[list(z) for _,z in itertools.groupby(g,key=lambda q:1000*q[2]//q[1])];best=99;hh=None
 for blocks in itertools.product(*(list(itertools.permutations(q)) for q in ties)):
  off=12;homes={}
  for block in blocks:
   for q in block:off+=q[1];homes.update({i:-off for i in q[0]})
  s=sum(weights[i] for i in want if homes[i]!=want[i])
  if s<best:best=s;hh={i:homes[i] for i in want}
 return best,hh
found=[];summaries=[]
for tag in ['old','base']:
 d=j[tag];c={int(k):set(v) for k,v in d['conflicts'].items()};cache={};counts={};best=99
 for row in range(77):
  for bit in range(77):
   if row==bit:continue
   cc=c.copy();cc[row]=c[row]^{bit};gg=pack(d['rows'],cc);sig=tuple((tuple(sorted(q[0])),q[1],q[2]) for q in gg)
   if sig not in cache:cache[sig]=score(gg)
   s,h=cache[sig];counts[s]=counts.get(s,0)+1
   if s<best:best=s;print(tag,'best',s,'toggle',row,bit,'remove',bit in c[row],h,flush=True)
   if s==0:found.append(dict(tag=tag,row=row,bit=bit,remove=bit in c[row],homes=h))
 summaries.append(dict(tag=tag,counts=counts,groupings=len(cache)))
p=Path(__file__).resolve().parent/'edge-results.json';p.write_text(json.dumps(dict(summaries=summaries,found=found),indent=2));print('summary',summaries,'zeros',len(found),flush=True)
