from pathlib import Path
import subprocess,sys,json
root=Path(__file__).resolve().parents[1];out=root/'build/foundation-bitset-oct20';tests={}
for profile in ['checked-Ob1-100','checked-Ob2-100']:
    base=(out/(profile+'.cpp')).read_text()
    for mode in ['outer','both','inner']:
        s=base
        if mode in ('outer','both'):s=s.replace('inline Object *GameLogic::findObjectByID','__declspec(noinline) inline Object *GameLogic::findObjectByID',1)
        if mode in ('inner','both'):s=s.replace('inline ObjectIdNode *ObjectIdMap::find','__declspec(noinline) inline ObjectIdNode *ObjectIdMap::find',1)
        tests[profile+'-'+mode]=s
name='?construct@FoundationAIUpdate@@UAEPAVObject@@PBVThingTemplate@@PBUCoord3D@@MPAVPlayer@@H_N@Z';results=[]
for key,s in tests.items():
    p=out/(key+'.cpp');p.write_bytes(s.encode())
    r=subprocess.run([sys.executable,'tools/explain_mismatch.py',name,'--rva','0x0045569D','--size','1174','--source',p.relative_to(root).as_posix()],cwd=root,capture_output=True,text=True,encoding='utf8',errors='replace',timeout=240)
    log=r.stdout+r.stderr;p.with_suffix('.log').write_bytes(log.encode())
    d={'key':key,'exit':r.returncode,'exact':'OK: bytes match' in log,'summary':[x for x in log.splitlines() if x.startswith(('compiled symbol size:','classification:','unresolved:','first diff:'))]}
    if r.returncode:d['error']=log[-700:]
    results.append(d);(out/'boundary-results.json').write_text(json.dumps(results,indent=2));print(json.dumps(d),flush=True)
