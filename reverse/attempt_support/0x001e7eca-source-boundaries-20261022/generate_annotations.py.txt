from search import *
import itertools
s=(W/'base.cpp').read_text();vs=[]
for ptr,attr,ret,normal,extra in itertools.product(range(4),range(4),range(2),range(4),range(2)):
 o=s
 o=o.replace('extern TerrainLogic*TheTerrainLogic;',['extern TerrainLogic*TheTerrainLogic;','extern TerrainLogic*volatile TheTerrainLogic;','extern TerrainLogic*const TheTerrainLogic;','extern TerrainLogic*volatile const TheTerrainLogic;'][ptr])
 at=['','__declspec(nothrow) ','__declspec(noinline) ','__declspec(nothrow) __declspec(noinline) '][attr]
 rt=['float','double'][ret]
 nt=['Coord3D*','Coord3D __unaligned *','Coord3D * __restrict','Coord3D * __w64'][normal]
 o=o.replace('virtual float ground(float,float,Coord3D*);','virtual '+at+rt+' ground(float,float,'+nt+');')
 o=o.replace('virtual float layer(float,float,int,Coord3D*,bool);','virtual '+at+rt+' layer(float,float,int,'+nt+',bool);')
 if extra:
  o=o.replace('virtual bool underwater(', 'virtual '+at+'unsigned char underwater(')
  o=o.replace(' int rva0028B511()const;', ' '+at+'int rva0028B511()const;')
 vs.append((f'annotations_{ptr}{attr}{ret}{normal}{extra}',o))
run(vs,'annotations256')
