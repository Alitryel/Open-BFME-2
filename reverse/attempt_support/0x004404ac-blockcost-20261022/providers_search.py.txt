def extract(text,mark):
 s=text.index(mark);b=text.index('{',s);level=1;e=b+1
 while level:
  level+=(text[e]=='{')-(text[e]=='}');e+=1
 return text[s:e]
win=(ROOT/'Code/GameEngine/Source/GameClient/GUI/GameWindowHide.cpp').read_text()
manager=win[win.index('typedef int Int;'):win.index('class GameWindow\n{')]
