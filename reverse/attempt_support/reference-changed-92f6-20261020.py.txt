from pathlib import Path
import sys,subprocess,json,hashlib,concurrent.futures,re
root=Path(__file__).resolve().parents[1];sys.path.insert(0,str(root/'tools'));import build
out=root/'build/reference-changed-92f6-oct20';out.mkdir(exist_ok=True)
revision='92f6bbe8e1dcdb46b5d27da0674313efc822b790'
assert subprocess.check_output(['git','-C',str(root/'reference/open-bfme-1'),'rev-parse','HEAD'],text=True).strip()==revision
paths=(root/'build/reference-changed-sources-92f6-oct20.txt').read_text(encoding='utf-8-sig').splitlines()
tasks=[];skipped=[]
for line in paths:
    line=line.strip()
    if not line or not line.endswith('.cpp'):continue
    donor=root/'reference/open-bfme-1'/line
    assert not any(x in donor.as_posix().lower() for x in ('vp6','on2','winamp','ffdshow','mfnode','bink','theora','video'))
    text=donor.read_text()
    if re.search(r'__declspec\s*\(\s*naked\s*\)|__emit',text):
        skipped.append({'source':line,'reason':'naked/emit source is not a clean C++ donor'});continue
    flags=build.source_extra_flags(donor)
    directives=[l for l in text.splitlines() if l.startswith('// compiler:')]
    for profile,extra in [('donor',[]),('target-O1',['-O1','-G7','-arch:SSE']),('target-O2',['-O2','-G7','-arch:SSE'])]:
        selected=flags+extra
        missing=[f for f in selected if f.startswith(('-I','/I')) and not (root/f[2:]).is_dir()]
        if missing:skipped.append({'source':line,'profile':profile,'reason':'missing intended include roots','roots':missing});continue
        def render(f):return f[:2]+'"'+f[2:]+'"' if f.startswith(('-I','/I')) and ' ' in f else f
        wrapper='// cl: '+' '.join(map(render,selected))+'\n'+'\n'.join(directives)+'\n'
        if build.source_needs_stlport(donor):wrapper+='// stlport\n'
        wrapper+='#include "'+donor.as_posix()+'"\n'
        trial=out/(profile+'_'+donor.name);trial.write_bytes(wrapper.encode())
        assert build.source_extra_flags(trial)==selected
        tasks.append({'donor':line,'sha256':hashlib.sha256(donor.read_bytes()).hexdigest(),'revision':revision,'profile':profile,'flags':selected,'trial':trial.relative_to(root).as_posix()})
(out/'inputs.json').write_bytes((json.dumps(tasks,indent=2)+'\n').encode())
(out/'skipped.json').write_bytes((json.dumps(skipped,indent=2)+'\n').encode())
def run(t):
    p=subprocess.run([sys.executable,'tools/place_bodies.py','--min','8',t['trial']],cwd=root,capture_output=True,text=True,encoding='utf8',errors='replace',timeout=300)
    output=p.stdout+p.stderr;(root/t['trial']).with_suffix('.log').write_bytes(output.encode())
    return {**t,'compile_failed':p.returncode!=0 or 'COMPILE FAILED' in output,'errors':[s for s in output.splitlines() if 'error ' in s][:4],'leads':[s for s in output.splitlines() if s.startswith(('BODY\t','PIN\t'))]}
results=[]
with concurrent.futures.ThreadPoolExecutor(max_workers=3) as pool:
    for r in pool.map(run,tasks):
        results.append(r);(out/'results.json').write_bytes((json.dumps(results,indent=2)+'\n').encode())
        print(Path(r['donor']).name,r['profile'],'failed',r['compile_failed'],'leads',r['leads'],'errors',r['errors'][:1],flush=True)
print('searched',len(results),'profiles; skipped',skipped,flush=True)
