from pathlib import Path
import json, sys
sys.path.insert(0, 'tools')
from variant_search import Scorer
from zh_sweep import carve

out = Path('build/local-vm/cookie-profiles')
scorer = Scorer('@__security_check_cookie@4', 0x629bab, 14, out.resolve())
results = []
for profile in ['', 'vc71-ddk-4035', 'vc71-sp1']:
    for flags in ['/O2 /GS-', '/O1 /GS-', '/O2 /Og- /GS-',
                  '/O1 /Og- /GS-', '/O2 /G5 /GS-', '/O2 /G7 /GS-']:
        name = (profile or 'rtm') + ' ' + flags
        source = ('// compiler: ' + profile + '\n' if profile else '')
        source += '// cl: ' + flags + '\n'
        source += '''extern "C" unsigned int __security_cookie;
extern "C" void report_failure();
extern "C" void __fastcall __security_check_cookie(unsigned int cookie) {
    if (cookie == __security_cookie) return;
    report_failure();
}
'''
        result = scorer(source)
        obj = out / ('c%d.obj' % scorer.n)
        extent = next((len(b) for n,b,_ in carve(obj.read_bytes())
                       if n == '@__security_check_cookie@4'), None) if obj.exists() else None
        record = dict(trial=name, extent=extent, **{k:v for k,v in result.items()
                      if k not in ('compiled', 'target')})
        record['compiled'] = result.get('compiled', b'').hex()
        results.append(record)
        print(record, flush=True)
        (out / 'results.json').write_text(json.dumps(results, indent=2)+'\n')
        if result['exact'] and extent == 14:
            (out / 'exact.cpp').write_text(source)
            raise SystemExit(0)
