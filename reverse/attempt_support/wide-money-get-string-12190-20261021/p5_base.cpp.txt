@FREE@
enum { LEAKS = 0 };
template <class T> class allocator { public: allocator() {} @ALLOCDTOR@ void deallocate(T *p, unsigned n) const { @DEALLOC@ } };
template <class P, class T, class A> class proxy : public A { public: P _M_data; @PROXYDTOR@ };
template <class C, class A>
class sbase
{
public:
	void dealloc_block() { if (_M_start != 0) _M_eos.deallocate(_M_start, _M_eos._M_data - _M_start); }
	~sbase() { dealloc_block(); }
	C *_M_start;
	C *_M_finish;
	proxy<C *, C, A> _M_eos;
};
inline void _Destroy(unsigned short *, unsigned short *) {}
inline void _Destroy(char *, char *) {}
template <class C>
class S : public sbase<C, allocator<C> >
{
public:
	S(int);
	S(const S &);
	~S() { _Destroy(this->_M_start, this->_M_finish + 1); }
	int size() const { return (int)(this->_M_finish - this->_M_start); }
};
@INST@
typedef S<@CT@> W;
W make(int);
int peek(int);
int f(bool intl, int k, int *err)
{
	W a = intl ? make(1) : make(2);
	W b = intl ? make(3) : make(4);
	W c(5);
	for (int i = 0; i < 4; ++i) {
		switch (peek(i)) {
		case 0:
			if (a.size()) { *err = 1; return k; }
			break;
		case 1:
			{
				W d = intl ? make(6) : make(7);
				k += d.size();
			}
			break;
		case 2:
			k += b.size();
			break;
		case 3:
			if (b.size()) { *err = 2; return k + 1; }
			break;
		}
	}
	if (c.size()) *err = 3;
	return k;
}
