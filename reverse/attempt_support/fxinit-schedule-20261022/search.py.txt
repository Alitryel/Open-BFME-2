def insns(blob,relocs=()):
    md=capstone.Cs(capstone.CS_ARCH_X86,capstone.CS_MODE_32)
    relocation={o for o,t,n in relocs}
    out=[]
    for x in md.disasm(blob,0):
        ops=x.op_str
        if x.mnemonic.startswith('j') or x.mnemonic=='call': ops='TARGET' if not '[' in ops else ops
        if any(x.address<=r<x.address+x.size for r in relocation):
            ops=re.sub(r'0x[0-9a-f]+|\b0\b','ADDR',ops)
        else:
            ops=re.sub(r'0x[0-9a-f]{6,}', 'ADDR',ops)
        ops=ops.replace('fs:[ADDR]','fs:[0]')
        out.append((x.address,x.mnemonic,ops,x.size))
    return out
TINS=insns(TARGET)

def run_one(item):
    label,src=item
    path=HERE/'trials'/f'{label}.cpp';path.parent.mkdir(exist_ok=True)
    path.write_text(src)
    obj=path.with_suffix('.obj')
    command,env=build.compiler_command(path,obj)
    p=subprocess.run(command,cwd=ROOT,env=env,capture_output=True,text=True)
    if p.returncode: return {'id':label,'error':(p.stdout+p.stderr)[-1800:]}
    raw,rels=build.read_object_symbol_bytes(obj,NAME,code_only=True)
    row={'name':NAME,'source':str(path.relative_to(ROOT)),'target_rva':hex(RVA),'target_size':str(SIZE),'notes':''}
    try:
        patch=build.compile_function(row,MAP,obj)
        full=patch['bytes']; unr=patch['unresolved']
        nd=sum(a!=b for a,b in zip(full,TARGET))+abs(len(full)-SIZE)
    except BaseException as e: nd=9999; unr=[str(e)]
    ours=insns(raw,rels)
    seq=difflib.SequenceMatcher(a=[x[1] for x in TINS],b=[x[1] for x in ours],autojunk=False)
    regions=0;struct=0;ops=0;delta=[]
    for tag,a,b,c,d in seq.get_opcodes():
        if tag!='equal':
            regions+=1;struct+=max(b-a,d-c)
            delta.append({'region':tag,'retail':[TINS[a:b]],'ours':[ours[c:d]]})
        else:
            for x,y in zip(TINS[a:b],ours[c:d]):
                if x[2]!=y[2]: ops+=1;delta.append({'retail':x,'ours':y})
    result={'id':label,'size':len(raw),'nd':nd,'regions':regions,'struct':struct,'ops':ops,'cost':struct+ops,'unresolved':unr}
    path.with_suffix('.json').write_text(json.dumps({'result':result,'diff':delta},indent=2))
    path.with_suffix('.dis').write_text('\n'.join(f'{a:04x} {m} {o}' for a,m,o,s in ours))
    return result

            pre='short *boneMap=remap.begin();\n' if pointer else ''
            base='boneMap' if pointer else 'remap'
            if access==0:
                pre+='unsigned l0=links[0][i];\nunsigned l1=links[1][i];\n'
                a,b=f'{base}[l0]',f'{base}[l1]'
            elif access==1:
                pre+='unsigned l1=links[1][i];\nunsigned l0=links[0][i];\n'
                a,b=f'{base}[l0]',f'{base}[l1]'
            else:a,b=f'{base}[links[0][i]]',f'{base}[links[1][i]]'
            ty=['int','const short &'][rtype]
            steps=[f'{ty} r0={a};',f'{ty} r1={b};','*(int *)(vertices+desc.Get_Basis_End())=r0+(r1<<8);',W]
            if sequence==1:steps[0],steps[1]=steps[1],steps[0]
            if sequence==2:steps[2],steps[3]=steps[3],steps[2]
            if sequence==3:steps=[steps[3],steps[0],steps[1],steps[2]]
            steps.insert(loc,br)
            body='{\n'+pre+'\n'.join(steps)+'\n'+P+'\n'+N+'\n}'
            arm=ARM[:ARM.index('      '+R)]+body+'\n     }\n    }\n'
            src=BASE[:START]+arm+BASE[END:]
            src=src.replace('bool FXShaderGeometry::Init(',dec+'bool FXShaderGeometry::Init(')
            yield f'bar_{access}{sequence}{bar}{loc}{rtype}{pointer}',src
