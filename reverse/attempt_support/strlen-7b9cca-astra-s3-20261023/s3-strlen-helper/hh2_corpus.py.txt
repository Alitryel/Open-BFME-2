# -*- coding: utf-8 -*-
# astra-s3 helper seat round 2: volatile if-assign guard (in-order, unfusable)
# crossed with latch predicates that may dodge MSVC's guard/latch unifier while
# still lowering to test dl,dl + jne.

FN = 'int rva007B9CCAStrLen(const char *s) { '
RET = ' return (int)(p - s - 1); }'

VARIANTS = []
def add(tag, body, flags='/O1 /G6 /MD', profile='', ext='cpp'):
    VARIANTS.append((tag, flags, profile, ext, body))

GVOL  = 'if ((c = *(volatile const char *)p++))'
GVOLN = 'if ((c = *(volatile const char *)p++) != 0)'
GPLAIN = 'if ((c = *p++))'

# LATCH forms: must be nonzero iff c != 0, hoping to dodge the unifier yet
# still fold late to "test dl,dl; jne".
LATCHES = [
 ('lbb1',    'c && 1'),
 ('lbbT',    'c && true'),
 ('lneq0eq', '(c == 0) == 0'),
 ('lneq0ne', '0 == (c == 0)'),
 ('lter01',  '(c == 0) ? 0 : 1'),
 ('lter10',  '(c == 0) ? 1 : 0'),   # wrong polarity control (semantics: c==0 -> 1) - diagnostic only
 ('lternz',  'c ? 1 : 0'),
 ('lboolne', '(bool)(c != 0)'),
 ('lnot',    '!(c == 0)'),
 ('lneqne',  '(c != 0) != 0'),
 ('lor0',    'c || 0'),
 ('lxor1',   '1 ^ (c == 0)'),
 ('landc',   'c && c'),             # control: expected to unify -> rotate
 ('lnz',     'nz(c)'),             # control: helper seen through -> rotate
 ('lucz',    '(unsigned char)c != 0'),
 ('lcmpeq',  '(c == 0) ? false : true'),
 ('lcompl',  '(c && 1) != 0'),
]
NZ = 'static inline int nz(char c) { return c != 0; }\n'

for lt, lx in LATCHES:
    add('v2_%s' % lt, FN + 'const char *p = s; char c; ' + GVOL + ' { do { c = *p++; } while (' + lx + '); }' + RET)

for lt, lx in LATCHES[:12]:
    add('v2n_%s' % lt, FN + 'const char *p = s; char c; ' + GVOLN + ' { do { c = *p++; } while (' + lx + '); }' + RET)

# plain (fusable) guard with new latches: does fusion preempt everything anyway?
for lt, lx in [('lbb1', 'c && 1'), ('lneq0eq', '(c == 0) == 0'), ('lter01', '(c == 0) ? 0 : 1')]:
    add('v2p_%s' % lt, FN + 'const char *p = s; char c; ' + GPLAIN + ' { do { c = *p++; } while (' + lx + '); }' + RET)

# double-test guard (second use of the loaded value blocks fusion if CSE is late)
DGUARDS = [
 ('dneq',   '(c = *p++) && (c != 0)'),
 ('dneq2',  '(c = *p++) && c'),
 ('dand1',  '(c = *p++) && (c && 1)'),
 ('dbb',    '(c = *p++) && !!c'),
 ('dorz',   '((c = *p++) != 0) && (c != 0)'),
 ('dnz',    '(c = *p++) && nz(c)'),
]
for i, (dt, dx) in enumerate(DGUARDS):
    add('v2d_%s' % dt, NZ + FN + 'const char *p = s; char c; if (' + dx + ') { do { c = *p++; } while (c); }' + RET)
    if i < 3:
        add('v2d_%s_lbb1' % dt, NZ + FN + 'const char *p = s; char c; if (' + dx + ') { do { c = *p++; } while (c && 1); }' + RET)

# separate-statement guard (parent's cond12 shape) with the new latch spellings:
# proves which latches make the unifier bail at all (expected test-before-lea sink).
for lt, lx in [('lbb1', 'c && 1'), ('lneq0eq', '(c == 0) == 0'), ('lter01', '(c == 0) ? 0 : 1'), ('lternz', 'c ? 1 : 0'), ('lboolne', '(bool)(c != 0)')]:
    add('v2s_%s' % lt, FN + 'const char *p = s; char c = *p++; if (c) { do { c = *p++; } while (' + lx + '); }' + RET)

# for-loop with the new latch spellings in the condition
for lt, lx in [('lbb1', 'c && 1'), ('lneq0eq', '(c == 0) == 0'), ('lter01', '(c == 0) ? 0 : 1')]:
    add('v2f_%s' % lt, FN + 'const char *p; char c = *s; for (p = s + 1; ' + lx + '; ++p) { c = *p; }' + RET)
    add('v2fv_%s' % lt, FN + 'const char *p; char c = *(volatile const char *)s; for (p = s + 1; ' + lx + '; ++p) { c = *p; }' + RET)

# walker with asymmetric methods and the new latch spellings
WK_T0T1 = ('struct Walker {\n    const char *cur;\n    const char *base;\n    Walker(const char *s) : cur(s), base(s) {}\n'
           '    char take0() { char c = *(volatile const char *)cur; cur = cur + 1; return c; }\n'
           '    char take1() { char c = *cur; cur = cur + 1; return c; }\n};\n')
for lt, lx in [('lbb1', 'c && 1'), ('lneq0eq', '(c == 0) == 0'), ('lter01', '(c == 0) ? 0 : 1')]:
    add('v2w_t0_%s' % lt,
        WK_T0T1 + 'int rva007B9CCAStrLen(const char *s) { Walker w(s); char c; if ((c = w.take0())) { do { c = w.take1(); } while (' + lx + '); } return (int)(w.cur - w.base - 1); }')
add('v2w_cc_bb1',
    WK_T0T1 + 'int rva007B9CCAStrLen(const char *s) { Walker w(s); if (w.take0()) { do {} while (w.take1() && 1); } return (int)(w.cur - w.base - 1); }')
add('v2w_cc_neq',
    WK_T0T1 + 'int rva007B9CCAStrLen(const char *s) { Walker w(s); if (w.take0()) { do {} while ((w.take1() != 0)); } return (int)(w.cur - w.base - 1); }')

# controls re-run for consistency
add('ctl_gvol_lwhile', FN + 'const char *p = s; char c; ' + GVOL + ' { do { c = *p++; } while (c); }' + RET)
add('ctl_sp_vol_uc', FN + 'const char *p = s; char c = *(volatile const char *)p; ++p; if (c) { do { c = *p++; } while ((unsigned char)c > 0); }' + RET)
