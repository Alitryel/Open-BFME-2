from pathlib import Path
from itertools import product
W=Path(__file__).resolve().parent;out=W/'coherent';out.mkdir(exist_ok=True)
b=(W/'control.cpp').read_text()
for idx,(prepcc,testcc,respcc,infword,copy,ctorcc,bridge,order,sites,pointerfield,th) in enumerate(product(range(2),repeat=11)):
 s=b;cc='__fastcall ' if ctorcc else ''
 pv='&p' if pointerfield else 'p';pt='const ContactVector *' if pointerfield else 'const ContactVector &'
 defs='struct PArg {'+pt+' p; '+cc+'PArg(const ContactVector &p):p('+pv+'){} };\n'
 defs+='struct PPair {PArg point,normal; '+cc+'PPair(const ContactVector &'+('n,const ContactVector &p' if order else 'p,const ContactVector &n')+'):point(p),normal(n){}'
 if copy:defs+=' '+cc+'PPair(const PPair &p):point(p.point),normal(p.normal){}'
 defs+='};\n'
 it='unsigned' if infword else 'void *';iv='(unsigned)p' if infword else 'p'
 defs+='struct IArg { '+it+' info; '+cc+'IArg(void *p):info('+iv+'){}'
 if copy:defs+=' '+cc+'IArg(const IArg &p):info(p.info){}'
 defs+='};\n'
 pp='PPair(n,p)' if order else 'PPair(p,n)'
 s=s.replace('struct ContactSource {',defs+'struct ContactSource {')
 s=s.replace('virtual void *prepare(const ContactVector &point, const ContactVector &normal)=0;', 'virtual void *'+('__fastcall ' if prepcc else '')+'preparePair(PPair value)'+(' throw()' if th else '')+'=0;\n __forceinline void *prepare(const ContactVector &p,const ContactVector &n){return preparePair('+pp+');}')
 s=s.replace('virtual bool test(void *info)=0;', 'virtual bool '+('__fastcall ' if testcc else '')+'test(IArg info)'+(' throw()' if th else '')+'=0;')
 s=s.replace('virtual void respond(void *info)=0;', 'virtual void '+('__fastcall ' if respcc else '')+'respond(IArg info)'+(' throw()' if th else '')+'=0;')
 helper='__forceinline void *bridge(ContactData *s,const ContactVector &p,const ContactVector &n){return s->source->prepare(p,n);}\n'
 s=s.replace('class BfmeThingCGD {',helper+'class BfmeThingCGD {')
 if bridge:s=s.replace('node->first->source->respond(node->second->source->prepare(node->point,node->normal));','node->first->source->respond(bridge(node->second,node->point,node->normal));')
 if sites:
  s=s.replace('node->second->source->prepare(node->point,node->normal)','node->second->source->preparePair('+('PPair(node->normal,node->point)' if order else 'PPair(node->point,node->normal)')+')')
  s=s.replace('node->first->source->prepare(node->point,ContactVector(node->normal).negate())','node->first->source->preparePair('+('PPair(ContactVector(node->normal).negate(),node->point)' if order else 'PPair(node->point,ContactVector(node->normal).negate())')+')')
 (out/('v%04d.cpp'%idx)).write_text(s)
print('generated',idx+1)
