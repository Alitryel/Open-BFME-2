"""Read-only observations of stock c2 scheduling graphs for FX Init."""
import sys,json,time,collections
from pathlib import Path
sys.path.insert(0,str(Path(__file__).parent))
from wdbg import Dbg
W=Path('build/fxinit-seat22')
src=sys.argv[1];tag=sys.argv[2]
d=Dbg(src,str(W/(tag+'.obj')),str(W/(tag+'.debug.log')))
graphs=[];functions=[];on=False
try:
 d.prompt();d.prompt('set $CanDeferOnBPByAddr = 1');d.prompt('break *0x10760539')
 for k in range(10000):
  o=d.prompt('cont',60)
  if 'terminated' in o or not o.strip():break
  ip=d.reg('eip')
  if ip==0x10760539:
   p=d.reg('eax');f=d.words(p,1)[0];fn=d.string(d.words(f+4,1)[0]);functions.append(fn)
   if on:break
   if '?Init@FXShaderGeometry' in fn:
    on=True;print('Init entered',flush=True);d.prompt('break *0x107b5338')
   continue
  if ip==0x107b5338 and on:
   graph=d.reg('eax');gw=d.words(graph,6);todo=[gw[0]];nodes={};tuples={};operands={}
   while todo:
    p=todo.pop()
    if p in nodes or not p:continue
    w=d.words(p,16);nodes[p]=w
    if len(w)<16:continue
    t=w[7]
    if t:
     tw=d.words(t,16);tuples[t]=tw
     # The expression/operand list is threaded at +0x28.
     if len(tw)==16:
      a=tw[10];seen=set()
      while 0x10000<a<0x10000000 and a not in seen and len(seen)<8:
       seen.add(a);aw=d.words(a,16);operands[a]=aw
       if len(aw)!=16:break
       a=aw[0]
    edge=w[3];seen=set()
    while edge and edge not in seen:
     seen.add(edge);ew=d.words(edge,4)
     if len(ew)<4:break
     todo.append(ew[3]);edge=ew[0]
   graphs.append(dict(graph=graph,words=gw,nodes=nodes,tuples=tuples,operands=operands))
   (W/(tag+'.graphs.json')).write_text(json.dumps(dict(functions=functions,graphs=graphs)))
   print('graph',len(graphs),'nodes',len(nodes),'ops',dict(collections.Counter(v[1] for v in tuples.values() if len(v)>1)),flush=True)
finally:
 (W/(tag+'.graphs.json')).write_text(json.dumps(dict(functions=functions,graphs=graphs)))
 if d.p.poll() is None:
  try:d.prompt('quit',10)
  except Exception:d.p.kill()
 d.log.close()
