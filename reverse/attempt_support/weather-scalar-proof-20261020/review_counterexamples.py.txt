"""Independent negative probes; materialize only after explicitly applying the bank."""
from pathlib import Path
import struct
import sys

ROOT = Path(__file__).resolve().parents[3]
sys.path[:0] = [str(ROOT / "tools"), str(ROOT / "tools/tests")]
from test_link_debt import scalar_call
from test_gate_exploits import TEXT, RDATA


def test_ebp_relative_popped_word(scalar_call):
    f = scalar_call
    prefix = bytes.fromhex("55 8bec 83ec04")
    f.raw = prefix + f.raw[:-1] + bytes.fromhex("8b45f8 c70001000000 c9 c3")
    f.relocs = [(len(prefix) + 6, 1, 0x14)]
    f.save()
    assert f.findings() == ["0x886655"]


def test_folded_pointer_in_other_exact_body(scalar_call):
    f = scalar_call
    f.text = "const int x=0x886655; void f(){consume(x);} int g(){return *(int*)(x+1);}\n"
    other = b"\xa1" + struct.pack("<I", f.value + 1) + b"\xc3"
    f.extra = [(".text", TEXT, other, [])]
    f.extra_symbols = [("_other", 0, 2, 0x20, 2, 0)]
    f.rows.append(dict(name="_other", source=f.source, target_rva="0x4000", target_size="6"))
    f.truth.ledger["_other"] = {0x4000}
    f.memory[0x4000] = other
    f.save()
    assert f.findings() == ["0x886655"]


def test_folded_pointer_in_static_data(scalar_call):
    f = scalar_call
    f.text = "const int x=0x886655; void f(){consume(x);} int *p=(int*)(x+1);\n"
    f.extra = [(".rdata", RDATA, struct.pack("<I", f.value + 1), [])]
    f.extra_symbols = [("_pointer", 0, 2, 0, 2, 0)]
    f.save()
    assert f.findings() == ["0x886655"]
