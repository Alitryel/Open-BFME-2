# -*- coding: utf-8 -*-
# astra-s3 helper seat round 3: WHILE-loop sources with split entry/body statements.
# Hypothesis: MSVC's while-inversion (guard block separate from preheader) yields
# [load, lea][test, je][body load+advance][test, jne] = the retail shape, while
# source-level if+do forms merge the guard block and sink the lea.

FN = 'int rva007B9CCAStrLen(const char *s) { '
RET = ' return (int)(p - s - 1); }'

VARIANTS = []
def add(tag, body, flags='/O1 /G6 /MD', profile='', ext='cpp'):
    VARIANTS.append((tag, flags, profile, ext, body))

# W1: split entry (c from s, p = s+1), plain while, single-statement body
add('w1_post',    FN + 'char c = *s; const char *p = s + 1; while (c) { c = *p++; }' + RET)
add('w1_ne',      FN + 'char c = *s; const char *p = s + 1; while (c != 0) { c = *p++; }' + RET)
add('w1_order2',  FN + 'const char *p = s + 1; char c = *s; while (c) { c = *p++; }' + RET)
add('w1_vol',     FN + 'char c = *(volatile const char *)s; const char *p = s + 1; while (c) { c = *p++; }' + RET)
add('w1_u8',      FN + 'unsigned char c = (unsigned char)*s; const char *p = s + 1; while (c) { c = (unsigned char)*p++; }' + RET)
add('w1_ptr',     FN + 'const char *p = s; char c = *p; p = p + 1; while (c) { c = *p++; }' + RET)

# W2: split body statements (load, then advance) inside the while
add('w2_split',   FN + 'char c = *s; const char *p = s + 1; while (c) { c = *p; p = p + 1; }' + RET)
add('w2_pre',     FN + 'char c = *s; const char *p = s + 1; while (c) { c = *p; ++p; }' + RET)
add('w2_postop',  FN + 'char c = *s; const char *p = s + 1; while (c) { c = *p; p++; }' + RET)
add('w2_add',     FN + 'char c = *s; const char *p = s + 1; while (c) { c = *p; p += 1; }' + RET)
add('w2_idx',     FN + 'char c = *s; const char *p = s + 1; while (c) { c = p[0]; p = &p[1]; }' + RET)
add('w2_ne',      FN + 'char c = *s; const char *p = s + 1; while (c != 0) { c = *p; p = p + 1; }' + RET)
add('w2_vol',     FN + 'char c = *(volatile const char *)s; const char *p = s + 1; while (c) { c = *p; p = p + 1; }' + RET)
add('w2_volbody', FN + 'char c = *s; const char *p = s + 1; while (c) { c = *(volatile const char *)p; p = p + 1; }' + RET)
add('w2_u8',      FN + 'unsigned char c = (unsigned char)*s; const char *p = s + 1; while (c) { c = (unsigned char)*p; p = p + 1; }' + RET)
add('w2_ptr',     FN + 'const char *p = s; char c = *p; p = p + 1; while (c) { c = *p; p = p + 1; }' + RET)
add('w2_order2',  FN + 'const char *p = s + 1; char c = *s; while (c) { c = *p; p = p + 1; }' + RET)

# W3: while with mismatching latch predicates (rotation blockers)
for lt, lx in [('bb1', 'c && 1'), ('neq0eq', '(c == 0) == 0'), ('ter01', '(c == 0) ? 0 : 1'),
                ('ternz', 'c ? 1 : 0'), ('ucz', '(unsigned char)c != 0'), ('ucgt', '(unsigned char)c > 0')]:
    add('w3_%s' % lt, FN + 'char c = *s; const char *p = s + 1; while (' + lx + ') { c = *p++; }' + RET)
    add('w3s_%s' % lt, FN + 'char c = *s; const char *p = s + 1; while (' + lx + ') { c = *p; p = p + 1; }' + RET)

# W4: while with mirrored second variable
add('w4_mirror',  FN + 'char c = *s; const char *p = s + 1; char d = c; while (d) { c = *p; p = p + 1; d = c; }' + RET)
add('w4_mirror2', FN + 'char c = *s; const char *p = s + 1; char d = c; while (d) { d = *p; p = p + 1; c = d; }' + RET)

# W5: while nested inside a guard if (redundant first test)
add('w5_in_if',   FN + 'char c = *s; const char *p = s + 1; if (c) { while (c) { c = *p++; } }' + RET)
add('w5_in_if2',  FN + 'char c = *s; const char *p = s + 1; if (c) { while (c) { c = *p; p = p + 1; } }' + RET)

# W6: do-while with guard inside loop-continue structures
add('w6_forever', FN + 'const char *p = s + 1; char c = *s; if (c) { for (;;) { c = *p; p = p + 1; if (!c) break; } }' + RET)
add('w6_forever2', FN + 'const char *p = s + 1; char c = *s; if (c) { for (;;) { c = *p++; if (!c) break; } }' + RET)

# W7: while reading guard char through p after advance (address through advanced reg)
add('w7_m1',      FN + 'const char *p = s + 1; char c = *(p - 1); while (c) { c = *p; p = p + 1; }' + RET)
add('w7_m1b',     FN + 'const char *p = s + 1; char c = p[-1]; while (c) { c = p[0]; p = &p[1]; }' + RET)

# W8: controls
add('ctl_loop0',  FN + 'const char *p = s; char c = *p++; while (c) c = *p++;' + RET)
add('ctl_split_ifdo', FN + 'char c = *s; const char *p = s + 1; if (c) { do { c = *p++; } while (c); }' + RET)
add('ctl_split_ifdo_vol', FN + 'char c = *(volatile const char *)s; const char *p = s + 1; if (c) { do { c = *p++; } while (c); }' + RET)
