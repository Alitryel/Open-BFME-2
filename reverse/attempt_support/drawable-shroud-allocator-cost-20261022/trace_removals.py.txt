import sys,json
from pathlib import Path
sys.path.insert(0,str(Path(__file__).resolve().parent))
from wdbg import Dbg

src,obj=sys.argv[1:3]
d=Dbg(src,obj,obj+'.debug.log')
d.prompt();d.prompt('set $CanDeferOnBPByAddr = 1')
base=0x10700000
bp=base+0x83acd
rem=base+0x9724c
added=False;focused=False;recorded=0
out=d.prompt('break *0x%x'%bp)
if 'Unable' not in out:added=True
while True:
    out=d.prompt('cont')
    if 'terminated' in out:break
    eip=d.reg('eip')
    if not added and eip!=bp:
        out=d.prompt('break *0x%x'%bp)
        if 'Unable' not in out:added=True
    if eip==bp:
        ebp=d.reg('ebp');f=d.words(ebp,1)[0]
        fn=d.string(d.words(f+4,1)[0])
        now='onComingOutOfShroud' in fn
        if now and not focused:
            d.prompt('break *0x%x'%rem);focused=True
            print(json.dumps({'enter':fn,'class':d.reg('ebx')}),flush=True)
        elif focused and not now:
            d.prompt('delete 2');focused=False
            print(json.dumps({'leave':fn,'removed_ranges':recorded}),flush=True)
    elif eip==rem and focused:
        first=d.reg('eax');end=d.reg('ebx');esp=d.reg('esp')
        ret=d.words(esp,1)[0];nodes=[];p=first
        while p!=end and p and len(nodes)<16:
            words=d.words(p,16)
            nodes.append({'p':hex(p),'words':['%08x'%w for w in words]})
            p=words[0]
        print(json.dumps({'caller':hex(ret),'first':hex(first),'end':hex(end),'nodes':nodes}),flush=True)
        recorded+=1
print(json.dumps({'done':True,'removed_ranges':recorded}),flush=True)
