import sys,struct,json
from pathlib import Path
sys.path.insert(0,str(Path(__file__).resolve().parents[2]/'tools'))
import build,capstone
D=Path(__file__).resolve().parent
image,sections=build.exe_image(); base=0x400000
md=capstone.Cs(capstone.CS_ARCH_X86,capstone.CS_MODE_32)
def body(r,n=40):
 try:return build.read_pe_bytes(image,sections,r,n)
 except ValueError:return b''
def listing(r,n=64):
 out=[]
 for i in md.disasm(body(r,n),r):
  out.append(f'{i.address:08x} {i.mnemonic} {i.op_str}')
  if i.mnemonic=='ret':break
 return out
code=next(s for s in sections if s['name']=='.text'); lo=base+code['rva'];hi=lo+code['virtual_size']
ans=[]
for s in sections:
 if s['name'] not in ('.rdata','.data'):continue
 raw=image[s['raw_pointer']:s['raw_pointer']+s['raw_size']]
 for p in range(0,len(raw)-40,4):
  a=struct.unpack_from('<10I',raw,p)
  if not all(lo<=x<hi for x in a):continue
  setter=list(md.disasm(body(a[6]-base,24),0));getter=list(md.disasm(body(a[7]-base,24),0))
  # The source interface's +18 setter and +1c getter bracket one pointer field.
  if len(setter)<3 or len(getter)<2:continue
  if not (setter[0].mnemonic=='mov' and setter[0].op_str=='eax, dword ptr [esp + 4]' and setter[1].mnemonic=='mov' and setter[1].op_str.endswith(', eax') and setter[2].mnemonic=='ret' and setter[2].op_str=='4'):continue
  if not (getter[0].mnemonic=='mov' and getter[0].op_str.startswith('eax, dword ptr [ecx') and getter[1].mnemonic=='ret' and not getter[1].op_str):continue
  r=s['rva']+p;ans.append({'table_rva':hex(r),'entries':[hex(x-base) for x in a],'slots':{str(j):listing(a[j]-base,120) for j in (0,1,3,5,6,7,8,9)}})
(D/'interface-candidates.json').write_text(json.dumps(ans,indent=2))
for a in ans:
 if a['table_rva']=='0x7fc290':print(json.dumps(a,indent=2))
print('candidates',len(ans))
