from pathlib import Path
import concurrent.futures,itertools,json,re,runpy,random
OUT=Path('build/canmake-alias-20261023/range-return').resolve();OUT.mkdir(parents=True,exist_ok=True)
n=runpy.run_path('reverse/attempt_support/canmake-early-cleanup-20261022/replay.py.txt');e=n['context']();r=e['r'];r['run'].__globals__['OUT']=OUT
text=__import__('subprocess').check_output(['git','cat-file','blob','20afdd90f70d238a5f035acc08f72ff23e28a844'],text=True).split('int AIBuildableWallSegment::canMake(Player *player)',1)[1];bank=text[text.index('{')+1:text.rindex('}')]
predicates=[
 'C>M','C>=M+1u','C-1u>=M','C-M!=0u','C!=0u','M!=0xffffffffu',
 '(C^M)!=0u','(C|M)!=0u','(C&~M)!=0u',
 '(unsigned __int64)C-M<=C','M%C==M','M/C==0u',
]
rng=random.Random(596e15)
for money,cost in [(0,1),(0,0xffffffff),(0xfffffffe,0xffffffff),(0x7fffffff,0x80000000)]+[sorted(rng.sample(range(0x100000000),2))for _ in range(10000)]:
 assert cost>money
 assert all([cost>money,cost>=money+1,cost-1>=money,cost-money!=0,cost!=0,money!=0xffffffff,cost^money,(cost|money)!=0,(cost&~money)!=0,cost-money<=cost,money%cost==money,money//cost==0])
forms=['v?2:0','2u*v','v+v','v<<1','(0u-v)&2u','2u-2u*!v','(v?9:0)-7','9u-7u*v']
tests=[]
for base,pred,shape,binding in itertools.product(range(3),range(12),range(8),range(4)):
 b=[bank,n['early'](e),e['win']][base]
 expr=predicates[pred].replace('C','((unsigned int)result.m_pad04)').replace('M','player->m_money')
 decl='';v='(unsigned int)('+expr+')'
 if binding:
  typ=['','bool','unsigned int','const bool &'][binding];decl=typ+' trueValue=('+expr+');'
  v='(unsigned int)trueValue'

 # The second return two is reached only after unsigned money < cost.
 positions=[m.start()for m in re.finditer(r'return 2;',b)];assert len(positions)==2

 tests.append((f'range-{base}-{pred}-{shape}-{binding}',e['prefix']+'\nint AIBuildableWallSegment::canMake(Player *player) {\n'+b+'\n}\n'))
results=[]
with concurrent.futures.ThreadPoolExecutor(max_workers=8)as pool:
 for i,d in enumerate(pool.map(r['run'],tests),1):
  results.append(d)
  if i%192==0 or d.get('exact_candidate'):print(json.dumps(dict(done=i,total=len(tests),best=sorted(results,key=lambda x:(x.get('nd',999),-x.get('score',0)))[:2])),flush=True)
(OUT/'results.json').write_text(json.dumps(results,indent=2)+'\n')
print(json.dumps(dict(trials=len(results),errors=sum('error'in x for x in results),unresolved=sum(bool(x.get('unresolved'))for x in results),exact=sum(x.get('exact_candidate',False)for x in results),bodies=len({x.get('masked_code_sha256')for x in results}))),flush=True)
