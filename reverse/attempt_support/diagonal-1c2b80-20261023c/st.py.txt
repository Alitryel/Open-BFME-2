import sys
def state(t):
    ids=list(map(int,open('tr/%s.cnt.ids'%t).read().split()))
    gs=list(map(int,open('tr/%s.cnt.g'%t).read().split()))
    A=next(i for i,x in enumerate(ids) if x>=0)
    B=next(i for i in range(A,len(ids)) if ids[i]<0)-A
    # kind-3 jumps within A phase: g increases between i-1 and i
    j3=[i for i in range(1,A+1) if gs[i]!=gs[i-1]]
    k15=[x for x in ids if x>=0]
    # last kind-3 refill index (allocation index whose bp saw the new G)
    import hashlib
    tail=hashlib.md5(' '.join('3' if x<0 else 'f' for x in ids[A+B:]).encode()).hexdigest()[:6]
    return dict(t=t,G0=gs[0],A=A,B=B,G1=gs[A+B],first15=k15[0],last15=k15[B-1],j3=j3[-3:],tail=tail,nj3=len(j3))
for t in sys.argv[1:]:
    print(state(t))
