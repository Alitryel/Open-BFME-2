import sys,os,re,subprocess,importlib.util
from concurrent.futures import ThreadPoolExecutor
from pathlib import Path
# scr.py SPEC.py : compile each variant under the c2 tracer (inj.py) and print size/nd plus the a8f54 set fingerprint
H=Path('/work/Open-BFME-2/build/c2tr'); D=H/'v3'
spec=importlib.util.spec_from_file_location('s',sys.argv[1]); m=importlib.util.module_from_spec(spec); spec.loader.exec_module(m)
base=Path(getattr(m,'BASE',str(H/'bank.cpp'))).read_text()
def fp(out):
    r=[]
    for l in out.splitlines():
        mm=re.match(r'R(\d+) call(\d+) .* n=(\d+) :: (.*)',l)
        if not mm: continue
        if mm.group(1)=='1': continue
        nodes=[x for x in mm.group(4).split() if x.startswith('node:')]
        r.append('R%s:%s%s'%(mm.group(1),mm.group(3),'N' if nodes else ''))
    import hashlib
    h=hashlib.md5('\n'.join(re.sub(r'set=\w+ a3=\w+','',l) for l in out.splitlines() if l.startswith('R')).encode()).hexdigest()[:6]
    return ' '.join(r)+' #'+h
def one(item):
    name,reps=item
    t=base
    if isinstance(reps,str): t=reps
    else:
        K='void HordeMeleeAmoeba::updateMeleeAttack'
        pre,post=t.split(K,1); post=K+post
        for a,b in reps:
            if a.startswith('PRE:'):
                a=a[4:]
                if a not in pre: return '%s MISSING %r'%(name,a[:60])
                pre=pre.replace(a,b,1); continue
            if a not in post: return '%s MISSING %r'%(name,a[:60])
            post=post.replace(a,b,1)
        t=pre+post
    f=D/(name+'.cpp'); f.write_text(t); obj=D/(name+'.obj')
    if obj.exists(): obj.unlink()
    r=subprocess.run(['timeout','600','python3',str(H/'inj.py'),str(f),str(obj)],capture_output=True,text=True,errors='replace',cwd=H)
    if not obj.exists(): return '%s ERR'%name
    s=subprocess.run(['python3',str(H/'objscore.py'),str(obj),str(f)],capture_output=True,text=True,errors='replace',cwd=H).stdout.strip()
    return '%s %s | %s'%(name,s,fp(r.stdout))
items=list(m.V.items())
with ThreadPoolExecutor(int(os.environ.get('J','8'))) as ex:
    for r in ex.map(one,items): print(r,flush=True)
