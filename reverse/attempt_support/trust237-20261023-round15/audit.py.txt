from pathlib import Path
import collections,hashlib,itertools,json,struct,sys
ROOT=Path.cwd();OUT=ROOT/'build/trust237-round15';sys.path.insert(0,str(ROOT/'tools'))
import build,capstone,pefile
from variant import resolve_bytes
NAME='?Rva00237D3DCheck@@YAHXZ';RVA=0x237d3d;SIZE=226
target=build.read_target_bytes(RVA,SIZE)
native=json.loads((OUT/'native.json').read_text())
links=json.loads((OUT/'ltcg.json').read_text())['results']
md=capstone.Cs(capstone.CS_ARCH_X86,capstone.CS_MODE_32);md.detail=True
api_symbols={'memset':'_memset','WinVerifyTrust':'_WinVerifyTrust@12','GetCommandLineW':'__imp__GetCommandLineW@0','CommandLineToArgvW':'__imp__CommandLineToArgvW@8','LocalFree':'__imp__LocalFree@4'}
groups={};receipts=[]
targetpe=pefile.PE(str(ROOT/'baselines/bfme2/workshop-vanilla-1.06/files/game.dat'))
targetimports={i.name.decode():i.address for dll in targetpe.DIRECTORY_ENTRY_IMPORT for i in dll.imports if i.name}
targets={'_memset':0x6291ae,'_WinVerifyTrust@12':0x655652}
reverse_apis={v:k for k,v in api_symbols.items()}
def check(row,raw,rel,family):
 ins=list(md.disasm(raw,0));assert sum(i.size for i in ins)==len(raw)
 assert [i.address for i in ins if i.mnemonic=='nop']==row['nops']==[]
 resolved,unres=resolve_bytes(raw,rel,target+b'\0'*max(0,len(raw)-SIZE),RVA,len(raw),{'_memset':[0x6291ae],'_WinVerifyTrust@12':[0x655652]})
 assert unres==[],unres
 actual=bytearray(raw)
 for off,typ,name in rel:
  if typ==6:value=targetimports[reverse_apis[name]]
  else:value=(targets[name]-RVA-off-4)&0xffffffff
  actual[off:off+4]=struct.pack('<I',value)
 comparison=resolved if family=='native' else actual
 counted=[i for i,(a,b) in enumerate(itertools.zip_longest(comparison,target)) if a!=b]
 assert len(counted)==row['diff'],(row['name'],len(counted),row['diff'])
 prefix=next((i for i,(a,b) in enumerate(zip(comparison,target)) if a!=b),min(len(raw),SIZE));assert prefix==row['prefix']
 if family=='ltcg':assert actual.hex()==row['normalized']
 diffs=[i for i,(a,b) in enumerate(itertools.zip_longest(actual,target)) if a!=b]
 canon=bytearray(raw)
 for off,typ,name in rel:
  assert canon[off:off+4]==b'\0'*4,(row['name'],off,typ,name)
 key=hashlib.sha256(canon+json.dumps(rel).encode()).hexdigest()
 if key not in groups:groups[key]=dict(raw=bytes(canon),relocations=rel,sources=[])
 groups[key]['sources'].append(row['name'])
 receipts.append(dict(name=row['name'],family=family,size=len(raw),sha256=hashlib.sha256(raw).hexdigest(),diff=len(diffs),prefix=next((i for i,(a,b) in enumerate(zip(actual,target)) if a!=b),min(len(raw),SIZE)),outside_nop_differences=sum(i<205 or i>=221 for i in diffs),canonical_code_reloc_sha256=key))

for row in native:
 assert 'error' not in row
 raw,rel=build.read_object_symbol_bytes(OUT/(row['name']+'.obj'),NAME,code_only=True)
 assert raw.hex()==row['raw'] and rel==[tuple(x) for x in row['relocations']]
 assert hashlib.sha256(raw).hexdigest()==row['sha256']
 check(row,raw,rel,'native')
for row in links:
 assert row['compile_returncode']==row['link_returncode']==0
 pe=pefile.PE(str(OUT/(row['name']+'.dll')));base=pe.OPTIONAL_HEADER.ImageBase
 start=next(x.address for x in pe.DIRECTORY_ENTRY_EXPORT.symbols if x.name.decode()==NAME)
 raw=pe.get_data(start,row['size']);assert raw.hex()==row['raw']
 imports={i.address:i.name.decode() for dll in pe.DIRECTORY_ENTRY_IMPORT for i in dll.imports if i.name}
 canon=bytearray(raw);rel=[];apis=[]
 for ins in md.disasm(raw,base+start):
  if ins.mnemonic!='call':continue
  off=ins.address-base-start
  if ins.operands[0].type==capstone.x86.X86_OP_MEM:
   api=imports[ins.operands[0].mem.disp];slot=off+2;typ=6
  else:
   thunk=pe.get_data(ins.operands[0].imm-base,6);assert thunk[:2]==b'\xff\x25'
   api=imports[struct.unpack('<I',thunk[2:])[0]];slot=off+1;typ=0x14
  apis.append([off,api]);rel.append((slot,typ,api_symbols[api]));canon[slot:slot+4]=b'\0'*4
 assert apis==row['calls']
 check(row,bytes(canon),rel,'ltcg')

probes=json.loads((OUT/'intrinsics.json').read_text())
assert len(probes)==6
for row in probes:
 assert 'C4163' in row['warnings'] and len(row['unresolved'])==16 and row['nops']==[]

result=dict(native_objects=len(native),ltcg_pairs=len(links),recognized_nop_intrinsics=0,unrecognized_intrinsic_probes=6,distinct_code_relocation_groups=len(groups),all_decoded_without_nops=True,best=sorted(receipts,key=lambda r:(r['outside_nop_differences'],r['diff'],-r['prefix']))[:6],receipts=receipts)
(OUT/'audit.json').write_text(json.dumps(result,indent=2)+'\n')
print({k:v for k,v in result.items() if k not in ['receipts','best']},flush=True)
