from itertools import product
from pathlib import Path
W=Path(__file__).resolve().parent
base=(W/'base.cpp').read_text()
out=W/'fasttable';out.mkdir(exist_ok=True)
for idx,(pack,ref,copy,ctorfast,prepraw,testraw,respraw,word) in enumerate(product(range(2),repeat=8)):
 cc='__fastcall ' if ctorfast else ''
 fld='const ContactVector &' if ref else 'const ContactVector *'
 val='p' if ref else '&p'
 defs='struct PArg { '+fld+' p; '+cc+'PArg(const ContactVector &p):p('+val+'){}'
 if copy:defs+=' '+cc+'PArg(const PArg &a):p(a.p){}'
 defs+=' };\n'
 it='unsigned' if word else 'void *'; iv='(unsigned)p' if word else 'p'
 defs+='struct IArg { '+it+' p; '+cc+'IArg(void *p):p('+iv+'){}'
 if copy:defs+=' '+cc+'IArg(const IArg &a):p(a.p){}'
 defs+=' };\n'
 if pack:
  defs+='struct PPair { PArg point,normal; '+cc+'PPair(const ContactVector &p,const ContactVector &n):point(p),normal(n){}'
  if copy:defs+=' '+cc+'PPair(const PPair &a):point(a.point),normal(a.normal){}'
  defs+=' };\n'
 argdecl='PPair args' if pack else 'PArg point,PArg normal'
 pv='PPair(point,normal)' if pack else 'point,normal'
 defs+='struct ContactSource;\nstruct ContactTable { void *slot0,*slot1,*slot2; void *(__fastcall *prepare)(ContactSource *self,'+argdecl+'); void *slot4,*slot5,*slot6,*slot7; bool (__fastcall *test)(ContactSource *self,IArg info); void (__fastcall *respond)(ContactSource *self,IArg info); };\n'
 defs+='struct ContactSource { ContactTable *table; void *prepare(const ContactVector &point,const ContactVector &normal) { return table->prepare(this,'+pv+'); } bool test(void *p) { return table->test(this,p); } void respond(void *p) { table->respond(this,p); } };\n'
 lo=base.index('struct ContactSource {');hi=base.index('struct ContactData')
 s=base[:lo]+defs+base[hi:]
 for source in ['node->first->source','node->second->source']:
  if prepraw:
   for normal in ['node->normal','ContactVector(node->normal).negate()']:
    old=source+'->prepare(node->point,'+normal+')'
    args='PPair(node->point,'+normal+')' if pack else 'node->point,'+normal
    s=s.replace(old,source+'->table->prepare('+source+','+args+')')
  if testraw:s=s.replace(source+'->test(',source+'->table->test('+source+',')
  if respraw:s=s.replace(source+'->respond(',source+'->table->respond('+source+',')
 (out/('v%04d.cpp'%idx)).write_text(s)
print('generated',idx+1)
