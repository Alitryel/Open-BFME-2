import ast,collections,json,re,sys
from pathlib import Path
D=Path(__file__).resolve().parent
base=(D/'base.cpp').read_text()
literals=lambda s:collections.Counter(re.findall(r'"(?:\\.|[^"\\])*"',s))
families=[('fire_order.py','addresses','fire-address-order',512),('fire_order.py','helpers','fire-format-order',480),('adapters.py','conventions','fire-conventions',960),('adapters.py','values','fire-values',496)]
for script,family,label,n in families:
 captured=[];namespace={'__name__':'verification','__file__':str(D/'trials.py')}
 ast.parse((D/script).read_text())
 exec(compile((D/'trials.py').read_text(),str(D/'trials.py'),'exec'),namespace)
 namespace['batch']=lambda items,label:captured.extend(items)
 old=sys.argv;sys.argv=['verify',family]
 exec(compile((D/script).read_text().replace('from trials import *',''),str(D/script),'exec'),namespace)
 sys.argv=old
 results=json.loads((D/(label+'-results.json')).read_text())
 assert len(captured)==len(results)==n
 for (tag,source),result in zip(captured,results):
  assert result['tag']==tag and 'error' not in result
  assert source==(D/(tag+'.cpp')).read_text(),tag
  assert literals(source)==literals(base),tag
  assert result['size']==4101 and result['regions']==0 and result['nd'] in (13,44)
 print(family,n,'exact regenerations, intact literal multiset, successful isolated compilation')
assert sum(x[3] for x in families)==2448
native=json.loads((D/'helper00000-before-pack.json').read_text())
rows=next(z for z in native if isinstance(z,list) and any(x['name'].startswith('_draw\n') for x in z))
prior=json.loads(Path('reverse/attempt_support/0x0004826c-sol6-allocator.json').read_text())['old']
keys=('index','size','weight','type')
rr=[dict(zip(keys,(x['data'][15],x['data'][8],x['data'][14],x['data'][11]))) for x in rows]
assert rr==[{k:x[k] for k in keys} for x in prior['rows']]
bs=lambda ns:sorted({n[0]+b for n in ns for b in range(32) if n[2]>>b&1})
conf={str(x['index']):bs(x['nodes']) for x in native[-1]['conflicts']}
delta={k:dict(before=prior['conflicts'][k],after=v) for k,v in conf.items() if v!=prior['conflicts'][k]}
assert delta=={'21':{'before':[0,1,5,9,21,70,73,74,75,76],'after':[21]}}
compact=dict(parent='0x0004826c-sol6-allocator.json:old',ordered_records_equal=True,conflict_delta=delta)
assert compact==json.loads((D/'helper-native-delta.json').read_text())
print('Native helper: same 77 ordered records; only canonical fontName conflict delta')
