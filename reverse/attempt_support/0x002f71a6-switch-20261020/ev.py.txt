"""ev.py f.cpp... : fit + shape flags E(entry push esi) C(cursor in edi) Z(xor edi,edi zero) R(rawTo esi post-loop)"""
import sys
sys.path.insert(0, '/work3/Open-BFME-2/build/h72')
from score import compile_, md, score
from concurrent.futures import ThreadPoolExecutor
def one(src):
    try: s = score(src)
    except Exception as e: return f'{src}: FAIL {e}'
    r, err = compile_(src)
    if r is None: return f'{src}: FAIL'
    L = list(md.disasm(bytes(r[0]['bytes']), 0))
    E = L[5].mnemonic == 'push' and L[5].op_str == 'esi'
    C = any(i.mnemonic == 'mov' and i.op_str == 'edi, eax' for i in L if 0xd0 <= i.address < 0x100)
    Z = any(i.mnemonic == 'xor' and i.op_str == 'edi, edi' for i in L if 0x1a0 <= i.address < 0x220)
    R = any(i.mnemonic == 'mov' and i.op_str == 'esi, dword ptr [ebp + 0x70]' for i in L if 0x1a0 <= i.address < 0x220)
    fl = ''.join(c if v else '.' for c, v in zip('ECZR', (E, C, Z, R)))
    return f'{fl} {s}'
with ThreadPoolExecutor(8) as ex:
    for l in ex.map(one, sys.argv[1:]): print(l, flush=True)
