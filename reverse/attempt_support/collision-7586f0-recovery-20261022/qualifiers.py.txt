from search import *
import itertools
cut=BASE.index('void CollisionManagerImpl::rva007586F0()')
pre,body=BASE[:cut],BASE[cut:]
items=[]
for n,(indexed,mask,typ,local) in enumerate(itertools.product(range(2),range(32),range(3),range(3))):
 s=body
 if indexed:s=s.replace('SortList *B = A + 1;','SortList *B = &lists[src+1];')
 # Qualify real stack-list reads separately. Their values and 32-bit memory widths stay fixed.
 def qual(base,member):
  return ['((volatile SortList *)'+base+')->'+member,
          '(*((const volatile unsigned int *)&'+base+'->'+member+'))',
          'readCount('+base+')'][typ]
 qpre=pre+'\ninline unsigned int readCount(const SortList *p) { return *(const volatile unsigned int *)&p->count; }\n'
 if mask&1:s=s.replace('while (A->count != 0)','while ('+qual('A','count')+' != 0)')
 if mask&2:s=s.replace('a != 0 && A->count != 0','a != 0 && '+qual('A','count')+' != 0')
 if mask&4:s=s.replace('b != 0 && B->count != 0','b != 0 && '+qual('B','count')+' != 0')
 if mask&8:s=s.replace('from->count--;',qual('from','count')+'--;') if typ!=2 else s.replace('from->count--;','--(*(volatile unsigned int *)&from->count);')
 if mask&16:s=s.replace('if (o->count == 0)','if ('+qual('o','count')+' == 0)')
 if local==1:
  s=s.replace('A->head->m_value','((const volatile Endpoint *)A->head)->m_value')
 if local==2:
  s=s.replace('A->head->m_value','A->head->m_value').replace('B->head->m_value','((const volatile Endpoint *)B->head)->m_value')
 # Unsigned int accessor cannot mutate through a const lvalue; keep type1 read-only casts for the decrement.
 if typ==1 and mask&8:s=s.replace(qual('from','count')+'--;','--(*(volatile unsigned int *)&from->count);')
 items.append(('qual-%04d'%n,qpre+s))
batch(items,'qualifiers')
