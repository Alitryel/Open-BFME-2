from pathlib import Path
from itertools import product
W=Path(__file__).resolve().parent;base=(W/'base.cpp').read_text();out=W/'bytereturn';out.mkdir(exist_ok=True)
for i,(t,r,p) in enumerate(product(['bool','char','signed char','unsigned char'],['void','bool','char','unsigned char','int','unsigned','void *','ContactSource *'],['void *','char *','unsigned','int'])):
 s=base.replace('struct ContactSource {','struct ContactSource;\nstruct ContactSource {')
 s=s.replace('virtual bool test(void *info)=0;','virtual '+t+' test('+p+' info)=0;')
 s=s.replace('virtual void respond(void *info)=0;','virtual '+r+' respond('+p+' info)=0;')
 s=s.replace('virtual void *prepare(','virtual '+p+' prepare(')
 (out/('v%04d.cpp'%i)).write_text(s)
print('generated',i+1)
