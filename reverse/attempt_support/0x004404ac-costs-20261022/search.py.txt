import concurrent.futures as cf
import difflib
import json
import pathlib
import subprocess
import sys
import time

ROOT = pathlib.Path('/work7/Open-BFME-2')
sys.path.insert(0, str(ROOT / 'tools'))
import build
from variant import resolve_bytes
import capstone

OUT = ROOT / 'build/hard-4404ac'
BASE = (ROOT / 'reverse/attempts/0x004404ac.cpp').read_text()
MARK = 'void AptMpGameSetup::rva004404AC(bool enable, int slot)'
HEAD, BODY = BASE.split(MARK)
SYM = '?rva004404AC@AptMpGameSetup@@QAEX_NH@Z'
RVA, SIZE = 0x4404ac, 526
TARGET = build.read_target_bytes(RVA, SIZE)
SMAP = build.load_symbol_map()
MD = capstone.Cs(capstone.CS_ARCH_X86, capstone.CS_MODE_32)
TINS = [(i.mnemonic, i.op_str) for i in MD.disasm(TARGET, RVA)]

def score(name, text):
    src, obj = OUT / (name + '.cpp'), OUT / (name + '.obj')
    reuse=src.exists() and obj.exists() and src.read_text()==text
    if not reuse:
        src.write_text(text)
        if obj.exists():obj.unlink()
        cmd, env = build.compiler_command(src, obj)
        p = subprocess.run(cmd, cwd=ROOT, env=env, capture_output=True, text=True)
        if p.returncode:
            (OUT / (name + '.log')).write_text(p.stdout + p.stderr)
            return dict(name=name, error=p.returncode)
    raw, relocs = build.read_object_symbol_bytes(obj, SYM, code_only=True)
    span=max(len(raw),SIZE)
    full_target=build.read_target_bytes(RVA, span)
    padded = raw.ljust(span, b'\0')
    code, unresolved = resolve_bytes(padded, relocs, full_target, RVA, span, SMAP)
    (OUT / (name + '.bin')).write_bytes(code)
    md = capstone.Cs(capstone.CS_ARCH_X86, capstone.CS_MODE_32)
    ins = [(i.mnemonic, i.op_str) for i in md.disasm(code[:len(raw)], RVA)]
    matcher = difflib.SequenceMatcher(None, [x[0] for x in ins], [x[0] for x in TINS], autojunk=False)
    regions, operands = 0, 0
    for tag,a,b,c,d in matcher.get_opcodes():
        if tag != 'equal':
            regions += 1
        else:
            operands += sum(ins[a+k][1] != TINS[c+k][1] for k in range(b-a))
    return dict(name=name, size=len(raw), nd=sum(a!=b for a,b in zip(code,TARGET)), regions=regions, operands=operands, unresolved=unresolved)

def safe_score(name,text):
    try:return score(name,text)
    except BaseException as e:return dict(name=name,error=str(e))

def qualify(bits):
    head, body = HEAD, BODY
    # Only existing scalar loads are qualified; no extra use or access.
    mods = [
        ('m_game->rva0043DA65()', '((Rva0043DA65 * volatile &)m_game)->rva0043DA65()'),
        ('m_rules.m_00', '((const volatile MpGameSetupRules &)m_rules).m_00'),
        ('m_rules.m_04', '((const volatile MpGameSetupRules &)m_rules).m_04'),
        ('m_hero[slot]', '((GameWindow * const volatile *)m_hero)[slot]'),
        ('m_playerTemplate[slot]', '((GameWindow * const volatile *)m_playerTemplate)[slot]'),
        ('m_handicap[slot]', '((GameWindow * const volatile *)m_handicap)[slot]'),
        ('game->getConstSlot(game->v13())->m_hasMap', '((const volatile GameSlot *)game->getConstSlot(game->v13()))->m_hasMap'),
    ]
    for i,(a,b) in enumerate(mods):
        if bits & (1<<i): body = body.replace(a,b)
    return head+MARK+body

def run(jobs, label):
    start=time.time()
    results=[]
    with cf.ThreadPoolExecutor(max_workers=8) as pool:
        futures = {pool.submit(safe_score,n,s):n for n,s in jobs}
        for f in cf.as_completed(futures):
            r=f.result(); results.append(r)
            if 'error' not in r and r['nd']==0 and r['size']==SIZE:
                print('EXACT',r,flush=True)
    results.sort(key=lambda r:(r.get('regions',999),r.get('operands',999),abs(r.get('size',0)-SIZE),r.get('nd',999)))
    (OUT/(label+'.json')).write_text(json.dumps(results,indent=2))
    print(json.dumps(dict(label=label, seconds=round(time.time()-start,1), errors=sum('error' in r for r in results), best=results[:12])),flush=True)

if __name__=='__main__':
    run([('qual-%03d'%n,qualify(n)) for n in range(128)],'qual')
