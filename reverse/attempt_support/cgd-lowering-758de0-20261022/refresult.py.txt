from pathlib import Path
from itertools import product
W=Path(__file__).resolve().parent;base=(W/'base.cpp').read_text();out=W/'refresult';out.mkdir(exist_ok=True)
for i,(cv,shape,pc,tc,rc,nt) in enumerate(product(['','const ','volatile ','const volatile '],range(2),range(2),range(2),range(2),range(2))):
 s=base.replace('struct ContactSource {','struct ContactInfo { unsigned data; };\nstruct ContactSource {')
 typ=cv+('ContactInfo &' if shape==0 else 'void *'+cv+'&')
 s=s.replace('virtual void *prepare(const ContactVector &point, const ContactVector &normal)=0;', 'virtual '+typ+' prepare(const ContactVector &point, const ContactVector &normal)'+(' const' if pc else '')+(' throw()' if nt else '')+'=0;')
 s=s.replace('virtual bool test(void *info)=0;', 'virtual bool test('+typ+'info)'+(' const' if tc else '')+'=0;')
 s=s.replace('virtual void respond(void *info)=0;', 'virtual void respond('+typ+'info)'+(' const' if rc else '')+'=0;')
 (out/('v%04d.cpp'%i)).write_text(s)
print('generated',i+1)
