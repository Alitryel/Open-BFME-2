from pathlib import Path
import sys,json,hashlib,difflib,collections
root=Path(__file__).resolve().parents[1];sys.path.insert(0,str(root/'tools'));import build,align_diff,capstone
out=root/'build/shroud-stlport-bitset-oct20';name='?onComingOutOfShroud@Drawable@@QAEXXZ'
md=capstone.Cs(capstone.CS_ARCH_X86,capstone.CS_MODE_32);native=align_diff.disasm(md,build.read_target_bytes(0x27893e,747));seen={};results=[]
for source in out.glob('*.cpp'):
    if source.name=='seed.cpp':continue
    obj=build.obj_path(source)
    if not obj.exists():continue
    try:raw,rels=build.read_object_symbol_bytes(obj,name,code_only=True)
    except Exception:continue
    sha=hashlib.sha256(raw).hexdigest()
    if sha in seen:seen[sha]['duplicates'].append(source.name);continue
    code=align_diff.disasm(md,raw);match=difflib.SequenceMatcher(a=[x[1] for x in native],b=[x[1] for x in code],autojunk=False)
    kinds=collections.Counter();regions=[];residue=[]
    for tag,i,j,k,l in match.get_opcodes():
        if tag!='equal':regions.append({'kind':tag,'retail':[list(x) for x in native[i:j]],'compiled':[list(x) for x in code[k:l]]});continue
        for a,b in zip(native[i:j],code[k:l]):
            if a[2]==b[2]:continue
            if any(b[0]<o+4 and o<b[0]+b[3] for o,t,n in rels):kinds['reloc']+=1;continue
            kind=align_diff.classify(a[2],b[2]);kinds[kind]+=1
            if kind not in ('branch','reloc'):residue.append({'kind':kind,'retail':a,'compiled':b})
    d={'source':source.name,'size':len(raw),'structural':len(regions),'kinds':dict(kinds),'residue':residue,'regions':regions,'duplicates':[],'sha256':sha}
    seen[sha]=d;results.append(d)
results.sort(key=lambda x:(x['structural'],sum(x['kinds'].get(k,0) for k in ['register','memory']),abs(x['size']-747)))
(out/'metrics.json').write_text(json.dumps(results,indent=2))
print('Unique parent objects',len(results))
for x in results[:18]:print(x['source'],x['size'],x['structural'],x['kinds'],'duplicate controls',len(x['duplicates']))
