// cl: /O1 /GS- /MD
extern "C" unsigned __cdecl _udiv64(unsigned __int64,unsigned,unsigned *);
extern "C" unsigned __int64 __cdecl __emulu(unsigned,unsigned);
extern "C" unsigned char __cdecl _addcarry_u32(unsigned char,unsigned,unsigned,unsigned *);
#pragma intrinsic(_udiv64,__emulu,_addcarry_u32)
union Wide { unsigned __int64 value; struct { unsigned low,high; } words; };
extern "C" unsigned __int64 __stdcall _aulldiv(unsigned __int64 n,unsigned __int64 d) {
 Wide a,b,result; a.value=n;b.value=d;
 if(!b.words.high) {
  unsigned rem,unused;
  result.words.high=_udiv64(a.words.high,b.words.low,&rem);
  Wide small;small.words.high=rem;small.words.low=a.words.low;
  result.words.low=_udiv64(small.value,b.words.low,&unused);
  return result.value;
 }
 Wide nn=a,dd=b;
 do { dd.value>>=1;nn.value>>=1; } while(dd.words.high);
 unsigned unused;
 unsigned q=_udiv64(nn.value,dd.words.low,&unused);
 unsigned cross=(unsigned)__emulu(q,b.words.high);
 Wide product;product.value=__emulu(q,b.words.low);
 unsigned high;
 if(_addcarry_u32(0,product.words.high,cross,&high)) --q;
 else if(high>a.words.high || (high==a.words.high && product.words.low>a.words.low)) --q;
 return q;
}
