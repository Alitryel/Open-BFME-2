from pathlib import Path
import sys,itertools,concurrent.futures,json
sys.path.insert(0,'build/df20-scalars-20261020');import trials
trials.scratch=Path.cwd()/'build/df20-scalars-20261020'
base=(trials.scratch/'range-x0-specialized-b1-r0-ttemporary.cpp').read_text();trials.symbol=base.splitlines()[0][3:]
first='''  if (__digits.size() == 0)
#pragma inline_depth(0)
    return __s;
#pragma inline_depth(255)'''
second='''  if (__cp == __digits_first)
#pragma inline_depth(1)
    return __s;
#pragma inline_depth(255)'''
specs=[]
for negint,unsigned,f,s,inside in itertools.product([0,1],repeat=5):
 if not f and not s and not inside:continue
 src=base
 if negint:src=src.replace('  bool __is_negative;','  int __is_negative;')
 if unsigned:src=src.replace('for (int __i = 0;','for (unsigned int __i = 0;')
 if f or s:
  start=src.index('#pragma inline_depth(0)\n  string_type __curr_sym');src=src[:start]+'  {\n'+src[start:]
  if f:src=src.replace(first,first.replace('return __s','goto money_exit'))
  if s:src=src.replace(second,second.replace('return __s','goto money_exit'))
  pos=src.rfind('  return __s;\n}')
  src=src[:pos]+src[pos:].replace('  return __s;\n}',('  return __s;\n' if inside else '')+'  }\nmoney_exit:\n  return __s;\n}',1)
 specs.append((f'recombine-n{negint}-u{unsigned}-f{f}-s{s}-i{inside}',src))
with concurrent.futures.ThreadPoolExecutor(max_workers=8) as p:r=list(p.map(trials.run,specs))
(trials.scratch/'recombine.json').write_text(json.dumps(r,indent=2))
