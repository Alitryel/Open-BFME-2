from pathlib import Path
import runpy,itertools,concurrent.futures,json
OUT=Path('build/canmake-joins-20261023/new-profiles').resolve();OUT.mkdir(parents=True,exist_ok=True)
r=runpy.run_path('reverse/attempt_support/canmake-00596e15-20261022/replay.py.txt');r['run'].__globals__['OUT']=OUT
base=runpy.run_path('reverse/attempt_support/canmake-alias-20261023/context.py.txt')['bank_text']()
tests=[]
for profile,og,os,ob,eh,scope in itertools.product(range(3),range(2),range(2),range(2),range(2),range(2)):
 b=base.replace('/EHsc',['/EHsc','/EHs'][eh]+' '+['/Og','/Og-'][og]+' '+['/Os','/Ot'][os]+' '+['/Ob1','/Ob2'][ob],1)
 if profile:b='// compiler: '+['','vc71-sp1','vc71-ddk-4035'][profile]+'\n'+b
 if scope:b=b.replace('MakeCodeLife selected;','').replace('Rva0039205C result;','MakeCodeLife selected;\nRva0039205C result;')
 tests.append((f'profile-{profile}{og}{os}{ob}{eh}{scope}',b))
import os
case=os.environ.get('CANMAKE_CASE')
if case:tests=[x for x in tests if x[0]==case];assert tests,case
results=[]
with concurrent.futures.ThreadPoolExecutor(max_workers=8)as pool:
 for i,d in enumerate(pool.map(r['run'],tests),1):
  results.append(d)
  if i%48==0 or d.get('exact_candidate'):print(json.dumps(dict(done=i,total=len(tests),best=sorted(results,key=lambda x:(x.get('nd',999),-x.get('score',0)))[:1])),flush=True)
(OUT/'results.json').write_text(json.dumps(results,indent=2)+'\n')
print(json.dumps(dict(trials=len(results),errors=sum('error'in x for x in results),unresolved=sum(bool(x.get('unresolved'))for x in results),exact=sum(x.get('exact_candidate',False)for x in results),bodies=len({x.get('masked_code_sha256')for x in results}))),flush=True)
