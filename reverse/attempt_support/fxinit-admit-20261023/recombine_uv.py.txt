import sys,json,time
from pathlib import Path
from concurrent.futures import ThreadPoolExecutor,as_completed
sys.path.insert(0,'build/fxinit-live22');import uvvisibility as u
h=u.h;h.HERE=Path('build/fxinit-admit23').resolve()
def variants():
 for base in ['valuebundlesafe_0000110','recombineview_101000','recombineview_002000']:
  h.BASE=(h.HERE/'trials'/(base+'.cpp')).read_text()
  for label,src in u.variants():yield base+'_'+label,src
if __name__=='__main__':
 t=time.time();items=list(variants());rs=[]
 with ThreadPoolExecutor(max_workers=8)as ex:
  for f in as_completed([ex.submit(h.run_one,x)for x in items]):rs.append(f.result())
 rs.sort(key=lambda z:(z.get('nd',99999),z.get('cost',99999)))
 (h.HERE/'recombineuv-results.json').write_text(json.dumps(rs,indent=2))
 print(json.dumps({'count':len(rs),'seconds':time.time()-t,'errors':sum('error'in z for z in rs),'best':rs[:6]},indent=2))
