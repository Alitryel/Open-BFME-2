import json
for f in ['base','diag']:
 ss=json.load(open('build/fxinit-split22/block'+f+'.obj.blocks.json'))
 print(f)
 for n,s in enumerate(ss):
  prev=ss[n-1]['tuples'] if n else {}
  print(s['ip'],'sets',s['sets'],'added tuples:')
  sy=s['symbols'];op=s['operands']
  def operand(p):
   a=op.get(str(p),[])
   if not a:return '?'
   k=a[2]&255;v=a[6];w=sy.get(str(v),[])
   if k==7:return 'imm('+str(v)+')'
   if k in [1,2]:return ('reg'if k==1 else'mem')+'('+hex(v)+('/serial'+str(w[7]) if len(w)>7 else'')+')'
   return 'kind'+str(k)+'('+hex(v)+')'
  def args(p):
   out=[];seen=set()
   while p and p not in seen:
    seen.add(p);out.append(operand(p));w=op.get(str(p),[]);p=w[0] if w else 0
   return ','.join(out)
  for p,w in s['tuples'].items():
   if p in prev or n==0:continue
   print(hex(int(p)),hex(w[1]),args(w[10]),'->',args(w[11]))
