from pathlib import Path
import sys,itertools,concurrent.futures,json
sys.path.insert(0,'build/df20-scalars-20261020');import trials
trials.scratch=Path.cwd()/'build/df20-scalars-20261020'
base=Path('build/df20-scalars-20261020/bank.cpp').read_text();trials.symbol=base.splitlines()[0][3:]
marker='#pragma inline_depth(255)\n}\nnamespace _STL {\ntypedef ostreambuf_iterator'
specs=[]
for body,depth,tag,temporary in itertools.product(['primary','two-arg'],[0,255],['named','temporary'],['reference','copy']):
 if body=='primary' and (depth!=0 or tag!='named' or temporary!='reference'):continue
 s=base.replace('// cl:','// cl: /D_STLP_NEEDS_EXTRA_TEMPLATE_CONSTRUCTORS',1)
 if body=='two-arg':
  tagdecl='forward_iterator_tag __tag;' if tag=='named' else ''
  tagexpr='__tag' if tag=='named' else 'forward_iterator_tag()'
  # The vendor's two-argument constructor passes a temporary allocator to the base.
  if temporary=='reference':init='allocator<wchar_t>()'
  else:init='allocator<wchar_t>(allocator<wchar_t>())'
  definition=f'''#pragma inline_depth({depth})
 template <> template <> __forceinline basic_string<wchar_t>::basic_string(const wchar_t* first,const wchar_t* last) : _String_base<wchar_t,allocator<wchar_t> >({init}) {{ {tagdecl} _M_range_initialize(first,last,{tagexpr}); }}
'''
  assert marker in s
  s=s.replace(marker,definition+marker,1)
 specs.append((f'constructor-{body}-d{depth}-t{tag}-a{temporary}',s))
with concurrent.futures.ThreadPoolExecutor(max_workers=8) as p:r=list(p.map(trials.run,specs))
(trials.scratch/'constructors.json').write_text(json.dumps(r,indent=2))
