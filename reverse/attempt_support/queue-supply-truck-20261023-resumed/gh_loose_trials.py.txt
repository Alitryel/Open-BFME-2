from pathlib import Path
import re,json,concurrent.futures
from probe import measure
root=Path(__file__).parent;jobs=[]
for control in ('guard-gh','clean-gh'):
 base=(root/(control+'.cpp')).read_text();cut=base.index('/* See if we have any "loose" harvesters')
 prefix,loose=base[:cut],base[cut:]
 mods={
  'sta_volatile':loose.replace('SupplyTruckAIInterface* supplyTruckAI','SupplyTruckAIInterface* volatile supplyTruckAI'),
  'sta_volatile_first':loose.replace('supplyTruckAI->getPreferredDockID()','(*(SupplyTruckAIInterface *volatile *)&supplyTruckAI)->getPreferredDockID()'),
  'sta_volatile_ferry':loose.replace('supplyTruckAI->isCurrentlyFerryingSupplies()','(*(SupplyTruckAIInterface *volatile *)&supplyTruckAI)->isCurrentlyFerryingSupplies()'),
  'sta_volatile_force':loose.replace('supplyTruckAI->isForcedIntoWantingState()','(*(SupplyTruckAIInterface *volatile *)&supplyTruckAI)->isForcedIntoWantingState()'),
  'obj_first_volatile':loose.replace('obj->getTemplate()','(*(Object *volatile *)&obj)->getTemplate()'),
  'obj_dock_volatile':loose.replace('obj->getAI()->getCommandInterface()','(*(Object *volatile *)&obj)->getAI()->getCommandInterface()'),
  'dock_volatile':loose.replace('ObjectID dock =','volatile ObjectID dock ='),
  'desired_limit':loose.replace('desiredGatherers*3','*(volatile Int *)&desiredGatherers*3'),
 }
 for tag,part in mods.items():
  p=root/('ghloose_'+control+'_'+tag+'.cpp');p.write_text(prefix+part);jobs.append(p)
def run(p):
 r=measure(p);p.with_suffix('.result.json').write_text(json.dumps(r,indent=2));print({k:v for k,v in r.items() if k not in ('body_hex','relocs')},flush=True)
with concurrent.futures.ThreadPoolExecutor(max_workers=3) as pool:list(pool.map(run,jobs))
