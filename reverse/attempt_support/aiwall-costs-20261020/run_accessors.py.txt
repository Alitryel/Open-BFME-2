from pathlib import Path
import sys,re,json,itertools
ROOT=Path(__file__).resolve().parents[2]
sys.path.insert(0,str(ROOT/'tools'))
from variant_search import Scorer
out=Path(__file__).parent/'accessors'
s=Scorer('?distSqToSegment@@YAMPBVCoord2D@@00@Z',0x4eacf5,197,out)
base=(ROOT/'reverse/attempts/0x004eadba.cpp').read_text().replace('../../Code/',str(ROOT/'Code').replace('\\','/')+'/')
definition='''
struct SegmentVector : Coord2D {
 __forceinline float getX() const { return x; }
 __forceinline float getY() const { return y; }
 __forceinline void setX(float v) { x=v; }
 __forceinline void setY(float v) { y=v; }
 __forceinline void set(float a,float b) { x=a;y=b; }
 __forceinline void subX(float v) { x-=v; }
 __forceinline void subY(float v) { y-=v; }
 __forceinline float &refX() { return x; }
 __forceinline float &refY() { return y; }
 __forceinline void difference(const Coord2D &a,const Coord2D &b) {x=a.x-b.x;y=a.y-b.y;}
 __forceinline float dot(const SegmentVector &other) const { return y*other.y+x*other.x; }
};
'''
rows=[]
for apwrite,apread,dirwrite,dirread in itertools.product(range(4),range(3),range(3),range(2)):
 text=base
 head,body=text.split('static __declspec(noinline) float distSqToSegment',1)
 a=(['ap.x','ap.getX()','ap.refX()'][apread],['ap.y','ap.getY()','ap.refY()'][apread])
 d=(['dir.x','dir.getX()'][dirread],['dir.y','dir.getY()'][dirread])
 body=re.sub(r'\bapx\b',a[0],body);body=re.sub(r'\bapy\b',a[1],body)
 body=re.sub(r'\bdx\b',d[0],body);body=re.sub(r'\bdy\b',d[1],body)
 old='\tfloat '+d[0]+' = lineEnd->x;\n\tfloat '+d[1]+' = lineEnd->y;'
 new=('\tdir.x=lineEnd->x;\n\tdir.y=lineEnd->y;' if dirwrite==0 else '\tdir.setX(lineEnd->x);\n\tdir.setY(lineEnd->y);' if dirwrite==1 else '\tdir.set(lineEnd->x,lineEnd->y);')
 assert old in body
 body=body.replace(old,new)
 old='\tfloat '+a[0]+' = pt->x - lineStart->x;\n\tfloat '+a[1]+' = pt->y - lineStart->y;'
 new=('\tap.x=pt->x-lineStart->x;\n\tap.y=pt->y-lineStart->y;' if apwrite==0 else '\tap.setX(pt->x-lineStart->x);\n\tap.setY(pt->y-lineStart->y);' if apwrite==1 else '\tap.set(pt->x-lineStart->x,pt->y-lineStart->y);' if apwrite==2 else '\tap.difference(*pt,*lineStart);')
 assert old in body
 body=body.replace(old,new)
 body=body.replace(d[0]+' -= lineStart->x;','dir.subX(lineStart->x);' if dirwrite else 'dir.x -= lineStart->x;')
 body=body.replace(d[1]+' -= lineStart->y;','dir.subY(lineStart->y);' if dirwrite else 'dir.y -= lineStart->y;')
 body=body.replace('\n{\n','\n{\n\tSegmentVector ap,dir;\n',1)
 text=head+definition+'static __declspec(noinline) float distSqToSegment'+body
 r=s(text);got=r.get('compiled',b'');target=r.get('target',b'')
 row={'forms':[apwrite,apread,dirwrite,dirread],'file':f'c{s.n}.cpp','size':len(got),'diff':sum(a!=b for a,b in zip(got,target))+abs(len(got)-len(target)),'exact':r['exact'],'fitness':r['fitness'],'class':r['class']}
 rows.append(row)
 if s.n==1 or r['exact'] or row['diff']<15:print(json.dumps(row),flush=True)
 if r['exact']:(out/'win.cpp').write_bytes(text.encode());break
(out/'results.json').write_text(json.dumps(rows,indent=2))
print('completed',len(rows),'best',min(rows,key=lambda r:(not r['exact'],-r['fitness'])),flush=True)
