from search import *
import re,itertools

def source(flags):
 gamemask,heromask,teamflag=flags
 body=BODY
 j=0
 def game_read(m):
  nonlocal j
  i=j;j+=1
  if not i:return 'game'
  return '((GameInfo *const volatile &)game)'if gamemask&(1<<(i-1))else 'game'
 body=re.sub(r'\bgame\b',game_read,body)
 j=0
 def hero_read(m):
  nonlocal j
  i=j;j+=1
  if not i:return 'hero'
  return '((GameWindow *const volatile &)hero)'if heromask&(1<<(i-1))else 'hero'
 body=re.sub(r'\bhero\b',hero_read,body)
 if teamflag:body=body.replace('if (*team)','if (*((GameWindow **const volatile &)team))')
 return HEAD+MARK+body
if __name__=='__main__':
 g=len(re.findall(r'\bgame\b',BODY))-1;h=len(re.findall(r'\bhero\b',BODY))-1
 print('reads',g,h,flush=True)
 flags=[(m,0,0)for m in range(1<<g)] + [(0,m,t)for m,t in itertools.product(range(1<<h),range(2))]
 run([('localread-%04d'%n,source(f))for n,f in enumerate(flags)],'local_reads')
