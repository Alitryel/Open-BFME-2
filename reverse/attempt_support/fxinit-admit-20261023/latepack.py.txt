import sys,itertools,json,time
from pathlib import Path
from concurrent.futures import ThreadPoolExecutor,as_completed
sys.path.insert(0,'build/fxinit-seat22');import search as h
h.HERE=Path('build/fxinit-admit23').resolve()
R,W,P,N=h.R,h.W,h.P,h.N
def pack(k,reverse):
 a,b='remap[links[0][i]]','remap[links[1][i]]'
 pre=''
 if k in [1,2,3,4,5,6,7]:
  ty=['','int','unsigned','__int64','unsigned __int64','short','const int &','const short &'][k]
  parts=[f'{ty} a={a};',f'{ty} b={b};']
  if reverse:parts.reverse()
  pre=''.join(parts);a,b='a','b'
 e=f'{a}+({b}<<8)'
 if k in [1,5]:e=f'{a}+{b}*256'
 if k==2:e=f'({a}+({b}<<8))'
 return '{'+pre+'*(int *)(vertices+desc.Get_Basis_End())=(int)('+e+');}'
def variants():
 # Source-distinct remap calculations (32/64-bit, reference/scalar and shift/
 # multiplication) under identical-effect predicates. Unlike earlier duplicate
 # P/N-copy trials, only the integer pack computation differs between arms.
 for kind,cond,scope,reverse,nested,invert in itertools.product(range(8),range(4),range(2),range(2),range(2),range(2)):
  pred=['positions[1]','normals[1]','links[1]','(m_FVF&0x10)'][cond]
  alt=pack(kind,reverse);orig=R
  if scope:alt+=W;orig+=W
  if nested:alt='if(basis){'+alt+'}else{'+orig+'}'
  if invert:pred='!('+pred+')';orig,alt=alt,orig
  new='if('+pred+'){'+orig+'}else{'+alt+'}'
  old=R+'\n      '+W if scope else R
  s=h.BASE.replace(old,new)
  yield f'latepack_{kind}{cond}{scope}{reverse}{nested}{invert}',s
if __name__=='__main__':
 t=time.time();rs=[]
 with ThreadPoolExecutor(max_workers=8)as ex:
  for f in as_completed([ex.submit(h.run_one,x)for x in variants()]):
   rs.append(f.result())
   if len(rs)%128==0:print('tested',len(rs),'best',min(rs,key=lambda z:(z.get('nd',99999),z.get('cost',99999))),flush=True)
 rs.sort(key=lambda z:(z.get('nd',99999),z.get('cost',99999)))
 (h.HERE/'latepack-results.json').write_text(json.dumps(rs,indent=2))
 print(json.dumps({'count':len(rs),'seconds':time.time()-t,'errors':sum('error'in z for z in rs),'best':rs[:6]},indent=2))
