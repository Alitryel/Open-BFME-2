from pathlib import Path

root=Path(__file__).parent
out=root/'field_reads';out.mkdir(exist_ok=True)
bases=[Path('reverse/attempts/0x0027893e.cpp').read_text(),Path('build/drawable27893e-20261022b/native_tail/e0n0f0s1.cpp').read_text()]
for bi,base in enumerate(bases):
    # Individual reads retain the same target field, width and runtime count.
    # Volatile is a codegen adaptation, never a claim about target layout.
    for bits in range(128):
        source=base
        source=source.replace('bool isEffectivelyDead() const { return (m_privateStatus & 1) != 0; }','bool isEffectivelyDead() const { return (m_privateStatus & 1) != 0; }\n bool readDeadInView() const { return (*(const volatile unsigned char *)&m_privateStatus & 1) != 0; }')
        start=source.index('void Drawable::onComingOutOfShroud()')
        head,body=source[:start],source[start:]
        pieces=body.split('obj->isEffectivelyDead()')
        assert len(pieces)==4
        body=pieces[0]
        for i,piece in enumerate(pieces[1:]):
            body+=('obj->readDeadInView()' if bits&(1<<i) else 'obj->isEffectivelyDead()')+piece
        if bits&8:body=body.replace('!(m_status & 0x40)','!(*(const volatile unsigned int *)&m_status & 0x40)').replace('!(castleDraw->m_status & 0x40)','!(*(const volatile unsigned int *)&castleDraw->m_status & 0x40)')
        if bits&16:body=body.replace('!(m_status & mask80)','!(*(const volatile unsigned int *)&m_status & mask80)')
        if bits&32:body=body.replace('m_status |= mask80','m_status = *(const volatile unsigned int *)&m_status | mask80')
        if bits&64:body=body.replace('!m_sightReported','!*(const volatile bool *)&m_sightReported').replace('m_useAltSightedEvent &&','*(const volatile bool *)&m_useAltSightedEvent &&')
        (out/f'b{bi}r{bits:03d}.cpp').write_text(head+body)
print(len(list(out.glob('*.cpp'))))
