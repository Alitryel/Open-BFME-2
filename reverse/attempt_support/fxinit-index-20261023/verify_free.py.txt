import sys,json
from pathlib import Path
sys.path.insert(0,'build/fxinit-seat22');import search as h
import eh_verify
here=Path('build/fxinit-index22');b=h.build.read_object_symbol_bytes(here/'trials/bank_control.obj',h.NAME,code_only=True)
results=json.loads((here/'free-results.json').read_text());count=0
for r in results:
 if r['nd']==9:
  assert not r['unresolved'] and h.build.read_object_symbol_bytes(here/'trials'/(r['id']+'.obj'),h.NAME,code_only=True)==b;count+=1
out={'bank_equal_controls':count,'helpers':{}}
for label in ['freevis_001','freevis_002','freevis_111','freevis_331']:
 obj=here/'trials'/(label+'.obj');out['helpers'][label]={}
 for name,rva,size in [('??1Rva0018C262@@QAE@XZ',0x18c3e6,56),('?free@@YAXPAX@Z',0x30830,17),('?Rva00030830GameFree@@YAXPAX@Z',0x30830,17)]:
  row={'name':name,'source':str(obj.with_suffix('.cpp')),'target_rva':hex(rva),'target_size':str(size),'notes':''}
  try:
   z=h.build.compile_function(row,h.MAP,obj)
   x={'size':len(z['bytes']),'exact':z['bytes']==h.build.read_target_bytes(rva,size),'unresolved':z['unresolved']}
   if rva==0x18c3e6:x['eh']=eh_verify.verify_row(eh_verify.Image(),eh_verify.Obj(obj),name,rva)
  except Exception as e:x={'error':str(e)}
  out['helpers'][label][name]=x
(here/'free-verification.json').write_text(json.dumps(out,indent=2)+'\n')
print(json.dumps(out,indent=2))
