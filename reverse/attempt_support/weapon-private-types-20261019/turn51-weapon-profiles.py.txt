from pathlib import Path
import subprocess,sys,json
out=Path('build/resume-eh-local')
s=Path('reverse/attempts/0x002ce30a.cpp').read_text()
begin=s.index('struct Coord3DBase {')
end=s.index('class Matrix3D',begin)
header='#define COORD3D_DV_CTOR_INLINE\n#define COORD3D_DV_COPY_INLINE\n#define COORD3D_DV_DTOR_INLINE\n#include "Code/Libraries/Include/Lib/Coord3DDerivedView.h"\n'
view=s[:begin]+header+s[end:]
variants={'shared-view':view,'ddk':'// compiler: vc71-ddk-4035\n'+view,'sp1':'// compiler: vc71-sp1\n'+view}
sym='?privateFireWeapon@Weapon@@QAE_NPBVObject@@PBUCoord3D@@0H1_N2IPAH@Z'
rs=[]
for name,src in variants.items():
 p=out/('turn51-weapon-'+name+'.cpp');p.write_text(src)
 proc=subprocess.run([sys.executable,'tools/align_diff.py',str(p),sym,'0x002ce30a','955'],stdout=subprocess.PIPE,stderr=subprocess.STDOUT,text=True)
 (out/('turn51-weapon-'+name+'.txt')).write_text(proc.stdout)
 lines=[l for l in proc.stdout.splitlines() if any(x in l.lower() for x in ('structural','memory','register','error','sub esp','lea edi','lea eax'))]
 print(name,proc.returncode,'\n'.join(lines[:14]),flush=True);rs.append({'name':name,'exit':proc.returncode,'summary':lines})
(out/'turn51-weapon-profiles-results.json').write_text(json.dumps(rs,indent=2))
