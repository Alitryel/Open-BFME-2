from search import *
import itertools

# WB's stack layout does not follow the bank's declaration order. Test whether
# front-end symbol ordering (spelling versus block scope) survives into c2.
pairs=[('foundation','foundationObject'),('obj','newObject'),('md','moduleData'),
 ('member','castleMember'),('member2','foundationMember'),('orientation','newAngle'),
 ('newMember','newCastleMember'),('oldMember','oldCastleMember'),('cost','buildCost'),
 ('self','foundationObj')]
tests={}
for mask in range(1024):
 s=BASE
 for i,(a,b) in enumerate(pairs):
  if mask&(1<<i):s=re.sub(r'\b'+a+r'\b',b,s)
 tests[f'names-{mask:04d}']=s
run(tests,'namescope')
