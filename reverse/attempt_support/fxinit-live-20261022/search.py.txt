import sys,json,itertools,time
from pathlib import Path
from concurrent.futures import ThreadPoolExecutor,as_completed
sys.path.insert(0,'build/fxinit-seat22')
import search as h
h.HERE=Path('build/fxinit-live22').resolve()
BASE=h.BASE
LOOP='   for(unsigned i=0;i<(unsigned)mesh->Get_Vertex_Count();++i) {'

def variants(mode):
 if mode=='groupcursor':
  streams=['positions[0]','normals[0]','tangents','binormals','positions[1]','normals[1]']
  for anchor,mask,typ,scope,deltaform in itertools.product(range(2),range(1,64),range(2),range(2),range(2)):
   src=BASE;origin=['binormals','tangents'][anchor]
   cursor='vectorCursor';expr='(unsigned)'+origin+'+i*sizeof(Vector3)'
   cexpr=cursor if typ==0 else '(unsigned)'+cursor
   init=('unsigned '+cursor+'='+expr+';' if typ==0 else 'const char *'+cursor+'=(const char *)('+expr+');')
   if scope:
    expr='(unsigned)'+origin
    init=('unsigned '+cursor+'='+expr+';' if typ==0 else 'const char *'+cursor+'=(const char *)('+expr+');')
   lo=src.index(LOOP);hi=src.index(' m_IB=new',lo);core=src[lo:hi]
   for k,stream in enumerate(streams):
    if mask>>k&1:
     if deltaform==0:e=f'{cexpr}+((unsigned){stream}-(unsigned){origin})'
     else:e=f'({cexpr}-(unsigned){origin})+(unsigned){stream}'
     core=core.replace(stream+'[i]','*(const Vector3 *)('+e+')')
   if scope:
    core='   '+init+'\n'+core
    advance=cursor+'=('+('unsigned'if typ==0 else'const char *')+')('+cexpr+'+sizeof(Vector3));'
    core=core.replace('    vertices+=desc.Get_FVF_Size();','    vertices+=desc.Get_FVF_Size();'+advance)
   else:core=core.replace(LOOP,LOOP+'\n'+init)
   src=src[:lo]+core+src[hi:]
   yield f'groupcursor_{anchor}_{mask:02}_{typ}{scope}{deltaform}',src
 if mode=='tailclone':
  for depth,vform,iform,order,prep in itertools.product(range(4),range(4),range(3),range(2),range(3)):
   src=BASE;tail=[];post=[];decl=''
   if vform==0:tail.append('vertices+=desc.Get_FVF_Size();')
   elif vform==1:
    decl+='char *nextVertex;';tail.append('nextVertex=vertices+desc.Get_FVF_Size();');post.append('vertices=nextVertex;')
   elif vform==2:
    decl+='unsigned nextVertex;';tail.append('nextVertex=(unsigned)vertices+desc.Get_FVF_Size();');post.append('vertices=(char *)nextVertex;')
   else:tail.append('vertices=desc.Get_FVF_Size()+vertices;')
   if iform==1:tail.append('++i;');src=src.replace(LOOP,LOOP.replace(';++i)', ';)'))
   if iform==2:
    decl+='unsigned nextIndex;';tail.append('nextIndex=i+1;');post.append('i=nextIndex;');src=src.replace(LOOP,LOOP.replace(';++i)', ';)'))
   if order:tail.reverse();post.reverse()
   tail=''.join(tail)
   # Repeated common-tail work is applied only after all vertex uses.
   normal=h.N
   if depth>=3:normal='if(normals[1]) { *(Vector3 *)(vertices+desc.Get_Normal_Offset(1))=normals[1][i];'+tail+'} else {'+tail+'}'
   two=h.R+'\n'+h.W+'\n'+h.P+'\n'+normal
   if depth==2:two+=tail
   one='*(int *)(vertices+desc.Get_Basis_End())=remap[links[0][i]];'
   inner='if(m_BonesPerVertex==1) {'+one+(tail if depth>=2 else '')+'} else if(m_BonesPerVertex==2) {'+two+'}'
   if depth>=2:inner+=' else {'+tail+'}'
   if depth==1:inner+=tail
   arm='if(m_Passes>1) { *(unsigned *)(vertices+desc.Get_Basis_End())=pass;'+(tail if depth>=1 else '')+'} else if(m_BonesPerVertex>0 && links[0]) {'+inner+'}'
   if depth>=1:arm+='else {'+tail+'}'
   if depth==0:arm+=tail
   if prep==1:
    # Stable stride address, read only at the original tail use.
    decl+='const unsigned *vertexStride=&desc.stride;';arm=arm.replace('desc.Get_FVF_Size()','*vertexStride')
   if prep==2:
    decl+='const unsigned &vertexStride=desc.stride;';arm=arm.replace('desc.Get_FVF_Size()','vertexStride')
   lo=src.index('    if(m_Passes>1) *(unsigned *)');hi=src.index('    vertices+=desc.Get_FVF_Size();',lo)+len('    vertices+=desc.Get_FVF_Size();')
   src=src[:lo]+decl+'\n'+arm+''.join(post)+src[hi:]
   yield f'tailclone_{depth}{vform}{iform}{order}{prep}',src
 if mode=='streamloops':
  streams=[('tangents','desc.Get_Basis_Offset()'),('binormals','desc.Get_Basis_Second_Offset()'),('positions[1]','desc.Get_Location_Offset(1)'),('normals[1]','desc.Get_Normal_Offset(1)')]
  for forms in itertools.product(range(6),repeat=4):
   src=BASE
   for k,(stream,offset) in enumerate(streams):
    old=f'if({stream}) *(Vector3 *)(vertices+{offset})={stream}[i];'
    p='optionalStream'+str(k);use=f'*(Vector3 *)(vertices+{offset})={p}[i];'
    form=forms[k]
    if form==0:new=old
    if form==1:new=f'for(const Vector3 *{p}={stream};{p};{p}=0) {{ {use} }}'
    if form==2:new=f'{{const Vector3 *{p}={stream};while({p}) {{ {use} {p}=0; }} }}'
    if form==3:new=f'{{const Vector3 *{p}={stream};do {{ if(!{p}) break; {use} }}while(false); }}'
    if form==4:new=f'for(const Vector3 *{p}={stream};{p};) {{ {use} break; }}'
    if form==5:new=f'{{const Vector3 *{p}={stream};if({p}) do {{ {use} {p}=0; }}while({p}); }}'
    assert old in src;src=src.replace(old,new)
   yield 'streamloops_'+''.join(map(str,forms)),src
 if mode=='byteivs':
  streams=['positions[0]','normals[0]','tangents','binormals','positions[1]','normals[1]']
  for mask,ty,scope,update in itertools.product(range(1,128),range(2),range(2),range(2)):
   src=BASE;typ=['unsigned','int'][ty];decl=[];inc=[]
   if mask&63:decl.append(typ+' vertexBytes=0;');inc.append('vertexBytes+=12')
   if mask&64:decl.append(typ+' linkBytes=0;');inc.append('linkBytes+=2')
   lo=src.index(LOOP);hi=src.index(' m_IB=new',lo);core=src[lo:hi]
   for k,stream in enumerate(streams):
    if mask>>k&1:core=core.replace(stream+'[i]','*(const Vector3 *)((const char *)'+stream+'+vertexBytes)')
   if mask&64:
    for stream in ['links[0]','links[1]','weights[0]','weights[1]']:
     core=core.replace(stream+'[i]','*(const unsigned short *)((const char *)'+stream+'+linkBytes)')
   if update==0:core=core.replace(';++i)', ';++i,'+','.join(inc)+')',1)
   else:core=core.replace('    vertices+=desc.Get_FVF_Size();','    vertices+=desc.Get_FVF_Size();\n'+ ';'.join(inc)+';')
   if scope==0:core='   '+''.join(decl)+'\n'+core
   else:
    src=src[:lo].replace('  for(unsigned pass=0;', '  '+''.join(decl)+'\n  for(unsigned pass=0;')+src[lo:]
    lo=src.index(LOOP);hi=src.index(' m_IB=new',lo)
    core='   '+''.join(x.split(' ',1)[1] for x in decl)+'\n'+core
   src=src[:lo]+core+src[hi:]
   yield f'byteivs_{mask:03}_{ty}{scope}{update}',src

if __name__=='__main__':
 mode=sys.argv[1];items=list(variants(mode));out=[];start=time.time()
 with ThreadPoolExecutor(max_workers=8) as pool:
  for fut in as_completed([pool.submit(h.run_one,x) for x in items]):
   r=fut.result();out.append(r)
 out.sort(key=lambda r:(r.get('nd',99999),r.get('cost',99999)))
 (h.HERE/(mode+'-results.json')).write_text(json.dumps(out,indent=2))
 print(json.dumps({'mode':mode,'count':len(out),'seconds':time.time()-start,'valid':sum('nd'in r and not r.get('unresolved') for r in out),'best':out[:6]},indent=2))
