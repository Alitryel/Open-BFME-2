# bh.py OBJ... -> group by hash of queueSupplyTruck bytes (section raw data, function range)
import struct,sys,hashlib,collections
SYM=b'?queueSupplyTruck@AIPlayer@@IAEXXZ'
def fbytes(p):
    d=open(p,'rb').read()
    nsec,=struct.unpack_from('<H',d,2); symp,nsym=struct.unpack_from('<II',d,8)
    secs=[struct.unpack_from('<8sIIIIIIHHI',d,20+40*i) for i in range(nsec)]
    strt=symp+18*nsym
    i=0; syms=[]
    while i<nsym:
        e=d[symp+18*i:symp+18*i+18]; nm=e[:8]
        if nm[:4]==b'\0\0\0\0':
            off,=struct.unpack_from('<I',nm,4); s=d[strt+off:d.index(b'\0',strt+off)]
        else: s=nm.rstrip(b'\0')
        val,sec=struct.unpack_from('<Ih',e,8); naux=e[17]
        syms.append((s,val,sec,e[16])); i+=1+naux
    me=[x for x in syms if x[0]==SYM and x[2]>0][0]
    sec=secs[me[2]-1]
    nxt=[x[1] for x in syms if x[2]==me[2] and x[1]>me[1] and x[3]==2]
    end=min(nxt) if nxt else sec[3]
    raw=d[sec[4]+me[1]:sec[4]+end]
    return raw
g=collections.defaultdict(list)
for p in sys.argv[1:]:
    try: r=fbytes(p); k=hashlib.md5(r).hexdigest()[:8]+' %d'%len(r)
    except Exception as e: k='FAIL'
    g[k].append(p.rsplit('/',1)[-1][:-4])
for k,v in sorted(g.items(),key=lambda x:-len(x[1])):
    print(k,len(v),' '.join(sorted(v,key=len)[:14]))
