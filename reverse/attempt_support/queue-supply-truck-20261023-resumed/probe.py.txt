from pathlib import Path
import sys,json,hashlib,difflib,time
root=Path(__file__).resolve().parents[2]
sys.path.insert(0,str(root/'tools'))
import build
import capstone
NAME='?queueSupplyTruck@AIPlayer@@IAEXXZ'
RVA=0x4f37b0
md=capstone.Cs(capstone.CS_ARCH_X86,capstone.CS_MODE_32)
def measure(src):
    src=Path(src).resolve();out=src.with_suffix('.obj')
    ok,msg,rc=build.try_compile_source(src,out)
    if not ok:return {'source':str(src),'ok':False,'error':msg,'rc':rc}
    raw,relocs=build.read_object_symbol_bytes(out,NAME)
    raw=raw.rstrip(b'\xcc')
    target=build.read_target_bytes(RVA,1537)
    a=list(md.disasm(target,0));b=list(md.disasm(raw,0))
    hoists=[]
    for x in b:
        if x.mnemonic=='mov' and x.op_str.split(',')[0] in ('esi','edi','ebx','ebp') and '0' in x.op_str:
            if any(off>=x.address and off<x.address+x.size and '??_7Rva000421C8' in name for off,kind,name in relocs):hoists.append((x.address,x.op_str))
    asm='\n'.join(f'{x.address:04x} {x.mnemonic:8} {x.op_str}' for x in b)
    src.with_suffix('.asm.txt').write_text(asm)
    matcher=difflib.SequenceMatcher(None,[x.mnemonic for x in a],[x.mnemonic for x in b],autojunk=False)
    regions=mem=reg=branch=exact=0
    for op,al,ar,bl,br in matcher.get_opcodes():
        if op!='equal':regions+=1;continue
        for x,y in zip(a[al:ar],b[bl:br]):
            if x.bytes==y.bytes:exact+=1
            if x.op_str==y.op_str:continue
            if 'ptr [' in x.op_str and 'ptr [' in y.op_str:mem+=1
            elif x.mnemonic.startswith(('j','call')):branch+=1
            elif not any(o>=y.address and o<y.address+y.size for o,k,n in relocs):reg+=1
    return {'source':src.name,'ok':True,'size':len(raw),'regions':regions,'mem':mem,'reg':reg,'exact_instructions':exact,'hoists':hoists,'hash':hashlib.md5(raw).hexdigest(),'body_hex':raw.hex(),'relocs':relocs}
if __name__=='__main__':
    for name in sys.argv[1:]:
        p=Path(name)
        if not p.exists():p=Path(__file__).parent/(name+'.cpp')
        start=time.monotonic();r=measure(p)
        p.with_suffix('.result.json').write_text(json.dumps(r,indent=2))
        print({k:v for k,v in r.items() if k not in ('body_hex','relocs')},'seconds',round(time.monotonic()-start,2),flush=True)
