#!/usr/bin/env python3
"""score.py file.cpp... : fit, len, normalized instruction diff for 0x2F71A6"""
import sys, re, difflib
from pathlib import Path
ROOT = Path('/work3/Open-BFME-2')
sys.path.insert(0, str(ROOT / 'tools'))
import build, capstone
from permute import fitness
from locate import object_functions
SYM = '?internal_findHierarchicalPath@Pathfinder@@IAEPAVPath@@_NHPAVObject@@PBUCoord3D@@200PAU4@@Z'
RVA, SIZE = 0x2F71A6, 4605
md = capstone.Cs(capstone.CS_ARCH_X86, capstone.CS_MODE_32)
_BIG = re.compile(r'0x[0-9a-f]{6,}')
def norm(i, n):
    m, o = i.mnemonic, i.op_str
    if m == 'call' and o.startswith('0x'): return m, 'X'
    if m.startswith('j') and o.startswith('0x'): return m, 'J'
    return m, _BIG.sub('A', o)
def ins(b): return [norm(i, len(b)) for i in md.disasm(bytes(b), 0)]
def compile_(src):
    src = Path(src).resolve(); obj = src.with_suffix('.obj')
    if obj.exists(): obj.unlink()
    ok, text, code = build.try_compile_source(src, obj)
    if not ok:
        return None, text
    row = {'name': SYM, 'target_rva': f'0x{RVA:08X}', 'target_size': str(SIZE),
           'source': src.relative_to(ROOT).as_posix(), 'notes': ''}
    try:
        got = build.compile_function(row, symmap, obj)
    except BaseException:
        row['target_size'] = str(SIZE + 256)
        got = build.compile_function(row, symmap, obj)
        got['target'] = got['target'][:SIZE]
    raw = None
    for n, span, rl in object_functions(obj):
        if n == SYM: raw = span
    return (got, raw), None
def score(src):
    r, err = compile_(src)
    if r is None:
        lines = [l for l in err.splitlines() if 'error' in l][:2]
        return f'{Path(src).name}: COMPILE FAIL {" | ".join(lines)[:300]}'
    got, raw = r
    comp, targ = got['bytes'], got['target']
    full = raw.rstrip(b'\xcc') if raw else b''
    f = fitness(comp, targ)
    ti, oi = ins(targ), ins(comp)
    sm = difflib.SequenceMatcher(None, ti, oi, autojunk=False)
    d = sum(max(i2-i1, j2-j1) for tg,i1,i2,j1,j2 in sm.get_opcodes() if tg != 'equal')
    mm = difflib.SequenceMatcher(None, [m for m,_ in ti], [m for m,_ in oi], autojunk=False)
    md_ = sum(max(i2-i1, j2-j1) for tg,i1,i2,j1,j2 in mm.get_opcodes() if tg != 'equal')
    return f'{Path(src).name}: fit={f:.5f} len={len(full)} nd={d} mnd={md_} unres={len(got["unresolved"])}'
def side(src, lo, hi):
    got, raw = compile_(src)[0]
    comp, targ = got['bytes'], got['target']
    ti = [(i.address, *norm(i, 0)) for i in md.disasm(bytes(targ), 0)]
    oi = [(i.address, *norm(i, 0)) for i in md.disasm(bytes(comp), 0)]
    sm = difflib.SequenceMatcher(None, [x[1:] for x in ti], [x[1:] for x in oi], autojunk=False)
    for tg,i1,i2,j1,j2 in sm.get_opcodes():
        if tg == 'equal' or not (lo <= ti[min(i1,len(ti)-1)][0] <= hi): continue
        print(f'--- {tg}')
        for k in range(max(i2-i1, j2-j1)):
            l = ti[i1+k] if i1+k < i2 else None; r = oi[j1+k] if j1+k < j2 else None
            ls = f'{l[0]:5x} {l[1]} {l[2]}' if l else ''; rs = f'{r[0]:5x} {r[1]} {r[2]}' if r else ''
            print(f'  {ls:<52} | {rs}')
symmap = build.load_symbol_map()
if __name__ == '__main__':
    if sys.argv[1] == '-s':
        side(sys.argv[4], int(sys.argv[2], 16), int(sys.argv[3], 16))
    else:
        for a in sys.argv[1:]:
            try: print(score(a), flush=True)
            except Exception as e: print(f'{a}: ERROR {e!r}', flush=True)
