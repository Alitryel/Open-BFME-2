from pathlib import Path
import sys,json,hashlib
sys.path.insert(0,'tools')
from variant_search import Scorer
from zh_sweep import carve
inputs=Path('build/dering-fine')
variants=json.loads((inputs/'manifest.json').read_text())
s=Scorer('_DeringBlockWeak_C',0x1c22c0,1526,Path('build/dering-fine-score').resolve())
results=[]
for trial in variants:
 src=(inputs/(trial['name']+'.cpp')).read_text()
 r=s(src);obj=s.dir/('c%d.obj'%s.n)
 extent=next((len(b) for n,b,_ in carve(obj.read_bytes()) if n==s.symbol),None) if obj.exists() else None
 diffs=[i for i,(a,b) in enumerate(zip(r.get('compiled',b''),r.get('target',b''))) if a!=b]
 item=dict(**trial,n=s.n,extent=extent,difference_count=len(diffs),first_differences=diffs[:24],source_sha256=hashlib.sha256(src.encode()).hexdigest(),**{k:v for k,v in r.items() if k not in ('compiled','target')})
 results.append(item);print(item,flush=True)
 (s.dir/'results.json').write_text(json.dumps(results,indent=2)+'\n')
 if r['exact'] and extent==1526:(s.dir/'exact.cpp').write_text(src);break
