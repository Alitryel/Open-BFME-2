from pathlib import Path
import sys,itertools,json,subprocess,concurrent.futures
sys.path.insert(0,'tools')
import build,capstone
OUT=Path('build/trust237-round3').resolve();ROOT=Path.cwd()
flags=['','/d2bzalign','/d2bzalign1','/d2bzalign2','/d2bzalign4','/d2bzalign8','/d2bzalign16','/d2bzalign32','/d2bzalign64']
def run(job):
 n,f=job;src=OUT/f'a{n:03}.cpp';obj=src.with_suffix('.obj')
 src.write_text('// cl: /O2 /G7 /arch:SSE /DNDEBUG /MD /EHsc '+f+'\nint alignmentProbe(unsigned n,int *p){for(unsigned i=0;i<n;++i) p[i]+=(int)(n-i);return (int)n;}')
 cmd,env=build.compiler_command(src,obj);p=subprocess.run(cmd,env=env,cwd=ROOT,capture_output=True,text=True)
 r=dict(n=n,flag=f,returncode=p.returncode,output=p.stdout+p.stderr)
 if p.returncode:return r
 name=next(s for s in build.defined_code_symbols(obj) if s.startswith('?alignmentProbe@'))
 raw,_=build.read_object_symbol_bytes(obj,name,code_only=True)
 md=capstone.Cs(capstone.CS_ARCH_X86,capstone.CS_MODE_32)
 r.update(size=len(raw),raw=raw.hex(),nops=[i.address for i in md.disasm(raw,0) if i.mnemonic=='nop'])
 return r
with concurrent.futures.ThreadPoolExecutor(max_workers=8) as pool:rows=list(pool.map(run,enumerate(flags)))
(OUT/'alignment_flags.json').write_text(json.dumps(rows,indent=2))
for r in rows:print({k:v for k,v in r.items() if k!='raw'})
