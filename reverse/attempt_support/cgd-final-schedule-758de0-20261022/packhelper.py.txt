from pathlib import Path
from itertools import product
W=Path(__file__).resolve().parent;base=(W/'base.cpp').read_text();out=W/'packhelper';out.mkdir(exist_ok=True)
for idx,(refs,copy,ctype,scope,cc,rt,site0,site1,site2) in enumerate(product(range(2),range(2),range(2),range(2),range(4),range(2),range(2),range(2),range(2))):
 field='const ContactVector &' if refs else 'const ContactVector *';p='p' if refs else '&p';n='n' if refs else '&n';pv='a.point' if refs else '*a.point';nv='a.normal' if refs else '*a.normal'
 defs='struct PrepareArgs { '+field+' point; '+field+' normal; PrepareArgs(const ContactVector &p,const ContactVector &n):point('+p+'),normal('+n+'){}'
 if copy:defs+=' PrepareArgs(const PrepareArgs &a):point(a.point),normal(a.normal){}'
 defs+=' };\n'
 arg='const PrepareArgs &a' if ctype else 'PrepareArgs a';r='unsigned' if rt else 'void *';cast='(unsigned)' if rt else ''
 conv=['','__fastcall ','__stdcall ','__cdecl '][cc]
 if scope:
  body='const ContactVector &point='+pv+'; const ContactVector &normal='+nv+'; return '+cast+'s->prepare(point,normal);'
 else:body='return '+cast+'s->prepare('+pv+','+nv+');'
 defs+='inline '+r+' '+conv+'packedPrepare(ContactSource *s,'+arg+') {'+body+'}\n'
 s=base.replace('struct ContactData',defs+'struct ContactData',1)
 p='node->second->source->prepare(node->point,node->normal)'
 for i,use in reversed(list(enumerate([site0,site1]))):
  if use:
   starts=[];pos=0
   while 1:
    pos=s.find(p,pos)
    if pos<0:break
    starts.append(pos);pos+=len(p)
   j=starts[i];s=s[:j]+'(void *)packedPrepare(node->second->source,PrepareArgs(node->point,node->normal))'+s[j+len(p):]
 if site2:s=s.replace('node->first->source->prepare(node->point,ContactVector(node->normal).negate())','(void *)packedPrepare(node->first->source,PrepareArgs(node->point,ContactVector(node->normal).negate()))')
 (out/('v%04d.cpp'%idx)).write_text(s)
print('generated',idx+1)
