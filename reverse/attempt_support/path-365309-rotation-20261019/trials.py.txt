# run.py
from pathlib import Path
import sys,json,time
root=Path.cwd();sys.path.insert(0,str(root/'tools'))
from variant_search import Scorer,diff_text
out=root/'build/lane-r-path';src=(out/'banked.cpp').read_text()
old='float s=(float)sin((double)angle);float c=(float)cos((double)angle);Rotate_Z(s,c);'
variants=[('baseline',src)]
for name,body in [
 ('nested_float','Rotate_Z(sinf(angle),cosf(angle));'),
 ('nested_cast','Rotate_Z((float)sin((double)angle),(float)cos((double)angle));'),
 ('reverse_locals','float c=(float)cos((double)angle);float s=(float)sin((double)angle);Rotate_Z(s,c);'),
 ('float_locals','float s=sinf(angle);float c=cosf(angle);Rotate_Z(s,c);'),
 ('nested_sin','float c=(float)cos((double)angle);Rotate_Z((float)sin((double)angle),c);'),
 ('nested_cos','float s=(float)sin((double)angle);Rotate_Z(s,(float)cos((double)angle));')]:
 variants.append((name,src.replace(old,body)))
scorer=Scorer('?rva00365309@Path@@QAE_NPAVObject@@PAVPathNode@@PBUCoord3D@@PAUDirectionOptions@@HM_N@Z',0x365309,2177,out/'objects')
results=[]
for name,s in variants:
 r=scorer(s);a,b=r.pop('compiled',b''),r.pop('target',b'')
 r.update(name=name,length=len(a),different_bytes=sum(x!=y for x,y in zip(a,b))+abs(len(a)-len(b)))
 (out/(name+'.cpp')).write_text(s)
 if a:(out/(name+'.diff.txt')).write_text(diff_text(a,b))
 results.append(r);(out/'results.json').write_text(json.dumps(results,indent=2));print(json.dumps(r),flush=True)

variants=[]
for profile in ['vc71-ddk-4035','vc71-sp1']:
 variants.append((profile,'// compiler: '+profile+'\n'+src))
for tag,add in [('add_scalar','dest.x+=node->pos.x;dest.y+=node->pos.y;dest.z+=node->pos.z;'),('add_reverse','dest.x=node->pos.x+dest.x;dest.y=node->pos.y+dest.y;dest.z=node->pos.z+dest.z;'),('add_ref','dest+=node->pos;'),('add_value','dest.add(node->pos);')]:
 variants.append((tag,src.replace('dest.add(&node->pos);',add)))
for tag,decl in [('inline','inline'),('implicit',''),('force','__forceinline')]:
 for args,call in [('locals',old),('nested','Rotate_Z(pathSin(angle),pathCos(angle));')]:
  helpers=decl+' float pathSin(float x){return (float)sin((double)x);}\n'+decl+' float pathCos(float x){return (float)cos((double)x);}\n'

  variants.append(('wrapper_'+tag+'_'+args,v))
for kind in ['inline','']:
 variants.append(('vector_'+(kind or 'implicit'),src.replace('__forceinline',kind)))
for params in ['const float &s_angle,const float &c_angle','float s_angle,const float &c_angle','const float &s_angle,float c_angle']:

 if a:(out/(name+'.diff.txt')).write_text(diff_text(a,b))
 results.append(r);(out/'shape-results.json').write_text(json.dumps(results,indent=2));print(json.dumps(r),flush=True)
