from pathlib import Path
import sys,subprocess,json,re,hashlib
sys.path.insert(0,'tools');import build
out=Path('build/resume-eh-local');variants=json.loads((out/'turn55-emotion-variants.json').read_text())
sym='?update@EmotionTrackerUpdate@@UAE?AW4UpdateSleepTime@@XZ';rva=0x4b1633;size=2200
retail=build.read_target_bytes(rva,size);results=[]
for name,src in variants.items():
 p=out/('turn55-emotion-'+name+'.cpp');p.write_bytes(src.encode())
 proc=subprocess.run([sys.executable,'tools/align_diff.py',str(p),sym,hex(rva),str(size)],stdout=subprocess.PIPE,stderr=subprocess.STDOUT,text=True)
 p.with_suffix('.txt').write_bytes(proc.stdout.encode())
 try:
  raw,relocs=build.read_object_symbol_bytes(build.obj_path(p),sym,code_only=True);mask=set()
  for off,kind,target in relocs:mask.update(range(off,off+4))
  diffs=[i for i,(a,b) in enumerate(zip(raw,retail)) if i not in mask and a!=b]
  row={'name':name,'extent':len(raw),'fixed_diff':len(diffs)+abs(len(raw)-size),'first':diffs[:8],'structural':re.search(r'== structural regions: (\d+)',proc.stdout)[1],'sha256':hashlib.sha256(raw).hexdigest()}
 except Exception as e:row={'name':name,'exit':proc.returncode,'error':str(e)}
 print(json.dumps(row),flush=True);results.append(row)
 (out/'turn55-emotion-results.json').write_bytes(json.dumps(results,indent=2).encode())
