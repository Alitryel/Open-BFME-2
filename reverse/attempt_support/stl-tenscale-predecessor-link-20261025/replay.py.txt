from pathlib import Path
import os,sys,json
snap=Path('/var/home/ostkannit/projects/Open-BFME-2/build/lcs-codex-filter/a7637e3b22')
dest=Path.cwd()/'build/stl-tenscale-owner-link-20261025'
os.environ['LINK_CYCLE_ROOT']=str(snap)
sys.dont_write_bytecode=True
sys.path.insert(0,str(snap/'tools'))
import link_cycle as lc
import closure_report as cr
original=lc.Measure.run

def run(self):
    result=original(self)
    selected=[(rec,ref) for rec,ref in self.pending if rec['row']['name']=='?_Stl_tenscale@_STL@@YAXAA_KHAAH@Z']
    table_refs=[(rec,ref) for rec,ref in selected if ref[2] in ['__Stl_tenpow','__Stl_twoexp']]
    addresses={ref[0] for _,ref in table_refs}
    native={ref[1] for _,ref in table_refs}
    def doc(rec,ref):
        key=(ref[0],ref[1],ref[2],id(ref[4]),ref[5])
        signed=ref[5]-(1<<32) if ref[5]&0x80000000 else ref[5]
        y=ref[3]
        return dict(row=rec['row']['name'],source=rec['row']['source'],linked_rva=hex(ref[0]),
            retail_rva=hex(ref[1]),symbol=ref[2],addend_signed=signed,
            symbol_section_offset=y.value if y else None,storage_class=y.cls if y else None,
            resolution=self.resolve(ref)[0],before_symbol_supported=key in self.before_refs)
    trace=dict(measured_snapshot=str(snap),table_refs=[doc(rec,ref) for rec,ref in table_refs],
        same_addresses=[doc(rec,ref) for rec,ref in self.pending if ref[0] in addresses or ref[1] in native],
        forward={hex(k):[hex(x) for x in sorted(self.fwd[k])] for k in addresses},
        backward={hex(k):[hex(x) for x in sorted(self.back[k])] for k in native})
    (dest/'native-reference-trace.json').write_text(json.dumps(trace,indent=2)+'\n')
    print('Table references',json.dumps(trace['table_refs']),flush=True)
    return result

lc.Measure.run=run
roots,table,facts=cr.replay(snap,verify=0)
(dest/'replay-roots.json').write_bytes(roots)
(dest/'replay-facts.json').write_text(json.dumps(facts,indent=2,default=str)+'\n')
print('ACTUAL READ-ONLY REPLAY',facts,flush=True)
