"""Materialize and optionally diagnose the best blocked sibling candidate.

This only writes under build/. It does not edit Code/, ledgers or gates.
Default execution uses normal explain_mismatch; --prepare-only does not compile.
"""
from pathlib import Path
import argparse
import hashlib
import json
import os
import re
import subprocess
import sys

HERE = Path(__file__).resolve().parent
ROOT = HERE.parents[3]
OUT = ROOT / "build/string-copy-context-replay/repair"
parser = argparse.ArgumentParser(description=__doc__)
parser.add_argument("--prepare-only", action="store_true")
args = parser.parse_args()
receipt = json.loads((HERE / "results.json").read_text())
base = ROOT / receipt["base"]["path"]
assert hashlib.sha256(base.read_bytes()).hexdigest() == receipt["base"]["sha256"], (
    "Production source changed; review the recorded patch against current source.")
original = base.read_text()
assert hashlib.sha256(original.encode()).hexdigest() == receipt["base"]["normalized_text_sha256"]
lines = original.splitlines(keepends=True)
patch = (HERE / "best-sibling22.patch").read_text().splitlines(keepends=True)
assert patch[0].startswith("--- ") and patch[1].startswith("+++ ")
cursor, i, output = 0, 2, []
while i < len(patch):
    match = re.fullmatch(r"@@ -(\d+)(?:,(\d+))? \+(\d+)(?:,(\d+))? @@\n", patch[i])
    assert match, patch[i]
    start = int(match.group(1)) - 1
    old_count = int(match.group(2) or "1")
    new_count = int(match.group(4) or "1")
    assert cursor <= start <= len(lines)
    output.extend(lines[cursor:start])
    cursor = start
    old_seen = new_seen = 0
    i += 1
    while i < len(patch) and not patch[i].startswith("@@ "):
        op, body = patch[i][0], patch[i][1:]
        assert op in " +-"
        if op in " -":
            assert cursor < len(lines) and lines[cursor] == body
            cursor += 1
            old_seen += 1
        if op in " +":
            output.append(body)
            new_seen += 1
        i += 1
    assert (old_seen, new_seen) == (old_count, new_count)
output.extend(lines[cursor:])
text = "".join(output)
OUT.mkdir(parents=True, exist_ok=True)
source = OUT / "candidate.cpp"
source.write_text(text, newline="\n")
assert hashlib.sha256(source.read_bytes()).hexdigest() == receipt["candidate"]["sha256"]
print(f"Prepared {source.relative_to(ROOT).as_posix()}", flush=True)
if args.prepare_only:
    raise SystemExit(0)

for relative, expected in receipt["candidate"]["dependency_receipt"]["deps"].items():
    path = ROOT / Path(relative)
    assert hashlib.md5(path.read_bytes()).hexdigest() == expected, (
        f"Recorded compiler input changed: {relative}")
temp = OUT / "temp"
temp.mkdir(exist_ok=True)
env = dict(os.environ, BFME_OBJSTORE="off", TEMP=str(temp), TMP=str(temp))
rows = [
    ("insert", "?insertRange@Rva008312E0String@@QAEXPAD00ABURva008312E0Tag@@@Z", "0x284D0", 848),
    ("replace", "?bfmeReplaceAliasedRange@Rva008312E0String@@QAEAAV1@PAD000ABURva008312E0Tag@@@Z", "0x2AE50", 511),
    ("copy", "??$Rva000179B0UninitializedCopy@PADPAD@_STL@@YAPADPAD00@Z", "0x179B0", 41),
]
for label, symbol, rva, size in rows:
    command = [sys.executable, "tools/explain_mismatch.py", symbol,
               "--rva", rva, "--size", str(size),
               "--source", source.relative_to(ROOT).as_posix(), "--context", "4"]
    done = subprocess.run(command, cwd=ROOT, env=env, stdout=subprocess.PIPE,
                          stderr=subprocess.STDOUT, text=True)
    (OUT / f"{label}.txt").write_text(done.stdout)
    print(done.stdout, end="", flush=True)
    assert done.returncode == 0
    assert "classification: exact match" in done.stdout
    assert f"compiled symbol size: {size} bytes" in done.stdout
print("Three recorded bodies reproduced. This candidate still has the documented link/ODR blockers.")
