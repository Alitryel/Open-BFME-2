from pathlib import Path
import sys,json,struct
sys.path.insert(0,'tools')
import build
root=Path('build/resume-eh-local/turn69-prereq-trials');maps=build.load_symbol_map();reports=[]
for variant in ['nested_declared_ctor','nested_inline_ctor','existing_record_declared_ctor','existing_record_inline_ctor']:
    p=root/(variant+'.cpp');obj=build.obj_path(p.resolve());symbols=build.defined_code_symbols(obj)
    for s in symbols:
        rva=size=None
        if s.startswith('?get_allocator@?$vector@'):rva,size=0x21983a,7
        elif s.startswith('??0?$_Vector_base@') and 'QAE@IABV' in s:rva,size=0x5c8c37,60
        elif s.startswith('??$__uninitialized_copy@PBU'):rva,size=0x4f5097,38
        elif s.startswith('??$_Construct@U'):rva,size=0x2cf99d,45
        if rva is None:continue
        raw,rel=build.read_object_symbol_bytes(obj,s)
        row={'name':s,'target_rva':hex(rva),'target_size':str(size),'source':p.as_posix(),'status':'matched','notes':''}
        c=build.compile_function(row,maps,obj)
        reports.append({'variant':variant,'symbol':s,'target_rva':hex(rva),'target_size':size,'extent':len(raw),'unresolved':c['unresolved'],'byte_diffs':sum(a!=b for a,b in zip(c['bytes'],c['target'])),'exact':len(raw)==size and not c['unresolved'] and c['bytes']==c['target']})
Path('build/resume-eh-local/turn69-prereq-helper-proof.json').write_text(json.dumps(reports,indent=2)+'\n')
for r in reports:print(json.dumps(r))
