import sys
sys.path.insert(0, '/work3/Open-BFME-2/build/h72')
from score import compile_, md, norm
from concurrent.futures import ThreadPoolExecutor
def one(src):
    r, err = compile_(src)
    if r is None: return f'{src}: FAIL'
    got, raw = r
    L = list(md.disasm(bytes(got['bytes']), 0))
    ent = ' / '.join(f'{i.mnemonic} {i.op_str}' for i in L[5:8])
    cur = [i for i in L if 0xe0 <= i.address < 0xf8]
    cs = ' / '.join(f'{i.mnemonic} {i.op_str}' for i in cur[:3])
    return f'{src.split("/")[-1]:14s} {ent:60s} | {cs}'
with ThreadPoolExecutor(8) as ex:
    for l in ex.map(one, sys.argv[1:]): print(l, flush=True)
