from search import *

def source(n):
    mode=n%8;pos=(n//8)%4;kind=n//32
    head=HEAD;body=BODY
    typ=['int','unsigned int','long','unsigned long'][kind%4]
    posmark=['\n\tGameInfo *game', '\n\tif (!game)', '\n\tif (slot == -1 || slot >= 8)', '\n\tbool observer'][pos]
    # Reference original parameter only once; all subsequent uses go through
    # an ordinary local. Qualifying the parameter load may prevent web merging.
    import re
    after=body.index(posmark)
    body=body[:after]+body[after:].replace('slot','index')
    decl='\n\t%s index = '%typ
    rhs=['slot','*(const volatile int *)&slot','(int)(unsigned int)slot','((const volatile int &)slot)',
         'slot','slot','slot','slot'][mode]
    if mode==4:typ='const int';decl='\n\tconst int &index = '
    if mode==5:decl='\n\tint &index = '
    if mode==6:decl='\n\tregister int index = '
    if mode==7:decl='\n\tconst int indexValue = slot;\n\tint index = ';rhs='indexValue'
    body=body[:after]+decl+rhs+';'+body[after:]
    if kind%4 in (1,3):body=body.replace('index >= 8','(int)index >= 8')
    return head+MARK+body

if __name__=='__main__':
    run([('param-%03d'%n,source(n)) for n in range(128)], 'parameter')
