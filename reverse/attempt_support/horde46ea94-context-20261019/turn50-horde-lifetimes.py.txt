from pathlib import Path
import json
import re
import subprocess
import sys

root = Path.cwd()
scratch = root / 'build/resume-eh-local'
base = (scratch / 'turn50-horde-whole-home-after.cpp').read_text()
at = base.index('bool HordeContain::rva0046EA94(Object *obj)')
prefix, body = base[:at], base[at:]
marker = 'if (!(*(Object * volatile *)&self) || other == self)'
assert body.count(marker) == 1

rebound = body.replace(marker, 'obj = other;\n\tif (!(*(Object * volatile *)&self) || obj == self)')
a = rebound.index('obj = other;')
rebound = rebound[:a] + rebound[a:].replace('other->', 'obj->')

a = body.index('Object *other = obj->rva002931F5(false);')
tail = body[a:]
tail = re.sub(r'\bobj\b', 'candidate', tail)
tail = re.sub(r'\bother\b', 'obj', tail)
tail = tail.replace('Object *obj = candidate->rva002931F5(false);',
                    'obj = candidate->rva002931F5(false);', 1)
formal = body[:a] + 'Object *candidate = obj;\n\t' + tail

old = 'if (rva0046970D(obj, m_26C, n2, false) && m_object->m_264->m_24 <= 1)\n\t\t\treturn true;'
assert old in body
reload = body.replace(old, '''if (rva0046970D(obj, m_26C, n2, false))
        {
            self = m_object;
            if (self->m_264->m_24 <= 1)
                return true;
        }''')
variants = {'reuse-formal-related-arm': rebound, 'reuse-formal-container': formal,
            'reload-owner-local': reload}
sys.path.insert(0, str(root / 'tools'))
import build
import eh_verify
image = eh_verify.Image()
symbol = '?rva0046EA94@HordeContain@@UAE_NPAVObject@@@Z'
results = []
for label, function in variants.items():
    path = scratch / ('turn50-horde-' + label + '.cpp')
    path.write_bytes((prefix + function).encode())
    run = subprocess.run([sys.executable, 'tools/align_diff.py',
        path.relative_to(root).as_posix(), symbol, '0x0046EA94', '488'],
        capture_output=True, text=True)
    (scratch / ('turn50-horde-' + label + '.txt')).write_bytes((run.stdout + run.stderr).encode())
    result = {'variant': label, 'exit': run.returncode}
    m = re.search(r'ours (\d+) B', run.stdout)
    result['size'] = int(m[1]) if m else None
    if m:
        for key in ('memory', 'register'):
            m = re.search(r'^\s+(\d+)\s+' + key + r'\s', run.stdout, re.M)
            result[key] = int(m[1]) if m else None
        m = re.search(r'== structural regions: (\d+)', run.stdout)
        result['structural'] = int(m[1]) if m else None
        result['eh'] = eh_verify.verify_row(image, eh_verify.Obj(build.obj_path(path)), symbol, 0x46ea94)
    else:
        result['error'] = (run.stdout + run.stderr)[-1200:]
    results.append(result)
    print(json.dumps(result), flush=True)
(scratch / 'turn50-horde-lifetimes-results.json').write_bytes((json.dumps(results, indent=2) + '\n').encode())
