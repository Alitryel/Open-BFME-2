from pathlib import Path
import sys,subprocess,json
root=Path(__file__).resolve().parents[1];sys.path.insert(0,str(root/'tools'))
out=root/'build/findpath-float-context-oct20';out.mkdir(exist_ok=True)
base=(root/'build/findpath-auto-ptr-oct20/baseline.cpp').read_text()
start=base.index('Path *Pathfinder::FindPath(')
before=base[:start];body=base[start:]
setbody='''inline void Coord3D::set(float X,float Y,float Z) {x=X;y=Y;z=Z;}
'''
variants={}
copies=[('destination','copyCoord3D(&destination, &adjusted);','destination.set(adjusted.x,adjusted.y,adjusted.z);'),('adjusted','copyCoord3D(&adjusted, rawTo);','adjusted.set(rawTo->x,rawTo->y,rawTo->z);'),('delta','copyCoord3D(&delta, &adjusted);','delta.set(adjusted.x,adjusted.y,adjusted.z);')]
for key,old,new in copies:
 assert old in body;variants['member-set-'+key]=before+setbody+body.replace(old,new)
s=body
for key,old,new in copies:s=s.replace(old,new)
variants['member-set-all']=before+setbody+s
old='destination.x != adjusted.x || destination.y != adjusted.y'
for key,params,expr,args in [
 ('xyxy','float ax,float ay,float bx,float by','ax != bx || ay != by','destination.x,destination.y,adjusted.x,adjusted.y'),
 ('xxyy','float ax,float bx,float ay,float by','ax != bx || ay != by','destination.x,adjusted.x,destination.y,adjusted.y'),
 ('pairs','const Coord3D &a,const Coord3D &b','a.x != b.x || a.y != b.y','destination,adjusted'),
]:
 helper='static __forceinline bool differentXY('+params+') {return '+expr+';}\n'
 variants['compare-'+key]=before+helper+body.replace(old,'differentXY('+args+')')
 if key!='pairs':variants['set-and-compare-'+key]=before+setbody+helper+body.replace(copies[0][1],copies[0][2]).replace(old,'differentXY('+args+')')
import build,capstone
from locate import object_functions
sym='?FindPath@Pathfinder@@QAEPAVPath@@PAVObject@@ABVLocomotorSet@@PBUCoord3D@@2PA_N@Z'
md=capstone.Cs(capstone.CS_ARCH_X86,capstone.CS_MODE_32)
rows=[]
for key,s in variants.items():
 src=out/(key+'.cpp');src.write_text(s)
 p=subprocess.run([sys.executable,'tools/explain_mismatch.py',sym,'--rva','0x2FD344','--size','781','--source',str(src)],cwd=root,capture_output=True,text=True,encoding='utf8',errors='replace',timeout=300)
 log=p.stdout+p.stderr;src.with_suffix('.mismatch.txt').write_text(log)
 obj=build.obj_path(src)
 if not obj.exists():print(key,'ERROR',log[-1300:],flush=True);continue
 f=next(b for n,b,r in object_functions(obj) if n==sym)
 latch=[(hex(i.address),i.mnemonic,i.op_str) for i in md.disasm(f,0) if 0x1bb<=i.address<=0x1d0]
 summary=[l for l in log.splitlines() if l.startswith(('classification:','first diff:','OK:','unresolved:'))]
 print(key,len(f),summary,latch,flush=True)
 rows.append(dict(key=key,bytes=len(f),summary=summary,latch=latch))
(out/'results.json').write_text(json.dumps(rows,indent=2))
