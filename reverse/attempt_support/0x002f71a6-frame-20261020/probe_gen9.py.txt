"""Frame-order probe 9 (big frame): IL reference weight of each Coord3D init/copy spelling.
p gets construct C + 2 x extp(&p); q (16B) gets k refs. Threshold k* -> p's IL ref count R."""
from pathlib import Path
import itertools
H = Path('/work3/Open-BFME-2/build/h72/p1')
HDR = """// cl: /O1 /DNDEBUG /MD /arch:SSE /G7 /EHsc
struct P3 { float x, y, z; void set(const P3 *a) { x = a->x; y = a->y; z = a->z; } };
struct HP : public P3 { HP(){} HP(const HP &c){x=c.x;y=c.y;z=c.z;} HP(const P3 &c){x=c.x;y=c.y;z=c.z;}
  HP &operator=(const P3 &c){*(P3 *)this=c;return *this;} };
static __forceinline void copy3(P3 *d, const P3 *s) { d->x = s->x; d->y = s->y; d->z = s->z; }
struct S4 { int a, b, c, d; };
void extp(P3 *); void extq(void *); void exta(int *); P3 *src(); HP *hsrc();
"""
C = {
    'none': ' HP p;',
    'hpctor': ' HP p(*src());',
    'hpcopy': ' HP p(*hsrc());',
    'hpassign': ' HP p; p = *src();',
    'fields': ' P3 p; { P3 *s = src(); p.x = s->x; p.y = s->y; p.z = s->z; }',
    'pod': ' P3 p = *src();',
    'podassign': ' P3 p; p = *src();',
    'set': ' P3 p; p.set(src());',
    'copy3': ' P3 p; copy3(&p, src());',
    'outpod': ' P3 p; extp(&p); *src() = p;',
    'outcopy3': ' P3 p; extp(&p); copy3(src(), &p);',
}
for f in H.glob('t9*.cpp'):
    f.unlink()
for (cn, cc), k in itertools.product(C.items(), range(1, 13)):
    body = ' int arr[40];\n exta(arr);\n' + cc + '\n S4 q;\n extp(&p); extp(&p);\n' + ' extq(&q);' * k + '\n'
    src = HDR + 'void probe()\n{\n' + body + '}\n'
    (H / f't9{cn}_k{k:02d}.cpp').write_text(src)
