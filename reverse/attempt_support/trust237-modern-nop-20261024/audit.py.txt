from pathlib import Path
import sys,json,hashlib,shutil
sys.path.insert(0,str(Path.cwd()/'tools'))
import build,capstone
md=capstone.Cs(capstone.CS_ARCH_X86,capstone.CS_MODE_32)
root=Path.cwd();out=root/'build/trust237-compiler-audit';out.mkdir(parents=True,exist_ok=True)
records=[]
for family,where in [('clang23','build/trust237-clang23-nop'),('msvc14','build/trust237-msvc14-nop')]:
    folder=root/where;rows=json.loads((folder/'results.json').read_text())
    for row in rows:
        src=folder/(row['name']+'.cpp')
        assert hashlib.sha256(src.read_bytes()).hexdigest()==row['source_sha256']
        obj=Path(row['object']) if family=='clang23' else src.with_suffix('.obj')
        raw,rel=build.read_object_symbol_bytes(obj,'?Rva00237D3DCheck@@YAHXZ',code_only=True)
        assert hashlib.sha256(raw).hexdigest()==row['raw_sha256']
        assert raw.hex()==row['raw_hex']
        assert [list(x) for x in rel]==row['relocations']
        ins=list(md.disasm(raw,0));assert sum(x.size for x in ins)==len(raw)
        nops=[x.address for x in ins if x.mnemonic=='nop'];assert nops==row['nops']
        if family=='clang23':
            refs=[r for r in rel if r[2]=='___nop'];assert len(refs)==16 and not nops
        else:assert len(nops)==16 and not any(r[2]=='___nop' for r in rel)
        assert len(raw)!=226
        records.append(dict(family=family,name=row['name'],size=len(raw),source_sha256=row['source_sha256'],raw_sha256=row['raw_sha256'],nop_offsets=nops,relocations=rel))
summary={}
for family in ['clang23','msvc14']:
    rs=[r for r in records if r['family']==family]
    summary[family]=dict(compiled=len(rs),distinct=len({r['raw_sha256'] for r in rs}),sizes=sorted({r['size'] for r in rs}),nop_counts=sorted({len(r['nop_offsets']) for r in rs}))
summary['wholefile_gate']=dict(functions='40/40',sources=3,strings=21,empty_strings=2,named_imports=7,data_rows=1)
summary['target_sha256']=hashlib.sha256(build.read_target_bytes(0x237D3D,226)).hexdigest()
summary['production_changed']=False;summary['exact_candidates']=0
(out/'audit.json').write_text(json.dumps(dict(summary=summary,objects=records),indent=2)+'\n',encoding='utf-8')
print(json.dumps(summary,indent=2))
