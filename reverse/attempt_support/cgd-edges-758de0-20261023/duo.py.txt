from pathlib import Path
from itertools import product
import json
W=Path(__file__).resolve().parent;b=(W/'base.cpp').read_text();D=W/'duo';D.mkdir(exist_ok=True)
rows=[]
for idx,(cc,ref,copy,dtor,ptr,sites,bridge,separate) in enumerate(product(range(2),range(2),range(4),range(2),range(2),range(8),range(2),range(2))):
 def carrier(name):
  ty='const ContactVector &' if ref else 'const ContactVector *'
  param='const ContactVector *v' if ptr else 'const ContactVector &v'
  val=('*v' if ptr else 'v') if ref else ('v' if ptr else '&v')
  text='struct '+name+' { '+ty+' p; '+name+'('+param+') : p('+val+') {} '
  if copy==1:text+=name+'(const '+name+' &v) : p(v.p) {} '
  elif copy==2:
   if ref:text+=name+'(const '+name+' &v) : p(v.p) { const ContactVector &q=v.p; } '
   else:text+=name+'(const '+name+' &v) { const ContactVector *q=v.p; memcpy(&p,&q,4); } '
  elif copy==3:text+=name+'(const '+name+' &v) throw() : p(v.p) {} '
  if dtor:text+='~'+name+'() {} '
  return text+'};\n'
 defs='extern "C" void *__cdecl memcpy(void *,const void *,unsigned);\n#pragma intrinsic(memcpy)\n'+carrier('PointArg')
 defs+=carrier('NormalArg') if separate else 'typedef PointArg NormalArg;\n'
 def pa(p):return 'PointArg('+('&('+p+')' if ptr else p)+')'
 def na(n):return 'NormalArg('+('&('+n+')' if ptr else n)+')'
 proto='virtual void *'+('__fastcall ' if cc else '')+'prepareArgs(PointArg p,NormalArg n)=0;\n __forceinline void *prepare(const ContactVector &p,const ContactVector &n){return prepareArgs('+pa('p')+','+na('n')+');}'
 s=b.replace('struct ContactSource {',defs+'struct ContactSource {')
 s=s.replace('virtual void *prepare(const ContactVector &point, const ContactVector &normal)=0;',proto)
 # All three calls have the same declared interface; no reverse-only ABI cast.
 s=s.replace('altPrepare(node->first->source,node->point,ContactVector(node->normal).negate())','node->first->source->prepare(node->point,ContactVector(node->normal).negate())')
 if not bridge:s=s.replace('bridge(node->second,node->point,node->normal)','node->second->source->prepare(node->point,node->normal)')
 elif sites&2:
  old='return s->source->prepare(p,n);'
  s=s.replace(old,'return s->source->prepareArgs('+pa('p')+','+na('n')+');')
 h,t=s.split('void BfmeThingCGD::bfmeTwoCGD() {',1)
 pos=0
 for i,(side,n) in enumerate([('second','node->normal'),('second','node->normal'),('first','ContactVector(node->normal).negate()')]):
  if i==1 and bridge:continue
  old='node->'+side+'->source->prepare(node->point,'+n+')'
  at=t.find(old,pos);assert at>=0,(idx,i)
  v='node->'+side+'->source->prepareArgs('+pa('node->point')+','+na(n)+')' if sites>>i&1 else old
  t=t[:at]+v+t[at+len(old):];pos=at+len(v)
 (D/('v%04d.cpp'%idx)).write_text(h+'void BfmeThingCGD::bfmeTwoCGD() {'+t)
 rows.append(dict(name='v%04d'%idx,fastcall=cc,reference=ref,copy=copy,dtor=dtor,pointer=ptr,sites=sites,bridge=bridge,separate=separate))
(W/'duo-models.json').write_text(json.dumps(rows,indent=1));print(len(rows),'consistent two-class argument models')
