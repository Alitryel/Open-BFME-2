from search import *
s=(W/'base.cpp').read_text()
vs=[]
for mask in range(256):
 o=s
 # Field reads present as calls in the debug twin, combined factorially with
 # complete bit-test chains rather than the previous one-getter-at-a-time tests.
 if mask&1:
  o=o.replace(' char pad[0x11f];\n unsigned char flags11f;', ' char pad[0x108];\n struct Bits {unsigned words[7]; unsigned test(int i)const {return words[i>>5]&(1U<<(i&31));}} kind;\n unsigned isKindOf(int k)const{return kind.test(k);}')
  o=o.replace(' bool isKindOf(KindOfType)const;', ' bool isKindOf(KindOfType)const;\n const LocoObjectTemplate *getTemplate()const{return definition;}')
  o=o.replace('object->definition->flags11f&8','object->getTemplate()->isKindOf(187)')
 if mask&2:
  o=o.replace(' bool isKindOf(KindOfType)const;', ' bool isKindOf(KindOfType)const;\n bool isHeld()const{return (disabled&8)!=0;}')
  o=o.replace('object->disabled&8','object->isHeld()')
 helpers=[]
 if mask&4:
  o=o.replace('float speed=speed40;', 'float speed=getCurrentSpeed();')
  helpers.append('float getCurrentSpeed()const{return speed40;}')
 if mask&8:
  i=o.index('bool Locomotor::rva001E7ECA');o=o[:i]+o[i:].replace('height48','getPreferredHeight()')
  helpers.append('float getPreferredHeight()const{return height48;}')
 if mask&16:
  i=o.index('bool Locomotor::rva001E7ECA');o=o[:i]+o[i:].replace('data->behavior','getBehavior()')
  helpers.append('int getBehavior()const{return data->behavior;}')
 if mask&32:
  i=o.index('bool Locomotor::rva001E7ECA');o=o[:i]+o[i:].replace('damping50','getDamping()')
  helpers.append('float getDamping()const{return damping50;}')
 if mask&64:
  # Direct callee visibility is untested for this Object helper (unlike
  # Locomotor same-unit helpers), so include its retail 20-byte behavior.
  o=o.replace(' int rva0028B511()const;', ' __declspec(noinline) int rva0028B511()const{if(*(unsigned char*)((char*)this+0x48c))return 1;return *(int*)((char*)this+0x40c);}')
 if mask&128:
  # The setters observed in WB take pointers. Bank currently uses a reference
  # adapter; retain native component stores while restoring the pointer chain.
  o=o.replace('void setTranslation(const Rva001E41D3Result&p)', 'void setTranslation(const Rva001E41D3Result*p)').replace('((LocoMatrix*)matrix)->set(p);','((LocoMatrix*)matrix)->set(*p);').replace('setTranslation(pos);','setTranslation(&pos);')
 if helpers:o=o.replace(' float getMaxSpeed(Object*)const;', ' float getMaxSpeed(Object*)const;\n'+'\n'.join(helpers))
 vs.append((f'access_{mask:03d}',o))
run(vs,'access256')
