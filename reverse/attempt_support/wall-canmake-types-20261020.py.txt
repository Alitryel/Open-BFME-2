from pathlib import Path
import subprocess,sys,json,itertools
root=Path(__file__).resolve().parents[1];out=root/'build/wall-canmake-closure-oct20'
base=(out/'baseline.cpp').read_text()
status='enum WallQuoteStatusView { QUOTE_SUCCESS=0, QUOTE_NO_MONEY=7, QUOTE_UNKNOWN=-1 };\n'
ret='enum WallMakeReturnView { MAKE_SUCCESS=0, MAKE_NO_MONEY=2, MAKE_FAILURE=9 };\n'
tests={}
for boolean,en,ret_en in itertools.product([False,True],repeat=3):
    s=base;tag='flags'+str(int(boolean))+'-status'+str(int(en))+'-return'+str(int(ret_en))
    if boolean:s=s.replace('unsigned char m_1C;','bool m_1C;').replace('unsigned char m_1D;','bool m_1D;')
    if en:s=status+s.replace('int m_20;','WallQuoteStatusView m_20;').replace('m_20 = -1;','m_20 = QUOTE_UNKNOWN;')
    if ret_en:
        s=ret+s
        s=s.replace('int canMake(Player *player);','WallMakeReturnView canMake(Player *player);').replace('int AIBuildableWallSegment::canMake(Player *player)','WallMakeReturnView AIBuildableWallSegment::canMake(Player *player)')
        start=s.index('WallMakeReturnView AIBuildableWallSegment::canMake');body=s[start:]
        for value,token in [(9,'MAKE_FAILURE'),(2,'MAKE_NO_MONEY'),(0,'MAKE_SUCCESS')]:body=body.replace('return '+str(value)+';','return '+token+';')
        s=s[:start]+body
    tests[tag]=s
    tests[tag+'-nondpodCoord']=s.replace('/Lib/Coord3D.h"','/Lib/Coord3DCtorView.h"').replace('#include "'+(root/'Code/Libraries/Include/Lib/Coord3DCtorView.h').as_posix()+'"','#define COORD3D_VIEW_DTOR\n#include "'+(root/'Code/Libraries/Include/Lib/Coord3DCtorView.h').as_posix()+'"')
results=[]
for key,s in tests.items():
    p=out/(key+'.cpp');p.write_bytes(s.encode())
    n='?canMake@AIBuildableWallSegment@@UAE'+('?AW4WallMakeReturnView@@' if '-return1' in key else 'H')+'PAVPlayer@@@Z'
    r=subprocess.run([sys.executable,'tools/explain_mismatch.py',n,'--rva','0x00596E15','--size','218','--source',p.relative_to(root).as_posix()],cwd=root,capture_output=True,text=True,encoding='utf8',errors='replace',timeout=240)
    log=r.stdout+r.stderr;p.with_suffix('.log').write_bytes(log.encode())
    d={'key':key,'name':n,'exit':r.returncode,'exact':'OK: bytes match' in log,'summary':[x for x in log.splitlines() if x.startswith(('compiled symbol size:','classification:','unresolved:'))]}
    if r.returncode:d['error']=log[-900:]
    results.append(d);(out/'type-results.json').write_text(json.dumps(results,indent=2));print(json.dumps(d),flush=True)
