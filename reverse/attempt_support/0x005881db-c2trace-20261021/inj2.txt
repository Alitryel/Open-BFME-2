B=0x10700000
bps=[0x83acd,0x82978,0xa8f54]; added=set()
for b in bps:
    o=d.prompt('break *0x%x'%(B+b))
    if 'Breakpoint' in o and 'Unable' not in o: added.add(b)
names={}
def nm(sym):
    if sym in names: return names[sym]
    sw=d.words(sym,1); pw=d.words(sw[0],2) if sw and sw[0]>0x100000 else []
    n=d.string(pw[1]).strip() if len(pw)>1 and pw[1]>0x100000 else '?'
    n=n.strip('"').lstrip('_'); names[sym]=n; return n
def rec_of(i):
    p=d.words(0x108807b8+(i&0x3ff)*4,1)[0]
    while p:
        w=d.words(p,13)
        if w[7]==i: return p,w
        p=w[12]
    return None,None
on=False;k=0;calls={};pend=None;skip=0
while True:

        if b not in added and b!=eip: d.prompt('break *0x%x'%(B+b)); added.add(b)

    if eip==0x83acd:
        ebp=d.reg('ebp'); f=d.words(ebp,1)[0]; curfn=d.string(d.words(f+4,1)[0]); on='updateMeleeAttack' in curfn and d.reg('ebx')==0; k=0; continue
    if not on: continue
    if eip==0x82978: k+=1; continue

            base,nx,bits=d.words(ch,3)

                if bits>>j&1:
                    i=base+j; p,w=rec_of(i)

        print('R%d call%d ra=%x set=%x a3=%x n=%d :: %s'%(k,c,ra,st[2],st[3],len(items),' '.join('%s:%x/w9=%d'%(nm(w[0]),i,w[9]&0xffff) if w else '%x'%i for i,p,w in items)),flush=True)
        if TR and int(TR)==k and int(TC)==c:
            if not any(w and nm(w[0])==TN for i,p,w in items):
                for b in d.words(0x108807b8,1024):

                        if nm(w[0])==TN and (not os.environ.get('IDX') or w[7]==int(os.environ['IDX'],16)) and (w[9]&0xffff)>=2: items.append((w[7],q,w))
                        q=w[12]
            for i,p,w in items:
                if w and nm(w[0])==TN:
                    ch=d.words(st[2],1)[0]
                    while ch:

                        ch=nx
                    pend=(ra,p,st[3],esp+16); print('moved',TN,hex(i),flush=True)
                    d.prompt('break *0x%x'%(B+ra)); added.add(ra); break
        continue
    if pend and eip==pend[0] and d.reg('esp')==pend[3]:
        ra,p,a3,esp0=pend; pend=None
        e=esp0-12
        d.prompt('set *(int*)0x%x = %d'%(e,B+ra)); d.prompt('set *(int*)0x%x = %d'%(e+4,p)); d.prompt('set *(int*)0x%x = %d'%(e+8,a3))
