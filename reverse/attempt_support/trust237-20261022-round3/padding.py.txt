from pathlib import Path
import sys,itertools,json,subprocess,concurrent.futures,hashlib
sys.path.insert(0,'tools')
import build,capstone
ROOT=Path.cwd();OUT=ROOT/'build/trust237-round3'
ops=['n ^= n>>7;','n += n<<3;','n=(n<<3)|(n>>29);','n*=0x45d9f3bu;','n^=0xa5a5a5a5u;','n+=n>>11;']
def run(job):
 n,mask,cpu,opt,bz=job
 prefix=''.join(op for k,op in enumerate(ops) if mask&(1<<k))
 text=f'// cl: /{opt} /G{cpu} /arch:SSE /DNDEBUG /MD /EHsc'+(' /d2bzalign' if bz else '')+'\nint alignmentProbe(unsigned n,int *p){'+prefix+'for(unsigned i=0;i<n;++i) p[i]+=(int)(n-i); return (int)n;}'
 src=OUT/f'pad{n:04}.cpp';obj=src.with_suffix('.obj');src.write_text(text)
 cmd,env=build.compiler_command(src,obj);p=subprocess.run(cmd,env=env,cwd=ROOT,capture_output=True,text=True)
 r=dict(n=n,mask=mask,cpu=cpu,opt=opt,bzalign=bz)
 if p.returncode:r['error']=(p.stdout+p.stderr)[-800:];return r
 name=next(s for s in build.defined_code_symbols(obj) if s.startswith('?alignmentProbe@'))
 raw,_=build.read_object_symbol_bytes(obj,name,code_only=True)
 md=capstone.Cs(capstone.CS_ARCH_X86,capstone.CS_MODE_32);md.detail=True;ins=list(md.disasm(raw,0))
 nops=[i.address for i in ins if i.mnemonic=='nop'];runs=[]
 for a in nops:
  if runs and a==runs[-1]['end']:runs[-1]['end']+=1
  else:runs.append(dict(start=a,end=a+1))
 backedges=[dict(at=i.address,to=i.operands[0].imm) for i in ins if i.mnemonic.startswith('j') and i.operands and i.operands[0].type==capstone.x86.X86_OP_IMM and i.operands[0].imm<i.address]
 return dict(**r,size=len(raw),sha256=hashlib.sha256(raw).hexdigest(),nops=nops,nop_runs=runs,backedges=backedges,raw=raw.hex())
jobs=[(n,*j) for n,j in enumerate(itertools.product(range(64),[5,6,7],['O1','O2'],range(2)))]
with concurrent.futures.ThreadPoolExecutor(max_workers=8) as pool:rows=list(pool.map(run,jobs))
(OUT/'padding.json').write_text(json.dumps(rows,indent=2))
good=[r for r in rows if 'error' not in r]
print('jobs',len(rows),'good',len(good),'unique',len({r['sha256'] for r in good}),'with NOP',sum(bool(r['nops']) for r in good),'max consecutive NOP',max((q['end']-q['start'] for r in good for q in r['nop_runs']),default=0))
for r in sorted(good,key=lambda r:max((q['end']-q['start'] for q in r['nop_runs']),default=0),reverse=True)[:4]:print({k:v for k,v in r.items() if k!='raw'})
