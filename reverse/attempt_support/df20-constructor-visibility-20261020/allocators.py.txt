from pathlib import Path
import sys,itertools,concurrent.futures,json
sys.path.insert(0,'build/df20-scalars-20261020');import trials
trials.scratch=Path.cwd()/'build/df20-scalars-20261020'
base=Path('build/df20-scalars-20261020/bank.cpp').read_text();trials.symbol=base.splitlines()[0][3:]
needle='  string_type __new_digits(__digits_first, __digits_last);'
specs=[]
for kind,loc in itertools.product(['temporary','from-digits','named','const','reference'],['local','before-locale','before-group','before-currency']):
 s=base
 if kind in ['temporary','from-digits']:
  if loc!='local':continue
  expr='allocator<wchar_t>()' if kind=='temporary' else '__digits.get_allocator()'
  s=s.replace(needle,needle[:-2]+', '+expr+');')
 else:
  decl={'named':'allocator<wchar_t> __digits_allocator;','const':'const allocator<wchar_t> __digits_allocator;','reference':'const allocator<wchar_t>& __digits_allocator = allocator<wchar_t>();'}[kind]
  marker={'local':needle,'before-locale':'  locale __loc =','before-group':'  string __grouping =','before-currency':'#pragma inline_depth(0)\n  string_type __curr_sym'}[loc]
  s=s.replace(marker,'  '+decl+'\n'+marker,1)
  s=s.replace(needle,needle[:-2]+', __digits_allocator);')
 specs.append((f'allocator-{kind}-{loc}',s))
with concurrent.futures.ThreadPoolExecutor(max_workers=8) as p:r=list(p.map(trials.run,specs))
(trials.scratch/'allocators.json').write_text(json.dumps(r,indent=2))
