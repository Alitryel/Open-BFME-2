from pathlib import Path
import json
import re
import subprocess
import sys

root = Path.cwd()
scratch = root / 'build/resume-eh-local'
home = root / 'Code/GameEngine/Source/GameLogic/Object/Contain/HordeContainIface11CSlots.cpp'
bank = (root / 'reverse/attempts/0x0046ea94.cpp').read_text()
source = home.read_text()
symbol = '?rva0046EA94@HordeContain@@UAE_NPAVObject@@@Z'
body = bank[bank.index('bool HordeContain::rva0046EA94(Object *obj)'):].strip() + '\n'

prefix = '''enum KindOfType { KINDOF_INVALID = -1 };
template <int N> class BitFlags;
template <> class BitFlags<11> {
    unsigned int m_bits[1];
public:
    bool any() const;
};
'''
source = source.replace('class Object\n{', prefix + '\nclass Object\n{', 1)
source = source.replace('bool affectedByUpgrade(const UpgradeTemplate *upgrade) const;',
    'bool affectedByUpgrade(const UpgradeTemplate *upgrade) const;\n'
    '\tbool isKindOf(KindOfType kind) const;\n'
    '\tbool isDisabled() const { return m_disabledMask.any(); }', 1)
source = source.replace('unsigned char m_pad114[0x244 - 0x114];',
    'unsigned char m_pad114[0x1C8 - 0x114];\n'
    '\tBitFlags<11> m_disabledMask;\n'
    '\tunsigned char m_pad1CC[0x244 - 0x1CC];', 1)
source = source.replace('unsigned char m_pad1D9[0x224 - 0x1D9];',
    'unsigned char m_pad1D9[0x1F4 - 0x1D9];\n'
    '\tRva00469851Names m_1F4;\n'
    '\tunsigned char m_pad200[0x218 - 0x200];\n'
    '\tRva00469851Names m_218;', 1)
source = source.replace('unsigned char m_pad238[0x254 - 0x238];',
    'bool m_238;\n\tunsigned char m_pad239[0x23C - 0x239];\n'
    '\tAsciiString m_23C;\n\tunsigned char m_pad240[0x254 - 0x240];', 1)
a = source.index('class Rva0046BB38Iface11C :')
b = source.index('class UpdateModule :', a)
view = source[a:b]
assert view.count('virtual void gap28() = 0;') == 1
view = view.replace('virtual void gap28() = 0;', 'virtual bool rva0046EA94(Object *obj) = 0;')
source = source[:a] + view + source[b:]
source = source.replace('virtual bool rva0046970D(Object *obj, int a2, const Rva00469851Names *names, bool sameGroup);',
    'virtual bool rva0046970D(Object *obj, int a2, const Rva00469851Names *names, bool sameGroup);\n'
    '\tvirtual bool rva0046EA94(Object *obj);', 1)

# Preserve the actual physical header behind every existing relative include.
resolved = []
def include(match):
    spelling = match[1]
    path = (home.parent / spelling).resolve()
    if not path.is_file():
        return match[0]
    resolved.append({'spelling': spelling, 'resolved': str(path)})
    return '#include "' + path.as_posix() + '"'
source = re.sub(r'#include\s+"([^"]+)"', include, source)

variants = {
    'bank': bank,
    'whole-home-after': source + '\n' + body,
    'whole-home-before': source.replace('bool HordeContain::rva0046BB38(Object *other)',
        body + '\nbool HordeContain::rva0046BB38(Object *other)', 1),
}
sys.path.insert(0, str(root / 'tools'))
import build
import eh_verify
image = eh_verify.Image()
results = []
for label, candidate in variants.items():
    path = scratch / ('turn50-horde-' + label + '.cpp')
    path.write_bytes(candidate.encode())
    run = subprocess.run([sys.executable, 'tools/align_diff.py',
        path.relative_to(root).as_posix(), symbol, '0x0046EA94', '488'],
        capture_output=True, text=True)
    (scratch / ('turn50-horde-' + label + '.txt')).write_bytes((run.stdout + run.stderr).encode())
    result = {'variant': label, 'exit': run.returncode}
    match = re.search(r'ours (\d+) B', run.stdout)
    result['size'] = int(match[1]) if match else None
    if match:
        for key in ('memory', 'register'):
            m = re.search(r'^\s+(\d+)\s+' + key + r'\s', run.stdout, re.M)
            result[key] = int(m[1]) if m else None
        m = re.search(r'== structural regions: (\d+)', run.stdout)
        result['structural'] = int(m[1]) if m else None
        result['eh'] = eh_verify.verify_row(image, eh_verify.Obj(build.obj_path(path)), symbol, 0x46ea94)
    else:
        result['error'] = (run.stdout + run.stderr)[-1400:]
    results.append(result)
    print(json.dumps(result), flush=True)
(scratch / 'turn50-horde-context-results.json').write_bytes((json.dumps({
    'physical_includes': resolved, 'results': results}, indent=2) + '\n').encode())
