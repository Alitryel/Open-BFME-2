from pathlib import Path
import sys,itertools
sys.path.insert(0,str(Path(__file__).resolve().parent));import search
D=search.DIR/'storage';D.mkdir(exist_ok=True);b=search.DIR.joinpath('base.cpp').read_text()
for n,(st,nt,cast,qual) in enumerate(itertools.product(range(8),range(8),range(4),range(4))):
 s=b
 ts=['ContactSource *','void *','char *','unsigned char *','unsigned','int','long','ContactData *'][st]
 ns=['ContactData *','void *','char *','unsigned char *','unsigned','int','long','ContactSource *'][nt]
 s=s.replace('ContactSource *source;',ts+'source;' if ts.endswith('*') else ts+' source;')
 if qual&1:s=s.replace('source;', 'source;') # no extra cv here; factors below influence casts instead
 s=s.replace('ContactData *first,*second;',ns+'first; '+ns+'second;' if ns.endswith('*') else ns+' first, second;')
 h,t=s.split('void BfmeThingCGD::bfmeTwoCGD() {')
 def cvt(typ,expr):
  if cast==0:return '(('+typ+' *)'+expr+')'
  if cast==1:return 'reinterpret_cast<'+typ+' *>('+expr+')'
  if cast==2:return '(('+typ+' *)(unsigned)'+expr+')'
  return '*reinterpret_cast<'+typ+' *const *>(&'+expr+')'
 for name in ['first','second']:
  raw='node->'+name
  data=raw if nt==0 and not qual&1 else cvt('ContactData',raw)
  src=data+'->source'
  val=src if st==0 and not qual&2 else cvt('ContactSource',src)
  t=t.replace(raw+'->source',val)
 s=h+'void BfmeThingCGD::bfmeTwoCGD() {'+t
 (D/('s%04d.cpp'%n)).write_text(s)
