from pathlib import Path
import sys,itertools
ROOT=Path.cwd();OUT=ROOT/'build/trust237-round12';OUT.mkdir(parents=True,exist_ok=True)
sys.path.insert(0,str(ROOT/'build/trust237-round9'));import frontends as f
f.OUT=OUT
# New family: return-value conversion objects, not the already-tested argv
# owner or empty lexical guards. All predicates return one for every u32.
jobs=[]
for member,cleanup,ctor,rettype,exc,predicate,force,cpu,named in itertools.product(
 ['value','reference','pointer'],['outside','conversion','destructor'],[False,True],
 ['int','bool'],[False,True],['range','or'],[False,True],[6,7],[False,True]):
    field={'value':'unsigned long status;','reference':'const unsigned long &status;','pointer':'const unsigned long *status;'}[member]
    arg='&u' if member=='pointer' else 'u'
    load='*status' if member=='pointer' else 'status'
    param='unsigned long' if member=='value' else ('const unsigned long &' if member=='reference' else 'const unsigned long *')
    exception=' throw()' if exc else ''
    hint='__forceinline ' if force else ''
    constructor=hint+'ReturnCarrier('+param+' s,unsigned short **p)'+exception+':status(s),argv(p){}' if ctor else ''
    dtor=hint+'~ReturnCarrier()'+exception+'{'+('LocalFree(argv);' if cleanup=='destructor' else '')+'}'
    body='unsigned long u='+load+';'
    if cleanup=='conversion':body+='LocalFree(argv);'
    if predicate=='range':body+='int r=0;if((u&1u)<=1u)goto yes;return r;yes:return r+1;'
    else:body+='return (u|1u)!=0u;'
    carrier='struct ReturnCarrier{'+field+'unsigned short **argv;'+constructor+dtor+hint+'operator '+rettype+'() const'+exception+'{'+body+'}};\n'
    expression='ReturnCarrier('+arg+',argv)' if ctor else 'result'
    decl=''
    if not ctor:decl='ReturnCarrier result={'+arg+',argv};'
    elif named:decl='ReturnCarrier result('+arg+',argv);';expression='result'
    # Even aggregate mode has a distinct direct/named scope choice.
    elif not named:pass
    tail=('LocalFree(argv);' if cleanup=='outside' else '')+decl+'return '+expression+';'
    if not ctor and not named:tail='{'+tail+'}'
    src=f.base.replace(f.marker,'long status=WinVerifyTrust(0,&action,&data);unsigned long u=(unsigned long)status;'+tail)
    src=src.replace('/O1 /G7',f'/O1 /G{cpu}').replace('int Rva00237D3DCheck() {',carrier+'int Rva00237D3DCheck() {')
    jobs.append((f'c{len(jobs):04}',src,dict(member=member,cleanup=cleanup,constructor=ctor,result=rettype,throw=exc,predicate=predicate,forceinline=force,cpu=cpu,named=named)))
f.execute('carriers',jobs)
