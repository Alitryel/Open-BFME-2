from pathlib import Path
import sys,itertools,concurrent.futures,json
sys.path.insert(0,'build/df20-scalars-20261020');import trials
trials.scratch=Path.cwd()/'build/df20-scalars-20261020'
base=(trials.scratch/'range-x0-specialized-b1-r0-ttemporary.cpp').read_text();trials.symbol=base.splitlines()[0][3:]
a='''  if (__digits.size() == 0)
#pragma inline_depth(0)
    return __s;
#pragma inline_depth(255)'''
b='''  if (__cp == __digits_first)
#pragma inline_depth(1)
    return __s;
#pragma inline_depth(255)'''
assert a in base and b in base
specs=[]
for first,second,visible in itertools.product(['return','guard','else'],['return','guard','else'],[0,1]):
 s=base
 if visible:s=s.replace('template <> _String_base<wchar_t, allocator<wchar_t> >::~_String_base();','')
 for mode,needle,condition,depth in [(second,b,'__cp == __digits_first',1),(first,a,'__digits.size() == 0',0)]:
  if mode=='return':continue
  # Transform from inside out; every path still copies the return iterator
  # before leaving the scope of its live formatting strings.
  at=s.index(needle);end=s.rindex('  return __s;')+len('  return __s;')
  rest=s[at+len(needle):end]
  fallback=f'#pragma inline_depth({depth})\n  return __s;\n#pragma inline_depth(255)'
  if mode=='guard':replacement='  if (!('+condition+')) {\n'+rest+'\n  }\n'+fallback
  else:replacement='  if ('+condition+') {\n'+fallback+'\n  } else {\n'+rest+'\n  }'
  s=s[:at]+replacement+s[end:]
 specs.append((f'guards-f{first}-s{second}-v{visible}',s))
with concurrent.futures.ThreadPoolExecutor(max_workers=8) as p:r=list(p.map(trials.run,specs))
(trials.scratch/'guard_scopes.json').write_text(json.dumps(r,indent=2))
