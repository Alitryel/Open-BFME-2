import sys
sys.path.insert(0,'/work2/Open-BFME-2/build/c2tr')
from wdbg import Dbg
# rec7.py SRC OBJ ROUND : dump all records in 82978's set (arg3) at that round's entry (updateMeleeAttack cls0)
src,obj,R=sys.argv[1],sys.argv[2],int(sys.argv[3])
d=Dbg(src,obj,obj+'.log')
d.prompt();d.prompt('set $CanDeferOnBPByAddr = 1')
bps=[0x83acd,0x82978]; added=set()
for b in bps:
    o=d.prompt('break *0x%x'%(0x10700000+b))
    if 'Breakpoint' in o and 'Unable' not in o: added.add(b)
names={}
def nm(sym):
    if sym in names: return names[sym]
    sw=d.words(sym,1); pw=d.words(sw[0],2) if sw and sw[0]>0x100000 else []
    n=d.string(pw[1]).strip() if len(pw)>1 and pw[1]>0x100000 else '?'
    n=n.strip('"').lstrip('_'); names[sym]=n; return n
def members(s):
    out=[]; c=d.words(s,1)[0]; m=0
    while c and m<64:
        m+=1; w=d.words(c,3)
        for i in range(32):
            if w[2]>>i&1: out.append(w[0]+i)
        c=w[1]
    return out
def rec(idx):
    p=d.words(0x108807b8+4*(idx&0x3ff),1)[0]
    while p:
        w=d.words(p,24)
        if w[7]==idx: return p,w
        p=w[12]
    return 0,[]
on=False;k=0
while True:
    out=d.prompt('cont')
    if 'terminated' in out: break
    eip=d.reg('eip')-0x10700000
    for b in bps:
        if b not in added and b!=eip: d.prompt('break *0x%x'%(0x10700000+b)); added.add(b)
    added.add(eip)
    if eip==0x83acd:
        ebp=d.reg('ebp'); f=d.words(ebp,1)[0]; curfn=d.string(d.words(f+4,1)[0]); on='updateMeleeAttack' in curfn and d.reg('ebx')==0; k=0; continue
    if not on: continue
    k+=1
    if k!=R: continue
    esp=d.reg('esp'); st=d.words(esp,5)
    a4=members(st[4]) if st[4] else []
    print('a4',' '.join('%x'%x for x in a4))
    for idx in members(st[3]):
        p,w=rec(idx)
        print('%3x %-14s @%x %s'%(idx,nm(w[0])[:14],p,' '.join('%x'%v for v in w)),flush=True)
    break
d.p.kill()
