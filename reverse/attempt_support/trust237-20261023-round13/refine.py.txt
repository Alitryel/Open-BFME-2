from pathlib import Path
import sys,itertools
ROOT=Path.cwd();OUT=ROOT/'build/trust237-round13'
sys.path.insert(0,str(ROOT/'build/trust237-round9'));import frontends as f
f.OUT=OUT
conditions=['(unsigned char)u==0x80u','(unsigned short)u==0x8000u',
 '(u&0x7fu)==0x40u','(u&0xffffu)==0x8000u','(unsigned char)u==2u',
 '(signed char)u==-128','(u>>31)==1u','(u&0x80000000u)!=0u']
predicates=['(u&1u)<=1u','(u&bound)<=bound','(unsigned char)u<=255u','(u|1u)!=0u']
jobs=[]
for condition,pred,shape,ref,exc,cpu in itertools.product(conditions,predicates,[0,1],[False,True],[False,True],[6,7]):
    decl='const unsigned long &bound=1u;' if ref else 'unsigned long bound=1u;'
    helper='__forceinline void releaseArgs(unsigned short **p)'+(' throw()' if exc else '')+'{LocalFree(p);}\n'
    if shape==0:
        tail='if(P){releaseArgs(argv);r=(int)(R);}else{releaseArgs(argv);r=(int)(R);}return r;'
    else:
        tail='if(P)goto left;goto right;left:releaseArgs(argv);r=(int)(R);goto done;right:releaseArgs(argv);r=(int)(R);goto done;done:return r;'
    tail=tail.replace('P',condition).replace('R',pred)
    src=f.base.replace(f.marker,'long status=WinVerifyTrust(0,&action,&data);unsigned long u=(unsigned long)status;int r=0;'+decl+tail)
    src=src.replace('int Rva00237D3DCheck() {',helper+'int Rva00237D3DCheck() {').replace('/O1 /G7',f'/O1 /G{cpu}')
    jobs.append((f'r{len(jobs):04}',src,dict(condition=condition,predicate=pred,shape=shape,bound_reference=ref,throw=exc,cpu=cpu)))
if __name__=='__main__':f.execute('refine',jobs)
