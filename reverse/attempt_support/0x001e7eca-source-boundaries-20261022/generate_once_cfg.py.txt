def wrap(x,k):
 return [x,'do {'+x+'} while(false);','for(int once=0; once<1; ++once){'+x+'}','for(bool once=true;once;once=false){'+x+'}','while(true){'+x+'break;}','for(;;){'+x+'break;}','switch(0){case 0:{'+x+'}break;}','for(int once=1; once; --once){'+x+'}'][k]
for p,q,place in itertools.product(range(8),range(8),range(2)):
 o=s
 sea=s[a:b];ground=s[c:d]

 else:
