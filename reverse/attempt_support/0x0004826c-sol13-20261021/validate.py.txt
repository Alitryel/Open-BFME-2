from pathlib import Path
import importlib, json, re, hashlib
from trials import BASE, DIR
# Added linkage-language string C is not a runtime string literal.
literal_set=lambda s:set(re.findall(r'"(?:[^"\\]|\\.)*"',s)) - {'"C"'}
out=[]
for family in ['zerocompare']:
 module=importlib.import_module(family)
 items=list(module.generate())
 results=json.loads((DIR/(family+'-results.json')).read_text())
 index={r['tag']:r for r in results}
 assert len(index)==len(results)==len(items),family
 hashes=[]
 for tag,source,*extra in items:
  assert (DIR/(tag+'.cpp')).read_text()==source,tag
  assert literal_set(source)==literal_set(BASE),tag
  assert 'error' not in index[tag],tag
  assert (DIR/(tag+'.obj')).exists(),tag
  hashes.append([tag,hashlib.sha256(source.encode()).hexdigest()])
 keys={(r['size'],r['nd'],r['regions'],r['slots']) for r in results}
 out.append({'family':family,'count':len(results),
  'source_hashes_sha256':hashlib.sha256(json.dumps(hashes,sort_keys=True).encode()).hexdigest(),
  'best':min(results,key=lambda r:(r['regions'],r['nd'])),
  'counts':{str(k):sum((r['size'],r['nd'],r['regions'],r['slots'])==k for r in results) for k in sorted(keys)},
  'validated_sources_and_literal_sets':True})
(DIR/'validated-results.json').write_text(json.dumps(out,indent=2)+'\n')
print('Validated',sum(r['count'] for r in out),'successful source trials')
