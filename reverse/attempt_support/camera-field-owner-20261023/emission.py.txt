from wdbg import *
W=Path(__file__).resolve().parent
src=W/sys.argv[1];tag=src.stem+'-emission'
d=Dbg(src,W/(tag+'.obj'),W/(tag+'.log'));rows=[]
try:
 d.prompt();d.prompt('set $CanDeferOnBPByAddr = 1');d.prompt('break *0x107b0010')
 for _ in range(200):
  o=d.prompt('cont',30)
  if 'terminated' in o or not o.strip():break
  f=d.reg('esi');fw=d.words(f,4)
  nm=d.string(d.words(fw[0],2)[1]);av=d.words(0x108a7b30,1)
  r=dict(function=nm,ebp_available=av);rows.append(r);print(r,flush=True)
 else:raise RuntimeError('too many emission functions')
finally:
 (W/(tag+'.json')).write_text(json.dumps(rows,indent=1))
 if d.p.poll() is None:
  try:d.prompt('quit')
  except Exception:d.p.kill()
 d.log.close()
