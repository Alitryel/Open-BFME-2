from regions import *
# New operand-type hypothesis: exact object-representation transfers through
# float/short/byte locals can have different early IL types from word copies.
helpers=[
 'float x; unsigned int r; memcpy(&x,&v,4); memcpy(&r,&x,4); return r;',
 'union Bits {unsigned int u; float f;}; Bits a,b; a.u=v; b.f=a.f; return b.u;',
 'struct Parts {unsigned short lo,hi;}; Parts a,b; unsigned int r; memcpy(&a,&v,4); b.lo=a.lo; b.hi=a.hi; memcpy(&r,&b,4); return r;',
 'struct Parts {unsigned char a,b,c,d;}; Parts x,y; unsigned int r; memcpy(&x,&v,4); y.a=x.a; y.b=x.b; y.c=x.c; y.d=x.d; memcpy(&r,&y,4); return r;',
]
def generate():
 for ei,helper in enumerate(helpers):
  for bits in itertools.product(range(2),repeat=7):
   dy,dn,oy,on,test,guard,contract=bits
   def cp(var,bit,typ):return '('+typ+' *)copyWord((unsigned int)('+var+'))' if bit else var
   arms='if('+('(unsigned)p>0' if test else 'p')+') {draw='+cp('p',dy,'Drawable')+'; obj='+cp('p->m_object',oy,'Object')+';} else {draw='+cp('p',dn,'Drawable')+'; obj='+cp('(Object*)p',on,'Object')+';}'
   src=BASE[:lo]+decl+arms+' if('+('p && obj' if guard else 'obj')+') {'+body+'\n} else {'+terrain+'}\n'+BASE[hi:]
   if not contract:src=src.replace('template <> void StringBase<char>::releaseBuffer() throw();\n','')
   src=src.replace('#define W(s)', 'extern "C" void *__cdecl memcpy(void *, const void *, unsigned int);\n#pragma intrinsic(memcpy)\n#define W(s)')
   src=src.replace('void W3DDisplay::gatherDebugStats(void)','static __forceinline unsigned int copyWord(unsigned int v) {'+helper+'}\nvoid W3DDisplay::gatherDebugStats(void)')
   yield ('tc'+str(ei)+''.join(map(str,bits)),src)
if __name__=='__main__':batch(list(generate()),'typedcopy')
