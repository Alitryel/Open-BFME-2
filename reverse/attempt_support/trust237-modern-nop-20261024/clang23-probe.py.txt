from pathlib import Path
import sys,json,itertools,hashlib
sys.path.insert(0,str(Path.cwd()/'tools'))
import build,capstone,recovery_experiments
out=Path('build/trust237-clang23-nop');out.mkdir(parents=True,exist_ok=True)
base=Path('reverse/attempt_support/trust237-20261022-round5/base.cpp.txt').read_text()
base=base[base.index('struct TrustGuid'):]
base=base.replace('int Rva00237D3DCheck() {','extern "C" void __nop(void);\n#pragma intrinsic(__nop)\nint Rva00237D3DCheck() {')
marker='    WinVerifyTrust(0,&action,&data);\n    LocalFree(argv);\n    return 1;'
assert marker in base
nops='\n'.join('    __nop();' for _ in range(16))
tails=[('plain', '    WinVerifyTrust(0,&action,&data);\n    LocalFree(argv);\n'+nops+'\n    return 1;'),('result_early','    WinVerifyTrust(0,&action,&data);\n    LocalFree(argv);\n    int result=0;\n'+nops+'\n    return ++result;'),('result_decl','    int result=0;\n    WinVerifyTrust(0,&action,&data);\n    LocalFree(argv);\n'+nops+'\n    return result+1;')]
rows=[];md=capstone.Cs(capstone.CS_ARCH_X86,capstone.CS_MODE_32)
for opt,frame,intr,(kind,tail) in itertools.product(['/O1','/O2','/Od'],['/Oy','/Oy-'],['/Oi','/Oi-'],tails):
    name=f'c{len(rows):03d}';src=out/(name+'.cpp')
    text='// compiler: clang23-x86\n// cl: '+f'{opt} {frame} {intr} /G6 /arch:SSE /DNDEBUG /MD /EHsc'+'\n'+base.replace(marker,tail)
    src.write_text(text,encoding='utf-8',newline='\n')
    try:r=recovery_experiments.run_experiment(src,'?Rva00237D3DCheck@@YAHXZ',0x237D3D,226,cache=out/'cache')
    except (Exception,SystemExit) as e:r={'status':'setup-error','error':str(e)}
    row={k:r.get(k) for k in ['status','object','emitted_size','first_mismatch','unresolved','bytes_match','candidate_exact','cache_hit','error','artifact_dir']}
    row.update(name=name,opt=opt,frame=frame,intr=intr,kind=kind,source_sha256=hashlib.sha256(text.encode()).hexdigest())
    if r.get('object'):
        raw,rel=build.read_object_symbol_bytes(Path(r['object']),'?Rva00237D3DCheck@@YAHXZ',code_only=True)
        ins=list(md.disasm(raw,0));row.update(raw_sha256=hashlib.sha256(raw).hexdigest(),raw_hex=raw.hex(),relocations=rel,nops=[i.address for i in ins if i.mnemonic=='nop'],instructions=[f'{i.address:03X} {i.mnemonic} {i.op_str}' for i in ins])
    rows.append(row);(out/'results.json').write_text(json.dumps(rows,indent=2)+'\n',encoding='utf-8')
    print(name,opt,frame,intr,kind,row.get('status'),row.get('emitted_size'),row.get('first_mismatch'),'nops',len(row.get('nops',[])),row.get('error',''),flush=True)
print('SUMMARY',len(rows),'outputs',sum(bool(r.get('object')) for r in rows),'exact',sum(bool(r.get('candidate_exact')) for r in rows),'distinct',len({r.get('raw_sha256') for r in rows if r.get('raw_sha256')}),flush=True)
