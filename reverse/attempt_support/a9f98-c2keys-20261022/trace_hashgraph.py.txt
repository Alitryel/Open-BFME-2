import sys,json
from pathlib import Path
import importlib.machinery
support=Path(__file__).resolve().parent
Dbg=importlib.machinery.SourceFileLoader('apt_trace_wdbg', str(support/'wdbg.py.txt')).load_module().Dbg
W=Path('build/a9f98-c2-trace');W.mkdir(parents=True,exist_ok=True)
src=Path(sys.argv[1]) if len(sys.argv)>1 else Path('reverse/attempts/0x000a9f98.cpp')
tag=sys.argv[2] if len(sys.argv)>2 else 'hashgraph-base'
d=Dbg(str(src),str(W/(tag+'.obj')),str(W/(tag+'.log')))
records=[];syms={};active=False
try:
 d.prompt();d.prompt('set $CanDeferOnBPByAddr = 1');d.prompt('break *0x10760539')
 for i in range(10000):
  o=d.prompt('cont',60)
  if 'terminated' in o or not o.strip():break
  if '0x10760539' in o:
   p=d.reg('eax');w=d.words(p,1);s=d.words(w[0],2);fn=d.string(s[1])
   if active:break
   if '?render@AptShapeContainer@@' in fn:
    active=True;d.prompt('break *0x10761546');print('render begin',flush=True)
   continue
  if active and '0x10761546' in o:
   p=d.reg('eax');s=d.words(p,24);ops=d.words(0x10882190,16);op=d.reg('ebp')
   ops=ops[:ops.index(0)] if 0 in ops else ops
   args=[]
   for a in ops:
    nw=d.words(a,16);arg={'p':a,'node':nw}
    if len(nw)>6 and nw[1] in [0x250,0x24f]:arg['sym']=d.words(nw[6],12)
    args.append(arg)
   records.append({'p':p,'w':s,'op':op,'args':args})
   if len(records)%250==0:print('records',len(records),'H',s[7],flush=True)
finally:
 (W/(tag+'.json')).write_text(json.dumps({'records':records,'syms':syms},indent=1))
 print('done',len(records),flush=True)
 if d.p.poll() is None:
  try:d.prompt('quit')
  except Exception:d.p.kill()
 d.log.close()
