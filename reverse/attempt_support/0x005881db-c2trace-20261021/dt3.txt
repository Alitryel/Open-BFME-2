import sys,os
sys.path.insert(0,'/work/Open-BFME-2/build/c2tr')
from wdbg import Dbg
# keep.py SRC OBJ ROUND:NAME[,NAME...] : at round ROUND's first a8f54 call keep only NAMEs in the split set
src,obj,tgt=sys.argv[1:4]
TR,TN=tgt.split(':'); TR=int(TR); TN=TN.split(',')
PS=[(int(a),b,int(c)) for a,b,c in (x.split(':') for x in os.environ.get('PSET','').split(',') if x)]
TRC=[int(x) for x in os.environ.get('TRACE','').split(',') if x]; WN=os.environ.get('WN','node')
d=Dbg(src,obj,obj+'.log')
d.prompt();d.prompt('set $CanDeferOnBPByAddr = 1')
B=0x10700000
bps=[0x83acd,0x82978,0xa8f54,0x831f7,0x8ba9d]; added=set()
for b in bps:
    o=d.prompt('break *0x%x'%(B+b))
    if 'Breakpoint' in o and 'Unable' not in o: added.add(b)
names={}
def nm(sym):
    if sym in names: return names[sym]
    sw=d.words(sym,1); pw=d.words(sw[0],2) if sw and sw[0]>0x100000 else []
    n=d.string(pw[1]).strip() if len(pw)>1 and pw[1]>0x100000 else '?'
    n=n.strip('"').lstrip('_'); names[sym]=n; return n
def rec_of(i):
    p=d.words(0x108807b8+(i&0x3ff)*4,1)[0]
    while p:
        w=d.words(p,13)
        if w[7]==i: return p,w
        p=w[12]
    return None,None

def members(hdr):
    out=[]; ch=d.words(hdr,1)[0] if hdr else 0
    while ch:
        ww=d.words(ch,3)
        if len(ww)<3: break
        base,nx,bits=ww; out+=[base+j for j in range(32) if bits>>j&1]; ch=nx
    return out
def s32(v): return v-(1<<32) if v>=1<<31 else v
on=False;k=0;calls={};nrounds=0
while True:
    out=d.prompt('cont')
    if 'terminated' in out: break
    eip=d.reg('eip')-B
    for b in bps:
        if b not in added and b!=eip: d.prompt('break *0x%x'%(B+b)); added.add(b)
    added.add(eip)
    if eip==0x83acd:
        ebp=d.reg('ebp'); f=d.words(ebp,1)[0]; curfn=d.string(d.words(f+4,1)[0]); on='updateMeleeAttack' in curfn and d.reg('ebx')==0; k=0; continue
    if not on: continue
    if eip==0x82978: k+=1; continue
    if eip==0x8ba9d:
        if PS:
            esi=d.reg('esi'); w=d.words(esi,18)
            if os.environ.get('DBG'): print('  8ba9d k=%d %s:%x w3=%d'%(k,nm(w[0]),w[7],w[3]),flush=True)
            for r,n,v in PS:
                if r==k and nm(w[0])==n:
                    print('  pset R%d %s:%x w3 %d -> %d'%(k,n,w[7],w[3],v),flush=True); d.prompt('set *(int*)0x%x = %d'%(esi+0xc,v))
        continue
    if eip==0x831f7:
        if k in TRC:
            esp=d.reg('esp'); a0,a1,a2=d.words(esp+4,3); w=d.words(a1,18)
            regrec=w[4]; reg=d.words(regrec+0x1c,1)[0] if regrec else -1
            nb=members(a2) if a2 else []
            s='  R%d col %s:%x w3=%d reg=%x nb=%d'%(k,nm(w[0]),w[7],w[3],reg,len(nb))
            for i in nb:
                p,ww=rec_of(i)
                if ww and nm(ww[0])==WN: s+=' | %s:%x allowed=%s'%(WN,i,members(ww[8]) if ww[8] else None)
            print(s,flush=True)
        continue
    if eip==0xa8f54:
        esp=d.reg('esp'); st=d.words(esp,4); ra=st[0]-B
        c=calls.get(k,0); calls[k]=c+1
        ch=d.words(st[2],1)[0]; items=[]
        while ch:
            base,nx,bits=d.words(ch,3)
            for j in range(32):
                if bits>>j&1:
                    i=base+j; p,w=rec_of(i); items.append((ch,base,i,w))
            ch=nx
        print('R%d call%d n=%d :: %s'%(k,c,len(items),' '.join('%s:%x'%(nm(w[0]),i) if w else '%x'%i for _,_,i,w in items)),flush=True)
        if k==TR and c==0:
            for chn,base,i,w in items:
                if w and nm(w[0]) in TN:
                    bits=d.words(chn,3)[2]; bits&=~(1<<(i-base)); d.prompt('set *(int*)0x%x = %d'%(chn+8,s32(bits)))
            print('dropped',TN,flush=True)
        if k>40: print('too many rounds');break
        continue
print('done',flush=True)
