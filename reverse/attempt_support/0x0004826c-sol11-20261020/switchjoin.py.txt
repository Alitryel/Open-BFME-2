from headmerge import *
# Prior switch trials used bool(p), which exposes pointer-null facts early.
# A direct integral selector has a distinct case-edge propagation path.
selectors=['(unsigned)p','(int)p','(unsigned __int64)(unsigned)p','(__int64)(int)p']
items=[]
for selector,reverse,jump,nul,order,init,contract in itertools.product(range(4),range(2),range(2),range(2),range(2),range(2),range(2)):
 null='(Object*)p' if nul else '0'
 a='draw=p; obj=p->m_object;' if not order else 'obj=p->m_object; draw=p;'
 b='draw=p; obj='+null+';'
 if init:b='draw=p;'+('obj='+null+';'if nul else '')
 if jump:
  cases=['case 0: goto nullcase;', 'default: goto objectcase;']
  end='objectcase: '+a+' goto joined; nullcase: '+b+' joined:;'
 else:
  cases=['case 0: '+b+' break;', 'default: '+a+' break;'];end=''
 if reverse:cases.reverse()
 form=('obj=0;'if init else '')+'switch('+selectors[selector]+') {'+' '.join(cases)+'} '+end+'if(obj) {'+body+'\n} else {'+terrain+'}\n'
 src=BASE[:lo]+decl+form+BASE[hi:]
 if not contract:src=src.replace('template <> void StringBase<char>::releaseBuffer() throw();\n','')
 items.append(('sj%d%d%d%d%d%d%d'%(selector,reverse,jump,nul,order,init,contract),src))
if __name__=='__main__':batch(items,'switchjoin')
