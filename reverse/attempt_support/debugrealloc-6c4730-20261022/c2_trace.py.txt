def symdesc(ptr):
    if not ptr:return None
    try:
        w=words(ptr,12)
        nm=string(words(words(w[0],1)[0]+4,1)[0]) if w[0] else ''
        return {'ptr':hex(ptr),'class':mem(ptr+4,1)[0],'name':nm,'words':list(w)}
    except OSError:return {'ptr':hex(ptr)}

def operand(ptr,depth=0):
    if not ptr:return None
    try:
        w=words(ptr,24)
        return {'ptr':hex(ptr),'kind':mem(ptr+8,1)[0],'id':w[6],
                'symbol':symdesc(w[7]),'symbol18':symdesc(w[6]) if (w[2]&255)==1 else None,'words':list(w),'base34':operand(w[13],depth+1) if (w[2]&255) in (5,6) and w[13] and depth<1 else None,'index38':operand(w[14],depth+1) if (w[2]&255) in (5,6) and w[14] and depth<1 else None}
    except OSError:return {'ptr':hex(ptr)}
