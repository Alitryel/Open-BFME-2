#!/bin/sh
# evalv2.sh src : name nd n15 c15T N cnd
cd /work/Open-BFME-2/build/db
n=$(basename "$1" .cpp)
nd=$(python3 dd.py "$1" | sed -n 's/.* nd \([0-9]*\) .*/\1/p')
c=$(timeout 900 python3 count15.py "$1" "e_$n" | sed -n 's/.* n15 \([0-9]*\) .*/\1/p')
t=$(python3 c15t.py "e_$n" | awk '{print $2}')
N=$(( ( (22 - t) % 32 + 32) % 32 ))
if [ "$N" -eq 0 ]; then cnd=$nd; else
timeout 900 python3 inject2.py "$1" "j_$n" "1:$N" > /dev/null 2>&1
cnd=$(python3 scoreobj.py tr/j_$n.obj 2>&1 | awk '{print $NF}'); rm -f tr/j_$n.obj tr/j_$n.log; fi
rm -f tr/e_$n.cnt.log
echo "$n nd=$nd n15=$c c15T=$t N=$N cnd=$cnd"
