def brace_end(s,start):
 opening=s.index('{',start);depth=1;i=opening+1
 while depth:
  depth+=(s[i]=='{')-(s[i]=='}');i+=1
 return i

regions=[]
a=BODY.index('\tif (combo.m_window)');regions.append((a,brace_end(BODY,a)))
a=BODY.index('\tif (m_handicap[slot])');b=BODY.index('\tGameWindow *hero',a);regions.append((a,b))
a=BODY.index('\tif (hero)');regions.append((a,brace_end(BODY,a)))
a=BODY.index('\tif (m_playerTemplate[slot])');b=BODY.index('\tGameWindow **team',a);regions.append((a,b))
a=BODY.index('\tif (*team)');regions.append((a,brace_end(BODY,a)))

def source(mask,predicate,form):
 text=BODY
 for n,(a,b) in reversed(list(enumerate(regions))):
  if mask>>n&1:
   block=text[a:b]
   cond=['enable','observer','slot == 0','!enable && observer'][predicate]
   if form==0:replacement='if ('+cond+') {\n'+block+'\n} else {\n'+block+'\n}\n'

   text=text[:a]+replacement+text[b:]
 return HEAD+MARK+text
