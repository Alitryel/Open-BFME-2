import sys,json,itertools,time
from pathlib import Path
from concurrent.futures import ThreadPoolExecutor,as_completed
sys.path.insert(0,'build/fxinit-seat22');import search as h
h.HERE=Path('build/fxinit-admit23').resolve()
def variants():
 for refs,vec,place,order,ctor,pguard in itertools.product(range(8),range(2),range(4),range(6),range(2),range(2)):
  if vec and refs&1:continue # The vector input is necessarily a reference.
  f=[('_STL::vector<short> &' if vec else 'short *const &' if refs&1 else 'short *','map','remap' if vec else 'remap._M_start'),('const unsigned short *const &' if refs&2 else 'const unsigned short *','a','links[0]'),('const unsigned short *const &' if refs&4 else 'const unsigned short *','b','links[1]')]
  fs=[f[i]for i in list(itertools.permutations(range(3)))[order]];args=','.join(x[2] for x in fs)
  definition='struct PackView{'+''.join(t+' '+n+';' for t,n,a in fs)
  if ctor:definition+='__forceinline PackView('+','.join(t+' p'+n for t,n,a in fs)+'):'+','.join(n+'(p'+n+')'for t,n,a in fs)+'{}'
  definition+='};\nstatic __forceinline int PackViewValue(PackView v,unsigned i){return v.map[v.a[i]]+(v.map[v.b[i]]<<8);}\n'
  pre='PackView view('+args+');' if ctor else 'PackView view={'+args+'};'
  s=h.BASE
  sites=['  for(unsigned pass=0;','    *(Vector3 *)(vertices+desc.Get_Location_Offset(0))=positions[0][i];','    if(basis) {','    if(m_Passes>1) *(unsigned *)']
  s=s.replace(sites[place],pre+'\n'+sites[place])
  s=s.replace(h.R,'*(int *)(vertices+desc.Get_Basis_End())=PackViewValue(view,i);')
  if pguard:s=s.replace('if(positions[1])','if(*(const Vector3 *volatile *)&positions[1])')
  s=s.replace('bool FXShaderGeometry::Init(',definition+'bool FXShaderGeometry::Init(')
  yield f'recombineview_{refs}{vec}{place}{order}{ctor}{pguard}',s
if __name__=='__main__':
 t=time.time();rs=[]
 with ThreadPoolExecutor(max_workers=8)as ex:
  for f in as_completed([ex.submit(h.run_one,x)for x in variants()]):
   rs.append(f.result())
   if len(rs)%256==0:print(len(rs),min(rs,key=lambda x:(x.get('nd',99999),x.get('cost',99999))),flush=True)
 rs.sort(key=lambda z:(z.get('nd',99999),z.get('cost',99999)))
 (h.HERE/'recombineview-results.json').write_text(json.dumps(rs,indent=2))
 print(json.dumps({'count':len(rs),'seconds':time.time()-t,'errors':sum('error'in z for z in rs),'best':rs[:6]},indent=2))
