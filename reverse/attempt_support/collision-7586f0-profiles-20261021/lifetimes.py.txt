from pathlib import Path
import sys,json
root=Path.cwd();sys.path.insert(0,str(root/'tools'))
from variant_search import Scorer,diff_text
from locate import object_functions
out=root/'build/collision-finish';src=(out/'banked.cpp').read_text()
sym='?rva007586F0@CollisionManagerImpl@@QAEXXZ'
variants=[]
for B in ['A + 1','&lists[src + 1]']:
 base=src.replace('A + 1',B)
 for cond in ['A[1].count','B->count']:
  code=base.replace('while (lists[src + 1].count != 0)\n\t\t{\n\t\t\tSortList *A = &lists[src];\n\t\t\tSortList *B = '+B+';', 'SortList *A = &lists[src]; SortList *B = '+B+';\n\t\twhile ('+cond+' != 0) {')
  code=code.replace('src ^= 2;','src ^= 2; A = &lists[src]; B = '+B+';')
  variants.append(('lifetime'+str(len(variants)),code))
 for form in ['&lists[src | 1]','&lists[src ^ 1]','(SortList *)((char *)lists + src * sizeof(SortList) + sizeof(SortList))','&lists[(unsigned)src + 1]']:
  variants.append(('index'+str(len(variants)),base.replace('SortList *B = '+B+';','SortList *B = '+form+';')))
 for repl in ['const SortList *A', 'SortList *const A']:
  # const pointee needs mutable source; const reference proves reads only, removal still through selected mutable node.
  if repl.startswith('const'):continue
  variants.append(('const'+str(len(variants)),base.replace('SortList *A',repl)))
 # Rotate the pool-array lifetime, not the pass CFG.
 for t in ['unsigned int','long','unsigned short']:
  variants.append(('source_type'+str(len(variants)),base.replace('int src = 0;',t+' src = 0;')))
scorer=Scorer(sym,0x7586f0,460,out/'objects2')
results=[]
for name,s in variants:
 r=scorer(s);a,b=r.pop('compiled',b''),r.pop('target',b'')
 extent=None
 obj=scorer.dir/('c'+str(scorer.n)+'.obj')
 if obj.exists():
  extent=next((len(v) for n,v,rel in object_functions(obj) if n==sym),None)
 r.update(name=name,extent=extent,different_bytes=sum(x!=y for x,y in zip(a,b))+abs(len(a)-len(b)))
 (out/(name+'.cpp')).write_text(s)
 if a:(out/(name+'.diff.txt')).write_text(diff_text(a,b,200))
 results.append(r);(out/'results2.json').write_text(json.dumps(results,indent=2));print(json.dumps(r),flush=True)
