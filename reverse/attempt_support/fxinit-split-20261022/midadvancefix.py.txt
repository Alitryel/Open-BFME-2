import sys,itertools,json,time,re
from pathlib import Path
from concurrent.futures import ThreadPoolExecutor
sys.path.insert(0,'build/fxinit-seat22');import search as h
h.HERE=Path('build/fxinit-split22').resolve()
LOOP='   for(unsigned i=0;i<(unsigned)mesh->Get_Vertex_Count();++i) {'
END='    vertices+=desc.Get_FVF_Size();'
def variants():
 for point,mode,iform,vform,scope,order,sgn in itertools.product(range(4),range(1,4),range(3),range(3),range(2),range(2),range(2)):
  src=h.BASE;old='\n      '.join([h.R,h.W,h.P,h.N]);at=old.index([h.R,h.W,h.P,h.N][point]);prefix=old[:at];tail=old[at:];decl=[];steps=[]
  if mode&1:
   src=src.replace(LOOP,LOOP.replace(';++i)', ';)'))
   if iform==0:decl.append(('int 'if sgn else'unsigned ')+'previousIndex;');steps.append('previousIndex=i++;');ix='previousIndex'
   elif iform==1:steps.append('++i;');ix='(i-1)'
   else:decl.append(('int 'if sgn else'unsigned ')+'previousIndex;');steps.append('previousIndex=i;i=previousIndex+1;');ix='previousIndex'
   tail=re.sub(r'\bi\b',ix,tail)
  if mode&2:
   if vform==0:decl.append('char *previousVertex;');steps.append('previousVertex=vertices;vertices+=desc.Get_FVF_Size();');vx='previousVertex'
   elif vform==1:steps.append('vertices+=desc.Get_FVF_Size();');vx='(vertices-desc.Get_FVF_Size())'
   else:decl.append('unsigned previousVertex;');steps.append('previousVertex=(unsigned)vertices;vertices=(char *)(previousVertex+desc.Get_FVF_Size());');vx='((char *)previousVertex)'
   tail=re.sub(r'\bvertices\b',vx,tail)
  if order:steps.reverse()
  if not mode&2:tail+='vertices+=desc.Get_FVF_Size();'
  tail+='goto AdvancedVertex;'
  new=prefix+''.join(steps)+tail
  if scope==0:new=''.join(decl)+new
  else:
   key=LOOP if not mode&1 else LOOP.replace(';++i)', ';)')
   src=src.replace(key,key+''.join(decl))
  assert old in src
  src=src.replace(END,END+('++i;' if mode&1 else '')+'\n AdvancedVertex:;').replace(old,new)
  yield f'midadvancefix_{point}{mode}{iform}{vform}{scope}{order}{sgn}',src
if __name__=='__main__':
 t=time.time()
 with ThreadPoolExecutor(max_workers=8)as pool:r=list(pool.map(h.run_one,variants()))
 r.sort(key=lambda x:(x.get('nd',99999),x.get('cost',99999)))
 (h.HERE/'midadvancefix-results.json').write_text(json.dumps(r,indent=2))
 print(json.dumps({'count':len(r),'valid':sum('nd'in x and not x.get('unresolved')for x in r),'seconds':time.time()-t,'best':r[:8]},indent=2))
