# usage: BPS="9f953:esi+1c,eax;9f958:eax" python3 bptrace.py src obj
import sys,os
sys.path.insert(0,os.path.dirname(os.path.abspath(__file__)))
from wdbg import Dbg
src,obj=sys.argv[1:3]
spec={}
for part in os.environ['BPS'].split(';'):
    a,exprs=part.split(':'); spec[int(a,16)]=[e for e in exprs.split(',') if e]
MAX=int(os.environ.get('MAX','100000'))
d=Dbg(src,obj,obj+'.log')
d.prompt();d.prompt('set $CanDeferOnBPByAddr = 1')
B=0x10700000
added=set()
def W(a,c):
    try: w=d.words(a,c)
    except Exception: w=[]
    return w+[0]*(c-len(w))
def ev(e):
    # reg, reg+off (deref dword at reg+off), reg+off+off2 (double deref), 'line'
    if e=='line': return W(B+0x1a9120,1)[0]&0xffff
    if e[0]=='q':
        ps=e[1:].split('+'); v=W(B+int(ps[0],16),1)[0]
        for off in ps[1:]:
            v=W(v+int(off,16),1)[0] if v else 0
        return v
    if e[0]=='m': return W(B+int(e[1:],16),1)[0]
    if e=='chain':
        x=d.reg('eax'); r=[]
        while x and len(r)<200:
            r.append('%x'%W(x+0x1c,1)[0]); x=W(x+0xc,1)[0]
        return '/'.join(r)
    if e=='stk': return '/'.join('%x'%(w-B) for w in W(d.reg('esp'),160) if B+0x1000<=w<B+0x1a0000)
    parts=e.split('+'); v=d.reg(parts[0])
    for off in parts[1:]:
        v=W(v+int(off,16),1)[0]
    return v
n=0
while n<MAX:
    for b in spec:
        if b in added: continue
        o=d.prompt('break *0x%x'%(B+b))
        if 'Breakpoint' in o and 'Unable' not in o: added.add(b)
    out=d.prompt('cont')
    if 'terminated' in out or 'exited' in out: break
    eip=d.reg('eip')-B
    if eip not in spec: continue
    vals=[(e,ev(e)) for e in spec[eip]]
    print('%x'%eip,' '.join(('%s=%s' if isinstance(v,str) else '%s=%x')%(e,v) for e,v in vals),flush=True); n+=1
d.p.kill()

import sys,subprocess,re,collections
from pathlib import Path
ROOT=Path('/work/Open-BFME-2'); sys.path.insert(0,str(ROOT/'tools'))
import build,capstone
SYM='?get_deformed_vertices@MeshModelClass@@QAEXPAVVector3@@PBVHTreeClass@@@Z'
def probe(srcpath, extra='', verbose=False):
    s=Path(srcpath).read_text().split('\n')
    for k,l in enumerate(s):
        if l.startswith('// cl:'):
            s[k]=l.replace('/DNDEBUG','/DNDEBUG /Od '+extra); break
    src=ROOT/'build/sk16/w19/dbg'/('o_'+Path(srcpath).stem+'.cpp'); src.parent.mkdir(exist_ok=True)
    src.write_text('\n'.join(s)); out=src.with_suffix('.obj')
    if out.exists(): out.unlink()
    cmd,env=build.compiler_command(src,out)
    p=subprocess.run(cmd,cwd=ROOT,env=env,capture_output=True,text=True)
    if p.returncode: return None,(p.stdout+p.stderr)[-800:]
    comp,rel=build.read_object_symbol_bytes(out,SYM,code_only=True)
    md=capstone.Cs(capstone.CS_ARCH_X86,capstone.CS_MODE_32)
    refs=collections.OrderedDict(); lines=[]
    for i in md.disasm(comp,0):
        lines.append('%x %s %s'%(i.address,i.mnemonic,i.op_str))
        for m in re.finditer(r'\[ebp ([-+]) (0x[0-9a-f]+|\d+)\]',i.op_str):
            off=int(m.group(2),0)*(-1 if m.group(1)=='-' else 1); refs.setdefault(off,[]).append('%x:%s %s'%(i.address,i.mnemonic,i.op_str))
    return (refs,lines),None
if __name__=='__main__':

def cvlocals(srcpath, extra=''):
    import struct
    s=Path(srcpath).read_text().split('\n')
    for k,l in enumerate(s):
        if l.startswith('// cl:'):
            s[k]=l.replace('/DNDEBUG','/DNDEBUG /Od /Z7 '+extra); break
    src=ROOT/'build/sk16/w19/dbg'/('z_'+Path(srcpath).stem+'.cpp'); src.parent.mkdir(exist_ok=True)
    src.write_text('\n'.join(s)); out=src.with_suffix('.obj')
    if out.exists(): out.unlink()
    cmd,env=build.compiler_command(src,out)
    p=subprocess.run(cmd,cwd=ROOT,env=env,capture_output=True,text=True)
    if p.returncode: return None,(p.stdout+p.stderr)[-800:]
    return out.read_bytes(),None
def frame(srcpath, extra='', fn='get_deformed_vertices'):
    import struct,re as _re
    d,e=cvlocals(srcpath,extra)
    if d is None: return None,e
    res={}; cur=None
    for m in _re.finditer(rb'(?s)(..)(\x0b\x11|\x10\x11|\x0f\x11)',d):
        ln=struct.unpack('<H',m.group(1))[0]; t=m.group(2)
        st=m.start()+4
        if t==b'\x0b\x11':
            off,ti=struct.unpack('<iI',d[st:st+8]); nm=d[st+8:d.index(b'\0',st+8)].decode('latin1')
            if cur is not None and len(nm)<40 and _re.match(r'^[A-Za-z_]\w*$',nm): res.setdefault(cur,[]).append((off,nm))
        else:
            # proc: name after fixed fields (35 bytes for 0x1110/0x110f in v3)
            nmst=st+35-4+4
            j=d.find(b"\0",st+35,st+300)
            cur=d[st+35:j].decode('latin1',errors='replace')
    out={}
    for k,v in res.items():
        if fn in k: out[k]=sorted(set(v),reverse=True)
    return out,None
