import sys, os, json, subprocess, concurrent.futures as cf, itertools, re, time, hashlib
from pathlib import Path
ROOT=Path('/work/Open-BFME-2'); sys.path.insert(0,str(ROOT/'tools'))
import build, capstone
from variant import resolve_bytes
HERE=Path(__file__).resolve().parent
SYM='?render@AptShapeContainer@@QAEXXZ'; RVA=0xa9f98; SIZE=1471
TARGET=build.read_target_bytes(RVA,SIZE)
MAP=build.load_symbol_map()
MAP['?rva001100A6@BfmeHub982@@QAIXPAX@Z']=[0x1100a6]
MAP['?rva001100C5@BfmeHub982@@QAIXPAX@Z']=[0x1100c5]
def asm(blob):
 md=capstone.Cs(capstone.CS_ARCH_X86,capstone.CS_MODE_32)
 return [(i.address,i.mnemonic,i.op_str) for i in md.disasm(blob,0)]
T=asm(TARGET)
def run(item):
 tag,src=item; path=HERE/(str(tag)+'.cpp'); obj=HERE/(str(tag)+'.obj'); path.write_text(src)
 command,env=build.compiler_command(path,obj)
 p=subprocess.run(command,cwd=ROOT,env=env,capture_output=True,text=True)
 if p.returncode:
  (HERE/(str(tag)+'.error')).write_text(p.stdout+p.stderr)
  return dict(tag=tag,score=100000,error=(p.stdout+p.stderr)[-500:])
 b,rels=build.read_object_symbol_bytes(obj,SYM,code_only=True)
 try:res,un=resolve_bytes(b,rels,TARGET,RVA,min(SIZE,len(b)),MAP)
 except SystemExit as e:
  result=dict(tag=tag,score=100000+abs(len(b)-SIZE),size=len(b),reason='extent/relocation',detail=str(e))
  (HERE/(str(tag)+'.json')).write_text(json.dumps(result))
  return result
 a=asm(res)
 diffs=[(t,c) for t,c in zip(T,a) if t!=c]
 score=len(diffs)+abs(len(a)-len(T))*2+abs(len(b)-SIZE)*2
 (HERE/(str(tag)+'.bin')).write_bytes(res)
 result=dict(tag=tag,score=score,size=len(b),nd=len(diffs),un=un,byte_diff=sum(x!=y for x,y in zip(res,TARGET)),regions=[sum(lo<=t[0]<hi for t,c in diffs) for lo,hi in [(0,0x180),(0x180,0x300),(0x300,0x460),(0x460,0x5bf)]])
 (HERE/(str(tag)+'.json')).write_text(json.dumps(result))
 return result

def batch(items):
 start=time.time()
 with cf.ThreadPoolExecutor(8) as ex:
  out=list(ex.map(run,items))
 (HERE/'results.json').write_text(json.dumps(out,indent=2))
 print('n',len(out),'seconds',round(time.time()-start,1),flush=True)
 for v in sorted(out,key=lambda x:x['score'])[:16]: print(json.dumps(v),flush=True)
 return out
if __name__=='__main__':
 batch([('baseline',(HERE/'base.cpp').read_text())])
