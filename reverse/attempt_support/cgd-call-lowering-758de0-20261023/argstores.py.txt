from itertools import product
from pathlib import Path
W=Path(__file__).resolve().parent
D=W/'argstores';D.mkdir(exist_ok=True)
bank=(W/'bank.cpp').read_text();control=(W/'uniform-control.cpp').read_text()
stores=['','node->first->source=node->first->source','node->second->source=node->second->source','node->first=node->first','node->second=node->second']
for i,(a,b,c,d,bridge,uniform) in enumerate(product(range(5),range(5),range(5),range(5),range(2),range(2))):
 s=control if uniform else bank
 def arg(j,v):return '('+stores[j]+','+v+')' if j else v
 old='node->second->source->prepare(node->point,node->normal)'
 new='node->second->source->prepare('+arg(a,'node->point')+','+arg(b,'node->normal')+')'
 assert old in s;s=s.replace(old,new,1)
 old='node->second->source->prepare(node->point,node->normal)' if uniform else 'bridge(node->second,node->point,node->normal)'
 new=('bridge(node->second,' if bridge else 'node->second->source->prepare(')+arg(c,'node->point')+','+arg(d,'node->normal')+')'
 assert old in s;s=s.replace(old,new,1)
 if bridge and uniform:
  at=s.index('class BfmeThingCGD')
  s=s[:at]+'__forceinline void *bridge(ContactData *s,const ContactVector &p,const ContactVector &n){return s->source->prepare(p,n);}\n'+s[at:]
 (D/('v%04d.cpp'%i)).write_text(s)
print('generated',i+1)
