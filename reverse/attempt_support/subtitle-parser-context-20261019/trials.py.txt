import json
import re
import subprocess
import sys
from pathlib import Path

root = Path(__file__).resolve().parents[2]
scratch = root / 'build/resume-eh-local'
base = (scratch / 'turn47-subtitle-baseline.cpp').read_text()
def without_definition(source, start):
    pos = source.index(start)
    brace = source.index('{', pos)
    depth = 0
    for end in range(brace, len(source)):
        if source[end] == '{': depth += 1
        elif source[end] == '}':
            depth -= 1
            if depth == 0:
                return source[:pos] + source[end + 1:]
    raise ValueError(start)

split = without_definition(base, 'void SubtitleManager::addSubtitle(')
split = without_definition(split, 'SubtitleManager::SubtitleManager(')

def opaque_callback(source):
    source = source.replace('void parseSubtitleText( INI *ini',
        'struct SubtitleFetchView { void *create; bool (__cdecl *fetch)(const char *, UnicodeString *); };\n\n'
        'void parseSubtitleText( INI *ini')
    source = source.replace('((Bool (__cdecl *)(const char *,UnicodeString *))manager->m_second)(label, &text)',
                            '((SubtitleFetchView *)manager)->fetch(label, &text)')
    return source

variants = {'separate-providers': split, 'callback-view': opaque_callback(base),
            'separate-and-callback': opaque_callback(split)}
results = []
for label, source in variants.items():
    # The bank's physical path sits outside the measured compiler region.
    # Preserve its established retail O2/SSE/G6 region settings explicitly.
    source = source.replace('// cl: ', '// cl: /O2 /arch:SSE /G6 ', 1)
    path = scratch / ('turn47-subtitle-' + label + '.cpp')
    path.write_text(source)
    proc = subprocess.run([sys.executable, str(root / 'tools/explain_mismatch.py'),
        '?parseSubtitleText@@YAXPAVINI@@PAX1PBX@Z', '--source',
        path.relative_to(root).as_posix(), '--rva', '0x00688B10', '--size', '1578'],
        cwd=root, capture_output=True, text=True)
    log = scratch / ('turn47-subtitle-' + label + '.txt')
    log.write_text(proc.stdout + proc.stderr)
    size = re.search(r'compiled symbol size: (\d+)', proc.stdout)
    diffs = [s for s in proc.stdout.splitlines() if s.startswith('!=')]
    result = {'variant': label, 'exit': proc.returncode,
              'compiled_size': int(size[1]) if size else None,
              'differing_instruction_rows': len(diffs),
              'exact': 'OK: bytes match' in proc.stdout,
              'log': log.relative_to(root).as_posix()}
    results.append(result)
    print(json.dumps(result), flush=True)
(scratch / 'turn47-subtitle-results.json').write_text(json.dumps(results, indent=2) + '\n')
