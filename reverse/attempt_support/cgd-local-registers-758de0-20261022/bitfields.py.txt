from pathlib import Path
import itertools
W=Path(__file__).resolve().parent;D=W/'bitfields';D.mkdir(exist_ok=True);b=(W/'base.cpp').read_text()
for k,(statefield,indexfield,stateunion,indexunion,signedstate,pack) in enumerate(itertools.product(range(2),range(2),range(2),range(2),range(2),[1,2,4,8])):
 st=('signed' if signedstate else 'unsigned')+' state'+(':32' if statefield else '')+';'
 ix='int index'+(':32' if indexfield else '')+';'
 if stateunion:st='union { '+st+' };'
 if indexunion:ix='union { '+ix+' };'
 s=b.replace('unsigned key0,key1,state;','unsigned key0,key1; '+st).replace('int index;',ix)
 s=s.replace('struct ContactVector {','#pragma pack(push,'+str(pack)+')\nstruct ContactVector {')+'\n#pragma pack(pop)\n'
 (D/('b%03d.cpp'%k)).write_text(s)
