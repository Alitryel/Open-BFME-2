import search
from search import *
import itertools

# Trial-only symbols for ABI-identical return-type spellings. This does not
# add pins or assert original source identity.
RETURNS=[
 ('int','H'), ('unsigned int','I'), ('long','J'), ('unsigned long','K'),
 ('void *','PAX'), ('GameInfo *','PAVGameInfo@@'),
 ('const GameInfo *','PBVGameInfo@@'), ('char *','PAD'),
]
for typ,mangle in RETURNS:
 search.SMAP['?rva0043DA65@Rva0043DA65@@QAE'+mangle+'XZ']=[0x43da65]

def source(flags):
 kind,local,context,split=flags
 typ,mangle=RETURNS[kind]
 head=HEAD.replace('class Rva0043DA65\n','class GameInfo;\nclass Rva0043DA65\n')
 head=head.replace('int rva0043DA65();',typ+' rva0043DA65();')
 body=BODY
 if local==1:
  body=body.replace('GameInfo *game = (GameInfo *)m_game->rva0043DA65();','Rva0043DA65 *holder=m_game; GameInfo *game=(GameInfo *)holder->rva0043DA65();')
 if local==2:
  body=body.replace('GameInfo *game = (GameInfo *)m_game->rva0043DA65();',typ+' result=m_game->rva0043DA65(); GameInfo *game=(GameInfo *)result;')
 if local==3:
  body=body.replace('GameInfo *game = (GameInfo *)m_game->rva0043DA65();','const unsigned long &result=(unsigned long)m_game->rva0043DA65(); GameInfo *game=(GameInfo *)result;')
 if context:
  # Body is exactly the established validated getter's source shape. The
  # one explicit layout is used only by these isolated trials.
  head=head.replace('unsigned char m_pad00[0x08];\n\tGameInfo *m_current;', 'virtual void dummy(); virtual bool validate(int); int m_pad04;\n\tGameInfo *m_current;')
  prefix=['','__declspec(noinline) ','__declspec(noinline) inline ','__forceinline '][context]
  definition=prefix+typ+' Rva0043DA65::rva0043DA65() { if (!validate((int)m_current)) m_current=0; return ('+typ+')m_current; }\n'
  head+=definition
 if split:
  p=body.index('\n\tif (!game)')
  body=body[:p]+'\n int index=slot;'+body[p:]
  import re
  p=body.index('\n\tif (slot == -1')
  body=body[:p]+re.sub(r'\bslot\b','index',body[p:])
 return head+MARK+body

if __name__=='__main__':
 run([('gettertype-%03d'%n,source(f))for n,f in enumerate(itertools.product(range(8),range(4),range(4),range(2)))], 'getter_types')
