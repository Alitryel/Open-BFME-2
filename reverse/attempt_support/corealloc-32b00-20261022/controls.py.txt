from pathlib import Path
import sys,json,hashlib
root=Path(__file__).resolve().parents[1];sys.path.insert(0,str(root/'tools'))
import build
from variant_search import Scorer,diff_text
from locate import object_functions

home=root/'Code/GameEngine/Source/Common/System/Rva00031010Insert.cpp'
source=home.read_text()
text='// cl: '+' '.join(build.source_extra_flags(home))+'\n'+source
out=root/'build/corealloc-revisit-20261022/acceptance-controls';out.mkdir(parents=True,exist_ok=True)
symbol='?rva00032B00@GeneralAllocator@Allocator@EA@@QAEPAXI@Z'
score=Scorer(symbol,0x32b00,343,out)
variants=[('production',text)]
scalar=text.replace('Rva00032B00ChunkLocation info(chunkSize, chunk);\n\t\trva00032B00InitializeChunk(info, chunkFlags);',
                    'chunk[1] = chunkSize | chunkFlags;\n\t\trva00032B00AddDoubleFencepost(chunk, 0);')
variants.append(('scalar-initialization',scalar))
variants.append(('without-post-zero-barrier',text.replace('\t\t_ReadWriteBarrier();\n\t\tRva00032B00ChunkLocation','\t\tRva00032B00ChunkLocation')))
variants.append(('without-ordered-prev-write',text.replace('*(CoreBlock *volatile *)&core->m_prev = head;','core->m_prev = head;')))
variants.append(('without-ordered-footer',text.replace('((volatile unsigned int *)fencepost)','((unsigned int *)fencepost)')))
records=[]
for name,variant in variants:
    result=score(variant)
    obj=out/f'c{score.n}.obj'
    raw=next(b.rstrip(b'\xcc') for n,b,r in object_functions(obj) if n==symbol)
    record=dict(name=name,extent=len(raw),exact=result['exact'] and len(raw)==343,fitness=result['fitness'],
                difference=('none' if result['exact'] else diff_text(result['compiled'],result['target'],100)),
                resolved_sha256=hashlib.sha256(result['compiled']).hexdigest(),
                retail_sha256=hashlib.sha256(result['target']).hexdigest())
    records.append(record)
    print(json.dumps(record),flush=True)
report=dict(input_revision='ab78405c8a56e382b3910e6a031cb8e8ee02ccb2',
            bfme1_revision='37c3022de1ab944c7ffe536440d637dc8199091c',
            source=home.relative_to(root).as_posix(),source_sha256=hashlib.sha256(home.read_bytes()).hexdigest(),
            compiler_flags=build.source_extra_flags(home),
            c2_sha256=hashlib.sha256((build.vc71_root()/'Vc7/bin/c2.dll').read_bytes()).hexdigest(),
            controls=records)
(out/'proof.json').write_text(json.dumps(report,indent=2)+'\n',newline='\n')
assert records[0]['exact'] and all(not r['exact'] for r in records[1:])
