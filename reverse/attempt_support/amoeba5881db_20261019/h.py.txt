import sys, os, re, json, difflib, importlib.util
from concurrent.futures import ThreadPoolExecutor
ROOT='D:/LAN/openbfme2/Open-BFME-2'
sys.path.insert(0,ROOT+'/tools'); os.chdir(ROOT)
import build, capstone
from pathlib import Path
SYM='?updateMeleeAttack@HordeMeleeAmoeba@@UAEXPAVObject@@@Z'
ROW={"name":SYM,"export_rva":"","target_rva":"0x005881DB","target_size":"2199","source":"","status":"matched","notes":""}
D=ROOT+'/build/nbr/am/b3/v/'
SM=build.load_symbol_map()
md=capstone.Cs(capstone.CS_ARCH_X86, capstone.CS_MODE_32)
def dis(b):
    out=[]
    for i in md.disasm(bytes(b),0):
        op=i.op_str
        if i.mnemonic.startswith('j') or i.mnemonic=='call': op='X'
        op=re.sub(r'0x[0-9a-f]{5,}','A',op); out.append(i.mnemonic+' '+op)
    return out
def score(path):
    src=Path(path).resolve(); obj=src.with_suffix('.obj')
    ok,text,code=build.try_compile_source(src,obj)
    if not ok: return None,'ERR '+[l for l in (text or '').splitlines() if 'error' in l][:1].__repr__()[-300:]
    row=dict(ROW); row['source']=src.relative_to(build.ROOT).as_posix()
    try: p=build.compile_function(row,SM,obj)
    except SystemExit as e: return None,'CFERR %s'%e
    t,c=p['target'],p['bytes']
    T,C=dis(t),dis(c)
    nd=sum(1 for l in difflib.unified_diff(T,C,lineterm='',n=0) if l[:1] in '+-' and not l.startswith(('+++','---')))
    return (len(c),nd,t==c),None
def run_spec(base,variants,workers=8):
    bt=open(base,newline='').read()
    def one(item):
        name,reps=item
        if isinstance(reps,str):
            f=D+name+'.cpp'; open(f,'w',newline='').write(reps); r,e=score(f)
            return name+' '+(e if e else 'size=%d nd=%d%s'%(r[0],r[1],' EXACT' if r[2] else ''))
        k=bt.index('void HordeMeleeAmoeba::updateMeleeAttack'); pre,t=bt[:k],bt[k:]
        for a,b in reps:
            if a.startswith('PRE:'):
                a=a[4:]
                if a not in pre: return name+' MISSING '+a[:60]
                pre=pre.replace(a,b,1); continue
            if a not in t: return name+' MISSING '+a[:60]
            t=t.replace(a,b,1)
        t=pre+t
        f=D+name+'.cpp'
        open(f,'w',newline='').write(t)
        r,e=score(f)
        return name+' '+(e if e else 'size=%d nd=%d%s'%(r[0],r[1],' EXACT' if r[2] else ''))
    with ThreadPoolExecutor(workers) as ex:
        for r in ex.map(one, variants.items()): print(r, flush=True)
if __name__=='__main__':
    if sys.argv[1]=='show': pass
    elif sys.argv[1]=='one':
        print(score(sys.argv[2]))
    else:
        spec=importlib.util.spec_from_file_location('s',sys.argv[1]); m=importlib.util.module_from_spec(spec); spec.loader.exec_module(m)
        run_spec(m.BASE,m.V)
def show(path):
    src=Path(path).resolve(); obj=src.with_suffix('.obj')
    ok,text,code=build.try_compile_source(src,obj)
    if not ok: print(text); return
    row=dict(ROW); row['source']=src.relative_to(build.ROOT).as_posix()
    p=build.compile_function(row,SM,obj)
    def d2(b): return ['%x %s %s'%(i.address,i.mnemonic,'X' if i.mnemonic.startswith('j') or i.mnemonic=='call' else i.op_str) for i in md.disasm(bytes(b),0)]
    T,C=d2(p['target']),d2(p['bytes'])
    st=lambda L:[l.split(' ',1)[1] for l in L]
    sm=difflib.SequenceMatcher(None,st(T),st(C),autojunk=False)
    for tag,i1,i2,j1,j2 in sm.get_opcodes():
        if tag=='equal': continue
        print('----',tag)
        for l in T[i1:i2]: print('  R',l)
        for l in C[j1:j2]: print('  O',l)
if __name__=='__main__' and sys.argv[1]=='show': show(sys.argv[2])
