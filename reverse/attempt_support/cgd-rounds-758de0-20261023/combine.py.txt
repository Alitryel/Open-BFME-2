from pathlib import Path
from itertools import product
import json
W=Path(__file__).resolve().parent
D=W/'combine';D.mkdir(exist_ok=True);rows=[]
# Recombine the first-response register-reservation winner with the zero-count
# intrinsic boundary that improved both call regions. These remain diagnostics.
for idx,(dummy,form,sites,cc,uniform) in enumerate(product([0,2,3],[3,5,7],range(64),range(2),range(2))):
 s=(W/'reservation'/('v%04d.cpp'%(dummy*64+16))).read_text()
 forms={3:'memcpy(p,q,0)',5:'memset(p,(int)(unsigned)q,0)',7:'memset(memcpy(p,q,0),0,0)'}
 decl='extern "C" void *__cdecl memcpy(void *,const void *,unsigned);\nextern "C" void *__cdecl memset(void *,int,unsigned);\n#pragma intrinsic(memcpy,memset)\n'
 decl+='__forceinline const ContactVector &'+('__fastcall ' if cc else '')+'zeroAddress(const ContactVector &a,const ContactVector &b){void *p=(void *)&a;void *q=(void *)&b;return *(const ContactVector *)('+forms[form]+');}\n'
 s=s.replace('class BfmeThingCGD {',decl+'class BfmeThingCGD {')
 if uniform:s=s.replace('altPrepare(node->first->source,node->point,ContactVector(node->normal).negate())','node->first->source->prepare(node->point,ContactVector(node->normal).negate())')
 h,t=s.split('void BfmeThingCGD::bfmeTwoCGD() {',1);start=0
 for i,n in enumerate(['node->normal','node->normal','ContactVector(node->normal).negate()']):
  old='node->point,'+n;at=t.find(old,start);assert at>=0,(idx,i)
  p='zeroAddress(node->point,node->normal)' if sites>>(2*i)&1 else 'node->point'
  normal='zeroAddress('+n+',node->point)' if sites>>(2*i+1)&1 else n
  new=p+','+normal;t=t[:at]+new+t[at+len(old):];start=at+len(new)
 (D/('v%04d.cpp'%idx)).write_text(h+'void BfmeThingCGD::bfmeTwoCGD() {'+t)
 rows.append(dict(name='v%04d'%idx,dummy=dummy,form=form,sites=sites,fastcall=cc,uniform=uniform))
(W/'combine-models.json').write_text(json.dumps(rows,indent=1));print(len(rows))
