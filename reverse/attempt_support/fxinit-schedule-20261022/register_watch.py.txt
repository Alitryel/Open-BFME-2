"""Read-only hardware watch of an operand's physical-register binding."""
import sys,json
from pathlib import Path
sys.path.insert(0,str(Path(__file__).parent))
from wdbg import Dbg
W=Path('build/fxinit-seat22');tag=sys.argv[2]
d=Dbg(sys.argv[1],str(W/(tag+'.obj')),str(W/(tag+'.debug.log')))
address=int(sys.argv[3],16);events=[];on=False
try:
 d.prompt();d.prompt('set $CanDeferOnBPByAddr = 1');d.prompt('break *0x10760539')
 for k in range(1000):
  o=d.prompt('cont',60)
  if 'terminated' in o or not o.strip():break
  ip=d.reg('eip')
  if ip==0x10760539:
   p=d.reg('eax');f=d.words(p,1)[0];fn=d.string(d.words(f+4,1)[0])
   if on:break
   if '?Init@FXShaderGeometry' in fn:
    on=True;print('Init entered',flush=True)
    d.prompt('break *0x10783acd')
    d.prompt('break *0x107b5338')
   continue
  if on and ip==0x10783acd:
   print(d.prompt('watch *(int *)0x%x'%address),flush=True)
   continue
  if on and ip==0x107b5338:break
  if on:
   value=d.words(address,1)
   event=dict(ip=hex(ip),value=list(map(hex,value)),output=o,backtrace=d.prompt('bt'))
   events.append(event);print('watch',len(events),hex(ip),event['value'],flush=True)
finally:
 (W/(tag+'.watch.json')).write_text(json.dumps(events,indent=2))
 if d.p.poll() is None:
  try:d.prompt('quit',10)
  except Exception:d.p.kill()
 d.log.close()
