from pathlib import Path
import sys,json,struct,collections
sys.path.insert(0,'tools');import build,eh_verify
folder=Path(__file__).resolve().parent
src=Path(sys.argv[1]).resolve() if len(sys.argv)>1 else folder/'mix-three-d0-ttemporary-adefault.cpp'
name=(folder/'bank.cpp').read_text().splitlines()[0][3:]
obj=eh_verify.Obj(build.obj_path(src));img=eh_verify.Image();symbol=obj.by_name[name]
sections=obj.eh_sections(symbol['section']);indices={s['index'] for s in sections}
places,conflicts=eh_verify.section_placements(img,obj,dict(symbol,retail_rva=0xdf20),sections)
mapping=build.load_symbol_map();calls=[]
for section in sections:
 at=places[section['index']]
 for off,index,kind in section['relocs']:
  target=obj.symbols[index]
  if kind!=eh_verify.REL32 or target['section'] in indices:continue
  site=at+off;addend=struct.unpack_from('<i',section['data'],off)[0]
  rva=(site+4+struct.unpack_from('<i',img.bytes,site)[0]-addend)&0xffffffff
  candidates=mapping.get(target['name'],[])
  calls.append({'site':hex(site),'name':target['name'],'retail_symbol_rva':hex(rva),'coff_addend':addend,'retail_opcode':hex(img.bytes[site-1]),'candidate_count':len(candidates),'admitted_related_candidates':[hex(x) for x in candidates if x in {0x7150,0xb3c0,0x142d70,0x629182}],'agrees':rva in candidates})
result={'source':str(src.relative_to(Path.cwd())),'conflicts':conflicts,'calls':calls}
(src.with_suffix('.eh-calls.json') if len(sys.argv)>1 else folder/'eh-calls.json').write_text(json.dumps(result,indent=2))
print('external EH calls',len(calls),'agrees',sum(x['agrees'] for x in calls))
for (n,r,a),count in collections.Counter((x['name'],x['retail_symbol_rva'],x['agrees']) for x in calls).items(): print(count,a,r,n)
