from search import *
import itertools

def source(flags):
 hierarchy,dtor,preview,slot,game=flags
 head=HEAD
 fields='\tunsigned char m_pad000[0x04];\n\tAptCommandMapAdder m_commandMaps; // +0x04\n\tAptExternHandlerAdder m_externHandlers; // +0x10\n\tunsigned char m_pad01c[0x58 - 0x1C];\n'
 poly='virtual ~AptMpGameSetup()'+(' {}' if dtor else ';')
 if hierarchy == 0:
  head=head.replace('class AptMpGameSetup\n{\npublic:', 'class AptMpGameSetup\n{\npublic:\n'+poly)
  head=head.replace('\tunsigned char m_pad000[0x04];\n','')
 else:
  base='class Rva005248D0 { public: virtual ~Rva005248D0(); protected: '
  if hierarchy==1:
   base+='char m_basePad[0x58-4]; } ;\n'
  elif hierarchy==2:
   base+=fields.replace('\tunsigned char m_pad000[0x04];\n','')+'};\n'
  else:
   base='class AptCallbackBase { public: virtual ~AptCallbackBase(); protected: AptCommandMapAdder m_commandMaps; AptExternHandlerAdder m_externHandlers; };\nclass Rva005248D0 : public AptCallbackBase { public: virtual ~Rva005248D0(); protected: char m_basePad[0x58-0x1c]; };\n'
  head=head.replace('class AptMpGameSetup\n{\npublic:',base+'class AptMpGameSetup : public Rva005248D0\n{\npublic:\n'+poly)
  assert fields in head
  head=head.replace(fields,'')
 if preview:
  head=head.replace('unsigned char m_pad00[0x1C];','virtual ~Rva0057E3DB(); unsigned char m_pad04[0x1C-4];')
 if slot:
  head=head.replace('unsigned char m_pad00[0x04];','virtual void reset();')
 if game:
  head=head.replace('virtual void v00();','virtual ~GameInfo();',1)
 return head+MARK+BODY

if __name__=='__main__':
 run([('hierarchy-%03d'%n,source(f)) for n,f in enumerate(itertools.product(range(4),range(2),range(2),range(2),range(2)))], 'hierarchy')
