"""Read-only current-camera scheduling trace; stock compiler decisions."""
import sys,json,re,collections
from pathlib import Path
sys.path.insert(0,str(Path(__file__).resolve().parent))
from wdbg import Dbg
src,obj=sys.argv[1:3];d=Dbg(src,obj,obj+'.debug.log')
d.prompt();d.prompt('set $CanDeferOnBPByAddr = 1');d.prompt('break *0x10760539')
on=False;bps=[];graphs=[];cycles=[];gn=0
while True:
    o=d.prompt('cont',60)
    if 'terminated' in o or not o.strip():break
    ip=d.reg('eip')
    if ip==0x10760539:
        p=d.reg('eax');f=d.words(p,1)[0];fn=d.string(d.words(f+4,1)[0])
        if on:
            on=False
            for n in bps:d.prompt('disable %d'%n)
        if '?rva00102F83@W3DCamTransform' in fn:
            on=True;print('ON',fn,flush=True)
            for a in [0x107b5338,0x107b4f08]:
                z=d.prompt('break *0x%x'%a);bps.append(int(re.search(r'Breakpoint (\d+)',z)[1]))
        continue
    if not on:continue
    if ip==0x107b5338:
        g=d.reg('eax');gw=d.words(g,6);todo=[gw[0]];nodes={};tuples={};operands={};gn+=1
        while todo:
            p=todo.pop()
            if not p or p in nodes:continue
            w=d.words(p,16);nodes[p]=w
            if len(w)<16:continue
            t=w[7]
            if t:
                tw=d.words(t,16);tuples[t]=tw
                if len(tw)==16:
                    for k in [10,11]:
                        a=tw[k];seen=set()
                        while 0x10000<a<0x10000000 and a not in seen and len(seen)<8:
                            seen.add(a);aw=d.words(a,16);operands[a]=aw
                            if len(aw)!=16:break
                            a=aw[0]
            edge=w[3];seen=set()
            while edge and edge not in seen:
                seen.add(edge);ew=d.words(edge,4)
                if len(ew)<4:break
                todo.append(ew[3]);edge=ew[0]
        graphs.append(dict(number=gn,graph=g,words=gw,nodes=nodes,tuples=tuples,operands=operands))
        print('graph',gn,'nodes',len(nodes),flush=True)
    if ip==0x107b4f08:
        if len(graphs[-1]['nodes'])<35:continue
        ptr=d.words(0x108a8044,1)[0];ready=[];seen=set()
        while ptr and ptr not in seen and len(seen)<150:
            seen.add(ptr);w=d.words(ptr,16);ready.append([ptr,w]);ptr=w[4] if len(w)>4 else 0
        cycle=d.words(d.reg('esp')+4,1)[0]
        cycles.append(dict(graph=gn,cycle=cycle,ready=ready))
Path(obj+'.graphs.json').write_text(json.dumps(dict(graphs=graphs,cycles=cycles)))
d.log.close();print('DONE graphs',len(graphs),'cycles',len(cycles),flush=True)
