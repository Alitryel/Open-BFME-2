from pathlib import Path
import json
W=Path(__file__).resolve().parent;base=(W/'bank.cpp').read_text();out=W/'checks';out.mkdir(exist_ok=True);meta={}
head,body=base.split('void BfmeThingCGD::bfmeTwoCGD() {')
head+='''
__forceinline ContactData *readData(ContactData *const volatile &p) { return p; }
__forceinline ContactSource *readSource(ContactSource *const volatile &p) { return p; }
__forceinline unsigned readState(const volatile unsigned &p) { return p; }
void BfmeThingCGD::bfmeTwoCGD() {
'''
old='if(node->first->source==0 || node->second->source==0) continue;'
a,b,c=body.split(old)
for control in range(2):
 for bits in range(1024):
  checks=[]
  for site in range(2):
   terms=[]
   for side_i,side in enumerate(['first','second']):
    p='node->'+side
    if bits>>(site*4+side_i*2)&1:p='readData('+p+')'
    p+='->source'
    if bits>>(site*4+side_i*2+1)&1:p='readSource('+p+')'
    terms.append(p+'==0')
   checks.append('if('+' || '.join(terms)+') continue;')
  s=head+a+checks[0]+b+checks[1]+c
  if bits>>8&1:s=s.replace('if(node->state==0)','if(readState(node->state)==0)')
  if bits>>9&1:s=s.replace('if(node->state==2)','if(readState(node->state)==2)')
  if control:s=s.replace('altPrepare(node->first->source,node->point,ContactVector(node->normal).negate())','node->first->source->prepare(node->point,ContactVector(node->normal).negate())')
  name='v%04d'%len(meta);(out/(name+'.cpp')).write_text(s);meta[name]=dict(control=control,bits=bits)
(W/'checks-meta.json').write_text(json.dumps(meta,indent=1));print('generated',len(meta))
