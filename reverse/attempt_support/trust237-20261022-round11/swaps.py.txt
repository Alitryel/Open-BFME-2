from pathlib import Path
import sys,itertools,json
ROOT=Path.cwd();OUT=ROOT/'build/trust237-round11'
sys.path.insert(0,str(ROOT/'build/trust237-round9'));import frontends as f
f.OUT=OUT
original_resolve=f.resolve_bytes
def with_helper_map(raw,rel,target,rva,size,extra):
    extra=dict(extra);extra['__allmul']=[0x629580]
    return original_resolve(raw,rel,target,rva,size,extra)
f.resolve_bytes=with_helper_map
bank=(ROOT/'reverse/attempts/0x00237d3d.cpp').read_text()
prototypes='''#include <stdlib.h>
#pragma intrinsic(_byteswap_ushort,_byteswap_ulong,_byteswap_uint64,_rotl64,_rotr64)
'''
# Each operation preserves the complete 32-bit unsigned u. Intrinsic lowering
# is distinct from the manual integer expressions tested in round6.
families=[
 ('swap16_pair','u=(u&0xffff0000u)|_byteswap_ushort(_byteswap_ushort((unsigned short)u));'),
 ('swap32_pair','u=_byteswap_ulong(_byteswap_ulong(u));'),
 ('swap64_pair','unsigned __int64 q=((unsigned __int64)u<<32)|u;q=_byteswap_uint64(_byteswap_uint64(q));u=(unsigned long)q;'),
 ('rotate_halves_pair','unsigned __int64 q=u;q=_rotl64(_rotl64(q,32),32);u=(unsigned long)q;'),
 ('rotate_equal_halves','unsigned __int64 q=((unsigned __int64)u<<32)|u;q=_rotl64(q,32);u=(unsigned long)q;'),
 ('rotate_inverse17','unsigned __int64 q=u;q=_rotr64(_rotl64(q,17),17);u=(unsigned long)q;'),
 ('swap16_equal_bytes','unsigned short w=(unsigned short)((u&255u)*0x101u);w=_byteswap_ushort(w);u=(u&0xffffff00u)|(w&255u);'),
 ('swap64_equal_bytes','unsigned __int64 q=(u&255u)*0x0101010101010101ui64;q=_byteswap_uint64(q);u=(u&0xffffff00u)|((unsigned long)q&255u);'),
]
old='long status=WinVerifyTrust(0,&action,&data); unsigned long u=(unsigned long)status; LocalFree(argv); int r=0; if(((u&1u)<=1u)) goto yes; return r; yes:return r+1;'
assert old in bank
jobs=[]
for (kind,stmt),count,pos,tail,helper,cpu,opt in itertools.product(families,[1,16],['before','after'],['bank','discard'],[False,True],[6,7],['O1','O2']):
    declaration=''
    if helper:
        declaration='__forceinline unsigned long transformStatus(unsigned long u){'+stmt+'return u;}\n'
        body='u=transformStatus(u);'*count
    else:body=('{'+stmt+'}')*count
    finish='int r=0;if((u&1u)<=1u)goto yes;return r;yes:return r+1;' if tail=='bank' else 'return 1;'
    text=bank.replace('/O1 /G6',f'/{opt} /G{cpu}')
    text=text.replace('int Rva00237D3DCheck() {',prototypes+declaration+'int Rva00237D3DCheck() {')
    code='long status=WinVerifyTrust(0,&action,&data);unsigned long u=(unsigned long)status;'
    code+=(body+'LocalFree(argv);' if pos=='before' else 'LocalFree(argv);'+body)+finish
    text=text.replace(old,code)
    jobs.append((f's{len(jobs):04}',text,dict(kind=kind,count=count,pos=pos,tail=tail,helper=helper,cpu=cpu,opt=opt)))
f.execute('swaps',jobs)
