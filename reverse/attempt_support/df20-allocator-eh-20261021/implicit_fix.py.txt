from pathlib import Path
import sys,json,concurrent.futures
sys.path.insert(0,str(Path(__file__).parent));import trials
trials.scratch=Path(__file__).resolve().parent
trials.symbol=(trials.scratch/'bank.cpp').read_text().splitlines()[0][3:]
r=json.loads((trials.scratch/'implicit_proxy.json').read_text());specs=[]
dtor='template <> _String_base<wchar_t, allocator<wchar_t> >::~_String_base();'
for x in r:
 if x['exit']==0:continue
 s=(trials.scratch/(x['label']+'.cpp')).read_text().replace(dtor,'')
 s=s.replace('#include <locale>','namespace _STL { '+dtor+' }\n#include <locale>',1)
 specs.append((x['label']+'-fixed',s))
with concurrent.futures.ThreadPoolExecutor(max_workers=8) as pool:fixed=list(pool.map(trials.run,specs))
(trials.scratch/'implicit_proxy_fixed.json').write_text(json.dumps(fixed,indent=2))
