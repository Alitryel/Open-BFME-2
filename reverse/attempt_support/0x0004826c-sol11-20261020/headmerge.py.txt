# earlier tests of an accessor alone. Conversion wrappers have no dummy reads,

 ('', 'p'),
 ('static __forceinline Drawable *copyDraw(Drawable *const &v) {return v;}\n','copyDraw(p)'),
 ('static __forceinline Drawable *copyDraw(Drawable *v) {return v;}\n','copyDraw(p)'),

]
