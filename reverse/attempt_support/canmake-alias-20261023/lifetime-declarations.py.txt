"""A full 2^10 declaration/lifetime factorial in the new two-byte-residue body."""
from pathlib import Path
import concurrent.futures,itertools,json,re,runpy
OUT=Path('build/canmake-alias-20261023/lifetime-declarations').resolve();OUT.mkdir(parents=True,exist_ok=True)
r=runpy.run_path('reverse/attempt_support/canmake-00596e15-20261022/replay.py.txt');r['run'].__globals__['OUT']=OUT
base=runpy.run_path('reverse/attempt_support/canmake-alias-20261023/context.py.txt')['bank_text']();prefix,body=base.rsplit('int AIBuildableWallSegment::canMake(Player *player)',1)
tests=[]
for choices in itertools.product(range(2),repeat=10):
 scope,init,typ,padding,ref,node,failed,money,receiver,condition=choices
 p=prefix;b=body
 if scope:b=b.replace('MakeCodeLife selected;','').replace('return AIBuildableStructure::canMake(player);','return AIBuildableStructure::canMake(player);\nMakeCodeLife selected;')
 if init:p=p.replace('MakeCodeLife():code(0)','MakeCodeLife():code(9)')
 if typ:p=p.replace('{ int code;','{ unsigned int code;')
 if padding:p=p.replace('{ int code;','{ int pad; int code;').replace('{ unsigned int code;','{ int pad; unsigned int code;')
 top=''
 if ref:
  b=b.replace('MakeCodeLife selected;','MakeCodeLife selected;\nconst '+('unsigned int'if typ else 'int')+'&picked=selected.code;').replace('return selected.code;','return picked;')
 if node:top+='Object *startNode;\n';b=b.replace('Object *startNode =','startNode =')
 if failed:top+='bool failed;\n';b=b.replace('bool failed =','failed =')
 if money:top+='unsigned int money;\n';b=b.replace('unsigned int money =','money =')
 if receiver:
  top+='AIBuildableWallSegment *const &self=this;\n'
  for member in ['m_startNodeID','getWorldPosition()','m_totalCost']:
   b=re.sub(r'(?<![\w.>])'+re.escape(member),lambda m:'self->'+m[0],b)
 if condition:
  b=b.replace('failed = !((BuildAssistant *)TheBuildAssistant)->quoteWall(','failed = ((BuildAssistant *)TheBuildAssistant)->quoteWall(').replace('if (failed)','if (!failed)')
 b=b.replace('{','{\n'+top,1)
 tests.append(('decl-'+''.join(map(str,choices)),p+'\nint AIBuildableWallSegment::canMake(Player *player)'+b))
results=[]
with concurrent.futures.ThreadPoolExecutor(max_workers=8)as pool:
 for i,d in enumerate(pool.map(r['run'],tests),1):
  results.append(d)
  if i%128==0 or d.get('exact_candidate'):print(json.dumps(dict(done=i,total=len(tests),best=sorted(results,key=lambda x:(x.get('nd',999),-x.get('score',0)))[:2])),flush=True)
(OUT/'results.json').write_text(json.dumps(results,indent=2)+'\n')
print(json.dumps(dict(trials=len(results),errors=sum('error'in x for x in results),unresolved=sum(bool(x.get('unresolved'))for x in results),exact=sum(x.get('exact_candidate',False)for x in results),bodies=len({x.get('masked_code_sha256')for x in results}))),flush=True)
