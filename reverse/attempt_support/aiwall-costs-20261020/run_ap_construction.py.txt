from pathlib import Path
import sys,json,itertools
ROOT=Path(__file__).resolve().parents[2]
sys.path.insert(0,str(ROOT/'tools'))
from variant_search import Scorer
out=Path(__file__).parent/'ap-construction'
s=Scorer('?distSqToSegment@@YAMPBVCoord2D@@00@Z',0x4eacf5,197,out)
base=(Path(__file__).parent/'partial-coords/c18.cpp').read_text()
old='ap.x = pt->x - lineStart->x;\n\tap.y = pt->y - lineStart->y;'
forms=[
 'ap = *pt; ap.x -= lineStart->x; ap.y -= lineStart->y;',
 'ap = *pt; ap.y -= lineStart->y; ap.x -= lineStart->x;',
 'ap.x=pt->x; ap.y=pt->y; ap.x-=lineStart->x; ap.y-=lineStart->y;',
 'ap.y=pt->y; ap.x=pt->x; ap.x-=lineStart->x; ap.y-=lineStart->y;',
 'ap.x=pt->x; ap.x-=lineStart->x; ap.y=pt->y; ap.y-=lineStart->y;',
 'ap.y=pt->y; ap.y-=lineStart->y; ap.x=pt->x; ap.x-=lineStart->x;',
 'ap.x=-lineStart->x; ap.y=-lineStart->y; ap.x+=pt->x; ap.y+=pt->y;',
 'ap.x=pt->x-lineStart->x; ap.y=pt->y-lineStart->y;',
]
rows=[]
for f,dxform,loc in itertools.product(range(len(forms)),range(4),range(2)):
 text=base.replace(old,forms[f])
 if dxform==1:
  text=text.replace('float dx = lineEnd->x;','const float dxEnd = lineEnd->x;').replace('float dy = lineEnd->y;','const float dyEnd = lineEnd->y;').replace('dx -= lineStart->x;','float dx = dxEnd - lineStart->x;').replace('dy -= lineStart->y;','float dy = dyEnd - lineStart->y;')
 elif dxform==2:
  text=text.replace('float dx = lineEnd->x;','float dx;').replace('float dy = lineEnd->y;','float dy;').replace('dx -= lineStart->x;','dx = lineEnd->x-lineStart->x;').replace('dy -= lineStart->y;','dy = lineEnd->y-lineStart->y;')
 elif dxform==3:
  text=text.replace('float dx = lineEnd->x;','float dx = (lineEnd->x);').replace('float dy = lineEnd->y;','float dy = (lineEnd->y);')
 if loc:text=text.replace('\tCoord2D ap;\n','').replace('\t'+forms[f],'\tCoord2D ap;\n\t'+forms[f])
 r=s(text);got=r.get('compiled',b'');target=r.get('target',b'')
 row={'forms':[f,dxform,loc],'file':f'c{s.n}.cpp','size':len(got),'diff':sum(a!=b for a,b in zip(got,target))+abs(len(got)-len(target)),'exact':r['exact'],'fitness':r['fitness'],'class':r['class']}
 rows.append(row)
 if s.n==1 or r['exact'] or row['diff']<12:print(json.dumps(row),flush=True)
 if r['exact']:(out/'win.cpp').write_bytes(text.encode());break
(out/'results.json').write_text(json.dumps(rows,indent=2))
print('completed',len(rows),'best',min(rows,key=lambda r:(not r['exact'],-r['fitness'])),flush=True)
