from pathlib import Path
import itertools

W=Path(__file__).resolve().parent
BASE=(W/'base.cpp').read_text()

def source(bits,kind='_ReadWriteBarrier',uniform=False):
    s=BASE
    if uniform:
        s=s.replace('altPrepare(node->first->source,node->point,ContactVector(node->normal).negate())',
                    'node->first->source->prepare(node->point,ContactVector(node->normal).negate())')
    decl=('extern "C" void '+kind+'(void);\n#pragma intrinsic('+kind+')\n')
    s=s.replace('struct ContactVector {',decl+'struct ContactVector {')
    helpers='\n__forceinline const ContactVector &bv(const ContactVector &v) { '+kind+'(); return v; }\n'
    helpers+='__forceinline ContactSource *bs(ContactSource *s) { '+kind+'(); return s; }\n'
    helpers+='__forceinline void *bi(void *p) { '+kind+'(); return p; }\n'
    s=s.replace('struct ContactData {',helpers+'struct ContactData {')
    def vec(x,on):return 'bv('+x+')' if on else x
    prepare='node->second->source->prepare('+vec('node->point',bits[0])+','+vec('node->normal',bits[1])+')'
    if bits[2]:prepare='bi('+prepare+')'
    s=s.replace('node->second->source->prepare(node->point,node->normal)',prepare)
    bridge=('bs(s->source)' if bits[3] else 's->source')+'->prepare('+vec('p',bits[4])+','+vec('n',bits[5])+')'
    if bits[6]:bridge='bi('+bridge+')'
    s=s.replace('return s->source->prepare(p,n);','return '+bridge+';')
    if bits[7]:s=s.replace('if(node->state==2) continue;',kind+'();\n  if(node->state==2) continue;')
    if bits[8]:s=s.replace('if(node->first->source==0 || node->second->source==0) continue;\n  node->second',kind+'();\n  if(node->first->source==0 || node->second->source==0) continue;\n  node->second')
    return s

def emit(family,choices,kind='_ReadWriteBarrier',uniform=False):
    d=W/family;d.mkdir(exist_ok=True)
    for n,bits in enumerate(choices):
        (d/f'v{n:04}.cpp').write_text(source(bits,kind,uniform))

if __name__=='__main__':
    singles=[(0,)*9]+[tuple(int(i==j) for i in range(9)) for j in range(9)]
    emit('readwrite',itertools.product(range(2),repeat=9))
    emit('read',itertools.product(range(2),repeat=9),'_ReadBarrier')
    emit('write',itertools.product(range(2),repeat=9),'_WriteBarrier')
    emit('uniform',itertools.product(range(2),repeat=9),uniform=True)
