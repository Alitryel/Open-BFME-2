import sys,subprocess,re
from pathlib import Path
sys.path.insert(0,'/work/Open-BFME-2/tools')
import build
src=Path(sys.argv[1]).resolve(); stem=str(src)+'.od'
cmd,env=build.compiler_command(src,Path(stem+'.obj'))
cmd=[c for c in cmd if c not in ('-O2','-O1','-GX-','-EHsc-','-G7')]
cmd[-1:-1]=['-Od','-GX','-FAs','-Fa'+str(Path(stem+'.asm').relative_to(build.ROOT))]
r=subprocess.run(cmd,cwd=build.ROOT,env=env,capture_output=True,text=True,errors='replace')
if r.returncode: print((r.stdout+r.stderr)[-1500:]); sys.exit(1)
txt=open(stem+'.asm',errors='replace').read()
i=txt.find('?updateMeleeAttack@HordeMeleeAmoeba@@UAEXPAVObject@@@Z PROC')
j=txt.rfind('_TEXT\tSEGMENT',0,i)
seg=txt[j:i]
loc=[]
for m in re.finditer(r'^(_?[A-Za-z_]\w*)\$(\d*)\s*=\s*(-?\d+)',seg,re.M):
    loc.append((int(m.group(3)),m.group(1),m.group(2)))
for off,n,s in sorted(loc,reverse=True): print('%5d %-4x %s$%s'%(off,(-off)&0xffff if off<0 else off,n,s))
