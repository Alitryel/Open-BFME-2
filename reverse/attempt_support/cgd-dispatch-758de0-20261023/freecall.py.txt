from pathlib import Path
from itertools import product
W=Path(__file__).resolve().parent
bank=(W/'bank.cpp').read_text();uniform=(W/'uniform-control.cpp').read_text()
D=W/'freecall';D.mkdir(exist_ok=True)
for idx,(opts,pack,control) in enumerate(product(product(range(5),repeat=4),range(4),range(2))):
 s=uniform if control else bank
 defs='''struct ContactSource;
struct RawPair { const ContactVector *p,*n; RawPair(const ContactVector &x,const ContactVector &y):p(&x),n(&y){} };
struct RawInfo {void *p;RawInfo(void *x):p(x){} };
typedef void *(__fastcall *RawPrepare)(ContactSource *,RawPair);
typedef bool (__fastcall *RawTest)(ContactSource *,RawInfo);
typedef void (__fastcall *RawRespond)(ContactSource *,RawInfo);
struct RawDispatch { void *pad0[3]; RawPrepare prepare;void *pad1[4];RawTest test;RawRespond respond;};
'''
 if pack&1:
  defs=defs.replace(' RawPair(const ContactVector &x,const ContactVector &y):p(&x),n(&y){}','')
  defs+='__forceinline RawPair pair(const ContactVector &p,const ContactVector &n) {RawPair a={&p,&n};return a;}\n'
 if pack&2:
  defs=defs.replace('RawInfo(void *x):p(x){}','')
  defs+='__forceinline RawInfo info(void *p) {RawInfo a={p};return a;}\n'
 inners=['node->second->source->prepare(node->point,node->normal)','node->second->source->prepare(node->point,node->normal)' if control else 'bridge(node->second,node->point,node->normal)']
 for j in range(2):
  outer='node->first->source';inner='node->second->source';method='test' if not j else 'respond'
  old=outer+'->'+method+'('+inners[j]+')';newinner=inners[j]
  for n in [2*j,2*j+1]:
   opt=opts[n]
   if not opt:continue
   meth='prepare' if n%2==0 else method
   tabtype='volatile RawDispatch *' if opt in [3,4] else 'RawDispatch *'
   readtype='RawDispatch *volatile *' if opt in [2,4] else 'RawDispatch **'
   tab='(('+tabtype+')*('+readtype+')s)'
   if meth=='prepare':
    pair=('pair' if pack&1 else 'RawPair')+'(p,n)'
    decl='void *raw'+str(n)+'(ContactSource *s,const ContactVector &p,const ContactVector &n)'
    body='return '+tab+'->prepare(s,'+pair+');'
    newinner='raw'+str(n)+'('+inner+',node->point,node->normal)'
   else:
    info=('info' if pack&2 else 'RawInfo')+'(v)'
    decl=('bool ' if meth=='test' else 'void ')+'raw'+str(n)+'(ContactSource *s,void *v)'
    body=('return ' if meth=='test' else '')+tab+'->'+meth+'(s,'+info+');'
   defs+='__forceinline '+decl+' {'+body+'}\n'
  new=('raw'+str(2*j+1)+'('+outer+','+newinner+')') if opts[2*j+1] else outer+'->'+method+'('+newinner+')'
  assert old in s,(idx,j)
  s=s.replace(old,new,1)
 a=s.index('struct ContactData {');s=s[:a]+defs+s[a:]
 (D/('v%04d.cpp'%idx)).write_text(s)
print('generated',idx+1)
