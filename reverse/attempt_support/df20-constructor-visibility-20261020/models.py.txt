from pathlib import Path
import sys,itertools,concurrent.futures,json,re
sys.path.insert(0,'build/df20-scalars-20261020');import trials
trials.scratch=Path.cwd()/'build/df20-scalars-20261020'
base=Path('build/df20-scalars-20261020/bank.cpp').read_text();trials.symbol=base.splitlines()[0][3:]
flags=['/EHsc','/EHs','/EHa','/GX','/EHsc /Og-','/EHsc /Oi-','/EHsc /Os','/EHsc /Oa','/EHsc /Ow']
specs=[]
for n,f in enumerate(flags):
 for c,opaque in itertools.product([0,1],repeat=2):
  s=base.replace('/EHsc /MD',f+' /MD',1)
  if c:s=s.replace('void __cdecl free(void *block);','extern "C" void __cdecl free(void *block);')
  if not opaque:s=s.replace('template <> _String_base<wchar_t, allocator<wchar_t> >::~_String_base();','')
  specs.append((f'model-{n}-c{c}-o{opaque}',s))
(trials.scratch/'models-flags.json').write_text(json.dumps(flags))
with concurrent.futures.ThreadPoolExecutor(max_workers=8) as p:r=list(p.map(trials.run,specs))
(trials.scratch/'models.json').write_text(json.dumps(r,indent=2))
