"""Test overlapping receiver and return local storage in the new EH context."""
from pathlib import Path
import concurrent.futures,itertools,json,re,runpy
OUT=Path('build/canmake-alias-20261023/receiver-union').resolve();OUT.mkdir(parents=True,exist_ok=True)
r=runpy.run_path('reverse/attempt_support/canmake-00596e15-20261022/replay.py.txt');r['run'].__globals__['OUT']=OUT
base=runpy.run_path('reverse/attempt_support/canmake-alias-20261023/context.py.txt')['bank_text']();prefix,body=base.rsplit('int AIBuildableWallSegment::canMake(Player *player)',1)
prefix=prefix[prefix.index('\n')+1:]
tests=[]
for bits,fallback,ctor,dtor in itertools.product(range(8),range(2),range(4),range(2)):
 b=body.replace('MakeCodeLife selected;','MakeCodeLife selected(this);')
 for i,member in enumerate(['m_startNodeID','getWorldPosition()','m_totalCost']):
  if bits&(1<<i):b=re.sub(r'(?<![\w.>])'+re.escape(member),lambda m:'selected.wall->'+m[0],b)
 if fallback:b=b.replace('return AIBuildableStructure::canMake(player);','return selected.wall->AIBuildableStructure::canMake(player);')
 constructors=['__forceinline MakeCodeLife(AIBuildableWallSegment*p):wall(p){}','__forceinline MakeCodeLife(AIBuildableWallSegment*p){wall=p;}','__forceinline MakeCodeLife(AIBuildableWallSegment*const &p):wall(p){}','__forceinline MakeCodeLife(AIBuildableWallSegment*p):wall(0){wall=p;}']
 helper='class AIBuildableWallSegment; struct MakeCodeLife { union{int code;AIBuildableWallSegment*wall;}; '+constructors[ctor]+' __forceinline ~MakeCodeLife(){'+('code=0;'if dtor else '')+'} __forceinline int get()const{return code;} };\n'
 tests.append((f'union-{bits}{fallback}{ctor}{dtor}',helper+prefix+'\nint AIBuildableWallSegment::canMake(Player *player)'+b))
results=[]
with concurrent.futures.ThreadPoolExecutor(max_workers=8)as pool:
 for i,d in enumerate(pool.map(r['run'],tests),1):
  results.append(d)
  if i%64==0 or d.get('exact_candidate'):print(json.dumps(dict(done=i,total=len(tests),best=sorted(results,key=lambda x:(x.get('nd',999),-x.get('score',0)))[:2])),flush=True)
(OUT/'results.json').write_text(json.dumps(results,indent=2)+'\n')
print(json.dumps(dict(trials=len(results),errors=sum('error'in x for x in results),unresolved=sum(bool(x.get('unresolved'))for x in results),exact=sum(x.get('exact_candidate',False)for x in results),bodies=len({x.get('masked_code_sha256')for x in results}))),flush=True)
