from search import *
import itertools
s=(W/'base.cpp').read_text();vs=[]
pre='''
struct TerrainCallable {
 float ground(float,float,Coord3D*);
 float layer(float,float,int,Coord3D*,bool);
 bool underwater(float,float,float*,float*,unsigned char*);
};
union GroundAddress { void *raw; float (TerrainCallable::*member)(float,float,Coord3D*); };
union LayerAddress { void *raw; float (TerrainCallable::*member)(float,float,int,Coord3D*,bool); };
union WaterAddress { void *raw; bool (TerrainCallable::*member)(float,float,float*,float*,unsigned char*); };
template<class A> __forceinline A makeAddress(void *p) {A a; a.raw=p;return a;}
'''
a=s.index('  case 1:');b=s.index('  case 4:',a);c=s.index('  case 6:');d=s.index('  case 7:',c)
for mode,site,byref,decl in itertools.product(range(4),range(3),range(2),range(4)):
 def region(blk):
  init=''
  for name,slot,ty in [('ground',6,'GroundAddress'),('layer',7,'LayerAddress'),('underwater',19,'WaterAddress')]:
   pat='((LocoTerrainView*)TheTerrainLogic)->'+name+'('
   if pat not in blk:continue
   recv='((TerrainCallable*)TheTerrainLogic)';vt='(*(void***)TheTerrainLogic)'
   nm=name+'Address'
   if mode==0:expr='makeAddress<'+ty+'>('+vt+'['+str(slot)+']).member'
   else:
    dec=ty+(' const &' if byref else ' ')+nm+'=makeAddress<'+ty+'>('+vt+'['+str(slot)+']);\n'
    if mode==1:init+=dec
    elif mode==2:blk=blk.replace('   requires=',dec+'   requires=',1)
    else:
     dec='void **'+('const &' if byref else '')+nm+'='+vt+';\n'
     blk=blk.replace('   requires=',dec+'   requires=',1)
    expr=nm+'.member' if mode<3 else 'makeAddress<'+ty+'>('+nm+'['+str(slot)+']).member'
   blk=blk.replace(pat,'('+recv+'->*('+expr+'))(')
  if init:blk=blk.replace('   float water;', '   float water;\n'+init) if 'float water;' in blk else blk.replace('   pos.z=',init+'   pos.z=',1)
  if decl:
   nm='terrainAlias';init='TerrainCallable *'+('const &' if decl==2 else '')+nm+'='+(' (TerrainCallable*const&)TheTerrainLogic' if decl==2 else '(TerrainCallable*)TheTerrainLogic')+';\n'
   blk=blk.replace('   requires=',init+'   requires=',1)
   blk=blk.replace('((TerrainCallable*)TheTerrainLogic)',nm)
   if decl==3:blk=blk.replace('(*(void***)TheTerrainLogic)','(*(void***)'+nm+')')
  return blk
 o=s
 if site in [0,2]:o=o[:c]+region(o[c:d])+o[d:]
 if site in [0,1]:o=o[:a]+region(o[a:b])+o[b:]
 o=o.replace('bool Locomotor::rva001E7ECA',pre+'bool Locomotor::rva001E7ECA')
 vs.append((f'dispatchsi_{mode}{site}{byref}{decl}',o))
run(vs,'dispatchsi96')
