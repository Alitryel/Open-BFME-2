# Test storage shared across the two paths instead of two candidate records.
def source(f):
 pred,kind,shared,guard,form=f
 ty=['bool','char','unsigned char','int'][kind]
 s=original.replace('bool locked',ty+' locked')
 decl=''
 if shared&1:s=s.replace(ty+' locked = m_rules.m_04 == 1;', 'locked = m_rules.m_04 == 1;');decl+=ty+' locked; '
 if shared&2:s=s.replace('bool teamEnable;', '');decl+='bool teamEnable; '
 if guard:
  # Retain the single original team-null guard while duplicating its interior.
  s=s[s.index('{')+1:s.rfind('}')]
 if form==0:region='if ('+PRED[pred]+') { '+s+' } else { '+s+' }'
 else:region='if ('+PRED[pred]+') goto pathA; {'+s+'} goto teamDone; pathA:; {'+s+'} teamDone:;'
 if guard:region='if (*team) {'+decl+region+'}'
 else:region=decl+region
 return HEAD+MARK+BODY[:a]+region+BODY[b:]
if __name__=='__main__':
 run([('sharedteam-%04d'%n,source(f)) for n,f in enumerate(itertools.product(range(8),range(4),range(4),range(2),range(2)))], 'shared_team')
