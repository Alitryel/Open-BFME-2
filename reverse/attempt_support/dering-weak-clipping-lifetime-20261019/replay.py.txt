from pathlib import Path
import difflib,json,struct,subprocess,sys
ROOT=Path(__file__).resolve().parents[2];OUT=Path(__file__).resolve().parent
sys.path.insert(0,str(ROOT/'tools'))
import build,capstone
# Copy to build/NAME/trials.py, then run phases clip, index, unsigned, shapes,
# coefficient, counter-lifetimes in that order. The seed predates this new bank.
seed=subprocess.check_output(['git','show','b235e6850fa490b134e90f2dffcef7ddd26ff294:reverse/attempts/0x001c22c0.cpp'],cwd=ROOT,text=True)
old='''\t\t\tif (value < 0)
\t\t\t\tvalue = 0;
\t\t\telse if (value > 255)
\t\t\t\tvalue = 255;
\t\t\tdestination[r * pitch + c] = (unsigned char)value;'''
assert old in seed
variants={}
for kind in ('unsigned char','int','unsigned int'):
    body='\t\t\t'+kind+' pixel;\n\t\t\tif (value < 0) pixel = 0;\n\t\t\telse if (value > 255) pixel = 255;\n\t\t\telse pixel = ('+kind+')value;\n\t\t\tdestination[r * pitch + c] = (unsigned char)pixel;'
    text=seed.replace(old,body)
    variants['clip-'+kind.replace(' ','-')]=text
    if kind=='unsigned char':variants['clip-byte-const-blend']=text.replace('int value =','const int value =')
phase=sys.argv[1] if len(sys.argv)>1 else 'clip'
if phase=='index':
    variants={}
    for label,base in [('bank',seed),('byte',(OUT/'clip-unsigned-char.cpp').read_text())]:
        start=base.index('\tconst unsigned char *vrow');end=base.index('\tconst unsigned char *hrow')
        segment=base[start:end]
        vertical=segment.replace('c = 0; c < 8; c++','c = 1; c <= 8; c++').replace('vrow[c]', 'vrow[c - 1]').replace('vrow[c - pitch]','vrow[c - pitch - 1]').replace('vweight[c]','vweight[c - 1]')
        text=base[:start]+vertical+base[end:]
        variants[label+'-vertical-one-based']=text
        start=base.index('\tconst unsigned char *hrow');end=base.index('\n\tfor (int r = 0;',start)
        segment=base[start:end]
        horizontal=segment.replace('c = 0; c < 9; c++','c = 1; c <= 9; c++').replace('hrow[c]', 'hrow[c - 1]').replace('hrow[c - 1] - hrow[c - 1]', 'hrow[c - 2] - hrow[c - 1]').replace('hrow[c - 1]) >', 'hrow[c - 2]) >').replace('hrow[c - 1]) :', 'hrow[c - 2]) :').replace('hweight[c]','hweight[c - 1]')
        # Rewrite the full difference explicitly to avoid a chained substitution changing its meaning.
        begin=horizontal.index('\t\t\tint difference =');finish=horizontal.index('\n',begin)
        horizontal=horizontal[:begin]+'\t\t\tint difference = (hrow[c - 1] - hrow[c - 2]) > 0 ? (hrow[c - 1] - hrow[c - 2]) : (hrow[c - 2] - hrow[c - 1]);'+horizontal[finish:]
        variants[label+'-horizontal-one-based']=base[:start]+horizontal+base[end:]
if phase=='unsigned':
    variants={}
    for label in ('clip-unsigned-char','clip-int','clip-unsigned-int'):
        text=(OUT/(label+'.cpp')).read_text()
        variants[label+'-unsigned-column']=text.replace('\tint c;','\tunsigned int c;')
if phase=='shapes':
    seed=(OUT/'clip-int.cpp').read_text();variants={}
    oldweights='\t\t\tint up = vertical[r * 8 + c];\n\t\t\tint down = vertical[r * 8 + c + 8];\n\t\t\tint left = horizontal[r * 9 + c];\n\t\t\tint right = horizontal[r * 9 + c + 1];'
    defs=dict(zip(('up','down','left','right'),oldweights.splitlines()))
    for order in [('right','down','up','left'),('down','right','up','left'),('up','down','right','left')]:
        variants['clip-weights-'+'-'.join(order)]=seed.replace(oldweights,'\n'.join(defs[k] for k in order))
    begin=seed.index('\tconst unsigned char *vrow');middle=seed.index('\tconst unsigned char *hrow');end=seed.index('\n\tfor (int r = 0;',middle)
    vloop=seed[begin:middle];hloop=seed[middle:end]
    variants['clip-phase-scopes']=seed[:begin]+'\t{\n'+vloop+'\t}\n\t{\n'+hloop+'\n\t}\n'+seed[end:]
    shared=hloop.replace('hrow','vrow').replace('hweight','vweight')
    variants['clip-shared-local-phase-scopes']=seed[:begin]+'\t{\n'+vloop+'\t}\n\t{\n'+shared+'\n\t}\n'+seed[end:]
    base=seed.replace('short vertical[72];','short vertical[9][8];').replace('short horizontal[72];','short horizontal[8][9];')
    base=base.replace('short *vweight = vertical;','short (*vweight)[8] = vertical;').replace('vweight[c]','(*vweight)[c]').replace('vweight += 8;','++vweight;')
    base=base.replace('short *hweight = horizontal;','short (*hweight)[9] = horizontal;').replace('hweight[c]','(*hweight)[c]').replace('hweight += 9;','++hweight;')
    base=base.replace('vertical[r * 8 + c + 8]','vertical[r + 1][c]').replace('vertical[r * 8 + c]','vertical[r][c]').replace('horizontal[r * 9 + c + 1]','horizontal[r][c + 1]').replace('horizontal[r * 9 + c]','horizontal[r][c]')
    variants['clip-dimensional-weights']=base
    variants['clip-dimensional-unsigned-column']=base.replace('\tint c;','\tunsigned int c;')
if phase=='counter-lifetimes':
    seed=(OUT/'clip-int.cpp').read_text();variants={}
    before,end=seed.split('\n\tfor (int r = 0;',1)
    for label,weights,outtype in [('local-unsigned-all',True,'unsigned int'),('local-unsigned-weights',True,'int'),('local-unsigned-output',False,'unsigned int')]:
        v=before.replace('\tint c;\n','').replace('for (c = 0; c < 8;', 'for ('+('unsigned int' if weights else 'int')+' c = 0; c < 8;').replace('for (c = 0; c < 9;', 'for ('+('unsigned int' if weights else 'int')+' c = 0; c < 9;')
        o=end.replace('for (c = 0; c < 8;', 'for ('+outtype+' c = 0; c < 8;')
        variants['clip-'+label]=v+'\n\tfor (int r = 0;'+o
    old='\t\t\tint pixel;\n\t\t\tif (value < 0) pixel = 0;\n\t\t\telse if (value > 255) pixel = 255;\n\t\t\telse pixel = (int)value;\n\t\t\tdestination[r * pitch + c] = (unsigned char)pixel;'
    variants['clip-expression-store']=seed.replace(old,'\t\t\tdestination[r * pitch + c] = (unsigned char)(value < 0 ? 0 : (value > 255 ? 255 : value));')
if phase=='coefficient':
    seed=(OUT/'clip-int.cpp').read_text();variants={}
    coefficient='128 - right - down - up - left'
    for label,expression in [('group-vertical','128 - right - (down + up) - left'),('group-all','128 - (right + down + up + left)'),('vertical-first','128 - down - up - right - left'),('source-pair','128 - up - down - left - right')]:
        variants['clip-coefficient-'+label]=seed.replace(coefficient,expression)
    start=seed.index('\t\t\tint up = vertical');end=seed.index('\t\t\tint value =',start)
    direct=seed[:start]+seed[end:]
    import re
    for name,expr in [('up','vertical[r * 8 + c]'),('down','vertical[r * 8 + c + 8]'),('left','horizontal[r * 9 + c]'),('right','horizontal[r * 9 + c + 1]')]:
        direct=re.sub(r'\b'+name+r'\b',expr,direct)
    variants['clip-direct-array-weights']=direct
    variants['clip-direct-array-unsigned']=direct.replace('\tint c;','\tunsigned int c;')
md=capstone.Cs(capstone.CS_ARCH_X86,capstone.CS_MODE_32)
target=build.read_target_bytes(0x1c22c0,1526);records=[]
for label,text in variants.items():
    source=OUT/(label+'.cpp');source.write_text(text)
    relative=source.relative_to(ROOT).as_posix()
    command=[sys.executable,'tools/explain_mismatch.py','_DeringBlockWeak_C','--rva','0x001C22C0','--size','1526','--source',relative,'--context','8']
    run=subprocess.run(command,cwd=ROOT,capture_output=True,text=True)
    (OUT/(label+'.txt')).write_text(run.stdout+run.stderr)
    if run.returncode:raise RuntimeError((label,run.stdout[-1600:],run.stderr[-1600:]))
    raw,relocs=build.read_object_symbol_bytes(build.obj_path(source),'_DeringBlockWeak_C',code_only=True)
    code=bytearray(raw)
    for offset,kind,name in relocs:
        assert (kind,name)==(6,'_Vp6FilterEdgeTagTable')
        struct.pack_into('<I',code,offset,struct.unpack_from('<I',code,offset)[0]+0xdb7a28)
    code=bytes(code)
    rec={'name':label,'size':len(code),'first':next((i for i,(a,b) in enumerate(zip(target,code)) if a!=b),None),'aligned':sum(m.size for m in difflib.SequenceMatcher(None,target,code,autojunk=False).get_matching_blocks()),'fixed_differences':sum(a!=b for a,b in zip(target,code))+abs(len(target)-len(code)),'exact':target==code}
    (OUT/(label+'.asm.txt')).write_text('\n'.join(f'{i.address:08x} {i.bytes.hex():20} {i.mnemonic} {i.op_str}' for i in md.disasm(code,0x5c22c0))+'\n')
    records.append(rec);print(rec,flush=True)
    (OUT/(phase+'-results.json')).write_text(json.dumps(records,indent=2)+'\n')
