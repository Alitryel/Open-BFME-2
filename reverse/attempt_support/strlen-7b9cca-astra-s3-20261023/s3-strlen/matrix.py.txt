import sys, json, subprocess, concurrent.futures, hashlib, itertools, time
from pathlib import Path
sys.path.insert(0,"tools")
import build
ROOT=Path.cwd()
OUT=ROOT/"build/s3-strlen"
target=build.read_target_bytes(0x7b9cca,24)
forms=[
("post","const char *p=s; while (*p++) {} return (int)(p-s-1);"),
("pre","const char *p=s; if (*p++) { do {} while (*p++); } return (int)(p-s-1);"),
("char","const char *p=s; char c=*p++; while(c) {c=*p++;} return (int)(p-s-1);"),
("goto","const char *p=s; char c=*p++; if(!c) goto done; again: c=*p++; if(c) goto again; done: return (int)(p-s-1);"),
("split","char c=*s; const char *p=s+1; if(c) do {c=*p++;}while(c); return (int)(p-s-1);"),
("assign","const char *p=s; char c; while((c=*p++)!=0) {} return (int)(p-s-1);"),
("two_reads","const char *p=s+1; if(*s) { while(*p) ++p; ++p; } return (int)(p-s-1);"),
("unsigned","const unsigned char *p=(const unsigned char*)s; unsigned char c=*p++; if(c) do {c=*p++;}while(c); return (int)(p-(const unsigned char*)s-1);"),
("signed","const signed char *p=(const signed char*)s; signed char c=*p++; if(c) do {c=*p++;}while(c); return (int)(p-(const signed char*)s-1);"),
("xor","const char *p=s; char c=*p++; while(c^0) c=*p++; return (int)(p-s-1);"),
("switch","const char *p=s; switch(*p++) {case 0: break; default: while(*p++) {} } return (int)(p-s-1);"),
("loopreturn","const char *p=s; char c=*p++; for (;;) {if(!c)return (int)(p-s-1); c=*p++;}"),
("volatile","const volatile char *p=s; char c=*p++; if(c) do {c=*p++;}while(c); return (int)(p-s-1);"),
("narrow","const char *p=s; unsigned c=(unsigned char)*p++; while(c) c=(unsigned char)*p++; return (int)(p-s-1);"),
("minus","const char *p=s; char c=*p++; if(c!=0) {do {c=*p++;}while(c!=0);} return (int)(p-1-s);"),
("intaddr","const char *p=s; char c=*p++; if(c) do {c=*p++;}while(c); return (int)((unsigned)p-(unsigned)s-1);"),
]
flags=[
"/O1 /G6", "/O2 /G6", "/O2 /G3", "/O2 /G4", "/O2 /G5", "/O2 /G7",
"/O1 /G3", "/O1 /G4", "/O1 /G5", "/O1 /G7",
"/Og /Ot /G3", "/Og /Ot /G4", "/Og /Os /G3", "/Og /Os /G4",
"/O2 /G3 /Ow", "/O2 /G4 /Ow", "/O2 /G3 /Oa", "/O2 /G4 /Oa",
]
def trial(arg):
 idx, (tag,body), flag, ext=arg
 src=OUT/("trial%04d.%s"%(idx,ext)); obj=src.with_suffix(".obj")
 src.write_text("// cl: "+flag+" /MD\nint rva007B9CCAStrLen(const char *s) { "+body+" }\n")
 command,env=build.compiler_command(src,obj)
 p=subprocess.run(command,cwd=ROOT,env=env,capture_output=True,text=True,timeout=90)
 rec=dict(index=idx,tag=tag,flags=flag,ext=ext,returncode=p.returncode,diagnostic=(p.stdout+p.stderr).strip())
 if p.returncode==0:
  symbols=build.defined_code_symbols(obj)
  sym=next(s for s in symbols if "rva007B9CCAStrLen" in s)
  buf,relocs=build.read_object_symbol_bytes(obj,sym,code_only=True)
  rec.update(size=len(buf),hex=buf.hex(),relocs=relocs,exact=buf==target,diff=sum(a!=b for a,b in itertools.zip_longest(buf,target)))
 return rec
jobs=[(i,form,flag,ext) for i,(form,flag,ext) in enumerate(itertools.product(forms,flags,["cpp","c"]))]
build.ensure_reference_current()
start=time.time()
with (OUT/"matrix.jsonl").open("w") as fh, concurrent.futures.ThreadPoolExecutor(max_workers=2) as pool:
 for i,rec in enumerate(pool.map(trial,jobs)):
  fh.write(json.dumps(rec)+"\n"); fh.flush()
  if rec.get("exact") or i%32==0: print(json.dumps(rec),flush=True)
print("DONE",len(jobs),"seconds",time.time()-start,flush=True)
