"""Symbolically run the world arm + merge through Translate; print each postMul/Translate sum's association.
Usage: assoc.py <bin>..."""
import re
import sys
import capstone

md = capstone.Cs(capstone.CS_ARCH_X86, capstone.CS_MODE_32)


def ins_list(b):
    return list(md.disasm(b, 0))


def mem(op):
    m = re.match(r"dword ptr \[(\w+)(?: ([+-]) (0x[0-9a-f]+|\d+))?\]", op)
    if not m:
        return None, None
    d = int(m.group(3), 0) if m.group(3) else 0
    if m.group(2) == '-':
        d = -d
    return m.group(1), d


def run(b):
    L = ins_list(b)
    idx = {i.address: n for n, i in enumerate(L)}
    # start: first load from [esi + 0x650] region (world arm) or first load of scale [esi+0x5cc]
    start = next(n for n, i in enumerate(L) if '0x5cc]' in i.op_str and i.mnemonic == 'movss')
    regs, stack, eaxoff = {}, {}, {}
    sums = []
    world_seen = False
    jumped = False
    n = start
    steps = 0
    while n < len(L) and steps < 600:
        i = L[n]; steps += 1
        mn = i.mnemonic
        ops = [o.strip() for o in i.op_str.split(",")]
        if mn == 'jmp' and not jumped and world_seen:
            tgt = int(ops[0], 16)
            n = idx[tgt]; jumped = True; continue
        if mn.startswith('j') or mn in ('call', 'ret'):
            if mn == 'call':
                break
            n += 1; continue

        def name(base, d):
            if base == 'esi':
                if 0x5d0 <= d < 0x600:
                    k = (d - 0x5d0) // 4; return ('b', k // 4, k % 4)
                if 0x650 <= d < 0x680:
                    k = (d - 0x650) // 4; return ('w', k // 4, k % 4)
                if d == 0x5cc:
                    return ('s',)
                return ('esi', d)
            if base == 'edi' and d in (0x4455c, 0x44560, 0x44564):
                return ('T', (d - 0x4455c) // 4)
            if base == 'ebp':
                return stack.get(d, ('ebp', d))
            if base in eaxoff:
                return name('esi', eaxoff[base] + d)
            return (base, d)

        def read(op):
            if op.startswith('xmm'):
                return regs.get(op, ('?', op))
            base, d = mem(op)
            if base is None:
                return ('?', op)
            v = name(base, d)
            if v and v[0] == 'w':
                pass
            return v
        if mn == 'lea':
            m = re.match(r"(\w+), \[esi \+ (0x[0-9a-f]+)\]", i.op_str)
            if m:
                eaxoff[m.group(1)] = int(m.group(2), 16)
            n += 1; continue
        if mn in ('movss', 'movaps'):
            if ops[0].startswith('xmm'):
                v = read(ops[1])
                if v and v[0] == 'w':
                    world_seen = True
                regs[ops[0]] = v
            else:
                base, d = mem(ops[0])
                if base == 'ebp':
                    stack[d] = regs.get(ops[1], ('?',))
            n += 1; continue
        if mn == 'mulss':
            x, y = regs.get(ops[0], ('?',)), read(ops[1])
            # scaled world element
            if x == ('s',) and y[0] == 'w':
                regs[ops[0]] = ('a', y[1], y[2])
            elif y == ('s',) and x[0] == 'w':
                regs[ops[0]] = ('a', x[1], x[2])
            else:
                regs[ops[0]] = ('*', x, y)
            n += 1; continue
        if mn == 'addss':
            v = ('+', regs.get(ops[0], ('?',)), read(ops[1]))
            regs[ops[0]] = v
            sums.append(v)
            n += 1; continue
        if mn == 'xorps':
            regs[ops[0]] = ('0',)
        n += 1
    return sums


def label(t):
    """Return (kind,row,col,label) for a product term, or None."""
    if t[0] == 'a' and t[2] == 3:
        return ('w', t[1], 3, 'W')
    if t[0] == 'w' and t[2] == 3:
        return ('w', t[1], 3, 'W')
    if t[0] != '*':
        return None
    x, y = t[1], t[2]
    for p, q in ((x, y), (y, x)):
        if p[0] == 'a' and q[0] == 'b' and p[2] == q[1]:
            return ('pm', p[1], q[2], f"t{p[2]}")
        if q[0] == 'T':
            r = rowof(p)
            return ('tr', r, 3, 'xyz'[q[1]])
    return None


def rowof(e):
    if e[0] in ('a', 'w'):
        return e[1]
    if e[0] == '+':
        return rowof(e[1])
    if e[0] == '*':
        l = label(e)
        return l[1] if l else None
    return None


def render(e):
    if e[0] == '+':
        return f"({render(e[1])}+{render(e[2])})"
    l = label(e)
    if l:
        return l[3]
    if e[0] == 'a' or e[0] == 'w':
        return f"{e[0]}{e[1]}{e[2]}"
    return '?'


def leaves(e):
    if e[0] == '+':
        return leaves(e[1]) + leaves(e[2])
    return [e]


def summary(b):
    out = {}
    for s in run(b):
        lv = leaves(s)
        labs = [label(x) for x in lv]
        if any(l is None for l in labs):
            continue
        kinds = {l[0] for l in labs if l[3] != 'W'}
        prods = [l for l in labs if l[3] != 'W']
        if len(prods) != 3 or len(kinds) != 1:
            continue
        k = kinds.pop()
        r = prods[0][1]
        c = prods[0][2]
        key = ('P' if k == 'pm' else 'T') + f"{r}{'XYZW'[c]}"
        out[key] = render(s)
    return out


if __name__ == '__main__':
    ret = summary(open(sys.argv[1], 'rb').read())
    keys = sorted(ret)
    for p in sys.argv[1:]:
        o = summary(open(p, 'rb').read())
        diff = [k for k in keys if o.get(k) != ret[k]]
        print(f"{p}: {len(keys)-len(diff)}/{len(keys)} match; diff: " + " ".join(f"{k}={o.get(k)}" for k in diff))


def norm(e):
    """'{a,b}+c' for an expression string like ((t2+t1)+t0) or (W+((t0+t2)+t1))."""
    if e is None:
        return None
    m = re.search(r"\(\((\w+)\+(\w+)\)\+(\w+)\)", e)
    if not m:
        return e
    a, b, c = m.groups()
    return "{" + ",".join(sorted([a, b])) + "}+" + c


def nscore(p, ret=None):
    ret = ret or summary(open('retail.bin', 'rb').read())
    o = summary(open(p, 'rb').read())
    full = sum(o.get(k) == ret[k] for k in ret)
    nm = sum(norm(o.get(k)) == norm(ret[k]) for k in ret)
    bad = [f"{k}={norm(o.get(k))}" for k in sorted(ret) if norm(o.get(k)) != norm(ret[k])]
    return full, nm, bad
