from pathlib import Path
import sys, subprocess, json
root=Path(__file__).resolve().parents[1];out=root/'build/shroud-finite-review-oct20'
base=(root/'build/shroud-stlport-bitset-oct20/whole-nearest-template-first.cpp').read_text()
tests={}
for kind,dec,defn,call,name in [
 ('free-fastcall','','__declspec(noinline) Object *__fastcall NearestEnemyInView(Object *origin)','NearestEnemyInView(obj)','?NearestEnemyInView@@YIPAVObject@@PAV1@@Z'),
 ('free-thiscall','','__declspec(noinline) Object *__thiscall NearestEnemyInView(Object *origin)','NearestEnemyInView(obj)','?NearestEnemyInView@@YEPAVObject@@PAV1@@Z'),
 ('static-fastcall','static __declspec(noinline) Object *__fastcall NearestEnemyInView(Object *origin);','__declspec(noinline) Object *__fastcall Drawable::NearestEnemyInView(Object *origin)','NearestEnemyInView(obj)','?NearestEnemyInView@Drawable@@SIPAVObject@@PAV2@@Z'),
]:
    s=base.replace('__declspec(noinline) Object *rva00272AD5();',dec)
    assert 'Object *Drawable::rva00272AD5()' in s
    s=s.replace('Object *Drawable::rva00272AD5()',defn,1)
    s=s.replace('Rva00261790 near((Arg00261790 *)this, true);','Rva00261790 near((Arg00261790 *)origin, true);',1)
    s=s.replace('getClosestObject(&m_pos,','getClosestObject(&((NearestOriginPositionView *)origin)->position,',1)
    s=s.replace(defn,'struct NearestOriginPositionView {char pad[0x38]; Coord3D position;};\n'+defn,1)
    s=s.replace('((Drawable *)obj)->rva00272AD5()',call)
    tests[kind]=(s,name)
results=[]
for key,(s,name) in tests.items():
    p=out/(key+'.cpp');p.write_bytes(s.encode())
    for label,symbol,rva,size in [('provider',name,'0x00272AD5','214'),('parent','?onComingOutOfShroud@Drawable@@QAEXXZ','0x0027893E','747')]:
        r=subprocess.run([sys.executable,'tools/explain_mismatch.py',symbol,'--rva',rva,'--size',size,'--source',p.relative_to(root).as_posix()],cwd=root,capture_output=True,text=True,encoding='utf8',errors='replace',timeout=240)
        log=r.stdout+r.stderr;p.with_suffix('.'+label+'.log').write_bytes(log.encode())
        d={'key':key,'label':label,'name':symbol,'exit':r.returncode,'exact':'OK: bytes match' in log,'summary':[x for x in log.splitlines() if x.startswith(('compiled symbol size:','classification:','unresolved:','first diff:'))]}
        if r.returncode:d['error']=log[-1500:]
        results.append(d);(out/'abi-results.json').write_text(json.dumps(results,indent=2));print(json.dumps(d),flush=True)
