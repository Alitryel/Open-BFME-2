import sys,json,re
from pathlib import Path
from wdbg import Dbg
out=Path('build/4404ac-blockcost');tag=sys.argv[1]
d=Dbg(out/(tag+'.cpp'),out/(tag+'-trace.obj'),out/(tag+'-trace.log'))
records=[];active=False;bid=0
try:
 d.prompt();d.prompt('set $CanDeferOnBPByAddr = 1');d.prompt('break *0x10760539')
 for i in range(2500):
  o=d.prompt('cont',40)
  if 'terminated' in o or not o.strip():break
  if '0x10760539' in o:
   w=d.words(d.reg('eax'),1);sy=d.words(w[0],2);fn=d.string(sy[1])
   if active:break
   if '?rva004404AC@AptMpGameSetup@@' in fn:
    active=True
    for a in [0x10782897,0x10782919,0x10783a45,0x10783a7b,0x10782b26]:d.prompt('break *0x%x'%a)
    print(tag,'active',flush=True)
   continue
  if not active:continue
  if '0x10782897' in o:
   bid+=1;s=d.words(d.reg('esp'),20)
   r={'kind':'block','block':bid,'stack':s}
   for j in [0x10,0x1c,0x20,0x24,0x28,0x40,0x44]:
    a=s[j//4]
    if a>0x10000 and a<0x10000000:r['at_%x'%j]={'ptr':a,'words':d.words(a,28)}
   records.append(r)
  elif any(a in o for a in ['0x10782919','0x10783a45','0x10783a7b']):
   a=d.reg('eax');w=d.words(a,28)
   records.append({'kind':('penalty' if '0x10783a7b' in o else 'cost'),'site':re.search(r'0x107[0-9a-f]+',o).group(),'block':bid,'addr':a,'words':w,'factor':d.reg('edx')})
  elif '0x10782b26' in o:
   a=d.reg('eax');w=d.words(a,28);sw=d.words(w[0],20)
   r={'kind':'insert','block':bid,'addr':a,'words':w,'symbol':sw,'stack':d.words(d.reg('esp'),12)}
   if sw[0] and sw[0]<0x10000000:
    fw=d.words(sw[0],16);r['front']=fw
    if len(fw)>1 and 0x10000<fw[1]<0x10000000:r['name']=d.string(fw[1])
   records.append(r)
finally:
 (out/(tag+'-blocks.json')).write_text(json.dumps(records,indent=1))
 if d.p.poll() is None:
  try:d.prompt('quit')
  except Exception:d.p.kill()
 d.log.close()
print(tag,len(records),'records',flush=True)
