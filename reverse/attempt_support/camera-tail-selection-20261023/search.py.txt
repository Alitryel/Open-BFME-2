import concurrent.futures as cf
import difflib,json,re,subprocess,sys,time,hashlib
from pathlib import Path
ROOT=Path(__file__).resolve().parents[2]
sys.path.insert(0,str(ROOT/'tools'))
import build,capstone
from variant import resolve_bytes
OUT=Path(__file__).resolve().parent
NAME='?adjustCameraMovement@PolygonTrigger@@QAE_NPBUCoord3D@@0PAU2@_N@Z'
RVA,SIZE=0x2e3aa8,723
BASE=(OUT/'base.cpp').read_text()
TARGET=build.read_target_bytes(RVA,SIZE)
SMAP=build.load_symbol_map()

def seq(raw,relocs=()):
 md=capstone.Cs(capstone.CS_ARCH_X86,capstone.CS_MODE_32);md.detail=True
 ins=list(md.disasm(raw,RVA));norm=[]
 for i in ins:
  op=i.op_str
  if i.group(capstone.CS_GRP_JUMP):op='BRANCH'
  elif i.group(capstone.CS_GRP_CALL) and i.operands[0].type==capstone.x86.X86_OP_IMM:op='CALL'
  elif any(i.address-RVA<=off<i.address-RVA+i.size for off,_,_ in relocs):
   op=re.sub(r'(?<![a-z0-9_])(?:0x[0-9a-f]+|\d+)(?![a-z0-9_])','ADDR',op)
  else:op=re.sub(r'0x[0-9a-f]{6,}','ADDR',op)
  norm.append((i.mnemonic,op))
 return ins,norm

TI,TN=seq(TARGET)
def metrics(obj):
 raw,rr=build.read_object_symbol_bytes(obj,NAME,code_only=True)
 ins,norm=seq(raw,rr);sm=difflib.SequenceMatcher(None,norm,TN,autojunk=False)
 nd=sum(max(b-a,d-c)for tag,a,b,c,d in sm.get_opcodes()if tag!='equal')
 fixed,unresolved=(resolve_bytes(raw,rr,TARGET,RVA,SIZE,SMAP)
                   if len(raw)==SIZE else (raw,[]))
 return dict(size=len(raw),nd=nd,fit=sm.ratio(),exact=len(raw)==SIZE and fixed==TARGET and not unresolved,
             unresolved=sorted(set(unresolved)),sha=hashlib.sha256(raw).hexdigest())

def run(item):
 label,text=item
 src,obj=OUT/(label+'.cpp'),OUT/(label+'.obj');src.write_text(text)
 command,env=build.compiler_command(src,obj)
 p=subprocess.run(command,cwd=ROOT,env=env,capture_output=True,text=True)
 if p.returncode:return dict(id=label,error=(p.stdout+p.stderr)[-600:])
 return dict(id=label,**metrics(obj))

PREDS=['slide','i & 1','candidate.x != 0.0f','along.y != 0.0f','n != 0','delta.x > 0.0f','start != to']
SITES=['along.scale(-1.0f);','deltaLen=delta.length();','used.sub(start);','usedLen=used.length();',
       'remaining=deltaLen-usedLen;','remaining*=0.4f;','along.normalize();','along.scale(remaining);',
       'next.add(&along);','candidate=next;']
def assignment_decls(text):
 return text.replace('float deltaLen=','float deltaLen;deltaLen=').replace('float usedLen=','float usedLen;usedLen=').replace('float remaining=','float remaining;remaining=')

def dup(text,needle,pred):
 assert text.count(needle)==1,(needle,text.count(needle))
 return text.replace(needle,'if ('+pred+') { '+needle+' } else { '+needle+' }')

def singles():
 head=assignment_decls(BASE)
 for si,site in enumerate(SITES):
  for pi,pred in enumerate(PREDS):
   yield f'single-s{si}-p{pi}',dup(head,site,pred)
 # Duplicate scopes retain user-written constructor calls on both paths.
 scopes=[
  'remaining*=0.4f;along.normalize();along.scale(remaining);Coord3D next(candidate);next.add(&along);if(rva002E3A39(next))candidate=next;',
  'along.scale(remaining);Coord3D next(candidate);next.add(&along);if(rva002E3A39(next))candidate=next;',
  'Coord3D next(candidate);next.add(&along);if(rva002E3A39(next))candidate=next;',
  'if(rva002E3A39(next))candidate=next;',
 ]
 for si,site in enumerate(scopes):
  for pi,pred in enumerate(PREDS):yield f'scope-s{si}-p{pi}',dup(BASE,site,pred)

def factorial():
 head=assignment_decls(BASE)
 # All sites use already initialised values, and every arm performs the
 # identical original operation. No speculative read or write is added.
 sites=SITES[1:9]
 for pi,pred in enumerate(PREDS):
  for mask in range(256):
   text=head
   for bit,site in enumerate(sites):
    if mask&(1<<bit):text=dup(text,site,pred)
   yield f'factor-p{pi}-{mask:02x}',text

def numbering():
 # Inert initialized declarations are discarded by the compiler, but can
 # change numbering of later scalar symbols and commutative operand ties.
 positions=['const Coord3D*start=from;',
            'Coord2D current,previous,intersection;',
            'Coord3D candidate;',
            'Coord3D along(previous.x,previous.y,0.0f);',
            'float deltaLen=delta.length();',
            'Coord3D used(candidate);',
            'float remaining=deltaLen-usedLen;',
            'remaining*=0.4f;']
 for ti,typ in enumerate(['int','float']):
  for pi,pos in enumerate(positions):
   for k in range(1,65):
    declarations=''.join(f'{typ} numbering{j}=0;' for j in range(k))
    yield f'number-t{ti}-p{pi}-{k:02d}',BASE.replace(pos,declarations+pos)

def layouts():
 # WB's MathCoord3D Sub/Add accept Coord3DBase. Isolate the effect of
 # expressing that actual base relationship rather than the flat view.
 import itertools
 for base,ctor,assign,dtor,force,formal in itertools.product(range(2),range(4),range(4),range(2),range(2),range(2)):
  if not base and formal:continue
  parent='struct Coord3DBase { float x,y,z; };' if base else ''
  fields=': public Coord3DBase {' if base else '{ float x,y,z;'
  ccopy=['x=o.x;y=o.y;z=o.z;', 'set(&o);', '*this=o;',
         'x=o.x;y=o.y;z=o.z;'][ctor]
  cfloat='x=ax;y=ay;z=az;'
  if ctor==3 and not base:
   ctors='Coord3D(const Coord3D&o):x(o.x),y(o.y),z(o.z){} Coord3D(float ax,float ay,float az):x(ax),y(ay),z(az){}'
  else:ctors='Coord3D(const Coord3D&o){'+ccopy+'} Coord3D(float ax,float ay,float az){'+cfloat+'}'
  assignment=['','Coord3D&operator=(const Coord3D&o){x=o.x;y=o.y;z=o.z;return *this;}',
              'Coord3D&operator=(const Coord3D&o){set(&o);return *this;}',
              'Coord3D&operator=(const Coord3D&o){if(this!=&o){x=o.x;y=o.y;z=o.z;}return *this;}'][assign]
  spec='__forceinline ' if force else ''
  typ='Coord3DBase' if formal else 'Coord3D'
  methods=spec+'void set(const '+typ+'*p){x=p->x;y=p->y;z=p->z;} '+spec+'void sub(const '+typ+'*p){x-=p->x;y-=p->y;z-=p->z;} '+spec+'void add(const '+typ+'*p){x+=p->x;y+=p->y;z+=p->z;} '+spec+'void scale(float s){x*=s;y*=s;z*=s;}'
  header=parent+'struct Coord3D '+fields+'public:Coord3D(){} '+ctors+assignment+('~Coord3D(){}' if dtor else '')+'float length()const;void normalize();'+methods+'};'
  text=BASE.replace('#include "Code/Libraries/Include/Lib/Coord3DCtorView.h"',header)
  yield f'layout-{base}{ctor}{assign}{dtor}{force}{formal}',text

def coarse():
 # Longer equal paths exercise the pre-merge cost phase, including
 # constructor references that single-statement variants omit.
 slide=BASE[BASE.index(' if(slide){'):BASE.index('\n *out=candidate;')]
 loop=BASE[BASE.index(' for(;i<n;'):BASE.index('\n return false;\n}')]
 main=BASE[BASE.index('\n const Coord3D*start='):BASE.rfind('\n}')]
 regions=[slide,loop,main,
          BASE[BASE.index(' Coord3D candidate;')+len(' Coord3D candidate;'):BASE.index('\n *out=candidate;')],
          '*out=candidate;return true;']
 predicates=['slide','from != to','out != to','out != from','this != 0']
 for ri,region in enumerate(regions):
  for pi,pred in enumerate(predicates):
   yield f'coarse-r{ri}-p{pi}',dup(BASE,region,pred)

def shapes():
 # Factorial around the WB temporary lifetime and CtorView scalar
 # replacement choices. Each edit preserves the existing math sequence.
 import itertools
 for bits in range(1024):
  text=BASE
  if bits&1:
   text=text.replace('float remaining=deltaLen-usedLen;\n  if(remaining>0.1f){remaining*=0.4f;',
                     'if(deltaLen-usedLen>0.1f){float remaining=(deltaLen-usedLen)*0.4f;')
  if bits&2:text=text.replace('Coord3D used(candidate);used.sub(start);\n  float usedLen=used.length();','float usedLen;{Coord3D used(candidate);used.sub(start);usedLen=used.length();}')
  if bits&4:text=text.replace('next.add(&along);','next.add(along);')
  if bits&8:text=text.replace('along.scale(remaining);Coord3D next(candidate);','Coord3D next(candidate);along.scale(remaining);')
  if bits&16:text=text.replace('Coord3D next(candidate);','Coord3D next(candidate.x,candidate.y,candidate.z);')
  if bits&32:text=text.replace('along.scale(remaining);','along.x=remaining*along.x;along.y=remaining*along.y;along.z=remaining*along.z;')
  if bits&64:text=text.replace('next.add(&along);','next.x=along.x+next.x;next.y=along.y+next.y;next.z=along.z+next.z;').replace('next.add(along);','next.x=along.x+next.x;next.y=along.y+next.y;next.z=along.z+next.z;')
  if bits&128:text=text.replace('float deltaLen=delta.length();','const float deltaLen=delta.length();').replace('float usedLen=used.length();','const float usedLen=used.length();')
  if bits&256:
   helper=text[text.index('inline __declspec(noinline) float Coord2D::length'):text.index('class CameraPlanar')]
   text=text.replace(helper,'')+'\n'+helper
  if bits&512:text=text.replace('Coord3D next(candidate);','const Coord3D &basePoint=candidate;Coord3D next(basePoint);').replace('Coord3D next(candidate.x,candidate.y,candidate.z);','const Coord3D &basePoint=candidate;Coord3D next(basePoint.x,basePoint.y,basePoint.z);')
  yield f'shape-{bits:03x}',text

def bits_paths():
 head=assignment_decls(BASE)
 predicates=['*(const unsigned*)&candidate.x != 0',
             '*(const unsigned*)&candidate.y != 0',
             '*(const unsigned*)&along.x != 0',
             '*(const unsigned*)&along.y != 0',
             '*(const unsigned*)&delta.x != 0',
             '*(const unsigned*)&delta.y != 0',
             'out != to', 'out != from']
 # Tail sites alone: the source-cost effect was sensitive to the
 # comparison's operand type in the first family.
 sites=SITES[5:]
 for pi,pred in enumerate(predicates):
  for mask in range(1,32):
   for shape in range(2):
    text=head
    for bit,site in enumerate(sites):
     if mask&(1<<bit):
      if shape==0:text=dup(text,site,pred)
      else:
       text=text.replace(site,'{bool branchCost'+str(bit)+'=('+pred+'); if(branchCost'+str(bit)+') {'+site+'} else {'+site+'}}')
    yield f'bits-p{pi}-{mask:02x}-s{shape}',text

def parentheses():
 # The older paren attempt preceded the bank's planar-constructor change.
 # Test interactions on the current symbol numbering, including mutator IL.
 for mask in range(2048):
  text=BASE
  header=(ROOT/'Code/Libraries/Include/Lib/Coord3DCtorView.h').read_text()
  replacements=[('normal.x*0.1f+intersection.x','(normal.x*0.1f)+intersection.x'),
   ('normal.y*0.1f+intersection.y','(normal.y*0.1f)+intersection.y'),
   ('along.x*delta.x+along.y*delta.y','(along.x*delta.x)+(along.y*delta.y)'),
   ('deltaLen-usedLen','(deltaLen-usedLen)'),
   ('remaining*=0.4f','remaining=(remaining*0.4f)'),
   ('x *= s; y *= s; z *= s;','x = (x*s); y = (y*s); z = (z*s);'),
   ('x += p->x; y += p->y; z += p->z;','x += (p->x); y += (p->y); z += (p->z);'),
   ('x += p->x; y += p->y; z += p->z;','x = (x + p->x); y = (y + p->y); z = (z + p->z);'),
   ('x=x*x','x=(x*x)'),
   ('sqrt(x*x+y*y)','sqrt((x*x)+(y*y))'),
   ('sqrt(x*x+y*y+z*z)','sqrt((x*x)+(y*y)+(z*z))')]
  # bit 8 adds parentheses to three sub expressions instead of a dummy.
  replacements[8]=('x -= p->x; y -= p->y; z -= p->z;','x = (x - p->x); y = (y - p->y); z = (z - p->z);')
  for bit,(old,new) in enumerate(replacements):
   if mask&(1<<bit):text=text.replace(old,new);header=header.replace(old,new)
  text=text.replace('#include "Code/Libraries/Include/Lib/Coord3DCtorView.h"',header)
  yield f'paren-{mask:03x}',text

def value_boundaries():
 # The other camera's read-only compiler traces establish that by-value
 # aggregate boundaries can prevent scalar-pseudo subsumption. Test that
 # mechanism at this bank's scale/add boundary, with ordinary inline C++.
 import itertools
 old='along.scale(remaining);Coord3D next(candidate);next.add(&along);'
 assert old in BASE
 types=['Coord3D','const Coord3D&','const Coord3D*']
 for ai,bi,order,body,stage in itertools.product(range(3),range(3),range(2),range(4),range(2)):
  a='(*a)' if ai==2 else 'a';b='(*b)' if bi==2 else 'b'
  adecl=types[ai]+' a';bdecl=types[bi]+' b'
  if body==0:code=f'Coord3D r({a});r.add(&{b});return r;'
  elif body==1:code=f'return Coord3D({a}.x+{b}.x,{a}.y+{b}.y,{a}.z+{b}.z);'
  elif body==2:code=f'Coord3D r({b});r.add(&{a});return r;'
  else:code=f'Coord3D r; r.x={a}.x+{b}.x;r.y={a}.y+{b}.y;r.z={a}.z+{b}.z;return r;'
  helper='static __forceinline Coord3D camera_sum('+adecl+','+bdecl+'){'+code+'}\n'
  args=['candidate','along'] if order==0 else ['along','candidate']
  args[0]=('&'+args[0]) if ai==2 else args[0]
  args[1]=('&'+args[1]) if bi==2 else args[1]
  tail='along.scale(remaining);Coord3D next(camera_sum('+','.join(args)+'));'
  if stage:
   helper='static __forceinline Coord3D camera_scaled(Coord3D p,float r){p.scale(r);return p;}\n'+helper
   args=[arg.replace('along','scaled') for arg in args]
   tail='Coord3D scaled(camera_scaled(along,remaining));Coord3D next(camera_sum('+','.join(args)+'));'
  text=BASE.replace('bool PolygonTrigger::adjustCameraMovement(',helper+'bool PolygonTrigger::adjustCameraMovement(').replace(old,tail)
  yield f'value-a{ai}-b{bi}-o{order}-m{body}-s{stage}',text

def main():
 mode=sys.argv[1]
 if mode=='diff':
  for name in sys.argv[2:]:
   p=Path(name).with_suffix('.obj');raw,rr=build.read_object_symbol_bytes(p,NAME,code_only=True)
   ins,norm=seq(raw,rr);sm=difflib.SequenceMatcher(None,TN,norm,autojunk=False)
   print(name,metrics(p))
   for tag,a,b,c,d in sm.get_opcodes():
    if tag=='equal':continue
    print(tag)
    for i in TI[max(0,a-1):min(len(TI),b+1)]:print('R',hex(i.address-RVA),i.mnemonic,i.op_str)
    for i in ins[max(0,c-1):min(len(ins),d+1)]:print('C',hex(i.address-RVA),i.mnemonic,i.op_str)
  return
 items=[('baseline',BASE)]
 if mode=='single':items+=list(singles())
 elif mode=='factor':items+=list(factorial())
 elif mode=='number':items+=list(numbering())
 elif mode=='layout':items+=list(layouts())
 elif mode=='coarse':items+=list(coarse())
 elif mode=='shape':items+=list(shapes())
 elif mode=='bits':items+=list(bits_paths())
 elif mode=='paren':items+=list(parentheses())
 elif mode=='value':items+=list(value_boundaries())
 elif mode!='base':raise ValueError(mode)
 start=time.time();rows=[]
 with (OUT/(mode+'.jsonl')).open('w')as log,cf.ThreadPoolExecutor(max_workers=8)as pool:
  for r in pool.map(run,items):
   rows.append(r);log.write(json.dumps(r)+'\n');log.flush()
   if r.get('exact'):print('EXACT',r['id'],flush=True)
 good=sorted([r for r in rows if 'error'not in r],key=lambda r:(r['nd'],abs(r['size']-SIZE)))
 print('count',len(rows),'errors',len(rows)-len(good),'distinct',len({r['sha']for r in good}),'seconds',round(time.time()-start,1))
 for r in good[:15]:print(json.dumps(r))

if __name__=='__main__':main()
