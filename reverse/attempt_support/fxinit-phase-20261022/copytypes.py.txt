# Preserve the optional-copy semantics, but make duplicate paths differ in
# high-level copy operations, asking whether machine lowering merges them late.
import sys,itertools,json,time
from pathlib import Path
from concurrent.futures import ThreadPoolExecutor,as_completed
sys.path.insert(0,'build/fxinit-seat22');import search as h
h.HERE=Path('build/fxinit-phase22').resolve()
def copy(src,off,mode):
 d=f'(vertices+desc.{off}(1))';s=f'(&{src}[1][i])'
 if mode==0:return f'*(Vector3 *){d}=*{s};'
 if mode==1:return f'memcpy({d},{s},sizeof(Vector3));'
 if mode==2:return '{Vector3 copyValue=*'+s+';*(Vector3 *)'+d+'=copyValue;}'
 if mode==3:return '{Vector3 *out=(Vector3 *)'+d+';const Vector3 *in='+s+';out->X=in->X;out->Y=in->Y;out->Z=in->Z;}'
 if mode==4:return '{char *out='+d+';const char *in=(const char *)'+s+';memcpy(out,in,4);memcpy(out+4,in+4,4);memcpy(out+8,in+8,4);}'
 if mode==5:return '{char *out='+d+';const char *in=(const char *)'+s+';unsigned first,second,third;memcpy(&first,in,4);memcpy(&second,in+4,4);memcpy(&third,in+8,4);memcpy(out,&first,4);memcpy(out+4,&second,4);memcpy(out+8,&third,4);}'
 if mode==6:return '{char *out='+d+';const char *in=(const char *)'+s+';for(unsigned k=0;k<12;++k)out[k]=in[k];}'
 if mode==7:return '{char *out='+d+';const char *in=(const char *)'+s+';__int64 first;unsigned last;memcpy(&first,in,8);memcpy(&last,in+8,4);memcpy(out,&first,8);memcpy(out+8,&last,4);}'
def variants():
 old=h.P+'\n      '+h.N
 for p,n,cond,form in itertools.product(range(8),range(8),range(4),range(2)):
  pred=['tangents','binormals','basis','(m_FVF&0x10)'][cond]
  cp=copy('positions','Get_Location_Offset',p);cn=copy('normals','Get_Normal_Offset',n)
  alt=f'if(positions[1]){{{cp}}}if(normals[1]){{{cn}}}'
  if form==0:body=f'if({pred}){{{h.P}{h.N}}}else{{{alt}}}'
  else:
   # Source-distinct copy paths converge before the following null guard.
   body='if(positions[1]){if('+pred+'){'+copy('positions','Get_Location_Offset',0)+'}else{'+cp+'}}if(normals[1]){if('+pred+'){'+copy('normals','Get_Normal_Offset',0)+'}else{'+cn+'}}'
  src=h.BASE.replace(old,body)
  # Intrinsic memcpy makes the exact-size copies visible; variable-size final
  # index memcpy must still resolve to the same retail call.
  if '--plain' not in sys.argv:src=src.replace('#include <string.h>','#include <string.h>\n#pragma intrinsic(memcpy)')
  assert src!=h.BASE
  yield ('copyplain_'if '--plain'in sys.argv else'copytypes_')+f'{p}{n}{cond}{form}',src
if __name__=='__main__':
 t=time.time();rs=[]
 with ThreadPoolExecutor(max_workers=8)as pool:
  for f in as_completed([pool.submit(h.run_one,x)for x in variants()]):
   rs.append(f.result())
   if len(rs)%64==0:print('tested',len(rs),'best',min(rs,key=lambda x:(x.get('nd',99999),x.get('cost',99999))),flush=True)
 rs.sort(key=lambda x:(x.get('nd',99999),x.get('cost',99999)))
 (h.HERE/('copyplain-results.json'if '--plain'in sys.argv else'copytypes-results.json')).write_text(json.dumps(rs,indent=2));print(json.dumps({'count':len(rs),'seconds':time.time()-t,'best':rs[:4]},indent=2))
