from headmerge import *
# Unlike the earlier ordering comparisons, these use equality of bijective
# 32-bit integer transforms. Their only false input is the null pointer.
# All arithmetic is unsigned to preserve high-bit pointers without overflow UB.
checks=[
 '-(unsigned)p != 0','(0u-(unsigned)p) != 0',
 '((unsigned)p+1u)!=1u','((unsigned)p-1u)!=0xffffffffu',
 '((unsigned)p+0x80000000u)!=0x80000000u',
 '((unsigned)p*3u)!=0','((unsigned)p*5u)!=0',
 '((unsigned)p*0xffffffffu)!=0',
 '((unsigned)p | (0u-(unsigned)p)) != 0',
 '((unsigned)p ^ 0xffffffffu) != 0xffffffffu',
 '((unsigned)p ^ ((unsigned)p >> 16)) != 0',
 '(((unsigned)p >> 16) | ((unsigned)p << 16)) != 0',
 '(((unsigned)p >> 31) | ((unsigned)p << 1)) != 0',
 '((unsigned)p & 0xffffu) || ((unsigned)p >> 16)',
 '(((unsigned)p | (0u-(unsigned)p)) >> 31) != 0',
 '!(((unsigned)p | (0u-(unsigned)p)) < 0x80000000u)',
]
items=[]
for f,c,nul,order,contract in itertools.product([8,9],range(len(checks)),range(2),range(2),range(2)):
 form=forms[f].replace('obj=0','obj='+( '(Object*)p' if nul else '0'))
 form=form.replace('p?p->m_object:0','p?p->m_object:'+( '(Object*)p' if nul else '0'))
 form=form.replace('if(p)','if('+checks[c]+')').replace('obj=p?','obj=('+checks[c]+')?')
 if order:form=form.replace('draw=p; obj=p->m_object;','obj=p->m_object; draw=p;')
 src=BASE[:lo]+decl+form+'\n'+BASE[hi:]
 if not contract:src=src.replace('template <> void StringBase<char>::releaseBuffer() throw();\n','')
 items.append(('aj%02d%02d%d%d%d'%(f,c,nul,order,contract),src))
if __name__=='__main__':batch(items,'arithjoin')
