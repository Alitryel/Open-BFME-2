import json,sys
from pathlib import Path
from wdbg import Dbg
W=Path(__file__).resolve().parent
tag=sys.argv[1]
d=Dbg(W/(tag+'.cpp'),W/(tag+'-rounds.obj'),W/(tag+'-rounds.log'))
on=False;seen=False;events=[];finished=False
def members(s):
 out=[];seen=set();p=d.words(s,1)[0] if s else 0
 while p and p not in seen:
  seen.add(p);w=d.words(p,3)
  for i in range(32):
   if w[2]>>i&1:out.append(w[0]+i)
  p=w[1]
 return out
try:
 d.prompt();d.prompt('set $CanDeferOnBPByAddr = 1')
 d.prompt('break *0x10783acd');added=False
 for _ in range(1000):
  o=d.prompt('cont',60)
  if 'terminated' in o:finished=True;print('compiler terminated',flush=True);break
  eip=d.reg('eip')
  if eip==0x10783acd:
   if not added:
    d.prompt('break *0x10782978');d.prompt('break *0x107a8f54');added=True
   f=d.words(d.reg('ebp'),1)[0];fn=d.string(d.words(f+4,1)[0])
   on='?bfmeTwoCGD@' in fn and d.reg('ebx')==0
   if on:
    seen=True;print(d.prompt('bt'),flush=True)
   elif seen:
    d.prompt('delete 1');d.prompt('delete 2');d.prompt('delete 3');break
   continue
  if not on:continue
  st=d.words(d.reg('esp'),6);indices=members(st[3] if eip==0x10782978 else st[2]);rows=[]
  for idx in indices:
   p=d.words(0x108807b8+4*(idx&1023),1)[0]
   while p:
    w=d.words(p,18)
    if w[7]==idx:
     rows.append(dict(idx=idx,p=p,w=w,sym=d.words(w[0],24)));break
    p=w[12]
  events.append(dict(eip=eip,stack=st,rows=rows))
  print('event',hex(eip),[(x['idx'],x['sym'][7],hex(x['w'][1])) for x in rows],flush=True)
 if not finished:
  for _ in range(200):
   o=d.prompt('cont',60)
   if 'terminated' in o:
    finished=True;print('compiler terminated',flush=True);break
  else:raise RuntimeError('compiler did not finish')
finally:
 (W/(tag+'-rounds.json')).write_text(json.dumps(events,indent=1))
 if d.p.poll() is None:
  try:d.prompt('quit')
  except Exception:d.p.kill()
 d.log.close()
