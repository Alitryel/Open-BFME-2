import json
import re
import subprocess
import sys
from pathlib import Path

root = Path(__file__).resolve().parents[2]
scratch = root / 'build/resume-eh-local'
base = (scratch / 'turn47-fxinit-baseline.cpp').read_text()
variants = {}
for profile in ('vc71-sp1', 'vc71-ddk-4035'):
    variants[profile] = '// compiler: ' + profile + '\n' + base

# These accessors carry the exact member-read semantics in the ZH dx8fvf.h
# declarations; BFME2's descriptor offsets stay as measured from retail.
basic = {'locationOffset': 'Get_Location_Offset',
         'normalOffset': 'Get_Normal_Offset',
         'diffuseOffset': 'Get_Diffuse_Offset',
         'stride': 'Get_FVF_Size'}
all_fields = dict(basic, extensionOffset='extensionOffsetValue',
                  extensionEnd='extensionEndValue', basisOffset='basisOffsetValue',
                  basisSecondOffset='basisSecondOffsetValue', basisEnd='basisEndValue',
                  extensionDataOffset='extensionDataOffsetValue')
for label, fields in [('fvf-basic', basic), ('fvf-all-views', all_fields)]:
    methods = ''.join(' unsigned %s() const { return %s; }\n' % (method, field)
                      for field, method in fields.items())
    source = base.replace(' unsigned Get_Tex_Offset(unsigned n) const',
                          methods + ' unsigned Get_Tex_Offset(unsigned n) const')
    for field, method in fields.items():
        source = re.sub(r'\bdesc\.' + field + r'\b', 'desc.' + method + '()', source)
    variants[label] = source
    # Existing native count getters cover all direct count uses too. This
    # preserves signedness and the measured layouts; no dead statements,
    # artificial temporaries or volatile operations are introduced.
    source = source.replace('mesh->VertexCount', 'mesh->Get_Vertex_Count()')
    source = source.replace('mesh->PolyCount', 'mesh->Get_Polygon_Count()')
    variants[label + '-counts'] = source

results = []
for label, source in variants.items():
    path = scratch / ('turn47-fx-' + label + '.cpp')
    path.write_text(source)
    proc = subprocess.run([sys.executable, str(root / 'tools/explain_mismatch.py'),
        '?Init@FXShaderGeometry@@QAE_NPAVMeshModelClass@@@Z', '--source',
        path.relative_to(root).as_posix(), '--rva', '0x0018C84E', '--size', '2333'],
        cwd=root, capture_output=True, text=True)
    log = scratch / ('turn47-fx-' + label + '.txt')
    log.write_text(proc.stdout + proc.stderr)
    size = re.search(r'compiled symbol size: (\d+)', proc.stdout)
    differences = [line for line in proc.stdout.splitlines() if line.startswith('!=')]
    result = {'variant': label, 'exit': proc.returncode,
              'compiled_size': int(size[1]) if size else None,
              'differing_instruction_rows_including_unresolved': len(differences),
              'log': log.relative_to(root).as_posix()}
    results.append(result)
    print(json.dumps(result), flush=True)
(scratch / 'turn47-fx-trials.json').write_text(json.dumps(results, indent=2) + '\n')
