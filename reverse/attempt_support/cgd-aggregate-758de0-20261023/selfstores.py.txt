from pathlib import Path
from itertools import product
W=Path(__file__).resolve().parent;b=(W/'bank.cpp').read_text()
D=W/'selfstores';D.mkdir(exist_ok=True)
for i,axes in enumerate(product(range(3),range(3),range(3),range(3),range(3),range(3),range(2),range(2))):
 *sites,abi,bridge=axes;s=b
 if not abi:
  a=s.index('struct PArg {');z=s.index('struct ContactData {');s=s[:a]+s[z:]
  s=s.replace('altPrepare(node->first->source,node->point,ContactVector(node->normal).negate())','node->first->source->prepare(node->point,ContactVector(node->normal).negate())')
 if not bridge:
  s=s.replace('__forceinline void *__cdecl bridge(ContactData *s,const ContactVector &p,const ContactVector &n) { return s->source->prepare(p,n); }\n','')
  s=s.replace('bridge(node->second,node->point,node->normal)','node->second->source->prepare(node->point,node->normal)')
 h,t=s.split('void BfmeThingCGD::bfmeTwoCGD() {')
 def receiver(role,k):
  d='node->'+role;r=d+'->source'
  return [r,'('+r+'='+r+')','('+d+'='+d+')->source'][k]
 # Site order is outer test, inner test, outer first response, inner first response,
 # outer reverse response, inner reverse response. The bridge receives the data view.
 parts=t.splitlines()
 for j,line in enumerate(parts):
  if 'node->state=node->first->source->test' in line:
   line=line.replace('node->first->source',receiver('first',sites[0])).replace('node->second->source',receiver('second',sites[1]))
  elif 'node->first->source->respond' in line:
   line=line.replace('node->first->source',receiver('first',sites[2]))
   if bridge:
    d='node->second'
    if sites[3]==1:
     # An identity store of the same source before the unchanged data helper.
     line=line.replace('bridge(node->second,','bridge((node->second->source=node->second->source,node->second),')
    elif sites[3]==2:line=line.replace('bridge(node->second,','bridge((node->second=node->second),')
   else:line=line.replace('node->second->source',receiver('second',sites[3]))
  elif 'node->second->source->respond' in line:
   line=line.replace('node->second->source',receiver('second',sites[4])).replace('node->first->source',receiver('first',sites[5]))
  parts[j]=line
 s=h+'void BfmeThingCGD::bfmeTwoCGD() {'+'\n'.join(parts)
 (D/('v%04d.cpp'%i)).write_text(s)
print('generated',i+1)
