import sys,concurrent.futures,itertools,json
from pathlib import Path
sys.path.insert(0,str(Path('build/trust237-session').resolve()))
import search
from search import OUT,ROOT,NAME,RVA,SIZE,TARGET,build,subprocess,capstone,resolve_bytes
codes=['0x800b0100L']
tails=[
 'bool result = false; if(status==0 || status==CODE) result=true; return result;',
 'int result = 0; if(status==0 || status==CODE) result=1; return result;',
 'int result; if(status==0 || status==CODE) result=1; else result=0; return result;',
 'int result = 1; if(status!=0 && status!=CODE) result=0; return result;',
 'if(status==0 || status==CODE) return 1; return 0;',
 'return (status==0 || status==CODE) ? 1 : 0;',
 'switch(status) {case 0: return 1; case CODE: return 1; default: return 0;}',
 'int result=0; if(status==0) goto yes; if(status!=CODE) goto done; yes: result=1; done:return result;',
]

def run(job):
 n,code,tail,early,kind,init=job
 src=OUT/f'r{n:04}.cpp';obj=src.with_suffix('.obj')
 text=search.make_source(n,0,early,0,0,0,0,'/O1 /G7 /arch:SSE',0)
 text=text.replace('LocalFree(argv); return 1;','LocalFree(argv); '+tail.replace('CODE',code))
 # make_source tail0 declares an unused status, and this diagnostic now uses it.
 text=text.replace('long status',kind+' status')
 if init:text=text.replace('    TrustFile file;',kind+' status=0;\n    TrustFile file;').replace(kind+' status=WinVerifyTrust','status=WinVerifyTrust').replace('    '+kind+' status;','')
 src.write_text(text)
 cmd,env=build.compiler_command(src,obj)
 proc=subprocess.run(cmd,env=env,cwd=ROOT,capture_output=True,text=True)
 if proc.returncode:return {'n':n,'error':proc.stdout+proc.stderr}
 raw,rel=build.read_object_symbol_bytes(obj,NAME,code_only=True)
 blob,un=resolve_bytes(raw,rel,TARGET,RVA,SIZE,{'_memset':[0x6291ae],'_WinVerifyTrust@12':[0x655652]})
 diff=[i for i,(a,b) in enumerate(itertools.zip_longest(blob,TARGET)) if a!=b]
 return {'n':n,'code':code,'tail':tail,'early':early,'kind':kind,'init':init,'size':len(raw),'outside':[i for i in diff if not 0xcd<=i<0xdd],'diff':len(diff),'last':blob[-42:].hex()}
jobs=[(n,*x) for n,x in enumerate(itertools.product(codes,tails,range(2),['const long','int','unsigned long','const unsigned long'],range(2)))]
jobs=[j for j in jobs if not(j[-1] and j[-2].startswith('const'))]
with concurrent.futures.ThreadPoolExecutor(max_workers=8) as p: results=list(p.map(run,jobs))
(OUT/'status_probe3.json').write_text(json.dumps(results,indent=2))
for r in sorted(results,key=lambda r:(len(r.get('outside',[None]*999)),r.get('diff',999)))[:8]:print(json.dumps(r))
