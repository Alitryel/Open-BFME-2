import sys,json,itertools,time
from pathlib import Path
from concurrent.futures import ThreadPoolExecutor
sys.path.insert(0,'build/fxinit-seat22');import search as h
h.HERE=Path('build/fxinit-index22').resolve()
base=(h.HERE/'trials/realvector_b1100.cpp').read_text()
a=base.index('struct _Rb_tree_node_base');b=base.index('template<> inline void allocator<short>',a)
base=base[:a]+base[b:]
base=base.replace('#include <functional>','#include <functional>\n#include <set>')
base=base.replace('template<class T> struct _Rb_global {static _Rb_tree_node_base *_M_increment(_Rb_tree_node_base *);};','')
base=base.replace('_M_t._M_leftmost()','_STL::set<int,_STL::less<int>,_STL::allocator<int> >::begin()._M_node')
base=base.replace('_M_t._M_header._M_data','_STL::set<int,_STL::less<int>,_STL::allocator<int> >::end()._M_node')
T='_Rb_tree<int,int,_Identity<int>,less<int>,allocator<int> >'
TC=T+'::_Rb_tree(const less<int>&cmp,const allocator<int>&a)'
SC='set<int,less<int>,allocator<int> >::set()'
def variants():
 for tree,setctor in itertools.product(range(4),repeat=2):
  ext=''
  if tree:
   ext+='template<> '+('__forceinline 'if tree==3 else '__declspec(noinline) ')+TC
   ext+= ';\n' if tree==1 else ':_Rb_tree_base<int,allocator<int> >(a),_M_node_count(0),_M_key_compare(cmp) {_M_empty_initialize();}\n'
  if setctor:
   ext+='template<> '+('__forceinline 'if setctor==3 else '__declspec(noinline) ')+SC
   ext+=';\n' if setctor==1 else ':_M_t(less<int>(),allocator<int>()) {}\n'
  x=base.replace('template<> inline void allocator<short>',ext+'\ntemplate<> inline void allocator<short>')
  yield 'realtree_'+str(tree)+str(setctor),x
if __name__=='__main__':
 t=time.time()
 with ThreadPoolExecutor(max_workers=8)as ex:r=list(ex.map(h.run_one,variants()))
 r.sort(key=lambda x:(x.get('cost',99999),x.get('nd',99999)))
 (h.HERE/'tree-results.json').write_text(json.dumps(r,indent=2))
 print(json.dumps({'count':len(r),'seconds':time.time()-t,'errors':sum('error'in x for x in r),'best':r[:4]},indent=2))
