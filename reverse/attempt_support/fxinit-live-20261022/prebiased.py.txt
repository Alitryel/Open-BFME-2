import sys,itertools,json,time
from pathlib import Path
from concurrent.futures import ThreadPoolExecutor
sys.path.insert(0,'build/fxinit-seat22')
import search as h
h.HERE=Path('build/fxinit-live22').resolve()
STREAMS=['positions[0]','normals[0]','tangents','binormals','positions[1]','normals[1]']
LOOP='   for(unsigned i=0;i<(unsigned)mesh->Get_Vertex_Count();++i) {'
def variants():
 for anchor,mask,bias,scope,typ in itertools.product(range(2),range(1,64),(4,8),range(3),range(2)):
  origin=['tangents','binormals'][anchor];src=h.BASE;decl=[]
  lo=src.index(LOOP);hi=src.index(' m_IB=new',lo);core=src[lo:hi]
  for k,stream in enumerate(STREAMS):
   if mask>>k&1:
    d='streamDelta'+str(k);decl.append('unsigned '+d+'=(unsigned)'+stream+'-((unsigned)'+origin+'+'+str(bias)+');')
    c='streamCursor' if typ==0 else '(unsigned)streamCursor'
    core=core.replace(stream+'[i]','*(const Vector3 *)('+c+'+'+d+')')
  if typ==0:init='unsigned streamCursor=(unsigned)'+origin+'+'+str(bias)+';';inc='streamCursor+=12;'
  else:init='const float *streamCursor=(const float *)((unsigned)'+origin+'+'+str(bias)+');';inc='streamCursor=(const float *)((unsigned)streamCursor+12);'
  core='   '+init+'\n'+core
  core=core.replace('    vertices+=desc.Get_FVF_Size();','    vertices+=desc.Get_FVF_Size();'+inc)
  ds=''.join(decl)
  if scope==0:core=ds+'\n'+core
  if scope==1:core=core.replace(LOOP,LOOP+'\n'+ds)
  if scope==2:
   pre=src[:lo];p=pre.index('  for(unsigned pass=0;');pre=pre[:p]+ds+'\n'+pre[p:]
   src=pre+src[lo:];lo=src.index(LOOP);hi=src.index(' m_IB=new',lo)
  src=src[:lo]+core+src[hi:]
  yield f'prebiased_{anchor}_{mask:02}_{bias}_{scope}{typ}',src
if __name__=='__main__':
 start=time.time()
 with ThreadPoolExecutor(max_workers=8) as pool:r=list(pool.map(h.run_one,variants()))
 r.sort(key=lambda x:(x.get('nd',99999),x.get('cost',99999)))
 (h.HERE/'prebiased-results.json').write_text(json.dumps(r,indent=2))
 print(json.dumps({'count':len(r),'valid':sum('nd'in x and not x.get('unresolved')for x in r),'seconds':time.time()-start,'best':r[:5]},indent=2))
