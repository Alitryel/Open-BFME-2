# Generate Transform_Vector row-grouping trials on top of the bank.
import sys, itertools
base, outdir = sys.argv[1], sys.argv[2]
s = open(base).read()
OLD = 'Matrix3D::Transform_Vector(lightTm, '
assert OLD in s
ANCH = '#include "matrix3d.h"\n'
def row(r, form):
    a = 'A[%d][0] * v->X' % r; b = 'A[%d][1] * v->Y' % r; c = 'A[%d][2] * v->Z' % r; d = 'A[%d][3]' % r
    return {
        'H': '(%s + %s + %s + %s)' % (a, b, c, d),
        'P1': '((%s + %s) + %s + %s)' % (a, b, c, d),
        'P2': '(((%s + %s) + %s) + %s)' % (a, b, c, d),
        'P3': '(%s + (%s + %s) + %s)' % (a, b, c, d),
        'F': '(float)(%s + %s + %s + %s)' % (a, b, c, d),
        'Q1': '((%s + %s) + %s + %s)' % (c, a, b, d),
        'Q2': '(%s + (%s + %s) + %s)' % (b, c, a, d),
        'Q3': '(((%s + %s) + %s) + %s)' % (c, a, b, d),
        'R1': '((%s + %s) + %s + %s)' % (b, a, c, d),
        'R2': '(((%s + %s) + %s) + %s)' % (b, a, c, d),
    }[form]
forms0 = sys.argv[3].split(',')
forms12 = sys.argv[4].split(',')
for f0 in forms0:
    for f1 in forms12:
        for f2 in forms12:
            helper = ('static WWINLINE void ShadowTransform(const Matrix3D & A,const Vector3 & in,Vector3 * out)\n{\n'
                      '\tVector3 tmp;\n\tVector3 * v;\n\tif (out == &in) {\n\t\ttmp = in;\n\t\tv = &tmp;\n\t} else {\n\t\tv = (Vector3 *)&in;\n\t}\n'
                      '\tout->X = %s;\n\tout->Y = %s;\n\tout->Z = %s;\n}\n' % (row(0, f0), row(1, f1), row(2, f2)))
            t = s.replace(ANCH, ANCH + helper, 1).replace(OLD, 'ShadowTransform(lightTm, ', 1)
            open('%s/tv_%s_%s_%s.cpp' % (outdir, f0, f1, f2), 'w').write(t)
