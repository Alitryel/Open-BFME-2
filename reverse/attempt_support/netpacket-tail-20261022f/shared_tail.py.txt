import itertools
from search import *
SIGN='NetCommandList *NetPacket::getCommandList()\n'; H,B=BANK.split(SIGN,1)
SEP="} else if (m_packet[i] == 'Z')"; D,Z=B.split(SEP,1)
PRED=['m_packetLen & 1','m_packetLen > offset','m_packetLen != i','relay & 1','commandType & 1','this != 0','m_packet[i-1] == 90','m_packetLen > 0']
def source(sel,form,scope):
 d,z=D,Z;c=PRED[sel]
 # A conditional jump to the existing D cursor tail survives through allocation.
 d=d.replace('NetCommandMsg *msg = 0;','NetCommandMsg *msg; msg = 0;').replace('NetCommandRef *ref = retval->addMessage(msg);','NetCommandRef *ref; ref = retval->addMessage(msg);')
 if scope==0:d=d.replace('i = offset;','sharedD: i = offset;')
 elif scope==1:d=d.replace('i = offset;','{ sharedD: i = offset; }')
 elif scope==2:d=d.replace('msg = 0;\n\t\t\ti = offset;','msg = 0;\n\t\t\t{ sharedD: i = offset; }')
 else:d=d.replace('i = offset;','if ('+c+') { sharedD: i = offset; } else i = offset;')
 forms=[
  'if ('+c+') goto sharedD; i = offset;',
  'if ('+c+') { i = offset; } else goto sharedD;',
  'if ('+c+') goto sharedD; else i = offset;',
  'if ('+c+') { i = offset; continue; } goto sharedD;',
  'switch (('+c+') ? 1 : 0) { case 0: goto sharedD; default: i = offset; break; }',
  'while ('+c+') goto sharedD; i = offset;',
  'do { if ('+c+') goto sharedD; i = offset; } while (false);',
  'if ('+c+') { i = offset; goto cursorDone; } goto sharedD; cursorDone:;',
 ]
 z=z.replace('i = offset;',forms[form])
 return H+SIGN+d+SEP+z
if __name__=='__main__':batch([('bank',BANK)]+[(f'dshared_{a}{b}{c}',source(a,b,c)) for a,b,c in itertools.product(range(8),range(8),range(4))],'shared-tail')
