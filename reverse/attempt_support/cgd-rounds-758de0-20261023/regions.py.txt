import json,difflib,collections,sys
from pathlib import Path
import search
W=Path(__file__).resolve().parent
bounds=[(0,0x94),(0x94,0xaa),(0xaa,0xbe),(0xbe,0xe1),(0xe1,0x136),(0x136,0x140),(0x140,333)]
def score(data):
 a=search.insns(data)
 sm=difflib.SequenceMatcher(None,[x[:2] for x in a],[x[:2] for x in search.T],autojunk=False)
 amounts=[0]*len(bounds)
 for tag,i,j,k,l in sm.get_opcodes():
  if tag=='equal':
   for t in search.T[k:l]:
    for m,(lo,hi) in enumerate(bounds):
     if lo<=t[3]<hi:amounts[m]+=t[2]
 return amounts
if __name__=='__main__':
 summaries={}
 for fam in sys.argv[1:]:
  rs=json.loads((W/(fam+'-results.json')).read_text());good=[r for r in rs if 'error' not in r]
  uniq={r['sha']:dict(r) for r in reversed(good)}
  for r in uniq.values():
   r['pair_regions']=score((W/fam/(r['name']+'.bin')).read_bytes());r['pair_sum']=sum(r['pair_regions'])
  rows=sorted(uniq.values(),key=lambda r:(-r['pair_sum'],r['positional']))
  info={'total':len(rs),'compiled':len(good),'distinct':len(rows),'top':rows[:12],'region_best':[max(rows,key=lambda r:r['pair_regions'][i]) for i in range(len(bounds))]}
  summaries[fam]=info
  print(fam,'total',len(rs),'compiled',len(good),'distinct',len(rows))
  for r in rows[:10]:print(r['name'],r['size'],r['positional'],r['pair_sum'],r['pair_regions'])
  print('region-best',[(r['name'],r['pair_regions'][i]) for i,r in enumerate(info['region_best'])])
 (W/'summary.json').write_text(json.dumps(summaries,indent=2))
