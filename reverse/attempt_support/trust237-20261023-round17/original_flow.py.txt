import original_check as common
import itertools,json,concurrent.futures

# Only the independently recovered comparison order is used. These controls
# can return zero and are never replacement candidates for the NOP baseline.
A='0x80096010L';B='0x800b0100L'
conds=['status=='+A+' || status=='+B,'status!='+A+' && status!='+B]
tails=[
 'int r=0;if(COND)goto done;++r;done:CLEAN;return r;',
 'int r=0;if(COND){CLEAN;return r;}CLEAN;++r;return r;',
 'int r=0;if(COND){CLEAN;goto done;}CLEAN;++r;done:return r;',
 'int r=1;if(COND)r=0;CLEAN;return r;',
 'int r=1;if(COND){CLEAN;r=0;}else{CLEAN;}return r;',
 'int r=0;if(COND){CLEAN;}else{CLEAN;++r;}return r;',
 'int r=0;CLEAN;if(COND)r=0;else r=1;return r;',
 'int r=0;CLEAN;if(COND)goto done;r=1;done:return r;',
]
jobs=[]
for ti,kind,cpu,th in itertools.product(range(len(tails)),['int','bool','unsigned char','unsigned short'],[6,7],range(2)):
 text=common.prefix
 if th:text=text.replace('void *__stdcall LocalFree(void *);','void *__stdcall LocalFree(void *) throw();')
 tail=tails[ti].replace('int r=',kind+' r=').replace('COND',conds[0]).replace('CLEAN','releaseArgs(argv)')
 text+='long status=WinVerifyTrust(0,&action,&data);'+tail+'\n}\n'
 text=text.replace('/G6',f'/G{cpu}')
 jobs.append((f'f{len(jobs):03}',text,dict(tail=ti,kind=kind,cpu=cpu,cleanup_nothrow=th)))
with concurrent.futures.ThreadPoolExecutor(max_workers=8) as pool:results=list(pool.map(common.run,jobs))
(common.OUT/'original-flow.json').write_text(json.dumps(results,indent=2)+'\n')
for r in sorted(results,key=lambda r:len(r.get('diff',[0]*999)))[:5]:print(json.dumps(r))
