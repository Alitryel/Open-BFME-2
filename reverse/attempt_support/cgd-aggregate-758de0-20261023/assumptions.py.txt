from pathlib import Path
from itertools import product
W=Path(__file__).resolve().parent;b=(W/'bank.cpp').read_text();D=W/'assumptions';D.mkdir(exist_ok=True)
for i,(mask,abi,bridge) in enumerate(product(range(512),range(2),range(2))):
 s=b
 if not abi:
  a=s.index('struct PArg {');z=s.index('struct ContactData {');s=s[:a]+s[z:]
  s=s.replace('altPrepare(node->first->source,node->point,ContactVector(node->normal).negate())','node->first->source->prepare(node->point,ContactVector(node->normal).negate())')
 if not bridge:
  s=s.replace('__forceinline void *__cdecl bridge(ContactData *s,const ContactVector &p,const ContactVector &n) { return s->source->prepare(p,n); }\n','')
  s=s.replace('bridge(node->second,node->point,node->normal)','node->second->source->prepare(node->point,node->normal)')
 sites=[('  ContactNode *node=current;','  __assume(current!=0);\n'),
 ('  if(node==0) return;','\n  __assume(node!=0);'),
 ('  if(node->first->source==0 || node->second->source==0) continue;','\n  __assume(node->first->source!=0);'),
 ('  if(node->first->source==0 || node->second->source==0) continue;','\n  __assume(node->second->source!=0);'),
 ('  if(node->state==2) continue;','\n  __assume(node->state!=0);'),
 ('  if(node->state==2) continue;','\n  __assume(node->state!=2);'),
 ('  if(node->state==2) continue;','\n  __assume(&node->point!=0);'),
 ('  if(node->state==2) continue;','\n  __assume(&node->normal!=0);'),
 ('  if(node->first->source==0 || node->second->source==0) continue;','\n  __assume(node->first->source!=0 && node->second->source!=0);')]
 # First site inserts before the node declaration, sites 2/3 only at the initial
 # guard, and the final bit only after the reverse guard. Other facts are after
 # the state-2 guard; all follow from reaching that original program point.
 for bit in range(9):
  if not(mask>>bit&1):continue
  anchor,add=sites[bit]
  if bit==0:s=s.replace(anchor,add+anchor,1)
  elif bit==8:
   p=s.rfind(anchor)+len(anchor);s=s[:p]+add+s[p:]
  else:s=s.replace(anchor,anchor+add,1)
 (D/('v%04d.cpp'%i)).write_text(s)
print('generated',i+1)
