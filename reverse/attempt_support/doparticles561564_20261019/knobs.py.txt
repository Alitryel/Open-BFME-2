import os,sys,re,subprocess,itertools,json
os.chdir(os.path.dirname(os.path.abspath(__file__)))
base=open(os.environ.get('BASE','base.cpp')).read()
E3='     out[i].x=offset[i].x+locs[i].X;\n     out[i].y=offset[i].y+locs[i].Y;\n     out[i].z=offset[i].z+locs[i].Z;'
ADD='Rva00561564AddP(&locs[i],&offset[i],&out[i]);'
CR='   Rva00561564CrossFirst(&axisC,&axisA,&axisB);\n   Rva00561564CrossSecond(&axisA,&axisB,&axisC);'
SETA='axisA.Set(locs[n-1].X-locs[0].X,locs[n-1].Y-locs[0].Y,locs[n-1].Z-locs[0].Z)'
REFE='     out[i].x=offset[i].x=locs[i].X;\n     out[i].y=offset[i].y=locs[i].Y;\n     out[i].z=offset[i].z=locs[i].Z;'
GAT='locs[n].X=p->m_pos.X;locs[n].Y=p->m_pos.Y;locs[n].Z=p->m_pos.Z;'
def build(fl):
    s=base
    # cached loop
    pre='';O='out[i].';F='offset[i].';L='locs[i].'
    if 'co' in fl: pre+='Rva00561C34Point *o=&out[i];';O='o->'
    if 'cf' in fl: pre+='Rva00561C34Point *f=&offset[i];';F='f->'
    if 'cl' in fl: pre+='Vector3 *l=&locs[i];';L='l->'
    body='     '+pre+'\n' if pre else ''
    body+='     %sx=%sx+%sX;\n     %sy=%sy+%sY;\n     %sz=%sz+%sZ;'%(O,F,L,O,F,L,O,F,L)
    s=s.replace(E3,body,1)
    a=ADD
    if 'al' in fl: a='Vector3 *l=&locs[i];'+a.replace('&locs[i]','l')
    if 'ao' in fl: a='Rva00561C34Point *f=&offset[i];'+a.replace('&offset[i]','f')
    if 'ar' in fl: a='Rva00561C34Point *o=&out[i];'+a.replace('&out[i]','o')
    s=s.replace(ADD,a,1)
    for f in fl:
        if f.startswith('X') and len(f)==13:
            # X<6 bits first><6 bits second>: stage flags per product
            def helper(name,bits):
                rows=[('y','z','z','y'),('z','x','x','z'),('x','y','y','x')]
                out=['static inline void %s(const FXParticleSystem::LightningDrawModule::Vec *a,const FXParticleSystem::LightningDrawModule::Vec *b,FXParticleSystem::LightningDrawModule::Vec *r){'%name]
                comp='xyz'
                for ri,(p1a,p1b,p2a,p2b) in enumerate(rows):
                    e1='a->%s*b->%s'%(p1a,p1b); e2='a->%s*b->%s'%(p2a,p2b)
                    if bits[2*ri]=='1': out.append(' float p%d=%s;'%(ri,e1)); e1='p%d'%ri
                    if bits[2*ri+1]=='1': out.append(' float q%d=%s;'%(ri,e2)); e2='q%d'%ri
                    out.append(' r->%s=%s-%s;'%(comp[ri],e1,e2))
                out.append('}')
                return chr(10).join(out)
            hs=helper('CrossK1',f[1:7])+chr(10)+helper('CrossK2',f[7:13])+chr(10)
            s=s.replace('int FXParticleSystem::LightningDrawModule::doParticles',hs+'int FXParticleSystem::LightningDrawModule::doParticles',1)
            s=s.replace('Rva00561564CrossFirst(&axisC','CrossK1(&axisC',1).replace('Rva00561564CrossSecond(&axisA','CrossK2(&axisA',1)
    for f in fl:
        if f[0] in 'PQ' and f[1:].isdigit():
            N=int(f[1:]); tgt='&axisB' if f[0]=='P' else '&axisC'
            old='Rva00561564CrossFirst(&axisC,&axisA,&axisB);' if f[0]=='P' else 'Rva00561564CrossSecond(&axisA,&axisB,&axisC);'
            decl='Vec *t%s0=%s;'%(f[0],tgt)+''.join('Vec *t%s%d=t%s%d;'%(f[0],k,f[0],k-1) for k in range(1,N))
            new=decl+old.replace(tgt+')','t%s%d)'%(f[0],N-1))
            assert old in s
            s=s.replace(old,new,1)
        if f.startswith('u') and f[1:].isdigit():
            N=int(f[1:]); s=s.replace(GAT,''.join('int un%d;'%k for k in range(N))+GAT,1)
    for f in fl:
        if f[0] in 'RC' and f[1:].isdigit():
            N=int(f[1:]); p=f[0]
            if p=='R':
                decl='Vector3 *tR0=&locs[i];'+''.join('Vector3 *tR%d=tR%d;'%(k,k-1) for k in range(1,N))
                s=s.replace(ADD,decl+ADD.replace('&locs[i]','tR%d'%(N-1)),1)
            else:
                decl='Rva00561C34Point *tC0=&offset[i];'+''.join('Rva00561C34Point *tC%d=tC%d;'%(k,k-1) for k in range(1,N))
                s=s.replace(E3,'     '+decl+chr(10)+E3.replace('offset[i].','tC%d->'%(N-1)),1)
    for f in fl:
        if f[0] in 'TJKLM' and f[1:].isdigit():
            N=int(f[1:]); p=f[0]
            if p=='T':
                decl='ShareBufferClass<Vector3> *tT0=g_00E065C8;'+''.join('ShareBufferClass<Vector3> *tT%d=tT%d;'%(k,k-1) for k in range(1,N))
                s=s.replace(' Vector3* locs=g_00E065C8->Get_Array();',' '+decl+'Vector3* locs=tT%d->Get_Array();'%(N-1),1)
            else:
                LOOPHEAD='Rva00561C34Point *out=pointsA[j]; Rva00561C34Point *offset=pointsB[j];'
                i0=s.index(LOOPHEAD)
                at = s.index(LOOPHEAD,i0+1) if p in 'JK' else i0
                arr='pointsB' if p in 'JL' else 'pointsA'
                decl='Rva00561C34Point *t%s0=%s[j];'%(p,arr)+''.join('Rva00561C34Point *t%s%d=t%s%d;'%(p,k,p,k-1) for k in range(1,N))
                if arr=='pointsB': new='Rva00561C34Point *out=pointsA[j]; '+decl+'Rva00561C34Point *offset=t%s%d;'%(p,N-1)
                else: new=decl+'Rva00561C34Point *out=t%s%d; Rva00561C34Point *offset=pointsB[j];'%(p,N-1)
                s=s[:at]+new+s[at+len(LOOPHEAD):]
    CS='axisC.x=0;axisC.y=0;axisC.z=1;'
    if 'w1' in fl: s=s.replace(CS,'axisC.x=0;axisC.y=0;float &cz=axisC.z;cz=1;',1)
    if 'w2' in fl: s=s.replace(CS,'axisC.x=axisC.y=0;axisC.z=1;',1)
    if 'w3' in fl: s=s.replace(CS,'Vec &C=axisC;C.x=0;C.y=0;C.z=1;',1).replace('CrossFirst(&axisC,','CrossFirst(&C,',1).replace(',&axisC);',',&C);',1)
    if 'w4' in fl: s=s.replace(CS,'axisC.x=0;axisC.y=0;float *cz=&axisC.z;*cz=1;',1)
    if 'w5' in fl: s=s.replace(CS,'Vec *C=&axisC;C->x=0;C->y=0;C->z=1;',1).replace('CrossFirst(&axisC,','CrossFirst(C,',1).replace(',&axisC);',',C);',1)
    if 'cset' in fl: s=s.replace('axisC.x=0;axisC.y=0;axisC.z=1;','axisC.Set(0,0,1);',1)
    if 'rfirst' in fl:
        for nm in ('Rva00561564CrossFirst','Rva00561564CrossSecond'):
            i=s.index('static inline void '+nm+'(')
            j=s.index('{',i)
            sig=s[i:j]
            new=sig.replace('(const FXParticleSystem::LightningDrawModule::Vec *a,const FXParticleSystem::LightningDrawModule::Vec *b,FXParticleSystem::LightningDrawModule::Vec *r)','(FXParticleSystem::LightningDrawModule::Vec *r,const FXParticleSystem::LightningDrawModule::Vec *a,const FXParticleSystem::LightningDrawModule::Vec *b)')
            assert new!=sig
            s=s[:i]+new+s[j:]
        s=s.replace('Rva00561564CrossFirst(&axisC,&axisA,&axisB);','Rva00561564CrossFirst(&axisB,&axisC,&axisA);',1)
        s=s.replace('Rva00561564CrossSecond(&axisA,&axisB,&axisC);','Rva00561564CrossSecond(&axisC,&axisA,&axisB);',1)
    if 'pb' in fl: s=s.replace(CR,'   Vec *pb=&axisB;'+chr(10)+CR.replace('&axisB','pb'),1)
    if 'pa' in fl: s=s.replace(CR,'   Vec *pa=&axisA;\n'+CR.replace('&axisA','pa'),1)
    if 'pc' in fl: s=s.replace(CR,'   Vec *pc=&axisC;\n'+CR.replace('&axisC','pc'),1)
    if 'last' in fl: s=s.replace(SETA,'Vector3 *last=&locs[n-1];'+SETA.replace('locs[n-1].','last->'),1)
    if 're' in fl: s=s.replace(REFE,'     Vector3 *l=&locs[i];\n'+REFE.replace('locs[i].','l->'),1)
    for f in fl:
        if f.startswith('g') and f[1:].isdigit():
            N=int(f[1:]); decl='Vector3 *d0=&locs[n];'+''.join('Vector3 *d%d=d%d;'%(k,k-1) for k in range(1,N))
            s=s.replace(GAT,decl+GAT.replace('locs[n].','d%d->'%(N-1)),1)
    if 'gat' in fl: s=s.replace(GAT,'Vector3 *d=&locs[n];'+GAT.replace('locs[n].','d->'),1)
    return s
SITES={'BxP1':0x5617d1,'BxP2':0x5617de,'X1':0x5617fd,'X2':0x561805,'X3':0x56181e,'X4':0x56182b,'Y1':0x561848,'Y2':0x561861,'Y3':0x56186e,'CzP1':0x561886,'Y4':0x56188a,'AA':0x561984,'CC':0x561a9a,'SetA':0x561776}
def run(name,s):
    open('k/'+name+'.cpp','w').write(s)
def main():
    os.makedirs('k',exist_ok=True)
    combos=[x.split(',') if x else [] for x in sys.argv[1:]]
    names=[]
    for fl in combos:
        nm='k_'+('_'.join(fl) or 'none')
        run(nm,build(fl)); names.append(nm)
    if not os.environ.get('NOCOMP'): subprocess.call(['sh','-c','printf "%s\n" '+' '.join('build/nbr/lt/b3/k/'+n+'.cpp' for n in names)+' | xargs -P 10 -n 1 sh run.sh > k/out.txt'])
    res={}
    for l in open('k/out.txt'):
        res[l.split()[0]]=l.rstrip()
    for n in names:
        l=res[n]; 
        sig=l.split('|')[-1].split()
        ad=set(int(x,16) for x in sig)
        bad=[k for k,v in SITES.items() if v in ad]
        other=len([x for x in ad if x not in SITES.values()])
        print('%-30s %s  wrong=%s'%(n,l.split('|')[0][14:].strip()[:30],' '.join(bad)))
main()
