from search import *
from context import getters, mutators, event_hierarchy
import itertools

calls=['(BfmeItemE63 *)foundation->findModule(memberKey)',
       '(BfmeItemE63 *)foundation->findModule(memberKey2)',
       '(BfmeItemE63 *)obj->findModule(memberKey)',
       '(BfmeItemE63 *)getObject()->findModule(memberKey)']
receivers=['foundation','foundation','obj','getObject()']
keys=['memberKey','memberKey2','memberKey','memberKey']

def wrap(s,form,mask):
    # Typed inline module accessors preserve the exact protected lookup call.
    defs=[
      'BfmeItemE63 *findCastleMember(NameKeyType key) const {return (BfmeItemE63*)findModule(key);}',
      'BfmeItemE63 *findCastleMember(const NameKeyType &key) const {return (BfmeItemE63*)findModule(key);}',
      'template<class T> T *findTypedModule(NameKeyType key) const {return (T*)findModule(key);}',
      'BfmeItemE63 *findCastleMember(NameKeyType key) {return (BfmeItemE63*)findModule(key);}',
      'static BfmeItemE63 *findCastleMember(const Object *o, NameKeyType key) {return (BfmeItemE63*)o->findModule(key);}',
      'static BfmeItemE63 *findCastleMember(NameKeyType key, const Object *o) {return (BfmeItemE63*)o->findModule(key);}',
      'static BfmeItemE63 *findCastleMember(Object &o, const NameKeyType &key) {return (BfmeItemE63*)o.findModule(key);}',
      'BfmeItemE63 *findCastleMember(NameKeyType key) const {Module *m=findModule(key); return static_cast<BfmeItemE63*>(m);}',
    ]
    s=s.replace('BodyModuleInterface *getBodyModule() const',defs[form]+'\n BodyModuleInterface *getBodyModule() const',1)
    for i,(call,o,k) in enumerate(zip(calls,receivers,keys)):
      if not mask&(1<<i):continue
      replacement=f'{o}->findCastleMember({k})'
      if form==2:replacement=f'{o}->findTypedModule<BfmeItemE63>({k})'
      elif form==4:replacement=f'Object::findCastleMember({o}, {k})'
      elif form==5:replacement=f'Object::findCastleMember({k}, {o})'
      elif form==6:replacement=f'Object::findCastleMember(*{o}, {k})'
      s=s.replace(call,replacement,1)
    return s

tests={}
for form,mask,context in itertools.product(range(8),range(1,16),range(4)):
 s=wrap(BASE,form,mask)
 if context&1:s=getters(s)
 if context&2:s=mutators(s)
 tests[f'wrapper-{form}-{mask:02d}-{context}']=s
run(tests,'wrappers')
