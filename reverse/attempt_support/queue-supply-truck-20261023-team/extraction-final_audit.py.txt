def cfg_restore(A,calls):
 byaddr={i.address:i for i in A};following={a.address:b.address for a,b in zip(A,A[1:])};callat={int(c['offset'],16):c['kind'] for c in calls}
 entry=A[0].address;todo=[(entry,False,False)];seen=set();errors=[];restores=set();exits=set()
 while todo:
  addr,active,alive=todo.pop()
  if (addr,active,alive) in seen:continue
  seen.add((addr,active,alive));i=byaddr[addr]
  if i.mnemonic=='mov' and i.op_str.startswith('byte ptr [') and '0x338]' in i.op_str:
   if i.op_str.endswith(', 1'):
    if active:errors.append((hex(addr),'enable while restore outstanding'))
    active=True
   else:
    if not active:errors.append((hex(addr),'restore without enable'))
    if alive:errors.append((hex(addr),'restore before string destruction'))
    active=False;restores.add(addr)
  call=callat.get(addr,'')
  if call=='direct:??0?$StringBase@D@@AAE@PBD@Z':alive=True
  if 'releaseBuffer@?$StringBase@D' in call:alive=False
  if i.mnemonic=='ret':
   exits.add((hex(addr),active,alive))
   if active:errors.append((hex(addr),'normal return without restore'))
   if alive:errors.append((hex(addr),'normal return before string destruction'))
   continue
  nxt=[]
  if i.mnemonic.startswith('j') and i.op_str.startswith('0x'):
   nxt.append(int(i.op_str,16))
   if i.mnemonic!='jmp':nxt.append(following[addr])
  else:nxt.append(following[addr])
  for dst in nxt:
   if dst not in byaddr:errors.append((hex(addr),'branch outside body '+hex(dst)))
   else:todo.append((dst,active,alive))
 return dict(errors=errors,restore_addresses=list(map(hex,sorted(restores))),exits=list(exits),visited_state_count=len(seen),scope='Normal CFG only; calls return normally; no exception edges or runtime validation.')
