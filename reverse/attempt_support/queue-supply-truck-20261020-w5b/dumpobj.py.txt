import sys
from pathlib import Path
sys.path.insert(0, str(Path('tools').resolve()))
import build, capstone
from locate import object_functions
# usage: dumpobj.py <src> [substring]
src = build.ROOT / sys.argv[1]
obj = build.obj_path(src)
if obj.exists():
    obj.unlink()
build.compile_source(src, obj)
md = capstone.Cs(capstone.CS_ARCH_X86, capstone.CS_MODE_32)
for n, s, r in object_functions(obj):
    if len(sys.argv) > 2 and sys.argv[2] not in n:
        continue
    print('==', n, len(s.rstrip(b'\xcc')))
    for i in md.disasm(s.rstrip(b'\xcc'), 0):
        print(f'  {i.address:04x} {i.mnemonic} {i.op_str}')
