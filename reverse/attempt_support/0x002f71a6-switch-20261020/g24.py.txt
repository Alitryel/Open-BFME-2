import itertools
from pathlib import Path
b = Path('sw2.cpp').read_text(encoding='latin-1')
NAME = "Path *Pathfinder::internal_findHierarchicalPath("
DEF = ("struct HPD : public Coord3D { HPD(){}\n HPD(const HPD &c) { x=c.x;y=c.y;z=c.z; }\n HPD(const Coord3D &c) { x=c.x;y=c.y;z=c.z; }\n"
       " HPD &operator=(const Coord3D &c) { *(Coord3D *)this=c;return *this; }\n ~HPD(){}\n};\n")
L = {'a': " HierarchyPoint adjustTo(*rawTo);", 'c': " HierarchyPoint clipFrom(*from);", 'o': " HierarchyPoint originalTo(adjustTo);", 'd': "     HierarchyPoint delta(way->location);"}
for s in L.values(): assert b.count(s) == 1
for bits in itertools.product((0,1), repeat=4):
    if not any(bits): continue
    u = b.replace(NAME, DEF + NAME, 1)
    for on, (k, s) in zip(bits, L.items()):
        if on: u = u.replace(s, s.replace('HierarchyPoint', 'HPD'))
    Path('v24/' + ''.join(k if on else '_' for on, k in zip(bits, L)) + '.cpp').write_text(u, encoding='latin-1', newline='')
