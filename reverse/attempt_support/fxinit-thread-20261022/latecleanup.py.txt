import sys,json,re,collections
from pathlib import Path
sys.path.insert(0,'build/netpacket-20261022e');from wdbg import Dbg
src,obj=sys.argv[1:3];d=Dbg(src,obj,obj+'.debug.log')
d.prompt();d.prompt('set $CanDeferOnBPByAddr = 1');d.prompt('break *0x10760539')
phases=[0x952ef,0x95308,0x95390,0x953a0,0x953a6,0x953ba,0x953c0,0x95407,0x95580,0x9558c,0x955f1,0x752f3]
on=False;done=False;bps={};snaps=[]
while True:
 out=d.prompt('cont',60)
 if 'terminated' in out: break
 ip=d.reg('eip')-0x10700000
 if ip==0x60539:
  p=d.reg('eax');f=d.words(p,1)[0];name=d.string(d.words(f+4,1)[0])
  if '?Init@FXShaderGeometry' in name and not done:
   on=True;fn=p
   for a in phases:
    s=d.prompt('break *0x%x'%(a+0x10700000));bps[a]=int(re.search(r'Breakpoint (\d+)',s)[1])
  continue
 if not on:continue
 sn={'ip':hex(ip),'func':fn,'fw':d.words(fn,48),'blocks':{},'tuples':{}}
 root=d.words(sn['fw'][2],3);p=root[0];seen=set()
 while p and p not in seen and len(seen)<400:
  seen.add(p);b=d.words(p,48);q=b[2];pr=[];ps=set()
  while q and q not in ps and len(ps)<20:
   ps.add(q);v=d.words(q,3);pr.append(v);q=v[1]
  s=b[7];last=b[8];order=[];ss=set()
  while s and s not in ss and len(ss)<4000:
   ss.add(s);t=d.words(s,16);sn['tuples'][s]=t;order.append(s)
   if s==last:break
   s=t[0]
  sn['blocks'][p]={'w':b,'preds':pr,'order':order};p=b[0]
 snaps.append(sn);Path(obj+'.phases.json').write_text(json.dumps(snaps))
 hist=collections.Counter(w[1]for w in sn['tuples'].values());print(hex(ip),'blocks',len(sn['blocks']),'tuples',len(sn['tuples']),'ops',hist.most_common(8),flush=True)
 if ip==0x752f3:
  for n in bps.values():d.prompt('disable %d'%n)
  on=False;done=True
print('DONE',flush=True)
