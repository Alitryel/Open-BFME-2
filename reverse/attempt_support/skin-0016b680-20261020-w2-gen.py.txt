import sys, os, re
from pathlib import Path
HERE = Path(__file__).resolve().parent
BASE = (HERE / os.environ.get('BASE', 't00_bank.cpp')).read_text()
OUT = HERE / 'v'
OUT.mkdir(exist_ok=True)
NL = '\n'

BLEND_OLD = BASE[BASE.index('static __forceinline void skinBlend'):BASE.index('class MeshModelClass')]
SINGLE_OLD = BASE[BASE.index('static __forceinline void skinTransform'):BASE.index('static __forceinline void skinBlend')]
ROWS = ['X', 'Y', 'Z']


def blend(rowfmt):
    s = 'static __forceinline void skinBlend(const BfmeSkinMatrix &a,const Vector3 &v,const BfmeSkinMatrix &b,const Vector3 &w,Vector3 &out) {' + NL
    for r, c in enumerate(ROWS):
        s += ' out.%s=%s;' % (c, rowfmt(r)) + NL
    return s + '}' + NL


def plain(r):
    return 'a[{r}][0]*v.X+a[{r}][1]*v.Y+a[{r}][2]*v.Z+a[{r}][3]+b[{r}][0]*w.X+b[{r}][1]*w.Y+b[{r}][2]*w.Z+b[{r}][3]'.format(r=r)


WBX = [' float x=Q.W*v.X+(Q.Y*v.Z-v.Y*Q.Z);', ' float y=Q.W*v.Y-(Q.X*v.Z-v.X*Q.Z);', ' float z=Q.W*v.Z+(Q.X*v.Y-v.X*Q.Y);', ' float w=-(Q.X*v.X+Q.Y*v.Y+Q.Z*v.Z);']
WBO = [' out.X=x*Q.W-w*Q.X+(z*Q.Y-y*Q.Z)+T.X;', ' out.Y=y*Q.W-w*Q.Y-(z*Q.X-x*Q.Z)+T.Y;', ' out.Z=z*Q.W-w*Q.Z+(y*Q.X-x*Q.Y)+T.Z;']
WBOP = [' out.X=(x*Q.W-w*Q.X+(z*Q.Y-y*Q.Z)+T.X);', ' out.Y=(y*Q.W-w*Q.Y-(z*Q.X-x*Q.Z)+T.Y);', ' out.Z=(z*Q.W-w*Q.Z+(y*Q.X-x*Q.Y)+T.Z);']
HDR = 'static __forceinline void skinTransform(const BfmeSkinTransform &t,const Vector3 &v,Vector3 &out) {' + NL


def single(q, body):
    pre = ' const Quaternion &q=t.Rotation;' + NL if q == 'q' else ''
    Q = 'q' if q == 'q' else 't.Rotation'
    return HDR + pre + NL.join(WBX + body).replace('Q.', Q + '.').replace('T.', 't.Translation.') + NL + '}' + NL


SINGLE = {'sq': single('q', WBO), 'sr': single('r', WBO), 'sqp': single('q', WBOP), 'srp': single('r', WBOP)}


def tpa(body):
    b = NL.join('  ' + l for l in WBX + body).replace('Q.', 'Rotation.').replace('T.', 'Translation.').replace('v.', 'v->').replace('out.', 'out->')
    return SINGLE_OLD + 'inline void BfmeSkinTransform::TransformPositionArray(const Vector3 *v,Vector3 *out,int count) const {' + NL + ' while(count--) {' + NL + b + NL + '  v++;out++;' + NL + ' }' + NL + '}' + NL


SINGLE_CALL = '   Vector3 *v=source[0],*out=dst;' + NL + '   int n=count;' + NL + '   while(n--) {skinTransform(t,*v++,*out++);}'


def apply(s, knobs):
    for k in knobs:
        if k.startswith('ch') and k[2:].isdigit():
            n = int(k[2:])
            if n:
                decl = 'const HTreeClass *t0=tree;' + ''.join('const HTreeClass *t%d=t%d;' % (i, i - 1) for i in range(1, n))
                s = s.replace(' int streams=1;' + NL, ' ' + decl + NL + ' int streams=1;' + NL, 1)
                s = s.replace('((const BfmeSkinTreeView*)tree)', '((const BfmeSkinTreeView*)t%d)' % (n - 1))
        elif k.startswith('cb') and k[2:].isdigit():
            n = int(k[2:])
            if n:
                decl = 'Vector3 *c0=source[0];' + ''.join('Vector3 *c%d=c%d;' % (i, i - 1) for i in range(1, n))
                old = '   Vector3 *v=source[0],*w=source[1],*out=dst;'
                assert old in s
                s = s.replace(old, '   ' + decl + NL + '   Vector3 *v=c%d,*w=source[1],*out=dst;' % (n - 1), 1)
        elif k == 'bo':
            s = s.replace(BLEND_OLD, blend(lambda r: '(' + plain(r) + ')'))
        elif k == 'bf':
            s = s.replace(BLEND_OLD, blend(lambda r: '(float)(' + plain(r) + ')'))
        elif k == 'bg':
            s = s.replace(BLEND_OLD, blend(lambda r: '(a[{r}][0]*v.X+a[{r}][1]*v.Y+a[{r}][2]*v.Z+a[{r}][3])+(b[{r}][0]*w.X+b[{r}][1]*w.Y+b[{r}][2]*w.Z+b[{r}][3])'.format(r=r)))
        elif k == 'bret':
            orders = [['a00', 'a01', 'b00', 'b01', 'b02', 'a02'], ['a11', 'b11', 'b12', 'a10', 'b10', 'a12'], ['a21', 'b21', 'b22', 'a20', 'b20', 'a22']]
            def f(r):
                terms = ['%s[%s][%s]*%s.%s' % (t[0], t[1], t[2], 'v' if t[0] == 'a' else 'w', 'XYZ'[int(t[2])]) for t in orders[r]]
                e = terms[0]
                for t in terms[1:]:
                    e = '(%s+%s)' % (e, t)
                return '(%s+b[%d][3])+a[%d][3]' % (e, r, r)
            s = s.replace(BLEND_OLD, blend(f))
        elif k == 'mp':
            s = re.sub(r'=Rotation\.(\w)\*Rotation\.(\w)\*2\.0f;', r'=(Rotation.\1*Rotation.\2)*2.0f;', s)
        elif k == 'mf':
            s = re.sub(r'=Rotation\.(\w)\*Rotation\.(\w)\*2\.0f;', r'=(float)(Rotation.\1*Rotation.\2*2.0f);', s)
        elif k in SINGLE:
            s = s.replace(SINGLE_OLD, SINGLE[k])
        elif k in ('mwb', 'mwbp'):
            s = s.replace(SINGLE_OLD, tpa(WBO if k == 'mwb' else WBOP))
            s = s.replace(' Quaternion Rotation;Vector3 Translation;', ' void TransformPositionArray(const Vector3 *v,Vector3 *out,int count) const;' + NL + ' Quaternion Rotation;Vector3 Translation;')
            assert SINGLE_CALL in s
            s = s.replace(SINGLE_CALL, '   t.TransformPositionArray(source[0],dst,count);')
        elif k in ('mwbf', 'mwbfp'):
            s = s.replace(SINGLE_OLD, tpa(WBO if k == 'mwbf' else WBOP).replace('inline void BfmeSkinTransform::TransformPositionArray', '__forceinline void BfmeSkinTransform::TransformPositionArray'))
            s = s.replace(' Quaternion Rotation;Vector3 Translation;', ' void TransformPositionArray(const Vector3 *v,Vector3 *out,int count) const;' + NL + ' Quaternion Rotation;Vector3 Translation;')
            assert SINGLE_CALL in s
            s = s.replace(SINGLE_CALL, '   t.TransformPositionArray(source[0],dst,count);')
        elif k == 'si':  # one int i shared by both stream loops (WB local_4)
            s = s.replace(' Vector3 *source[2];' + NL, ' int i;' + NL + ' Vector3 *source[2];' + NL, 1)
            s = s.replace(' for(int k=0;k<streams;++k) {if(vertices[k])source[k]=vertices[k]->Get_Array();else source[k]=0;}', ' for(i=0;i<streams;++i) {if(vertices[i])source[i]=vertices[i]->Get_Array();else source[i]=0;}')
            s = s.replace('  for(int k=0;k<streams;++k)source[k]+=count;', '  for(i=0;i<streams;++i)source[i]+=count;')
        elif k in ('mb', 'mbp'):  # blend as WB member of the first matrix with the loop inside
            rows = []
            for r, c in enumerate(ROWS):
                e = 'v[{r}][0]*in0->X+v[{r}][1]*in0->Y+v[{r}][2]*in0->Z+v[{r}][3]+tm1.v[{r}][0]*in1->X+tm1.v[{r}][1]*in1->Y+tm1.v[{r}][2]*in1->Z+tm1.v[{r}][3]'.format(r=r)
                if k == 'mbp':
                    e = '(' + e + ')'
                rows.append('  out->%s=%s;' % (c, e))
            helper = 'void BfmeSkinMatrix::Blend(const Vector3 *in0,const BfmeSkinMatrix &tm1,const Vector3 *in1,Vector3 *out,int count) const {' + NL + ' while(count--) {' + NL + NL.join(rows) + NL + '  in0++;in1++;out++;' + NL + ' }' + NL + '}' + NL
            s = s.replace(BLEND_OLD, '__forceinline ' + helper)
            s = s.replace('__forceinline const float *operator[](int i) const {return v[i];} };', '__forceinline const float *operator[](int i) const {return v[i];} void Blend(const Vector3 *in0,const BfmeSkinMatrix &tm1,const Vector3 *in1,Vector3 *out,int count) const; };')
            old = '   Vector3 *v=source[0],*w=source[1],*out=dst;' + NL + '   int n=count;' + NL + '   while(n--) {skinBlend(a,*v++,b,*w++,*out++);}'
            assert old in s
            s = s.replace(old, '   a.Blend(source[0],b,source[1],dst,count);')
        elif k == 'vr':  # Vector4-like rows: scale loop as WB tm0[j]*=w0; tm1[j]*=w1;
            old = 'struct BfmeSkinMatrix { float v[3][4]; __forceinline float *operator[](int i) {return v[i];} __forceinline const float *operator[](int i) const {return v[i];}'
            assert old in s
            row = 'struct BfmeSkinRow { float X,Y,Z,W; __forceinline float &operator[](int i) {return (&X)[i];} __forceinline const float &operator[](int i) const {return (&X)[i];} __forceinline BfmeSkinRow &operator*=(float k) {X*=k;Y*=k;Z*=k;W*=k;return *this;} };' + NL
            s = s.replace(old, row + 'struct BfmeSkinMatrix { BfmeSkinRow v[3]; __forceinline BfmeSkinRow &operator[](int i) {return v[i];} __forceinline const BfmeSkinRow &operator[](int i) const {return v[i];}')
            old = '   for(int r=0;r<3;++r) {a[r][0]*=firstWeight;a[r][1]*=firstWeight;a[r][2]*=firstWeight;a[r][3]*=firstWeight;b[r][0]*=secondWeight;b[r][1]*=secondWeight;b[r][2]*=secondWeight;b[r][3]*=secondWeight;}'
            assert old in s
            s = s.replace(old, '   for(int r=0;r<3;++r) {a[r]*=firstWeight;b[r]*=secondWeight;}')
            s = s.replace('tm1.v[0][', 'tm1.v[0][').replace('v[{r}]', 'v[{r}]')
        elif k == 'n1':  # the tree view held in a local, as WB's htree
            s = s.replace(' int streams=1;' + NL, ' const BfmeSkinTreeView *htree=(const BfmeSkinTreeView*)tree;' + NL + ' int streams=1;' + NL, 1)
            s = s.replace('((const BfmeSkinTreeView*)tree)', 'htree')
        elif k == 'n1b':  # the view local declared just before the loop
            s = s.replace(' int total=Get_Vertex_Count();', ' const BfmeSkinTreeView *htree=(const BfmeSkinTreeView*)tree;' + NL + ' int total=Get_Vertex_Count();', 1)
            s = s.replace('((const BfmeSkinTreeView*)tree)', 'htree')
        elif k in ('n3', 'n4'):
            idx = 'first' if k == 'n3' else 'second'
            mat = 'a' if k == 'n3' else 'b'
            old = '   ((const BfmeSkinTreeView*)tree)->Get_Transform(%s).Get_Matrix3D(%s);' % (idx, mat)
            assert old in s, k
            s = s.replace(old, '   const BfmeSkinTransform &t%s=((const BfmeSkinTreeView*)tree)->Get_Transform(%s);t%s.Get_Matrix3D(%s);' % (mat, idx, mat, mat))
        elif k == 'n5':  # single path calls through the accessor without the reference local
            old = '   const BfmeSkinTransform &t=((const BfmeSkinTreeView*)tree)->Get_Transform(first);'
            assert old in s
            s = s.replace(old, '')
            s = s.replace('   t.TransformPositionArray(source[0],dst,count);', '   ((const BfmeSkinTreeView*)tree)->Get_Transform(first).TransformPositionArray(source[0],dst,count);')
        elif k == 'ct':  # empty inline default constructor on the matrix (WB Matrix3D ctor calls)
            old = 'struct BfmeSkinMatrix { BfmeSkinRow v[3];'
            assert old in s
            s = s.replace(old, old + ' __forceinline BfmeSkinMatrix() {}')
        elif k == 'ctr':
            old = 'struct BfmeSkinRow { float X,Y,Z,W;'
            assert old in s
            s = s.replace(old, old + ' __forceinline BfmeSkinRow() {}')
        elif k == 'bm':  # blend reads rows through members as WB 0xA0ECA0 (no operator[] calls)
            for r in range(3):
                for c, m in enumerate('XYZW'):
                    s = s.replace('tm1.v[%d][%d]' % (r, c), 'tm1.v[%d].%s' % (r, m)).replace('=v[%d][%d]' % (r, c), '=v[%d].%s' % (r, m)).replace('+v[%d][%d]' % (r, c), '+v[%d].%s' % (r, m))
        elif k == 'gp':  # Get_Matrix3D element expressions in outer parentheses
            for e in ['1.0f-yy-zz', 'xy-zw', 'xz+yw', 'xy+zw', '1.0f-zz-xx', 'yz-xw', 'xz-yw', 'yz+xw', '1.0f-yy-xx']:
                assert ('=' + e + ';') in s, e
                s = s.replace('=' + e + ';', '=(' + e + ');')
        elif k == 'tp':  # TPA x,y,z,w in outer parentheses
            for v_ in ['x', 'y', 'z']:
                s = re.sub(r'(  float %s=)(Rotation\.W\*v->.\+?-?\(.*?\));' % v_, r'\1(\2);', s)
        elif k == 'pp':  # products in Get_Matrix3D written 2.0f*A*B
            s = re.sub(r'=Rotation\.(\w)\*Rotation\.(\w)\*2\.0f;', r'=2.0f*Rotation.\1*Rotation.\2;', s)
        else:
            raise SystemExit('unknown knob ' + k)
    return s


if __name__ == '__main__':
    pre = os.environ.get('PRE', '')
    for spec in sys.argv[1:]:
        knobs = [x for x in spec.split(',') if x]
        name = pre + ('_'.join(knobs) or 'none')
        s = apply(BASE, knobs)
        (OUT / (name + '.cpp')).write_text(s)
        print('build/w2-16b680/v/%s.cpp' % name)
