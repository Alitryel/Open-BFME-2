from pathlib import Path
from itertools import product
W=Path(__file__).resolve().parent
base=(W/'base.cpp').read_text()
out=W/'compound2';out.mkdir(exist_ok=True)
for index,(kind,copy,align,shape,ret) in enumerate(product(range(6),range(2),range(2),range(8),range(2))):
 s=base
 if kind<4:
  ty=['const ContactVector *','const ContactVector &','unsigned','const void *'][kind]
  cv=['&p','p','(unsigned)&p','&p'][kind]
  nv=['&n','n','(unsigned)&n','&n'][kind]
  pair='struct '+('__declspec(align(8)) ' if align else '')+'ContactPair { '+ty+' point; '+ty+' normal; ContactPair(const ContactVector &p,const ContactVector &n) : point('+cv+'),normal('+nv+') {}'
  if copy:pair+=' ContactPair(const ContactPair &v) : point(v.point),normal(v.normal) {}'
  pair+=' };\n'
  param='ContactPair'
  construct='ContactPair(p,n)'
  direct=lambda p,n:'ContactPair('+p+','+n+')'
 else:
  pair='typedef '+('unsigned ' if kind==4 else '')+'__int64 ContactPair;\n'
  param='ContactPair'
  construct='(ContactPair)(unsigned)&p | ((ContactPair)(unsigned)&n<<32)'
  if copy:construct='((ContactPair)(unsigned)&n<<32) + (unsigned)&p'
  direct=lambda p,n:'((ContactPair)(unsigned)&('+p+') '+('+' if copy else '|')+' ((ContactPair)(unsigned)&('+n+')<<32))'
 if ret:
  rty='unsigned';cast='(void *)'
 else:rty='void *';cast=''
 decl='virtual '+rty+'preparePair('+param+' pair)=0;\n void *prepare(const ContactVector &p,const ContactVector &n) { return '+cast+'preparePair('+construct+'); }'
 # Ensure a space between an integer return type and the name.
 decl=decl.replace('unsignedpreparePair','unsigned preparePair')
 s=s.replace('struct ContactSource {',pair+'struct ContactSource {')
 s=s.replace('virtual void *prepare(const ContactVector &point, const ContactVector &normal)=0;',decl)
 calls=[('node->second->source','node->point','node->normal'),('node->second->source','node->point','node->normal'),('node->first->source','node->point','ContactVector(node->normal).negate()')]
 # Give each of the three call sites its own direct versus inline-adapter choice.
 pos=0
 for site,(recv,p,n) in enumerate(calls):
  old=recv+'->prepare('+p+','+n+')'
  at=s.find(old,pos)
  new=(cast+recv+'->preparePair('+direct(p,n)+')') if shape&(1<<site) else old
  s=s[:at]+new+s[at+len(old):];pos=at+len(new)
 (out/('v%04d.cpp'%index)).write_text(s)
print('generated',index+1)
