from pathlib import Path
import sys,subprocess,json,concurrent.futures,difflib,itertools,time,hashlib
sys.path.insert(0,str(Path('tools').resolve()))
import build,capstone
from variant import resolve_bytes
ROOT=Path.cwd();OUT=ROOT/'build/trust237-round5';RVA=0x237d3d;SIZE=226
NAME='?Rva00237D3DCheck@@YAHXZ'
TARGET=build.read_target_bytes(RVA,SIZE)
base=(OUT/'base.cpp').read_text();base=base[base.index('// cl'):]
marker='    WinVerifyTrust(0,&action,&data);\n    LocalFree(argv);\n    return 1;'
assert marker in base
md=capstone.Cs(capstone.CS_ARCH_X86,capstone.CS_MODE_32)
retins=[i.mnemonic for i in md.disasm(TARGET,0)]
PRED=[
 '(bool)(u|1u)',
 '(unsigned char)(u|1u)!=0',
 '(signed char)(u|1u)!=0',
 '(unsigned short)(u|1u)!=0',
 '(short)(u|1u)!=0',
 '((u&~1u)+1u)!=0',
 '(u^(u+1u))!=0',
 '(u-(u^1u))!=0',
 '((u&1u)+((u&1u)^1u))==1u',
 '(u+1u)>u || u==0xffffffffu',
 '((u|(0u-u))>>31)!=0 || u==0u',
 'u+(u^0xffffffffu)==0xffffffffu',
 '(u&0x7fffffffu)<=0x7fffffffu',
 '(unsigned char)u<=255u',
 '((u|1u)&0xffffu)!=0',
 '((u|1u)&255u)!=0',
 '((u&1u)==0u) | ((u&1u)==1u)',
 '(bool)(u&1u) != (bool)((u&1u)^1u)',
 '((u&1u)+!(u&1u))!=0',
 '((u|1u)^0u)!=0',
 '!!(unsigned short)(u|1u)',
 '((u|1u)/1u)!=0',
 '((u|1u)%2u)==1u',
 '((u|1u)&1u)==1u',
]
SHAPES=[
 'return P;',
 'bool p=P; return p;',
 'unsigned char p=(unsigned char)(P); return p;',
 'const bool &p=P; return p;',
 'bool p=P; return (int)p;',
 'int r=0; if(P) ++r; return r;',
 'switch((int)(P)){case 0:return 0; case 1:return 1; default:return 0;}',
 'for(unsigned int i=(P);i;--i) return 1; return 0;',
]
def source(pred,shape,timing,helper,flags='/O1 /G7 /arch:SSE',debug=''):
    helpers=''
    expr=pred
    if helper:
        directive='__forceinline' if helper==1 else 'inline'
        helpers=directive+' bool latePredicate(unsigned long u) { return '+pred+'; }\n'
        expr='latePredicate(u)'
    consume=shape.replace('P',expr)
    if timing:
        # Store the predicate across LocalFree in a narrow, rather than long, value.
        consume='bool held='+expr+'; LocalFree(argv); '+shape.replace('P','held')
    else:consume='LocalFree(argv); '+consume
    text=base.replace(marker,'    long status=WinVerifyTrust(0,&action,&data);\n    unsigned long u=(unsigned long)status;\n    '+consume)
    text=text.replace('int Rva00237D3DCheck() {',helpers+'int Rva00237D3DCheck() {')
    text=text.replace('/O1 /G7 /arch:SSE',flags+' '+debug)
    return text

def run(job):
    name,text,meta=job
    src=OUT/(name+'.cpp');obj=src.with_suffix('.obj');src.write_text(text)
    try:
        cmd,env=build.compiler_command(src,obj)
        cmd.insert(-1,"-Fd"+str(obj.with_suffix(".pdb").relative_to(ROOT)))
        p=subprocess.run(cmd,env=env,cwd=ROOT,capture_output=True,text=True)
        if p.returncode:return dict(name=name,meta=meta,error=(p.stdout+p.stderr)[-1200:])
        raw,rel=build.read_object_symbol_bytes(obj,NAME,code_only=True)
        resolved,unres=resolve_bytes(raw,rel,TARGET+b'\0'*max(0,len(raw)-SIZE),RVA,len(raw),{'_memset':[0x6291ae],'_WinVerifyTrust@12':[0x655652]})
        ins=[i.mnemonic for i in capstone.Cs(capstone.CS_ARCH_X86,capstone.CS_MODE_32).disasm(raw,0)]
        sm=difflib.SequenceMatcher(a=ins,b=retins,autojunk=False)
        eq=sum(x[2]-x[1] for x in sm.get_opcodes() if x[0]=='equal')
        prefix=next((i for i,(a,b) in enumerate(zip(resolved,TARGET)) if a!=b),min(len(raw),SIZE))
        return dict(name=name,meta=meta,size=len(raw),diff=sum(a!=b for a,b in itertools.zip_longest(resolved,TARGET)),prefix=prefix,eq=eq,regions=sum(x[0]!='equal' for x in sm.get_opcodes()),nop16=b'\x90'*16 in raw,nops=sum(i=='nop' for i in ins),tail=raw[-48:].hex(),sha256=hashlib.sha256(raw).hexdigest(),unresolved=unres)
    except BaseException as e:return dict(name=name,meta=meta,error=str(e))

def execute(mode,jobs):
    results=[];start=time.monotonic()
    with concurrent.futures.ThreadPoolExecutor(max_workers=8) as pool:
        for r in pool.map(run,jobs):
            results.append(r)
            if len(results)%256==0:print(mode,'done',len(results),'/',len(jobs),'seconds',round(time.monotonic()-start),flush=True)
    (OUT/(mode+'.json')).write_text(json.dumps(results,indent=2))
    good=[r for r in results if 'error' not in r]
    print('TOTAL',len(results),'GOOD',len(good),'UNIQUE',len({r['sha256'] for r in good}),'NOP16',sum(r['nop16'] for r in good),'SOME_NOP',sum(r['nops']>0 for r in good))
    best={r['sha256']:r for r in sorted(good,key=lambda r:r['name'],reverse=True)}
    for r in sorted(best.values(),key=lambda r:(r['diff'],-r['prefix'],r['regions']))[:12]:print(json.dumps(r))
    for r in [r for r in results if 'error' in r][:3]:print(json.dumps(r))
    return results

if __name__=='__main__':
    mode=sys.argv[1] if len(sys.argv)>1 else 'boolean'
    if mode=='boolean':
        choices=itertools.product(range(len(PRED)),range(len(SHAPES)),range(2),range(3))
        jobs=[(f'b{n:04}',source(PRED[p],SHAPES[s],t,h),dict(pred=p,shape=s,timing=t,helper=h)) for n,(p,s,t,h) in enumerate(choices)]
    elif mode=='debug':
        flags=['/Z7','/Zi','/Zd','/ZI','/Z7 /Ob2','/Zi /Oy-','/Z7 /Gm','/Z7 /GF-']
        choices=itertools.product([0,1,5,6,8,12,16,22],range(2),range(len(flags)))
        jobs=[(f'd{n:04}',source(PRED[p],SHAPES[5],t,0,flags=('/Od /G7 /arch:SSE' if flags[f]=='/ZI' else '/O1 /G7 /arch:SSE'),debug=flags[f]),dict(pred=p,timing=t,debug=flags[f])) for n,(p,t,f) in enumerate(choices)]
    else:raise SystemExit(mode)
    execute(mode,jobs)
