import itertools
from pathlib import Path
base = Path('base.cpp').read_text()
FREE = {'tc': 'extern "C" void __cdecl free(void *block) throw(...);', 'nc': 'extern "C" void __cdecl free(void *block);'}
PROXY = {'p0': '', 'pl': '~proxy() { if (LEAKS) free(_M_data); }', 'pe': '~proxy() {}'}
ALLOC = {'a0': '', 'ae': '~allocator() {}', 'al': '~allocator() { if (LEAKS) free(0); }'}
DEALLOC = {'d1': 'if (p != 0) free(p);'}
INST = {'i0': '', 'ib': 'template class sbase<@CT@, allocator<@CT@> >;'}
CT = {'w': 'unsigned short', 'n': 'char'}
for (fk, fv), (pk, pv), (ak, av), (ik, iv), (ck, cv) in itertools.product(FREE.items(), PROXY.items(), ALLOC.items(), INST.items(), CT.items()):
    s = base.replace('@FREE@', fv).replace('@PROXYDTOR@', pv).replace('@ALLOCDTOR@', av).replace('@DEALLOC@', DEALLOC['d1']).replace('@INST@', iv).replace('@CT@', cv)
    Path(f'v_{fk}{pk}{ak}{ik}{ck}.cpp').write_text(s)
