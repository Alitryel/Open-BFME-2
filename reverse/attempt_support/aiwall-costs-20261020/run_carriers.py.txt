from pathlib import Path
import sys, re, json, time
ROOT=Path(__file__).resolve().parents[2]
sys.path.insert(0,str(ROOT/'tools'))
from variant_search import Scorer, diff_text
symbol='?distSqToSegment@@YAMPBVCoord2D@@00@Z'
out=Path(__file__).parent/'carriers'
scorer=Scorer(symbol,0x4eacf5,197,out)
base=(ROOT/'reverse/attempts/0x004eadba.cpp').read_text()
base=base.replace('../../Code/',str(ROOT/'Code').replace('\\','/')+'/')
rows=[]
for mask in range(16):
 text=base
 head,body=text.split('static __declspec(noinline) float distSqToSegment',1)
 for i,var in enumerate(('apx','apy','dx','dy')):
  if not mask&(1<<i):continue
  body=re.sub(r'\b'+var+r'\b',var+'.value',body)
  body=re.sub(r'float '+var+r'\.value = ([^;]+);',r'CoordinateValue '+var+r' = { \1 };',body)
 text=head+'struct CoordinateValue { float value; };\nstatic __declspec(noinline) float distSqToSegment'+body
 r=scorer(text)
 got=r.get('compiled',b'');target=r.get('target',b'')
 row={'mask':mask,'file':f'c{scorer.n}.cpp','size':len(got),'diff':sum(a!=b for a,b in zip(got,target))+abs(len(got)-len(target)),'exact':r['exact'],'fitness':r['fitness'],'class':r['class']}
 rows.append(row)
 print(json.dumps(row),flush=True)
 if r['exact']:
  (out/'win.cpp').write_bytes(text.encode())
  break
(out/'results.json').write_text(json.dumps(rows,indent=2))
