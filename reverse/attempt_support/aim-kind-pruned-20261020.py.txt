from pathlib import Path
import subprocess,sys,json,re
root=Path(__file__).resolve().parents[1];out=root/'build/aim-kind-context-oct20';tests={}
for p in out.glob('generic-*.cpp'):
    s=p.read_text()
    for method in ('3C','89','96'):
        s,n=re.subn(r' Bool isKindOf'+method+r'\(\) const\s*\{\s*return [^;]+;\s*\}', '',s,count=1)
        assert n==1,(p,method)
    tests[p.stem]=s
    if '-std-' in p.stem:tests[p.stem+'-index']=s.replace('return m_kindOf.test(kind);','return m_kindOf[kind];')
name='?updateInternal@AIAttackAimAtTargetState@@AAE?AW4StateReturnType@@_N@Z';results=[]
for key,s in tests.items():
    p=out/('pruned-'+key+'.cpp');p.write_text(s)
    r=subprocess.run([sys.executable,'tools/explain_mismatch.py',name,'--rva','0x003508C5','--size','1104','--source',p.relative_to(root).as_posix()],cwd=root,capture_output=True,text=True,encoding='utf8',errors='replace',timeout=240)
    log=r.stdout+r.stderr;p.with_suffix('.log').write_text(log)
    d={'key':key,'exit':r.returncode,'exact':'OK: bytes match' in log,'summary':[x for x in log.splitlines() if x.startswith(('compiled symbol size:','classification:','unresolved:','first diff:'))]}
    if r.returncode:d['error']=log[-700:]
    results.append(d);(out/'pruned-results.json').write_text(json.dumps(results,indent=2));print(json.dumps(d),flush=True)
