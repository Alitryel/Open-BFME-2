from pathlib import Path
import sys,json,struct,pefile,capstone
from capstone.x86_const import X86_OP_IMM
sys.path.insert(0,'tools');import build,eh_verify
root=Path.cwd();dest=root/'build/df20-context-20261020'
name=(dest/'baseline.cpp').read_text().splitlines()[0][3:]
game=eh_verify.Image();pe=pefile.PE(str(root/'baselines/bfme2/workshop-vanilla-1.06/files/worldbuilder.exe'))
wb=eh_verify.Image.__new__(eh_verify.Image);wb.base=pe.OPTIONAL_HEADER.ImageBase;wb.bytes=pe.get_memory_mapped_image();wb.size=len(wb.bytes)
md=capstone.Cs(capstone.CS_ARCH_X86,capstone.CS_MODE_32);md.detail=True

def frame(image,rva):
 handler,fi=image.frame_funcinfo(rva);maxstate=image.dword(fi+4);maprva=image.dword(fi+8)-image.base
 entries=[struct.unpack_from('<iI',image.bytes,maprva+8*i) for i in range(maxstate)]
 actions=[p-image.base for s,p in entries if p];low=min(actions)
 return {'handler':handler,'funcinfo':fi,'states':maxstate,'map':maprva,'to_states':[s for s,p in entries],'actions':[hex(p-image.base) if p else None for s,p in entries],'funclet_start':low,'funclet_bytes':handler-low}
a=frame(game,0xdf20);b=frame(wb,0x6e71a0-wb.base)
assert a['states']==b['states'] and a['to_states']==b['to_states']
size=a['funclet_bytes'];assert size==b['funclet_bytes']
x=list(md.disasm(game.bytes[a['funclet_start']:a['handler']],game.base+a['funclet_start']));y=list(md.disasm(wb.bytes[b['funclet_start']:b['handler']],wb.base+b['funclet_start']))
assert len(x)==len(y)
bad=[]
for g,w in zip(x,y):
 off=g.address-game.base-a['funclet_start'];wx=w.address-wb.base-b['funclet_start']
 if off!=wx or g.size!=w.size or g.mnemonic!=w.mnemonic:bad.append({'offset':hex(off),'kind':'shape'});continue
 gm=bytearray(g.bytes);wm=bytearray(w.bytes)
 if g.mnemonic in ('call','jmp') and g.operands[0].type==w.operands[0].type==X86_OP_IMM:
  o=g.encoding.imm_offset;n=g.encoding.imm_size;assert n==4
  gm[o:o+n]=wm[o:o+n]=bytes(n)
 if gm!=wm:bad.append({'offset':hex(off),'game':g.mnemonic+' '+g.op_str,'wb':w.mnemonic+' '+w.op_str})
obj=eh_verify.Obj(build.obj_path(dest/'baseline.cpp'));sy=obj.by_name[name];eh=obj.eh_sections(sy['section']);places,conflicts=eh_verify.section_placements(game,obj,dict(sy,retail_rva=0xdf20),eh)
sections=[];operand_diffs=[]
for s in eh:
 at=places[s['index']];body=s['data'];theirs=game.bytes[at:at+len(body)];masked=set()
 for o,i,t in s['relocs']:masked.update(range(o,o+4))
 diff=[o for o in range(len(body)) if o not in masked and body[o]!=theirs[o]]
 sections.append({'name':s['name'],'rva':hex(at),'length':len(body),'nonreloc_byte_differences':len(diff)})
 if s['name'].startswith('.text'):
  oi=list(md.disasm(body,at));ti=list(md.disasm(theirs,at));assert len(oi)==len(ti)
  for o,t in zip(oi,ti):
   delta=o.address-at
   if any(delta<=d<delta+o.size for d in diff):operand_diffs.append({'rva':hex(o.address),'ours':o.mnemonic+' '+o.op_str,'retail':t.mnemonic+' '+t.op_str})
result={'game':a,'wb':b,'wb_funclet_instructions':len(x),'wb_funclet_remaining_differences':bad,'bank_eh_sections':sections,'bank_eh_relocation_conflicts':conflicts,'bank_funclet_operand_differences':operand_diffs}
(dest/'eh-graph.json').write_text(json.dumps(result,indent=2)+'\n')
print(json.dumps({k:v for k,v in result.items() if k not in ['game','wb','bank_funclet_operand_differences']},indent=2))
print('Funclet operand differences:',json.dumps(operand_diffs,indent=2))
