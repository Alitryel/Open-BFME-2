from pathlib import Path

for version,opt,frame in itertools.product(['14.50.35717','14.51.36231'],['/O1','/O2','/Od'],['/Oy','/Oy-']):
    n=f'm{len(rows):03d}';src=out/(n+'.cpp');obj=src.with_suffix('.obj');src.write_text('// cl: '+opt+' '+frame+' /GS- /G6 /arch:SSE /DNDEBUG /MD /EHsc\n'+body,encoding='utf-8',newline='\n')
    command,env=build.compiler_command(src,obj)
    compiler=Path('C:/Program Files (x86)/Microsoft Visual Studio/18/BuildTools/VC/Tools/MSVC')/version/'bin/Hostx64/x86/cl.exe'
    command[0]=str(compiler);env['PATH']=str(compiler.parent)+';'+env['PATH']

    row=dict(name=n,version=version,opt=opt,frame=frame,command=command,compiler_sha256=hashlib.sha256(compiler.read_bytes()).hexdigest(),source_sha256=hashlib.sha256(src.read_bytes()).hexdigest(),exit_code=p.returncode)

    rows.append(row);(out/'results.json').write_text(json.dumps(rows,indent=2)+'\n',encoding='utf-8');print(n,version,opt,frame,'rc',p.returncode,'size',row.get('size'),'nops',len(row.get('nops',[])),flush=True)
print('SUMMARY',len(rows),'compiled',sum(r['exit_code']==0 for r in rows),'distinct',len({r.get('raw_sha256') for r in rows if r.get('raw_sha256')}),flush=True)
