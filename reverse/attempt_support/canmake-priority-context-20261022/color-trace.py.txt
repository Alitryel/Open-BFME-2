import sys,os,select,re,time,subprocess,json
from pathlib import Path
ROOT=Path.cwd().resolve()
sys.path.insert(0,str(ROOT/'tools'))
import build
class Dbg:
    def __init__(self,src,obj,log):
        cmd,env=build.compiler_command(Path(src).resolve(),Path(obj).resolve())
        cmd[0]='/usr/bin/winedbg'
        self.p=subprocess.Popen(cmd,cwd=ROOT,env=env,stdin=subprocess.PIPE,stdout=subprocess.PIPE,stderr=subprocess.STDOUT)
        self.log=open(log,'wb')
    def prompt(self,command=None,timeout=300):
        p=self.p
        if command is not None:
            p.stdin.write((command+'\n').encode());p.stdin.flush()
        data=b'';deadline=time.time()+timeout
        while not data.endswith(b'Wine-dbg>'):
            if time.time()>deadline: raise TimeoutError(data[-300:])
            if not select.select([p.stdout],[],[],1)[0]: continue
            chunk=os.read(p.stdout.fileno(),65536)
            if not chunk: return data.decode(errors='replace')
            data+=chunk
        self.log.write(data);self.log.flush()
        return data.decode(errors='replace')
    def words(self,addr,count):
        out=self.prompt('x/%dx 0x%x'%(count,addr))
        vals=[]
        for line in out.splitlines():
            m=re.match(r'\s*0x([0-9a-f]+):\s+(.*)',line)
            if not m: continue
            vals+= [int(v,16) for v in re.findall(r'\b([0-9a-f]{8})\b',m.group(2))]
        if not vals:
            vals=[int(v,16) for v in re.findall(r'\b([0-9a-f]{8})\b',out.replace('Wine-dbg>',''))]
        return vals[:count]
    def reg(self,name):
        out=self.prompt('print/x $'+name)
        out=out.replace('Wine-dbg>','').strip()
        m=re.search(r'(0x[0-9a-f]+|\b[0-9]+\b)',out); return int(m.group(1),0)
    def string(self,addr):
        out=self.prompt('x/s 0x%x'%addr)
        return out.replace('Wine-dbg>','').strip()
