import json,sys,capstone
from pathlib import Path
W=Path(__file__).parent;m=capstone.Cs(capstone.CS_ARCH_X86,capstone.CS_MODE_32)
allrows=[];seen=set()
for p in W.glob('*.json'):
 if p.stem.endswith(('regions','trace','controls','insert','color')):continue
 rows=json.loads(p.read_text())
 if not isinstance(rows,list):continue
 for r in rows:
  if not isinstance(r,dict) or 'sha'not in r or r['sha']in seen or 'error'in r:continue
  seen.add(r['sha']);f=W/(r['label']+'.bin')
  if not f.exists():continue
  a=list(m.disasm(f.read_bytes()[:max(0,r['size']-124)],0x593f15));at={x.address:i for i,x in enumerate(a)}
  ds=[i for i,x in enumerate(a) if x.mnemonic=='call'and x.op_str=='0x4d55bc']
  if not ds:continue
  tail=a[ds[-1]+1:ds[-1]+5];r=dict(r)
  r['detach_count']=len(ds);r['tailins']=[x.mnemonic+' '+x.op_str for x in tail]
  r['targetins']=[];r['backshared']=False;r['exact_order']=False
  if len(tail)>1 and tail[0].mnemonic=='mov' and tail[1].mnemonic=='jmp':
   try:target=int(tail[1].op_str,16)
   except ValueError:target=0
   if target in at:
    t=a[at[target]:at[target]+3];r['targetins']=[x.mnemonic+' '+x.op_str for x in t]
    r['backshared']=target<tail[0].address and t[0].mnemonic=='mov' and t[1].mnemonic=='jmp'
    r['exact_order']=r['backshared'] and tail[0].op_str=='ebx, dword ptr [ebp - 0x2c]' and t[0].op_str=='esi, dword ptr [ebp - 0x14]'
  allrows.append(r)
(W/'tail-findings.json').write_text(json.dumps(allrows,indent=2))
for key in ['backshared','exact_order']:
 rs=[r for r in allrows if r[key]];print(key,len(rs))
 for r in sorted(rs,key=lambda r:-r['fit'])[:8]:print(r['label'],r['size'],r['diff'],round(r['fit'],6),r['tailins'][:2],r['targetins'][:2])
