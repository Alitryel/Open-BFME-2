import concurrent.futures as cf
import difflib
import itertools
import json
import pathlib
import re
import subprocess
import sys
import time

ROOT = pathlib.Path('/work7/Open-BFME-2')
sys.path.insert(0, str(ROOT / 'tools'))
import build
from variant import resolve_bytes
import capstone

OUT = ROOT / 'build/655b90-mergeweights'
BASE = (OUT / 'control.cpp').read_text()
A = BASE.index('int Rva007E8B00(')
Z = BASE.index("\n// The record's", A)
HEAD, BODY, TAIL = BASE[:A], BASE[A:Z], BASE[Z:]
SYM = '?Rva007E8B00@@YAHPAEHPBD@Z'
RVA, SIZE = 0x655b90, 549
TARGET = build.read_target_bytes(RVA, SIZE)
SMAP = build.load_symbol_map()
MD = capstone.Cs(capstone.CS_ARCH_X86, capstone.CS_MODE_32)
TINS = [(i.mnemonic, i.op_str) for i in MD.disasm(TARGET, RVA)]

def score(name, text):
    src, obj = OUT / (name + '.cpp'), OUT / (name + '.obj')
    if not (src.exists() and obj.exists() and src.read_text() == text):
        src.write_text(text)
        if obj.exists(): obj.unlink()
        cmd, env = build.compiler_command(src, obj)
        p = subprocess.run(cmd, cwd=ROOT, env=env, capture_output=True, text=True)
        if p.returncode:
            (OUT / (name + '.log')).write_text(p.stdout + p.stderr)
            return dict(name=name, error=p.returncode)
    raw, relocs = build.read_object_symbol_bytes(obj, SYM, code_only=True)
    span = max(len(raw), SIZE)
    code, unresolved = resolve_bytes(raw.ljust(span, b'\0'), relocs,
        build.read_target_bytes(RVA, span), RVA, span, SMAP)
    (OUT / (name + '.bin')).write_bytes(code)
    md = capstone.Cs(capstone.CS_ARCH_X86, capstone.CS_MODE_32)
    ins = [(i.mnemonic, i.op_str) for i in md.disasm(code[:len(raw)], RVA)]
    regions = operands = 0
    sm = difflib.SequenceMatcher(None, [x[0] for x in ins], [x[0] for x in TINS], autojunk=False)
    for tag,a,b,c,d in sm.get_opcodes():
        if tag != 'equal': regions += 1
        else: operands += sum(ins[a+k][1] != TINS[c+k][1] for k in range(b-a))
    return dict(name=name, size=len(raw), nd=sum(a!=b for a,b in zip(code,TARGET)),
                regions=regions, operands=operands, unresolved=unresolved)

def safe_score(name, text):
    try: return score(name, text)
    except BaseException as e: return dict(name=name, error=str(e))

def duplicate(t, pred, form):
    if form == 0: return 'if ('+pred+') {\n'+t+'\n} else {\n'+t+'\n}'
    return 'switch (bool('+pred+')) {case true: {\n'+t+'\n} break; default: {\n'+t+'\n} break;}'

def replace_region(body, start, end, pred, form):
    a = body.index(start)
    if start.startswith('/*'): a += len(start)
    z = body.index(end, a)
    return body[:a]+duplicate(body[a:z], pred, form)+body[z:]

def gen_merge():
    yield 'baseline', BASE
    for mask, form in itertools.product(range(256), range(2)):
        b = BODY.replace('\tfor ( i = 0; (unsigned int)field[ i ]',
            '/*LENGTH*/\n\tfor ( i = 0; (unsigned int)field[ i ]')
        b = b.replace('\tiDelta =', '/*DELTA*/\n\tiDelta =')
        pred = 'pValue == pTail'
        if mask & 1:
            b = re.sub(r'pTail = p;\s*pValue = pTail;',
                       lambda m: duplicate(m[0], 'p == pBuffer', form), b)
        if mask & 2:
            t = 'iDelta = i - ( pTail - pValue );'
            b = b.replace(t, duplicate(t, pred, form))
        if mask & 4:
            b = replace_region(b, '\tif ( iDelta == 0 && memcmp', '\n\tif ( iDelta > 0 )', pred, form)
        if mask & 8:
            b = replace_region(b, '\tif ( iDelta > 0 )', '\n\tmemcpy(', pred, form)
        if mask & 16:
            t = 'memcpy( pValue, field, i );'
            b = b.replace(t, duplicate(t, pred, form))
        if mask & 32:
            b = replace_region(b, '\tif ( pValue[ i ] == 0', '\n\treturn i - 1;', pred, form)
        if mask & 64:
            b = replace_region(b, '/*LENGTH*/', '/*DELTA*/', pred, form)
        if mask & 128:
            b = replace_region(b, '/*LENGTH*/', '\n\treturn i - 1;', pred, form)
        yield 'merge-%03d-%d' % (mask, form), HEAD+b+TAIL

def gen_assign():
    predicates = ['pTail == pBuffer', 'pTail == 0', '*pTail == 0',
                  "*field == '~'", 'size > 0', 'pTail == p']
    for n,(mask,pred,form,arm) in enumerate(itertools.product(range(1,16),predicates,range(3),range(2))):
        counter = [0]
        def change(m):
            bit = counter[0]; counter[0] += 1
            if not mask & (1 << bit): return m[0]
            left = 'pValue = pTail;'
            right = 'pValue = p;' if arm else left
            if form == 0: t='if('+pred+'){'+left+'}else{'+right+'}'
            elif form == 1:
                t='if('+pred+')goto yes%d; '%bit+right+' goto end%d; yes%d: '%(bit,bit)+left+' end%d:;'%bit
            else: t='pValue = ('+pred+') ? pTail : '+('p' if arm else 'pTail')+';'
            return 'pTail = p;\n'+t
        b=re.sub(r'pTail = p;\s*pValue = pTail;',change,BODY)
        assert counter[0] == 4
        yield 'assign-%04d'%n, HEAD+b+TAIL

def gen_tail():
    predicates = ['pValue == pTail','pValue == pBuffer','pValue != 0',
                  'pValue[0] != 0','iDelta == 0','i > 0','size > 0',"*field == '~'"]
    zero='pValue[ i - 1 ] = 0;'
    vol='pValue[ i - 1 ] = *(volatile char *)&g_bfmeFeslFieldSeparator;'
    for n,(pred,form,depth,mask) in enumerate(itertools.product(predicates,range(3),range(1,5),range(1,4))):
        # C++ labels have function scope. The original broad generator's
        # recursive goto text reused labels; exclude those malformed trials.
        if form==2 and (depth>1 or mask==3): continue
        b=BODY
        def dup(t):
            for d in range(depth):
                if form<2:t=duplicate(t,pred,form)
                else:t='if('+pred+')goto side%d; {'%d+t+'}goto done%d;side%d:{'%(d,d)+t+'}done%d:;'%d
            return t
        if mask&1:b=b.replace(zero,dup(zero))
        if mask&2:
            b=b.replace(vol,'char separator=*(volatile char *)&g_bfmeFeslFieldSeparator;\n'+dup('pValue[ i - 1 ] = separator;'))
        yield 'tail-%04d'%n,HEAD+b+TAIL

def gen_loops():
    start=BODY.index('\tfor ( i = 0; (unsigned int)field[ i ]')
    end=BODY.index('\n\tiDelta',start)
    cond="(unsigned int)field[i] >= ' '"
    loops=[
        'for(i=0;'+cond+';i++){} i++;',
        'i=0;while('+cond+'){i++;} i++;',
        'for(i=0;;i++){if(!('+cond+'))break;}i++;',
        'i=0;for(;;){if(!('+cond+'))break;i++;}i++;',
        'i=0;do{if(!('+cond+'))break;i++;}while(true);i++;',
        'i=0;while(true){if('+cond+')i++;else break;}i++;',
        'i=0;test:if('+cond+'){i++;goto test;}i++;',
        'i=0;for(;;){if(!('+cond+')){i++;break;}i++;}',
        'i=0;for(;;){if('+cond+'){i++;}else{i++;break;}}',
        'i=0;while('+cond+'){i++;}if(pValue==pTail)i++;else i++;',
        'i=0;while('+cond+'){i++;}if(pValue==pBuffer)i++;else i++;',
        'i=0;while('+cond+'){i++;}if(iDelta==0)i++;else i++;',
        'i=0;while('+cond+'){i++;}i=1+i;',
        'i=0;while('+cond+'){i++;}int length=i+1;i=length;',
        'i=0;while('+cond+'){i++;}unsigned length=i+1;i=length;',
        'i=0;while('+cond+'){i++;}int &length=i;++length;',
    ]
    # iDelta is uninitialized at the loop, so replace that diagnostic predicate.
    loops=[s.replace('iDelta==0','size==0') for s in loops]
    for n,(loop,sep,decl,head) in enumerate(itertools.product(loops,range(3),range(4),range(2))):
        b=BODY[:start]+loop+BODY[end:]
        if sep==1:b=b.replace('g_bfmeFeslFieldSeparator !=', '*(volatile char *)&g_bfmeFeslFieldSeparator !=').replace('*(volatile char *)&g_bfmeFeslFieldSeparator;', 'g_bfmeFeslFieldSeparator;',1)
        if sep==2:b=b.replace('*(volatile char *)&g_bfmeFeslFieldSeparator;', 'g_bfmeFeslFieldSeparator;')
        if decl&1:b=b.replace('\tint i;','').replace('\tint iDelta;','\tint iDelta;\n\tint i;')
        if decl&2:b=b.replace('\tunsigned char *pValue;','').replace('\tint iDelta;','\tunsigned char *pValue;\n\tint iDelta;')
        if head:b=b.replace('pTail = p;\n\t\t\t\t\tpValue = pTail;', 'pValue = p;\n\t\t\t\t\tpTail = pValue;')
        yield 'loops-%04d'%n,HEAD+b+TAIL

def run(jobs, label):
    start = time.time()
    results = []
    with cf.ThreadPoolExecutor(max_workers=8) as pool:
        futures = [pool.submit(safe_score, n, s) for n,s in jobs]
        for f in cf.as_completed(futures):
            r = f.result(); results.append(r)
            if r.get('size') == SIZE and r.get('nd') == 0 and not r.get('unresolved'):
                print('EXACT', r, flush=True)
    results.sort(key=lambda r:(r.get('regions',999),r.get('operands',999),abs(r.get('size',0)-SIZE),r.get('nd',999)))
    (OUT/(label+'.json')).write_text(json.dumps(results,indent=2))
    print(json.dumps(dict(label=label, seconds=round(time.time()-start,1), errors=sum('error' in r for r in results), best=results[:8])),flush=True)

if __name__ == '__main__':
    label=sys.argv[1] if len(sys.argv)>1 else 'merge'
    run(list({'merge':gen_merge,'assign':gen_assign,'tail':gen_tail,'loops':gen_loops}[label]()), label)
