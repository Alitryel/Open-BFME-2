import json,sys
from pathlib import Path
from wdbg import Dbg
W=Path(__file__).resolve().parent;tag=sys.argv[1]
d=Dbg(W/(tag+'.cpp'),W/(tag+'-schedule.obj'),W/(tag+'-schedule.log'))
events=[];active=False;finished=False;window=None

def snapshot(first,last):
 nodes={};syms={};order=[]
 def operand(p,depth=0):
  if p<0x10000 or p in nodes or depth>6:return
  w=d.words(p,20);nodes[p]=w
  if len(w)<16:return
  k=w[2]&255
  if k in [1,2,4]:
   for sp in [w[6],w[7]]:
    if sp>0x10000 and sp not in syms:syms[sp]=d.words(sp,24)
  if k in [5,6]:operand(w[13],depth+1)
  elif k in [13,18]:operand(w[10],depth+1);operand(w[11],depth+1)
 p=first
 while p and p not in order and len(order)<300:
  order.append(p);operand(p)
  if p==last:break
  p=nodes[p][0]
 return dict(first=first,last=last,order=order,nodes=nodes,syms=syms)

def graph():
 root=d.words(0x108a8054,1)[0];rw=d.words(root,12);p=rw[0];nodes={};edges={}
 while p and p not in nodes and len(nodes)<300:
  w=d.words(p,16);nodes[p]=w;e=w[3];seen=set()
  while e and e not in seen and len(seen)<300:
   seen.add(e);ew=d.words(e,6);edges[e]=ew;e=ew[0]
  if p==rw[1]:break
  p=w[0]
 return dict(root=root,rw=rw,nodes=nodes,edges=edges)

try:
 d.prompt();d.prompt('set $CanDeferOnBPByAddr = 1');d.prompt('break *0x107b0087')
 for _ in range(500):
  out=d.prompt('cont',60)
  if 'terminated' in out:finished=True;break
  e=d.reg('eip')
  if e==0x107b0087:
   f=d.reg('esi');fn=d.string(d.words(d.words(f,1)[0]+4,1)[0])
   if '?bfmeTwoCGD@' not in fn:continue
   active=True;d.prompt('delete 1')
   for a in (0x107b4e11,0x107b4e20,0x107b4e2a,0x107b29e0):d.prompt('break *0x%x'%a)
   continue
  if not active:continue
  if e==0x107b29e0:
   for i in (2,3,4,5):d.prompt('delete '+str(i))
   break
  if e==0x107b4e11:
   first=d.reg('ebp')
   window=dict(before=snapshot(first,d.reg('edi')),predecessor=d.words(first+16,1)[0]);events.append(window)
   print('window',len(events),'instructions',len(window['before']['order']),flush=True)
  elif e==0x107b4e20:
   window['graph']=graph()
   print('graph nodes',len(window['graph']['nodes']),'edges',len(window['graph']['edges']),flush=True)
  elif e==0x107b4e2a:
   # Keep the pre-window predecessor: 0x108a8050 is the first IR node.
   pred=window['predecessor']
   window['after']=snapshot(d.words(pred,1)[0],d.words(0x108a804c,1)[0])
 if not finished:
  for _ in range(300):
   out=d.prompt('cont',60)
   if 'terminated' in out:finished=True;print('compiler terminated',flush=True);break
  else:raise RuntimeError('did not finish')
finally:
 (W/(tag+'-schedule.json')).write_text(json.dumps(events,indent=1))
 if d.p.poll() is None:
  try:d.prompt('quit')
  except Exception:d.p.kill()
 d.log.close()
