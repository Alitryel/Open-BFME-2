from search import *
import itertools
s=(W/'base.cpp').read_text();vs=[]
# Keep ABI and call effects; test front-end type identities and complete receiver identity.
for normal,lit,merge,qual,pure in itertools.product(range(4),range(8),range(2),range(2),range(2)):
 o=s
 ty=['Coord3D*','const Coord3D*','volatile Coord3D*','const volatile Coord3D*'][normal]
 o=o.replace('ground(float,float,Coord3D*)','ground(float,float,'+ty+')').replace('layer(float,float,int,Coord3D*,bool)','layer(float,float,int,'+ty+',bool)')
 value=['0','0i64','0ui64','(unsigned __int64)0','(__int64)0','(short)0','(unsigned char)0','(const int)0'][lit]
 o=o.replace('->ground(pos.x,pos.y,0)','->ground(pos.x,pos.y,'+value+')').replace('->layer(pos.x,pos.y,object->rva0028B511(),0,true)','->layer(pos.x,pos.y,object->rva0028B511(),'+value+',true)')
 if qual:
  o=o.replace('virtual float ground(float,float,'+ty+');','virtual float ground(float,float,'+ty+')const;').replace('virtual float layer(float,float,int,'+ty+',bool);','virtual float layer(float,float,int,'+ty+',bool)const;')
 if pure:
  for line in list(o.splitlines()):
   if 'virtual' in line and ((' ground(' in line) or (' layer(' in line) or (' underwater(' in line)):
    o=o.replace(line,line[:-1]+'=0;')
 if merge:
  start=o.index('class TerrainLogic\n');end=o.index('class LocoTerrainView:',start)
  o=o[:start]+o[end:]
  o=o.replace('class LocoTerrainView:','class TerrainLogic:')
  o=o.replace('class TerrainLogic:public LocoSlots<6>\n{\n public:', 'class TerrainLogic:public LocoSlots<6>\n{\n public:\n PathfindLayerEnum getHighestLayerForDestination(const Coord3D*,bool);')
  at=o.index('class BfmeFixedStorage0004543D')
  o=o[:at]+'extern TerrainLogic*TheTerrainLogic;\n'+o[at:]
  o=o.replace('LocoTerrainView','TerrainLogic')
 vs.append((f'ident_{normal}{lit}{merge}{qual}{pure}',o))
run(vs,'identity512')
