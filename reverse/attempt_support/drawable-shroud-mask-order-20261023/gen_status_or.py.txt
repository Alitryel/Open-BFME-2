from pathlib import Path

forms=[
 '!(V&0x40U)',
 '((V|~0x40U)!=~0U)',
 '(((V|~0x40U)+1U)!=0)',
 '((V|~0x40U)<~0U)',
 '((V|~0x40U)==~0x40U)',
 '((V|~0x40U)<=~0x40U)',
 '(((V|~0x40U)^~0U)!=0)',
 '(((V|~0x40U)^~0x40U)==0)',
 '(((V&0x40U)+~0x40U)==~0x40U)',
 '(((V&0x40U)|~0x40U)==~0x40U)',
 '(((V|~0x40U)+0x40U)!=0x3FU)',
 '((unsigned char)(V|~0x40U)!=255U)',
 '((signed char)(V|~0x40U)!=(signed char)-1)',
 '((unsigned char)(V|~0x40U)==191U)',
 '((signed char)(V|~0x40U)==(signed char)-65)',
 '((unsigned short)(V|~0x40U)!=65535U)',
 '((unsigned short)(V|~0x40U)==65471U)',
 '(((V&0x40U)^0x40U)>0)',
 '(((V>>6)%2U)==0)',
 '((V%128U)<64U)',
]
for i,j in itertools.product(range(20),repeat=2):
