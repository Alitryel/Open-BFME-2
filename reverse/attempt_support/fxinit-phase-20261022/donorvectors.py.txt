import sys,json,itertools
from pathlib import Path
from concurrent.futures import ThreadPoolExecutor
sys.path.insert(0,'build/fxinit-seat22');import search as h
h.HERE=Path('build/fxinit-phase22').resolve()
view='class Vector3 {\npublic:\n float X,Y,Z;\n Vector3 &operator=(const Vector3 &v) {X=v.X;Y=v.Y;Z=v.Z;return *this;}\n};'
roots=['reference/open-bfme-1/game/Libraries/Source/WWVegas','reference/open-bfme-1/inputs/reference/CnC_Generals_Zero_Hour/GeneralsMD/Code/Libraries/Source/WWVegas']
def variants():
 for donor,kind,order in itertools.product(range(2),range(2),range(2)):
  root=roots[donor]
  for d in ['WWLib','WWMath','WWDebug']:assert (Path(root)/d).is_dir(),(root,d)
  for f in ['vector2.h','vector3.h']:assert(Path(root)/'WWMath'/f).is_file()
  extras=' '.join('/I'+root+'/'+x for x in ['WWLib','WWMath','WWDebug'])+' /Ireference/shims/sweep'
  src=h.BASE.replace('// cl: /I.', '// cl: '+extras+' /I.').replace(view,'')
  hdr=root+'/WWMath/'
  if kind==0:
   includes='\n'.join('#include "'+hdr+x+'"'for x in (['vector2.h','vector3.h']if not order else['vector3.h','vector2.h']))
  else:
   # Actual Vector3 and math context before the canonical BFME2 Vector2.
   includes='#include "'+hdr+'vector3.h"\n#include "reference/shims/wwmath/Vector2.h"'
   if order:includes='#include "'+hdr+'wwmath.h"\n'+includes
  src=src.replace('#include "reference/shims/wwmath/Vector2.h"',includes)
  src=src.replace('struct TriIndex {','typedef char verify_Vector3_size[sizeof(Vector3)==12?1:-1];\nstruct TriIndex {')
  yield f'donorvectors_{donor}{kind}{order}',src
with ThreadPoolExecutor(max_workers=8)as p:rs=list(p.map(h.run_one,variants()))
(h.HERE/'donorvectors-results.json').write_text(json.dumps(rs,indent=2));print(json.dumps(rs,indent=2))
