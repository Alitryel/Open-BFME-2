from pathlib import Path
import json,struct,hashlib,re,pefile
out=Path('build/trust237-round4');out.mkdir(parents=True,exist_ok=True)
p=Path('reference/open-bfme-1/inputs/toolchains/vs2003/Program Files/Microsoft Visual Studio .NET 2003/Vc7/bin/c2.dll');b=p.read_bytes();pe=pefile.PE(data=b);base=pe.OPTIONAL_HEADER.ImageBase
rows=[]
for n in range(1,16):
 addr=struct.unpack('<I',pe.get_data(0x10870c5c-base+4*n,4))[0]
 rows.append(dict(length=n,va=hex(addr),raw=pe.get_data(addr-base,n).hex()))
def pad(n):return bytes.fromhex(rows[14]['raw'])*(n//15)+(bytes.fromhex(rows[n%15-1]['raw']) if n%15 else b'')
recipes=json.loads(Path('reverse/attempt_support/trust237-20261022-round3/padding_recipes.json').read_text())
checks=[dict(cpu=r['cpu'],length=r['size'],matches_table=bytes.fromhex(r['bytes'])==pad(r['size'])) for r in recipes]
result=dict(path=str(p),sha256=hashlib.sha256(b).hexdigest(),image_base=hex(base),table_base='0x10870c5c',table=rows,emitter_ranges=[['0x107a7e32','0x107a7e86'],['0x107e8640','0x107e869d']],inferred_npad16=pad(16).hex(),observed_recipe_comparison=checks)
(out/'compiler-padding.json').write_text(json.dumps(result,indent=2)+'\n')
results=[]
for p in [Path('baselines/bfme2/workshop-vanilla-1.06/files/worldbuilder.exe'),Path('reference/open-bfme-1/inputs/baselines/bfme1/workshop-vanilla-1.03/files/worldbuilder.exe')]:
 pe=pefile.PE(str(p));b=p.read_bytes();imports=[]
 for dll in getattr(pe,'DIRECTORY_ENTRY_IMPORT',[]):
  for i in dll.imports:
   if i.name and i.name in [b'WinVerifyTrust',b'LocalFree',b'CommandLineToArgvW',b'GetCommandLineW']:
    blob=b'\xff\x15'+i.address.to_bytes(4,'little')
    sites=[hex(pe.get_rva_from_offset(m.start())) for m in re.finditer(re.escape(blob),b)]
    imports.append(dict(dll=dll.dll.decode(),name=i.name.decode(),iat=hex(i.address),calls=sites))
 guids=[hex(pe.get_rva_from_offset(m.start())) for m in re.finditer(re.escape(bytes.fromhex('6bc5aa0044cdd0118cc200c04fc295ee')),b)]
 results.append(dict(path=str(p),base=hex(pe.OPTIONAL_HEADER.ImageBase),imports=imports,guids=guids))
(out/'sibling-image-probe.json').write_text(json.dumps(results,indent=2)+'\n')
