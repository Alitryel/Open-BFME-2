from pathlib import Path
from itertools import product

W=Path(__file__).resolve().parent
BASE=(W/'base.cpp').read_text()
FORMS=[
 'u',
 '_rotl(_rotr(u,1),1)',
 '_rotl(_rotr(u,7),7)',
 '_rotl(_rotr(u,16),16)',
 '_rotr(_rotl(u,31),31)',
 '_lrotl(_lrotr(u,8),8)',
 '_rotl(_rotl(u,16),16)',
 '_rotl(_rotl(u,13),19)',
 '_rotl(_rotl(_rotl(_rotl(u,8),8),8),8)',
 '((u&0xff00ff00u)|((u&0x00ff00ffu)))',
 '((u&0xfffffffcu)+(u&3u))',
 '((u&0xffff0000u)|(unsigned short)u)',
 '(unsigned)(((unsigned __int64)u<<31)>>31)',
 '(unsigned)(((unsigned __int64)u<<17)>>17)',
 '(unsigned)(((unsigned __int64)u<<63>>63)|((unsigned __int64)(u>>1)<<1))',
 '(u*0xaaaaaaabu)*3u',
 '(u*0xcccccccdu)*5u',
 '(u*0xb6db6db7u)*7u',
 '((u<<1)|(u>>31))>>1|((u>>31)<<31)',
 '((u>>1)|(u<<31))<<1|((u&1u))',
 '(unsigned)(__int64)(int)u',
 '((u&0x55555555u)+(u&0xaaaaaaaau))',
 '((u&0x33333333u)+(u&0xccccccccu))',
 '((u>>24)<<24)|((u<<8)>>8)',
]

decl='''extern "C" unsigned int __cdecl _rotl(unsigned int,int);
extern "C" unsigned int __cdecl _rotr(unsigned int,int);
extern "C" unsigned long __cdecl _lrotl(unsigned long,int);
extern "C" unsigned long __cdecl _lrotr(unsigned long,int);
#pragma intrinsic(_rotl,_rotr,_lrotl,_lrotr)
'''

def source(form,sites,before,cc):
 s=BASE
 if before:
  body='unsigned u=(unsigned)n;u='+FORMS[form]+';return *(const ContactVector *)(u+20);'
 else:
  body='unsigned u=(unsigned)&n->point;return *(const ContactVector *)('+FORMS[form]+');'
 helper=decl+'__forceinline const ContactVector &'+('__fastcall ' if cc else '')+'latePoint(ContactNode *n){'+body+'}\n'
 s=s.replace('class BfmeThingCGD {',helper+'class BfmeThingCGD {')
 h,t=s.split('void BfmeThingCGD::bfmeTwoCGD() {',1)
 start=0
 for i in range(3):
  at=t.find('node->point',start);assert at>=0
  value='latePoint(node)' if sites>>i&1 else 'node->point'
  t=t[:at]+value+t[at+len('node->point'):];start=at+len(value)
 return h+'void BfmeThingCGD::bfmeTwoCGD() {'+t

if __name__=='__main__':
 out=W/'identity';out.mkdir(exist_ok=True)
 choices=list(product(range(len(FORMS)),range(8),range(2),range(2)))
 for idx,args in enumerate(choices):(out/f'v{idx:04}.cpp').write_text(source(*args))
 print('generated',len(choices))
