import sys,json
sys.path.insert(0,'/work/Open-BFME-2/build/c2tr')  # wdbg.py.txt from this directory
from wdbg import Dbg
# inject2.py src tag "AT:N,AT:N,..."  consume N kind-15 ids after the AT-th kind-15 allocation (deferred across chunk refills)
src=sys.argv[1]; tag=sys.argv[2]
plan=sorted((int(a),int(n)) for a,n in (x.split(':') for x in sys.argv[3].split(',')))
W='/work/Open-BFME-2/build/db/tr/'
d=Dbg(src,W+tag+'.obj',W+tag+'.log')
active=False; cnt=0; pend=0; log=[]
try:
    d.prompt();d.prompt('set $CanDeferOnBPByAddr = 1');d.prompt('break *0x10760539')
    while True:
        o=d.prompt('cont',300)
        if 'terminated' in o or not o.strip(): break
        if '0x10760539' in o:
            if active: continue
            w=d.words(d.reg('eax'),1);s=d.words(w[0],2);fn=d.string(s[1])
            if 'DiagonalBlur' in fn:
                active=True;d.prompt('break *0x1076045d')
            continue
        if active and '0x1076045d' in o:
            cnt+=1
            while plan and plan[0][0]==cnt: pend+=plan.pop(0)[1]
            if pend:
                d.prompt('finish',60)
                ar=d.words(0x108a79b0,1)[0]; cur,end=d.words(ar+4,2)
                room=(end-cur)//0x60; t=min(room,pend)
                d.prompt('set *(int*)0x%x = 0x%x'%(ar+4,cur+t*0x60)); pend-=t; log.append((cnt,t))
            if not plan and not pend: d.prompt('delete')
            continue
finally:
    print('log',log,flush=True)
    if d.p.poll() is None:
        try: d.prompt('quit')
        except Exception: d.p.kill()
