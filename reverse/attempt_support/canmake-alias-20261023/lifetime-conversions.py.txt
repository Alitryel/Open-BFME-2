"""Test implicit conversions and value-copy lifetime at the return boundary."""
from pathlib import Path
import concurrent.futures,itertools,json,re,runpy
OUT=Path('build/canmake-alias-20261023/lifetime-conversions').resolve();OUT.mkdir(parents=True,exist_ok=True)
r=runpy.run_path('reverse/attempt_support/canmake-00596e15-20261022/replay.py.txt');r['run'].__globals__['OUT']=OUT
base=runpy.run_path('reverse/attempt_support/canmake-alias-20261023/context.py.txt')['bank_text']();prefix,body=base.rsplit('int AIBuildableWallSegment::canMake(Player *player)',1);prefix=prefix[prefix.index('\n')+1:]
tests=[]
for typ,init,conv,copy in itertools.product(range(4),range(2),range(8),range(4)):
 type_=['int','unsigned int','MakeCodeEnum','unsigned char'][typ];b=body
 helper='enum MakeCodeEnum{MAKE_ZERO=0,MAKE_TWO=2,MAKE_NINE=9};\n'
 helper+='struct MakeCodeLife { '+type_+' code; __forceinline MakeCodeLife()'+(':code(('+type_+')0)'if init else '')+'{} '
 helper+='__forceinline MakeCodeLife(const MakeCodeLife&r):code(r.code){} __forceinline ~MakeCodeLife(){} '
 helper+='__forceinline int get()const{return code;} __forceinline operator int()const{return code;} '
 helper+='__forceinline MakeCodeLife value()const{return *this;} __forceinline const MakeCodeLife &alias()const{return *this;} };\n'
 helper+='__forceinline int copyCode(MakeCodeLife value){return value.code;}\n'
 helper+='__forceinline int aliasCode(const MakeCodeLife &value){return value.code;}\n'
 expr=['selected.code','selected.get()','int(selected)','selected','selected.value().code','int(selected.value())','copyCode(selected)','aliasCode(selected)'][conv]
 if copy==1:expr='MakeCodeLife(selected).code'if conv%2==0 else 'int(MakeCodeLife(selected))'
 elif copy==2:expr='selected.alias().code'if conv%2==0 else 'int(selected.alias())'
 elif copy==3:expr='copyCode(MakeCodeLife(selected))'if conv%2==0 else 'aliasCode(MakeCodeLife(selected))'
 b=re.sub(r'selected.code=(\d+);',lambda m:'selected.code=('+type_+')'+m[1]+';',b).replace('return selected.code;','return '+expr+';')
 tests.append((f'conversion-{typ}{init}{conv}{copy}',helper+prefix+'\nint AIBuildableWallSegment::canMake(Player *player)'+b))
results=[]
with concurrent.futures.ThreadPoolExecutor(max_workers=8)as pool:
 for i,d in enumerate(pool.map(r['run'],tests),1):
  results.append(d)
  if i%128==0 or d.get('exact_candidate'):print(json.dumps(dict(done=i,total=len(tests),best=sorted(results,key=lambda x:(x.get('nd',999),-x.get('score',0)))[:2])),flush=True)
(OUT/'results.json').write_text(json.dumps(results,indent=2)+'\n')
print(json.dumps(dict(trials=len(results),errors=sum('error'in x for x in results),unresolved=sum(bool(x.get('unresolved'))for x in results),exact=sum(x.get('exact_candidate',False)for x in results),bodies=len({x.get('masked_code_sha256')for x in results}))),flush=True)
