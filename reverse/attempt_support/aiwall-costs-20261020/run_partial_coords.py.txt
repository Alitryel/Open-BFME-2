from pathlib import Path
import sys,re,json
ROOT=Path(__file__).resolve().parents[2]
sys.path.insert(0,str(ROOT/'tools'))
from variant_search import Scorer,diff_text
out=Path(__file__).parent/'partial-coords'
s=Scorer('?distSqToSegment@@YAMPBVCoord2D@@00@Z',0x4eacf5,197,out)
base=(ROOT/'reverse/attempts/0x004eadba.cpp').read_text().replace('../../Code/',str(ROOT/'Code').replace('\\','/')+'/')
rows=[]
for mode in ('separate','paired','onefield'):
 for mask in range(1,16):
  text=base
  head,body=text.split('static __declspec(noinline) float distSqToSegment',1)
  declarations=[]
  for i,var in enumerate(('apx','apy','dx','dy')):
   if not mask&(1<<i):continue
   carrier=(('ap' if i<2 else 'dir') if mode=='paired' else var+'value')
   field=('x' if i%2==0 else 'y')
   declaration=('CoordinateValue' if mode=='onefield' else 'Coord2D')+' '+carrier+';'
   if declaration not in declarations:declarations.append(declaration)
   body=re.sub(r'\b'+var+r'\b',carrier+'.'+field,body)
   body=body.replace('float '+carrier+'.'+field+' = ',carrier+'.'+field+' = ')
  body=body.replace('\n{\n','\n{\n\t'+'\n\t'.join(declarations)+'\n',1)
  text=head+'struct CoordinateValue { float x; float y; };\nstatic __declspec(noinline) float distSqToSegment'+body
  r=s(text);got=r.get('compiled',b'');target=r.get('target',b'')
  row={'mode':mode,'mask':mask,'file':f'c{s.n}.cpp','size':len(got),'diff':sum(a!=b for a,b in zip(got,target))+abs(len(got)-len(target)),'exact':r['exact'],'fitness':r['fitness'],'class':r['class']}
  rows.append(row);print(json.dumps(row),flush=True)
  if r['exact']:(out/'win.cpp').write_bytes(text.encode());break
 if r['exact']:break
(out/'results.json').write_text(json.dumps(rows,indent=2))
