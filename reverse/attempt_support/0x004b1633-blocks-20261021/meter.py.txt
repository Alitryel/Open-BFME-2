"""emo meter: for each source variant, find update's fidx, trace the A numbering (agAO tl.py) and report the
number X of the critical symbol (the last A take before the 2nd status6 inline = the inlined getObject() temp
at the case-3 compare). Exact needs X in [927, 934] (validated by symshift on the GameMakeColor and literal banks).
Also scores the plain compile. DIAGNOSTIC ONLY.
usage: meter.py SRC [SRC...] [-j N]"""
import sys, json, subprocess, argparse
from pathlib import Path
from concurrent.futures import ThreadPoolExecutor
ROOT = Path(__file__).resolve().parents[2]
PY = sys.executable
NAME = '?update@EmotionTrackerUpdate@@UAE?AW4UpdateSleepTime@@XZ'


def run(src):
    src = Path(src)
    od = ROOT / 'build/emo/meter' / src.stem
    od.mkdir(parents=True, exist_ok=True)
    r = subprocess.run([PY, str(ROOT / 'build/agAK/symshift.py'), str(src), '--fidx', '-1', '--tag', 'scan', '--outdir', str(od)],
                       cwd=ROOT, capture_output=True, text=True)
    try:
        names = json.load(open(od / (src.stem + '.scan.shift.json')))['meta']['names_at_p0']
    except Exception as ex:
        return '%s: scan failed %s %s' % (src.name, ex, (r.stdout + r.stderr)[-300:])
    fidx = next((i for i, n in enumerate(names) if NAME in n), None)
    if fidx is None:
        return '%s: update not found in scan' % src.name
    out = od / 'tl.json'
    subprocess.run([PY, str(ROOT / 'build/agAO/tl.py'), str(src), '--fidx', str(fidx), '--out', str(out)], cwd=ROOT,
                   capture_output=True, text=True)
    ev = json.load(open(out))['ev']
    lastA = None; st = 0; crit = None; nA = 0; afresh = 0; cf = 0; cfil = None; bf = 0
    for e in ev:
        if e[0] == 'k' and e[2] == 'Cf': cf += 1
        if e[0] == 'k' and e[2] == 'Bf': bf += 1
        if e[0] == 'I' and cfil is None: cfil = cf
        if e[0] == 'k' and e[2][0] == 'A':
            lastA = (e[1], e[2], e[3]); nA += 1
            if e[2] == 'Af': afresh += 1
        if e[0] == 'I' and 'status6' in e[2]:
            st += 1
            if st == 2:
                crit = lastA; break
    s = subprocess.run([PY, str(ROOT / 'build/xs/score.py'), str(src), NAME, '0x4B1633', '2200', '--objdir', str(od / 'obj')],
                       cwd=ROOT, capture_output=True, text=True)
    sc = (s.stdout + s.stderr).strip().splitlines()
    sc = sc[-1] if sc else '?'
    if crit is None:
        return '%-28s fidx %d critical not found | %s' % (src.name, fidx, sc)
    X = crit[2]
    need = 'OK' if 927 <= X <= 934 else ('+%d..+%d' % (927 - X, 934 - X) if X < 927 else '-%d' % (X - 934))
    return '%-28s fidx %d crit k%d %s X=%d need %s (Af %d Cf %d CfIL %d Bf %d) | %s' % (src.name, fidx, crit[0], crit[1], X, need, afresh, cf, cfil or -1, bf, sc)


if __name__ == '__main__':
    ap = argparse.ArgumentParser(); ap.add_argument('src', nargs='+'); ap.add_argument('-j', type=int, default=4)
    a = ap.parse_args()
    with ThreadPoolExecutor(a.j) as ex:
        for line in ex.map(run, a.src): print(line, flush=True)
