from pathlib import Path
import sys,itertools,json,subprocess,concurrent.futures,hashlib
ROOT=Path.cwd();OUT=ROOT/'build/trust237-round7'
sys.path.insert(0,str(ROOT/'tools'));sys.path.insert(0,str(ROOT/'build/trust237-round5'))
import build,capstone,search
search.OUT=OUT
probe=(OUT/'qi-control.cpp').read_text()
bank=(ROOT/'reverse/attempts/0x00237d3d.cpp').read_text()
baseline=[line for line in bank.splitlines() if line.startswith('// cl:')][0]
choices=[
 ('default',''),('qi','/QI0f'),('qi_off','/QI0f-'),
 ('managed_qi','/QI0f /d2Gmanaged'),
 ('managed_off','/QI0f- /d2Gmanaged'),
 ('clr_qi','/QI0f /clr'),
]
def run(job):
    n,kind,cpu,label,extra=job
    flags=f'// cl: /O1 /G{cpu} /DNDEBUG /MD /EHsc '+extra
    text=(probe.replace(probe.splitlines()[0],flags) if kind=='probe' else bank.replace(baseline,flags))
    name=f'm{n:03}'
    if kind=='target':return search.run((name,text,dict(kind=kind,cpu=cpu,label=label,extra=extra)))
    src=OUT/(name+'.cpp');obj=src.with_suffix('.obj');src.write_text(text)
    cmd,env=build.compiler_command(src,obj);r=subprocess.run(cmd,cwd=ROOT,env=env,capture_output=True,text=True)
    row=dict(name=name,meta=dict(kind=kind,cpu=cpu,label=label,extra=extra),command=cmd)
    if r.returncode:return dict(**row,error=(r.stdout+r.stderr)[-1200:])
    raw,rel=build.read_object_symbol_bytes(obj,'_boundaryProbe',code_only=True)
    ins=list(capstone.Cs(capstone.CS_ARCH_X86,capstone.CS_MODE_32).disasm(raw,0))
    return dict(**row,size=len(raw),sha256=hashlib.sha256(raw).hexdigest(),raw=raw.hex(),relocations=[list(x) for x in rel],nops=[i.address for i in ins if i.mnemonic=='nop'],warnings=(r.stdout+r.stderr)[-1200:])
jobs=[(n,*x) for n,x in enumerate(itertools.product(['probe','target'],[5,6,7],choices))]
# Flatten label/extra paired choices.
jobs=[(n,k,c,choice[0],choice[1]) for n,k,c,choice in jobs]
with concurrent.futures.ThreadPoolExecutor(max_workers=8) as pool:rows=list(pool.map(run,jobs))
(OUT/'qi-modes.json').write_text(json.dumps(rows,indent=2)+'\n')
for r in rows:
    if 'error' in r:print(r['name'],r['meta'],'ERROR',r['error'].splitlines()[-1])
    else:print(r['name'],r['meta'],'size',r['size'],'nops',r['nops'],'diff',r.get('diff'),r['sha256'][:16])
