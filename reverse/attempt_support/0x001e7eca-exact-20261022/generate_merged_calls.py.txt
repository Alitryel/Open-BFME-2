from search import *
import itertools
s=(W/'base.cpp').read_text()
call='((LocoTerrainView*)TheTerrainLogic)->layer(pos.x,pos.y,object->rva0028B511(),0,true)'
gcall='((LocoTerrainView*)TheTerrainLogic)->ground(pos.x,pos.y,0)'
selectors=['object->disabled&1','data->behavior==1','mode==1','object->definition!=0']
def dup(expr,cond,form):
 if form==0:return 'if('+cond+'){pos.z='+expr+';}else{pos.z='+expr+';}'
 if form==1:return 'pos.z=('+cond+')?'+expr+':'+expr+';'
 if form==2:return 'float q;if('+cond+'){q='+expr+';}else{q='+expr+';}pos.z=q;'
 if form==3:return 'switch(('+cond+')?1:0){case 0:pos.z='+expr+';break;default:pos.z='+expr+';break;}'
forms=[None]+[(c,f) for c in selectors for f in range(4)]
vs=[]
for i,j in itertools.product(range(len(forms)),repeat=2):
 o=s
 if i:
  c,f=forms[i];o=o.replace('else pos.z='+call+';', 'else {'+dup(call,c,f)+'}')
 if j:
  c,f=forms[j];o=o.replace('pos.z='+gcall+';',dup(gcall,c,f))
 vs.append((f'mergedcalls_{i:02d}_{j:02d}',o))
run(vs,'mergedcalls289')
