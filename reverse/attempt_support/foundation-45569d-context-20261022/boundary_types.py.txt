from search import *
from context import getters,mutators
import itertools

def hierarchy(s,form):
 if not form:return s
 start=s.index('class BehaviorModule\n{')
 end=s.index('struct BehaviorModuleInterface',start)
 s=s[:start]+s[end:]
 s=s.replace('private:\n\tconst ModuleData *m_moduleData; // +0x04','protected:\n\tconst ModuleData *m_moduleData; // +0x04',1)
 if form in (1,2):
  defs='class ObjectModule : public Module {public: Object *getObject() const {return m_object;} protected: Object *m_object;};\n class BehaviorModule : public ObjectModule {};\n'
 else:
  defs='class BehaviorModule : public Module {public: Object *getObject() const {return m_object;} protected: Object *m_object;};\n'
 s=s.replace('// The CastleMemberBehavior module:',defs+'\n// The CastleMemberBehavior module:',1)
 if form in (1,3):
  s=s.replace('class BfmeItemE63 : public Module','class BfmeItemE63 : public BehaviorModule',1)
  s=s.replace('unsigned char m_pad08[0x14 - 0x08];','unsigned char m_pad0c[0x14 - 0x0c];',1)
 return s

def boundary(s,mask):
 if mask&1:s=s.replace('Object *m_object;','Object *const m_object;',1)
 if mask&2:s=s.replace('const ModuleData *m_moduleData;','const ModuleData *const m_moduleData;')
 if mask&4:s=s.replace('Int rva0033A69A(const Player *player, Int builder, Int a3) const;','UnsignedInt rva0033A69A(const Player *player, Int builder, Int a3) const;',1)
 if mask&8:
  s=s.replace('Int builder, Int a3','const Object *builder, Int a3',1)
  s=s.replace('(Int)getObject(), -1','getObject(), -1',1)
 if mask&16:
  s=s.replace('Rva0039BAD2Input *what, Int cost','const ThingTemplate *what, Int cost',1)
  s=s.replace('(Rva0039BAD2Input *)what, cost','what, cost',1)
 if mask&32:s=s.replace('UnsignedInt withdraw(UnsignedInt amount','UnsignedInt withdraw(Int amount',1)
 if mask&64:s=s.replace('void onStructureCreated(Object *builder','void onStructureCreated(const Object *builder',1)
 if mask&128:
  s=s.replace('BfmeDelayedLuaEventList *eventList','DelayedLuaEventList *eventList',1)
  s=s.replace('(BfmeDelayedLuaEventList *)&list','&list',1)
 if mask&256:s=getters(s)
 if mask&512:s=mutators(s)
 return s

if __name__=='__main__':
 tests={}
 for form,mask in itertools.product(range(4),range(1024)):
  s=boundary(hierarchy(BASE,form),mask)
  tests[f'types-{form}-{mask:04d}']=s
 run(tests,'types')
