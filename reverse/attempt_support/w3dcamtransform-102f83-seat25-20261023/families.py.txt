import search as h
import itertools,re,sys

def angle():
    for bits in itertools.product(range(2),repeat=10):
        s=h.BASE
        a=s.index('    __forceinline void mulCameraAngleVector');b=s.index('    __forceinline void Rotate_X',a)
        part=s[a:b]
        for bit,(r,k) in zip(bits,itertools.product(range(3),range(3))):
            term='Row[%d][%d]*v.%s'%(r,k,'XYZ'[k])
            if bit:part=part.replace(term,'('+term+')')
        if bits[9]:
            lines=part.splitlines();ids=[i for i,l in enumerate(lines) if re.search(r'float [xyz]=',l)]
            vals=[lines[i] for i in ids]
            for i,v in zip(ids,reversed(vals)):lines[i]=v
            part='\n'.join(lines)+'\n'
        yield 'angle_'+''.join(map(str,bits)),s[:a]+part+s[b:]
h.angle=angle


def copies():
    for bits in itertools.product(range(2),repeat=10):
        s=h.BASE
        for j,(start,end) in enumerate([('Vector3(const Vector3 &v)','Vector3 &operator=(const Coord3D'),('Vector3 &operator=(const Vector3 &v)','void Set(Real x'),('void Set(Real x','__forceinline Vector3 &operator+=')]):
            a=s.index(start);b=s.index(end,a+len(start));part=s[a:b]
            for i,c in enumerate('XYZ'):
                value='v.'+c if j<2 else c.lower()
                if bits[j*3+i]:part=part.replace(c+' = '+value+';',c+' = ('+value+');')
            if bits[9]:
                lines=part.splitlines();ids=[i for i,l in enumerate(lines) if re.search(r'^[ \t]*[XYZ] =',l)]
                vals=[lines[i] for i in ids]
                for i,v in zip(ids,reversed(vals)):lines[i]=v
                part='\n'.join(lines)+'\n\t'
            s=s[:a]+part+s[b:]
        yield 'copies_'+''.join(map(str,bits)),s
h.copies=copies


def zoomregion():
    for bits in itertools.product(range(2),repeat=9):
        s=h.BASE
        pos=s.index('void W3DCamTransform::rva00102F83(CameraTransformInput *v)')
        a=s.index('    if(v->realZoom) {',s.index('// The following code',pos));b=s.index('    } else if(v->fxPitch',a)
        body=s[a+len('    if(v->realZoom) {'):b]
        if bits[0]:body=body.replace('sourcePos=pitchCamera(sourcePos,targetPos,v->fxPitch);','sourcePos.X=(sourcePos.X-targetPos.X)/v->fxPitch+targetPos.X;sourcePos.Y=(sourcePos.Y-targetPos.Y)/v->fxPitch+targetPos.Y;')
        if bits[1]:body=body.replace('sourcePos.Z*=capped*0.75f+0.25f;','sourcePos.Z=sourcePos.Z*(capped*0.75f+0.25f);')
        if bits[2]:body=body.replace('float capped=v->zoom;','float capped; capped=v->zoom;').replace('v->fxPitch=1.0f*(0.25f+pitchAdjust*0.75f);','float adjusted=0.25f+pitchAdjust*0.75f;v->fxPitch=adjusted;')
        if bits[3]:body=body.replace('1.0f*(0.25f+pitchAdjust*0.75f)','(1.0f*(0.25f+pitchAdjust*0.75f))')
        st='Vector3 sourcePos' if bits[4] else 'Vector3 &sourcePos'
        tt='Vector3 targetPos' if bits[5] else 'const Vector3 &targetPos'
        vt='CameraTransformInput *v'
        params=[st,tt,vt];args=['sourcePos','targetPos','v']
        if bits[6]:params.reverse();args.reverse()
        ret='Vector3' if bits[4] else 'void'
        func='static __forceinline '+ret+' zoomCameraRegion('+','.join(params)+') {'+body+('return sourcePos;' if bits[4] else '')+'}\n'
        call=('sourcePos=' if bits[4] else '')+'zoomCameraRegion('+','.join(args)+');'
        s=s[:a]+'    if(v->realZoom) {'+call+'\n'+s[b:]
        s=s[:pos]+func+s[pos:]

        if bits[8]:
            q=s.index('    __forceinline void mulVector3');r=s.index('    __forceinline void Rotate_X',q)
            part=re.sub(r'(float [xyz]=)([^;]+);',r'\1(\2);',s[q:r]);s=s[:q]+part+s[r:]

h.zoomregion=zoomregion


def copyabi():
    for ctor,assign,ret,params in itertools.product(range(4),repeat=4):
        s=h.BASE
        for kind,k in [('Vector3(const Vector3 &v)',ctor),('Vector3 &operator=(const Vector3 &v)',assign)]:
            a=s.index('\t'+kind);b=s.index('\n\t}',a)+3
            if k==1:piece=''
            elif k==2:piece='\t'+kind+' {Set(v.X,v.Y,v.Z);'+('return *this;' if 'operator' in kind else '')+'}'
            elif k==3:piece='\t'+kind+' {memcpy(this,&v,sizeof(Vector3));'+('return *this;' if 'operator' in kind else '')+'}'
            else:continue
            s=s[:a]+piece+s[b:]
        if ctor==3 or assign==3:s=s.replace('#include "ascii_string.h"','#include <string.h>\n#include "ascii_string.h"')
        if ret&1:s=s.replace('return p;}','return Vector3(p);}')
        if ret&2:s=s.replace('return s;}','return Vector3(s);}')
        if params&1:s=s.replace('translated(Vector3 p,','translated(const Vector3 in,').replace('const CameraTransformInput *v) {p.X','const CameraTransformInput *v) {Vector3 p(in);p.X')
        if params&2:s=s.replace('pitchCamera(Vector3 s,','pitchCamera(const Vector3 in,').replace('const Vector3 &t,float p) {s.X','const Vector3 &t,float p) {Vector3 s(in);s.X')
        yield 'copyabi_%d%d%d%d'%(ctor,assign,ret,params),s
h.copyabi=copyabi


def value_nodes():
    forms=['{lhs}+{rhs}','+({lhs}+{rhs})','(+{lhs})+{rhs}','{lhs}+(+{rhs})']
    for cfg in itertools.product(range(4),repeat=4):
      for pitch,helper in itertools.product(range(2),repeat=2):
        s=h.BASE
        if helper:
            for axis,i in [('X',0),('Y',1)]:
                old='p.'+axis+'+=v->pos.'+axis.lower()+';'
                exp=forms[cfg[i]].format(lhs='p.'+axis,rhs='v->pos.'+axis.lower())
                s=s.replace(old,'p.'+axis+'='+exp+';')
        else:
            out=''
            for i,(var,axis) in enumerate(itertools.product(['sourcePos','targetPos'],'XY')):
                lhs=var+'.'+axis;rhs='v->pos.'+axis.lower()
                out+=lhs+'='+forms[cfg[i]].format(lhs=lhs,rhs=rhs)+';'
            s=s.replace('sourcePos=translated(sourcePos,v);targetPos=translated(targetPos,v);',out)
        if pitch:
            out=''
            for axis,i in [('X',2),('Y',3)]:
                lhs='sourcePos.'+axis;rhs='targetPos.'+axis
                ex=['('+lhs+'-'+rhs+')/v->fxPitch+'+rhs,'+(('+lhs+'-'+rhs+')/v->fxPitch+'+rhs+')','(('+lhs+'-'+rhs+')/v->fxPitch)+'+rhs,'('+lhs+'-'+rhs+')/v->fxPitch+(+'+rhs+')'][cfg[i]]
                out+=lhs+'='+ex+';'
            s=s.replace('sourcePos=pitchCamera(sourcePos,targetPos,v->fxPitch);',out)
        yield 'value_nodes_'+''.join(map(str,cfg))+str(pitch)+str(helper),s
h.value_nodes=value_nodes


def lifetimes():
    for bits in itertools.product(range(2),repeat=10):
        s=h.BASE
        if bits[0]:s=s.replace('float groundFraction=','const float &groundFraction=')
        if bits[1]:s=s.replace('float factor=','const float &factor=')
        for i,(name,axis,angle) in enumerate([('angleTransform','0,0,1','angle'),('pitchTransform','1,0,0','pitch'),('angleTransform2','0,0,1','angle'),('pitchTransform2','1,0,0','pitch')]):
            if bits[2+i]:s=s.replace('Matrix3D '+name+'(Vector3('+axis+'),v->'+angle+');','const Matrix3D &'+name+'=Matrix3D(Vector3('+axis+'),v->'+angle+');')
        if bits[6]:s=s.replace('operator*=(float k)','operator*=(const float &k)')
        if bits[7]:s=s.replace('Matrix3D(const Vector3 &a, float angle)','Matrix3D(const Vector3 &a, const float &angle)')
        if bits[8]:s=s.replace('const Vector3 &t,float p) {s.X','const Vector3 &t,const float &p) {s.X')
        if bits[9]:s=s.replace('targetPos.Z+=v->pos.z+v->targetZAdjustment;','float liveTargetHeight=v->pos.z+v->targetZAdjustment;targetPos.Z+=liveTargetHeight;')
        yield 'lifetimes_'+''.join(map(str,bits)),s
h.lifetimes=lifetimes


def matrix_aliases():
    for bits in itertools.product(range(2),repeat=8):
        s=h.BASE
        for method,enabled in [('mulVector3',bits[6]),('mulCameraAngleVector',bits[7])]:
            if not enabled:continue
            a=s.index('    __forceinline void '+method);b=s.index('\n    }',a)+6;part=s[a:b]
            start=part.index('{')+1;body=part[start:];decl=''
            if bits[0]:
                body=body.replace('Row[','matrix.Row[');decl+='const Matrix3D &matrix=*this;'
            if bits[1]:
                body=body.replace('v.X','input.X').replace('v.Y','input.Y').replace('v.Z','input.Z');decl+='const Vector3 &input=v;'
            for i in range(3):
                if bits[2+i]:
                    row=('matrix.' if bits[0] else '')+'Row[%d]'%i
                    body=body.replace(row,'row%d'%i);decl+='const Vector4 &row%d=%s;'%(i,row)
            if bits[5]:body=body.replace('v.Set(x,y,z);','output.Set(x,y,z);');decl+='Vector3 &output=v;' if bits[5] else ''
            part=part[:start]+decl+body;s=s[:a]+part+s[b:]
        yield 'matrix_aliases_'+''.join(map(str,bits)),s
h.matrix_aliases=matrix_aliases
h.run(sys.argv[1])
