import sys,itertools,json,time
from pathlib import Path
from concurrent.futures import ThreadPoolExecutor
sys.path.insert(0,'build/fxinit-seat22')
import search as h
h.HERE=Path('build/fxinit-live22').resolve()
UV='''Vector2 *MeshMatDescClass::Get_UV_Array_By_Index(int index,bool create) {
 if(create && !UV[index]) UV[index]=new UVBufferClass(VertexCount,"MeshMatDescClass::UV");
 if(!create && m_unknown0C!=0 && index<2) return Get_UV_Array(0,index);
 if(UV[index]) return UV[index]->Get_Array();
 return 0;
}
'''
LEG='''Vector2 *MeshMatDescClass::Get_UV_Array(int pass,int stage) SUFFIX {
 if(UVSource[pass][stage]==-1) return 0;
 if(UV[UVSource[pass][stage]]) return UV[UVSource[pass][stage]]->Get_Array();
 return 0;
}
'''
def variants():
 for visible,uvattr,legacy,placement,modelthrow in itertools.product(range(2),range(3),range(4),range(2),range(2)):
  src=h.BASE
  src=src.replace('class MeshMatDescClass {','class UVBufferClass : public ShareBufferClass<Vector2> {public: UVBufferClass(int,const char *);char pad10[8];unsigned crc;};\nclass MeshMatDescClass {')
  suffix='throw()'if legacy==3 else ''
  src=src.replace(' Vector2 *Get_UV_Array_By_Index(int,bool);',' Vector2 *Get_UV_Array_By_Index(int,bool);\n Vector2 *Get_UV_Array(int,int) '+suffix+';')
  src=src.replace(' char m_pad08[0x50-0x08];',' int PolyCount; int m_unknown0C; UVBufferClass *UV[8];int UVSource[4][2];')
  body=''
  if legacy:body+=('inline __declspec(noinline) 'if legacy==2 else'inline ')+LEG.replace('SUFFIX',suffix)
  if visible:body+=['inline ','inline __declspec(noinline) ','__forceinline '][uvattr]+UV
  if placement==0:src=src.replace('inline unsigned *MeshMatDescClass::Get_Color_Array',body+'\ninline unsigned *MeshMatDescClass::Get_Color_Array')
  else:src+='\n'+body
  if modelthrow:src=src.replace('const Vector2 *Get_UV_Array_By_Index(int index) {','const Vector2 *Get_UV_Array_By_Index(int index) throw() {')
  yield f'uvvisibility_{visible}{uvattr}{legacy}{placement}{modelthrow}',src
if __name__=='__main__':
 start=time.time()
 with ThreadPoolExecutor(max_workers=8) as pool:r=list(pool.map(h.run_one,variants()))
 r.sort(key=lambda x:(x.get('nd',99999),x.get('cost',99999)))
 (h.HERE/'uvvisibility-results.json').write_text(json.dumps(r,indent=2))
 print(json.dumps({'count':len(r),'valid':sum('nd'in x and not x.get('unresolved')for x in r),'seconds':time.time()-start,'best':r[:5]},indent=2))
