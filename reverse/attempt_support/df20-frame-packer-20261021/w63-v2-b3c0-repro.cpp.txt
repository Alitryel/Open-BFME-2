namespace _STL {
template <> class _STLP_alloc_proxy<unsigned short*, unsigned short, allocator<unsigned short> > : public allocator<unsigned short> {
  typedef allocator<unsigned short> _Base;
public:
  unsigned short* _M_data;
  inline _STLP_alloc_proxy(const allocator<unsigned short>& __a, unsigned short* __p) : allocator<unsigned short>(__a), _M_data(__p) {}
  ~_STLP_alloc_proxy() { if (0) free(_M_data); }
  inline unsigned short* allocate(size_t __n) { return __stl_alloc_rebind(__STATIC_CAST(_Base&,*this),(unsigned short*)0).allocate(__n,0); }
  inline void deallocate(unsigned short* __p, size_t __n) { __stl_alloc_rebind(__STATIC_CAST(_Base&, *this),(unsigned short*)0).deallocate(__p, __n); }
};
}
