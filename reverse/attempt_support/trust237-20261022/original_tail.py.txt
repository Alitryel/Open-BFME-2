from pathlib import Path
import sys,json,subprocess,concurrent.futures,itertools
sys.path.insert(0,str(Path('build/trust237-session').resolve()))
import search
from search import OUT,ROOT,NAME,RVA,SIZE,TARGET,build,resolve_bytes
K='0x800b0100L'
tails=[
 'return status==0 || status==K;',
 'int r=0; if(status==0) r=1; if(status==K) r=1; return r;',
 'int r=0; if(status==0) r=1; else if(status==K) r=1; return r;',
 'int r=1; if(status!=0) { if(status!=K) r=0;} return r;',
 'int r=0; do {if(status==0) break; if(status==K) break; return 0;} while(0); return r+1;',
 'int r=0; while(status!=0) { if(status==K) break; return 0;} return r+1;',
 'int r=0; if(status!=0 && status!=K) goto done; ++r; done:return r;',
 'int r=0; if(status==0) goto yes; if(status==K) goto yes; return r; yes: return r+1;',
 'int r=0; if(status==0) goto yes; if(status!=K) goto done; yes:++r; done:return r;',
 'return (status==0) | (status==K);',
 'return (status==0) + (status==K);',
 'int r=0; if(status==0 || status==K) r=1; return r;',
]
for repeats in range(1,9):
 cond=' || '.join(['status==0','status==K']*repeats)
 tails+=['return '+cond+';', 'int r=0; if('+cond+') ++r; return r;']
 cond=' && '.join(['status!=0','status!=K']*repeats)
 tails+=['if('+cond+') return 0; return 1;']

def run(job):
 n,tail,copy,boolean,early=job
 src=OUT/f'u{n:04}.cpp';obj=src.with_suffix('.obj')
 text=search.make_source(n,0,early,0,0,0,0,'/O1 /G7 /arch:SSE',0)
 tail=tail.replace('K',K)
 if boolean:tail=tail.replace('int r=','bool r=')
 decl=['','const long copied=status; ','long copied=status; ','const long &copied=status; '][copy]
 if copy:tail=tail.replace('status','copied')
 text=text.replace('LocalFree(argv); return 1;',decl+'LocalFree(argv); '+tail)
 src.write_text(text)
 cmd,env=build.compiler_command(src,obj);proc=subprocess.run(cmd,env=env,cwd=ROOT,capture_output=True,text=True)
 if proc.returncode:return {'n':n,'error':(proc.stdout+proc.stderr)[-600:]}
 raw,rel=build.read_object_symbol_bytes(obj,NAME,code_only=True)
 blob,un=resolve_bytes(raw,rel,TARGET+b'\0'*max(0,len(raw)-SIZE),RVA,len(raw),{'_memset':[0x6291ae],'_WinVerifyTrust@12':[0x655652]})
 diff=[i for i,(a,b) in enumerate(itertools.zip_longest(blob,TARGET)) if a!=b]
 return {'n':n,'tail':tail,'copy':copy,'boolean':boolean,'early':early,'size':len(raw),'outside':[i for i in diff if not 0xcd<=i<0xdd],'diff':len(diff),'last':blob[-42:].hex()}
jobs=[(n,*j) for n,j in enumerate(itertools.product(tails,range(4),range(2),range(2)))]
with concurrent.futures.ThreadPoolExecutor(max_workers=8) as pool:rows=list(pool.map(run,jobs))
(OUT/'original_tail.json').write_text(json.dumps(rows,indent=2))
good=[r for r in rows if 'error' not in r];print('TOTAL',len(rows),'GOOD',len(good))
for r in sorted(good,key=lambda r:(len(r['outside']),r['diff']))[:8]:print(json.dumps(r))
