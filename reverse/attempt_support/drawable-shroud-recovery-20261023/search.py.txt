import concurrent.futures as cf
import difflib
import json
import re
import subprocess
import sys
import time
from pathlib import Path

ROOT = Path(__file__).resolve().parents[2]
sys.path.insert(0, str(ROOT / 'tools'))
import build
import capstone

OUT = Path(__file__).resolve().parent
NAME = '?onComingOutOfShroud@Drawable@@AAEXXZ'
RVA, SIZE = 0x27893e, 747
BASE = subprocess.check_output(['git','show',
    '060ad76780dced93c6d7c28bc7a3fda246b98464:reverse/attempts/0x0027893e.cpp'],cwd=ROOT,text=True)
TARGET = build.read_target_bytes(RVA, SIZE)
SYMBOL_MAP = build.load_symbol_map()

def insts(blob, relocs=()):
    md = capstone.Cs(capstone.CS_ARCH_X86, capstone.CS_MODE_32)
    md.detail = True
    seq = list(md.disasm(blob, 0))
    index = {i.address:n for n,i in enumerate(seq)}
    sites = {offset for offset, _, _ in relocs}
    normalized = []
    for i in seq:
        op = i.op_str
        if i.group(capstone.CS_GRP_CALL) and i.operands[0].type==capstone.x86.X86_OP_IMM:
            op='CALL'
        elif i.group(capstone.CS_GRP_JUMP):
            op='BRANCH'
        elif any(i.address <= r < i.address+i.size for r in sites):
            op=re.sub(r'(?<![a-z0-9_])(?:0x[0-9a-f]+|\d+)(?![a-z0-9_])','ADDR',op)
        else:
            op=re.sub(r'0x[0-9a-f]{6,}', 'ADDR', op)
        normalized.append((i.mnemonic, op))
    return seq, normalized

TSEQ, TNORM = insts(TARGET)

def score(obj):
    raw, relocs = build.read_object_symbol_bytes(obj, NAME, code_only=True)
    # The parent owns one COMDAT; trailing CC is padding rather than code.
    raw = raw.rstrip(b'\xcc')
    from variant import resolve_bytes
    resolved, unresolved = resolve_bytes(raw,relocs,TARGET,RVA,SIZE,SYMBOL_MAP)
    seq, norm = insts(raw, relocs)
    target = list(TNORM)
    sm = difflib.SequenceMatcher(None, norm, target, autojunk=False)
    distance = sum(max(a2-a1,b2-b1) for tag,a1,a2,b1,b2 in sm.get_opcodes() if tag!='equal')
    return {'size':len(raw), 'insns':len(seq), 'nd':distance,
            'exact':len(raw)==SIZE and resolved==TARGET and not unresolved,
            'fit':round(sm.ratio(),9), 'sha':build.hashlib.sha256(raw).hexdigest()}

def run(item):
    label, text = item
    src, obj = OUT/(label+'.cpp'), OUT/(label+'.obj')
    src.write_text(text)
    cmd, env = build.compiler_command(src, obj)
    p = subprocess.run(cmd, cwd=ROOT, env=env, capture_output=True, text=True)
    if p.returncode:
        return {'id':label, 'error':(p.stdout+p.stderr)[-900:]}
    result = score(obj)
    result['id'] = label
    return result

def replacements(text, needle, choices):
    parts = text.split(needle)
    assert len(parts)==len(choices)+1, (needle,len(parts),len(choices))
    return ''.join(p+c for p,c in zip(parts, choices))+parts[-1]

def qualified(mask):
    text = BASE
    regular = '!obj->isEffectivelyDead()'
    volatile = '!(*(const volatile unsigned char *)((const char *)obj + 0x438) & 1)'
    text = replacements(text, regular, [volatile if mask & (1<<i) else regular for i in range(3)])
    # The two shared-tail predecessor status reads are distinct real loads.
    text = replacements(text, '!(m_status & mask80)', [
        '!(*(const volatile unsigned int *)&m_status & mask80)' if mask&(1<<(3+i)) else '!(m_status & mask80)'
        for i in range(2)])
    for bit, field in [(5,'m_revealedEvent'),(6,'m_shroudedEvent')]:
        if mask&(1<<bit):
            text=text.replace('notifyEvent = tmpl->'+field+';',
                              'notifyEvent = *(const volatile int *)&tmpl->'+field+';')
    if mask&128:
        text=text.replace('Object *obj = m_object;', 'Object *obj = *(Object * const volatile *)&m_object;')
    return text

def split_tail(text):
    start=text.index('completeNotify:')
    work=text[start+len('completeNotify:'):text.rfind('}')].strip()
    work=work.replace('notifyEvent','event')
    text=text[:start]+'}\n'
    for field in ['m_revealedEvent','m_shroudedEvent']:
        old='notifyEvent = tmpl->'+field+';\n\t\t\tgoto completeNotify;'
        new='int event = tmpl->'+field+';\n'+work
        assert old in text
        text=text.replace(old,new)
    return text.replace('\tint notifyEvent;\n','')

def duplicate(text,site,pred):
    sites=[
      'Rva004A1828Holder *holder = (Rva004A1828Holder *)Rva004A1828Get((Rva004A1828Owner *)obj);',
      'Player *controller = obj->getControllingPlayer();',
      'data.event = dataEvent;',
      'Object *nearest = ((Drawable *)obj)->rva00272AD5();',
      'm_sightReported = true;',
      'Object *holderObject = TheGameLogic->findObjectByID(holder->getID());',
      'Drawable *draw = holderObject->getDrawable();',
      'Module *member = obj->findModule(key_CastleMemberBehavior);',
      'Object *castle = TheGameLogic->findObjectByID(member->m_castleID);',
      'Module *castleBehavior = castle->findModule(CastleBehavior::rva0003955DA());',
      'Drawable *castleDraw = castle->getDrawable();',
      '((Rva003974CE *)castleBehavior)->apply(rva002706F0, 0);',
      'm_status |= mask80;',
    ]
    predicates=['m_useAltSightedEvent','m_sightReported','obj != 0',
                'tmpl != 0','(m_status & 0x40) != 0','obj->isEffectivelyDead()']
    needle=sites[site];condition=predicates[pred]
    if needle.count(' = ') and '*' in needle.split(' = ')[0]:
        lhs,rhs=needle[:-1].split(' = ',1)
        name=lhs.split('*')[-1].strip()
        rep=lhs+'; if ('+condition+') '+name+' = '+rhs+'; else '+name+' = '+rhs+';'
    else:
        rep='if ('+condition+') { '+needle+' } else { '+needle+' }'
    assert needle in text
    return text.replace(needle,rep)

def main():
    family = sys.argv[1] if len(sys.argv)>1 else 'qual'
    if family=='rescore':
        results=[dict(score(obj),id=obj.stem) for obj in OUT.glob('*.obj')]
        (OUT/'rescored.json').write_text(json.dumps(results,indent=1))
        print(json.dumps(sorted(results,key=lambda r:(r['nd'],abs(r['size']-SIZE)))[:12],indent=1))
        return
    items = [('base', BASE)]
    if family=='qual':
        items += [('qual-%04d'%m,qualified(m)) for m in range(1,256)]
    elif family=='duplicate':
        items=[('shared',BASE),('split',split_tail(BASE))]
        for form,base in [('shared',BASE),('split',split_tail(BASE))]:
            items += [('%s-d%02d-p%d'%(form,site,pred),duplicate(base,site,pred))
                      for site in range(13) for pred in range(6)]
    else:
        raise SystemExit(family)
    start=time.time()
    results=[]
    with cf.ThreadPoolExecutor(max_workers=8) as pool, (OUT/(family+'.jsonl')).open('w') as log:
        for result in pool.map(run,items):
            results.append(result)
            log.write(json.dumps(result)+'\n'); log.flush()
            if result.get('exact'):
                print('EXACT',json.dumps(result),flush=True)
    valid=[r for r in results if 'error' not in r]
    print(json.dumps({'family':family, 'generated':len(items), 'compiled':len(valid),
                      'seconds':round(time.time()-start),
                      'unique':len({r['sha'] for r in valid}),
                      'best':sorted(valid,key=lambda r:(r['nd'],abs(r['size']-SIZE)))[:12],
                      'errors':[r for r in results if 'error' in r][:3]},indent=2),flush=True)

if __name__=='__main__': main()
