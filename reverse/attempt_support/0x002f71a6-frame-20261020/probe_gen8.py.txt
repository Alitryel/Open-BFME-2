"""Frame-order probe 8 (big frame): exact density ties between different sizes; which wins, by use order."""
from pathlib import Path
import itertools
H = Path('/work3/Open-BFME-2/build/h72/p1')
HDR = """// cl: /O1 /DNDEBUG /MD /arch:SSE /G7 /EHsc
struct P3 { float x, y, z; };
struct S2 { int a, b; };
struct S4 { int a, b, c, d; };
struct S6 { int a, b, c, d, e, f; };
void extp(P3 *); void extq(void *); void exta(int *);
"""
for f in H.glob('t8*.cpp'):
    f.unlink()
CASES = {'S4a': ('S4', 3, 4), 'S4b': ('S4', 6, 8), 'S6a': ('S6', 2, 4), 'S6b': ('S6', 1, 2), 'S2a': ('S2', 3, 2), 'S2b': ('S2', 6, 4)}
for (cn, (qt, r, k)), use in itertools.product(CASES.items(), ('pq', 'qp', 'mix')):
    body = f' P3 p; {qt} q; int arr[40];\n exta(arr);\n'
    if use == 'pq':
        body += ' extp(&p);' * r + '\n' + ' extq(&q);' * k + '\n'
    elif use == 'qp':
        body += ' extq(&q);' * k + '\n' + ' extp(&p);' * r + '\n'
    else:
        body += ' extq(&q);' + ' extp(&p);' * r + ' extq(&q);' * (k - 1) + '\n'
    src = HDR + 'void probe()\n{\n' + body + '}\n'
    (H / f't8{cn}_r{r}k{k}_{use}.cpp').write_text(src)
