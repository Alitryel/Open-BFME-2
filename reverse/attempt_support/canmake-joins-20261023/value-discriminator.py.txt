from pathlib import Path
import runpy,itertools,concurrent.futures,json,re
OUT=Path('build/canmake-joins-20261023/value-discriminator').resolve();OUT.mkdir(parents=True,exist_ok=True)
r=runpy.run_path('reverse/attempt_support/canmake-00596e15-20261022/replay.py.txt');r['run'].__globals__['OUT']=OUT
text=runpy.run_path('reverse/attempt_support/canmake-alias-20261023/context.py.txt')['bank_text']()
prefix,body=text.rsplit('int AIBuildableWallSegment::canMake(Player *player)',1)
conditions=['failed','result.m_20==7','result.m_20!=0 && result.m_20!=7','player->m_money<(unsigned int)result.m_pad04']
pattern=r'\{selected.code=(2|9);return selected.code;\}'
assert len(re.findall(pattern,body))==4

def replacement(form,c,v,local,expr,site):
 choice='chosen'+str(site)if local else 'selected.code'
 decl='int '+choice+';'if local else ''
 c='(unsigned int)('+c+')'
 x=[f'({c})?{v}:0',f'({c})?{v}:{11-v}',f'({c})*{v}',f'({v}u*2u)>>({c})'][expr]
 head=decl+choice+'='+x+';'
 ret='selected.code='+choice+';return selected.code;'if local else 'return selected.code;'
 statements=[
  f'switch({choice}){{case {v}:{{{ret}}}default:return 0;}}',
  f'switch({choice}){{case {v}:selected.code={v};return selected.code;default:return 0;}}',
  f'if({choice}=={v}){{{ret}}}return 0;',
  f'if({choice}){{{ret}}}return 0;',
  f'do{{if({choice}!={v})break;{ret}}}while(0);return 0;',
  f'while({choice}){{{ret}}}return 0;',
  f'if({choice}!={v}){choice}={v};{ret}',
  f'while({choice}!={v}){choice}={v};{ret}',
  f'do{{if({choice}=={v})break;{choice}={v};}}while(1);{ret}',
  f'switch({choice}){{case 0:break;case {11-v}:break;default:{{{ret}}}}}return 0;',
  f'if({choice}=={v}){{selected.code={v};return selected.code;}}return 0;',
  f'if({choice}!={v})return 0;{ret}',
 ]
 return '{'+head+statements[form]+'}'
tests=[]
for site,form,local,expr in itertools.product(range(5),range(12),range(2),range(4)):
 counter=iter(range(4))
 def replace(m):
  i=next(counter)
  return replacement(form,conditions[i],int(m[1]),local,expr,i)if site in (i,4)else m[0]
 b=re.sub(pattern,replace,body)
 tests.append((f'discrim-{site}-{form}-{local}{expr}',prefix+'\nint AIBuildableWallSegment::canMake(Player *player)'+b))
import os
case=os.environ.get('CANMAKE_CASE')
if case:tests=[x for x in tests if x[0]==case];assert tests,case
results=[]
with concurrent.futures.ThreadPoolExecutor(max_workers=8)as pool:
 for i,d in enumerate(pool.map(r['run'],tests),1):
  results.append(d)
  if i%120==0 or d.get('exact_candidate'):print(json.dumps(dict(done=i,total=len(tests),best=sorted(results,key=lambda x:(x.get('nd',999),-x.get('score',0)))[:1])),flush=True)
(OUT/'results.json').write_text(json.dumps(results,indent=2)+'\n')
print(json.dumps(dict(trials=len(results),errors=sum('error'in x for x in results),unresolved=sum(bool(x.get('unresolved'))for x in results),exact=sum(x.get('exact_candidate',False)for x in results),bodies=len({x.get('masked_code_sha256')for x in results}))),flush=True)
