from pathlib import Path
import itertools,json
W=Path(__file__).resolve().parent
base=(W/'base.cpp').read_text()
head,old=base.split('void BfmeThingCGD::bfmeTwoCGD() {',1)
D=W/'edges';D.mkdir(exist_ok=True)
preds=['node->state==1','node->state!=2','node->state&1','node->key0&1',
       'node->key1<node->key0','node->first==node->second','node->index_dummy',
       'node->first->source==node->second->source']
# A deliberately inert comparison may have either value, but each arm performs
# the same address assignment. No added source read is volatile or effectful.
preds[6]='index&1'
test='node->state=node->first->source->test(node->second->source->prepare(node->point,node->normal))?1:2;'
rows=[]
for n,(shape,pred,fields,scope,rev,bridge,mixed) in enumerate(itertools.product(range(8),range(8),range(3),range(2),range(2),range(2),range(2))):
 decl='const ContactVector *p; const ContactVector *n;'
 assign=['p=&node->point; n=&node->normal;',
         'p=&node->point; n=0;', 'n=&node->normal; p=0;'][fields]
 # Unused null aliases are never dereferenced.
 a=assign;P=preds[pred]
 block=[
  f'if(node->state==0) {{ {test} if({P}) goto ready; {a} }} else {{ if({P}) {{ ready: {a} }} else {{ {a} }} }}',
  f'if(node->state!=0) {{ if({P}) {{ ready: {a} }} else {{ {a} }} }} else {{ {test} if({P}) goto ready; {a} }}',
  f'if(node->state==0) {{ {test} if({P}) goto ready; }} else if({P}) goto ready; {a} goto prepared; ready: {a} prepared:;',
  f'if(node->state==0) {{ {test} if({P}) {{ ready: {a} }} else {{ {a} }} }} else {{ if({P}) goto ready; {a} }}',
  f'if(node->state==0) {{ {test} {a} }} else {{ {a} }}',
  f'if(node->state==0) {{ {a} {test} }} else {{ {a} }}',
  f'if(node->state==0) {{ {test} }} if({P}) {{ ready: {a} }} else {{ {a} }}',
  f'if(node->state==0) {{ {test} }} if({P}) goto ready; {a} goto prepared; ready: {a} prepared:;'
 ][shape]
 point='*p' if fields!=2 else 'node->point'
 normal='*n' if fields!=1 else 'node->normal'
 first=f'bridge(node->second,{point},{normal})' if bridge else f'node->second->source->prepare({point},{normal})'
 rpoint=point if rev else 'node->point'
 rnormal='ContactVector('+normal+').negate()' if rev else 'ContactVector(node->normal).negate()'
 last=f'altPrepare(node->first->source,{rpoint},{rnormal})' if mixed else f'node->first->source->prepare({rpoint},{rnormal})'
 body='''index=0; current=buckets[0];
 for(;;) {
  while(current==0) {int i=index+1; if(i==0x493) return; index=i; current=buckets[i];}
  ContactNode *node=current; current=current->next;
  if(node==0) return;
  if(node->first->source==0 || node->second->source==0) continue;
  BLOCK
  if(node->state==2) continue;
  node->first->source->respond(FIRST);
  if(node->first->source==0 || node->second->source==0) continue;
  node->second->source->respond(LAST);
 }
}'''.replace('BLOCK',('' if scope else decl)+'\n'+block).replace('FIRST',first).replace('LAST',last)
 body=('\n'+decl+'\n' if scope else '\n')+body
 src=head+'void BfmeThingCGD::bfmeTwoCGD() {'+body+'\n'
 (D/('v%04d.cpp'%n)).write_text(src)
 rows.append(dict(name='v%04d'%n,shape=shape,pred=pred,fields=fields,scope=scope,reverse=rev,bridge=bridge,mixed=mixed))
(W/'edges-models.json').write_text(json.dumps(rows,indent=1))
print(len(rows),'edge/phi candidates')
