import sys, subprocess, json, concurrent.futures, difflib
from pathlib import Path
sys.path.insert(0,str(Path.cwd()/'tools'))
import build, capstone
from locate import object_functions
ROOT=Path.cwd(); NAME='?onComingOutOfShroud@Drawable@@AAEXXZ'; RVA=0x27893e; SIZE=747
TARGET=build.read_target_bytes(RVA,SIZE)
def score(obj):
    raw,rel=next((x[1],x[2]) for x in object_functions(obj) if x[0]==NAME)
    md=capstone.Cs(capstone.CS_ARCH_X86,capstone.CS_MODE_32)
    a=list(md.disasm(raw,0));b=list(md.disasm(TARGET,0))
    ai={i.address:n for n,i in enumerate(a)};bi={i.address:n for n,i in enumerate(b)}
    mask=set(k for o,t,n in rel for k in range(o,o+4))
    sm=difflib.SequenceMatcher(a=[i.mnemonic for i in a],b=[i.mnemonic for i in b],autojunk=False)
    ops=sm.get_opcodes();amap={i:j for tag,x1,x2,y1,y2 in ops if tag=='equal' for i,j in zip(range(x1,x2),range(y1,y2))}
    def normalized(x,y):
        if x.mnemonic.startswith('j') and y.mnemonic==x.mnemonic:
            try:return ('branch',amap.get(ai.get(int(x.op_str,16)))),('branch',bi.get(int(y.op_str,16)))
            except ValueError:pass
        rr=[k-x.address for k in range(x.address,x.address+x.size) if k in mask]
        if rr:
            aa=bytearray(x.bytes);bb=bytearray(y.bytes)
            for k in rr:
                if k<len(bb):aa[k]=bb[k]=0
            return bytes(aa),bytes(bb)
        return bytes(x.bytes),bytes(y.bytes)
    nd=0;regions=0;diff=[]
    for tag,x1,x2,y1,y2 in ops:
        if tag!='equal':
            nd+=max(x2-x1,y2-y1);regions+=1
            diff.append([tag,b[y1].address if y1<len(b) else SIZE,x2-x1,y2-y1]);continue
        for i,j in zip(range(x1,x2),range(y1,y2)):
            na,nb=normalized(a[i],b[j])
            if na!=nb:
                nd+=1;diff.append([b[j].address,a[i].mnemonic+' '+a[i].op_str,b[j].mnemonic+' '+b[j].op_str])
    return dict(size=len(raw),insns=len(a),nd=nd,regions=regions,diff=diff)
def trial(src):
    obj=src.with_suffix('.obj')
    command,env=build.compiler_command(src,obj)
    try:
        if '--score' not in sys.argv:
            p=subprocess.run(command,cwd=ROOT,env=env,capture_output=True,text=True,timeout=100)
            if p.returncode:return dict(src=str(src.relative_to(ROOT)),error=(p.stdout+p.stderr)[-700:])
        r=score(obj);r['src']=str(src.relative_to(ROOT));return r
    except BaseException as e:return dict(src=str(src.relative_to(ROOT)),error=str(e))
if __name__=='__main__':
    path=Path(sys.argv[1]);sources=sorted(path.glob(next((x[10:]for x in sys.argv if x.startswith('--pattern=')),'*.cpp'))) if path.is_dir() else [path]
    out=Path(next((x[9:]for x in sys.argv if x.startswith('--output=')),str(path.with_suffix('.jsonl'))))
    with out.open('w') as f,concurrent.futures.ThreadPoolExecutor(max_workers=8) as pool:
        for r in pool.map(trial,[s.resolve() for s in sources]):
            f.write(json.dumps(r)+'\n');f.flush()
    rs=[json.loads(l) for l in out.read_text().splitlines()]
    good=[r for r in rs if 'nd'in r]
    print(json.dumps(dict(trials=len(rs),compiled=len(good),best=sorted(good,key=lambda r:(r['nd'],abs(r['size']-SIZE)))[:8]),indent=2))
