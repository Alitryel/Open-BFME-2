from pathlib import Path
import sys,itertools,concurrent.futures,json,re
sys.path.insert(0,'build/df20-scalars-20261020');import trials
trials.scratch=Path.cwd()/'build/df20-scalars-20261020'
base=(trials.scratch/'range-x0-specialized-b1-r0-ttemporary.cpp').read_text();trials.symbol=base.splitlines()[0][3:]
start=base.index('  string_type __sign =');end=base.index('#pragma inline_depth(255)',start)
old=base[start:end]
specs=[]
for form,depth,visible,which in itertools.product(['ternary','if','direct'],[0,1,255],[0,1],['sign','currency']):
 s=base
 if visible:s=s.replace('template <> _String_base<wchar_t, allocator<wchar_t> >::~_String_base();','')
 if which=='sign':
  if form=='ternary':expr='return neg ? p.negative_sign() : p.positive_sign();'
  elif form=='if':expr='if (neg) return p.negative_sign(); return p.positive_sign();'
  else:expr='return p.negative_sign();' # each arm uses a helper with its respective direct getter
  if form!='direct':
   helper=f'template<class Punct> __forceinline basic_string<wchar_t> __money_sign(const Punct& p, bool neg) {{ {expr} }}'
   call='  string_type __sign = __intl ? __money_sign(__punct_intl,__is_negative) : __money_sign(__punct,__is_negative);\n'
  else:
   helper='template<class Punct> __forceinline basic_string<wchar_t> __money_negative(const Punct& p) { return p.negative_sign(); }\ntemplate<class Punct> __forceinline basic_string<wchar_t> __money_positive(const Punct& p) { return p.positive_sign(); }'
   call=old
   for p in ['__punct_intl','__punct']:
    call=call.replace(p+'.negative_sign()',f'__money_negative({p})').replace(p+'.positive_sign()',f'__money_positive({p})')
  s=s.replace(old,call)
 else:
  # A direct-return helper can forward the hidden result; a ternary helper
  # tests whether placing the conditional inside it changes temporary cleanup.
  if form=='ternary':
   helper='template<class Punct, class Intl> __forceinline basic_string<wchar_t> __money_currency(const Punct& p,const Intl& i,bool intl) { return intl ? i.curr_symbol() : p.curr_symbol(); }'
   replacement='__money_currency(__punct,__punct_intl,__intl)'
  elif form=='if':
   helper='template<class Punct, class Intl> __forceinline basic_string<wchar_t> __money_currency(const Punct& p,const Intl& i,bool intl) { if(intl) return i.curr_symbol(); return p.curr_symbol(); }'
   replacement='__money_currency(__punct,__punct_intl,__intl)'
  else:
   helper='template<class Punct> __forceinline basic_string<wchar_t> __money_currency(const Punct& p) { return p.curr_symbol(); }'
   replacement='__intl ? __money_currency(__punct_intl) : __money_currency(__punct)'
  pat=r'__intl \? __punct_intl.curr_symbol\(\)\s*: __punct.curr_symbol\(\)'
  s,n=re.subn(pat,replacement,s);assert n==1
 helper=f'#pragma inline_depth({depth})\n'+helper+'\n#pragma inline_depth(255)\n'
 marker='template <> _MoneyOut money_put<wchar_t, _MoneyOut>::do_put('
 s=s.replace(marker,helper+marker,1)
 specs.append((f'signcontext-{which}-{form}-d{depth}-v{visible}',s))
with concurrent.futures.ThreadPoolExecutor(max_workers=8) as p:r=list(p.map(trials.run,specs))
(trials.scratch/'sign_context.json').write_text(json.dumps(r,indent=2))
