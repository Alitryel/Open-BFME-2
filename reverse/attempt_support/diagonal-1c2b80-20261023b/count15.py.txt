import sys,os
sys.path.insert(0,'/work/Open-BFME-2/build/c2tr')  # wdbg.py.txt from this directory
from wdbg import Dbg
# count15.py src tag : count kind-15 and kind-3 symbol allocations while compiling DiagonalBlur
src=sys.argv[1]; tag=sys.argv[2]
W='/work/Open-BFME-2/build/db/tr/'
d=Dbg(src,W+tag+'.cnt.obj',W+tag+'.cnt.log')
active=False; n15=0; ids=[]; gs=[]
try:
    d.prompt();d.prompt('set $CanDeferOnBPByAddr = 1');d.prompt('break *0x10760539')
    while True:
        o=d.prompt('cont',300)
        if 'terminated' in o or not o.strip(): break
        if '0x10760539' in o:
            if active: continue
            w=d.words(d.reg('eax'),1);s=d.words(w[0],2);fn=d.string(s[1])
            if 'DiagonalBlur' in fn:
                active=True;d.prompt('break *0x1076045d');d.prompt('break *0x10760853')
            continue
        if active and '0x1076045d' in o:
            ar=d.words(0x108a79b0,1)[0]
            if ar:
                cur,end=d.words(ar+4,2)
                ids.append(d.words(cur+0x1c,1)[0] if cur<end else d.words(0x108a79d0,1)[0])
            else: ids.append(d.words(0x108a79d0,1)[0])
            gs.append(d.words(0x108a79d0,1)[0])
            continue
        if active and '0x10760853' in o:
            ids.append(-1); gs.append(d.words(0x108a79d0,1)[0])
            continue
finally:
    # kind-15 hits only via 0x1076045d path? 0x10760476 is shared by kind-3 refill path too
    k15=[x for x in ids if x>=0]
    f=next(i for i,x in enumerate(ids) if x>=0); b=next((i for i in range(f,len(ids)) if ids[i]<0),len(ids))
    print(tag,'n15',len(k15),'burst',b-f,'n3',ids.count(-1),flush=True)
    open(W+tag+'.cnt.ids','w').write(' '.join(map(str,ids)))
    open(W+tag+'.cnt.g','w').write(' '.join(map(str,gs)))
    os.path.exists(W+tag+'.cnt.obj') and os.unlink(W+tag+'.cnt.obj')
    if d.p.poll() is None:
        try: d.prompt('quit')
        except Exception: d.p.kill()
