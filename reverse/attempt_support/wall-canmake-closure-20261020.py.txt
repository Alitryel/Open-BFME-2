from pathlib import Path
import subprocess,sys,re,json
root=Path(__file__).resolve().parents[1]
out=root/'build/wall-canmake-closure-oct20';out.mkdir(exist_ok=True)
home=root/'Code/GameEngine/Source/GameLogic/SkirmishAI/AIWallBuilder'
seed=out/'seed.cpp'
if not seed.exists():seed.write_bytes((root/'reverse/attempts/0x00596e15.cpp').read_bytes())
base=seed.read_text()
base=re.sub(r'#include "([^"]+)"',lambda m:'#include "'+(home/m[1]).resolve().as_posix()+'"',base)
base=base.replace('register int zero = 0;','').replace('m_startNodeID == zero','m_startNodeID == 0').replace('if (0) { AsciiString dummy; }','')
base=base.replace('int rva0055ADD8(int ignored);','int failBuild(int ignored);').replace('->rva0055ADD8((int)player)','->failBuild((int)player)')
element='''class Rva004D9A3C { public: ~Rva004D9A3C(); private: char data[0x1C]; };\nvoid operator delete[](void *p);\n'''
cleanup='''\nRva0039205C::~Rva0039205C() { if(m_array08) { delete[] m_array08; m_array08=0; m_unk00=0; } }\n'''
tests={'baseline':base}
for tag,attr in [('visible-noinline','__declspec(noinline) '),('visible',''),('visible-forceinline','__forceinline ')]:
    s=base.replace('class Rva004D9A3C;',element,1).replace('~Rva0039205C();',attr+'~Rva0039205C();',1)+cleanup
    tests[tag]=s
    for budget in [40,100,200]:tests[tag+'-budget'+str(budget)]=s.replace('/O1 /G7','/O1 /Ob2 /d2inlS'+str(budget)+' /G7',1)
tests['baseline-inline100']=base.replace('/O1 /G7','/O1 /Ob2 /d2inlS100 /G7',1)
name='?canMake@AIBuildableWallSegment@@UAEHPAVPlayer@@@Z'
results=[]
for key,s in tests.items():
    p=out/(key+'.cpp');p.write_bytes(s.encode())
    r=subprocess.run([sys.executable,'tools/explain_mismatch.py',name,'--rva','0x00596E15','--size','218','--source',p.relative_to(root).as_posix()],cwd=root,capture_output=True,text=True,encoding='utf8',errors='replace',timeout=240)
    log=r.stdout+r.stderr;p.with_suffix('.log').write_bytes(log.encode())
    d={'key':key,'exit':r.returncode,'exact':'OK: bytes match' in log,'summary':[x for x in log.splitlines() if x.startswith(('compiled symbol size:','classification:','first diff:','unresolved:'))]}
    if r.returncode:d['error']=log[-900:]
    results.append(d);(out/'results.json').write_text(json.dumps(results,indent=2));print(json.dumps(d),flush=True)
