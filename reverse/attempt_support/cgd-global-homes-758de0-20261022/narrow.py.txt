from pathlib import Path
from itertools import product
W=Path(__file__).resolve().parent;out=W/'narrow';out.mkdir(exist_ok=True)
banks=[(W/'control.cpp').read_text(),(W/'base.cpp').read_text()]
for idx,(ty,ret,bank,nt) in enumerate(product(['bool','char','signed char','unsigned char','short','unsigned short','int','unsigned'],['bool','char','unsigned char','int'],range(2),range(2))):
 s=banks[bank]
 s=s.replace('virtual void *','virtual '+ty+' ')
 s=s.replace('inline void *','inline '+ty+' ')
 s=s.replace('virtual bool test(void *info)=0;','virtual '+ret+' test('+ty+' info)'+(' throw()' if nt else '')+'=0;')
 s=s.replace('virtual void respond(void *info)=0;','virtual void respond('+ty+' info)'+(' throw()' if nt else '')+'=0;')
 if nt:
  s=s.replace('const ContactVector &normal)=0;','const ContactVector &normal) throw()=0;')
 (out/('v%04d.cpp'%idx)).write_text(s)
print('generated',idx+1)
