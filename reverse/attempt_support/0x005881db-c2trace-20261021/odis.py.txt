import sys,capstone
sys.path.insert(0,'/work2/Open-BFME-2/tools')
import build
from pathlib import Path
SYM='?updateMeleeAttack@HordeMeleeAmoeba@@UAEXPAVObject@@@Z'
obj=Path(sys.argv[1]).resolve(); src=sys.argv[2]; which=sys.argv[3] if len(sys.argv)>3 else 'c'
lo,hi=(int(x,16) for x in (sys.argv[4] if len(sys.argv)>4 else '0-900').split('-'))
row={"name":SYM,"target_rva":"0x005881DB","target_size":"2199","source":src,"notes":""}
got=build.compile_function(row,build.load_symbol_map(),obj)
b=got['bytes'] if which=='c' else got['target']
md=capstone.Cs(capstone.CS_ARCH_X86,capstone.CS_MODE_32)
for i in md.disasm(bytes(b),0):
    if lo<=i.address<hi: print('%4x %s %s'%(i.address,i.mnemonic,i.op_str))
