import sys,struct
def members(lib):
    d=open(lib,'rb').read(); assert d[:8]==b'!<arch>\n'; p=8; lnames=None
    while p<len(d):
        h=d[p:p+60]; name=h[:16].decode('latin1').strip(); size=int(h[48:58]); body=d[p+60:p+60+size]
        if name=='//': lnames=body
        elif name.startswith('/') and name[1:].strip().isdigit() and lnames is not None:
            off=int(name[1:]); name=lnames[off:lnames.index(b'\0',off)].decode('latin1')
        yield name,body
        p+=60+size+(size&1)
def secs(obj):
    m,n=struct.unpack_from('<HH',obj,0); opt=struct.unpack_from('<H',obj,16)[0]
    for i in range(n):
        o=20+opt+40*i; nm=obj[o:o+8].rstrip(b'\0'); sz,ptr=struct.unpack_from('<II',obj,o+16)
        yield nm,obj[ptr:ptr+sz]
def dump(obj,want):
    out=[]
    for nm,data in secs(obj):
        if nm!=b'.debug$S': continue
        p=4
        while p+8<=len(data):
            t,l=struct.unpack_from('<II',data,p); body=data[p+8:p+8+l]; p+=8+l; p=(p+3)&~3
            if t!=0xf1: continue
            q=0; recs=[]; 
            while q+4<=len(body):
                rl,rt=struct.unpack_from('<HH',body,q); r=body[q+4:q+2+rl]; q+=2+rl
                def cstr(b): return b[:b.index(b'\0')].decode('latin1') if b'\0' in b else b.decode('latin1')
                if rt in (0x1110,0x110f): recs.append('PROC '+cstr(r[35:]))
                elif rt==0x110b: off,typ=struct.unpack_from('<iI',r,0); recs.append('BPREL %d %s'%(off,cstr(r[8:])))
                elif rt in (0x110c,0x110d): recs.append('DATA '+cstr(r[10:]))
                elif rt==0x1103: recs.append('BLOCK')
                elif rt==0x0006: recs.append('END')
                elif rt==0x1111: off,typ,reg=struct.unpack_from('<iIH',r,0); recs.append('REGREL %d r%d %s'%(off,reg,cstr(r[10:])))
                elif rt==0x1105: recs.append('LABEL '+cstr(r[7:]))
                else: recs.append('rec%04x'%rt)
            if any(want in x for x in recs if x.startswith('PROC')): out+=recs
    return out
if __name__=='__main__':
    src=sys.argv[1]
    if src.endswith('.lib'):
        for name,body in members(src):
            if 'pdblkup' in name.lower(): obj=body; break
    else: obj=open(src,'rb').read()
    for r in dump(obj,sys.argv[2]): print(r)
