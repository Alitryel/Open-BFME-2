COMMAND,ENV=build.compiler_command(ROOT/'reverse/attempts/0x0004826c.cpp',DIR/'unused.obj')

def run(item):
    tag,source,*extra=item
    src=DIR/(tag+'.cpp'); obj=DIR/(tag+'.obj')
    src.write_text(source)
    command=COMMAND[:-2]+(extra[0] if extra else [])+['-Fo'+obj.relative_to(ROOT).as_posix(),src.relative_to(ROOT).as_posix()]
    p=subprocess.run(command,cwd=ROOT,env=ENV,stdout=subprocess.PIPE,stderr=subprocess.STDOUT,text=True)
    (DIR/(tag+'.log')).write_text(p.stdout)
    if p.returncode: return {'tag':tag,'error':p.stdout[-1600:]}
    raw,relocs=build.read_object_symbol_bytes(obj,NAME)
    raw=raw.rstrip(b'\xcc')
    mask={j for off,kind,name in relocs for j in range(off,off+4)}
    nd=sum(x!=y for i,(x,y) in enumerate(zip(raw,TARGET)) if i not in mask)+abs(len(raw)-len(TARGET))
    ours=list(MD.disasm(raw,0))
    blocks=list(difflib.SequenceMatcher(None,[i.mnemonic for i in TINS],[i.mnemonic for i in ours],autojunk=False).get_opcodes())
    regions=sum(kind!='equal' for kind,a,b,c,d in blocks)
    slots=[]; others=[]
    for kind,a,b,c,d in blocks:
        if kind=='equal':
            for t,o in zip(TINS[a:b],ours[c:d]):
                if t.op_str!=o.op_str and not any(o.address<=j<o.address+o.size for j in mask):
                    line=[t.address,t.mnemonic,t.op_str,o.op_str]
                    (slots if 'ebp' in t.op_str and 'ebp' in o.op_str else others).append(line)
    result={'tag':tag,'size':len(raw),'nd':nd,'regions':regions,'slots':len(slots),'others':len(others),'sig':slots[:5]}
    (DIR/(tag+'.json')).write_text(json.dumps(dict(result,diff=slots+others),indent=2))
    return result

def batch(items,label):
    start=time.time()
    with concurrent.futures.ThreadPoolExecutor(max_workers=8) as pool:
        results=list(pool.map(run,items))
    (DIR/(label+'-results.json')).write_text(json.dumps(results,indent=2))
    ok=[r for r in results if 'error' not in r]
    print(json.dumps({'label':label,'n':len(results),'secs':round(time.time()-start,1),'errors':[r for r in results if 'error' in r][:2], 'best':sorted(ok,key=lambda r:(r['regions'],r['nd']))[:8], 'counts':{str(k):sum((r['size'],r['nd'],r['regions'])==k for r in ok) for k in sorted(set((r['size'],r['nd'],r['regions']) for r in ok))}},indent=2),flush=True)
    return results
