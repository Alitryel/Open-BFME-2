import sys,itertools,re
sys.path.insert(0,'build/netpacket-20261022');from search import *
from next import single_index
single=single_index(BANK)
# Disassembly shows only memcpy arms induce the stores that spoil single-index
# register residency. Capture the consumed offset in a local value across calls.
fields=['commandType','timestamp','frame','playerID','relay','commandID']
types=['UnsignedByte','UnsignedInt','UnsignedInt','UnsignedByte','UnsignedByte','UnsignedShort']
items=[]
for modes in itertools.product(range(3),repeat=6):
 b=single
 for field,typ,mode in zip(fields,types,modes):
  old='\t\t\t++i;\n\t\t\tmemcpy(&'+field+', m_packet + i, sizeof('+typ+'));\n\t\t\ti += sizeof('+typ+');'
  assert old in b
  if mode==0:continue
  if mode==1:
   new='\t\t\tInt consumed = i + 1;\n\t\t\tmemcpy(&'+field+', m_packet + consumed, sizeof('+typ+'));\n\t\t\ti = consumed + sizeof('+typ+');'
  else:
   new='\t\t\tInt consumed = ++i;\n\t\t\tmemcpy(&'+field+', m_packet + consumed, sizeof('+typ+'));\n\t\t\ti = consumed + sizeof('+typ+');'
  b=b.replace(old,new,1)
 items.append(('copies_'+''.join(map(str,modes)),b))
batch(items,'copy-lifetimes')
