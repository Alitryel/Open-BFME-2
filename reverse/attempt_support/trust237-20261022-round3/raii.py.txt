from pathlib import Path
import itertools,sys
sys.path.insert(0,str(Path('build/trust237-round3').resolve()))
import search
# Ownership cleanup calls exactly the same API once, including on normal return.
# All predicates below are true for 32-bit unsigned long and have no memory effect.
preds=['true','(u&1u)<=1u','(unsigned char)u<=255u','((u|1u)&255u)!=0u','(u^(u+1u))!=0u','(unsigned __int64)u<=0xffffffffui64']
forms=['return P;','int r=0; if(P)++r; return r;','int r=0;if(P)goto yes;return r;yes:return r+1;','bool r=P;return r;']
jobs=[]
for pi,fi,life,cpu,directive,eh,decl in itertools.product(range(6),range(4),range(3),range(2),range(2),range(2),range(2)):
 hint=['','__forceinline '][directive]
 exc=['',' throw()'][eh]
 member=['void *value;','unsigned short **value;'][decl]
 owner='struct ArgvOwner {'+member+hint+'~ArgvOwner()'+exc+'{LocalFree(value);} };\n'
 tail=forms[fi].replace('P','('+preds[pi]+')')
 status='long status=WinVerifyTrust(0,&action,&data); unsigned long u=(unsigned long)status; '
 if life==0:
  tail=status+'{ArgvOwner owner={argv};} '+tail
 elif life==1:
  tail='ArgvOwner owner={argv}; '+status+tail
 else:
  tail='long status; {ArgvOwner owner={argv};status=WinVerifyTrust(0,&action,&data);} unsigned long u=(unsigned long)status; '+tail
 s=search.base.replace(search.marker,'    '+tail)
 s=s.replace('int Rva00237D3DCheck() {',owner+'int Rva00237D3DCheck() {')
 if not cpu:s=s.replace('/G7','/G6')
 jobs.append((f'r{len(jobs):04}',s,dict(pred=pi,form=fi,lifetime=life,cpu=cpu,forceinline=directive,throw=eh,pointertype=decl)))
search.execute('raii',jobs)
