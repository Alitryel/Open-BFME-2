import json,difflib,hashlib,sys
from pathlib import Path
import search
W=Path(__file__).resolve().parent
T=[(x[0],x[1],x[2]) for x in search.T]
summary={}
for family in ['fasttable','hybrid','packhelper','thiscall','altvirtual','fastpair']:
 rows=json.loads((W/(family+'-results.json')).read_text());unique={}
 for r in rows:
  if 'error' in r or r['sha'] in unique:continue
  data=(W/family/(r['name']+'.bin')).read_bytes();A=[(x[0],x[1],x[2]) for x in search.insns(data)]
  m=difflib.SequenceMatcher(None,[x[:2] for x in A],[x[:2] for x in T],autojunk=False)
  r['pair_bytes']=sum(sum(x[2] for x in T[c:d]) for tag,a,b,c,d in m.get_opcodes() if tag=='equal')
  r['fixed_regions']={label:sum(x!=y for x,y in zip(data[a:z],search.target[a:z]))+abs(len(data[a:z])-len(search.target[a:z])) for label,a,z in [('prefix',0,0x94),('test',0x94,0xaa),('response1',0xbe,0xe1),('reverse_tail',0x136,0x140)]}
  unique[r['sha']]=r
 best=sorted(unique.values(),key=lambda r:(-r['pair_bytes'],r['positional']))
 summary[family]={'variants':len(rows),'compiled':sum('error' not in r for r in rows),'distinct':len(unique),'best':best[:1]}
 print(family,summary[family]['variants'],summary[family]['compiled'],len(unique),best[0]['name'],best[0]['size'],best[0]['pair_bytes'],best[0]['positional'])
(W/'new-trials-summary.json').write_text(json.dumps(summary,indent=2)+'\n')
