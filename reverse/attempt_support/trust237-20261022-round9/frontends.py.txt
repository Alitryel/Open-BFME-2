from pathlib import Path
import sys,itertools,json,hashlib,concurrent.futures,subprocess,difflib
ROOT=Path.cwd();OUT=ROOT/'build/trust237-round9'
sys.path.insert(0,str(ROOT/'tools'))
import build,capstone
from variant import resolve_bytes
RVA=0x237d3d;SIZE=226;NAME='?Rva00237D3DCheck@@YAHXZ';TARGET=build.read_target_bytes(RVA,SIZE)
md=capstone.Cs(capstone.CS_ARCH_X86,capstone.CS_MODE_32)
retins=[i.mnemonic for i in md.disasm(TARGET,0)]
base=(ROOT/'reverse/attempt_support/trust237-20261022-round5/base.cpp.txt').read_text()
marker='    WinVerifyTrust(0,&action,&data);\n    LocalFree(argv);\n    return 1;'
assert marker in base
preds=['(u&1u)<=1u','(unsigned char)u<=255u','((u* (u+1u))&1u)==0u','(u|1u)!=0u']
tails=['return P;','{int r=0;if(P)goto yes;return r;yes:return r+1;}','{int r=0;if(!(P))goto done;++r;done:return r;}']

def c_source(text,hoist):
    for name in ['TrustGuid','TrustFile','TrustData']:
        text=text.replace('struct '+name+' {','typedef struct '+name+' {')
    text=text.replace('unsigned char d[8]; };','unsigned char d[8]; } TrustGuid;').replace('TrustGuid *subject; };','TrustGuid *subject; } TrustFile;').replace('unsigned long flags; };','unsigned long flags; } TrustData;')
    text=text.replace('extern "C" {','').replace('\n}\nint Rva','\nint Rva')
    # VS7.1 C requires declarations to precede statements in each block.
    text=text.replace('    int argc=0;','    {int argc=0;').replace('    TrustGuid action=','    {TrustGuid action=')
    text=text.replace('    long status=','    {long status=')
    text=text.rstrip()+'\n}\n}\n}\n'
    if hoist:
        text=text.replace('    TrustFile file;','    TrustFile file;\n    int argc;\n    unsigned short **argv;\n    TrustGuid action;\n    TrustData data;\n    long status;\n    unsigned long u;')
        text=text.replace('{int argc=0;','{argc=0;').replace('unsigned short **argv=','argv=')
        init='TrustGuid action={0x00AAC56B,0xCD44,0x11D0,{0x8C,0xC2,0,0xC0,0x4F,0xC2,0x95,0xEE}};'
        stores='action.a=0x00AAC56B;action.b=0xCD44;action.c=0x11D0;action.d[0]=0x8C;action.d[1]=0xC2;action.d[2]=0;action.d[3]=0xC0;action.d[4]=0x4F;action.d[5]=0xC2;action.d[6]=0x95;action.d[7]=0xEE;'
        text=text.replace(init,stores).replace('    TrustData data;\n    memset','    memset').replace('{long status=','{status=').replace('unsigned long u=(unsigned long)status;','u=(unsigned long)status;')
    return text

def run(job):
    name,text,meta=job
    src=OUT/(name+'.cpp');obj=src.with_suffix('.obj');src.write_text(text)
    sym=meta.get('symbol',NAME)
    try:
        cmd,env=build.compiler_command(src,obj)
        cmd.insert(-1,'-Fd'+str(obj.with_suffix('.pdb').relative_to(ROOT)))
        p=subprocess.run(cmd,cwd=ROOT,env=env,capture_output=True,text=True)
        row=dict(name=name,meta=meta,command=cmd,warnings=(p.stdout+p.stderr)[-1600:])
        if p.returncode:return dict(**row,error=row['warnings'])
        raw,rel=build.read_object_symbol_bytes(obj,sym,code_only=True)
        res,unres=resolve_bytes(raw,rel,TARGET+b'\0'*max(0,len(raw)-SIZE),RVA,len(raw),{'_memset':[0x6291ae],'_WinVerifyTrust@12':[0x655652]})
        decoder=capstone.Cs(capstone.CS_ARCH_X86,capstone.CS_MODE_32)
        ins=list(decoder.disasm(raw,0));mn=[i.mnemonic for i in ins]
        sm=difflib.SequenceMatcher(a=mn,b=retins,autojunk=False)
        eq=sum(c-b for a,b,c,d,e in sm.get_opcodes() if a=='equal')
        return dict(**row,size=len(raw),diff=sum(a!=b for a,b in itertools.zip_longest(res,TARGET)),prefix=next((i for i,(a,b) in enumerate(zip(res,TARGET)) if a!=b),min(len(raw),SIZE)),eq=eq,nop16=b'\x90'*16 in raw,nops=[i.address for i in ins if i.mnemonic=='nop'],pauses=[i.address for i in ins if i.mnemonic=='pause'],ret=raw[-8:].hex(),raw=raw.hex(),sha256=hashlib.sha256(raw).hexdigest(),unresolved=unres,relocations=rel)
    except BaseException as e:return dict(name=name,meta=meta,error=str(e))

def execute(label,jobs):
    with concurrent.futures.ThreadPoolExecutor(max_workers=8) as pool:rows=list(pool.map(run,jobs))
    (OUT/(label+'.json')).write_text(json.dumps(rows,indent=2)+'\n')
    good=[r for r in rows if 'error' not in r]
    print(label,'attempts',len(rows),'good',len(good),'distinct',len({r['sha256'] for r in good}),'nop16',sum(r['nop16'] for r in good),'with_nops',sum(bool(r['nops']) for r in good),flush=True)
    unique={r['sha256']:r for r in good}
    for r in sorted(unique.values(),key=lambda r:(r['diff'],-r['prefix']))[:8]:print(r['name'],r['meta'],'size',r['size'],'diff',r['diff'],'prefix',r['prefix'],'eq',r['eq'],'tail',r['ret'],flush=True)
    for r in [r for r in rows if 'error' in r][:5]:print('ERROR',r['name'],r['meta'],r['error'][-800:],flush=True)
    return rows

if __name__=='__main__':
    mode=sys.argv[1]
    repair_cpp=mode=='repair_cpp'
    if repair_cpp:mode='frontends'
    jobs=[]
    if mode=='frontends':
        for pred,tail,hoist,cpu,opt,frontend in itertools.product(range(4),range(3),range(2),[6,7],['O1','O2'],['cpp','externcpp','c']):
            text=base.replace(marker,'    long status=WinVerifyTrust(0,&action,&data);unsigned long u=(unsigned long)status;LocalFree(argv);'+tails[tail].replace('P',preds[pred]))
            text=c_source(text,hoist).replace('/O1 /G7',f'/{opt} /G{cpu}')
            # The same C89-compatible body enters C and C++ front ends.
            if frontend=='c':text=text.replace('// cl:','// cl: /TC')
            elif frontend=='externcpp':text='extern "C" {\n'+text+'\n}\n'
            else:text=text.replace('void *__cdecl memset','extern "C" {\nvoid *__cdecl memset').replace('\nint Rva00237D3DCheck()', '\n}\nint Rva00237D3DCheck()')
            sym='_Rva00237D3DCheck' if frontend in ['c','externcpp'] else NAME
            jobs.append((f'f{len(jobs):04}',text,dict(predicate=pred,tail=tail,hoist=hoist,cpu=cpu,opt=opt,frontend=frontend,symbol=sym)))
    elif mode=='mixed':
        bank=(ROOT/'reverse/attempts/0x00237d3d.cpp').read_text()
        probe=(ROOT/'reverse/attempt_support/trust237-20261022-round7/qi-control.cpp.txt').read_text()
        for kind,pragma,opt,cpu,extra in itertools.product(['bank','probe'],['none','unmanaged','managed_off','managed_on_off'],['O1','O2'],[6,7],['','/d2Gmanaged','/QI0f /d2Gmanaged']):
            text=bank if kind=='bank' else probe
            orig=next(l for l in text.splitlines() if l.startswith('// cl:'))
            text=text.replace(orig,f'// cl: /{opt} /G{cpu} /DNDEBUG /MD /EHsc '+extra)
            at='int Rva00237D3DCheck() {' if kind=='bank' else 'extern "C" int boundaryProbe'
            directive={'none':'','unmanaged':'#pragma unmanaged\n','managed_off':'#pragma managed(push,off)\n','managed_on_off':'#pragma managed\n#pragma unmanaged\n'}[pragma]
            text=text.replace(at,directive+at)
            if pragma=='managed_off':text+='\n#pragma managed(pop)\n'
            jobs.append((f'm{len(jobs):04}',text,dict(kind=kind,pragma=pragma,cpu=cpu,opt=opt,extra=extra,symbol=NAME if kind=='bank' else '_boundaryProbe')))
    elif mode=='hints':
        bank=(ROOT/'reverse/attempts/0x00237d3d.cpp').read_text()
        for count,cpu,opt,hint in itertools.product([1,8,16],[3,5,6,7],['O1','O2'],['_mm_pause','_mm_sfence','_mm_lfence','_mm_mfence']):
            text=bank.replace('/O1 /G6',f'/{opt} /G{cpu}')
            text=text.replace('int Rva00237D3DCheck() {','extern "C" void '+hint+'(void);\n#pragma intrinsic('+hint+')\nint Rva00237D3DCheck() {')
            text=text.replace('LocalFree(argv);','LocalFree(argv);'+(hint+'();')*count)
            jobs.append((f'h{len(jobs):04}',text,dict(count=count,cpu=cpu,opt=opt,hint=hint)))
    else:raise SystemExit(mode)
    if repair_cpp:
        old=json.loads((OUT/'frontends.json').read_text())
        fixed=execute('cpp-linkage-repair',[j for j in jobs if j[2]['frontend']=='cpp'])
        table={r['name']:r for r in fixed}
        rows=[table.get(r['name'],r) for r in old]
        (OUT/'frontends.json').write_text(json.dumps(rows,indent=2)+'\n')
    else:execute(mode,jobs)
