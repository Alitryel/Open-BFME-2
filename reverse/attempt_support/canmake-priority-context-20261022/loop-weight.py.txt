from pathlib import Path
import runpy,itertools,concurrent.futures,json
n=runpy.run_path('reverse/attempt_support/canmake-early-cleanup-20261022/replay.py.txt');e=n['context']();r=e['r']
OUT=Path('build/canmake-loop-weight-20261022').resolve();OUT.mkdir(exist_ok=True);r['run'].__globals__['OUT']=OUT
preds=['failed','result.m_20==7','result.m_20!=0 && result.m_20!=7','player->m_money<(unsigned int)result.m_pad04']

def loop(sh,c,v,depth):
 s=f'if(!({c}))break;return {v};'
 for i in range(depth):
  # All loops either return immediately or leave on their first iteration.
  # Even the counter forms never evaluate an increment on a returning path.
  k=f'iteration{i}'
  s=[f'do{{{s}}}while(0);',
     f'do{{{s}}}while({c});',
     f'while(1){{{s}}}',
     f'for(;;){{{s}}}',
     f'for(int {k}=0;{k}<1;++{k}){{{s}}}',
     f'for(int {k}=1;{k}>0;--{k}){{{s}}}',
     f'while({c}){{{s}}}',
     f'do{{if({c}){{{s}}}else break;}}while(0);',
     f'do{{if(!({c}))continue;{s}}}while(0);',
     f'do{{{s}}}while(!({c}));',
     f'for(bool {k}=({c});{k};{k}=false){{{s}}}',
     f'for(bool {k}=true;{k};{k}=false){{{s}}}',
     ][sh]
 return s+'return 0;'

def tests():
 head=e['body'][:e['body'].index('if (failed)')]
 for base,site,shape,depth,repeat in itertools.product(range(2),range(5),range(12),[1,2,3,4],range(2)):
  vals=[9,2,9,2];f=[f'return {v};'for v in vals]
  for i in range(4):
   if i==site or site==4:
    c=preds[i]
    if repeat:c=f'(({c}) && ({c}))'
    f[i]=loop(shape,c,vals[i],depth)
  b=head+'if(failed){'+f[0]+'}'
  if base:b+='switch(result.m_20){case 0:break;case 7:{'+f[1]+'}default:{'+f[2]+'}}'
  else:b+='if(result.m_20==7){'+f[1]+'}if(result.m_20!=0){'+f[2]+'}'
  if site not in (3,4):f[3]='do{if(!(player->m_money<(unsigned int)result.m_pad04))break;return 2;}while(0);return 0;'
  b+='unsigned int money=player->m_money;if(money<(unsigned int)result.m_pad04)goto insufficient;m_totalCost=result.m_pad04;return 0;insufficient:{'+f[3]+'}'
  yield f'loop-{base}{site}-{shape}-{depth}-{repeat}',e['prefix']+'\nint AIBuildableWallSegment::canMake(Player *player){'+b+'}\n'

if __name__=='__main__':
 items=list({s:(k,s)for k,s in tests()}.values());ds=[]
 with concurrent.futures.ThreadPoolExecutor(max_workers=8)as pool:

 (OUT/'loop-weight.json').write_text(json.dumps(ds,indent=2)+'\n')
 print(json.dumps({'trials':len(ds),'errors':sum('error'in d for d in ds),'unresolved':sum(bool(d.get('unresolved'))for d in ds),'exact':sum(d.get('exact_candidate',False)for d in ds),'unique':len({d.get('masked_code_sha256')for d in ds})}),flush=True)
