from pathlib import Path
import itertools, json, sys
sys.path.insert(0,str(Path('build/trust237-round5').resolve()))
import search
search.OUT=Path('build/trust237-round6').resolve()

# Representative distinct, status-consuming families from the isolated probes.
preds=[
    '(((u+0u)*(u+1u))&1u)==0u',
    '(((u+0u)*(u+1u)*(u+2u))&1u)==0u',
    '(((u+0u)*(u+1u)*(u+2u)*(u+3u))&7u)==0u',
    '(u/3u)*3u+(u%3u)==u',
    '(u/3u)*3u<=u',
    '(u&(0u-u))<=u',
    '(u/(u|1u))<=1u',
    '((u+0x7fffffffu)^0x80000000u)==(u-1u)',
]
shapes=[
    'return P;',
    'int r=0; if(P) goto yes; return r; yes:return r+1;',
    'int r=0; if(!(P))goto done; ++r; done:return r;',
]
jobs=[]
for pi,shape,timing,helper,cpu,scope in itertools.product(range(len(preds)),range(3),range(2),range(2),[6,7],range(2)):
    expr=preds[pi];helpers=''
    if helper:
        helpers='__forceinline bool arithmeticPredicate(unsigned long u){return '+expr+';}\n'
        expr='arithmeticPredicate(u)'
    if timing:
        tail='bool held=('+expr+');LocalFree(argv);'+shapes[shape].replace('P','held')
    else:tail='LocalFree(argv);'+shapes[shape].replace('P','('+expr+')')
    if scope:tail='{'+tail+'}'
    text=search.base.replace(search.marker,'long status=WinVerifyTrust(0,&action,&data);unsigned long u=(unsigned long)status;'+tail)
    text=text.replace('int Rva00237D3DCheck() {',helpers+'int Rva00237D3DCheck() {').replace('/G7',f'/G{cpu}')
    jobs.append((f't{len(jobs):04}',text,dict(predicate=pi,shape=shape,timing=timing,helper=helper,cpu=cpu,scope=scope)))
if __name__=='__main__':search.execute('full-arithmetic',jobs)
