import sys,itertools
from pathlib import Path
sys.path.insert(0,str(Path('build/trust237-round2').resolve()))
import search
preds=[
 '(u&255u)<=255u', '(u&65535u)<=65535u',
 '(unsigned char)u==255u || (unsigned char)u<255u',
 '(unsigned short)u==65535u || (unsigned short)u<65535u',
 '(u&1u)<=1u', '(u&1u)==0u || (u&1u)==1u',
 '(u&0x7fffffffu)<0x80000000u', '(u>>31u)<=1u',
 '(u%16u)<16u', '(u&15u)==0u || (u&15u)>0u',
 '(u&255u)<255u || (u&255u)==255u',
 '(signed char)u>=-128 && (signed char)u<=127',
 '(short)u>=-32768 && (short)u<=32767',
 '(unsigned char)(u|1u)>0u', 'u==0u || ((u|(0u-u))>>31u)!=0u',
]
forms=[
 'int r=0; if(P) ++r; return r;',
 'int r=0; if(!(P)) goto done; ++r; done:return r;',
 'int r=0; if(P) goto yes; return r; yes:return r+1;',
 'int r=0; do{if(!(P))break;++r;}while(0);return r;',
 'int r=0; while(P){++r;break;}return r;',
 'int r=0; if(P) r++; else r=0; return r;',
 'int r=1; if(!(P)) r--; return r;',
 'int r=0; switch((bool)(P)){case true:++r;break;case false:break;}return r;',
]
jobs=[]
for pi,fi,early,cpu,helper in itertools.product(range(len(preds)),range(len(forms)),range(3),range(2),range(2)):
 tail=forms[fi].replace('P','('+preds[pi]+')')
 if early==1:tail=tail.replace('int r=','r=');tail='int r; LocalFree(argv); '+tail
 elif early==2:tail=tail.replace('int r=','r=');tail='LocalFree(argv); '+tail
 else:tail='LocalFree(argv); '+tail
 text=search.base.replace(search.marker,'    long status=WinVerifyTrust(0,&action,&data); unsigned long u=(unsigned long)status; '+tail)
 if early==2:text=text.replace('    TrustFile file;', '    int r;\n    TrustFile file;')
 if not cpu:text=text.replace('/G7','/G6')
 if helper:
  text=text.replace('int Rva00237D3DCheck() {','__forceinline bool rangePredicate(unsigned long u){return '+preds[pi]+';}\nint Rva00237D3DCheck() {')
  text=text.replace('('+preds[pi]+')','rangePredicate(u)')
 jobs.append((f'p{len(jobs):04}',text,dict(pred=pi,form=fi,early=early,cpu=cpu,helper=helper)))
search.execute('partitions',jobs)
