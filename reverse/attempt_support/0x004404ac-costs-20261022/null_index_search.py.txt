from search import *
import re

INIT=[
 'int index = game ? slot : 0;\n if (!game) return;',
 'int index = game ? slot : -1;\n if (!game) return;',
 'int index = game ? slot : 8;\n if (!game) return;',
 'int index = !game ? 0 : slot;\n if (!game) return;',
 'int index;\n if (game) index=slot;\n if (!game) return;',
 'int index=0;\n if (game) index=slot;\n if (!game) return;',
 'int index=-1;\n if (game) index=slot;\n if (!game) return;',
 'int index=slot;\n if (!game) index=0;\n if (!game) return;',
 'int index=slot;\n if (!game) index=-1;\n if (!game) return;',
 'int index;\n if (!game) return; else index=slot;',
 'int index;\n if (game) index=slot; else return;',
 'int index;\n if (game) {index=slot; goto ValidGame;} else return;\n ValidGame:;',
 'int index=slot;\n if (game) goto ValidGame; return;\n ValidGame:;',
 'int index;\n do {if (!game) return; index=slot;} while (false);',
 'int index=slot;\n do {if (!game) return;} while (false);',
 'int index;\n if ((game != 0) && ((index=slot),true)) {} else return;',
]

def source(n):
    kind=n%len(INIT);mode=n//len(INIT)
    body=BODY
    a=body.index('\tif (!game)');b=body.index('\tif (slot == -1 || slot >= 8)')
    tail=re.sub(r'\bslot\b','index',body[b:])
    init=INIT[kind]
    if mode==1:init=init.replace('int index','long index')
    if mode==2:init=init.replace('int index','unsigned int index');tail=tail.replace('index >= 8','(int)index >= 8')
    if mode==3:init=init.replace('int index','register int index')
    return HEAD+MARK+body[:a]+init+'\n'+tail

if __name__=='__main__':run([('nullidx-%02d'%n,source(n)) for n in range(64)],'null_index')
