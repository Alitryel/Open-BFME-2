import sys,itertools,importlib.util
from pathlib import Path
# fgen.py SPEC.py OUTDIR : SPEC defines CH=[[(old,new),...alts...],...] where alt 0 is usually [] (bank); writes every combination
spec=importlib.util.spec_from_file_location('s',sys.argv[1]); m=importlib.util.module_from_spec(spec); spec.loader.exec_module(m)
base=Path(getattr(m,'BASE','bank.cpp')).read_text(); out=Path(sys.argv[2]); out.mkdir(exist_ok=True)
K='void HordeMeleeAmoeba::updateMeleeAttack'
for combo in itertools.product(*[range(len(c)) for c in m.CH]):
    pre,post=base.split(K,1); post=K+post; ok=True
    for ci,ai in enumerate(combo):
        for a,b in m.CH[ci][ai]:
            if a.startswith('PRE:'):
                a=a[4:]
                if a not in pre: print('MISSING',ci,ai,a[:50]); ok=False; break
                pre=pre.replace(a,b,1)
            else:
                if a not in post: print('MISSING',ci,ai,a[:50]); ok=False; break
                post=post.replace(a,b,1)
    if ok: (out/('v'+''.join(map(str,combo))+'.cpp')).write_text(pre+post)
