// compiler:clang23-x86
// cl: /O1 /G6 /GS- /MD
// The two little-endian qwords occupy four consecutive stack dwords.
extern "C" unsigned __int64 __stdcall _allmul(volatile unsigned a_low,
    volatile unsigned a_high, volatile unsigned b_low,
    volatile unsigned b_high) __asm__("__allmul");
unsigned __int64 __stdcall _allmul(volatile unsigned a_low,
    volatile unsigned a_high, volatile unsigned b_low, volatile unsigned b_high)
{
    unsigned ah = a_high;
    unsigned bh = b_high;
    unsigned bl = b_low;
    if ((ah | bh) == 0)
        return (unsigned __int64)a_low * bl;
    unsigned cross = ah * bl;
    cross += a_low * b_high;
    unsigned __int64 low = (unsigned __int64)a_low * bl;
    return low + ((unsigned __int64)cross << 32);
}
