import sys,itertools,json,time
from pathlib import Path
from concurrent.futures import ThreadPoolExecutor
sys.path.insert(0,'build/fxinit-seat22')
import search as h
h.HERE=Path('build/fxinit-live22').resolve()
def variants():
 for a,b,c,d,p,n in itertools.product(range(4),range(4),range(2),range(2),range(3),range(3)):
  src=h.BASE;lo=h.START;hi=h.END;arm=src[lo:hi]
  for value,form in [(1,a),(2,b)]:
   new=[f'm_BonesPerVertex=={value}',f'(unsigned)m_BonesPerVertex=={value}u',f'(unsigned long)m_BonesPerVertex=={value}ul',f'(__int64)m_BonesPerVertex=={value}i64'][form]
   arm=arm.replace('m_BonesPerVertex=='+str(value),new)
  if c:arm=arm.replace('m_Passes>1','(long)m_Passes>1L')
  if d:arm=arm.replace('m_BonesPerVertex>0','(long)m_BonesPerVertex>0L')
  for stream,form in [('positions',p),('normals',n)]:
   ptr=stream+'[1]';expr=[ptr,'(unsigned long)'+ptr+'!=0UL','(unsigned __int64)(unsigned)'+ptr+'!=0ui64'][form]
   arm=arm.replace('if('+ptr+')','if('+expr+')')
  src=src[:lo]+arm+src[hi:]
  yield 'literalweb_'+''.join(map(str,(a,b,c,d,p,n))),src
if __name__=='__main__':
 start=time.time()
 with ThreadPoolExecutor(max_workers=8) as pool:r=list(pool.map(h.run_one,variants()))
 r.sort(key=lambda x:(x.get('nd',99999),x.get('cost',99999)))
 (h.HERE/'literalweb-results.json').write_text(json.dumps(r,indent=2))
 print(json.dumps({'count':len(r),'valid':sum('nd'in x and not x.get('unresolved')for x in r),'seconds':time.time()-start,'best':r[:5]},indent=2))
